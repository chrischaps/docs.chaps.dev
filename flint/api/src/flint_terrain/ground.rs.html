1//! A height field placed in the world, and a ray against it (ADR 0095).
2//!
3//! [`Heightmap`] knows its samples but not where it lies: the editor and
4//! [`Terrain`](crate::Terrain) put it at the origin, `width` by `depth`.
5//! A game that builds its own ground (from a function, a server, a mesh it
6//! makes itself) needs to ask the same questions of it at any place and
7//! scale: how high is the ground here, and where does this ray meet it.
8//! [`GroundField`] answers both, so click-to-move and a camera rig can find
9//! the real ground on a hill instead of a plane at some guessed height.
10//!
11//! Samples sit on the vertices of a grid spanning `origin .. origin + size`
12//! in x and z, as `Heightmap::sample` reads them: column 0 at `origin[0]`,
13//! the last column at `origin[0] + size[0]`. Between them the ground is
14//! bilinear. World y is `sample × height_scale + base_y`.
15
16use crate::heightmap::Heightmap;
17
18/// A [`Heightmap`] placed in the world. Build it once (it scans the samples
19/// for their range) and share it; every query takes `&self`.
20pub struct GroundField {
21    heightmap: Heightmap,
22    origin: [f32; 2],
23    size: [f32; 2],
24    height_scale: f32,
25    base_y: f32,
26    /// World y of the lowest and highest sample: the box a ray is clipped to.
27    y_range: [f32; 2],
28}
29
30impl GroundField {
31    /// Place `heightmap` with its first sample at `origin` (world x, z) and
32    /// its last at `origin + size`. A sample `h` stands at world y
33    /// `h × height_scale + base_y`.
34    ///
35    /// Panics if the heightmap is narrower than 2 × 2 samples, or `size`
36    /// isn't positive: there'd be no ground between the samples.
37    pub fn new(
38        heightmap: Heightmap,
39        origin: [f32; 2],
40        size: [f32; 2],
41        height_scale: f32,
42        base_y: f32,
43    ) -> Self {
44        assert!(
45            heightmap.width >= 2 && heightmap.depth >= 2,
46            "a ground field needs at least 2x2 samples, got {}x{}",
47            heightmap.width,
48            heightmap.depth
49        );
50        assert!(
51            size[0] > 0.0 && size[1] > 0.0,
52            "a ground field's size must be positive, got {size:?}"
53        );
54        let (lo, hi) = heightmap
55            .heights()
56            .iter()
57            .fold((f32::INFINITY, f32::NEG_INFINITY), |(lo, hi), &h| {
58                (lo.min(h), hi.max(h))
59            });
60        let (a, b) = (lo * height_scale + base_y, hi * height_scale + base_y);
61        Self {
62            heightmap,
63            origin,
64            size,
65            height_scale,
66            base_y,
67            y_range: [a.min(b), a.max(b)],
68        }
69    }
70
71    pub fn heightmap(&self) -> &Heightmap {
72        &self.heightmap
73    }
74
75    pub fn origin(&self) -> [f32; 2] {
76        self.origin
77    }
78
79    pub fn size(&self) -> [f32; 2] {
80        self.size
81    }
82
83    /// World y of the lowest and highest sample.
84    pub fn y_range(&self) -> [f32; 2] {
85        self.y_range
86    }
87
88    /// The spacing of the samples in world units, along x and z.
89    pub fn texel(&self) -> [f32; 2] {
90        [
91            self.size[0] / (self.heightmap.width - 1) as f32,
92            self.size[1] / (self.heightmap.depth - 1) as f32,
93        ]
94    }
95
96    /// Whether (x, z) lies over the field, edges included.
97    pub fn contains(&self, x: f32, z: f32) -> bool {
98        let (u, v) = self.uv(x, z);
99        (0.0..=1.0).contains(&u) && (0.0..=1.0).contains(&v)
100    }
101
102    /// World y of the ground at (x, z), or `None` off the field.
103    pub fn height_at(&self, x: f32, z: f32) -> Option<f32> {
104        self.contains(x, z).then(|| self.height_clamped(x, z))
105    }
106
107    /// Where a ray first comes down onto the ground, as a world point, or
108    /// `None` if it misses. `dir` needn't be unit length.
109    ///
110    /// The ray is clipped to the field's box (its x/z extent and the range
111    /// of its heights) first, so it can start anywhere: far off the field,
112    /// or high above it. One that comes in through the field's side below
113    /// the ground meets the ground at the edge, as if the field had a
114    /// wall there. One that starts under the ground is reported where it
115    /// next comes down onto it from above, if it does. The hit is
116    /// found by stepping one texel at a time over the ground and bisecting
117    /// the step that crosses it, so a ridge thinner than a texel can be
118    /// stepped over.
119    pub fn ray(&self, origin: [f32; 3], dir: [f32; 3]) -> Option<[f32; 3]> {
120        let lo = [self.origin[0], self.y_range[0], self.origin[1]];
121        let hi = [
122            self.origin[0] + self.size[0],
123            self.y_range[1],
124            self.origin[1] + self.size[1],
125        ];
126        let [tx, tz] = self.texel();
127        march(origin, dir, lo, hi, tx.min(tz), |x, z| {
128            self.height_clamped(x, z)
129        })
130    }
131
132    fn uv(&self, x: f32, z: f32) -> (f32, f32) {
133        (
134            (x - self.origin[0]) / self.size[0],
135            (z - self.origin[1]) / self.size[1],
136        )
137    }
138
139    fn height_clamped(&self, x: f32, z: f32) -> f32 {
140        let (u, v) = self.uv(x, z);
141        self.heightmap.sample(u, v) * self.height_scale + self.base_y
142    }
143}

145/// The first point where a ray comes down onto the ground `height(x, z)`,
146/// within the box `lo..hi`. `step` is the ground's texel: the ray advances
147/// that far over the ground per sample, then the crossing step is bisected.
148pub(crate) fn march(
149    origin: [f32; 3],
150    dir: [f32; 3],
151    lo: [f32; 3],
152    hi: [f32; 3],
153    step: f32,
154    height: impl Fn(f32, f32) -> f32,
155) -> Option<[f32; 3]> {
156    let len = (dir[0] * dir[0] + dir[1] * dir[1] + dir[2] * dir[2]).sqrt();
157    if len.is_nan() || len <= 1e-12 || step.is_nan() || step <= 0.0 {
158        return None;
159    }
160    let d = [dir[0] / len, dir[1] / len, dir[2] / len];
161
162    // Slab clip to the box. A flat field (lo y == hi y) still has a box: a
163    // ray through its plane enters and leaves at the same t.
164    let (mut t_in, mut t_out) = (0.0f32, f32::INFINITY);
165    for axis in 0..3 {
166        if d[axis].abs() < 1e-12 {
167            if origin[axis] < lo[axis] || origin[axis] > hi[axis] {
168                return None;
169            }
170            continue;
171        }
172        let a = (lo[axis] - origin[axis]) / d[axis];
173        let b = (hi[axis] - origin[axis]) / d[axis];
174        t_in = t_in.max(a.min(b));
175        t_out = t_out.min(a.max(b));
176    }
177    if t_in.is_nan() || t_out.is_nan() || t_in > t_out {
178        return None;
179    }
180
181    let at = |t: f32| {
182        [
183            origin[0] + d[0] * t,
184            origin[1] + d[1] * t,
185            origin[2] + d[2] * t,
186        ]
187    };
188    // Above the ground: positive. On it: zero.
189    let above = |t: f32| {
190        let p = at(t);
191        p[1] - height(p[0], p[2])
192    };
193
194    // One texel over the ground per step; a vertical ray takes the box's
195    // height in one step, which is exact, since the ground under it is one
196    // height.
197    let over = (d[0] * d[0] + d[2] * d[2]).sqrt();
198    let span = t_out - t_in;
199    let dt = if over > 1e-9 {
200        (step / over).min(span)
201    } else {
202        span
203    };
204    let steps = if dt > 0.0 {
205        ((span / dt).ceil() as u64).clamp(1, 1 << 22)
206    } else {
207        1
208    };
209
210    let mut prev_t = t_in;
211    let mut prev = above(t_in);
212    // On the ground as it enters, or coming in through the field's side
213    // under the ground: it meets the field's edge.
214    if prev == 0.0 || (prev < 0.0 && t_in > 0.0) {
215        let p = at(t_in);
216        return Some([p[0], height(p[0], p[2]), p[2]]);
217    }
218    for i in 1..=steps {
219        let t = if i == steps {
220            t_out
221        } else {
222            t_in + dt * i as f32
223        };
224        let cur = above(t);
225        if prev > 0.0 && cur <= 0.0 {
226            let (mut a, mut b) = (prev_t, t);
227            for _ in 0..24 {
228                let mid = 0.5 * (a + b);
229                if above(mid) > 0.0 {
230                    a = mid;
231                } else {
232                    b = mid;
233                }
234            }
235            let p = at(0.5 * (a + b));
236            return Some([p[0], height(p[0], p[2]), p[2]]);
237        }
238        prev_t = t;
239        prev = cur;
240    }
241    None
242}

248    /// A ramp along x: y = x / 4 over 0..100, sampled every `texel` units,
249    /// placed at `origin`.
250    fn ramp(origin: [f32; 2], texel: f32) -> GroundField {
251        let n = (100.0 / texel) as u32 + 1;
252        let mut heights = Vec::with_capacity((n * n) as usize);
253        for _row in 0..n {
254            for col in 0..n {
255                heights.push(col as f32 * texel / 4.0);
256            }
257        }
258        GroundField::new(
259            Heightmap::from_raw(heights, n, n),
260            origin,
261            [100.0, 100.0],
262            1.0,
263            0.0,
264        )
265    }

267    /// Hills: a sum of sines, sampled every 2 units over 200 x 200.
268    fn hills() -> (GroundField, impl Fn(f32, f32) -> f32) {
269        let f = |x: f32, z: f32| 12.0 * (x * 0.031).sin() * (z * 0.047).cos() + 15.0;
270        let n = 101u32;
271        let mut heights = Vec::with_capacity((n * n) as usize);
272        for row in 0..n {
273            for col in 0..n {
274                heights.push(f(-50.0 + col as f32 * 2.0, 30.0 + row as f32 * 2.0));
275            }
276        }
277        let field = GroundField::new(
278            Heightmap::from_raw(heights, n, n),
279            [-50.0, 30.0],
280            [200.0, 200.0],
281            1.0,
282            0.0,
283        );
284        (field, f)
285    }

361    #[test]
362    fn hits_on_hills_lie_on_the_ground_within_a_texel() {
363        let (g, f) = hills();
364        let eye = [-120.0f32, 90.0, -40.0];
365        let mut n = 0;
366        for i in 0..20 {
367            for j in 0..20 {
368                // Aim a little under points on the true surface (the
369                // bilinear ground can sit a hair below the sines), so every
370                // ray comes down by the target. The first hit may be a hill
371                // in front of it, so check the hit is on the ground and no
372                // farther than the target.
373                let tx = -40.0 + i as f32 * 9.5;
374                let tz = 40.0 + j as f32 * 9.5;
375                let target = [tx, f(tx, tz) - 0.1, tz];
376                let dir = [target[0] - eye[0], target[1] - eye[1], target[2] - eye[2]];
377                let hit = g.ray(eye, dir).unwrap();
378                assert!((hit[1] - g.height_at(hit[0], hit[2]).unwrap()).abs() < 1e-3);
379                let d_hit = (hit[0] - eye[0]).hypot(hit[2] - eye[2]);
380                let d_target = (tx - eye[0]).hypot(tz - eye[2]);
381                assert!(d_hit <= d_target + 2.0, "{hit:?} past {target:?}");
382                if d_target - d_hit < 2.0 {
383                    n += 1;
384                }
385            }
386        }
387        // Most targets are in plain view of the eye, and land on themselves.
388        assert!(n > 200, "{n}");
389    }
