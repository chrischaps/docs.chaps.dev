6/// Configuration for terrain generation, parsed from the `terrain` component
7pub struct TerrainConfig {
8    /// Path to the heightmap PNG
9    pub heightmap_path: String,
10    /// World-space X extent
11    pub width: f32,
12    /// World-space Z extent
13    pub depth: f32,
14    /// Maximum Y height (heightmap 1.0 maps to this)
15    pub height_scale: f32,
16    /// Number of quads per chunk edge (vertices = resolution + 1)
17    pub chunk_resolution: u32,
18    /// UV tiling factor for layer textures
19    pub texture_tile: f32,
20    /// Path to the RGBA splat map
21    pub splat_map_path: String,
22    /// Paths to 4 layer textures (R, G, B, A channels of splat map)
23    pub layer_textures: [String; 4],
24    /// PBR metallic factor
25    pub metallic: f32,
26    /// PBR roughness factor
27    pub roughness: f32,
28}

30/// A complete terrain composed of chunks
31pub struct Terrain {
32    /// The source heightmap
33    pub heightmap: Heightmap,
34    /// Number of chunks along X axis
35    pub chunks_x: u32,
36    /// Number of chunks along Z axis
37    pub chunks_z: u32,
38    /// All generated chunks
39    pub chunks: Vec<TerrainChunk>,
40}
41
42impl Terrain {
43    /// Generate all terrain chunks from a heightmap and config.
44    ///
45    /// The chunk grid is sized so that each chunk covers approximately
46    /// `chunk_resolution` quads, with the heightmap distributed across
47    /// all chunks. For small heightmaps, a single chunk is used.
48    pub fn generate(heightmap: &Heightmap, config: &TerrainConfig) -> Self {
49        let res = config.chunk_resolution.max(2);
50
51        // Determine how many chunks we need based on heightmap resolution
52        // Each chunk covers `res` quads, so we need ceil(hm_pixels / res) chunks
53        let chunks_x = ((heightmap.width - 1) as f32 / res as f32).ceil().max(1.0) as u32;
54        let chunks_z = ((heightmap.depth - 1) as f32 / res as f32).ceil().max(1.0) as u32;
55
56        let mut chunks = Vec::with_capacity((chunks_x * chunks_z) as usize);
57
58        for row in 0..chunks_z {
59            for col in 0..chunks_x {
60                let chunk = generate_chunk(
61                    heightmap, config, col, row, res, chunks_x, chunks_z,
62                );
63                chunks.push(chunk);
64            }
65        }
66
67        Self {
68            heightmap: Heightmap::from_raw(
69                // Clone the heightmap data so Terrain owns it
70                heightmap.clone_heights(),
71                heightmap.width,
72                heightmap.depth,
73            ),
74            chunks_x,
75            chunks_z,
76            chunks,
77        }
78    }
79
80    /// Sample world-space height at (x, z).
81    /// Returns the interpolated Y value at that position.
82    pub fn sample_height(&self, x: f32, z: f32, config: &TerrainConfig) -> f32 {
83        self.heightmap.sample_world(x, z, config)
84    }
85
86    /// Export all terrain geometry as a single trimesh for physics.
87    /// Returns (vertices, triangle_indices) suitable for Rapier's TriMesh collider.
88    pub fn trimesh_data(&self) -> (Vec<[f32; 3]>, Vec<[u32; 3]>) {
89        let total_verts: usize = self.chunks.iter().map(|c| c.positions.len()).sum();
90        let total_tris: usize = self.chunks.iter().map(|c| c.indices.len() / 3).sum();
91
92        let mut vertices = Vec::with_capacity(total_verts);
93        let mut triangles = Vec::with_capacity(total_tris);
94        let mut base_index: u32 = 0;
95
96        for chunk in &self.chunks {
97            vertices.extend_from_slice(&chunk.positions);
98
99            for tri in chunk.indices.chunks(3) {
100                triangles.push([
101                    tri[0] + base_index,
102                    tri[1] + base_index,
103                    tri[2] + base_index,
104                ]);
105            }
106
107            base_index += chunk.positions.len() as u32;
108        }
109
110        (vertices, triangles)
111    }
