1//! Grass configuration — parsed from `[grass]` TOML section
2
3use serde::{Deserialize, Serialize};
4
5#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
6#[serde(rename_all = "lowercase")]
7pub enum DensitySource {
8    Splat,
9    Map,
10}
11
12impl Default for DensitySource {
13    fn default() -> Self { Self::Splat }
14}
15
16#[derive(Debug, Clone, Serialize, Deserialize)]
17pub struct GrassConfig {
18    #[serde(default)]
19    pub enabled: bool,
20    #[serde(default = "default_density")]
21    pub density: f32,
22    #[serde(default = "default_max_distance")]
23    pub max_distance: f32,
24    #[serde(default = "default_fade_start")]
25    pub fade_start: f32,
26    #[serde(default = "default_blade_width")]
27    pub blade_width: f32,
28    #[serde(default = "default_blade_height")]
29    pub blade_height: f32,
30    #[serde(default = "default_height_variation")]
31    pub height_variation: f32,
32    #[serde(default = "default_color_base")]
33    pub color_base: [f32; 3],
34    #[serde(default = "default_color_tip")]
35    pub color_tip: [f32; 3],
36    #[serde(default = "default_color_dry")]
37    pub color_dry: [f32; 3],
38    #[serde(default = "default_dry_amount")]
39    pub dry_amount: f32,
40    #[serde(default = "default_wind_direction")]
41    pub wind_direction: [f32; 3],
42    #[serde(default = "default_wind_speed")]
43    pub wind_speed: f32,
44    #[serde(default = "default_wind_strength")]
45    pub wind_strength: f32,
46    #[serde(default = "default_bend_radius")]
47    pub bend_radius: f32,
48    #[serde(default = "default_bend_strength")]
49    pub bend_strength: f32,
50    #[serde(default)]
51    pub density_source: DensitySource,
52    #[serde(default)]
53    pub density_layer: u32,
54    #[serde(default = "default_density_threshold")]
55    pub density_threshold: f32,
56}

58impl Default for GrassConfig {
59    fn default() -> Self {
60        Self {
61            enabled: false, density: default_density(), max_distance: default_max_distance(),
62            fade_start: default_fade_start(), blade_width: default_blade_width(),
63            blade_height: default_blade_height(), height_variation: default_height_variation(),
64            color_base: default_color_base(), color_tip: default_color_tip(),
65            color_dry: default_color_dry(), dry_amount: default_dry_amount(),
66            wind_direction: default_wind_direction(), wind_speed: default_wind_speed(),
67            wind_strength: default_wind_strength(), bend_radius: default_bend_radius(),
68            bend_strength: default_bend_strength(), density_source: DensitySource::default(),
69            density_layer: 0, density_threshold: default_density_threshold(),
70        }
71    }
72}
73
74impl GrassConfig {
75    pub fn max_instances(&self, terrain_width: f32, terrain_depth: f32) -> u32 {
76        let area = terrain_width * terrain_depth;
77        (self.density * area * 0.5).ceil() as u32
78    }
79}
