5/// Parameters for brush operations
6#[derive(Debug, Clone, Copy)]
7pub struct BrushParams {
8    /// Brush radius in UV space (0..1)
9    pub radius: f32,
10    /// Brush strength (0..1)
11    pub strength: f32,
12    /// Falloff curve (0 = hard edge, 1 = smooth gaussian)
13    pub falloff: f32,
14}
15
16impl Default for BrushParams {
17    fn default() -> Self {
18        Self {
19            radius: 0.05,
20            strength: 0.5,
21            falloff: 0.7,
22        }
23    }
24}

43/// Apply a height brush stroke at UV position (u, v) on the heightmap buffer.
44pub fn apply_height_brush(
45    heights: &mut [f32],
46    res: u32,
47    u: f32,
48    v: f32,
49    mode: HeightBrushMode,
50    params: &BrushParams,
51    dt: f32,
52    seed: u64,
53) {
54    let res = res as usize;
55    let cx = (u * (res - 1) as f32) as i32;
56    let cy = (v * (res - 1) as f32) as i32;
57    let pixel_radius = (params.radius * res as f32).ceil() as i32;
58
59    let rate = params.strength * dt;
60
61    // For flatten: compute average height under brush
62    let flatten_target = if mode == HeightBrushMode::Flatten {
63        compute_brush_average(heights, res, cx, cy, pixel_radius, params.falloff)
64    } else {
65        0.0
66    };
67
68    let noise_gen = if mode == HeightBrushMode::Noise {
69        Some(Perlin::new((seed & 0xFFFF_FFFF) as u32))
70    } else {
71        None
72    };
73
74    for dy in -pixel_radius..=pixel_radius {
75        for dx in -pixel_radius..=pixel_radius {
76            let px = cx + dx;
77            let py = cy + dy;
78
79            if px < 0 || px >= res as i32 || py < 0 || py >= res as i32 {
80                continue;
81            }
82
83            let dist = ((dx * dx + dy * dy) as f32).sqrt() / pixel_radius.max(1) as f32;
84            if dist > 1.0 {
85                continue;
86            }
87
88            let weight = falloff_weight(dist, params.falloff);
89            let idx = py as usize * res + px as usize;
90
91            match mode {
92                HeightBrushMode::Raise => {
93                    heights[idx] += rate * weight;
94                }
95                HeightBrushMode::Lower => {
96                    heights[idx] -= rate * weight;
97                }
98                HeightBrushMode::Smooth => {
99                    let avg = local_average(heights, res, px, py);
100                    let diff = avg - heights[idx];
101                    heights[idx] += diff * rate * weight;
102                }
103                HeightBrushMode::Flatten => {
104                    let diff = flatten_target - heights[idx];
105                    heights[idx] += diff * rate * weight;
106                }
107                HeightBrushMode::Noise => {
108                    if let Some(ref ng) = noise_gen {
109                        let nx = px as f64 * 0.1;
110                        let ny = py as f64 * 0.1;
111                        let n = ng.get([nx, ny]) as f32;
112                        heights[idx] += n * rate * weight;
113                    }
114                }
115            }
116        }
117    }
118}

120/// Apply a splat brush stroke at UV position (u, v) on the splat buffer.
121pub fn apply_splat_brush(
122    splat: &mut [u8],
123    res: u32,
124    u: f32,
125    v: f32,
126    layer: u8,
127    mode: SplatBrushMode,
128    params: &BrushParams,
129    dt: f32,
130) {
131    if layer > 3 {
132        return;
133    }
134
135    let res = res as usize;
136    let cx = (u * (res - 1) as f32) as i32;
137    let cy = (v * (res - 1) as f32) as i32;
138    let pixel_radius = (params.radius * res as f32).ceil() as i32;
139    let rate = params.strength * dt;
140
141    for dy in -pixel_radius..=pixel_radius {
142        for dx in -pixel_radius..=pixel_radius {
143            let px = cx + dx;
144            let py = cy + dy;
145
146            if px < 0 || px >= res as i32 || py < 0 || py >= res as i32 {
147                continue;
148            }
149
150            let dist = ((dx * dx + dy * dy) as f32).sqrt() / pixel_radius.max(1) as f32;
151            if dist > 1.0 {
152                continue;
153            }
154
155            let weight = falloff_weight(dist, params.falloff) * rate;
156            let idx = (py as usize * res + px as usize) * 4;
157
158            match mode {
159                SplatBrushMode::Paint => {
160                    // Increase target layer, decrease others proportionally
161                    let current = splat[idx + layer as usize] as f32;
162                    let add = (255.0 - current) * weight;
163                    splat[idx + layer as usize] =
164                        (current + add).round().clamp(0.0, 255.0) as u8;
165
166                    // Redistribute remaining to other channels
167                    let new_val = splat[idx + layer as usize] as f32;
168                    let others_total: f32 = (0..4)
169                        .filter(|&c| c != layer as usize)
170                        .map(|c| splat[idx + c] as f32)
171                        .sum();
172
173                    let remaining = 255.0 - new_val;
174                    if others_total > 0.0 {
175                        for c in 0..4 {
176                            if c != layer as usize {
177                                splat[idx + c] = (splat[idx + c] as f32 * remaining
178                                    / others_total)
179                                    .round()
180                                    .clamp(0.0, 255.0)
181                                    as u8;
182                            }
183                        }
184                    }
185                }
186                SplatBrushMode::Erase => {
187                    // Decrease target layer, increase layer 0 as fallback
188                    let current = splat[idx + layer as usize] as f32;
189                    let sub = current * weight;
190                    splat[idx + layer as usize] =
191                        (current - sub).round().clamp(0.0, 255.0) as u8;
192
193                    // Give the removed weight to layer 0 (unless erasing layer 0)
194                    let fallback = if layer == 0 { 1 } else { 0 };
195                    splat[idx + fallback] =
196                        (splat[idx + fallback] as f32 + sub).round().clamp(0.0, 255.0) as u8;
197                }
198            }
199        }
200    }
201}

203/// Cast a ray against a heightmap and find the UV intersection point.
204///
205/// Uses ray marching with binary search refinement.
206/// Returns `Some((u, v))` if the ray hits the terrain.
207pub fn ray_heightmap_intersect(
208    origin: [f32; 3],
209    dir: [f32; 3],
210    heights: &[f32],
211    res: u32,
212    width: f32,
213    depth: f32,
214    height_scale: f32,
215) -> Option<(f32, f32)> {
216    let res = res as usize;
217
218    // Normalize direction
219    let len = (dir[0] * dir[0] + dir[1] * dir[1] + dir[2] * dir[2]).sqrt();
220    if len < 1e-8 {
221        return None;
222    }
223    let d = [dir[0] / len, dir[1] / len, dir[2] / len];
224
225    // March along ray with adaptive step
226    let max_dist = (width * width + depth * depth + (height_scale * height_scale)).sqrt() * 2.0;
227    let step = (width.min(depth)) / res as f32;
228
229    let mut t = 0.0f32;
230    let mut prev_above = true;
231
232    while t < max_dist {
233        let px = origin[0] + d[0] * t;
234        let py = origin[1] + d[1] * t;
235        let pz = origin[2] + d[2] * t;
236
237        // Convert to UV
238        let u = px / width;
239        let v = pz / depth;
240
241        if !(-0.1..=1.1).contains(&u) || !(-0.1..=1.1).contains(&v) {
242            // Way out of bounds — if we were in bounds before, we've exited
243            if t > step * 2.0 {
244                return None;
245            }
246            t += step;
247            continue;
248        }
249
250        let terrain_h = sample_bilinear(heights, res, u.clamp(0.0, 1.0), v.clamp(0.0, 1.0))
251            * height_scale;
252        let above = py > terrain_h;
253
254        if !above && prev_above && t > 0.0 {
255            // Binary search refinement
256            let mut lo = t - step;
257            let mut hi = t;
258            for _ in 0..16 {
259                let mid = (lo + hi) * 0.5;
260                let mx = origin[0] + d[0] * mid;
261                let my = origin[1] + d[1] * mid;
262                let mz = origin[2] + d[2] * mid;
263                let mu = (mx / width).clamp(0.0, 1.0);
264                let mv = (mz / depth).clamp(0.0, 1.0);
265                let mh = sample_bilinear(heights, res, mu, mv) * height_scale;
266                if my > mh {
267                    lo = mid;
268                } else {
269                    hi = mid;
270                }
271            }
272
273            let final_t = (lo + hi) * 0.5;
274            let fu = ((origin[0] + d[0] * final_t) / width).clamp(0.0, 1.0);
275            let fv = ((origin[2] + d[2] * final_t) / depth).clamp(0.0, 1.0);
276            return Some((fu, fv));
277        }
278
279        prev_above = above;
280        t += step;
281    }
282
283    None
284}

286fn falloff_weight(dist: f32, falloff: f32) -> f32 {
287    if falloff <= 0.0 {
288        // Hard edge
289        if dist <= 1.0 {
290            1.0
291        } else {
292            0.0
293        }
294    } else {
295        // Gaussian-ish falloff
296        let sigma = falloff * 0.5;
297        (-dist * dist / (2.0 * sigma * sigma)).exp()
298    }
299}

301fn local_average(heights: &[f32], res: usize, x: i32, y: i32) -> f32 {
302    let mut sum = 0.0f32;
303    let mut count = 0;
304
305    for dy in -1..=1 {
306        for dx in -1..=1 {
307            let nx = x + dx;
308            let ny = y + dy;
309            if nx >= 0 && nx < res as i32 && ny >= 0 && ny < res as i32 {
310                sum += heights[ny as usize * res + nx as usize];
311                count += 1;
312            }
313        }
314    }
315
316    if count > 0 {
317        sum / count as f32
318    } else {
319        heights[y as usize * res + x as usize]
320    }
321}

323fn compute_brush_average(
324    heights: &[f32],
325    res: usize,
326    cx: i32,
327    cy: i32,
328    pixel_radius: i32,
329    falloff: f32,
330) -> f32 {
331    let mut sum = 0.0f32;
332    let mut weight_sum = 0.0f32;
333
334    for dy in -pixel_radius..=pixel_radius {
335        for dx in -pixel_radius..=pixel_radius {
336            let px = cx + dx;
337            let py = cy + dy;
338            if px < 0 || px >= res as i32 || py < 0 || py >= res as i32 {
339                continue;
340            }
341            let dist = ((dx * dx + dy * dy) as f32).sqrt() / pixel_radius.max(1) as f32;
342            if dist > 1.0 {
343                continue;
344            }
345            let w = falloff_weight(dist, falloff);
346            sum += heights[py as usize * res + px as usize] * w;
347            weight_sum += w;
348        }
349    }
350
351    if weight_sum > 0.0 {
352        sum / weight_sum
353    } else {
354        0.0
355    }
356}

358fn sample_bilinear(heights: &[f32], res: usize, u: f32, v: f32) -> f32 {
359    let u = u.clamp(0.0, 1.0);
360    let v = v.clamp(0.0, 1.0);
361    let fx = u * (res - 1) as f32;
362    let fy = v * (res - 1) as f32;
363    let x0 = (fx as usize).min(res - 2);
364    let y0 = (fy as usize).min(res - 2);
365    let x1 = x0 + 1;
366    let y1 = y0 + 1;
367    let tx = fx - x0 as f32;
368    let ty = fy - y0 as f32;
369    let h00 = heights[y0 * res + x0];
370    let h10 = heights[y0 * res + x1];
371    let h01 = heights[y1 * res + x0];
372    let h11 = heights[y1 * res + x1];
373    let h0 = h00 * (1.0 - tx) + h10 * tx;
374    let h1 = h01 * (1.0 - tx) + h11 * tx;
375    h0 * (1.0 - ty) + h1 * ty
376}
