294impl TerrainSpec {
295    /// Load a terrain spec from a `.terrain.toml` file.
296    pub fn from_file(path: &Path) -> Result<Self, String> {
297        let content = std::fs::read_to_string(path)
298            .map_err(|e| format!("Failed to read '{}': {}", path.display(), e))?;
299        toml::from_str(&content)
300            .map_err(|e| format!("Failed to parse '{}': {}", path.display(), e))
301    }
302
303    /// Save the spec to a `.terrain.toml` file.
304    pub fn save(&self, path: &Path) -> Result<(), String> {
305        let content = toml::to_string_pretty(self)
306            .map_err(|e| format!("Failed to serialize: {}", e))?;
307        std::fs::write(path, content)
308            .map_err(|e| format!("Failed to write '{}': {}", path.display(), e))
309    }
310
311    /// Create a minimal default spec.
312    pub fn default_spec(name: &str) -> Self {
313        Self {
314            meta: MetaConfig {
315                name: name.to_string(),
316                version: "1.0.0".to_string(),
317            },
318            seed: SeedConfig::default(),
319            geometry: GeometryConfig::default(),
320            textures: TextureConfig::default(),
321            height_layers: vec![HeightLayer::Noise(NoiseLayer {
322                noise_type: NoiseType::Perlin,
323                octaves: 6,
324                frequency: 2.0,
325                lacunarity: 2.0,
326                persistence: 0.5,
327                amplitude: 1.0,
328                blend: BlendMode::Add,
329            })],
330            splat_rules: Vec::new(),
331            grass: None,
332        }
333    }
334
335    /// Get layer texture paths as a 4-element array.
336    pub fn layer_paths(&self) -> [String; 4] {
337        [
338            self.textures.layers.layer0.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
339            self.textures.layers.layer1.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
340            self.textures.layers.layer2.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
341            self.textures.layers.layer3.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
342        ]
343    }
344}

378        let toml_str = r#"
379[meta]
380name = "all_layers"
381
382[[height_layers]]
383op = "noise"
384noise_type = "ridged"
385octaves = 4
386frequency = 3.0
387
388[[height_layers]]
389op = "flatten"
390height = 0.2
391mode = "above"
392
393[[height_layers]]
394op = "erosion"
395kind = "hydraulic"
396iterations = 100
397
398[[splat_rules]]
399layer = 0
400height_min = 0.0
401height_max = 0.5
402"#;
403        let spec: TerrainSpec = toml::from_str(toml_str).unwrap();
404        assert_eq!(spec.height_layers.len(), 3);
405        assert_eq!(spec.splat_rules.len(), 1);
406
407        match &spec.height_layers[0] {
408            HeightLayer::Noise(n) => {
409                assert_eq!(n.noise_type, NoiseType::Ridged);
410                assert_eq!(n.octaves, 4);
411            }
412            _ => panic!("Expected noise layer"),
413        }
414        match &spec.height_layers[1] {
415            HeightLayer::Flatten(f) => assert_eq!(f.mode, FlattenMode::Above),
416            _ => panic!("Expected flatten layer"),
417        }
418        match &spec.height_layers[2] {
419            HeightLayer::Erosion(e) => assert_eq!(e.kind, ErosionKind::Hydraulic),
420            _ => panic!("Expected erosion layer"),
421        }
