291impl TerrainSpec {
292    /// Load a terrain spec from a `.terrain.toml` file.
293    pub fn from_file(path: &Path) -> Result<Self, String> {
294        let content = std::fs::read_to_string(path)
295            .map_err(|e| format!("Failed to read '{}': {}", path.display(), e))?;
296        toml::from_str(&content)
297            .map_err(|e| format!("Failed to parse '{}': {}", path.display(), e))
298    }
299
300    /// Save the spec to a `.terrain.toml` file.
301    pub fn save(&self, path: &Path) -> Result<(), String> {
302        let content = toml::to_string_pretty(self)
303            .map_err(|e| format!("Failed to serialize: {}", e))?;
304        std::fs::write(path, content)
305            .map_err(|e| format!("Failed to write '{}': {}", path.display(), e))
306    }
307
308    /// Create a minimal default spec.
309    pub fn default_spec(name: &str) -> Self {
310        Self {
311            meta: MetaConfig {
312                name: name.to_string(),
313                version: "1.0.0".to_string(),
314            },
315            seed: SeedConfig::default(),
316            geometry: GeometryConfig::default(),
317            textures: TextureConfig::default(),
318            height_layers: vec![HeightLayer::Noise(NoiseLayer {
319                noise_type: NoiseType::Perlin,
320                octaves: 6,
321                frequency: 2.0,
322                lacunarity: 2.0,
323                persistence: 0.5,
324                amplitude: 1.0,
325                blend: BlendMode::Add,
326            })],
327            splat_rules: Vec::new(),
328        }
329    }
330
331    /// Get layer texture paths as a 4-element array.
332    pub fn layer_paths(&self) -> [String; 4] {
333        [
334            self.textures.layers.layer0.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
335            self.textures.layers.layer1.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
336            self.textures.layers.layer2.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
337            self.textures.layers.layer3.as_ref().map(|l| l.path.clone()).unwrap_or_default(),
338        ]
339    }
340}

372    #[test]
373    fn parse_all_layer_types() {
374        let toml_str = r#"
375[meta]
376name = "all_layers"
377
378[[height_layers]]
379op = "noise"
380noise_type = "ridged"
381octaves = 4
382frequency = 3.0
383
384[[height_layers]]
385op = "flatten"
386height = 0.2
387mode = "above"
388
389[[height_layers]]
390op = "erosion"
391kind = "hydraulic"
392iterations = 100
393
394[[splat_rules]]
395layer = 0
396height_min = 0.0
397height_max = 0.5
398"#;
399        let spec: TerrainSpec = toml::from_str(toml_str).unwrap();
400        assert_eq!(spec.height_layers.len(), 3);
401        assert_eq!(spec.splat_rules.len(), 1);
402
403        match &spec.height_layers[0] {
404            HeightLayer::Noise(n) => {
405                assert_eq!(n.noise_type, NoiseType::Ridged);
406                assert_eq!(n.octaves, 4);
407            }
408            _ => panic!("Expected noise layer"),
409        }
410        match &spec.height_layers[1] {
411            HeightLayer::Flatten(f) => assert_eq!(f.mode, FlattenMode::Above),
412            _ => panic!("Expected flatten layer"),
413        }
414        match &spec.height_layers[2] {
415            HeightLayer::Erosion(e) => assert_eq!(e.kind, ErosionKind::Hydraulic),
416            _ => panic!("Expected erosion layer"),
417        }
418    }
