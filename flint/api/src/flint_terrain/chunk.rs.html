1//! Terrain chunk mesh generation
2
3use crate::heightmap::Heightmap;
4use crate::terrain::TerrainConfig;
5
6/// A single chunk of terrain geometry
7pub struct TerrainChunk {
8    /// Grid position (column, row) in the chunk grid
9    pub grid_pos: (u32, u32),
10    /// Vertex positions in world space
11    pub positions: Vec<[f32; 3]>,
12    /// Vertex normals
13    pub normals: Vec<[f32; 3]>,
14    /// UV coordinates (normalized over entire terrain for splat map lookup)
15    pub uvs: Vec<[f32; 2]>,
16    /// Triangle indices (CCW winding)
17    pub indices: Vec<u32>,
18    /// AABB minimum corner
19    pub aabb_min: [f32; 3],
20    /// AABB maximum corner
21    pub aabb_max: [f32; 3],
22}
23
24/// Generate mesh data for a single terrain chunk.
25///
26/// `col` and `row` identify which chunk in the grid.
27/// `resolution` is the number of quads per chunk edge (vertices = resolution + 1).
28/// `chunks_x` and `chunks_z` are the total number of chunks in each direction.
29pub fn generate_chunk(
30    heightmap: &Heightmap,
31    config: &TerrainConfig,
32    col: u32,
33    row: u32,
34    resolution: u32,
35    chunks_x: u32,
36    chunks_z: u32,
37) -> TerrainChunk {
38    let verts_per_edge = resolution + 1;
39    let vert_count = (verts_per_edge * verts_per_edge) as usize;
40
41    let mut positions = Vec::with_capacity(vert_count);
42    let mut normals = Vec::with_capacity(vert_count);
43    let mut uvs = Vec::with_capacity(vert_count);
44
45    let mut aabb_min = [f32::MAX; 3];
46    let mut aabb_max = [f32::MIN; 3];
47
48    for vz in 0..verts_per_edge {
49        for vx in 0..verts_per_edge {
50            // Normalized position within the entire terrain [0..1]
51            let u = (col as f32 + vx as f32 / resolution as f32) / chunks_x as f32;
52            let v = (row as f32 + vz as f32 / resolution as f32) / chunks_z as f32;
53
54            let world_x = u * config.width;
55            let world_z = v * config.depth;
56            let height = heightmap.sample(u, v) * config.height_scale;
57
58            let normal =
59                heightmap.compute_normal(u, v, config.width, config.depth, config.height_scale);
60
61            let pos = [world_x, height, world_z];
62
63            // Update AABB
64            for i in 0..3 {
65                aabb_min[i] = aabb_min[i].min(pos[i]);
66                aabb_max[i] = aabb_max[i].max(pos[i]);
67            }
68
69            positions.push(pos);
70            normals.push(normal);
71            uvs.push([u, v]);
72        }
73    }
74
75    // Generate indices (two triangles per quad, CCW winding)
76    let index_count = (resolution * resolution * 6) as usize;
77    let mut indices = Vec::with_capacity(index_count);
78
79    for qz in 0..resolution {
80        for qx in 0..resolution {
81            let tl = qz * verts_per_edge + qx;
82            let tr = tl + 1;
83            let bl = tl + verts_per_edge;
84            let br = bl + 1;
85
86            // First triangle (top-left, bottom-left, bottom-right)
87            indices.push(tl);
88            indices.push(bl);
89            indices.push(br);
90
91            // Second triangle (top-left, bottom-right, top-right)
92            indices.push(tl);
93            indices.push(br);
94            indices.push(tr);
95        }
96    }
97
98    TerrainChunk {
99        grid_pos: (col, row),
100        positions,
101        normals,
102        uvs,
103        indices,
104        aabb_min,
105        aabb_max,
106    }
107}
