1//! A model read back from a glTF file, in the shape the validator checks.
2//!
3//! [`validate_output`](crate::validate_output) checks what a generator made
4//! in memory; [`validate_model`](crate::validate::validate_model) checks a
5//! [`Model`], which is what a file on disk holds. `flint gen --validate`
6//! reads its own GLB back through here (so an exporter bug fails the
7//! checks), and `flint asset validate` reads an imported or AI-generated
8//! model through the same path, so one set of rules covers every model.
9//!
10//! The file is read with [`flint_import::import_gltf`], the importer the
11//! renderer uses, so the validator sees what the game will load.
12
13use std::path::Path;

15use flint_import::{ImportResult, ImportedMesh};

18use crate::types::{BoundingBox, ImageData, MaterialData, MeshData, SubMesh, Vertex};

20/// One mesh of a [`Model`], with what the file left out.
21///
22/// glTF makes normals and UVs optional, and a [`MeshData`] can't say an
23/// attribute is absent, so the absences are counted here: the missing
24/// vertices hold zeros in `mesh`, and the checks report them as missing
25/// instead of as broken.
26#[derive(Debug, Clone)]
27pub struct ModelMesh {
28    pub mesh: MeshData,
29    /// Vertices whose primitive has no `NORMAL` attribute.
30    pub normals_missing: usize,
31    /// Vertices whose primitive has no `TEXCOORD_0` attribute.
32    pub uvs_missing: usize,
33    /// Per material of `mesh`: whether it samples a texture (base colour,
34    /// metallic-roughness or normal). UVs on a textured material may tile
35    /// past [0,1]; glTF samplers repeat by default.
36    pub textured: Vec<bool>,
37}
38
39/// How a model's meshes relate.
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub enum ModelLayout {
42    /// One mesh: every mesh node of the file, merged.
43    Single,
44    /// One mesh per level of detail (`LOD_0`, `LOD_1`, ... as
45    /// [`mesh_lods_to_glb`](crate::mesh_lods_to_glb) writes them).
46    Lods,
47    /// One skinned mesh.
48    Skinned,
49}
50
51/// The meshes of a model file, ready for
52/// [`validate_model`](crate::validate::validate_model).
53#[derive(Debug, Clone)]
54pub struct Model {
55    pub layout: ModelLayout,
56    pub meshes: Vec<ModelMesh>,
57    /// glTF materials in the file (a primitive with none uses glTF's
58    /// default material, which isn't counted).
59    pub material_count: usize,
60}
61
62impl Model {
63    /// Triangles over every mesh (every level, for a LOD chain).
64    pub fn triangle_count(&self) -> usize {
65        self.meshes.iter().map(|m| m.mesh.triangle_count()).sum()
66    }
67}

69/// Read a `.glb` / `.gltf` file into a [`Model`].
70pub fn read_model(path: &Path) -> Result<Model> {
71    let import = flint_import::import_gltf(path).map_err(|e| {
72        ProcGenError::InvalidMesh(format!("could not import {}: {e}", path.display()))
73    })?;
74    model_from_import(&import)
75}
76
77/// Build a [`Model`] from an import.
78///
79/// Mesh nodes named `LOD_0`, `LOD_1`, ... in order (two or more) are a LOD
80/// chain, one mesh each. Anything else is merged into one mesh, primitives
81/// becoming submeshes, so the style's triangle ceiling and palette apply
82/// to the whole model. Node transforms are not applied: every check reads
83/// shape (relative sliver area, unit normals, indices), not placement.
84pub fn model_from_import(import: &ImportResult) -> Result<Model> {
85    let materials: Vec<MaterialData> = import
86        .materials
87        .iter()
88        .map(|m| MaterialData {
89            name: m.name.clone(),
90            base_color: m.base_color,
91            metallic: m.metallic,
92            roughness: m.roughness,
93            normal_map: None,
94        })
95        .collect();
96    let textured: Vec<bool> = import
97        .materials
98        .iter()
99        .map(|m| {
100            m.base_color_texture.is_some()
101                || m.metallic_roughness_texture.is_some()
102                || m.normal_texture.is_some()
103        })
104        .collect();
105
106    // Primitives grouped by the node that holds them, in file order.
107    let nodes: Vec<(&str, Vec<&ImportedMesh>)> = import
108        .nodes
109        .iter()
110        .filter(|n| !n.mesh_primitive_indices.is_empty())
111        .map(|n| {
112            let prims = n
113                .mesh_primitive_indices
114                .iter()
115                .filter_map(|&i| import.meshes.get(i))
116                .collect();
117            (n.name.as_str(), prims)
118        })
119        .collect();
120    if nodes.is_empty() {
121        return Err(ProcGenError::InvalidMesh("the file holds no mesh".into()));
122    }
123
124    let is_lod_chain = nodes.len() > 1
125        && nodes
126            .iter()
127            .enumerate()
128            .all(|(i, (name, _))| *name == format!("LOD_{i}"));
129    let skinned = import.meshes.iter().any(|m| m.skin_index.is_some());
130
131    let (layout, meshes) = if is_lod_chain {
132        let meshes = nodes
133            .iter()
134            .map(|(_, prims)| merge(prims, &materials, &textured))
135            .collect();
136        (ModelLayout::Lods, meshes)
137    } else {
138        let prims: Vec<&ImportedMesh> = nodes.into_iter().flat_map(|(_, p)| p).collect();
139        let layout = if skinned {
140            ModelLayout::Skinned
141        } else {
142            ModelLayout::Single
143        };
144        (layout, vec![merge(&prims, &materials, &textured)])
145    };
146
147    Ok(Model {
148        layout,
149        meshes,
150        material_count: import.materials.len(),
151    })
152}

154/// glTF's default material, for a primitive that names none: white, fully
155/// metallic, fully rough (glTF 2.0 §3.9.6).
156fn gltf_default_material() -> MaterialData {
157    MaterialData {
158        name: "default (glTF)".into(),
159        base_color: [1.0; 4],
160        metallic: 1.0,
161        roughness: 1.0,
162        normal_map: None,
163    }
164}

166/// Merge primitives into one mesh, a submesh each. Primitives that share a
167/// vertex buffer (the exporter's submeshes do) share it here too, so the
168/// vertex count matches the mesh that was written.
169fn merge(prims: &[&ImportedMesh], materials: &[MaterialData], textured: &[bool]) -> ModelMesh {
170    let mut out = ModelMesh {
171        mesh: MeshData {
172            vertices: Vec::new(),
173            indices: Vec::new(),
174            materials: Vec::new(),
175            submeshes: Vec::new(),
176            bounding_box: BoundingBox::from_positions(&[]),
177        },
178        normals_missing: 0,
179        uvs_missing: 0,
180        textured: Vec::new(),
181    };
182    // The file's material index (None = glTF default) for each slot.
183    let mut slots: Vec<Option<usize>> = Vec::new();
184    let mut previous: Option<(&ImportedMesh, u32)> = None;
185
186    for prim in prims {
187        let base = match previous {
188            Some((p, base)) if same_vertices(p, prim) => base,
189            _ => {
190                let base = out.mesh.vertices.len() as u32;
191                append_vertices(&mut out, prim);
192                base
193            }
194        };
195        previous = Some((prim, base));
196
197        let slot = match slots.iter().position(|s| *s == prim.material_index) {
198            Some(slot) => slot,
199            None => {
200                slots.push(prim.material_index);
201                let (mat, tex) = match prim.material_index.and_then(|i| materials.get(i)) {
202                    Some(m) => (m.clone(), textured[prim.material_index.unwrap()]),
203                    None => (gltf_default_material(), false),
204                };
205                out.mesh.materials.push(mat);
206                out.textured.push(tex);
207                slots.len() - 1
208            }
209        };
210
211        let index_start = out.mesh.indices.len() as u32;
212        if prim.indices.is_empty() {
213            // Non-indexed: every three vertices are a triangle.
214            out.mesh
215                .indices
216                .extend((0..prim.positions.len() as u32).map(|i| base + i));
217        } else {
218            out.mesh
219                .indices
220                .extend(prim.indices.iter().map(|&i| base.saturating_add(i)));
221        }
222        out.mesh.submeshes.push(SubMesh {
223            index_start,
224            index_count: out.mesh.indices.len() as u32 - index_start,
225            material_index: slot,
226        });
227    }
228
229    out.mesh.recompute_bounds();
230    out
231}

233fn same_vertices(a: &ImportedMesh, b: &ImportedMesh) -> bool {
234    a.positions == b.positions && a.normals == b.normals && a.uvs == b.uvs
235}
236
237fn append_vertices(out: &mut ModelMesh, prim: &ImportedMesh) {
238    let n = prim.positions.len();
239    let has_normals = prim.normals.len() == n;
240    let has_uvs = prim.uvs.len() == n;
241    if !has_normals {
242        out.normals_missing += n;
243    }
244    if !has_uvs {
245        out.uvs_missing += n;
246    }
247    out.mesh.vertices.extend(
248        prim.positions
249            .iter()
250            .enumerate()
251            .map(|(i, &position)| Vertex {
252                position,
253                normal: if has_normals {
254                    prim.normals[i]
255                } else {
256                    [0.0; 3]
257                },
258                tangent: [0.0; 4],
259                uv: if has_uvs { prim.uvs[i] } else { [0.0; 2] },
260            }),
261    );
262}

264/// Read a PNG back into an [`ImageData`] (RGBA8), so a written texture can
265/// be validated as it lies on disk. The file can't say what the image
266/// means, so the caller names its `semantics`.
267pub fn read_png(path: &Path, semantics: crate::ChannelSemantics) -> Result<ImageData> {
268    let bytes = flint_core::vfs::read(path)?;
269    let img = image::load_from_memory(&bytes)
270        .map_err(|e| ProcGenError::ImageError(format!("could not read {}: {e}", path.display())))?
271        .into_rgba8();
272    Ok(ImageData {
273        width: img.width(),
274        height: img.height(),
275        pixels: img.into_raw(),
276        channel_semantics: semantics,
277    })
278}

289    fn lines(report: &crate::ValidationReport) -> Vec<String> {
290        report
291            .checks
292            .iter()
293            .map(|c| format!("[{}] {}: {}", c.status, c.name, c.detail))
294            .collect()
295    }

297    /// The GLB the exporter writes validates exactly as the output it was
298    /// written from: every check, status and detail. A difference is an
299    /// exporter (or importer) bug.
300    fn assert_round_trip(output: &GeneratorOutput, spec: &ProcGenSpec, name: &str) -> Model {
301        let glb = match output {
302            GeneratorOutput::Mesh(m) => crate::mesh_to_glb(m),
303            GeneratorOutput::MeshWithLods(l) => crate::mesh_lods_to_glb(l),
304            GeneratorOutput::SkinnedMesh(m) => crate::skinned_mesh_to_glb(m),
305            _ => unreachable!(),
306        }
307        .unwrap();
308        let dir = std::env::temp_dir().join(format!("flint_model_{name}_{}", std::process::id()));
309        std::fs::create_dir_all(&dir).unwrap();
310        let path = dir.join(format!("{name}.glb"));
311        std::fs::write(&path, glb).unwrap();
312        let model = read_model(&path).unwrap();
313        std::fs::remove_dir_all(&dir).ok();
314
315        let constraints = ValidationConstraints {
316            palette: vec!["#5a3d22".into(), "#4f6b2a".into()],
317            roughness_range: Some([0.5, 1.0]),
318            ..Default::default()
319        };
320        let memory = validate_output(output, spec, &constraints);
321        let disk = validate_model(&model, Some(spec), &constraints);
322        assert_eq!(lines(&disk), lines(&memory));
323        model
324    }

326    #[test]
327    fn the_oak_validates_the_same_from_disk() {
328        let spec =
329            ProcGenSpec::parse(include_str!("../../../specs/oak_tree.procgen.toml")).unwrap();
330        let output = crate::TreeGenerator
331            .generate(&spec, &mut SeededRng::new(42))
332            .unwrap();
333        let lods = output.as_mesh_lods().unwrap().len();
334        let model = assert_round_trip(&output, &spec, "oak");
335        assert_eq!(model.layout, ModelLayout::Lods);
336        assert_eq!(model.meshes.len(), lods);
337        // The bark's normal map comes back as a texture, so its UVs tile.
338        assert!(model.meshes[0].textured.iter().any(|&t| t));
339    }
