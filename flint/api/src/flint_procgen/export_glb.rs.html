1//! GLB binary export for procedurally generated meshes.
2//!
3//! Converts [`MeshData`] into the GLB binary container format using the
4//! `gltf::json` AST types. The output is a self-contained `.glb` file with
5//! interleaved vertex data (position, normal, tangent, UV) and `u32` indices.

21/// Bytes per interleaved vertex: position(12) + normal(12) + tangent(16) + uv(8) = 48.
22const VERTEX_STRIDE: usize = 48;
23
24/// Convert a single [`MeshData`] into a self-contained GLB binary.
25pub fn mesh_to_glb(mesh: &MeshData) -> Result<Vec<u8>> {
26    meshes_to_glb_inner(&[mesh])
27}
28
29/// Convert multiple [`MeshData`] (e.g. LOD levels) into a single GLB binary.
30///
31/// Each mesh becomes a separate glTF mesh node. LOD 0 is the first mesh.
32pub fn mesh_lods_to_glb(meshes: &[MeshData]) -> Result<Vec<u8>> {
33    let refs: Vec<&MeshData> = meshes.iter().collect();
34    meshes_to_glb_inner(&refs)
35}

37/// Bytes per skinned vertex: position(12) + normal(12) + tangent(16) + uv(8) + joints(8) + weights(16) = 72.
38const SKINNED_VERTEX_STRIDE: usize = 72;
39
40/// Convert a [`SkinnedMeshData`] into a self-contained GLB binary with skeleton.
41pub fn skinned_mesh_to_glb(mesh: &SkinnedMeshData) -> Result<Vec<u8>> {
42    mesh.validate().map_err(|e| {
43        crate::ProcGenError::InvalidMesh(format!("skinned mesh validation failed: {e}"))
44    })?;
45
46    let bin_data = build_skinned_buffer(mesh);
47    let root = build_skinned_gltf_root(mesh, bin_data.len());
48    write_glb(&root, &bin_data)
49}

51/// Pack skinned vertex + index + inverse bind matrix data into a single binary buffer.
52fn build_skinned_buffer(mesh: &SkinnedMeshData) -> Vec<u8> {
53    let mut buf = Vec::new();
54
55    // Vertices: interleaved pos(3) + normal(3) + tangent(4) + uv(2) + joints(4 u16) + weights(4 f32)
56    for v in &mesh.vertices {
57        buf.extend_from_slice(bytemuck_f32_slice(&v.position));
58        buf.extend_from_slice(bytemuck_f32_slice(&v.normal));
59        buf.extend_from_slice(bytemuck_f32_slice(&v.tangent));
60        buf.extend_from_slice(bytemuck_f32_slice(&v.uv));
61        // Joint indices as u16
62        for &j in &v.joint_indices {
63            buf.extend_from_slice(&j.to_le_bytes());
64        }
65        // Joint weights as f32
66        buf.extend_from_slice(bytemuck_f32_slice(&v.joint_weights));
67    }
68
69    // Pad to 4-byte alignment
70    while buf.len() % 4 != 0 {
71        buf.push(0);
72    }
73
74    // Indices: u32
75    for &idx in &mesh.indices {
76        buf.extend_from_slice(&idx.to_le_bytes());
77    }
78
79    // Pad to 4-byte alignment
80    while buf.len() % 4 != 0 {
81        buf.push(0);
82    }
83
84    // Inverse bind matrices: 16 floats per bone
85    let ibms = mesh.skeleton.compute_inverse_bind_matrices();
86    for ibm in &ibms {
87        buf.extend_from_slice(bytemuck_f32_slice(ibm));
88    }
89
90    // Pad to 4-byte alignment
91    while buf.len() % 4 != 0 {
92        buf.push(0);
93    }
94
95    buf
96}

98/// Build the glTF JSON root for a skinned mesh with skeleton.
99fn build_skinned_gltf_root(mesh: &SkinnedMeshData, buffer_byte_length: usize) -> json::Root {
100    let mut root = json::Root::default();
101    root.asset.generator = Some("Flint Procgen".into());
102
103    let buffer_idx = root.push(json::Buffer {
104        byte_length: USize64::from(buffer_byte_length),
105        name: None,
106        uri: None,
107        extensions: None,
108        extras: Default::default(),
109    });
110
111    // ─── Compute buffer layout ──────────────────────────────────────────
112    let vertex_bytes = mesh.vertices.len() * SKINNED_VERTEX_STRIDE;
113    let padded_vertex_bytes = (vertex_bytes + 3) & !3;
114
115    let index_bytes = mesh.indices.len() * mem::size_of::<u32>();
116    let index_offset = padded_vertex_bytes;
117    let padded_index_bytes = (index_bytes + 3) & !3;
118
119    let ibm_offset = index_offset + padded_index_bytes;
120    let ibm_bytes = mesh.skeleton.bones.len() * 16 * mem::size_of::<f32>();
121
122    // ─── Buffer views ───────────────────────────────────────────────────
123    let vtx_view_idx = root.push(json::buffer::View {
124        buffer: buffer_idx,
125        byte_length: USize64::from(vertex_bytes),
126        byte_offset: Some(USize64::from(0usize)),
127        byte_stride: Some(json::buffer::Stride(SKINNED_VERTEX_STRIDE)),
128        name: None,
129        target: Some(Checked::Valid(Target::ArrayBuffer)),
130        extensions: None,
131        extras: Default::default(),
132    });
133
134    let idx_view_idx = root.push(json::buffer::View {
135        buffer: buffer_idx,
136        byte_length: USize64::from(index_bytes),
137        byte_offset: Some(USize64::from(index_offset)),
138        byte_stride: None,
139        name: None,
140        target: Some(Checked::Valid(Target::ElementArrayBuffer)),
141        extensions: None,
142        extras: Default::default(),
143    });
144
145    let ibm_view_idx = root.push(json::buffer::View {
146        buffer: buffer_idx,
147        byte_length: USize64::from(ibm_bytes),
148        byte_offset: Some(USize64::from(ibm_offset)),
149        byte_stride: None,
150        name: None,
151        target: None,
152        extensions: None,
153        extras: Default::default(),
154    });
155
156    // ─── Accessors ──────────────────────────────────────────────────────
157    let bb = &mesh.bounding_box;
158    let pos_min = serde_json::json!([bb.min.x, bb.min.y, bb.min.z]);
159    let pos_max = serde_json::json!([bb.max.x, bb.max.y, bb.max.z]);
160    let vertex_count = mesh.vertices.len();
161
162    // POSITION (offset 0)
163    let pos_acc = root.push(json::Accessor {
164        buffer_view: Some(vtx_view_idx),
165        byte_offset: Some(USize64::from(0u64)),
166        count: USize64::from(vertex_count),
167        component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
168        type_: Checked::Valid(Type::Vec3),
169        min: Some(pos_min),
170        max: Some(pos_max),
171        name: None,
172        normalized: false,
173        sparse: None,
174        extensions: None,
175        extras: Default::default(),
176    });
177
178    // NORMAL (offset 12)
179    let norm_acc = root.push(json::Accessor {
180        buffer_view: Some(vtx_view_idx),
181        byte_offset: Some(USize64::from(12u64)),
182        count: USize64::from(vertex_count),
183        component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
184        type_: Checked::Valid(Type::Vec3),
185        min: None,
186        max: None,
187        name: None,
188        normalized: false,
189        sparse: None,
190        extensions: None,
191        extras: Default::default(),
192    });
193
194    // TANGENT (offset 24)
195    let tang_acc = root.push(json::Accessor {
196        buffer_view: Some(vtx_view_idx),
197        byte_offset: Some(USize64::from(24u64)),
198        count: USize64::from(vertex_count),
199        component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
200        type_: Checked::Valid(Type::Vec4),
201        min: None,
202        max: None,
203        name: None,
204        normalized: false,
205        sparse: None,
206        extensions: None,
207        extras: Default::default(),
208    });
209
210    // TEXCOORD_0 (offset 40)
211    let uv_acc = root.push(json::Accessor {
212        buffer_view: Some(vtx_view_idx),
213        byte_offset: Some(USize64::from(40u64)),
214        count: USize64::from(vertex_count),
215        component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
216        type_: Checked::Valid(Type::Vec2),
217        min: None,
218        max: None,
219        name: None,
220        normalized: false,
221        sparse: None,
222        extensions: None,
223        extras: Default::default(),
224    });
225
226    // JOINTS_0 (offset 48, u16x4)
227    let joints_acc = root.push(json::Accessor {
228        buffer_view: Some(vtx_view_idx),
229        byte_offset: Some(USize64::from(48u64)),
230        count: USize64::from(vertex_count),
231        component_type: Checked::Valid(GenericComponentType(ComponentType::U16)),
232        type_: Checked::Valid(Type::Vec4),
233        min: None,
234        max: None,
235        name: None,
236        normalized: false,
237        sparse: None,
238        extensions: None,
239        extras: Default::default(),
240    });
241
242    // WEIGHTS_0 (offset 56, f32x4)
243    let weights_acc = root.push(json::Accessor {
244        buffer_view: Some(vtx_view_idx),
245        byte_offset: Some(USize64::from(56u64)),
246        count: USize64::from(vertex_count),
247        component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
248        type_: Checked::Valid(Type::Vec4),
249        min: None,
250        max: None,
251        name: None,
252        normalized: false,
253        sparse: None,
254        extensions: None,
255        extras: Default::default(),
256    });
257
258    // Index accessor
259    let idx_acc = root.push(json::Accessor {
260        buffer_view: Some(idx_view_idx),
261        byte_offset: Some(USize64::from(0u64)),
262        count: USize64::from(mesh.indices.len()),
263        component_type: Checked::Valid(GenericComponentType(ComponentType::U32)),
264        type_: Checked::Valid(Type::Scalar),
265        min: None,
266        max: None,
267        name: None,
268        normalized: false,
269        sparse: None,
270        extensions: None,
271        extras: Default::default(),
272    });
273
274    // Inverse bind matrices accessor
275    let ibm_acc = root.push(json::Accessor {
276        buffer_view: Some(ibm_view_idx),
277        byte_offset: Some(USize64::from(0u64)),
278        count: USize64::from(mesh.skeleton.bones.len()),
279        component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
280        type_: Checked::Valid(Type::Mat4),
281        min: None,
282        max: None,
283        name: None,
284        normalized: false,
285        sparse: None,
286        extensions: None,
287        extras: Default::default(),
288    });
289
290    // ─── Vertex attributes ──────────────────────────────────────────────
291    let mut attributes = BTreeMap::new();
292    attributes.insert(Checked::Valid(Semantic::Positions), pos_acc);
293    attributes.insert(Checked::Valid(Semantic::Normals), norm_acc);
294    attributes.insert(Checked::Valid(Semantic::Tangents), tang_acc);
295    attributes.insert(Checked::Valid(Semantic::TexCoords(0)), uv_acc);
296    attributes.insert(Checked::Valid(Semantic::Joints(0)), joints_acc);
297    attributes.insert(Checked::Valid(Semantic::Weights(0)), weights_acc);
298
299    // ─── Primitives & materials ─────────────────────────────────────────
300    let primitives = if mesh.submeshes.is_empty() {
301        let mat_data = mesh.materials.first().cloned().unwrap_or_default();
302        let mat_idx = root.push(material_from_data(&mat_data));
303        vec![Primitive {
304            attributes,
305            extensions: None,
306            extras: Default::default(),
307            indices: Some(idx_acc),
308            material: Some(mat_idx),
309            mode: Checked::Valid(Mode::Triangles),
310            targets: None,
311        }]
312    } else {
313        mesh.submeshes
314            .iter()
315            .map(|sub| {
316                let mat_data = mesh
317                    .materials
318                    .get(sub.material_index)
319                    .cloned()
320                    .unwrap_or_default();
321                let mat_idx = root.push(material_from_data(&mat_data));
322
323                let sub_idx_acc = root.push(json::Accessor {
324                    buffer_view: Some(idx_view_idx),
325                    byte_offset: Some(USize64::from(
326                        sub.index_start as usize * mem::size_of::<u32>(),
327                    )),
328                    count: USize64::from(sub.index_count as usize),
329                    component_type: Checked::Valid(GenericComponentType(ComponentType::U32)),
330                    type_: Checked::Valid(Type::Scalar),
331                    min: None,
332                    max: None,
333                    name: None,
334                    normalized: false,
335                    sparse: None,
336                    extensions: None,
337                    extras: Default::default(),
338                });
339
340                Primitive {
341                    attributes: attributes.clone(),
342                    extensions: None,
343                    extras: Default::default(),
344                    indices: Some(sub_idx_acc),
345                    material: Some(mat_idx),
346                    mode: Checked::Valid(Mode::Triangles),
347                    targets: None,
348                }
349            })
350            .collect()
351    };
352
353    // ─── Bone nodes ─────────────────────────────────────────────────────
354    let bone_node_base = root.nodes.len() as u32;
355    let mut joint_indices = Vec::new();
356
357    for bone in &mesh.skeleton.bones {
358        let [tx, ty, tz] = bone.rest_translation;
359        let [rx, ry, rz, rw] = bone.rest_rotation;
360        let [sx, sy, sz] = bone.rest_scale;
361
362        let node_idx = root.push(json::Node {
363            name: Some(bone.name.clone()),
364            translation: Some([tx, ty, tz]),
365            rotation: Some(json::scene::UnitQuaternion([rx, ry, rz, rw])),
366            scale: Some([sx, sy, sz]),
367            children: None,
368            mesh: None,
369            ..Default::default()
370        });
371        joint_indices.push(node_idx);
372    }
373
374    // Set up parent-child relationships
375    for (i, bone) in mesh.skeleton.bones.iter().enumerate() {
376        if let Some(parent) = bone.parent {
377            let parent_node: json::Index<json::Node> =
378                json::Index::new(bone_node_base + parent as u32);
379            let child_node: json::Index<json::Node> =
380                json::Index::new(bone_node_base + i as u32);
381            if let Some(ref mut children) = root.nodes[parent_node.value()].children {
382                children.push(child_node);
383            } else {
384                root.nodes[parent_node.value()].children = Some(vec![child_node]);
385            }
386        }
387    }
388
389    // ─── Skin ───────────────────────────────────────────────────────────
390    let skin_idx = root.push(json::Skin {
391        name: Some("Armature".into()),
392        inverse_bind_matrices: Some(ibm_acc),
393        joints: joint_indices.clone(),
394        skeleton: joint_indices.first().copied(),
395        extensions: None,
396        extras: Default::default(),
397    });
398
399    // ─── Mesh node (with skin reference) ────────────────────────────────
400    let mesh_idx = root.push(json::Mesh {
401        extensions: None,
402        extras: Default::default(),
403        name: Some("Mesh".into()),
404        primitives,
405        weights: None,
406    });
407
408    let mesh_node_idx = root.push(json::Node {
409        mesh: Some(mesh_idx),
410        skin: Some(skin_idx),
411        name: Some("Mesh".into()),
412        ..Default::default()
413    });
414
415    // ─── Scene ──────────────────────────────────────────────────────────
416    // Scene nodes: mesh node + root bone nodes (bones without parents)
417    let mut scene_nodes = vec![mesh_node_idx];
418    for (i, bone) in mesh.skeleton.bones.iter().enumerate() {
419        if bone.parent.is_none() {
420            scene_nodes.push(json::Index::new(bone_node_base + i as u32));
421        }
422    }
423
424    let scene_idx = root.push(json::Scene {
425        extensions: None,
426        extras: Default::default(),
427        name: Some("Scene".into()),
428        nodes: scene_nodes,
429    });
430    root.scene = Some(scene_idx);
431
432    root
433}

435/// Internal: build a GLB from one or more meshes sharing a single binary buffer.
436fn meshes_to_glb_inner(meshes: &[&MeshData]) -> Result<Vec<u8>> {
437    if meshes.is_empty() {
438        return Err(crate::ProcGenError::InvalidMesh(
439            "no meshes provided for GLB export".into(),
440        ));
441    }
442
443    for (i, mesh) in meshes.iter().enumerate() {
444        mesh.validate().map_err(|e| {
445            crate::ProcGenError::InvalidMesh(format!("mesh {i} validation failed: {e}"))
446        })?;
447    }
448
449    let bin_data = build_buffer(meshes);
450    let root = build_gltf_root(meshes, bin_data.len());
451    write_glb(&root, &bin_data)
452}

454/// Pack all mesh vertex + index data into a single binary buffer.
455///
456/// Layout per mesh: [vertices (interleaved)] [padding to 4-byte] [indices (u32)]
457/// Each mesh section is appended sequentially.
458fn build_buffer(meshes: &[&MeshData]) -> Vec<u8> {
459    let mut buf = Vec::new();
460
461    for mesh in meshes {
462        // Vertices: interleaved pos(3) + normal(3) + tangent(4) + uv(2) = 12 floats = 48 bytes
463        for v in &mesh.vertices {
464            buf.extend_from_slice(bytemuck_f32_slice(&v.position));
465            buf.extend_from_slice(bytemuck_f32_slice(&v.normal));
466            buf.extend_from_slice(bytemuck_f32_slice(&v.tangent));
467            buf.extend_from_slice(bytemuck_f32_slice(&v.uv));
468        }
469
470        // Pad vertex section to 4-byte alignment (already aligned for f32, but be safe)
471        while buf.len() % 4 != 0 {
472            buf.push(0);
473        }
474
475        // Indices: u32
476        for &idx in &mesh.indices {
477            buf.extend_from_slice(&idx.to_le_bytes());
478        }
479
480        // Pad to 4-byte alignment after indices
481        while buf.len() % 4 != 0 {
482            buf.push(0);
483        }
484    }
485
486    buf
487}

489/// Reinterpret an `f32` slice as little-endian bytes.
490fn bytemuck_f32_slice(data: &[f32]) -> &[u8] {
491    let ptr = data.as_ptr() as *const u8;
492    let len = data.len() * mem::size_of::<f32>();
493    // SAFETY: f32 has no alignment requirements stricter than u8, and the
494    // original slice is valid for len bytes.
495    unsafe { std::slice::from_raw_parts(ptr, len) }
496}

498/// Build the glTF JSON root with buffer views, accessors, meshes, materials,
499/// nodes, and a single scene.
500fn build_gltf_root(meshes: &[&MeshData], buffer_byte_length: usize) -> json::Root {
501    let mut root = json::Root::default();
502    root.asset.generator = Some("Flint Procgen".into());
503
504    // Single buffer covering all mesh data
505    let buffer_idx = root.push(json::Buffer {
506        byte_length: USize64::from(buffer_byte_length),
507        name: None,
508        uri: None,
509        extensions: None,
510        extras: Default::default(),
511    });
512
513    let mut byte_offset: usize = 0;
514    let mut node_indices = Vec::new();
515
516    for (mesh_i, mesh) in meshes.iter().enumerate() {
517        let vertex_bytes = mesh.vertices.len() * VERTEX_STRIDE;
518        let vertex_offset = byte_offset;
519
520        // Pad vertex section to 4-byte alignment
521        let padded_vertex_bytes = (vertex_bytes + 3) & !3;
522
523        let index_bytes = mesh.indices.len() * mem::size_of::<u32>();
524        let index_offset = vertex_offset + padded_vertex_bytes;
525
526        let padded_index_bytes = (index_bytes + 3) & !3;
527        byte_offset = index_offset + padded_index_bytes;
528
529        // Buffer view for vertex data (interleaved)
530        let vtx_view_idx = root.push(json::buffer::View {
531            buffer: buffer_idx,
532            byte_length: USize64::from(vertex_bytes),
533            byte_offset: Some(USize64::from(vertex_offset)),
534            byte_stride: Some(json::buffer::Stride(VERTEX_STRIDE)),
535            name: None,
536            target: Some(Checked::Valid(Target::ArrayBuffer)),
537            extensions: None,
538            extras: Default::default(),
539        });
540
541        // Buffer view for index data
542        let idx_view_idx = root.push(json::buffer::View {
543            buffer: buffer_idx,
544            byte_length: USize64::from(index_bytes),
545            byte_offset: Some(USize64::from(index_offset)),
546            byte_stride: None,
547            name: None,
548            target: Some(Checked::Valid(Target::ElementArrayBuffer)),
549            extensions: None,
550            extras: Default::default(),
551        });
552
553        // Compute bounding box for POSITION accessor
554        let bb = &mesh.bounding_box;
555        let pos_min = serde_json::json!([bb.min.x, bb.min.y, bb.min.z]);
556        let pos_max = serde_json::json!([bb.max.x, bb.max.y, bb.max.z]);
557        let vertex_count = mesh.vertices.len();
558
559        // Accessor: POSITION (offset 0 in stride)
560        let pos_acc = root.push(json::Accessor {
561            buffer_view: Some(vtx_view_idx),
562            byte_offset: Some(USize64::from(0u64)),
563            count: USize64::from(vertex_count),
564            component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
565            type_: Checked::Valid(Type::Vec3),
566            min: Some(pos_min),
567            max: Some(pos_max),
568            name: None,
569            normalized: false,
570            sparse: None,
571            extensions: None,
572            extras: Default::default(),
573        });
574
575        // Accessor: NORMAL (offset 12)
576        let norm_acc = root.push(json::Accessor {
577            buffer_view: Some(vtx_view_idx),
578            byte_offset: Some(USize64::from(12u64)),
579            count: USize64::from(vertex_count),
580            component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
581            type_: Checked::Valid(Type::Vec3),
582            min: None,
583            max: None,
584            name: None,
585            normalized: false,
586            sparse: None,
587            extensions: None,
588            extras: Default::default(),
589        });
590
591        // Accessor: TANGENT (offset 24)
592        let tang_acc = root.push(json::Accessor {
593            buffer_view: Some(vtx_view_idx),
594            byte_offset: Some(USize64::from(24u64)),
595            count: USize64::from(vertex_count),
596            component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
597            type_: Checked::Valid(Type::Vec4),
598            min: None,
599            max: None,
600            name: None,
601            normalized: false,
602            sparse: None,
603            extensions: None,
604            extras: Default::default(),
605        });
606
607        // Accessor: TEXCOORD_0 (offset 40)
608        let uv_acc = root.push(json::Accessor {
609            buffer_view: Some(vtx_view_idx),
610            byte_offset: Some(USize64::from(40u64)),
611            count: USize64::from(vertex_count),
612            component_type: Checked::Valid(GenericComponentType(ComponentType::F32)),
613            type_: Checked::Valid(Type::Vec2),
614            min: None,
615            max: None,
616            name: None,
617            normalized: false,
618            sparse: None,
619            extensions: None,
620            extras: Default::default(),
621        });
622
623        // Accessor: indices
624        let idx_acc = root.push(json::Accessor {
625            buffer_view: Some(idx_view_idx),
626            byte_offset: Some(USize64::from(0u64)),
627            count: USize64::from(mesh.indices.len()),
628            component_type: Checked::Valid(GenericComponentType(ComponentType::U32)),
629            type_: Checked::Valid(Type::Scalar),
630            min: None,
631            max: None,
632            name: None,
633            normalized: false,
634            sparse: None,
635            extensions: None,
636            extras: Default::default(),
637        });
638
639        // Shared vertex attributes for all primitives in this mesh
640        let mut base_attributes = BTreeMap::new();
641        base_attributes.insert(Checked::Valid(Semantic::Positions), pos_acc);
642        base_attributes.insert(Checked::Valid(Semantic::Normals), norm_acc);
643        base_attributes.insert(Checked::Valid(Semantic::Tangents), tang_acc);
644        base_attributes.insert(Checked::Valid(Semantic::TexCoords(0)), uv_acc);
645
646        let primitives = if mesh.submeshes.is_empty() {
647            // Single primitive using full index buffer (legacy path)
648            let mat_data = mesh.materials.first().cloned().unwrap_or_default();
649            let mat_idx = root.push(material_from_data(&mat_data));
650
651            vec![Primitive {
652                attributes: base_attributes,
653                extensions: None,
654                extras: Default::default(),
655                indices: Some(idx_acc),
656                material: Some(mat_idx),
657                mode: Checked::Valid(Mode::Triangles),
658                targets: None,
659            }]
660        } else {
661            // One primitive per submesh, each with its own index accessor
662            mesh.submeshes
663                .iter()
664                .map(|sub| {
665                    let mat_data = mesh
666                        .materials
667                        .get(sub.material_index)
668                        .cloned()
669                        .unwrap_or_default();
670                    let mat_idx = root.push(material_from_data(&mat_data));
671
672                    // Accessor into a sub-range of the shared index buffer view
673                    let sub_idx_acc = root.push(json::Accessor {
674                        buffer_view: Some(idx_view_idx),
675                        byte_offset: Some(USize64::from(
676                            sub.index_start as usize * mem::size_of::<u32>(),
677                        )),
678                        count: USize64::from(sub.index_count as usize),
679                        component_type: Checked::Valid(GenericComponentType(ComponentType::U32)),
680                        type_: Checked::Valid(Type::Scalar),
681                        min: None,
682                        max: None,
683                        name: None,
684                        normalized: false,
685                        sparse: None,
686                        extensions: None,
687                        extras: Default::default(),
688                    });
689
690                    Primitive {
691                        attributes: base_attributes.clone(),
692                        extensions: None,
693                        extras: Default::default(),
694                        indices: Some(sub_idx_acc),
695                        material: Some(mat_idx),
696                        mode: Checked::Valid(Mode::Triangles),
697                        targets: None,
698                    }
699                })
700                .collect()
701        };
702
703        let mesh_name = if meshes.len() > 1 {
704            Some(format!("LOD_{mesh_i}"))
705        } else {
706            Some("Mesh".into())
707        };
708
709        let mesh_idx = root.push(json::Mesh {
710            extensions: None,
711            extras: Default::default(),
712            name: mesh_name.clone(),
713            primitives,
714            weights: None,
715        });
716
717        let node_idx = root.push(json::Node {
718            mesh: Some(mesh_idx),
719            name: mesh_name,
720            ..Default::default()
721        });
722        node_indices.push(node_idx);
723    }
724
725    // Scene
726    let scene_idx = root.push(json::Scene {
727        extensions: None,
728        extras: Default::default(),
729        name: Some("Scene".into()),
730        nodes: node_indices,
731    });
732    root.scene = Some(scene_idx);
733
734    root
735}

737/// Convert a [`MaterialData`] into a glTF JSON material.
738fn material_from_data(mat: &MaterialData) -> json::Material {
739    use gltf::json::material::{PbrBaseColorFactor, PbrMetallicRoughness, StrengthFactor};
740
741    json::Material {
742        name: Some(mat.name.clone()),
743        pbr_metallic_roughness: PbrMetallicRoughness {
744            base_color_factor: PbrBaseColorFactor(mat.base_color),
745            metallic_factor: StrengthFactor(mat.metallic),
746            roughness_factor: StrengthFactor(mat.roughness),
747            ..Default::default()
748        },
749        ..Default::default()
750    }
751}
752
753/// Assemble the final GLB binary from the JSON root and binary buffer.
754fn write_glb(root: &json::Root, bin_data: &[u8]) -> Result<Vec<u8>> {
755    let json_bytes = json::serialize::to_vec(root).map_err(|e| {
756        crate::ProcGenError::InvalidMesh(format!("failed to serialize glTF JSON: {e}"))
757    })?;
758
759    let glb = gltf::binary::Glb {
760        header: gltf::binary::Header {
761            magic: *b"glTF",
762            version: 2,
763            length: 0, // overwritten by to_vec()
764        },
765        json: Cow::Borrowed(&json_bytes),
766        bin: Some(Cow::Borrowed(bin_data)),
767    };
768
769    glb.to_vec().map_err(|e| {
770        crate::ProcGenError::InvalidMesh(format!("failed to write GLB container: {e}"))
771    })
772}
