1//! Frame-budgeted generation queue for procedural assets.
2//!
3//! The [`ProcGenQueue`] spreads generation work across frames within a
4//! configurable time budget, prioritizing assets closest to the camera.
5

18/// Whether the queued asset is a model or texture.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum AssetKind {
21    Model,
22    Texture,
23}
24
25/// A pending generation request in the queue.
26#[derive(Debug, Clone)]
27pub struct GenerationRequest {
28    pub spec_name: String,
29    pub spec: ProcGenSpec,
30    pub content_hash: ContentHash,
31    pub seed: Seed,
32    /// Distance squared from camera; lower = higher priority.
33    pub priority: f32,
34    pub entity_position: [f32; 3],
35    pub asset_kind: AssetKind,
36}
37
38/// A completed asset ready for GPU upload.
39pub struct CompletedAsset {
40    pub spec_name: String,
41    /// Shared with the cache: uploading it copies nothing.
42    pub output: Arc<GeneratorOutput>,
43    pub asset_kind: AssetKind,
44}

46/// A queued asset whose generation failed, kept for the caller to act on
47/// (fall back to a file, keep a placeholder) rather than only logged.
48#[derive(Debug, Clone, PartialEq)]
49pub struct FailedAsset {
50    pub spec_name: String,
51    pub asset_kind: AssetKind,
52    pub error: String,
53}
54
55/// Statistics about the generation queue.
56#[derive(Debug, Clone, Copy, Default)]
57pub struct QueueStats {
58    pub pending_count: usize,
59    pub completed_this_frame: usize,
60    pub budget_used_ms: f64,
61    pub total_completed: u64,
62}
63
64/// Frame-budgeted generation queue.
65///
66/// Each frame, [`process_frame`](ProcGenQueue::process_frame) pops the highest-priority
67/// (closest to camera) pending requests and generates them until the time budget is
68/// exhausted. Results are returned for GPU upload.
69pub struct ProcGenQueue {
70    pending: Vec<GenerationRequest>,
71    budget_ms: f64,
72    last_camera_pos: [f32; 3],
73    /// Camera must move more than sqrt(this) units to trigger reprioritization.
74    reprioritize_threshold_sq: f32,
75    total_completed: u64,
76    last_stats: QueueStats,
77    /// Failures since the last [`take_failures`](Self::take_failures).
78    failures: Vec<FailedAsset>,
79}
80
81impl ProcGenQueue {
82    pub fn new(budget_ms: f64) -> Self {
83        Self {
84            pending: Vec::new(),
85            budget_ms,
86            last_camera_pos: [0.0; 3],
87            reprioritize_threshold_sq: 4.0,
88            total_completed: 0,
89            last_stats: QueueStats::default(),
90            failures: Vec::new(),
91        }
92    }
93
94    /// The requests that failed to generate since the last call, each once.
95    pub fn take_failures(&mut self) -> Vec<FailedAsset> {
96        std::mem::take(&mut self.failures)
97    }
98
99    pub fn set_budget_ms(&mut self, budget_ms: f64) {
100        self.budget_ms = budget_ms;
101    }
102
103    /// Enqueue a generation request. Deduplicates by `spec_name`.
104    pub fn enqueue(&mut self, request: GenerationRequest) {
105        if self.is_pending(&request.spec_name) {
106            return;
107        }
108        self.pending.push(request);
109    }
110
111    /// Check if a spec is already pending in the queue.
112    pub fn is_pending(&self, spec_name: &str) -> bool {
113        self.pending.iter().any(|r| r.spec_name == spec_name)
114    }
115
116    pub fn pending_count(&self) -> usize {
117        self.pending.len()
118    }
119
120    /// Process the queue for one frame within the time budget.
121    ///
122    /// Reprioritizes if the camera has moved significantly, then generates
123    /// assets in priority order (closest first) until the budget is exhausted.
124    pub fn process_frame(
125        &mut self,
126        camera_pos: [f32; 3],
127        registry: &GeneratorRegistry,
128        cache: &mut ProcGenCache,
129    ) -> Vec<CompletedAsset> {
130        if self.pending.is_empty() {
131            self.last_stats = QueueStats {
132                pending_count: 0,
133                completed_this_frame: 0,
134                budget_used_ms: 0.0,
135                total_completed: self.total_completed,
136            };
137            return Vec::new();
138        }
139
140        // Reprioritize if camera moved significantly
141        let dx = camera_pos[0] - self.last_camera_pos[0];
142        let dy = camera_pos[1] - self.last_camera_pos[1];
143        let dz = camera_pos[2] - self.last_camera_pos[2];
144        let cam_dist_sq = dx * dx + dy * dy + dz * dz;
145
146        if cam_dist_sq > self.reprioritize_threshold_sq {
147            for req in &mut self.pending {
148                let ex = req.entity_position[0] - camera_pos[0];
149                let ey = req.entity_position[1] - camera_pos[1];
150                let ez = req.entity_position[2] - camera_pos[2];
151                req.priority = ex * ex + ey * ey + ez * ez;
152            }
153            self.last_camera_pos = camera_pos;
154        }
155
156        // Sort by priority ascending (closest first)
157        self.pending.sort_by(|a, b| {
158            a.priority
159                .partial_cmp(&b.priority)
160                .unwrap_or(std::cmp::Ordering::Equal)
161        });
162
163        let start = Instant::now();
164        let mut completions = Vec::new();
165
166        while !self.pending.is_empty() {
167            let req = &self.pending[0];
168
169            // Check cache first. A memory hit costs nothing; one read from
170            // the disk tier costs a file read and a decode, so it counts.
171            if let Some(cached) = cache.get(req.content_hash, &req.seed) {
172                let completed = CompletedAsset {
173                    spec_name: req.spec_name.clone(),
174                    output: cached,
175                    asset_kind: req.asset_kind,
176                };
177                completions.push(completed);
178                self.pending.remove(0);
179                self.total_completed += 1;
180                if start.elapsed().as_secs_f64() * 1000.0 >= self.budget_ms {
181                    break;
182                }
183                continue;
184            }
185
186            // Check budget before generating
187            let elapsed_ms = start.elapsed().as_secs_f64() * 1000.0;
188            if elapsed_ms >= self.budget_ms && !completions.is_empty() {
189                break;
190            }
191
192            // Generate
193            let req = self.pending.remove(0);
194            match registry.generate_from_spec(&req.spec) {
195                Ok(output) => {
196                    let output = cache.insert(req.content_hash, &req.seed, output);
197                    completions.push(CompletedAsset {
198                        spec_name: req.spec_name,
199                        output,
200                        asset_kind: req.asset_kind,
201                    });
202                    self.total_completed += 1;
203                }
204                Err(e) => {
205                    self.failures.push(FailedAsset {
206                        spec_name: req.spec_name,
207                        asset_kind: req.asset_kind,
208                        error: e.to_string(),
209                    });
210                }
211            }
212
213            let elapsed_ms = start.elapsed().as_secs_f64() * 1000.0;
214            if elapsed_ms >= self.budget_ms {
215                break;
216            }
217        }
218
219        let budget_used_ms = start.elapsed().as_secs_f64() * 1000.0;
220        self.last_stats = QueueStats {
221            pending_count: self.pending.len(),
222            completed_this_frame: completions.len(),
223            budget_used_ms,
224            total_completed: self.total_completed,
225        };
226
227        completions
228    }
229
230    pub fn clear(&mut self) {
231        self.pending.clear();
232    }
233
234    pub fn stats(&self) -> QueueStats {
235        self.last_stats
236    }
237}

271    fn make_request(name: &str, position: [f32; 3]) -> GenerationRequest {
272        let spec = ProcGenSpec::parse(&MOCK_SPEC_TOML.replace("test_item", name)).unwrap();
273        let seed = spec.seed.to_seed().unwrap();
274        let content_hash = ContentHash::from_str(&toml::to_string(&spec).unwrap_or_default());
275        let dx = position[0];
276        let dy = position[1];
277        let dz = position[2];
278        GenerationRequest {
279            spec_name: name.to_string(),
280            spec,
281            content_hash,
282            seed,
283            priority: dx * dx + dy * dy + dz * dz,
284            entity_position: position,
285            asset_kind: AssetKind::Model,
286        }
287    }

361    fn budget_enforcement() {
362        // Use a tiny budget (0.001ms) — mock generator is fast but we should
363        // still break after at least 1 item if the budget is exhausted.
364        let mut queue = ProcGenQueue::new(0.001);
365        let registry = make_registry();
366        let mut cache = make_cache();
367
368        for i in 0..10 {
369            queue.enqueue(make_request(&format!("item_{i}"), [i as f32, 0.0, 0.0]));

414    fn camera_move_reprioritizes() {
415        let mut queue = ProcGenQueue::new(1000.0);
416
417        // Entity A is at (10,0,0), entity B is at (1,0,0).
418        // Camera starts at origin → B is closer.
419        queue.enqueue(make_request("entity_a", [10.0, 0.0, 0.0]));
420        queue.enqueue(make_request("entity_b", [1.0, 0.0, 0.0]));
421
422        // Move camera to (9,0,0) — now A is closer (dist=1) than B (dist=8)
423        queue.last_camera_pos = [0.0, 0.0, 0.0]; // far enough to trigger reprioritization
424        let registry = make_registry();
425        let mut cache = make_cache();
426
427        let completed = queue.process_frame([9.0, 0.0, 0.0], &registry, &mut cache);
428        assert_eq!(completed.len(), 2);
429        assert_eq!(completed[0].spec_name, "entity_a"); // closer to camera at (9,0,0)
430        assert_eq!(completed[1].spec_name, "entity_b");
