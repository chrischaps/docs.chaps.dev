1use crate::types::Vertex;
2use flint_core::Vec3;
3
4/// Compute area-weighted smooth normals for a triangle mesh.
5///
6/// Each triangle contributes its (unnormalized) cross product to its three vertices.
7/// Because `|cross(e0, e1)| = 2 * triangle_area`, larger faces contribute more.
8/// After accumulation every vertex normal is normalized to unit length.
9/// Degenerate vertices (zero accumulated normal) fall back to `[0, 1, 0]`.
10pub fn compute_normals(vertices: &mut [Vertex], indices: &[u32]) {
11    // Zero out existing normals
12    for v in vertices.iter_mut() {
13        v.normal = [0.0, 0.0, 0.0];
14    }
15
16    // Accumulate face-area-weighted normals
17    for tri in indices.chunks_exact(3) {
18        let i0 = tri[0] as usize;
19        let i1 = tri[1] as usize;
20        let i2 = tri[2] as usize;
21
22        let p0 = Vec3::from_array(vertices[i0].position);
23        let p1 = Vec3::from_array(vertices[i1].position);
24        let p2 = Vec3::from_array(vertices[i2].position);
25
26        let e0 = p1 - p0;
27        let e1 = p2 - p0;
28        let face_normal = e0.cross(&e1); // unnormalized = 2 * area * unit_normal
29
30        for &idx in &[i0, i1, i2] {
31            vertices[idx].normal[0] += face_normal.x;
32            vertices[idx].normal[1] += face_normal.y;
33            vertices[idx].normal[2] += face_normal.z;
34        }
35    }
36
37    // Normalize
38    for v in vertices.iter_mut() {
39        let n = Vec3::from_array(v.normal);
40        let len = n.length();
41        if len > 1e-10 {
42            let unit = n * (1.0 / len);
43            v.normal = unit.to_array();
44        } else {
45            v.normal = [0.0, 1.0, 0.0]; // degenerate fallback
46        }
47    }
48}

64    #[test]
65    fn flat_quad_normals_equal_face_normal() {
66        // Two triangles forming a flat quad on the XZ plane (CCW from +Y)
67        let mut vertices = vec![
68            default_vertex([0.0, 0.0, 0.0]),
69            default_vertex([1.0, 0.0, 0.0]),
70            default_vertex([1.0, 0.0, 1.0]),
71            default_vertex([0.0, 0.0, 1.0]),
72        ];
73        // Winding: 0,3,2 and 0,2,1 → CCW from above → normal +Y
74        let indices = vec![0, 3, 2, 0, 2, 1];
75
76        compute_normals(&mut vertices, &indices);
77
78        for v in &vertices {
79            assert!(
80                (v.normal[0]).abs() < 1e-5,
81                "normal.x should be ~0, got {}",
82                v.normal[0]
83            );
84            assert!(
85                (v.normal[1] - 1.0).abs() < 1e-5,
86                "normal.y should be ~1, got {}",
87                v.normal[1]
88            );
89            assert!(
90                (v.normal[2]).abs() < 1e-5,
91                "normal.z should be ~0, got {}",
92                v.normal[2]
93            );
94        }

140    fn area_weighting_favors_large_triangles() {
141        // Shared vertex between a small and large triangle on different planes.
142        // The large triangle's normal should dominate.
143        let mut vertices = vec![
144            // Small triangle in XY plane (area = 0.005)
145            default_vertex([0.0, 0.0, 0.0]), // 0 (shared)
146            default_vertex([0.1, 0.0, 0.0]), // 1
147            default_vertex([0.0, 0.1, 0.0]), // 2
148            // Large triangle in XZ plane (area = 50)
149            default_vertex([10.0, 0.0, 0.0]), // 3
150            default_vertex([0.0, 0.0, 10.0]), // 4
151        ];
152        // Small tri: 0,1,2 → normal +Z
153        // Large tri: 0,3,4 → normal +Y
154        let indices = vec![0, 1, 2, 0, 3, 4];
155
156        compute_normals(&mut vertices, &indices);
157
158        let shared = Vec3::from_array(vertices[0].normal);
159        // Y component (from large tri) should dominate over Z component (from small tri)
160        assert!(
161            shared.y.abs() > shared.z.abs() * 10.0,
162            "large triangle should dominate: y={}, z={}",
163            shared.y,
164            shared.z
165        );
