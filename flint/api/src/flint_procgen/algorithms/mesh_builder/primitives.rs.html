53/// Generate a tapered cylinder with linearly interpolated radius.
54///
55/// Body vertices: `(radial_segments + 1) * (height_segments + 1)`
56/// Body triangles: `radial_segments * height_segments * 2`
57/// Each cap adds `radial_segments + 1` vertices and `radial_segments` triangles.
58///
59/// The UV seam is handled by duplicating the first column with u=1.0.
60/// Normals are analytically computed from the cone slope.
61pub fn tapered_cylinder(params: &TaperedCylinderParams) -> MeshData {
62    let r = params.radial_segments;
63    let h = params.height_segments;
64
65    let mut vertices = Vec::new();
66    let mut indices = Vec::new();
67
68    // Slope for normal Y-component: the normal tilts outward based on taper
69    let slope = (params.radius_bottom - params.radius_top) / params.height;
70
71    // Body vertices: rings along height
72    for j in 0..=h {
73        let t = j as f32 / h as f32;
74        let y = t * params.height;
75        let radius = params.radius_bottom + (params.radius_top - params.radius_bottom) * t;
76
77        for i in 0..=r {
78            let theta = (i as f32 / r as f32) * TAU;
79            let cos_t = theta.cos();
80            let sin_t = theta.sin();
81
82            let x = radius * cos_t;
83            let z = radius * sin_t;
84
85            // Normal: outward radial + slope component
86            let nx = cos_t;
87            let nz = sin_t;
88            let ny = slope;
89            let n_len = (nx * nx + ny * ny + nz * nz).sqrt();
90            let (nx, ny, nz) = if n_len > 1e-10 {
91                (nx / n_len, ny / n_len, nz / n_len)
92            } else {
93                (0.0, 1.0, 0.0)
94            };
95
96            let u = i as f32 / r as f32;
97            let v = t;
98
99            vertices.push(Vertex {
100                position: [x, y, z],
101                normal: [nx, ny, nz],
102                tangent: [0.0, 0.0, 0.0, 1.0], // computed later if needed
103                uv: [u, v],
104            });
105        }
106    }
107
108    // Body indices
109    for j in 0..h {
110        for i in 0..r {
111            let row_width = r + 1;
112            let a = j * row_width + i;
113            let b = a + 1;
114            let c = (j + 1) * row_width + i;
115            let d = c + 1;
116
117            indices.push(a);
118            indices.push(c);
119            indices.push(b);
120
121            indices.push(b);
122            indices.push(c);
123            indices.push(d);
124        }
125    }
126
127    // Bottom cap
128    if params.caps && params.radius_bottom > 1e-10 {
129        let center_idx = vertices.len() as u32;
130        vertices.push(Vertex {
131            position: [0.0, 0.0, 0.0],
132            normal: [0.0, -1.0, 0.0],
133            tangent: [1.0, 0.0, 0.0, 1.0],
134            uv: [0.5, 0.5],
135        });
136
137        let first_ring = vertices.len() as u32;
138        for i in 0..=r {
139            let theta = (i as f32 / r as f32) * TAU;
140            let cos_t = theta.cos();
141            let sin_t = theta.sin();
142            let x = params.radius_bottom * cos_t;
143            let z = params.radius_bottom * sin_t;
144            vertices.push(Vertex {
145                position: [x, 0.0, z],
146                normal: [0.0, -1.0, 0.0],
147                tangent: [1.0, 0.0, 0.0, 1.0],
148                uv: [cos_t * 0.5 + 0.5, sin_t * 0.5 + 0.5],
149            });
150        }
151
152        for i in 0..r {
153            // Winding: clockwise when looking down (-Y)
154            indices.push(center_idx);
155            indices.push(first_ring + i + 1);
156            indices.push(first_ring + i);
157        }
158    }
159
160    // Top cap
161    if params.caps && params.radius_top > 1e-10 {
162        let center_idx = vertices.len() as u32;
163        vertices.push(Vertex {
164            position: [0.0, params.height, 0.0],
165            normal: [0.0, 1.0, 0.0],
166            tangent: [1.0, 0.0, 0.0, 1.0],
167            uv: [0.5, 0.5],
168        });
169
170        let first_ring = vertices.len() as u32;
171        for i in 0..=r {
172            let theta = (i as f32 / r as f32) * TAU;
173            let cos_t = theta.cos();
174            let sin_t = theta.sin();
175            let x = params.radius_top * cos_t;
176            let z = params.radius_top * sin_t;
177            vertices.push(Vertex {
178                position: [x, params.height, z],
179                normal: [0.0, 1.0, 0.0],
180                tangent: [1.0, 0.0, 0.0, 1.0],
181                uv: [cos_t * 0.5 + 0.5, sin_t * 0.5 + 0.5],
182            });
183        }
184
185        for i in 0..r {
186            // Winding: counter-clockwise when looking down (+Y)
187            indices.push(center_idx);
188            indices.push(first_ring + i);
189            indices.push(first_ring + i + 1);
190        }
191    }
192
193    let bbox =
194        BoundingBox::from_positions(&vertices.iter().map(|v| v.position).collect::<Vec<_>>());
195
196    MeshData {
197        vertices,
198        indices,
199        materials: vec![MaterialData::default()],
200        submeshes: vec![],
201        bounding_box: bbox,
202    }
203}

217/// Generate a UV sphere.
218///
219/// Vertices: `(segments + 1) * (rings + 1)`
220/// Triangles: `segments * rings * 2`
221pub fn sphere(params: &SphereParams) -> MeshData {
222    let s = params.segments;
223    let ri = params.rings;
224
225    let mut vertices = Vec::new();
226    let mut indices = Vec::new();
227
228    for j in 0..=ri {
229        let phi = std::f32::consts::PI * j as f32 / ri as f32;
230        let sin_phi = phi.sin();
231        let cos_phi = phi.cos();
232
233        for i in 0..=s {
234            let theta = TAU * i as f32 / s as f32;
235            let sin_theta = theta.sin();
236            let cos_theta = theta.cos();
237
238            let x = sin_phi * cos_theta;
239            let y = cos_phi;
240            let z = sin_phi * sin_theta;
241
242            let u = i as f32 / s as f32;
243            let v = j as f32 / ri as f32;
244
245            vertices.push(Vertex {
246                position: [x * params.radius, y * params.radius, z * params.radius],
247                normal: [x, y, z], // normalized position = normal for unit sphere
248                tangent: [0.0, 0.0, 0.0, 1.0],
249                uv: [u, v],
250            });
251        }
252    }
253
254    let row_width = s + 1;
255    for j in 0..ri {
256        for i in 0..s {
257            let a = j * row_width + i;
258            let b = a + 1;
259            let c = (j + 1) * row_width + i;
260            let d = c + 1;
261
262            indices.push(a);
263            indices.push(c);
264            indices.push(b);
265
266            indices.push(b);
267            indices.push(c);
268            indices.push(d);
269        }
270    }
271
272    let bbox =
273        BoundingBox::from_positions(&vertices.iter().map(|v| v.position).collect::<Vec<_>>());
274
275    MeshData {
276        vertices,
277        indices,
278        materials: vec![MaterialData::default()],
279        submeshes: vec![],
280        bounding_box: bbox,
281    }
282}

284/// Generate an ellipsoid (UV sphere with per-axis radii).
285///
286/// Vertices: `(segments + 1) * (rings + 1)`
287/// Triangles: `segments * rings * 2`
288pub fn ellipsoid(params: &EllipsoidParams) -> MeshData {
289    let s = params.segments;
290    let ri = params.rings;
291
292    let mut vertices = Vec::new();
293    let mut indices = Vec::new();
294
295    for j in 0..=ri {
296        let phi = std::f32::consts::PI * j as f32 / ri as f32;
297        let sin_phi = phi.sin();
298        let cos_phi = phi.cos();
299
300        for i in 0..=s {
301            let theta = TAU * i as f32 / s as f32;
302            let sin_theta = theta.sin();
303            let cos_theta = theta.cos();
304
305            // Unit sphere direction
306            let ux = sin_phi * cos_theta;
307            let uy = cos_phi;
308            let uz = sin_phi * sin_theta;
309
310            // Position scaled by per-axis radii
311            let px = ux * params.radius_x;
312            let py = uy * params.radius_y;
313            let pz = uz * params.radius_z;
314
315            // Normal: gradient of the implicit ellipsoid surface x²/a² + y²/b² + z²/c² = 1
316            // ∇ = (2x/a², 2y/b², 2z/c²), normalized
317            let nx = ux / params.radius_x;
318            let ny = uy / params.radius_y;
319            let nz = uz / params.radius_z;
320            let n_len = (nx * nx + ny * ny + nz * nz).sqrt();
321            let (nx, ny, nz) = if n_len > 1e-10 {
322                (nx / n_len, ny / n_len, nz / n_len)
323            } else {
324                (0.0, 1.0, 0.0)
325            };
326
327            let u = i as f32 / s as f32;
328            let v = j as f32 / ri as f32;
329
330            vertices.push(Vertex {
331                position: [px, py, pz],
332                normal: [nx, ny, nz],
333                tangent: [0.0, 0.0, 0.0, 1.0],
334                uv: [u, v],
335            });
336        }
337    }
338
339    let row_width = s + 1;
340    for j in 0..ri {
341        for i in 0..s {
342            let a = j * row_width + i;
343            let b = a + 1;
344            let c = (j + 1) * row_width + i;
345            let d = c + 1;
346
347            indices.push(a);
348            indices.push(c);
349            indices.push(b);
350
351            indices.push(b);
352            indices.push(c);
353            indices.push(d);
354        }
355    }
356
357    let bbox =
358        BoundingBox::from_positions(&vertices.iter().map(|v| v.position).collect::<Vec<_>>());
359
360    MeshData {
361        vertices,
362        indices,
363        materials: vec![MaterialData::default()],
364        submeshes: vec![],
365        bounding_box: bbox,
366    }
367}

369/// Generate a cone (tapered cylinder with radius_top = 0).

371/// Apex vertices share position but get per-face slant normals

377        height: params.height,

381    })

727        let top_ring_start = (8 + 1) * 4; // row_width * height_segments
