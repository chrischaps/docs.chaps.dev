30/// Incremental mesh builder for composing geometry from primitives and raw data.
31pub struct MeshBuilder {
32    vertices: Vec<Vertex>,
33    indices: Vec<u32>,
34    materials: Vec<MaterialData>,
35    submeshes: Vec<SubMesh>,
36}
37
38impl MeshBuilder {
39    /// Create an empty mesh builder.
40    pub fn new() -> Self {
41        Self {
42            vertices: Vec::new(),
43            indices: Vec::new(),
44            materials: Vec::new(),
45            submeshes: Vec::new(),
46        }
47    }
48
49    /// Create with pre-allocated capacity.
50    pub fn with_capacity(verts: usize, indices: usize) -> Self {
51        Self {
52            vertices: Vec::with_capacity(verts),
53            indices: Vec::with_capacity(indices),
54            materials: Vec::new(),
55            submeshes: Vec::new(),
56        }
57    }
58
59    /// Add a single vertex, returning its index.
60    pub fn add_vertex(&mut self, vertex: Vertex) -> u32 {
61        let idx = self.vertices.len() as u32;
62        self.vertices.push(vertex);
63        idx
64    }
65
66    /// Add a triangle from three vertex indices.
67    pub fn add_triangle(&mut self, i0: u32, i1: u32, i2: u32) {
68        self.indices.push(i0);
69        self.indices.push(i1);
70        self.indices.push(i2);
71    }
72
73    /// Add a quad as two triangles (i0-i1-i2, i0-i2-i3).
74    pub fn add_quad(&mut self, i0: u32, i1: u32, i2: u32, i3: u32) {
75        self.add_triangle(i0, i1, i2);
76        self.add_triangle(i0, i2, i3);
77    }
78
79    /// Add a material, returning its index.
80    pub fn add_material(&mut self, material: MaterialData) -> usize {
81        let idx = self.materials.len();
82        self.materials.push(material);
83        idx
84    }
85
86    /// Merge another MeshData into this builder, offsetting indices.
87    pub fn merge(&mut self, mesh: &MeshData) {
88        let base = self.vertices.len() as u32;
89        self.vertices.extend_from_slice(&mesh.vertices);
90        self.indices.extend(mesh.indices.iter().map(|i| i + base));
91    }
92
93    /// Merge another MeshData and record a [`SubMesh`] for the added indices.
94    pub fn merge_as_submesh(&mut self, mesh: &MeshData, material_index: usize) {
95        let index_start = self.indices.len() as u32;
96        self.merge(mesh);
97        let index_count = mesh.indices.len() as u32;
98        self.submeshes.push(SubMesh {
99            index_start,
100            index_count,
101            material_index,
102        });
103    }
104
105    /// Translate all vertex positions by an offset.
106    pub fn translate(&mut self, offset: Vec3) {
107        for v in &mut self.vertices {
108            let p = Vec3::from_array(v.position) + offset;
109            v.position = p.to_array();
110        }
111    }
112
113    /// Uniformly scale all vertex positions around the origin.
114    pub fn scale(&mut self, factor: f32) {
115        for v in &mut self.vertices {
116            let p = Vec3::from_array(v.position) * factor;
117            v.position = p.to_array();
118        }
119    }
120
121    /// Compute area-weighted smooth normals for the current geometry.
122    pub fn compute_normals(&mut self) {
123        normals::compute_normals(&mut self.vertices, &self.indices);
124    }
125
126    /// Compute MikkTSpace-style tangents for the current geometry.
127    pub fn compute_tangents(&mut self) {
128        tangents::compute_tangents(&mut self.vertices, &self.indices);
129    }
130
131    /// Finalize into a MeshData with computed bounding box.
132    pub fn build(self) -> MeshData {
133        let bbox = BoundingBox::from_positions(
134            &self.vertices.iter().map(|v| v.position).collect::<Vec<_>>(),
135        );
136        MeshData {
137            vertices: self.vertices,
138            indices: self.indices,
139            materials: if self.materials.is_empty() {
140                vec![MaterialData::default()]
141            } else {
142                self.materials
143            },
144            submeshes: self.submeshes,
145            bounding_box: bbox,
146        }
147    }
148
149    /// Finalize with normals and tangents computed automatically.
150    pub fn build_with_computed_normals_and_tangents(mut self) -> MeshData {
151        self.compute_normals();
152        self.compute_tangents();
153        self.build()
154    }
155
156    /// Generate a cylinder and merge it into the builder.
157    pub fn add_cylinder(&mut self, params: &CylinderParams) {
158        let mesh = cylinder(params);
159        self.merge(&mesh);
160    }
161
162    /// Generate a sphere and merge it into the builder.
163    pub fn add_sphere(&mut self, params: &SphereParams) {
164        let mesh = sphere(params);
165        self.merge(&mesh);
166    }
167
168    /// Generate a cone and merge it into the builder.
169    pub fn add_cone(&mut self, params: &ConeParams) {
170        let mesh = cone(params);
171        self.merge(&mesh);
172    }
173
174    /// Generate a tapered cylinder and merge it into the builder.
175    pub fn add_tapered_cylinder(&mut self, params: &TaperedCylinderParams) {
176        let mesh = tapered_cylinder(params);
177        self.merge(&mesh);
178    }
179
180    /// Extrude a tube along a curve and merge it into the builder.
181    pub fn add_extruded_tube(&mut self, params: &ExtrudeParams) -> crate::Result<()> {
182        let mesh = extrude_along_curve(params)?;
183        self.merge(&mesh);
184        Ok(())
185    }
186}

207    #[test]
208    fn builder_add_and_build() {
209        let mut b = MeshBuilder::new();
210        let i0 = b.add_vertex(default_vertex([0.0, 0.0, 0.0]));
211        let i1 = b.add_vertex(default_vertex([1.0, 0.0, 0.0]));
212        let i2 = b.add_vertex(default_vertex([0.0, 1.0, 0.0]));
213        b.add_triangle(i0, i1, i2);
214
215        let mesh = b.build();
216        assert_eq!(mesh.vertex_count(), 3);
217        assert_eq!(mesh.triangle_count(), 1);

233            rings: 3,

237        let total_tris = mesh1.triangle_count() + mesh2.triangle_count();
