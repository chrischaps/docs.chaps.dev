86    /// Merge another MeshData into this builder, offsetting indices.
87    pub fn merge(&mut self, mesh: &MeshData) {
88        let base = self.vertices.len() as u32;
89        self.vertices.extend_from_slice(&mesh.vertices);
90        self.indices.extend(mesh.indices.iter().map(|i| i + base));
91    }

93    /// Merge another MeshData with a column-major affine transform applied
94    /// (the layout of `Transform::to_matrix`).
95    ///
96    /// Positions take the full matrix. Normals take the inverse-transpose of
97    /// its linear part, so non-uniform scale keeps them perpendicular to the
98    /// surface; tangents take the linear part. A mirroring transform
99    /// (negative determinant) reverses each triangle's winding and flips the
100    /// tangent handedness, so the merged copy still faces outward.
101    pub fn merge_transformed(&mut self, mesh: &MeshData, matrix: &[[f32; 4]; 4]) {
102        let base = self.vertices.len() as u32;
103        let m = matrix;
104        // Linear part as rows: lin[row][col] = m[col][row].
105        let lin = [
106            [m[0][0], m[1][0], m[2][0]],
107            [m[0][1], m[1][1], m[2][1]],
108            [m[0][2], m[1][2], m[2][2]],
109        ];
110        // Cofactor matrix = det * inverse-transpose; the det factor washes out
111        // on normalisation, except its sign, which `cof_sign` restores.
112        let cof = [
113            [
114                lin[1][1] * lin[2][2] - lin[1][2] * lin[2][1],
115                lin[1][2] * lin[2][0] - lin[1][0] * lin[2][2],
116                lin[1][0] * lin[2][1] - lin[1][1] * lin[2][0],
117            ],
118            [
119                lin[0][2] * lin[2][1] - lin[0][1] * lin[2][2],
120                lin[0][0] * lin[2][2] - lin[0][2] * lin[2][0],
121                lin[0][1] * lin[2][0] - lin[0][0] * lin[2][1],
122            ],
123            [
124                lin[0][1] * lin[1][2] - lin[0][2] * lin[1][1],
125                lin[0][2] * lin[1][0] - lin[0][0] * lin[1][2],
126                lin[0][0] * lin[1][1] - lin[0][1] * lin[1][0],
127            ],
128        ];
129        let det = lin[0][0] * cof[0][0] + lin[0][1] * cof[0][1] + lin[0][2] * cof[0][2];
130        let mirrored = det < 0.0;
131        let cof_sign = if mirrored { -1.0 } else { 1.0 };
132
133        let apply = |mat: &[[f32; 3]; 3], v: [f32; 3]| -> Vec3 {
134            Vec3::new(
135                mat[0][0] * v[0] + mat[0][1] * v[1] + mat[0][2] * v[2],
136                mat[1][0] * v[0] + mat[1][1] * v[1] + mat[1][2] * v[2],
137                mat[2][0] * v[0] + mat[2][1] * v[1] + mat[2][2] * v[2],
138            )
139        };
140
141        self.vertices.extend(mesh.vertices.iter().map(|v| {
142            let p = apply(&lin, v.position) + Vec3::new(m[3][0], m[3][1], m[3][2]);
143            let n = apply(&cof, v.normal) * cof_sign;
144            let t = apply(&lin, [v.tangent[0], v.tangent[1], v.tangent[2]]);
145            let n = if n.length() > 1e-12 {
146                n.normalized()
147            } else {
148                n
149            };
150            let t = if t.length() > 1e-12 {
151                t.normalized()
152            } else {
153                t
154            };
155            Vertex {
156                position: p.to_array(),
157                normal: n.to_array(),
158                tangent: [t.x, t.y, t.z, v.tangent[3] * cof_sign],
159                uv: v.uv,
160            }
161        }));
162        for tri in mesh.indices.chunks_exact(3) {
163            if mirrored {
164                self.add_triangle(tri[0] + base, tri[2] + base, tri[1] + base);
165            } else {
166                self.add_triangle(tri[0] + base, tri[1] + base, tri[2] + base);
167            }
168        }
169    }

171    /// Merge another MeshData and record a [`SubMesh`] for the added indices.
172    pub fn merge_as_submesh(&mut self, mesh: &MeshData, material_index: usize) {
173        let index_start = self.indices.len() as u32;
174        self.merge(mesh);
175        let index_count = mesh.indices.len() as u32;
176        self.submeshes.push(SubMesh {
177            index_start,
178            index_count,
179            material_index,
180        });
181    }

183    /// Merge another MeshData with its materials, keeping its submeshes.
184    ///
185    /// The mesh's materials are appended to this builder's, and each of its
186    /// submeshes (or the whole mesh, when it has none) is recorded against
187    /// its material's new index.
188    pub fn merge_with_materials(&mut self, mesh: &MeshData) {
189        let material_base = self.materials.len();
190        let index_base = self.indices.len() as u32;
191        self.materials.extend(mesh.materials.iter().cloned());
192        self.merge(mesh);
193        if mesh.submeshes.is_empty() {
194            if !mesh.indices.is_empty() {
195                self.submeshes.push(SubMesh {
196                    index_start: index_base,
197                    index_count: mesh.indices.len() as u32,
198                    material_index: material_base,
199                });
200            }
201        } else {
202            self.submeshes
203                .extend(mesh.submeshes.iter().map(|s| SubMesh {
204                    index_start: index_base + s.index_start,
205                    index_count: s.index_count,
206                    material_index: material_base + s.material_index,
207                }));
208        }
209    }

211    /// Translate all vertex positions by an offset.
212    pub fn translate(&mut self, offset: Vec3) {
213        for v in &mut self.vertices {
214            let p = Vec3::from_array(v.position) + offset;
215            v.position = p.to_array();
216        }
217    }
218
219    /// Uniformly scale all vertex positions around the origin.
220    pub fn scale(&mut self, factor: f32) {
221        for v in &mut self.vertices {
222            let p = Vec3::from_array(v.position) * factor;
223            v.position = p.to_array();
224        }
225    }

237    /// Finalize into a MeshData with computed bounding box.
238    pub fn build(self) -> MeshData {
239        let bbox = BoundingBox::from_positions(
240            &self.vertices.iter().map(|v| v.position).collect::<Vec<_>>(),
241        );
242        MeshData {
243            vertices: self.vertices,
244            indices: self.indices,
245            materials: if self.materials.is_empty() {
246                vec![MaterialData::default()]
247            } else {
248                self.materials
249            },
250            submeshes: self.submeshes,
251            bounding_box: bbox,
252        }
253    }
254
255    /// Finalize with normals and tangents computed automatically.
256    pub fn build_with_computed_normals_and_tangents(mut self) -> MeshData {
257        self.compute_normals();
258        self.compute_tangents();
259        self.build()
260    }

472    /// Every triangle's vertex normals are perpendicular to its edges (for a
473    /// flat-shaded mesh) and agree with its winding.
474    fn assert_flat_normals_match_faces(mesh: &MeshData) {
475        for tri in mesh.indices.chunks_exact(3) {
476            let p = |i: u32| Vec3::from_array(mesh.vertices[i as usize].position);
477            let e1 = p(tri[1]) - p(tri[0]);
478            let e2 = p(tri[2]) - p(tri[0]);
479            let face = e1.cross(&e2).normalized();
480            for &i in tri {
481                let v = &mesh.vertices[i as usize];
482                let n = Vec3::from_array(v.normal);
483                assert!((n.length() - 1.0).abs() < 1e-5, "normal not unit");
484                assert!(n.dot(&face) > 0.9999, "normal off face: {}", n.dot(&face));
485                let t = Vec3::new(v.tangent[0], v.tangent[1], v.tangent[2]);
486                assert!(t.dot(&n).abs() < 1e-5, "tangent leaves the surface");
487            }
488        }
489    }

511    fn merge_matches_transform_to_matrix() {
512        let t = flint_core::Transform::default()
513            .with_position(Vec3::new(1.0, 2.0, 3.0))
514            .with_rotation(Vec3::new(30.0, 45.0, 10.0))
515            .with_scale(Vec3::new(2.0, 2.0, 2.0));
516        let cube = cuboid(1.0, 1.0, 1.0);

524        let expect: Vec<f32> = (0..3)

529        }
