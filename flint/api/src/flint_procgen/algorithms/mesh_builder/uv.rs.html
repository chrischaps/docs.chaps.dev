4/// Projection axis for planar UV mapping.
5#[derive(Debug, Clone, Copy, PartialEq, Eq)]
6pub enum ProjectionAxis {
7    /// Project onto XY plane (u = x, v = y)
8    XY,
9    /// Project onto XZ plane (u = x, v = z)
10    XZ,
11    /// Project onto YZ plane (u = y, v = z)
12    YZ,
13}
14
15/// Apply cylindrical UV mapping around the Y axis.
16///
17/// - `u = atan2(dz, dx) / TAU + 0.5` (wraps 0..1 around the cylinder)
18/// - `v = dy / height` (0 at bottom, 1 at top)
19///
20/// `center` is the base center of the cylinder, `height` is the total height.
21/// `uv_scale` multiplies the final UV coordinates for tiling.
22pub fn apply_cylindrical_uv(
23    vertices: &mut [Vertex],
24    center: Vec3,
25    height: f32,
26    uv_scale: [f32; 2],
27) {
28    let inv_height = if height.abs() > 1e-10 {
29        1.0 / height
30    } else {
31        1.0
32    };
33
34    for v in vertices.iter_mut() {
35        let p = Vec3::from_array(v.position);
36        let d = p - center;
37
38        let u = d.z.atan2(d.x) / std::f32::consts::TAU + 0.5;
39        let vy = d.y * inv_height;
40
41        v.uv = [u * uv_scale[0], vy * uv_scale[1]];
42    }
43}
44
45/// Apply planar UV projection along the given axis.
46///
47/// The two remaining axes become u and v, scaled by `scale`.
48pub fn apply_planar_uv(vertices: &mut [Vertex], axis: ProjectionAxis, scale: f32) {
49    let inv_scale = if scale.abs() > 1e-10 {
50        1.0 / scale
51    } else {
52        1.0
53    };
54
55    for v in vertices.iter_mut() {
56        let p = v.position;
57        let (u, vy) = match axis {
58            ProjectionAxis::XY => (p[0], p[1]),
59            ProjectionAxis::XZ => (p[0], p[2]),
60            ProjectionAxis::YZ => (p[1], p[2]),
61        };
62        v.uv = [u * inv_scale, vy * inv_scale];
63    }
64}

80    #[test]
81    fn cylindrical_uv_wraps_zero_to_one() {
82        let mut vertices = vec![
83            default_vertex([1.0, 0.0, 0.0]),  // atan2(0, 1) = 0 → u = 0.5
84            default_vertex([0.0, 0.0, 1.0]),  // atan2(1, 0) = PI/2 → u = 0.75
85            default_vertex([-1.0, 0.0, 0.0]), // atan2(0, -1) = PI → u = 1.0 (or 0.0)
86            default_vertex([0.0, 5.0, 0.0]),  // top center → v = 1.0
87        ];
88
89        apply_cylindrical_uv(&mut vertices, Vec3::ZERO, 5.0, [1.0, 1.0]);
90
91        // u should be in [0, 1]
92        for v in &vertices {
93            assert!(
94                v.uv[0] >= -0.01 && v.uv[0] <= 1.01,
95                "u should be in [0,1], got {}",
96                v.uv[0]
97            );
98        }
99
100        // Check specific values
101        assert!((vertices[0].uv[0] - 0.5).abs() < 1e-4, "atan2(0,1) → u=0.5");
102        assert!(
103            (vertices[1].uv[0] - 0.75).abs() < 1e-4,
104            "atan2(1,0) → u=0.75"
105        );
106        assert!((vertices[3].uv[1] - 1.0).abs() < 1e-4, "top → v=1.0");
107    }
