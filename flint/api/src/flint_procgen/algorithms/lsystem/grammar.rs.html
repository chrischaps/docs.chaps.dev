1//! Text L-systems: grammars written as strings, the way they are printed in
2//! *The Algorithmic Beauty of Plants* (Prusinkiewicz & Lindenmayer).
3//!
4//! ```text
5//! axiom:  X
6//! rules:  X -> F+[[X]-X]-F[-FX]+X
7//!         F -> FF
8//! ```
9//!
10//! [`Grammar::expand`] rewrites the axiom; [`interpret`] walks the result with
11//! a 3D turtle and returns [`BranchSegment`]s.
12//!
13//! # Rules
14//!
15//! A rule is `"<symbol> -> <successor>"`. Several rules may share a
16//! predecessor, each with a weight: `"F (0.33) -> F[+F]F"`. When a symbol has
17//! more than one rule, one is drawn per occurrence with probability
18//! proportional to its weight; a symbol with a single rule never touches the
19//! RNG, so deterministic grammars ignore the seed. `→` works as the arrow too,
20//! and whitespace is ignored everywhere.
21//!
22//! A symbol written with arguments, `F(2,0.5)`, is still the symbol `F` to the
23//! rewriter: a rule for `F` replaces it, arguments and all, and a symbol with
24//! no rule is copied as written.
25//!
26//! # Turtle symbols
27//!
28//! | Symbol | Meaning | Argument |
29//! |---|---|---|
30//! | `F`, `G` | draw forward | `F(length)` or `F(length, radius)` |
31//! | `f` | move forward without drawing | `f(length)` |
32//! | `+` / `-` | yaw left / right | angle in degrees |
33//! | `^` / `&` | pitch up / down | angle |
34//! | `/` / `\` | roll clockwise / counter-clockwise | angle |
35//! | `\|` | turn around (yaw 180°) | |
36//! | `[` / `]` | push / pop the turtle | |
37//! | `!` | thin the branch by `radius_decay` | `!(radius)` sets it |
38//!
39//! Any other symbol (`X`, `A`, `L`...) only drives rewriting; the turtle skips
40//! it. Without arguments, a step is `step_length` long and the rotations use
41//! the configured angles.
42//!
43//! Length and radius decay by **nesting level**: every `[` multiplies the
44//! default step and the current radius by `length_decay` and `radius_decay`,
45//! and `]` restores them. A branch's children are shorter and thinner than it
46//! is however many rewriting passes produced them. Explicit arguments are
47//! used as written.
48

59/// One symbol of an L-system string, with its arguments if it has any.
60#[derive(Debug, Clone, Copy, PartialEq)]
61struct Token<'a> {
62    symbol: char,
63    /// The text between the parentheses, if the symbol was written `X(...)`.
64    args: Option<&'a str>,
65    /// The symbol exactly as written, arguments included.
66    text: &'a str,
67    /// Byte offset in the source string, for error messages.
68    offset: usize,
69}
70
71/// Split an L-system string into symbols. Whitespace is skipped.
72fn tokenize(s: &str) -> Result<Vec<Token<'_>>> {
73    let mut tokens = Vec::with_capacity(s.len());
74    let mut chars = s.char_indices().peekable();
75    while let Some((offset, symbol)) = chars.next() {
76        if symbol.is_whitespace() {
77            continue;
78        }
79        if symbol == '(' || symbol == ')' {
80            return Err(syntax_error(s, offset, "parenthesis without a symbol"));
81        }
82        let mut end = offset + symbol.len_utf8();
83        let mut args = None;
84        if let Some(&(open, '(')) = chars.peek() {
85            let close = s[open..]
86                .find(')')
87                .map(|i| open + i)
88                .ok_or_else(|| syntax_error(s, open, "unclosed '('"))?;
89            args = Some(&s[open + 1..close]);
90            end = close + 1;
91            while chars.peek().is_some_and(|&(i, _)| i < end) {
92                chars.next();
93            }
94        }
95        tokens.push(Token {
96            symbol,
97            args,
98            text: &s[offset..end],
99            offset,
100        });
101    }
102    Ok(tokens)
103}

105/// Parse a comma-separated argument list: `"1.5, 0.3"` -> `[1.5, 0.3]`.
106fn parse_args(args: &str) -> std::result::Result<Vec<f32>, String> {
107    args.split(',')
108        .map(|a| {
109            let a = a.trim();
110            a.parse::<f32>()
111                .ok()
112                .filter(|v| v.is_finite())
113                .ok_or_else(|| format!("'{a}' is not a number"))
114        })
115        .collect()
116}

118fn syntax_error(s: &str, offset: usize, reason: &str) -> ProcGenError {
119    ProcGenError::InvalidParameter {
120        name: "lsystem".into(),
121        reason: format!("{reason} at byte {offset} of \"{}\"", clip(s)),
122    }
123}
124
125/// The string, shortened for an error message.
126fn clip(s: &str) -> String {
127    const MAX: usize = 60;
128    if s.chars().count() <= MAX {
129        s.to_string()
130    } else {
131        format!("{}...", s.chars().take(MAX).collect::<String>())
132    }
133}
134
135/// Tokenize `s`, parse every argument list, and check that brackets balance.
136/// Returns the string with whitespace removed.
137fn validate(s: &str, what: &str) -> Result<String> {
138    let tokens = tokenize(s)?;
139    let mut depth = 0i32;
140    for t in &tokens {
141        if let Some(args) = t.args {
142            parse_args(args).map_err(|e| syntax_error(s, t.offset, &e))?;
143        }
144        match t.symbol {
145            '[' => depth += 1,
146            ']' => {
147                depth -= 1;
148                if depth < 0 {
149                    return Err(syntax_error(s, t.offset, "']' without a matching '['"));
150                }
151            }
152            _ => {}
153        }
154    }
155    if depth != 0 {
156        return Err(ProcGenError::InvalidParameter {
157            name: "lsystem".into(),
158            reason: format!("{what} \"{}\" leaves {depth} '[' unclosed", clip(s)),
159        });
160    }
161    Ok(tokens.iter().map(|t| t.text).collect())
162}

166/// A production: every occurrence of `predecessor` is replaced by `successor`.
167#[derive(Debug, Clone, PartialEq)]
168pub struct ProductionRule {
169    /// The symbol this rule rewrites.
170    pub predecessor: char,
171    /// What it becomes. Brackets balance; may be empty (the symbol is erased).
172    pub successor: String,
173    /// Relative weight among rules with the same predecessor (stochastic
174    /// L-systems). Ignored when the predecessor has only one rule.
175    pub weight: f32,
176}
177
178impl ProductionRule {
179    /// A deterministic rule `predecessor -> successor`.
180    pub fn new(predecessor: char, successor: &str) -> Result<Self> {
181        Self::weighted(predecessor, successor, 1.0)
182    }
183
184    /// A rule that competes with others for the same predecessor.
185    pub fn weighted(predecessor: char, successor: &str, weight: f32) -> Result<Self> {
186        if predecessor.is_whitespace() || "[]()".contains(predecessor) {
187            return Err(ProcGenError::InvalidParameter {
188                name: "lsystem rule".into(),
189                reason: format!("'{predecessor}' cannot be rewritten"),
190            });
191        }
192        if !(weight.is_finite() && weight > 0.0) {
193            return Err(ProcGenError::InvalidParameter {
194                name: "lsystem rule".into(),
195                reason: format!("weight {weight} for '{predecessor}' must be positive"),
196            });
197        }
198        Ok(Self {
199            predecessor,
200            successor: validate(successor, "successor")?,
201            weight,
202        })
203    }
204
205    /// Parse `"F -> FF"` or, with a weight, `"F (0.33) -> F[+F]F"`.
206    pub fn parse(rule: &str) -> Result<Self> {
207        let bad = |reason: &str| ProcGenError::InvalidParameter {
208            name: "lsystem rule".into(),
209            reason: format!("{reason} in \"{}\"", clip(rule)),
210        };
211        let (lhs, successor) = rule
212            .split_once("->")
213            .or_else(|| rule.split_once('→'))
214            .ok_or_else(|| bad("expected '->'"))?;
215        let lhs = lhs.trim();
216        let mut chars = lhs.chars();
217        let predecessor = chars.next().ok_or_else(|| bad("missing predecessor"))?;
218        let rest = chars.as_str().trim();
219        let weight = if rest.is_empty() {
220            1.0
221        } else {
222            rest.strip_prefix('(')
223                .and_then(|r| r.strip_suffix(')'))
224                .and_then(|w| w.trim().parse::<f32>().ok())
225                .ok_or_else(|| bad("expected one symbol, optionally followed by a (weight)"))?
226        };
227        Self::weighted(predecessor, successor, weight)
228    }
229}

231/// A text L-system: an axiom and the rules that rewrite it.
232#[derive(Debug, Clone, PartialEq)]
233pub struct Grammar {
234    /// The starting string.
235    pub axiom: String,
236    /// Productions, in the order they were added.
237    pub rules: Vec<ProductionRule>,
238    /// Budget guard: expansion fails once the string grows past this many
239    /// bytes. Rewriting is exponential; this stops a typo from eating memory.
240    pub max_length: usize,
241}
242
243impl Grammar {
244    /// Default for [`Grammar::max_length`]: 4 MB of symbols.
245    pub const DEFAULT_MAX_LENGTH: usize = 4 << 20;
246
247    /// A grammar with no rules yet.
248    pub fn new(axiom: &str) -> Result<Self> {
249        Ok(Self {
250            axiom: validate(axiom, "axiom")?,
251            rules: Vec::new(),
252            max_length: Self::DEFAULT_MAX_LENGTH,
253        })
254    }
255
256    /// Parse an axiom and rules written as `"F -> FF"` strings.
257    pub fn parse<S: AsRef<str>>(axiom: &str, rules: &[S]) -> Result<Self> {
258        let mut grammar = Self::new(axiom)?;
259        for rule in rules {
260            grammar.rules.push(ProductionRule::parse(rule.as_ref())?);
261        }
262        Ok(grammar)
263    }
264
265    /// Add a rule.
266    pub fn rule(mut self, rule: ProductionRule) -> Self {
267        self.rules.push(rule);
268        self
269    }
270
271    /// Rewrite the axiom `iterations` times.
272    pub fn expand(&self, iterations: usize, rng: &mut SeededRng) -> Result<String> {
273        // Per predecessor: its successors with their running weight totals.
274        let mut choices: HashMap<char, Vec<(f32, &str)>> = HashMap::new();
275        for r in &self.rules {
276            let list = choices.entry(r.predecessor).or_default();
277            let total = list.last().map_or(0.0, |&(t, _)| t);
278            list.push((total + r.weight, &r.successor));
279        }
280
281        let mut current = self.axiom.clone();
282        for _ in 0..iterations {
283            let mut next = String::with_capacity(current.len() * 2);
284            for token in tokenize(&current)? {
285                let replacement = match choices.get(&token.symbol).map(Vec::as_slice) {
286                    None => token.text,
287                    Some([(_, only)]) => only,
288                    Some(list) => {
289                        let total = list.last().map_or(0.0, |&(t, _)| t);
290                        let pick = rng.next_f32() * total;
291                        list.iter()
292                            .find(|&&(t, _)| pick < t)
293                            .or(list.last())
294                            .map_or(token.text, |&(_, s)| s)
295                    }
296                };
297                next.push_str(replacement);
298                if next.len() > self.max_length {
299                    return Err(ProcGenError::BudgetExceeded(format!(
300                        "L-system string passed max_length {} bytes",
301                        self.max_length
302                    )));
303                }
304            }
305            current = next;
306        }
307        Ok(current)
308    }
309
310    /// Expand and interpret in one call.
311    pub fn generate(
312        &self,
313        iterations: usize,
314        config: &TurtleConfig,
315        rng: &mut SeededRng,
316    ) -> Result<Vec<BranchSegment>> {
317        interpret(&self.expand(iterations, rng)?, config)
318    }
319}

323/// How the turtle reads an expanded string.
324#[derive(Debug, Clone, PartialEq)]
325pub struct TurtleConfig {
326    /// Length of a bare `F` at nesting level 0.
327    pub step_length: f32,
328    /// Branch radius at nesting level 0.
329    pub radius: f32,
330    /// Step length multiplier per `[`.
331    pub length_decay: f32,
332    /// Radius multiplier per `[` (and per `!`).
333    pub radius_decay: f32,
334    /// Default angle for `+` and `-`, in degrees.
335    pub yaw_deg: f32,
336    /// Default angle for `^` and `&`, in degrees.
337    pub pitch_deg: f32,
338    /// Default angle for `/` and `\`, in degrees.
339    pub roll_deg: f32,
340    /// Direction branches bend toward after each drawn step: down for
341    /// gravity, up for phototropism. Ignored while `tropism_strength` is 0.
342    pub tropism: Vec3,
343    /// How strongly the heading bends toward `tropism`: each step turns it
344    /// by `strength * |heading x tropism|` radians, as in ABOP §2.2.
345    pub tropism_strength: f32,
346    /// Budget guard on emitted segments.
347    pub max_segments: u32,
348}

377/// Turtle state plus the per-branch scale that `[` and `]` save and restore.
378#[derive(Clone)]
379struct Frame {
380    turtle: TurtleState,
381    step: f32,
382    radius: f32,
383}
384
385/// Walk an expanded L-system string with a 3D turtle.
386///
387/// The turtle starts at the origin heading +Y. A `]` with no matching `[`
388/// is an error; a `[` left open at the end is not.
389pub fn interpret(expanded: &str, config: &TurtleConfig) -> Result<Vec<BranchSegment>> {
390    let positive = |v: f32| v.is_finite() && v > 0.0;
391    if !positive(config.length_decay) || !positive(config.radius_decay) {
392        return Err(ProcGenError::InvalidParameter {
393            name: "TurtleConfig".into(),
394            reason: "length_decay and radius_decay must be positive".into(),
395        });
396    }
397
398    let tropism = config.tropism.normalized();
399    let mut segments = Vec::new();
400    let mut frame = Frame {
401        turtle: TurtleState::default(),
402        step: config.step_length,
403        radius: config.radius,
404    };
405    let mut stack: Vec<Frame> = Vec::new();
406
407    for token in tokenize(expanded)? {
408        let args = match token.args {
409            Some(a) => parse_args(a).map_err(|e| syntax_error(expanded, token.offset, &e))?,
410            None => Vec::new(),
411        };
412        let arg = |i: usize, default: f32| args.get(i).copied().unwrap_or(default);
413        let t = &mut frame.turtle;
414        match token.symbol {
415            'F' | 'G' => {
416                let length = arg(0, frame.step);
417                let radius = arg(1, frame.radius);
418                let end = t.ahead(length);
419                segments.push(BranchSegment {
420                    start: t.position,
421                    end,
422                    radius_start: radius,
423                    radius_end: radius,
424                    depth: t.depth,
425                });
426                if segments.len() > config.max_segments as usize {
427                    return Err(ProcGenError::BudgetExceeded(format!(
428                        "segment count {} exceeds max_segments {}",
429                        segments.len(),
430                        config.max_segments
431                    )));
432                }
433                t.position = end;
434                if config.tropism_strength != 0.0 {
435                    bend_toward(t, tropism, config.tropism_strength);
436                }
437            }
438            'f' => t.position = t.ahead(arg(0, frame.step)),
439            '+' => t.yaw(arg(0, config.yaw_deg)),
440            '-' => t.yaw(-arg(0, config.yaw_deg)),
441            '^' => t.pitch(arg(0, config.pitch_deg)),
442            '&' => t.pitch(-arg(0, config.pitch_deg)),
443            '/' => t.roll(arg(0, config.roll_deg)),
444            '\\' => t.roll(-arg(0, config.roll_deg)),
445            '|' => t.yaw(180.0),
446            '!' => frame.radius = arg(0, frame.radius * config.radius_decay),
447            '[' => {
448                stack.push(frame.clone());
449                frame.turtle.depth += 1;
450                frame.step *= config.length_decay;
451                frame.radius *= config.radius_decay;
452            }
453            ']' => {
454                frame = stack.pop().ok_or_else(|| {
455                    syntax_error(expanded, token.offset, "']' without a matching '['")
456                })?;
457            }
458            _ => {}
459        }
460    }
461
462    Ok(segments)
463}

465/// Turn the turtle's frame toward `direction` by `strength * |H x T|`.
466fn bend_toward(t: &mut TurtleState, direction: Vec3, strength: f32) {
467    let axis = t.heading.cross(&direction);
468    let sin = axis.length();
469    if sin < 1e-6 {
470        return;
471    }
472    let axis = axis * (1.0 / sin);
473    let angle = strength * sin;
474    t.heading = rotate_around_axis(t.heading, axis, angle);
475    t.up = rotate_around_axis(t.up, axis, angle);
476    t.right = rotate_around_axis(t.right, axis, angle);
477}

667    #[test]
668    fn rule_parsing() {
669        let r = ProductionRule::parse("F (0.33) -> F[+F]F").unwrap();
670        assert_eq!(r.predecessor, 'F');
671        assert_eq!(r.successor, "F[+F]F");
672        assert!((r.weight - 0.33).abs() < 1e-6);
673        let r = ProductionRule::parse("X → F-X").unwrap();
674        assert_eq!((r.predecessor, r.weight), ('X', 1.0));
675
676        for bad in [
677            "F FF",          // no arrow
678            "-> FF",         // no predecessor
679            "FG -> F",       // two predecessors
680            "F (x) -> F",    // weight not a number
681            "F (0) -> F",    // weight not positive
682            "[ -> F",        // brackets aren't rewritable
683            "F -> F[+F",     // unbalanced successor
684            "F -> F]",       // ditto
685            "F -> F(1,",     // unclosed argument list
686            "F -> F(1,two)", // bad argument
687        ] {
688            assert!(ProductionRule::parse(bad).is_err(), "accepted {bad:?}");
689        }
690    }
