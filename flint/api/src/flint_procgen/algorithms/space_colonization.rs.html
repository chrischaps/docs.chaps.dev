1//! Space colonization algorithm for organic branching structures.
2//!
3//! Grows a tree graph toward a cloud of attraction points scattered within a
4//! crown volume (Runions et al., *Modeling Trees with a Space Colonization
5//! Algorithm*, 2007). Branches compete for the space they grow into, so the
6//! canopy fills its volume with adaptive branching density, which reads as
7//! more natural than an L-system for broadleaf crowns, roots and veins.
8//!
9//! # Algorithm Overview
10//!
11//! 1. Scatter attraction points inside a [`CrownVolume`] via rejection
12//!    sampling ([`scatter_attraction_points`])
13//! 2. Grow a straight trunk from the root upward
14//! 3. Each iteration, every attraction point pulls on its closest node within
15//!    `influence_distance`; every pulled node grows one `step_size` toward the
16//!    mean of its pulls
17//! 4. Remove attraction points within `kill_distance` of a node
18//! 5. Stop when no points remain, none can pull, or `max_iterations` is hit
19//! 6. Assign branching depth and radii via the pipe model
20//!    (`r_parent^n = sum(r_child^n)`, da Vinci's rule at n = 2)
21//! 7. Convert the graph to `Vec<BranchSegment>` ([`branch_graph_to_segments`]),
22//!    the same format the L-system produces
23//!
24//! Each attraction point remembers its closest node; nodes are never removed,
25//! so an iteration only compares points against the nodes it just added. That
26//! keeps a step at O(points x new nodes) instead of O(points x all nodes).
27

33// ─── A. Crown Volume ────────────────────────────────────────────────────────
34
35/// Defines the 3D shape within which attraction points are scattered.
36#[derive(Debug, Clone, Copy, PartialEq)]
37pub enum CrownVolume {
38    /// A sphere centered at `center` with the given `radius`.
39    Sphere { center: Vec3, radius: f32 },
40    /// An ellipsoid centered at `center` with semi-axis lengths `radii` (rx, ry, rz).
41    Ellipsoid { center: Vec3, radii: Vec3 },
42    /// Upper hemisphere (y >= center.y) centered at `center` with the given `radius`.
43    Hemisphere { center: Vec3, radius: f32 },
44    /// An axis-aligned box from `min` to `max` (hedges, espaliers, root beds).
45    Box { min: Vec3, max: Vec3 },
46    /// An upright cone: a disc of `radius` at `base`, narrowing to a point
47    /// `height` above it (conifers).
48    Cone {
49        base: Vec3,
50        radius: f32,
51        height: f32,
52    },
53}
54
55impl CrownVolume {
56    /// Check the volume encloses some space, so sampling terminates.
57    fn validate(&self) -> Result<()> {
58        let ok = match *self {
59            CrownVolume::Sphere { radius, .. } | CrownVolume::Hemisphere { radius, .. } => {
60                radius > 0.0
61            }
62            CrownVolume::Ellipsoid { radii, .. } => radii.x > 0.0 && radii.y > 0.0 && radii.z > 0.0,
63            CrownVolume::Box { min, max } => min.x <= max.x && min.y <= max.y && min.z <= max.z,
64            CrownVolume::Cone { radius, height, .. } => radius > 0.0 && height > 0.0,
65        };
66        if ok {
67            Ok(())
68        } else {
69            Err(ProcGenError::InvalidParameter {
70                name: "crown".into(),
71                reason: "volume must have positive extent".into(),
72            })
73        }
74    }
75
76    /// Sample a random point inside this volume using rejection sampling.
77    fn sample(&self, rng: &mut SeededRng) -> Vec3 {
78        match *self {
79            CrownVolume::Sphere { center, radius } => loop {
80                let x = rng.next_range(-radius, radius);
81                let y = rng.next_range(-radius, radius);
82                let z = rng.next_range(-radius, radius);
83                if x * x + y * y + z * z <= radius * radius {
84                    return center + Vec3::new(x, y, z);
85                }
86            },
87            CrownVolume::Ellipsoid { center, radii } => loop {
88                let x = rng.next_range(-radii.x, radii.x);
89                let y = rng.next_range(-radii.y, radii.y);
90                let z = rng.next_range(-radii.z, radii.z);
91                let nx = x / radii.x;
92                let ny = y / radii.y;
93                let nz = z / radii.z;
94                if nx * nx + ny * ny + nz * nz <= 1.0 {
95                    return center + Vec3::new(x, y, z);
96                }
97            },
98            CrownVolume::Hemisphere { center, radius } => loop {
99                let x = rng.next_range(-radius, radius);
100                let y = rng.next_range(0.0, radius);
101                let z = rng.next_range(-radius, radius);
102                if x * x + y * y + z * z <= radius * radius {
103                    return center + Vec3::new(x, y, z);
104                }
105            },
106            CrownVolume::Box { min, max } => Vec3::new(
107                rng.next_range(min.x, max.x),
108                rng.next_range(min.y, max.y),
109                rng.next_range(min.z, max.z),
110            ),
111            CrownVolume::Cone {
112                base,
113                radius,
114                height,
115            } => loop {
116                let x = rng.next_range(-radius, radius);
117                let y = rng.next_range(0.0, height);
118                let z = rng.next_range(-radius, radius);
119                let r = radius * (1.0 - y / height);
120                if x * x + z * z <= r * r {
121                    return base + Vec3::new(x, y, z);
122                }
123            },
124        }
125    }
126}

128/// Scatter `count` attraction points inside `volume`, deterministically from `rng`.
129pub fn scatter_attraction_points(
130    volume: &CrownVolume,
131    count: u32,
132    rng: &mut SeededRng,
133) -> Vec<Vec3> {
134    (0..count).map(|_| volume.sample(rng)).collect()
135}

139/// Parameters controlling the space colonization algorithm.
140#[derive(Debug, Clone, PartialEq)]
141pub struct SpaceColonizationParams {
142    /// Shape of the attraction point cloud.
143    pub crown: CrownVolume,
144    /// Number of attraction points to scatter (default 500). Zero grows the
145    /// trunk alone.
146    pub num_attraction_points: u32,
147    /// Remove a point when a node is this close (default 0.5).
148    pub kill_distance: f32,
149    /// Maximum distance a point can attract a node (default 4.0).
150    pub influence_distance: f32,
151    /// New nodes are placed this far from their parent (default 0.3).
152    pub step_size: f32,
153    /// Root node position (default origin).
154    pub root_position: Vec3,
155    /// Number of straight-up trunk nodes before colonization (default 5).
156    pub trunk_steps: u32,
157    /// Safety cap on the main colonization loop (default 200).
158    pub max_iterations: u32,
159    /// Maximum number of tree nodes — budget guard (default 5000).
160    pub max_nodes: u32,
161    /// Root radius for the pipe model (default 0.3).
162    pub base_radius: f32,
163    /// Terminal node radius before scaling to `base_radius` (default 0.02).
164    pub tip_radius: f32,
165    /// Pipe model exponent `n` in `r_parent^n = sum(r_child^n)` (default 2.0,
166    /// da Vinci's rule). Larger values keep inner branches thicker.
167    pub pipe_exponent: f32,
168}

194/// One node of a grown branch graph. Node 0 is the root; every other node's
195/// `parent` has a lower index, so a forward pass visits parents first.
196#[derive(Debug, Clone, Copy, PartialEq)]
197pub struct BranchNode {
198    /// World-space position.
199    pub position: Vec3,
200    /// Pipe-model radius at this node.
201    pub radius: f32,
202    /// Index of the parent node; `None` only for the root.
203    pub parent: Option<usize>,
204    /// Branching depth: 0 on the trunk, +1 past each fork.
205    pub depth: u32,
206}

208/// The outcome of a colonization run.
209#[derive(Debug, Clone)]
210pub struct Colonization {
211    /// The grown tree graph (empty when nothing grew).
212    pub nodes: Vec<BranchNode>,
213    /// Growth iterations run.
214    pub iterations: u32,
215    /// Attraction points left unconsumed (out of reach, or the cap was hit).
216    pub unconsumed: usize,
217}

221/// Space colonization tree generator.
222///
223/// Grows organic branching structures by iteratively extending a tree graph
224/// toward a cloud of attraction points scattered within a crown volume.
225#[derive(Debug)]
226pub struct SpaceColonization {
227    params: SpaceColonizationParams,
228}
229
230impl SpaceColonization {
231    /// Create a new generator, validating parameters.
232    pub fn new(params: SpaceColonizationParams) -> Result<Self> {
233        if params.kill_distance <= 0.0 {
234            return Err(ProcGenError::InvalidParameter {
235                name: "kill_distance".into(),
236                reason: "must be > 0".into(),
237            });
238        }
239        if params.kill_distance >= params.influence_distance {
240            return Err(ProcGenError::InvalidParameter {
241                name: "kill_distance".into(),
242                reason: "must be less than influence_distance".into(),
243            });
244        }
245        if params.step_size <= 0.0 {
246            return Err(ProcGenError::InvalidParameter {
247                name: "step_size".into(),
248                reason: "must be > 0".into(),
249            });
250        }
251        if params.pipe_exponent < 1.0 {
252            return Err(ProcGenError::InvalidParameter {
253                name: "pipe_exponent".into(),
254                reason: "must be >= 1".into(),
255            });
256        }
257        params.crown.validate()?;
258        Ok(Self { params })
259    }
260
261    /// The validated parameters.
262    pub fn params(&self) -> &SpaceColonizationParams {
263        &self.params
264    }
265
266    /// Scatter attraction points in the crown and grow the graph toward them.
267    pub fn grow(&self, rng: &mut SeededRng) -> Result<Colonization> {
268        let p = &self.params;
269        let points = scatter_attraction_points(&p.crown, p.num_attraction_points, rng);
270        self.colonize(&points)
271    }
272
273    /// Grow the graph from `root_position` toward the given attraction points.
274    ///
275    /// Deterministic: the same points always give the same graph. Fails with
276    /// [`ProcGenError::BudgetExceeded`] past `max_nodes`.
277    pub fn colonize(&self, attractors: &[Vec3]) -> Result<Colonization> {
278        let p = &self.params;
279        let trunk: Vec<Vec3> = (0..=p.trunk_steps)
280            .map(|i| p.root_position + Vec3::new(0.0, p.step_size * i as f32, 0.0))
281            .collect();
282        self.colonize_from(&trunk, attractors)
283    }
284
285    /// Grow toward the attraction points from a given chain of nodes, which
286    /// become nodes `0..chain.len()`, each the parent of the next.
287    ///
288    /// Any node of the chain can sprout, so a tree's real trunk, passed as
289    /// the chain, grows branches all the way up rather than only from its
290    /// tip. `root_position` and `trunk_steps` are ignored.
291    pub fn colonize_from(&self, chain: &[Vec3], attractors: &[Vec3]) -> Result<Colonization> {
292        let p = &self.params;
293        let kill_sq = p.kill_distance * p.kill_distance;
294        let influence_sq = p.influence_distance * p.influence_distance;
295        if chain.is_empty() {
296            return Err(ProcGenError::InvalidParameter {
297                name: "chain".into(),
298                reason: "colonization needs at least a root node".into(),
299            });
300        }
301
302        let mut graph = Graph::default();
303        for (i, &pos) in chain.iter().enumerate() {
304            graph.push(pos, i.checked_sub(1));
305        }
306        graph.check_budget(p.max_nodes)?;
307
308        // Every live attraction point with its closest node so far.
309        let mut live: Vec<Attractor> = attractors
310            .iter()
311            .map(|&point| Attractor {
312                point,
313                closest: 0,
314                dist_sq: f32::INFINITY,
315            })
316            .collect();
317        let mut seen = 0; // nodes below this index are folded into `live`
318
319        let mut iterations = 0;
320        while iterations < p.max_iterations {
321            // Fold in the nodes added since last time, and kill what they reach.
322            let fresh = seen..graph.positions.len();
323            seen = fresh.end;
324            live.retain_mut(|a| {
325                for ni in fresh.clone() {
326                    let d = dist_sq(graph.positions[ni], a.point);
327                    if d < a.dist_sq {
328                        a.dist_sq = d;
329                        a.closest = ni;
330                    }
331                }
332                a.dist_sq >= kill_sq
333            });
334            if live.is_empty() {
335                break;
336            }
337
338            // Sum each node's pulls toward the points it is closest to, and
339            // remember the nearest of them.
340            let n = graph.positions.len();
341            let mut pull: Vec<Vec3> = vec![Vec3::ZERO; n];
342            let mut nearest: Vec<Option<&Attractor>> = vec![None; n];
343            for a in live.iter().filter(|a| a.dist_sq <= influence_sq) {
344                let dir = a.point - graph.positions[a.closest];
345                pull[a.closest] = pull[a.closest] + dir * (1.0 / a.dist_sq.sqrt());
346                if nearest[a.closest].is_none_or(|b| a.dist_sq < b.dist_sq) {
347                    nearest[a.closest] = Some(a);
348                }
349            }
350
351            let before = graph.positions.len();
352            for (ni, dir) in pull.into_iter().enumerate() {
353                let Some(near) = nearest[ni] else { continue };
354                let tolerance = p.step_size * 0.05;
355                let from = graph.positions[ni];
356                let len = dir.length();
357                let mut pos = from + dir * (p.step_size / len.max(1e-6));
358                if len < 1e-6 || graph.has_child_near(ni, pos, tolerance) {
359                    // Pulls that cancel out, or that already grew a child no
360                    // closer to them, would stall; step toward the nearest
361                    // point instead, which always gets closer to it.
362                    let toward = near.point - from;
363                    pos = from + toward * (p.step_size.min(near.dist_sq.sqrt()) / toward.length());
364                    if graph.has_child_near(ni, pos, tolerance) {
365                        continue;
366                    }
367                }
368                graph.push(pos, Some(ni));
369            }
370            graph.check_budget(p.max_nodes)?;
371            iterations += 1;
372            if graph.positions.len() == before {
373                break; // stalled: the rest is out of reach
374            }
375        }
376
377        // A lone root is not a branch.
378        if graph.positions.len() < 2 {
379            return Ok(Colonization {
380                nodes: Vec::new(),
381                iterations,
382                unconsumed: live.len(),
383            });
384        }
385
386        let depths = graph.depths();
387        let radii = pipe_radii(&graph.parents, p.tip_radius, p.pipe_exponent, p.base_radius);
388        let nodes = (0..graph.positions.len())
389            .map(|i| BranchNode {
390                position: graph.positions[i],
391                radius: radii[i],
392                parent: graph.parents[i],
393                depth: depths[i],
394            })
395            .collect();
396        Ok(Colonization {
397            nodes,
398            iterations,
399            unconsumed: live.len(),
400        })
401    }

403    /// Run the space colonization algorithm and produce branch segments.
404    pub fn generate_segments(&self, rng: &mut SeededRng) -> Result<Vec<BranchSegment>> {
405        Ok(branch_graph_to_segments(&self.grow(rng)?.nodes))
406    }
407}

409/// An attraction point and the closest node it has seen.
410struct Attractor {
411    point: Vec3,
412    closest: usize,
413    dist_sq: f32,
414}

416/// The growing graph, struct-of-arrays, with child lists for fork detection.
417#[derive(Default)]
418struct Graph {
419    positions: Vec<Vec3>,
420    parents: Vec<Option<usize>>,
421    children: Vec<Vec<usize>>,
422}
423
424impl Graph {
425    fn push(&mut self, position: Vec3, parent: Option<usize>) {
426        let idx = self.positions.len();
427        if let Some(p) = parent {
428            self.children[p].push(idx);
429        }
430        self.positions.push(position);
431        self.parents.push(parent);
432        self.children.push(Vec::new());
433    }
434
435    fn has_child_near(&self, node: usize, pos: Vec3, tolerance: f32) -> bool {
436        self.children[node]
437            .iter()
438            .any(|&c| dist_sq(self.positions[c], pos) < tolerance * tolerance)
439    }
440
441    fn check_budget(&self, max_nodes: u32) -> Result<()> {
442        if self.positions.len() > max_nodes as usize {
443            return Err(ProcGenError::BudgetExceeded(format!(
444                "node count {} exceeds max_nodes {}",
445                self.positions.len(),
446                max_nodes
447            )));
448        }
449        Ok(())
450    }
451
452    /// Branching depth per node: +1 below each fork (2+ children).
453    fn depths(&self) -> Vec<u32> {
454        let mut depths = vec![0u32; self.positions.len()];
455        for i in 1..self.positions.len() {
456            let parent = self.parents[i].expect("non-root node has a parent");
457            let fork = self.children[parent].len() >= 2;
458            depths[i] = depths[parent] + u32::from(fork);
459        }
460        depths
461    }
462}

464// ─── E. Pipe Model Radii ────────────────────────────────────────────────────
465
466/// Compute radii using the pipe model.
467///
468/// Leaves get `tip_radius`. Working bottom-up, each parent's radius is
469/// `(sum(r_child^n))^(1/n)`. Finally, all radii are scaled so the root
470/// gets exactly `base_radius`. `parents` must list parents before children.
471fn pipe_radii(parents: &[Option<usize>], tip_radius: f32, n: f32, base_radius: f32) -> Vec<f32> {
472    let mut sum = vec![0.0f32; parents.len()];
473    let mut radii = vec![0.0f32; parents.len()];
474    for i in (0..parents.len()).rev() {
475        radii[i] = if sum[i] > 0.0 {
476            sum[i].powf(1.0 / n)
477        } else {
478            tip_radius
479        };
480        if let Some(p) = parents[i] {
481            sum[p] += radii[i].powf(n);
482        }
483    }
484    if let Some(&root) = radii.first() {
485        if root > 1e-10 {
486            let scale = base_radius / root;
487            radii.iter_mut().for_each(|r| *r *= scale);
488        }
489    }
490    radii
491}

493// ─── F. Graph to Segments ───────────────────────────────────────────────────
494
495/// Convert a branch graph into one [`BranchSegment`] per parent-child edge,
496/// the format the L-system produces, so the tree mesher takes either.
497pub fn branch_graph_to_segments(nodes: &[BranchNode]) -> Vec<BranchSegment> {
498    nodes
499        .iter()
500        .filter_map(|node| {
501            let parent = &nodes[node.parent?];
502            Some(BranchSegment {
503                start: parent.position,
504                end: node.position,
505                radius_start: parent.radius,
506                radius_end: node.radius,
507                depth: node.depth,
508            })
509        })
510        .collect()
511}

515/// Squared distance between two points.
516fn dist_sq(a: Vec3, b: Vec3) -> f32 {
517    let d = a - b;
518    d.dot(&d)
519}

523/// Create a deciduous tree with a spherical crown.
524///
525/// The crown sphere sits atop a straight trunk of the given height.
526pub fn deciduous_tree(
527    trunk_height: f32,
528    crown_radius: f32,
529    base_radius: f32,
530) -> Result<SpaceColonization> {
531    let step_size = 0.3;
532    let trunk_steps = (trunk_height / step_size).round().max(1.0) as u32;
533    SpaceColonization::new(SpaceColonizationParams {
534        crown: CrownVolume::Sphere {
535            center: Vec3::new(0.0, trunk_height + crown_radius, 0.0),
536            radius: crown_radius,
537        },
538        trunk_steps,
539        step_size,
540        base_radius,
541        ..Default::default()
542    })
543}

545/// Create a columnar tree with a tall, narrow ellipsoid crown.
546///
547/// Good for cypresses, poplars, and other upright forms.
548pub fn columnar_tree(
549    trunk_height: f32,
550    crown_height: f32,
551    crown_width: f32,
552    base_radius: f32,
553) -> Result<SpaceColonization> {
554    let step_size = 0.3;
555    let trunk_steps = (trunk_height / step_size).round().max(1.0) as u32;
556    let half_height = crown_height * 0.5;
557    SpaceColonization::new(SpaceColonizationParams {
558        crown: CrownVolume::Ellipsoid {
559            center: Vec3::new(0.0, trunk_height + half_height, 0.0),
560            radii: Vec3::new(crown_width * 0.5, half_height, crown_width * 0.5),
561        },
562        trunk_steps,
563        step_size,
564        base_radius,
565        ..Default::default()
566    })
567}
