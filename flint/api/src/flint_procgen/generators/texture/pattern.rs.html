1//! Pattern field types and trait for texture pattern generators.
2//!
3//! A [`PatternField`] is the intermediate representation that all pattern
4//! generators produce — one [`PatternCell`] per pixel. Downstream stages
5//! (Task 4.2) consume these fields to derive albedo, normal, and roughness
6//! textures.

14/// A pattern generator that fills a 2D field with cell IDs, heights, and
15/// edge distances. Implementations must be deterministic for a given seed.
16pub trait Pattern: Send + Sync {
17    /// Generate a pattern field of the given dimensions.
18    ///
19    /// When `seamless` is true, the pattern tiles without visible seams.
20    fn generate(
21        &self,
22        width: u32,
23        height: u32,
24        seamless: bool,
25        rng: &mut SeededRng,
26    ) -> PatternField;
27}

33/// A 2D grid of [`PatternCell`] values produced by a pattern generator.
34///
35/// Width × height cells, stored in row-major order.
36#[derive(Debug, Clone)]
37pub struct PatternField {
38    /// Width in pixels.
39    pub width: u32,
40    /// Height in pixels.
41    pub height: u32,
42    /// Cell data, length = `width * height`, stored row-major.
43    pub cells: Vec<PatternCell>,
44}
45
46impl PatternField {
47    /// Create a new field with all cells zeroed.
48    pub fn new(width: u32, height: u32) -> Self {
49        Self {
50            width,
51            height,
52            cells: vec![PatternCell::default(); (width * height) as usize],
53        }
54    }
55
56    /// Get a reference to the cell at `(x, y)`.
57    ///
58    /// # Panics
59    /// Panics if `x >= width` or `y >= height`.
60    pub fn get(&self, x: u32, y: u32) -> &PatternCell {
61        debug_assert!(x < self.width && y < self.height);
62        &self.cells[(y * self.width + x) as usize]
63    }
64
65    /// Get a mutable reference to the cell at `(x, y)`.
66    ///
67    /// # Panics
68    /// Panics if `x >= width` or `y >= height`.
69    pub fn get_mut(&mut self, x: u32, y: u32) -> &mut PatternCell {
70        debug_assert!(x < self.width && y < self.height);
71        let idx = (y * self.width + x) as usize;
72        &mut self.cells[idx]
73    }
74
75    /// Sample the field at normalised coordinates, treating it as a tile.
76    ///
77    /// `(u, v)` wrap into `[0, 1)`, and pixel centres sit at `(i + 0.5) / n`,
78    /// so `get_normalized(0.0, v)` is halfway between the last column and the
79    /// first: the same value whichever way round the tile you approach it.
80    /// `height` and `distance_to_edge` are bilinear; `cell_id` (a label, not a
81    /// quantity) comes from the nearest pixel.
82    pub fn get_normalized(&self, u: f32, v: f32) -> PatternCell {
83        let (w, h) = (self.width as i64, self.height as i64);
84        let fx = u.rem_euclid(1.0) * w as f32 - 0.5;
85        let fy = v.rem_euclid(1.0) * h as f32 - 0.5;
86        let (x0, y0) = (fx.floor() as i64, fy.floor() as i64);
87        let (tx, ty) = (fx - x0 as f32, fy - y0 as f32);
88        let at = |x: i64, y: i64| self.get(x.rem_euclid(w) as u32, y.rem_euclid(h) as u32);
89        let (c00, c10, c01, c11) = (
90            at(x0, y0),
91            at(x0 + 1, y0),
92            at(x0, y0 + 1),
93            at(x0 + 1, y0 + 1),
94        );
95        let bi = |f: fn(&PatternCell) -> f32| {
96            let top = f(c00) + (f(c10) - f(c00)) * tx;
97            let bottom = f(c01) + (f(c11) - f(c01)) * tx;
98            top + (bottom - top) * ty
99        };
100        let nearest = at((fx + 0.5).floor() as i64, (fy + 0.5).floor() as i64);
101        PatternCell {
102            cell_id: nearest.cell_id,
103            height: bi(|c| c.height),
104            distance_to_edge: bi(|c| c.distance_to_edge),
105        }
106    }
107
108    /// How visible the tile's seams are, measured on `height`.
109    ///
110    /// The last column and the first are neighbours once the tile repeats, so
111    /// in a seamless field the step across the wrap looks like a step between
112    /// any other pair of neighbouring columns; they are not equal values.
113    /// For every pair of adjacent columns (and rows) this takes the mean
114    /// absolute height step down the pair. The seam's pair is compared with
115    /// the roughest interior pair, not the average one, because a pattern
116    /// may put an edge on the seam on purpose (a grid's grooves sit on its
117    /// tile boundaries, and so on the wrap).
118    pub fn seam_steps(&self) -> SeamSteps {
119        let (w, h) = (self.width, self.height);
120        let hgt = |x: u32, y: u32| self.get(x, y).height;
121        let col_pair =
122            |a: u32, b: u32| (0..h).map(|y| (hgt(a, y) - hgt(b, y)).abs()).sum::<f32>() / h as f32;
123        let row_pair =
124            |a: u32, b: u32| (0..w).map(|x| (hgt(x, a) - hgt(x, b)).abs()).sum::<f32>() / w as f32;
125        let cols: Vec<f32> = (1..w).map(|x| col_pair(x - 1, x)).collect();
126        let rows: Vec<f32> = (1..h).map(|y| row_pair(y - 1, y)).collect();
127        let max = |v: &[f32]| v.iter().copied().fold(0.0f32, f32::max);
128        SeamSteps {
129            across_x: col_pair(w - 1, 0),
130            across_y: row_pair(h - 1, 0),
131            interior_x: max(&cols),
132            interior_y: max(&rows),
133            interior: cols.iter().sum::<f32>() / cols.len().max(1) as f32,
134        }
135    }
136
137    /// Convert the height channel to a grayscale RGBA8 [`ImageData`] for
138    /// debugging and visual inspection.
139    pub fn to_height_image(&self) -> ImageData {
140        let mut img = ImageData::new(self.width, self.height, ChannelSemantics::Height);
141        for (i, cell) in self.cells.iter().enumerate() {
142            let byte = (cell.height.clamp(0.0, 1.0) * 255.0) as u8;
143            let offset = i * 4;
144            img.pixels[offset] = byte;
145            img.pixels[offset + 1] = byte;
146            img.pixels[offset + 2] = byte;
147            img.pixels[offset + 3] = 255;
148        }
149        img
150    }
151
152    /// Convert the cell_id channel to a color-coded RGBA8 [`ImageData`] for
153    /// debugging. Each cell gets a deterministic pseudo-random color.
154    pub fn to_cell_id_image(&self) -> ImageData {
155        let mut img = ImageData::new(self.width, self.height, ChannelSemantics::Mask);
156        for (i, cell) in self.cells.iter().enumerate() {
157            let (r, g, b) = id_to_color(cell.cell_id);
158            let offset = i * 4;
159            img.pixels[offset] = r;
160            img.pixels[offset + 1] = g;
161            img.pixels[offset + 2] = b;
162            img.pixels[offset + 3] = 255;
163        }
164        img
165    }
166
167    /// Convert the distance_to_edge channel to a grayscale RGBA8 [`ImageData`].
168    pub fn to_edge_distance_image(&self) -> ImageData {
169        let mut img = ImageData::new(self.width, self.height, ChannelSemantics::Mask);
170        for (i, cell) in self.cells.iter().enumerate() {
171            let byte = (cell.distance_to_edge.clamp(0.0, 1.0) * 255.0) as u8;
172            let offset = i * 4;
173            img.pixels[offset] = byte;
174            img.pixels[offset + 1] = byte;
175            img.pixels[offset + 2] = byte;
176            img.pixels[offset + 3] = 255;
177        }
178        img
179    }
180}

182/// Mean absolute height steps across a field's seams and inside it; see
183/// [`PatternField::seam_steps`].
184#[derive(Debug, Clone, Copy, PartialEq)]
185pub struct SeamSteps {
186    /// Across the left/right seam (last column to first).
187    pub across_x: f32,
188    /// Across the top/bottom seam (last row to first).
189    pub across_y: f32,
190    /// The roughest pair of adjacent interior columns.
191    pub interior_x: f32,
192    /// The roughest pair of adjacent interior rows.
193    pub interior_y: f32,
194    /// Mean over all pairs of adjacent interior columns.
195    pub interior: f32,
196}
197
198impl SeamSteps {
199    /// The worse seam as a multiple of the roughest interior pair on its
200    /// axis: at most about 1 for a seamless field, well above it for a seam.
201    pub fn ratio(&self) -> f32 {
202        let x = self.across_x / self.interior_x.max(1e-6);
203        let y = self.across_y / self.interior_y.max(1e-6);
204        x.max(y)
205    }
206}

212/// Per-pixel data produced by a pattern generator.
213#[derive(Debug, Clone, Copy, PartialEq)]
214pub struct PatternCell {
215    /// Which cell/region this pixel belongs to.
216    pub cell_id: u32,
217    /// Height value in \[0, 1\] for normal map derivation.
218    pub height: f32,
219    /// Distance to nearest cell edge in \[0, 1\], where 0 = on the edge.
220    pub distance_to_edge: f32,
221}
222
223impl Default for PatternCell {
224    fn default() -> Self {
225        Self {
226            cell_id: 0,
227            height: 0.0,
228            distance_to_edge: 0.0,
229        }
230    }
231}

237/// Hash a cell ID into a deterministic RGB color for debug visualization.
238fn id_to_color(id: u32) -> (u8, u8, u8) {
239    // Simple hash based on mixing bits — visually distinct for low cell counts.
240    let mut h = id.wrapping_mul(2654435761); // Knuth multiplicative hash
241    let r = (h & 0xFF) as u8;
242    h = h.wrapping_mul(2246822519);
243    let g = ((h >> 8) & 0xFF) as u8;
244    h = h.wrapping_mul(3266489917);
245    let b = ((h >> 16) & 0xFF) as u8;
246    // Ensure minimum brightness so cells are visible.
247    (r.max(40), g.max(40), b.max(40))
248}

254/// Where pattern tests write their debug PNGs: `<target>/test-output/`,
255/// honouring `CARGO_TARGET_DIR`.
256#[cfg(test)]
257pub(crate) fn debug_output_dir() -> std::path::PathBuf {
258    let target = std::env::var_os("CARGO_TARGET_DIR")
259        .map(std::path::PathBuf::from)
260        .unwrap_or_else(|| std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../../target"));
261    let dir = target.join("test-output");
262    std::fs::create_dir_all(&dir).unwrap();
263    dir
264}
265
266/// Write an [`ImageData`] as a PNG into [`debug_output_dir`].
267#[cfg(test)]
268pub(crate) fn write_debug_png(name: &str, img: &ImageData) {
269    let path = debug_output_dir().join(format!("{name}.png"));
270    image::save_buffer(
271        &path,
272        &img.pixels,
273        img.width,
274        img.height,
275        image::ColorType::Rgba8,
276    )
277    .unwrap();
278    println!("wrote {}", path.display());
279}

281/// Shared acceptance checks for any pattern's output: dimensions, no NaN,
282/// height and edge distance in \[0, 1\].
283#[cfg(test)]
284pub(crate) fn assert_valid(field: &PatternField, width: u32, height: u32) {
285    assert_eq!((field.width, field.height), (width, height));
286    assert_eq!(field.cells.len(), (width * height) as usize);
287    for c in &field.cells {
288        assert!(
289            c.height.is_finite() && (0.0..=1.0).contains(&c.height),
290            "height {}",
291            c.height
292        );
293        assert!(
294            c.distance_to_edge.is_finite() && (0.0..=1.0).contains(&c.distance_to_edge),
295            "distance_to_edge {}",
296            c.distance_to_edge
297        );
298    }
299}
