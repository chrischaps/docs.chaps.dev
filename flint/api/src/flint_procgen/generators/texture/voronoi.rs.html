1//! Voronoi-based brick/stone pattern generator.
2//!
3//! Produces irregular stone and cobblestone layouts via Voronoi tessellation
4//! with optional Lloyd's relaxation for regularity control. Each cell gets a
5//! unique ID, height variation, and edge distance suitable for mortar grooves.
6//!
7//! This is a custom implementation (not `noise::Worley`) because we need cell
8//! IDs and precise edge distances, not just distance values.
9

21/// Parameters for the Voronoi brick/stone pattern.
22#[derive(Debug, Clone)]
23pub struct VoronoiBrickParams {
24    /// Number of Voronoi cells (seed points). ~12 for a stone wall.
25    pub cell_count: u32,
26    /// Width of the mortar groove as a fraction of the texture. 0.02 typical.
27    pub mortar_width: f32,
28    /// Per-cell height randomness in \[0, 1\]. 0 = flat, 1 = max variation.
29    pub height_variation: f32,
30    /// Regularity: 0 = random Voronoi, 1 = fully grid-relaxed (Lloyd's).
31    /// Intermediate values blend between random and relaxed positions.
32    pub regularity: f32,
33}
34
35impl Default for VoronoiBrickParams {
36    fn default() -> Self {
37        Self {
38            cell_count: 12,
39            mortar_width: 0.02,
40            height_variation: 0.3,
41            regularity: 0.0,
42        }
43    }
44}
45
46impl VoronoiBrickParams {
47    /// Parse voronoi brick parameters from a TOML `params` value.
48    ///
49    /// Missing fields fall back to defaults. Invalid values produce errors.
50    pub fn from_toml(params: &toml::Value) -> Result<Self> {
51        let table = params
52            .as_table()
53            .ok_or_else(|| ProcGenError::InvalidParameter {
54                name: "params".into(),
55                reason: "expected a TOML table".into(),
56            })?;
57
58        let mut p = Self::default();
59
60        if let Some(v) = table.get("cell_count") {
61            p.cell_count = toml_u32(v, "cell_count")?;
62            if p.cell_count == 0 {
63                return Err(ProcGenError::InvalidParameter {
64                    name: "cell_count".into(),
65                    reason: "must be positive".into(),
66                });
67            }
68        }
69        if let Some(v) = table.get("mortar_width") {
70            p.mortar_width = toml_f32(v, "mortar_width")?;
71            if p.mortar_width < 0.0 || p.mortar_width > 1.0 {
72                return Err(ProcGenError::InvalidParameter {
73                    name: "mortar_width".into(),
74                    reason: "must be in [0.0, 1.0]".into(),
75                });
76            }
77        }
78        if let Some(v) = table.get("height_variation") {
79            p.height_variation = toml_f32(v, "height_variation")?;
80            if p.height_variation < 0.0 || p.height_variation > 1.0 {
81                return Err(ProcGenError::InvalidParameter {
82                    name: "height_variation".into(),
83                    reason: "must be in [0.0, 1.0]".into(),
84                });
85            }
86        }
87        if let Some(v) = table.get("regularity") {
88            p.regularity = toml_f32(v, "regularity")?;
89            if p.regularity < 0.0 || p.regularity > 1.0 {
90                return Err(ProcGenError::InvalidParameter {
91                    name: "regularity".into(),
92                    reason: "must be in [0.0, 1.0]".into(),
93                });
94            }
95        }
96
97        Ok(p)
98    }
99}

105/// Voronoi-based stone/brick pattern generator.
106///
107/// Scatters seed points and computes per-pixel Voronoi cell membership, edge
108/// distance, and height. Supports seamless tiling via toroidal distance.
109pub struct VoronoiBrickPattern {
110    pub params: VoronoiBrickParams,
111}
112
113impl VoronoiBrickPattern {
114    pub fn new(params: VoronoiBrickParams) -> Self {
115        Self { params }
116    }
117}
118
119impl Pattern for VoronoiBrickPattern {
120    fn generate(
121        &self,
122        width: u32,
123        height: u32,
124        seamless: bool,
125        rng: &mut SeededRng,
126    ) -> PatternField {
127        let p = &self.params;
128        let cell_count = p.cell_count.max(1) as usize;
129
130        // --- Step 1: Generate seed points in [0, 1)² ---
131        let mut points = voronoi_util::generate_seed_points(rng, cell_count);
132
133        // --- Step 2: Apply Lloyd's relaxation for regularity ---
134        if p.regularity > 0.0 {
135            let iterations = (p.regularity * 10.0).ceil() as usize;
136            let relaxed = voronoi_util::lloyds_relaxation(&points, iterations, seamless);
137            // Blend between random and relaxed based on regularity
138            for i in 0..points.len() {
139                points[i].0 = voronoi_util::lerp(points[i].0, relaxed[i].0, p.regularity);
140                points[i].1 = voronoi_util::lerp(points[i].1, relaxed[i].1, p.regularity);
141            }
142        }
143
144        // --- Step 3: Build spatial acceleration grid ---
145        let grid = SpatialGrid::new(&points, seamless);
146
147        // --- Step 4: Per-cell heights via hash + noise ---
148        let mut height_rng = rng.fork("cell_heights");
149        let cell_base_heights: Vec<f32> = (0..cell_count)
150            .map(|_| {
151                let base = 0.5 + (height_rng.next_f32() - 0.5) * p.height_variation;
152                base.clamp(0.0, 1.0)
153            })
154            .collect();
155
156        // Surface variation noise (subtle per-cell bumps)
157        let noise_rng = rng.fork("surface_noise");
158        let surface_noise = Fbm::new(PerlinNoise::new(noise_rng.seed()))
159            .with_octaves(4)
160            .with_frequency(8.0)
161            .with_persistence(0.4);
162
163        // --- Step 5: Fill the pattern field ---
164        let mut field = PatternField::new(width, height);
165
166        for y in 0..height {
167            for x in 0..width {
168                let u = (x as f32 + 0.5) / width as f32;
169                let v = (y as f32 + 0.5) / height as f32;
170
171                // Find closest and second-closest seed points
172                let (closest_id, dist1, dist2) = grid.find_two_nearest(u, v, &points, seamless);
173
174                // Edge distance: difference between closest and second-closest
175                let raw_edge_dist = dist2 - dist1;
176                // Normalize: mortar_width defines the 0→1 transition zone
177                let mortar_w = p.mortar_width.max(1e-6);
178                let edge_normalized = (raw_edge_dist / mortar_w).min(1.0);
179
180                // Height: base cell height + subtle surface noise
181                let base_h = cell_base_heights[closest_id];
182                let noise_val = surface_noise.sample_2d(u as f64 * 16.0, v as f64 * 16.0) as f32;
183                let surface_h = base_h + noise_val * 0.05;
184
185                // In mortar regions, lower the height
186                let final_height = if edge_normalized < 1.0 {
187                    // Smooth blend into mortar groove
188                    let mortar_depth = 0.15;
189                    let t = edge_normalized; // 0 = on edge, 1 = fully inside cell
190                    let mortar_h = (surface_h - mortar_depth).max(0.0);
191                    // Smooth step for nice transition
192                    let s = t * t * (3.0 - 2.0 * t);
193                    voronoi_util::lerp(mortar_h, surface_h, s)
194                } else {
195                    surface_h
196                };
197
198                let cell = field.get_mut(x, y);
199                cell.cell_id = closest_id as u32;
200                cell.height = final_height.clamp(0.0, 1.0);
201                cell.distance_to_edge = edge_normalized;
202            }
203        }
204
205        field
206    }
207}

346    #[test]
347    fn voronoi_seamless_continuity() {
348        let pattern = default_pattern();
349        let mut rng = SeededRng::new(42);
350        let field = pattern.generate(64, 64, true, &mut rng);
351
352        // Compare top row vs bottom row
353        let tolerance = 0.15; // Allow some tolerance due to pixel discretization
354        let mut top_bottom_matches = 0;
355        for x in 0..field.width {
356            let top = field.get(x, 0);
357            let bottom = field.get(x, field.height - 1);
358            if top.cell_id == bottom.cell_id {
359                top_bottom_matches += 1;
360            }
361            // Height should be close at boundaries
362            assert!(
363                (top.height - bottom.height).abs() < tolerance || top.cell_id != bottom.cell_id,
364                "seamless height mismatch at x={}: top={}, bottom={}",
365                x,
366                top.height,
367                bottom.height
368            );
369        }
370
371        // Compare left column vs right column
372        let mut left_right_matches = 0;
373        for y in 0..field.height {
374            let left = field.get(0, y);
375            let right = field.get(field.width - 1, y);
376            if left.cell_id == right.cell_id {
377                left_right_matches += 1;
378            }
379        }
380
381        // At least some boundary pixels should share cell IDs for seamless tiling
382        assert!(
383            top_bottom_matches > 0 || left_right_matches > 0,
384            "no seamless continuity found at boundaries"
385        );
386    }

388    #[test]
389    fn voronoi_regularity() {
390        let regular = VoronoiBrickPattern::new(VoronoiBrickParams {
391            cell_count: 9,
392            regularity: 1.0,
393            ..Default::default()
394        });
395        let mut rng = SeededRng::new(42);
396        let field = regular.generate(64, 64, false, &mut rng);
397
398        // With high regularity, cells should be roughly equal-sized.
399        let mut cell_sizes: std::collections::HashMap<u32, u32> = Default::default();
400        for cell in &field.cells {
401            *cell_sizes.entry(cell.cell_id).or_default() += 1;
402        }
403        let sizes: Vec<u32> = cell_sizes.values().copied().collect();
404        let mean = sizes.iter().sum::<u32>() as f32 / sizes.len() as f32;
405        let max_deviation = sizes
406            .iter()
407            .map(|&s| (s as f32 - mean).abs() / mean)
408            .fold(0.0f32, f32::max);
409
410        // With Lloyd's relaxation, max deviation from mean should be moderate
411        assert!(
412            max_deviation < 1.0,
413            "high regularity should produce more uniform cells, max deviation: {}",
414            max_deviation
415        );
416    }
