1//! PBR texture map derivation from pattern fields.
2//!
3//! Bridges the gap between the abstract [`PatternField`] (per-pixel cell ID,
4//! height, edge distance) and usable PBR textures — **albedo**, **normal**, and
5//! **roughness** maps. This is the core "material from math" step that turns
6//! procedural patterns into something beautiful.
7//!
8//! # Pipeline
9//!
10//! ```text
11//! PatternField ──► derive_albedo()     ──► ImageData (Color, sRGB RGBA8)
12//!              ├─► derive_normal()     ──► ImageData (Normal, linear RGBA8)
13//!              └─► derive_roughness()  ──► ImageData (Roughness, linear RGBA8)
14//! ```
15

26/// Parameters controlling PBR map derivation from a pattern field.
27#[derive(Debug, Clone)]
28pub struct TextureMapParams {
29    /// Texture width in pixels.
30    pub width: u32,
31    /// Texture height in pixels.
32    pub height: u32,
33    /// Which maps to generate: `"albedo"`, `"normal"`, `"roughness"`.
34    pub output_maps: Vec<String>,
35    /// Pattern type: `"voronoi_brick"`, `"perlin_organic"`, `"tiling_grid"`.
36    pub pattern: String,
37    /// Base albedo color as linear RGBA.
38    pub base_color: [f32; 4],
39    /// Per-cell color shift magnitude.
40    pub color_variation: f32,
41    /// Color for mortar/gap regions (linear RGBA).
42    pub mortar_color: [f32; 4],
43    /// Mortar region threshold — pixels with `distance_to_edge` below this
44    /// value blend toward mortar color.
45    pub mortar_threshold: f32,
46    /// Base surface roughness (0–1).
47    pub roughness_base: f32,
48    /// Per-cell roughness shift magnitude.
49    pub roughness_variation: f32,
50    /// Roughness in mortar regions (defaults smoother than surface).
51    pub roughness_mortar: f32,
52    /// Normal map strength multiplier.
53    pub normal_strength: f32,
54    /// FBM frequency for micro-detail overlay.
55    pub detail_scale: f32,
56    /// FBM amplitude (0 = no detail noise).
57    pub detail_strength: f32,
58    /// Whether the texture should tile seamlessly.
59    pub seamless: bool,
60}

68            pattern: "voronoi_brick".into(),

80        }
81    }

85    /// Parse map derivation parameters from a TOML `params` value.
86    ///
87    /// Missing fields fall back to sensible defaults. Invalid values produce
88    /// descriptive errors.
89    pub fn from_toml(params: &toml::Value) -> Result<Self> {
90        let table = params
91            .as_table()
92            .ok_or_else(|| ProcGenError::InvalidParameter {
93                name: "params".into(),
94                reason: "expected a TOML table".into(),
95            })?;
96
97        let mut p = Self::default();
98
99        if let Some(v) = table.get("width") {
100            p.width = toml_u32(v, "width")?;
101            if p.width == 0 {
102                return Err(ProcGenError::InvalidParameter {
103                    name: "width".into(),
104                    reason: "must be positive".into(),
105                });
106            }
107        }
108        if let Some(v) = table.get("height") {
109            p.height = toml_u32(v, "height")?;
110            if p.height == 0 {
111                return Err(ProcGenError::InvalidParameter {
112                    name: "height".into(),
113                    reason: "must be positive".into(),
114                });
115            }
116        }
117        if let Some(v) = table.get("output_maps") {
118            p.output_maps = toml_string_array(v, "output_maps")?;
119            for map_name in &p.output_maps {
120                if !["albedo", "normal", "roughness"].contains(&map_name.as_str()) {
121                    return Err(ProcGenError::InvalidParameter {
122                        name: "output_maps".into(),
123                        reason: format!(
124                            "unknown map type \"{map_name}\"; expected \"albedo\", \"normal\", or \"roughness\""
125                        ),
126                    });
127                }
128            }
129        }
130        if let Some(v) = table.get("pattern") {
131            p.pattern = toml_string(v, "pattern")?;
132        }
133        if let Some(v) = table.get("base_color") {
134            let hex = v.as_str().ok_or_else(|| ProcGenError::InvalidParameter {
135                name: "base_color".into(),
136                reason: "expected a hex color string".into(),
137            })?;
138            p.base_color = parse_hex_color(hex)?;
139        }
140        if let Some(v) = table.get("color_variation") {
141            p.color_variation = toml_f32(v, "color_variation")?;
142        }
143        if let Some(v) = table.get("mortar_color") {
144            let hex = v.as_str().ok_or_else(|| ProcGenError::InvalidParameter {
145                name: "mortar_color".into(),
146                reason: "expected a hex color string".into(),
147            })?;
148            p.mortar_color = parse_hex_color(hex)?;
149        }
150        if let Some(v) = table.get("mortar_threshold") {
151            p.mortar_threshold = toml_f32(v, "mortar_threshold")?;
152        }
153        if let Some(v) = table.get("roughness_base") {
154            p.roughness_base = toml_f32(v, "roughness_base")?;
155        }
156        if let Some(v) = table.get("roughness_variation") {
157            p.roughness_variation = toml_f32(v, "roughness_variation")?;
158        }
159        if let Some(v) = table.get("roughness_mortar") {
160            p.roughness_mortar = toml_f32(v, "roughness_mortar")?;
161        }
162        if let Some(v) = table.get("normal_strength") {
163            p.normal_strength = toml_f32(v, "normal_strength")?;
164        }
165        if let Some(v) = table.get("detail_scale") {
166            p.detail_scale = toml_f32(v, "detail_scale")?;
167        }
168        if let Some(v) = table.get("detail_strength") {
169            p.detail_strength = toml_f32(v, "detail_strength")?;
170        }
171        if let Some(v) = table.get("seamless") {
172            p.seamless = v.as_bool().ok_or_else(|| ProcGenError::InvalidParameter {
173                name: "seamless".into(),
174                reason: "expected a boolean".into(),
175            })?;
176        }
177
178        Ok(p)
179    }

184/// Deterministic per-cell color from cell ID, using a hash-based approach
185/// to avoid dependence on pixel traversal order.
186fn cell_color_for_id(
187    cell_id: u32,
188    base_color: [f32; 4],
189    variation: f32,
190    rng_seed: u64,
191) -> [f32; 4] {
192    if variation <= 0.0 {
193        return base_color;
194    }
195    // Combine the master seed with the cell ID for a deterministic per-cell seed.
196    let cell_seed = rng_seed
197        .wrapping_mul(0x9E37_79B9_7F4A_7C15)
198        .wrapping_add(cell_id as u64);
199    let mut cell_rng = SeededRng::new(cell_seed);
200    cell_rng.next_color_variation(base_color, variation)
201}
202
203/// Deterministic per-cell roughness shift from cell ID.
204fn cell_roughness_shift(cell_id: u32, variation: f32, rng_seed: u64) -> f32 {
205    if variation <= 0.0 {
206        return 0.0;
207    }
208    let cell_seed = rng_seed
209        .wrapping_mul(0x517C_C1B7_2722_0A95)
210        .wrapping_add(cell_id as u64);
211    let mut cell_rng = SeededRng::new(cell_seed);
212    cell_rng.next_range(-variation, variation)
213}

215// ─── sRGB Conversion ──────────────────────────────────────────────────────

217/// Gamma-correct a linear float to an sRGB byte.
218fn linear_to_srgb(v: f32) -> u8 {
219    let s = if v <= 0.003_130_8 {
220        v * 12.92
221    } else {
222        1.055 * v.powf(1.0 / 2.4) - 0.055
223    };
224    (s.clamp(0.0, 1.0) * 255.0) as u8
225}

227// ─── Heightfield Sampling ─────────────────────────────────────────────────

229/// Sample the combined height at (x, y) with toroidal wrapping for seamless
230/// tiling. Combines the pattern field height with an optional FBM detail layer.
231fn combined_height(
232    field: &PatternField,
233    x: i32,
234    y: i32,
235    detail_noise: Option<&dyn NoiseSource>,
236    detail_strength: f32,
237    detail_scale: f32,
238) -> f32 {
239    let w = field.width as i32;
240    let h = field.height as i32;
241    let wx = ((x % w) + w) % w;
242    let wy = ((y % h) + h) % h;
243
244    let base = field.get(wx as u32, wy as u32).height;
245
246    if let Some(noise) = detail_noise {
247        let nx = wx as f64 / w as f64 * detail_scale as f64;
248        let ny = wy as f64 / h as f64 * detail_scale as f64;
249        base + detail_strength * noise.sample_2d(nx, ny) as f32
250    } else {
251        base
252    }
253}

255// ─── Albedo Derivation ────────────────────────────────────────────────────
256
257/// Derive an albedo (base color) map from a pattern field.
258///
259/// For each pixel:
260/// - Look up cell_id and distance_to_edge
261/// - Compute per-cell color variation deterministically via hash
262/// - Blend toward mortar_color near edges
263/// - Overlay subtle FBM noise for micro-detail
264/// - Output sRGB RGBA8
265pub fn derive_albedo(
266    field: &PatternField,
267    params: &TextureMapParams,
268    rng: &mut SeededRng,
269) -> ImageData {
270    let w = field.width;
271    let h = field.height;
272    let mut img = ImageData::new(w, h, ChannelSemantics::Color);
273    let seed = rng.seed();
274
275    // FBM detail noise (optional)
276    let detail_noise = if params.detail_strength > 0.0 {
277        let noise_rng = rng.fork("albedo_detail");
278        Some(perlin_fbm(noise_rng.seed(), 4, params.detail_scale as f64))
279    } else {
280        None
281    };
282
283    for y in 0..h {
284        for x in 0..w {
285            let cell = field.get(x, y);
286
287            // Per-cell color variation
288            let mut color = cell_color_for_id(
289                cell.cell_id,
290                params.base_color,
291                params.color_variation,
292                seed,
293            );
294
295            // Mortar blend: smooth transition near edges
296            if cell.distance_to_edge < params.mortar_threshold && params.mortar_threshold > 0.0 {
297                let t = cell.distance_to_edge / params.mortar_threshold;
298                // t=0 at edge (full mortar), t=1 at threshold (full surface)
299                for (c, &mc) in color.iter_mut().zip(params.mortar_color.iter()).take(3) {
300                    *c = mc * (1.0 - t) + *c * t;
301                }
302            }
303
304            // Detail overlay: subtle brightness modulation
305            if let Some(ref noise) = detail_noise {
306                let nx = x as f64 / w as f64 * params.detail_scale as f64;
307                let ny = y as f64 / h as f64 * params.detail_scale as f64;
308                let detail = noise.sample_2d(nx, ny) as f32 * params.detail_strength;
309                for c in color.iter_mut().take(3) {
310                    *c = (*c + detail).clamp(0.0, 1.0);
311                }
312            }
313
314            // Encode as sRGB RGBA8
315            let offset = ((y * w + x) * 4) as usize;
316            img.pixels[offset] = linear_to_srgb(color[0]);
317            img.pixels[offset + 1] = linear_to_srgb(color[1]);
318            img.pixels[offset + 2] = linear_to_srgb(color[2]);
319            img.pixels[offset + 3] = (color[3].clamp(0.0, 1.0) * 255.0) as u8;
320        }
321    }
322
323    img
324}

326// ─── Normal Derivation ────────────────────────────────────────────────────
327
328/// Derive a tangent-space normal map from a pattern field.
329///
330/// Builds a combined heightfield (pattern height + optional FBM detail),
331/// then applies a finite-difference gradient to compute tangent-space normals.
332/// The Sobel-like kernel wraps toroidally for seamless tiling.
333///
334/// Output is linear RGBA8 with the standard encoding:
335/// `((nx * 0.5 + 0.5) * 255, (ny * 0.5 + 0.5) * 255, (nz * 0.5 + 0.5) * 255, 255)`
336pub fn derive_normal(
337    field: &PatternField,
338    params: &TextureMapParams,
339    rng: &mut SeededRng,
340) -> ImageData {
341    let w = field.width;
342    let h = field.height;
343    let mut img = ImageData::new(w, h, ChannelSemantics::Normal);
344
345    // FBM detail noise (optional)
346    let detail_noise: Option<Box<dyn NoiseSource>> = if params.detail_strength > 0.0 {
347        let noise_rng = rng.fork("normal_detail");
348        Some(Box::new(perlin_fbm(
349            noise_rng.seed(),
350            4,
351            params.detail_scale as f64,
352        )))
353    } else {
354        None
355    };
356    let noise_ref = detail_noise.as_deref();
357
358    for y in 0..h {
359        for x in 0..w {
360            let ix = x as i32;
361            let iy = y as i32;
362
363            // Finite-difference gradient (2-sample Sobel-like)
364            let h_left = combined_height(
365                field,
366                ix - 1,
367                iy,
368                noise_ref,
369                params.detail_strength,
370                params.detail_scale,
371            );
372            let h_right = combined_height(
373                field,
374                ix + 1,
375                iy,
376                noise_ref,
377                params.detail_strength,
378                params.detail_scale,
379            );
380            let h_up = combined_height(
381                field,
382                ix,
383                iy - 1,
384                noise_ref,
385                params.detail_strength,
386                params.detail_scale,
387            );
388            let h_down = combined_height(
389                field,
390                ix,
391                iy + 1,
392                noise_ref,
393                params.detail_strength,
394                params.detail_scale,
395            );
396
397            let dx = (h_right - h_left) * params.normal_strength;
398            let dy = (h_down - h_up) * params.normal_strength;
399
400            // Tangent-space normal: Z starts at 1.0 before normalization,
401            // guaranteeing nz > 0 in the output.
402            let nx = -dx;
403            let ny = -dy;
404            let nz = 1.0_f32;
405            let len = (nx * nx + ny * ny + nz * nz).sqrt();
406            let nx = nx / len;
407            let ny = ny / len;
408            let nz = nz / len;
409
410            // Encode to RGBA8 (linear — no gamma)
411            let offset = ((y * w + x) * 4) as usize;
412            img.pixels[offset] = ((nx * 0.5 + 0.5).clamp(0.0, 1.0) * 255.0) as u8;
413            img.pixels[offset + 1] = ((ny * 0.5 + 0.5).clamp(0.0, 1.0) * 255.0) as u8;
414            img.pixels[offset + 2] = ((nz * 0.5 + 0.5).clamp(0.0, 1.0) * 255.0) as u8;
415            img.pixels[offset + 3] = 255;
416        }
417    }
418
419    img
420}

422// ─── Roughness Derivation ─────────────────────────────────────────────────
423
424/// Derive a roughness map from a pattern field.
425///
426/// - Surface regions use `roughness_base` with per-cell variation
427/// - Mortar/gap regions use `roughness_mortar`
428/// - Smooth blend between mortar and surface via `distance_to_edge`
429/// - Optional FBM detail overlay (less aggressive than on albedo)
430/// - Output is grayscale RGBA8 (R=G=B=roughness, A=255), linear
431pub fn derive_roughness(
432    field: &PatternField,
433    params: &TextureMapParams,
434    rng: &mut SeededRng,
435) -> ImageData {
436    let w = field.width;
437    let h = field.height;
438    let mut img = ImageData::new(w, h, ChannelSemantics::Roughness);
439    let seed = rng.seed();
440
441    // FBM detail noise (optional)
442    let detail_noise = if params.detail_strength > 0.0 {
443        let noise_rng = rng.fork("roughness_detail");
444        Some(perlin_fbm(noise_rng.seed(), 4, params.detail_scale as f64))
445    } else {
446        None
447    };
448
449    for y in 0..h {
450        for x in 0..w {
451            let cell = field.get(x, y);
452
453            // Per-cell roughness variation
454            let cell_shift = cell_roughness_shift(cell.cell_id, params.roughness_variation, seed);
455            let surface_rough = (params.roughness_base + cell_shift).clamp(0.0, 1.0);
456
457            // Mortar blend
458            let roughness = if cell.distance_to_edge < params.mortar_threshold
459                && params.mortar_threshold > 0.0
460            {
461                let t = cell.distance_to_edge / params.mortar_threshold;
462                params.roughness_mortar * (1.0 - t) + surface_rough * t
463            } else {
464                surface_rough
465            };
466
467            // Detail overlay (half strength compared to albedo)
468            let roughness = if let Some(ref noise) = detail_noise {
469                let nx = x as f64 / w as f64 * params.detail_scale as f64;
470                let ny = y as f64 / h as f64 * params.detail_scale as f64;
471                let detail = noise.sample_2d(nx, ny) as f32 * params.detail_strength * 0.5;
472                (roughness + detail).clamp(0.0, 1.0)
473            } else {
474                roughness
475            };
476
477            // Encode as grayscale RGBA8 (linear — no gamma)
478            let byte = (roughness.clamp(0.0, 1.0) * 255.0) as u8;
479            let offset = ((y * w + x) * 4) as usize;
480            img.pixels[offset] = byte;
481            img.pixels[offset + 1] = byte;
482            img.pixels[offset + 2] = byte;
483            img.pixels[offset + 3] = 255;
484        }
485    }
486
487    img
488}

490// ─── Orchestrator ──────────────────────────────────────────────────────────
491
492/// Derive PBR texture maps from a pattern field according to `params.output_maps`.
493///
494/// Returns one [`ImageData`] per requested map, in the order they appear in
495/// `output_maps`. Each map is generated with a forked RNG for determinism.
496pub fn derive_maps(
497    field: &PatternField,
498    params: &TextureMapParams,
499    rng: &mut SeededRng,
500) -> Vec<ImageData> {
501    let mut results = Vec::with_capacity(params.output_maps.len());
502
503    for map_name in &params.output_maps {
504        match map_name.as_str() {
505            "albedo" => {
506                let mut map_rng = rng.fork("derive_albedo");
507                results.push(derive_albedo(field, params, &mut map_rng));
508            }
509            "normal" => {
510                let mut map_rng = rng.fork("derive_normal");
511                results.push(derive_normal(field, params, &mut map_rng));
512            }
513            "roughness" => {
514                let mut map_rng = rng.fork("derive_roughness");
515                results.push(derive_roughness(field, params, &mut map_rng));
516            }
517            _ => {} // Invalid names filtered by from_toml validation
518        }
519    }
520
521    results
522}

531    /// Create a synthetic pattern field for testing:
532    /// 4 quadrant cells with varying heights and edge distances.
533    fn test_field(size: u32) -> PatternField {
534        let mut field = PatternField::new(size, size);
535        let half = size / 2;
536
537        for y in 0..size {
538            for x in 0..size {
539                let cell_id = match (x < half, y < half) {
540                    (true, true) => 0,
541                    (false, true) => 1,
542                    (true, false) => 2,
543                    (false, false) => 3,
544                };
545
546                // Distance to the nearest quadrant boundary
547                let dx = (x as f32 - half as f32).abs() / half as f32;
548                let dy = (y as f32 - half as f32).abs() / half as f32;
549                let dist_to_edge = dx.min(dy).min(
550                    (x as f32 / size as f32)
551                        .min(y as f32 / size as f32)
552                        .min((1.0 - x as f32 / size as f32).min(1.0 - y as f32 / size as f32)),
553                );
554
555                // Height varies across the field
556                let height = (x as f32 / size as f32 + y as f32 / size as f32) * 0.5;
557
558                *field.get_mut(x, y) = PatternCell {
559                    cell_id,
560                    height,
561                    distance_to_edge: dist_to_edge.clamp(0.0, 1.0),
562                };
563            }
564        }
565
566        field
567    }

619            for x in 0..field.width {

621                let offset = ((y * field.width + x) * 4) as usize;
622                let rgb = [
623                    albedo.pixels[offset],
624                    albedo.pixels[offset + 1],
625                    albedo.pixels[offset + 2],
626                ];
627
628                if cell.distance_to_edge < 0.01 && edge_pixel.is_none() {
629                    edge_pixel = Some(rgb);
630                }
631                if cell.distance_to_edge > 0.3 && interior_pixel.is_none() {
632                    interior_pixel = Some(rgb);
633                }
634            }
635        }
636
637        if let (Some(edge), Some(interior)) = (edge_pixel, interior_pixel) {
638            // At least one channel should differ
639            let differs =
640                edge[0] != interior[0] || edge[1] != interior[1] || edge[2] != interior[2];
641            assert!(
642                differs,
643                "edge {edge:?} should differ from interior {interior:?}"
644            );
645        }
646    }
