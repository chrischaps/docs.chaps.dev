13/// Squared distance between two points, with optional toroidal wrapping.
14pub fn dist_sq(ax: f32, ay: f32, bx: f32, by: f32, seamless: bool) -> f32 {
15    if seamless {
16        let mut dx = (ax - bx).abs();
17        let mut dy = (ay - by).abs();
18        if dx > 0.5 {
19            dx = 1.0 - dx;
20        }
21        if dy > 0.5 {
22            dy = 1.0 - dy;
23        }
24        dx * dx + dy * dy
25    } else {
26        let dx = ax - bx;
27        let dy = ay - by;
28        dx * dx + dy * dy
29    }
30}

32/// Linear interpolation helper.
33#[inline]
34pub fn lerp(a: f32, b: f32, t: f32) -> f32 {
35    a + (b - a) * t
36}

42pub fn generate_seed_points(rng: &mut SeededRng, count: usize) -> Vec<(f32, f32)> {
43    let mut point_rng = rng.fork("voronoi_points");
44    (0..count)
45        .map(|_| (point_rng.next_f32(), point_rng.next_f32()))
46        .collect()
47}

53/// Perform Lloyd's relaxation iterations on the seed points.
54/// Moves each point toward the centroid of its Voronoi cell.
55pub fn lloyds_relaxation(
56    points: &[(f32, f32)],
57    iterations: usize,
58    seamless: bool,
59) -> Vec<(f32, f32)> {
60    let mut current = points.to_vec();
61    let resolution = 64u32; // Sample grid for centroid estimation
62
63    for _ in 0..iterations {
64        // Accumulate centroid sums per cell
65        let mut sum_x = vec![0.0f64; current.len()];
66        let mut sum_y = vec![0.0f64; current.len()];
67        let mut count = vec![0u32; current.len()];
68
69        for sy in 0..resolution {
70            for sx in 0..resolution {
71                let u = (sx as f32 + 0.5) / resolution as f32;
72                let v = (sy as f32 + 0.5) / resolution as f32;
73
74                let mut best_id = 0;
75                let mut best_dist = f32::INFINITY;
76
77                for (id, &(px, py)) in current.iter().enumerate() {
78                    let d = dist_sq(u, v, px, py, seamless);
79                    if d < best_dist {
80                        best_dist = d;
81                        best_id = id;
82                    }
83                }
84
85                sum_x[best_id] += u as f64;
86                sum_y[best_id] += v as f64;
87                count[best_id] += 1;
88            }
89        }
90
91        for i in 0..current.len() {
92            if count[i] > 0 {
93                let cx = (sum_x[i] / count[i] as f64) as f32;
94                let cy = (sum_y[i] / count[i] as f64) as f32;
95                // Keep in [0, 1) for seamless wrapping
96                current[i] = (cx.rem_euclid(1.0), cy.rem_euclid(1.0));
97            }
98        }
99    }
100
101    current
102}

108/// A spatial hash grid for accelerating nearest-neighbor queries on seed points.
109///
110/// For seamless tiling, the grid stores references to all 9 toroidal copies
111/// of each seed point so boundary queries are handled naturally.
112pub struct SpatialGrid {
113    /// Number of grid cells per axis.
114    grid_size: usize,
115    /// For each grid cell, a list of (point_index, wrapped_x, wrapped_y).
116    cells: Vec<Vec<(usize, f32, f32)>>,
117}
118
119impl SpatialGrid {
120    pub fn new(points: &[(f32, f32)], seamless: bool) -> Self {
121        let grid_size = (points.len() as f32).sqrt().ceil().max(2.0) as usize;
122        let mut cells = vec![Vec::new(); grid_size * grid_size];
123        let inv = grid_size as f32;
124
125        let offsets: &[(f32, f32)] = if seamless {
126            &[
127                (0.0, 0.0),
128                (-1.0, 0.0),
129                (1.0, 0.0),
130                (0.0, -1.0),
131                (0.0, 1.0),
132                (-1.0, -1.0),
133                (-1.0, 1.0),
134                (1.0, -1.0),
135                (1.0, 1.0),
136            ]
137        } else {
138            &[(0.0, 0.0)]
139        };
140
141        for (idx, &(px, py)) in points.iter().enumerate() {
142            for &(ox, oy) in offsets {
143                let wx = px + ox;
144                let wy = py + oy;
145                // Only insert if the wrapped point could be relevant (near [0,1]²)
146                if (-0.5..=1.5).contains(&wx) && (-0.5..=1.5).contains(&wy) {
147                    let gx =
148                        ((wx * inv).floor() as isize).clamp(0, grid_size as isize - 1) as usize;
149                    let gy =
150                        ((wy * inv).floor() as isize).clamp(0, grid_size as isize - 1) as usize;
151                    cells[gy * grid_size + gx].push((idx, wx, wy));
152                }
153            }
154        }
155
156        Self { grid_size, cells }
157    }
158
159    /// Find the nearest and second-nearest seed point to `(u, v)`.
160    /// Returns `(closest_id, distance_to_closest, distance_to_second_closest)`.
161    pub fn find_two_nearest(
162        &self,
163        u: f32,
164        v: f32,
165        _points: &[(f32, f32)],
166        _seamless: bool,
167    ) -> (usize, f32, f32) {
168        let inv = self.grid_size as f32;
169        let gx = ((u * inv).floor() as isize).clamp(0, self.grid_size as isize - 1) as usize;
170        let gy = ((v * inv).floor() as isize).clamp(0, self.grid_size as isize - 1) as usize;
171
172        let mut best_id = 0;
173        let mut best_dist = f32::INFINITY;
174        let mut second_dist = f32::INFINITY;
175
176        // Search the 3x3 neighborhood of grid cells
177        for dy in -1i32..=1 {
178            for dx in -1i32..=1 {
179                let nx = (gx as i32 + dx).clamp(0, self.grid_size as i32 - 1) as usize;
180                let ny = (gy as i32 + dy).clamp(0, self.grid_size as i32 - 1) as usize;
181
182                for &(idx, wx, wy) in &self.cells[ny * self.grid_size + nx] {
183                    let dx = u - wx;
184                    let dy = v - wy;
185                    let d = (dx * dx + dy * dy).sqrt();
186
187                    if d < best_dist {
188                        second_dist = best_dist;
189                        best_dist = d;
190                        best_id = idx;
191                    } else if d < second_dist {
192                        second_dist = d;
193                    }
194                }
195            }
196        }
197
198        (best_id, best_dist, second_dist)
199    }
