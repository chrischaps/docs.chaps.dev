1//! Leaf geometry generation for procedural trees.
2//!
3//! Generates billboard quads or diamond-shaped mesh clusters at branch tips,
4//! with configurable density, size variation, and color variation.
5//!
6//! Every leaf is a *card*: a front face and a back face with opposite
7//! winding, so it shows from both sides under back-face culling and each
8//! side is lit by its own normal. Cards are stored whole and in order
9//! ([`card_index_count`] indices each), which is what lets LODs thin
10//! foliage a leaf at a time instead of collapsing edges through it.
11//!
12//! Leaves are coloured by a small palette of shades
13//! ([`LeafParams::shades`]): each shade is its own material and submesh,
14//! tinted by the season ([`season_tint`]) and spread in lightness by
15//! `color_variation`. Every leaf wears one shade, picked at random.
16

28/// Vertices and triangles one leaf card takes, front and back together.
29pub fn card_size(style: LeafStyle) -> (u64, u64) {
30    match style {
31        // 4 corners a side
32        LeafStyle::BillboardCluster => (8, 4),
33        // a centre and 4 tips a side
34        LeafStyle::MeshCluster => (10, 8),
35        LeafStyle::None => (0, 0),
36    }
37}

39/// How many leaf clusters sit along a branch segment `seg_len` long when it
40/// carries leaves along its length: longer segments get several.
41pub fn along_branch_clusters(seg_len: f32, params: &LeafParams) -> usize {
42    let spacing = (params.leaf_spread_radius * 2.0).max(0.2);
43    ((seg_len / spacing).ceil() as usize).max(1)
44}

46/// Indices one leaf card takes in a leaf mesh, front and back faces together.
47pub fn card_index_count(style: LeafStyle) -> usize {
48    match style {
49        // 2 triangles a side
50        LeafStyle::BillboardCluster => 12,
51        // 4 triangles a side
52        LeafStyle::MeshCluster => 24,
53        LeafStyle::None => 0,
54    }
55}

57/// Generate leaf geometry at branch tips and optionally along branch segments.
58///
59/// Dispatches to billboard or mesh cluster generation based on `params.style`.
60/// Returns an empty mesh for [`LeafStyle::None`]. The result has one material
61/// and one submesh per shade that any leaf wears.
62pub fn generate_leaves(
63    tips: &[Vec3],
64    tip_directions: &[Vec3],
65    segments: &[BranchSegment],
66    params: &LeafParams,
67    rng: &mut SeededRng,
68) -> Result<MeshData> {
69    if params.style == LeafStyle::None {
70        return Ok(MeshBuilder::new().build());
71    }
72
73    // Collect along-branch positions
74    let (along_positions, along_directions) = sample_along_branch_positions(segments, params, rng);
75
76    let tip_count = tips.len().min(tip_directions.len());
77    let all_positions: Vec<Vec3> = tips[..tip_count]
78        .iter()
79        .copied()
80        .chain(along_positions)
81        .collect();
82    let all_directions: Vec<Vec3> = tip_directions[..tip_count]
83        .iter()
84        .copied()
85        .chain(along_directions)
86        .collect();
87
88    let palette = leaf_palette(params, rng);
89    let mut shades: Vec<MeshBuilder> = palette.iter().map(|_| MeshBuilder::new()).collect();
90
91    for i in 0..all_positions.len() {
92        if !rng.next_bool(params.density as f64) {
93            continue;
94        }
95
96        let tip = all_positions[i];
97        let dir = safe_normalize(all_directions[i], Vec3::UP);
98
99        for _ in 0..params.leaves_per_tip {
100            match params.style {
101                LeafStyle::BillboardCluster => billboard_card(tip, dir, params, rng, &mut shades),
102                _ => diamond_card(tip, dir, params, rng, &mut shades),
103            }
104        }
105    }
106
107    let mut out = MeshBuilder::new();
108    for (k, shade) in shades.into_iter().enumerate() {
109        let mesh = shade.build();
110        if mesh.indices.is_empty() {
111            continue;
112        }
113        let material = out.add_material(MaterialData {
114            name: format!("leaf_{k}"),
115            base_color: palette[k],
116            metallic: 0.0,
117            roughness: 0.8,
118            normal_map: None,
119        });
120        out.merge_as_submesh(&mesh, material);
121    }
122    Ok(out.build())
123}

125/// Linear RGB of the season's stops after the base colour: yellow at 0.4,
126/// orange at 0.7, red at 1.0 (from sRGB).
127fn season_stops() -> [(f32, [f32; 3]); 3] {
128    let lin = |r: f32, g: f32, b: f32| [srgb_to_linear(r), srgb_to_linear(g), srgb_to_linear(b)];
129    [
130        (0.4, lin(0.82, 0.66, 0.14)),
131        (0.7, lin(0.84, 0.40, 0.09)),
132        (1.0, lin(0.64, 0.12, 0.07)),
133    ]
134}

136/// Tint a linear leaf colour toward the season: 0 leaves it as it is, then it
137/// runs through yellow (0.4) and orange (0.7) to red (1.0), blending in linear
138/// light. Alpha is kept.
139pub fn season_tint(base: [f32; 4], season: f32) -> [f32; 4] {
140    let s = season.clamp(0.0, 1.0);
141    let mut from = (0.0, [base[0], base[1], base[2]]);
142    for (at, rgb) in season_stops() {
143        if s <= at {
144            let t = (s - from.0) / (at - from.0);
145            let mix = |a: f32, b: f32| a + (b - a) * t;
146            return [
147                mix(from.1[0], rgb[0]),
148                mix(from.1[1], rgb[1]),
149                mix(from.1[2], rgb[2]),
150                base[3],
151            ];
152        }
153        from = (at, rgb);
154    }
155    [from.1[0], from.1[1], from.1[2], base[3]]
156}

158/// The leaf shades: `params.shades` colours spread evenly across
159/// `season ± season_variation`, each with its own lightness drawn from
160/// `color_variation`. The lightness step is taken in sRGB, where a step
161/// looks the same on a dark leaf as on a light one; in linear light ±0.08
162/// turned a dark green into chartreuse.
163fn leaf_palette(params: &LeafParams, rng: &mut SeededRng) -> Vec<[f32; 4]> {
164    let n = params.shades.max(1);
165    (0..n)
166        .map(|k| {
167            let spread = if n == 1 {
168                0.0
169            } else {
170                k as f32 / (n - 1) as f32 * 2.0 - 1.0
171            };
172            let season = params.season + spread * params.season_variation;
173            let [r, g, b, a] = season_tint(params.color_base, season);
174            let srgb = [linear_to_srgb(r), linear_to_srgb(g), linear_to_srgb(b), a];
175            let [r, g, b, a] = rng.next_color_variation_hsl(srgb, params.color_variation);
176            [srgb_to_linear(r), srgb_to_linear(g), srgb_to_linear(b), a]
177        })
178        .collect()
179}

181/// Convert a single linear channel value to sRGB (inverse of [`srgb_to_linear`]).
182fn linear_to_srgb(c: f32) -> f32 {
183    if c <= 0.0031308 {
184        c * 12.92
185    } else {
186        1.055 * c.powf(1.0 / 2.4) - 0.055
187    }
188}

190/// Normalize a Vec3, returning `fallback` if zero-length.
191fn safe_normalize(v: Vec3, fallback: Vec3) -> Vec3 {
192    let n = v.normalized();
193    if n.length() < 1e-6 {
194        fallback
195    } else {
196        n
197    }
198}

200/// Sample leaf placement positions along branch segments.
201///
202/// Filters segments by `min_depth`, then probabilistically samples each one
203/// using `density`. Places leaves at the segment midpoint (or at multiple
204/// evenly-spaced positions for long segments relative to `spread_radius`).
205fn sample_along_branch_positions(
206    segments: &[BranchSegment],
207    params: &LeafParams,
208    rng: &mut SeededRng,
209) -> (Vec<Vec3>, Vec<Vec3>) {
210    let mut positions = Vec::new();
211    let mut directions = Vec::new();
212
213    if params.leaf_along_branch_density <= 0.0 {
214        return (positions, directions);
215    }
216
217    for seg in segments {
218        if seg.depth < params.leaf_along_branch_min_depth {
219            continue;
220        }
221
222        if !rng.next_bool(params.leaf_along_branch_density as f64) {
223            continue;
224        }
225
226        let seg_len = seg.length();
227        if seg_len < 1e-6 {
228            continue;
229        }
230
231        let dir = seg.direction();
232        let num_points = along_branch_clusters(seg_len, params);
233
234        for j in 0..num_points {
235            let t = if num_points == 1 {
236                0.5
237            } else {
238                (j as f32 + 0.5) / num_points as f32
239            };
240            let pos = seg.start + (seg.end - seg.start) * t;
241            positions.push(pos);
242            directions.push(dir);
243        }
244    }
245
246    (positions, directions)
247}

249/// Uniform in `[-radius, radius)`, or 0 for no radius (an empty range panics).
250fn jitter(rng: &mut SeededRng, radius: f32) -> f32 {
251    if radius > 0.0 {
252        rng.next_range(-radius, radius)
253    } else {
254        0.0
255    }
256}

258/// Pick the shade a leaf wears.
259fn pick_shade(rng: &mut SeededRng, shades: usize) -> usize {
260    ((rng.next_f32() * shades as f32) as usize).min(shades - 1)
261}

263/// Add one leaf card to `builder`: the front face as given (counter-clockwise
264/// about `normal`), then the same corners again with the normal flipped and
265/// the winding reversed for the back.
266fn add_card(
267    builder: &mut MeshBuilder,
268    corners: &[(Vec3, [f32; 2])],
269    triangles: &[[u32; 3]],
270    normal: Vec3,
271    tangent: Vec3,
272) {
273    for front in [true, false] {
274        let sign = if front { 1.0 } else { -1.0 };
275        let mut base = None;
276        for &(p, uv) in corners {
277            let i = builder.add_vertex(Vertex {
278                position: p.to_array(),
279                normal: (normal * sign).to_array(),
280                tangent: [tangent.x, tangent.y, tangent.z, sign],
281                uv,
282            });
283            base.get_or_insert(i);
284        }
285        let base = base.unwrap_or(0);
286        for t in triangles {
287            if front {
288                builder.add_triangle(base + t[0], base + t[1], base + t[2]);
289            } else {
290                builder.add_triangle(base + t[0], base + t[2], base + t[1]);
291            }
292        }
293    }
294}

296/// One flat quad leaf facing along the branch with a random spin.
297fn billboard_card(
298    tip: Vec3,
299    dir: Vec3,
300    params: &LeafParams,
301    rng: &mut SeededRng,
302    shades: &mut [MeshBuilder],
303) {
304    let spread = params.leaf_spread_radius;
305
306    // Find a perpendicular basis
307    let perp = perpendicular_to(dir);
308    let bitangent = safe_normalize(dir.cross(&perp), Vec3::new(0.0, 0.0, 1.0));
309
310    let scale = params.leaf_size
311        * (1.0 + rng.next_range(-params.leaf_size_variation, params.leaf_size_variation));
312    let half = scale * 0.5;
313
314    // Random rotation around the branch direction
315    let angle = rng.next_range(0.0, std::f32::consts::TAU);
316    let u = rotate_around_axis(perp, dir, angle);
317    let v = rotate_around_axis(bitangent, dir, angle);
318
319    // Random offset within the spread radius sphere
320    let offset = dir * jitter(rng, spread) + u * jitter(rng, spread) + v * jitter(rng, spread);
321    let center = tip + offset;
322
323    let shade = pick_shade(rng, shades.len());
324    // The quad spans u and v, so it faces along u × v (the branch direction).
325    let normal = safe_normalize(u.cross(&v), dir);
326    add_card(
327        &mut shades[shade],
328        &[
329            (center - u * half - v * half, [0.0, 0.0]),
330            (center + u * half - v * half, [1.0, 0.0]),
331            (center + u * half + v * half, [1.0, 1.0]),
332            (center - u * half + v * half, [0.0, 1.0]),
333        ],
334        &[[0, 1, 2], [0, 2, 3]],
335        normal,
336        u,
337    );
338}

340/// One diamond leaf, tilted off the branch direction by a random pitch/yaw.
341fn diamond_card(
342    tip: Vec3,
343    dir: Vec3,
344    params: &LeafParams,
345    rng: &mut SeededRng,
346    shades: &mut [MeshBuilder],
347) {
348    let spread = params.leaf_spread_radius;
349
350    let scale = params.leaf_size
351        * (1.0 + rng.next_range(-params.leaf_size_variation, params.leaf_size_variation));
352    let half = scale * 0.5;
353
354    // Random orientation offset
355    let pitch_offset = rng.next_range(-0.5, 0.5);
356    let yaw_offset = rng.next_range(-0.5, 0.5);
357
358    let perp = perpendicular_to(dir);
359    let bitangent = safe_normalize(dir.cross(&perp), Vec3::new(0.0, 0.0, 1.0));
360
361    // Apply random pitch/yaw to the leaf direction
362    let leaf_dir = safe_normalize(dir + perp * pitch_offset + bitangent * yaw_offset, dir);
363    let leaf_perp = perpendicular_to(leaf_dir);
364    let leaf_bi = safe_normalize(leaf_dir.cross(&leaf_perp), Vec3::new(0.0, 0.0, 1.0));
365
366    // Random offset within the spread radius sphere
367    let offset =
368        dir * jitter(rng, spread) + perp * jitter(rng, spread) + bitangent * jitter(rng, spread);
369    let center = tip + offset;
370
371    let shade = pick_shade(rng, shades.len());
372    // Diamond: center, +X, +Y, -X, -Y in leaf space. perp × bi = leaf_dir, so
373    // the fan below winds counter-clockwise about the leaf's direction.
374    add_card(
375        &mut shades[shade],
376        &[
377            (center, [0.5, 0.5]),
378            (center + leaf_perp * half, [1.0, 0.5]),
379            (center + leaf_bi * half, [0.5, 0.0]),
380            (center - leaf_perp * half, [0.0, 0.5]),
381            (center - leaf_bi * half, [0.5, 1.0]),
382        ],
383        &[[0, 1, 2], [0, 2, 3], [0, 3, 4], [0, 4, 1]],
384        leaf_dir,
385        leaf_perp,
386    );
387}

389/// Find a vector perpendicular to `v`.
390fn perpendicular_to(v: Vec3) -> Vec3 {
391    let candidate = if v.x.abs() < 0.9 {
392        Vec3::RIGHT
393    } else {
394        Vec3::UP
395    };
396    let result = v.cross(&candidate).normalized();
397    if result.length() < 1e-6 {
398        Vec3::RIGHT
399    } else {
400        result
401    }
402}

428    /// A ring of `n` tips around a trunk at y = 4, each pointing outward.
429    fn ring_of_tips(n: usize) -> (Vec<Vec3>, Vec<Vec3>) {
430        (0..n)
431            .map(|i| {
432                let a = i as f32 / n as f32 * std::f32::consts::TAU;
433                let out = Vec3::new(a.cos(), 0.3, a.sin()).normalized();
434                (Vec3::new(0.0, 4.0, 0.0) + out * 2.0, out)
435            })
436            .unzip()
437    }

535    #[test]
536    fn cards_face_both_ways_and_normals_match_winding() {
537        // Each card's front faces along its branch (outward from the trunk),
538        // the back faces the other way, and every triangle's winding agrees
539        // with its vertices' normal, so lighting and culling see the same side.
540        for style in [LeafStyle::BillboardCluster, LeafStyle::MeshCluster] {
541            let (tips, dirs) = ring_of_tips(24);
542            let params = LeafParams {
543                style,
544                density: 1.0,
545                leaves_per_tip: 2,
546                leaf_spread_radius: 0.0,
547                ..LeafParams::default()
548            };
549            let mesh =
550                generate_leaves(&tips, &dirs, &[], &params, &mut SeededRng::new(11)).unwrap();
551            let pos = |i: u32| Vec3::from_array(mesh.vertices[i as usize].position);
552            let per_card = card_index_count(style);
553            let mut outward = 0;
554            for card in mesh.indices.chunks(per_card) {
555                let (front, back) = card.split_at(per_card / 2);
556                let nf = Vec3::from_array(mesh.vertices[front[0] as usize].normal);
557                let nb = Vec3::from_array(mesh.vertices[back[0] as usize].normal);
558                assert!(
559                    nf.dot(&nb) < -0.99,
560                    "{style:?}: back normal should oppose front"
561                );
562                for tri in card.chunks(3) {
563                    let face = (pos(tri[1]) - pos(tri[0])).cross(&(pos(tri[2]) - pos(tri[0])));
564                    let n = Vec3::from_array(mesh.vertices[tri[0] as usize].normal);
565                    assert!(
566                        face.dot(&n) > 0.0,
567                        "{style:?}: winding disagrees with normal"
568                    );
569                }
570                let centre = pos(front[0]);
571                let away = (centre - Vec3::new(0.0, centre.y, 0.0)).normalized();
572                if nf.dot(&away) > 0.0 {
573                    outward += 1;
574                }
575            }
576            let total = mesh.indices.len() / per_card;
577            assert!(
578                outward * 10 >= total * 9,
579                "{style:?}: only {outward}/{total} fronts face away from the trunk"
580            );
581        }
582    }

586        let (tips, dirs) = ring_of_tips(100);

606            }

618    #[test]
619    fn season_runs_green_to_red() {
620        let green = [0.03, 0.26, 0.004, 1.0];
621        assert_eq!(season_tint(green, 0.0), green);
622        let yellow = season_tint(green, 0.4);
623        let orange = season_tint(green, 0.7);
624        let red = season_tint(green, 1.0);
625        assert!(
626            yellow[0] > yellow[2] && yellow[1] > 0.3,
627            "yellow: {yellow:?}"
628        );
629        assert!(orange[0] > 2.0 * orange[1], "orange: {orange:?}");
630        assert!(red[0] > 10.0 * red[1], "red: {red:?}");
631        // Red's share of the colour climbs monotonically through the year.
632        let mut last = 0.0;
633        for i in 0..=20 {
634            let c = season_tint(green, i as f32 / 20.0);
635            let share = c[0] / (c[0] + c[1] + c[2]);
636            assert!(
637                share >= last - 1e-6,
638                "red share fell at season {}",
639                i as f32 / 20.0
640            );
641            last = share;
642        }
643    }
