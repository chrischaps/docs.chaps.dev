27/// Typed trunk configuration extracted from a `ProcGenSpec`'s params table.
28#[derive(Debug, Clone)]
29pub struct TrunkParams {
30    /// Total trunk height in meters.
31    pub height: f32,
32    /// Radius at the base of the trunk.
33    pub radius_base: f32,
34    /// Radius at the top of the trunk.
35    pub radius_top: f32,
36    /// Number of segments along the spine (vertical resolution).
37    pub segments: u32,
38    /// Number of radial segments in the cross-section.
39    pub radial_segments: u32,
40    /// Perlin displacement magnitude for trunk curvature.
41    pub curve_noise: f32,
42    /// Bark base color as linear RGBA.
43    pub bark_color: [f32; 4],
44    /// Bark roughness factor.
45    pub bark_roughness: f32,
46    /// Normal map strength multiplier.
47    pub bark_normal_strength: f32,
48    /// Normal map resolution (width = height).
49    pub bark_normal_resolution: u32,
50}

69impl TrunkParams {
70    /// Parse trunk parameters from a TOML `params` value.
71    ///
72    /// Missing fields fall back to sensible defaults. Invalid values produce
73    /// descriptive errors.
74    pub fn from_toml(params: &toml::Value) -> Result<Self> {
75        let table = params
76            .as_table()
77            .ok_or_else(|| ProcGenError::InvalidParameter {
78                name: "params".into(),
79                reason: "expected a TOML table".into(),
80            })?;
81
82        let mut p = Self::default();
83
84        if let Some(v) = table.get("trunk_height") {
85            p.height = toml_f32(v, "trunk_height")?;
86            if p.height <= 0.0 {
87                return Err(ProcGenError::InvalidParameter {
88                    name: "trunk_height".into(),
89                    reason: "must be positive".into(),
90                });
91            }
92        }
93
94        if let Some(v) = table.get("trunk_radius_base") {
95            p.radius_base = toml_f32(v, "trunk_radius_base")?;
96            if p.radius_base <= 0.0 {
97                return Err(ProcGenError::InvalidParameter {
98                    name: "trunk_radius_base".into(),
99                    reason: "must be positive".into(),
100                });
101            }
102        }
103
104        if let Some(v) = table.get("trunk_radius_top") {
105            p.radius_top = toml_f32(v, "trunk_radius_top")?;
106            if p.radius_top < 0.0 {
107                return Err(ProcGenError::InvalidParameter {
108                    name: "trunk_radius_top".into(),
109                    reason: "must be non-negative".into(),
110                });
111            }
112        }
113
114        if let Some(v) = table.get("trunk_segments") {
115            p.segments = toml_u32(v, "trunk_segments")?;
116            if p.segments < 2 {
117                return Err(ProcGenError::InvalidParameter {
118                    name: "trunk_segments".into(),
119                    reason: "must be at least 2".into(),
120                });
121            }
122        }
123
124        if let Some(v) = table.get("radial_segments") {
125            p.radial_segments = toml_u32(v, "radial_segments")?;
126            if p.radial_segments < 3 {
127                return Err(ProcGenError::InvalidParameter {
128                    name: "radial_segments".into(),
129                    reason: "must be at least 3".into(),
130                });
131            }
132        }
133
134        if let Some(v) = table.get("trunk_curve_noise") {
135            p.curve_noise = toml_f32(v, "trunk_curve_noise")?;
136            if p.curve_noise < 0.0 {
137                return Err(ProcGenError::InvalidParameter {
138                    name: "trunk_curve_noise".into(),
139                    reason: "must be non-negative".into(),
140                });
141            }
142        }
143
144        if let Some(v) = table.get("bark_color_base") {
145            let hex = v.as_str().ok_or_else(|| ProcGenError::InvalidParameter {
146                name: "bark_color_base".into(),
147                reason: "expected a hex color string (e.g. \"#8B4513\")".into(),
148            })?;
149            p.bark_color = parse_hex_color(hex)?;
150        }
151
152        if let Some(v) = table.get("bark_roughness") {
153            p.bark_roughness = toml_f32(v, "bark_roughness")?;
154        }
155
156        if let Some(v) = table.get("bark_normal_strength") {
157            p.bark_normal_strength = toml_f32(v, "bark_normal_strength")?;
158        }
159
160        if let Some(v) = table.get("bark_normal_resolution") {
161            p.bark_normal_resolution = toml_u32(v, "bark_normal_resolution")?;
162            if !p.bark_normal_resolution.is_power_of_two() || p.bark_normal_resolution < 16 {
163                return Err(ProcGenError::InvalidParameter {
164                    name: "bark_normal_resolution".into(),
165                    reason: "must be a power of 2 and at least 16".into(),
166                });
167            }
168        }
169
170        Ok(p)
171    }
172}

620/// Generate a complete tree (trunk + branches) from parameters and RNG.
621///
622/// 1. Generates the trunk via [`generate_trunk`]
623/// 2. Generates branches via [`generate_branches`] attached to the trunk tip
624/// 3. Merges both meshes and returns the combined result
625pub fn generate_tree(
626    trunk_params: &TrunkParams,
627    branch_params: &BranchParams,
628    rng: &mut SeededRng,
629) -> Result<TreeOutput> {
630    let mut trunk_rng = rng.fork("trunk");
631    let trunk = generate_trunk(trunk_params, &mut trunk_rng)?;
632
633    let mut branch_rng = rng.fork("branches");
634    let branch_output = generate_branches(
635        branch_params,
636        trunk.tip_position,
637        trunk.tip_direction,
638        trunk_params.radius_top,
639        &mut branch_rng,
640    )?;
641
642    // Merge trunk + branch meshes
643    let mut builder = MeshBuilder::new();
644    builder.merge(&trunk.mesh);
645    builder.merge(&branch_output.mesh);
646    builder.compute_normals();
647    builder.compute_tangents();
648
649    // Preserve bark material
650    let mut mesh = builder.build();
651    mesh.materials = trunk.mesh.materials.clone();
652
653    Ok(TreeOutput {
654        mesh,
655        normal_map: trunk.normal_map,
656        branch_tips: branch_output.tip_positions,
657        branch_tip_directions: branch_output.tip_directions,
658        branch_segments: branch_output.segments,
659    })
660}

662// ─── TreeGenerator ──────────────────────────────────────────────────────────
663
664/// Procedural tree generator — produces trunk, branches, and leaves with
665/// optional LOD levels.
666///
667/// Registered as `"tree_v1"` in the generator registry. All parameters are
668/// parsed from the spec's `params` table via [`TrunkParams`], [`BranchParams`],
669/// and [`LeafParams`].
670pub struct TreeGenerator;
671
672impl Generator for TreeGenerator {
673    fn type_name(&self) -> &str {
674        "tree_v1"
675    }
676
677    fn output_kind(&self) -> OutputKind {
678        OutputKind::Mesh
679    }
680
681    fn generate(&self, spec: &ProcGenSpec, rng: &mut SeededRng) -> Result<GeneratorOutput> {
682        // 1. Parse params
683        let trunk_params = TrunkParams::from_toml(&spec.params)?;
684        let branch_params = BranchParams::from_toml(&spec.params)?;
685        let leaf_params = LeafParams::from_toml(&spec.params)?;
686
687        // 2. Generate tree (trunk + branches)
688        let tree = generate_tree(&trunk_params, &branch_params, rng)?;
689
690        // 3. Generate leaves
691        let leaf_mesh = generate_leaves(
692            &tree.branch_tips,
693            &tree.branch_tip_directions,
694            &tree.branch_segments,
695            &leaf_params,
696            &mut rng.fork("leaves"),
697        )?;
698
699        // 4. Merge tree + leaves (with submesh tracking for multi-material)
700        let mut builder = MeshBuilder::new();
701        builder.merge_as_submesh(&tree.mesh, 0);
702        if leaf_mesh.vertex_count() > 0 {
703            builder.merge_as_submesh(&leaf_mesh, 1);
704        }
705        builder.compute_normals();
706        builder.compute_tangents();
707        let mut full_mesh = builder.build();
708
709        // Merge materials from bark + leaf
710        let mut materials = tree.mesh.materials.clone();
711        if leaf_mesh.vertex_count() > 0 {
712            materials.extend(leaf_mesh.materials.iter().cloned());
713        }
714        full_mesh.materials = materials;
715
716        // 5. Generate LODs if specified
717        if let Some(ref lod_levels) = spec.lod {
718            if !lod_levels.is_empty() {
719                let targets: Vec<u32> = lod_levels.iter().map(|l| l.target_triangles).collect();
720                let lods = generate_lods(&full_mesh, &targets);
721                return Ok(GeneratorOutput::MeshWithLods(lods));
722            }
723        }
724
725        Ok(GeneratorOutput::Mesh(full_mesh))
726    }
727
728    fn param_schema(&self) -> serde_json::Value {
729        serde_json::json!({
730            "type": "object",
731            "properties": {
732                "trunk_height": { "type": "number", "minimum": 0.01, "default": 5.0 },
733                "trunk_radius_base": { "type": "number", "minimum": 0.01, "default": 0.3 },
734                "trunk_radius_top": { "type": "number", "minimum": 0.0, "default": 0.1 },
735                "trunk_segments": { "type": "integer", "minimum": 2, "default": 10 },
736                "radial_segments": { "type": "integer", "minimum": 3, "default": 8 },
737                "trunk_curve_noise": { "type": "number", "minimum": 0.0, "default": 0.1 },
738                "bark_color_base": { "type": "string", "pattern": "^#?[0-9a-fA-F]{6,8}$", "default": "#573214" },
739                "bark_roughness": { "type": "number", "minimum": 0.0, "maximum": 1.0, "default": 0.9 },
740                "bark_normal_strength": { "type": "number", "default": 1.0 },
741                "bark_normal_resolution": { "type": "integer", "default": 256 },
742                "branch_method": { "type": "string", "enum": ["lsystem", "space_colonization"], "default": "lsystem" },
743                "branch_levels": { "type": "integer", "minimum": 1, "default": 3 },
744                "branch_angle_min": { "type": "number", "default": 20.0 },
745                "branch_angle_max": { "type": "number", "default": 40.0 },
746                "branch_length_falloff": { "type": "number", "exclusiveMinimum": 0.0, "maximum": 1.0, "default": 0.7 },
747                "branch_radius_falloff": { "type": "number", "exclusiveMinimum": 0.0, "maximum": 1.0, "default": 0.6 },
748                "leaf_style": { "type": "string", "enum": ["billboard", "mesh", "none"], "default": "billboard" },
749                "leaf_density": { "type": "number", "minimum": 0.0, "maximum": 1.0, "default": 0.8 },
750                "leaf_color_base": { "type": "string", "pattern": "^#?[0-9a-fA-F]{6,8}$", "default": "#2D8C0D" },
751                "leaf_color_variation": { "type": "number", "minimum": 0.0, "default": 0.08 },
752                "leaf_size": { "type": "number", "minimum": 0.01, "default": 0.15 },
753                "leaf_size_variation": { "type": "number", "minimum": 0.0, "maximum": 1.0, "default": 0.3 },
754                "leaves_per_tip": { "type": "integer", "minimum": 1, "default": 5 },
755                "leaf_spread_radius": { "type": "number", "minimum": 0.0, "default": 0.3 },
756                "leaf_along_branch_density": { "type": "number", "minimum": 0.0, "maximum": 1.0, "default": 0.0 },
757                "leaf_along_branch_min_depth": { "type": "integer", "minimum": 0, "default": 2 }
758            },

760        })

762
763    fn estimate_cost(&self, spec: &ProcGenSpec) -> GenerationCost {
764        let trunk = TrunkParams::from_toml(&spec.params).unwrap_or_default();
765        let branch = BranchParams::from_toml(&spec.params).unwrap_or_default();
766
767        let trunk_verts = trunk.segments * trunk.radial_segments;
768        let branch_verts = branch.max_segments * branch.radial_segments;
769        let total_verts = trunk_verts + branch_verts;
770
771        GenerationCost {
772            estimated_vertices: total_verts,
773            estimated_triangles: total_verts, // rough estimate
774            estimated_texture_bytes: (trunk.bark_normal_resolution as u64)
775                .pow(2)
776                .saturating_mul(4),
777            estimated_generation_ms: 50.0 + branch.max_segments as f64 * 0.01,
778        }
779    }
