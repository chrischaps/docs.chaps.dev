30/// Generate a trunk mesh from the given parameters and RNG.
31///
32/// This is the main entry point for Task 3.1. It:
33/// 1. Generates a noise-displaced vertical spine
34/// 2. Extrudes a tapered tube along the spine
35/// 3. Recomputes normals and tangents
36/// 4. Generates a procedural bark normal map
37/// 5. Sets up the bark material
38pub fn generate_trunk(params: &TrunkParams, rng: &mut SeededRng) -> Result<TrunkOutput> {
39    // --- Spine generation ---
40    let mut spine_rng = rng.fork("trunk_spine");
41    let noise_x = PerlinNoise::new(spine_rng.fork("noise_x").seed());
42    let noise_z = PerlinNoise::new(spine_rng.fork("noise_z").seed());
43
44    let num_points = params.segments + 1;
45    let freq = 1.0; // frequency for spine noise sampling
46    let mut positions = Vec::with_capacity(num_points as usize);
47
48    for i in 0..num_points {
49        let t = i as f32 / params.segments as f32;
50        let y = t * params.height;
51
52        let x = if params.curve_noise > 0.0 {
53            noise_x.sample_2d(y as f64 * freq, 0.0) as f32 * params.curve_noise
54        } else {
55            0.0
56        };
57        let z = if params.curve_noise > 0.0 {
58            noise_z.sample_2d(y as f64 * freq, 0.0) as f32 * params.curve_noise
59        } else {
60            0.0
61        };
62
63        positions.push(Vec3::new(x, y, z));
64    }
65
66    // Record tip info before extrusion
67    let tip_position = *positions.last().unwrap();
68    let tip_direction = if positions.len() >= 2 {
69        let n = positions.len();
70        let d = positions[n - 1] - positions[n - 2];
71        let len = d.length();
72        if len > 1e-10 {
73            Vec3::new(d.x / len, d.y / len, d.z / len)
74        } else {
75            Vec3::UP
76        }
77    } else {
78        Vec3::UP
79    };
80
81    // --- Mesh extrusion ---
82    let spine = build_spine_from_positions(&positions, params.radius_base, params.radius_top);
83    let mut mesh = extrude_along_curve(&ExtrudeParams {
84        spine,
85        radial_segments: params.radial_segments,
86        caps: true,
87        uv_length_scale: 1.0,
88    })?;
89
90    // Recompute normals and tangents for correct lighting
91    compute_normals(&mut mesh.vertices, &mesh.indices);
92    compute_tangents(&mut mesh.vertices, &mesh.indices);
93
94    // --- Material ---
95    mesh.materials = vec![MaterialData {
96        name: "bark".into(),
97        base_color: params.bark_color,
98        metallic: 0.0,
99        roughness: params.bark_roughness,
100    }];
101
102    // --- Bark normal map ---
103    let mut bark_rng = rng.fork("bark");
104    let normal_map = generate_bark_normal_map(
105        &mut bark_rng,
106        params.bark_normal_resolution,
107        params.bark_normal_strength,
108    );
109
110    Ok(TrunkOutput {
111        mesh,
112        normal_map,
113        tip_position,
114        tip_direction,
115    })
116}

126    #[test]
127    fn mesh_validity() {
128        let params = default_params();
129        let mut rng = SeededRng::new(42);
130        let output = generate_trunk(&params, &mut rng).unwrap();
131        assert!(
132            output.mesh.validate().is_ok(),
133            "mesh should be valid: {:?}",
134            output.mesh.validate()
135        );
136
137        // Check no degenerate triangles (zero-area)
138        let verts = &output.mesh.vertices;
139        let indices = &output.mesh.indices;
140        for tri in indices.chunks_exact(3) {
141            let p0 = Vec3::from_array(verts[tri[0] as usize].position);
142            let p1 = Vec3::from_array(verts[tri[1] as usize].position);
143            let p2 = Vec3::from_array(verts[tri[2] as usize].position);
144
145            let e0 = p1 - p0;
146            let e1 = p2 - p0;
147            let cross = e0.cross(&e1);
148            let area = cross.length() * 0.5;
149            assert!(
150                area > 1e-10,
151                "degenerate triangle with area {area}: [{:?}, {:?}, {:?}]",
152                p0,
153                p1,
154                p2
155            );
156        }
157    }

159    #[test]
160    fn bounding_box_sanity() {
161        let params = TrunkParams {
162            height: 8.0,
163            radius_base: 0.5,
164            ..default_params()
165        };
166        let mut rng = SeededRng::new(42);
167        let output = generate_trunk(&params, &mut rng).unwrap();
168
169        let size = output.mesh.bounding_box.size();
170
171        // Height should be roughly trunk_height (within 20% for noise displacement)
172        assert!(
173            (size.y - params.height).abs() < params.height * 0.3,
174            "bbox height {:.2} should be near trunk height {:.2}",
175            size.y,
176            params.height
177        );
178
179        // Width should be roughly 2 * radius_base (plus some noise margin)
180        assert!(
181            size.x < params.radius_base * 4.0,
182            "bbox width {:.2} too large for radius_base {:.2}",
183            size.x,
184            params.radius_base
185        );
186        assert!(
187            size.z < params.radius_base * 4.0,
188            "bbox depth {:.2} too large for radius_base {:.2}",
189            size.z,
190            params.radius_base
191        );
192    }

194    #[test]
195    fn determinism_same_seed() {
196        let params = default_params();
197        let mut rng1 = SeededRng::new(12345);
198        let mut rng2 = SeededRng::new(12345);
199
200        let out1 = generate_trunk(&params, &mut rng1).unwrap();
201        let out2 = generate_trunk(&params, &mut rng2).unwrap();
202
203        // Vertex-for-vertex equality
204        assert_eq!(
205            out1.mesh.vertices.len(),
206            out2.mesh.vertices.len(),
207            "vertex count should match"
208        );
209        for (i, (v1, v2)) in out1
210            .mesh
211            .vertices
212            .iter()
213            .zip(out2.mesh.vertices.iter())
214            .enumerate()
215        {
216            assert_eq!(v1.position, v2.position, "vertex {i} position differs");
217        }
218        assert_eq!(out1.mesh.indices, out2.mesh.indices);
219    }
