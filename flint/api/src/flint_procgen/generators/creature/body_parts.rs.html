1//! Generates geometry for body parts attached to bones.
2//!
3//! Each body part specifies a bone, a shape, dimensions, and a material.
4//! The geometry is positioned in the bone's local space and rigidly skinned.
5

14/// Parsed body part definition.
15#[derive(Debug, Clone)]
16pub struct BodyPartDef {
17    pub bone: String,
18    pub shape: String,
19    pub dimensions: [f32; 3],
20    pub offset: [f32; 3],
21    pub material: String,
22}
23
24/// Parse a body part definition from a TOML table.
25pub fn parse_body_part(table: &toml::value::Table, idx: usize) -> Result<BodyPartDef> {
26    let prefix = format!("body_parts[{idx}]");
27
28    let bone = table
29        .get("bone")
30        .ok_or_else(|| ProcGenError::MissingParameter(format!("{prefix}.bone")))?;
31    let bone = toml_string(bone, &format!("{prefix}.bone"))?;
32
33    let shape = table
34        .get("shape")
35        .ok_or_else(|| ProcGenError::MissingParameter(format!("{prefix}.shape")))?;
36    let shape = toml_string(shape, &format!("{prefix}.shape"))?;
37
38    let dimensions = if let Some(v) = table.get("dimensions") {
39        let arr = v.as_array().ok_or_else(|| ProcGenError::InvalidParameter {
40            name: format!("{prefix}.dimensions"),
41            reason: "expected an array of 3 numbers".into(),
42        })?;
43        if arr.len() != 3 {
44            return Err(ProcGenError::InvalidParameter {
45                name: format!("{prefix}.dimensions"),
46                reason: format!("expected 3 elements, got {}", arr.len()),
47            });
48        }
49        [
50            toml_f32(&arr[0], &format!("{prefix}.dimensions[0]"))?,
51            toml_f32(&arr[1], &format!("{prefix}.dimensions[1]"))?,
52            toml_f32(&arr[2], &format!("{prefix}.dimensions[2]"))?,
53        ]
54    } else {
55        [1.0, 1.0, 1.0]
56    };
57
58    let offset = if let Some(v) = table.get("offset") {
59        let arr = v.as_array().ok_or_else(|| ProcGenError::InvalidParameter {
60            name: format!("{prefix}.offset"),
61            reason: "expected an array of 3 numbers".into(),
62        })?;
63        if arr.len() != 3 {
64            return Err(ProcGenError::InvalidParameter {
65                name: format!("{prefix}.offset"),
66                reason: format!("expected 3 elements, got {}", arr.len()),
67            });
68        }
69        [
70            toml_f32(&arr[0], &format!("{prefix}.offset[0]"))?,
71            toml_f32(&arr[1], &format!("{prefix}.offset[1]"))?,
72            toml_f32(&arr[2], &format!("{prefix}.offset[2]"))?,
73        ]
74    } else {
75        [0.0, 0.0, 0.0]
76    };
77
78    let material = table
79        .get("material")
80        .and_then(|v| v.as_str())
81        .unwrap_or("default")
82        .to_string();
83
84    Ok(BodyPartDef {
85        bone,
86        shape,
87        dimensions,
88        offset,
89        material,
90    })
91}

93/// Generate geometry for a body part, returning skinned vertices and indices.
94///
95/// The geometry is generated centered at origin, then offset. Bone transforms
96/// are applied via the skeleton (not baked into vertex positions), so vertices
97/// are in the bone's local space.
98pub fn generate_body_part(
99    part: &BodyPartDef,
100    skeleton: &SkeletonDef,
101    world_transforms: &[[f32; 16]],
102) -> Result<(Vec<SkinnedVertex>, Vec<u32>, String)> {
103    let bone_idx = skeleton.find_bone(&part.bone).ok_or_else(|| {
104        ProcGenError::InvalidParameter {
105            name: "body_part.bone".into(),
106            reason: format!("bone '{}' not found in skeleton", part.bone),
107        }
108    })?;
109
110    let mesh = generate_shape(&part.shape, &part.dimensions)?;
111
112    // Transform vertices to world space (baked) and assign rigid skinning to this bone
113    let world = &world_transforms[bone_idx];
114    let skinned_verts: Vec<SkinnedVertex> = mesh
115        .vertices
116        .iter()
117        .map(|v| {
118            let local_pos = [
119                v.position[0] + part.offset[0],
120                v.position[1] + part.offset[1],
121                v.position[2] + part.offset[2],
122            ];
123            let world_pos = transform_point(world, &local_pos);
124            let world_norm = transform_normal(world, &v.normal);
125
126            SkinnedVertex {
127                position: world_pos,
128                normal: world_norm,
129                tangent: v.tangent,
130                uv: v.uv,
131                joint_indices: [bone_idx as u16, 0, 0, 0],
132                joint_weights: [1.0, 0.0, 0.0, 0.0],
133            }
134        })
135        .collect();
136
137    Ok((skinned_verts, mesh.indices, part.material.clone()))
138}

140fn generate_shape(shape: &str, dimensions: &[f32; 3]) -> Result<MeshData> {
141    match shape {
142        "ellipsoid" => Ok(ellipsoid(&EllipsoidParams {
143            radius_x: dimensions[0],
144            radius_y: dimensions[1],
145            radius_z: dimensions[2],
146            segments: 16,
147            rings: 12,
148        })),
149        "sphere" => {
150            let radius = dimensions[0];
151            Ok(sphere(&SphereParams {
152                radius,
153                segments: 16,
154                rings: 12,
155            }))
156        }
157        "capsule" => {
158            // Approximate capsule as tapered cylinder with hemispherical ends
159            let radius = dimensions[0];
160            let height = dimensions[1];
161            Ok(tapered_cylinder(&TaperedCylinderParams {
162                radius_bottom: radius,
163                radius_top: radius,
164                height,
165                radial_segments: 12,
166                height_segments: 4,
167                caps: true,
168            }))
169        }
170        "tapered_cylinder" => {
171            let radius_bottom = dimensions[0];
172            let radius_top = dimensions[1];
173            let height = dimensions[2];
174            Ok(tapered_cylinder(&TaperedCylinderParams {
175                radius_bottom,
176                radius_top,
177                height,
178                radial_segments: 12,
179                height_segments: 4,
180                caps: true,
181            }))
182        }
183        _ => Err(ProcGenError::InvalidParameter {
184            name: "body_part.shape".into(),
185            reason: format!(
186                "unknown shape '{shape}'; valid: ellipsoid, sphere, capsule, tapered_cylinder"
187            ),
188        }),
189    }
190}

192/// Transform a point by a column-major 4x4 matrix.
193pub(crate) fn transform_point(m: &[f32; 16], p: &[f32; 3]) -> [f32; 3] {
194    [
195        m[0] * p[0] + m[4] * p[1] + m[8] * p[2] + m[12],
196        m[1] * p[0] + m[5] * p[1] + m[9] * p[2] + m[13],
197        m[2] * p[0] + m[6] * p[1] + m[10] * p[2] + m[14],
198    ]
199}
200
201/// Transform a normal by a column-major 4x4 matrix (using upper-left 3x3, then normalize).
202pub(crate) fn transform_normal(m: &[f32; 16], n: &[f32; 3]) -> [f32; 3] {
203    let x = m[0] * n[0] + m[4] * n[1] + m[8] * n[2];
204    let y = m[1] * n[0] + m[5] * n[1] + m[9] * n[2];
205    let z = m[2] * n[0] + m[6] * n[1] + m[10] * n[2];
206    let len = (x * x + y * y + z * z).sqrt();
207    if len > 1e-10 {
208        [x / len, y / len, z / len]
209    } else {
210        [0.0, 1.0, 0.0]
211    }
212}

214#[cfg(test)]
215mod tests {
216    use super::*;
217    use crate::skinning::BoneDef;
218
219    #[test]
220    fn generate_ellipsoid_body_part() {
221        let skeleton = SkeletonDef {
222            bones: vec![BoneDef {
223                name: "root".into(),
224                parent: None,
225                rest_translation: [0.0, 0.0, 0.0],
226                rest_rotation: [0.0, 0.0, 0.0, 1.0],
227                rest_scale: [1.0, 1.0, 1.0],
228            }],
229        };
230        let world_transforms = skeleton.compute_world_transforms();
231
232        let part = BodyPartDef {
233            bone: "root".into(),
234            shape: "ellipsoid".into(),
235            dimensions: [0.5, 0.3, 0.4],
236            offset: [0.0, 0.0, 0.0],
237            material: "test".into(),
238        };
239
240        let (verts, indices, mat) = generate_body_part(&part, &skeleton, &world_transforms).unwrap();
241        assert!(!verts.is_empty());
242        assert!(!indices.is_empty());
243        assert_eq!(mat, "test");
244
245        // All vertices should be skinned to bone 0
246        for v in &verts {
247            assert_eq!(v.joint_indices[0], 0);
248            assert_eq!(v.joint_weights[0], 1.0);
249        }
250    }
