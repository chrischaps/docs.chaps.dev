31/// Build a skeleton from TOML bone definitions and limb chains.
32///
33/// Returns the skeleton and a mapping from limb chain info for geometry generation.
34pub fn build_skeleton(
35    bone_defs: &[toml::Value],
36    limb_chains: &[LimbChainDef],
37) -> Result<(SkeletonDef, Vec<LimbChainBones>)> {
38    let mut bones = Vec::new();
39    let mut name_to_index: HashMap<String, usize> = HashMap::new();
40
41    // 1. Add explicit bones from the bone table
42    for (i, def) in bone_defs.iter().enumerate() {
43        let table = def
44            .as_table()
45            .ok_or_else(|| ProcGenError::InvalidParameter {
46                name: format!("bones[{i}]"),
47                reason: "expected a table".into(),
48            })?;
49
50        let name = table
51            .get("name")
52            .ok_or_else(|| ProcGenError::MissingParameter(format!("bones[{i}].name")))?;
53        let name = toml_string(name, &format!("bones[{i}].name"))?;
54
55        let parent = if let Some(parent_val) = table.get("parent") {
56            let parent_name = toml_string(parent_val, &format!("bones[{i}].parent"))?;
57            let idx =
58                name_to_index
59                    .get(&parent_name)
60                    .ok_or_else(|| ProcGenError::InvalidParameter {
61                        name: format!("bones[{i}].parent"),
62                        reason: format!(
63                            "bone '{parent_name}' not found (must be defined before use)"
64                        ),
65                    })?;
66            Some(*idx)
67        } else {
68            None
69        };
70
71        let position = parse_vec3_opt(table, "position", &format!("bones[{i}]"))?;
72        let rotation_deg = parse_vec3_opt(table, "rotation", &format!("bones[{i}]"))?;
73        let rotation_quat = euler_deg_to_quat(&rotation_deg);
74
75        let idx = bones.len();
76        name_to_index.insert(name.clone(), idx);
77        bones.push(BoneDef {
78            name,
79            parent,
80            rest_translation: position,
81            rest_rotation: rotation_quat,
82            rest_scale: [1.0, 1.0, 1.0],
83        });
84    }
85
86    // 2. Add bones from limb chains
87    let mut chain_bones = Vec::new();
88
89    for chain in limb_chains {
90        let sides = if chain.mirror {
91            vec![("_R", 1.0f32), ("_L", -1.0f32)]
92        } else {
93            vec![("", 1.0f32)]
94        };
95
96        for (suffix, mirror_sign) in &sides {
97            let parent_idx = name_to_index.get(&chain.parent_bone).ok_or_else(|| {
98                ProcGenError::InvalidParameter {
99                    name: format!("limb_chain.{}.parent_bone", chain.name),
100                    reason: format!("bone '{}' not found", chain.parent_bone),
101                }
102            })?;
103
104            let mut segment_bone_indices = Vec::new();
105            let mut current_parent = *parent_idx;
106
107            // First bone: at attach offset
108            let attach = [
109                chain.attach_offset[0] * mirror_sign,
110                chain.attach_offset[1],
111                chain.attach_offset[2],
112            ];
113
114            for (seg_i, seg) in chain.segments.iter().enumerate() {
115                let bone_name = format!("{}{suffix}_{seg_i}", chain.name);
116                let translation = if seg_i == 0 {
117                    attach
118                } else {
119                    // Each subsequent bone is offset along the local -Y axis (length of previous segment)
120                    [0.0, -chain.segments[seg_i - 1].length, 0.0]
121                };
122
123                // Mirror: flip the yaw and roll for mirrored side
124                let angles = if *mirror_sign < 0.0 {
125                    [seg.rest_angles[0], -seg.rest_angles[1], -seg.rest_angles[2]]
126                } else {
127                    seg.rest_angles
128                };
129                let rotation = euler_deg_to_quat(&angles);
130
131                let idx = bones.len();
132                name_to_index.insert(bone_name.clone(), idx);
133                bones.push(BoneDef {
134                    name: bone_name,
135                    parent: Some(current_parent),
136                    rest_translation: translation,
137                    rest_rotation: rotation,
138                    rest_scale: [1.0, 1.0, 1.0],
139                });
140                segment_bone_indices.push(idx);
141                current_parent = idx;
142            }
143
144            chain_bones.push(LimbChainBones {
145                chain_name: format!("{}{suffix}", chain.name),
146                bone_indices: segment_bone_indices,
147                segments: chain.segments.clone(),
148                material: chain.material.clone(),
149            });
150        }
151    }
152
153    Ok((SkeletonDef { bones }, chain_bones))
154}

165/// Parse a limb chain definition from a TOML table.
166pub fn parse_limb_chain(table: &toml::value::Table, idx: usize) -> Result<LimbChainDef> {
167    let prefix = format!("limb_chains[{idx}]");
168
169    let name = table
170        .get("name")
171        .ok_or_else(|| ProcGenError::MissingParameter(format!("{prefix}.name")))?;
172    let name = toml_string(name, &format!("{prefix}.name"))?;
173
174    let parent_bone = table
175        .get("parent_bone")
176        .ok_or_else(|| ProcGenError::MissingParameter(format!("{prefix}.parent_bone")))?;
177    let parent_bone = toml_string(parent_bone, &format!("{prefix}.parent_bone"))?;
178
179    let attach_offset = if let Some(v) = table.get("attach_offset") {
180        parse_f32_array3(v, &format!("{prefix}.attach_offset"))?
181    } else {
182        [0.0, 0.0, 0.0]
183    };
184
185    let mirror = table
186        .get("mirror")
187        .and_then(|v| v.as_bool())
188        .unwrap_or(false);
189
190    let material = table
191        .get("material")
192        .and_then(|v| v.as_str())
193        .unwrap_or("default")
194        .to_string();
195
196    let segments_val = table
197        .get("segments")
198        .ok_or_else(|| ProcGenError::MissingParameter(format!("{prefix}.segments")))?;
199    let segments_arr = segments_val
200        .as_array()
201        .ok_or_else(|| ProcGenError::InvalidParameter {
202            name: format!("{prefix}.segments"),
203            reason: "expected an array".into(),
204        })?;
205
206    let mut segments = Vec::new();
207    for (si, seg_val) in segments_arr.iter().enumerate() {
208        let seg_table = seg_val
209            .as_table()
210            .ok_or_else(|| ProcGenError::InvalidParameter {
211                name: format!("{prefix}.segments[{si}]"),
212                reason: "expected a table".into(),
213            })?;
214
215        let length = seg_table
216            .get("length")
217            .ok_or_else(|| {
218                ProcGenError::MissingParameter(format!("{prefix}.segments[{si}].length"))
219            })
220            .and_then(|v| toml_f32(v, &format!("{prefix}.segments[{si}].length")))?;
221
222        let radius_start = seg_table
223            .get("radius_start")
224            .ok_or_else(|| {
225                ProcGenError::MissingParameter(format!("{prefix}.segments[{si}].radius_start"))
226            })
227            .and_then(|v| toml_f32(v, &format!("{prefix}.segments[{si}].radius_start")))?;
228
229        let radius_end = seg_table
230            .get("radius_end")
231            .ok_or_else(|| {
232                ProcGenError::MissingParameter(format!("{prefix}.segments[{si}].radius_end"))
233            })
234            .and_then(|v| toml_f32(v, &format!("{prefix}.segments[{si}].radius_end")))?;
235
236        let rest_angles = if let Some(v) = seg_table.get("rest_angles") {
237            parse_f32_array3(v, &format!("{prefix}.segments[{si}].rest_angles"))?
238        } else {
239            [0.0, 0.0, 0.0]
240        };
241
242        segments.push(LimbSegmentDef {
243            length,
244            radius_start,
245            radius_end,
246            rest_angles,
247        });
248    }
249
250    Ok(LimbChainDef {
251        name,
252        parent_bone,
253        attach_offset,
254        mirror,
255        segments,
256        material,
257    })
258}

262fn parse_vec3_opt(table: &toml::value::Table, key: &str, prefix: &str) -> Result<[f32; 3]> {
263    if let Some(v) = table.get(key) {
264        parse_f32_array3(v, &format!("{prefix}.{key}"))
265    } else {
266        Ok([0.0, 0.0, 0.0])
267    }
268}
269
270fn parse_f32_array3(v: &toml::Value, name: &str) -> Result<[f32; 3]> {
271    let arr = v.as_array().ok_or_else(|| ProcGenError::InvalidParameter {
272        name: name.into(),
273        reason: "expected an array of 3 numbers".into(),
274    })?;
275    if arr.len() != 3 {
276        return Err(ProcGenError::InvalidParameter {
277            name: name.into(),
278            reason: format!("expected 3 elements, got {}", arr.len()),
279        });
280    }
281    Ok([
282        toml_f32(&arr[0], name)?,
283        toml_f32(&arr[1], name)?,
284        toml_f32(&arr[2], name)?,
285    ])
286}

288/// Convert Euler angles (degrees, pitch/yaw/roll = XYZ) to a quaternion [x, y, z, w].
289pub fn euler_deg_to_quat(angles_deg: &[f32; 3]) -> [f32; 4] {
290    let [pitch, yaw, roll] = *angles_deg;
291    let (px, py, pz) = (
292        pitch.to_radians() * 0.5,
293        yaw.to_radians() * 0.5,
294        roll.to_radians() * 0.5,
295    );
296
297    let (sp, cp) = px.sin_cos();
298    let (sy, cy) = py.sin_cos();
299    let (sr, cr) = pz.sin_cos();
300
301    // XYZ rotation order
302    let x = sp * cy * cr + cp * sy * sr;
303    let y = cp * sy * cr - sp * cy * sr;
304    let z = cp * cy * sr + sp * sy * cr;
305    let w = cp * cy * cr - sp * sy * sr;
306
307    [x, y, z, w]
308}
