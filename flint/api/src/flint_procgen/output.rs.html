1//! Generator output types.
2//!
3//! A generator produces a [`GeneratorOutput`] which wraps the concrete data
4//! (mesh, image, or raw audio bytes) along with accessors for type-safe
5//! extraction.
6
7use serde::{Deserialize, Serialize};
8
9use crate::skinning::SkinnedMeshData;
10use crate::types::{ImageData, MeshData};
11
12/// The kind of asset a generator produces.
13#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
14pub enum OutputKind {
15    /// 3D mesh geometry with materials.
16    Mesh,
17    /// 2D image / texture.
18    Image,
19    /// Raw audio data.
20    Sound,
21}
22
23/// The concrete output of a procedural generator.
24#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
25pub enum GeneratorOutput {
26    /// Generated 3D mesh.
27    Mesh(MeshData),
28    /// Generated 3D mesh with multiple LOD levels (index 0 = highest detail).
29    MeshWithLods(Vec<MeshData>),
30    /// Generated image / texture.
31    Image(ImageData),
32    /// Multiple related images (e.g. albedo + normal + roughness maps).
33    ImageSet(Vec<ImageData>),
34    /// Generated skinned mesh with skeleton (single LOD).
35    SkinnedMesh(SkinnedMeshData),
36    /// Generated audio (raw bytes, format TBD).
37    Sound(Vec<u8>),
38}
39
40impl GeneratorOutput {
41    /// Estimated heap memory usage in bytes.
42    pub fn estimated_bytes(&self) -> usize {
43        match self {
44            Self::Mesh(m) => m.estimated_bytes(),
45            Self::MeshWithLods(lods) => lods.iter().map(|m| m.estimated_bytes()).sum(),
46            Self::Image(i) => i.estimated_bytes(),
47            Self::ImageSet(images) => images.iter().map(|i| i.estimated_bytes()).sum(),
48            Self::SkinnedMesh(m) => m.estimated_bytes(),
49            Self::Sound(bytes) => bytes.len(),
50        }
51    }
52
53    /// Which kind of output this is.
54    pub fn kind(&self) -> OutputKind {
55        match self {
56            Self::Mesh(_) | Self::MeshWithLods(_) | Self::SkinnedMesh(_) => OutputKind::Mesh,
57            Self::Image(_) | Self::ImageSet(_) => OutputKind::Image,
58            Self::Sound(_) => OutputKind::Sound,
59        }
60    }
61
62    /// Borrow the mesh data, if this is a `Mesh` variant.
63    pub fn as_mesh(&self) -> Option<&MeshData> {
64        match self {
65            Self::Mesh(m) => Some(m),
66            _ => None,
67        }
68    }
69
70    /// Borrow the LOD mesh slices, if this is a `MeshWithLods` variant.
71    pub fn as_mesh_lods(&self) -> Option<&[MeshData]> {
72        match self {
73            Self::MeshWithLods(lods) => Some(lods),
74            _ => None,
75        }
76    }
77
78    /// Borrow the image data, if this is an `Image` variant.
79    pub fn as_image(&self) -> Option<&ImageData> {
80        match self {
81            Self::Image(i) => Some(i),
82            _ => None,
83        }
84    }
85
86    /// Consume and return the mesh data, if this is a `Mesh` variant.
87    pub fn into_mesh(self) -> Option<MeshData> {
88        match self {
89            Self::Mesh(m) => Some(m),
90            _ => None,
91        }
92    }
93
94    /// Consume and return the LOD meshes, if this is a `MeshWithLods` variant.
95    pub fn into_mesh_lods(self) -> Option<Vec<MeshData>> {
96        match self {
97            Self::MeshWithLods(lods) => Some(lods),
98            _ => None,
99        }
100    }
101
102    /// Consume and return the image data, if this is an `Image` variant.
103    pub fn into_image(self) -> Option<ImageData> {
104        match self {
105            Self::Image(i) => Some(i),
106            _ => None,
107        }
108    }
109
110    /// Borrow the image set, if this is an `ImageSet` variant.
111    pub fn as_image_set(&self) -> Option<&[ImageData]> {
112        match self {
113            Self::ImageSet(images) => Some(images),
114            _ => None,
115        }
116    }
117
118    /// Consume and return the image set, if this is an `ImageSet` variant.
119    pub fn into_image_set(self) -> Option<Vec<ImageData>> {
120        match self {
121            Self::ImageSet(images) => Some(images),
122            _ => None,
123        }
124    }
125
126    /// Borrow the skinned mesh data, if this is a `SkinnedMesh` variant.
127    pub fn as_skinned_mesh(&self) -> Option<&SkinnedMeshData> {
128        match self {
129            Self::SkinnedMesh(m) => Some(m),
130            _ => None,
131        }
132    }
133
134    /// Consume and return the skinned mesh data, if this is a `SkinnedMesh` variant.
135    pub fn into_skinned_mesh(self) -> Option<SkinnedMeshData> {
136        match self {
137            Self::SkinnedMesh(m) => Some(m),
138            _ => None,
139        }
140    }
141}

143#[cfg(test)]
144mod tests {
145    use super::*;
146    use crate::types::*;
147    use flint_core::Vec3;
148
149    fn sample_mesh() -> MeshData {
150        MeshData {
151            vertices: vec![Vertex {
152                position: [0.0, 0.0, 0.0],
153                normal: [0.0, 1.0, 0.0],
154                tangent: [1.0, 0.0, 0.0, 1.0],
155                uv: [0.0, 0.0],
156            }],
157            indices: vec![],
158            materials: vec![],
159            submeshes: vec![],
160            bounding_box: BoundingBox {
161                min: Vec3::ZERO,
162                max: Vec3::ZERO,
163            },
164        }
165    }
166
167    #[test]
168    fn output_kind_mesh() {
169        let out = GeneratorOutput::Mesh(sample_mesh());
170        assert_eq!(out.kind(), OutputKind::Mesh);
171    }
172
173    #[test]
174    fn output_kind_image() {
175        let img = ImageData::new(2, 2, ChannelSemantics::Color);
176        let out = GeneratorOutput::Image(img);
177        assert_eq!(out.kind(), OutputKind::Image);
178    }
179
180    #[test]
181    fn output_kind_sound() {
182        let out = GeneratorOutput::Sound(vec![0u8; 16]);
183        assert_eq!(out.kind(), OutputKind::Sound);
184    }
185
186    #[test]
187    fn as_mesh_returns_some_for_mesh() {
188        let out = GeneratorOutput::Mesh(sample_mesh());
189        assert!(out.as_mesh().is_some());
190        assert!(out.as_image().is_none());
191    }
192
193    #[test]
194    fn into_image_returns_some_for_image() {
195        let img = ImageData::new(4, 4, ChannelSemantics::Normal);
196        let out = GeneratorOutput::Image(img);
197        assert!(out.into_image().is_some());
198    }
199
200    #[test]
201    fn output_kind_mesh_with_lods() {
202        let out = GeneratorOutput::MeshWithLods(vec![sample_mesh(), sample_mesh()]);
203        assert_eq!(out.kind(), OutputKind::Mesh);
204    }
205
206    #[test]
207    fn as_mesh_lods_returns_some() {
208        let out = GeneratorOutput::MeshWithLods(vec![sample_mesh()]);
209        assert!(out.as_mesh_lods().is_some());
210        assert_eq!(out.as_mesh_lods().unwrap().len(), 1);
211        assert!(out.as_mesh().is_none());
212    }

214    #[test]

216        let out = GeneratorOutput::Mesh(sample_mesh());
217        assert!(out.estimated_bytes() > 0);
218    }

221    fn estimated_bytes_image() {
222        let img = ImageData::new(4, 4, ChannelSemantics::Color);
223        let out = GeneratorOutput::Image(img);
224        assert_eq!(out.estimated_bytes(), 4 * 4 * 4);
225    }
226
227    #[test]
228    fn estimated_bytes_sound() {
229        let out = GeneratorOutput::Sound(vec![0u8; 100]);
230        assert_eq!(out.estimated_bytes(), 100);
231    }
232
233    #[test]
234    fn estimated_bytes_mesh_with_lods() {
235        let out = GeneratorOutput::MeshWithLods(vec![sample_mesh(), sample_mesh()]);
236        let single = GeneratorOutput::Mesh(sample_mesh()).estimated_bytes();
237        assert_eq!(out.estimated_bytes(), single * 2);
238    }
239
240    #[test]
241    fn estimated_bytes_image_set() {
242        let img = ImageData::new(2, 2, ChannelSemantics::Color);
243        let single = img.estimated_bytes();
244        let out = GeneratorOutput::ImageSet(vec![
245            ImageData::new(2, 2, ChannelSemantics::Color),
246            ImageData::new(2, 2, ChannelSemantics::Normal),
247        ]);

249    }

253        let out = GeneratorOutput::MeshWithLods(vec![sample_mesh(), sample_mesh()]);
254        let lods = out.into_mesh_lods().unwrap();
255        assert_eq!(lods.len(), 2);
256    }
