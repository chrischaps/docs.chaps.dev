1//! Runtime resolver — discovers procgen specs and generates assets on demand.
2//!
3//! The [`ProcGenResolver`] indexes `*.procgen.toml` files by their `meta.name`
4//! and provides cached generation through the [`ProcGenCache`].
5
6use std::collections::HashMap;
7use std::path::Path;
8
9use crate::cache::{ProcGenCache, ProcGenCacheConfig};
10use crate::output::GeneratorOutput;
11use crate::queue::{AssetKind, CompletedAsset, GenerationRequest, ProcGenQueue, QueueStats};
12use crate::registry::GeneratorRegistry;
13use crate::spec::ProcGenSpec;
14use crate::{register_built_in_generators, CacheStats, ProcGenError, Result};
15use flint_core::ContentHash;
16
17/// Discovers procgen specs by name and generates assets with caching.
18pub struct ProcGenResolver {
19    /// Indexed specs: `meta.name` → spec
20    specs: HashMap<String, ProcGenSpec>,
21    /// Generator registry with all built-in generators
22    registry: GeneratorRegistry,
23    /// In-memory LRU cache for generated outputs
24    cache: ProcGenCache,
25    /// Frame-budgeted generation queue
26    queue: ProcGenQueue,
27}
28
29impl ProcGenResolver {
30    /// Create a new resolver with default cache config and built-in generators.
31    pub fn new() -> Self {
32        let config = ProcGenCacheConfig::default();
33        let budget_ms = config.runtime_max_generation_ms_per_frame;
34        let mut registry = GeneratorRegistry::new();
35        register_built_in_generators(&mut registry);
36        // Also register mock generator for testing/validation
37        registry.register(Box::new(crate::mock::MockGenerator));
38        Self {
39            specs: HashMap::new(),
40            registry,
41            cache: ProcGenCache::new(config),
42            queue: ProcGenQueue::new(budget_ms),
43        }
44    }
45
46    /// Scan directories for `*.procgen.toml` files and index by `meta.name`.
47    ///
48    /// Later directories override earlier ones when specs share the same name.
49    pub fn discover_and_index(&mut self, dirs: &[&Path]) {
50        for dir in dirs {
51            if let Ok(found) = crate::spec::discover_specs(dir) {
52                for spec in found {
53                    self.specs.insert(spec.meta.name.clone(), spec);
54                }
55            }
56        }
57    }
58
59    /// Check whether a spec with the given name is indexed.
60    pub fn has_spec(&self, name: &str) -> bool {
61        self.specs.contains_key(name)
62    }
63
64    /// Number of indexed specs.
65    pub fn spec_count(&self) -> usize {
66        self.specs.len()
67    }
68
69    /// Generate (or retrieve from cache) the output for the named spec.
70    pub fn generate(&mut self, name: &str) -> Result<GeneratorOutput> {
71        let spec = self
72            .specs
73            .get(name)
74            .ok_or_else(|| ProcGenError::UnknownGenerator(format!("no spec named '{name}'")))?
75            .clone();
76
77        let seed = spec.seed.to_seed()?;
78        let content_hash = ContentHash::from_str(&toml::to_string(&spec).unwrap_or_default());
79
80        // Check cache first
81        if let Some(cached) = self.cache.get(content_hash, &seed) {
82            return Ok(cached.clone());
83        }
84
85        // Generate
86        let output = self.registry.generate_from_spec(&spec)?;
87
88        // Cache the result
89        self.cache.insert(content_hash, &seed, output.clone());
90
91        Ok(output)
92    }
93
94    /// Current cache statistics.
95    pub fn cache_stats(&self) -> CacheStats {
96        self.cache.stats()
97    }
98
99    /// Enqueue a named spec for frame-budgeted generation.
100    ///
101    /// Returns `true` if the spec was enqueued (not already cached or pending).
102    /// Returns `false` if the spec doesn't exist, is already cached, or is
103    /// already pending.
104    pub fn enqueue(
105        &mut self,
106        name: &str,
107        entity_position: [f32; 3],
108        asset_kind: AssetKind,
109    ) -> bool {
110        let spec = match self.specs.get(name) {
111            Some(s) => s.clone(),
112            None => return false,
113        };
114
115        let seed = match spec.seed.to_seed() {
116            Ok(s) => s,
117            Err(_) => return false,
118        };
119        let content_hash = ContentHash::from_str(&toml::to_string(&spec).unwrap_or_default());
120
121        // Already cached — no need to queue
122        if self.cache.get(content_hash, &seed).is_some() {
123            return false;
124        }
125
126        // Already pending
127        if self.queue.is_pending(name) {
128            return false;
129        }
130
131        let dx = entity_position[0];
132        let dy = entity_position[1];
133        let dz = entity_position[2];
134        let priority = dx * dx + dy * dy + dz * dz;
135
136        self.queue.enqueue(GenerationRequest {
137            spec_name: name.to_string(),
138            spec,
139            content_hash,
140            seed,
141            priority,
142            entity_position,
143            asset_kind,
144        });
145        true
146    }
147
148    /// Process the generation queue for one frame within the time budget.
149    pub fn process_frame(&mut self, camera_pos: [f32; 3]) -> Vec<CompletedAsset> {
150        self.queue
151            .process_frame(camera_pos, &self.registry, &mut self.cache)
152    }
153
154    /// Check if a spec name is pending in the queue.
155    pub fn is_pending(&self, name: &str) -> bool {
156        self.queue.is_pending(name)
157    }
158
159    /// Check if a spec name has a cached output.
160    pub fn is_cached(&mut self, name: &str) -> bool {
161        let spec = match self.specs.get(name) {
162            Some(s) => s,
163            None => return false,
164        };
165        let seed = match spec.seed.to_seed() {
166            Ok(s) => s,
167            Err(_) => return false,
168        };
169        let content_hash = ContentHash::from_str(&toml::to_string(spec).unwrap_or_default());
170        self.cache.get(content_hash, &seed).is_some()
171    }
172
173    /// Current queue statistics.
174    pub fn queue_stats(&self) -> QueueStats {
175        self.queue.stats()
176    }
177
178    /// Clear all pending items from the queue.
179    pub fn clear_queue(&mut self) {
180        self.queue.clear();
181    }
182}

207    fn temp_dir(suffix: &str) -> std::path::PathBuf {
208        let dir = std::env::temp_dir().join(format!("flint_resolver_test_{suffix}"));
209        let _ = fs::remove_dir_all(&dir);
210        fs::create_dir_all(&dir).unwrap();
211        dir
212    }

214    #[test]
215    fn discover_specs_by_name() {
216        let dir = temp_dir("discover");
217        fs::write(dir.join("cube.procgen.toml"), MOCK_SPEC).unwrap();
218
219        let spec2 = MOCK_SPEC.replace("test_cube", "test_sphere");
220        fs::write(dir.join("sphere.procgen.toml"), &spec2).unwrap();
221
222        // Non-spec file should be ignored
223        fs::write(dir.join("readme.txt"), "hello").unwrap();
224
225        let mut resolver = ProcGenResolver::new();
226        resolver.discover_and_index(&[dir.as_path()]);
227
228        assert_eq!(resolver.spec_count(), 2);
229        assert!(resolver.has_spec("test_cube"));
230        assert!(resolver.has_spec("test_sphere"));
231        assert!(!resolver.has_spec("nonexistent"));
232
233        let _ = fs::remove_dir_all(&dir);
234    }
235
236    #[test]
237    fn generate_and_cache() {
238        let dir = temp_dir("gen_cache");
239        fs::write(dir.join("cube.procgen.toml"), MOCK_SPEC).unwrap();
240
241        let mut resolver = ProcGenResolver::new();
242        resolver.discover_and_index(&[dir.as_path()]);
243
244        // First generation — cache miss
245        let output1 = resolver.generate("test_cube").unwrap();
246        assert_eq!(resolver.cache_stats().misses, 1);
247        assert_eq!(resolver.cache_stats().hits, 0);
248
249        // Second generation — cache hit
250        let output2 = resolver.generate("test_cube").unwrap();
251        assert_eq!(resolver.cache_stats().hits, 1);
252        assert_eq!(output1, output2);
253
254        let _ = fs::remove_dir_all(&dir);
255    }

263    #[test]
264    fn later_dir_overrides_earlier() {
265        let dir_a = temp_dir("override_a");
266        let dir_b = temp_dir("override_b");
267
268        // Both dirs have a spec named "shared_name" but with different seeds
269        let spec_a = r#"
270generator = "mock"
271[meta]
272name = "shared_name"
273version = "1.0.0"
274[seed]
275mode = "fixed"
276value = 1
277"#;
278        let spec_b = r#"
279generator = "mock"
280[meta]
281name = "shared_name"
282version = "2.0.0"
283[seed]
284mode = "fixed"
285value = 999
286"#;
287        fs::write(dir_a.join("a.procgen.toml"), spec_a).unwrap();
288        fs::write(dir_b.join("b.procgen.toml"), spec_b).unwrap();
289
290        let mut resolver = ProcGenResolver::new();
291        resolver.discover_and_index(&[dir_a.as_path(), dir_b.as_path()]);
292
293        assert_eq!(resolver.spec_count(), 1);
294        assert!(resolver.has_spec("shared_name"));
295
296        // The later dir (dir_b) should win — version "2.0.0"
297        let spec = resolver.specs.get("shared_name").unwrap();
298        assert_eq!(spec.meta.version, "2.0.0");
299
300        let _ = fs::remove_dir_all(&dir_a);
301        let _ = fs::remove_dir_all(&dir_b);
302    }
