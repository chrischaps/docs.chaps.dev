1//! Generator registry — maps type names to generator implementations.
2//!
3//! The registry is the central dispatch point: given a [`ProcGenSpec`], it
4//! resolves the generator by name, creates a [`SeededRng`], and delegates
5//! to [`Generator::generate()`].
6
7use std::collections::HashMap;
8
9use crate::generator::Generator;
10use crate::output::GeneratorOutput;
11use crate::rng::SeededRng;
12use crate::spec::ProcGenSpec;
13use crate::{ProcGenError, Result};
14
15/// A registry of procedural generators, keyed by type name.
16pub struct GeneratorRegistry {
17    generators: HashMap<String, Box<dyn Generator>>,
18}
19
20impl GeneratorRegistry {
21    /// Create an empty registry.
22    pub fn new() -> Self {
23        Self {
24            generators: HashMap::new(),
25        }
26    }
27
28    /// Register a generator. Replaces any existing generator with the same
29    /// type name.
30    pub fn register(&mut self, generator: Box<dyn Generator>) {
31        let name = generator.type_name().to_string();
32        self.generators.insert(name, generator);
33    }
34
35    /// Look up a generator by type name.
36    pub fn get(&self, type_name: &str) -> Option<&dyn Generator> {
37        self.generators.get(type_name).map(|g| g.as_ref())
38    }
39
40    /// All registered generator type names.
41    pub fn generator_names(&self) -> Vec<&str> {
42        self.generators.keys().map(|k| k.as_str()).collect()
43    }
44
45    /// A Markdown parameter reference for every registered generator, in
46    /// name order: a `###` heading with the output kind, then
47    /// [`crate::param_reference_markdown`] of its schema. The crate README's
48    /// catalog is this for the built-ins (`cargo run -p flint-procgen
49    /// --example param_catalog` prints it).
50    pub fn reference_markdown(&self) -> String {
51        let mut names = self.generator_names();
52        names.sort_unstable();
53        let mut out = String::new();
54        for name in names {
55            let generator = self.get(name).unwrap();
56            let kind = match generator.output_kind() {
57                crate::OutputKind::Mesh => "mesh, written as GLB",
58                crate::OutputKind::Image => "image set, written as PNGs",
59                crate::OutputKind::Sound => "sound, written as WAV",
60            };
61            out.push_str(&format!("### `{name}` ({kind})\n\n"));
62            out.push_str(&crate::param_reference_markdown(&generator.param_schema()));
63            out.push('\n');
64        }
65        out
66    }
67
68    /// Resolve the generator from a spec, create a seeded RNG, and run
69    /// generation.
70    pub fn generate_from_spec(&self, spec: &ProcGenSpec) -> Result<GeneratorOutput> {
71        let generator = self
72            .generators
73            .get(&spec.generator)
74            .ok_or_else(|| ProcGenError::UnknownGenerator(spec.generator.clone()))?;
75
76        let seed = spec.resolve_seed()?;
77        let mut rng = SeededRng::new(seed);
78
79        generator.generate(spec, &mut rng)
80    }
81}
82
83impl Default for GeneratorRegistry {
84    fn default() -> Self {
85        Self::new()
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92    use crate::mock::MockGenerator;
93    use crate::output::OutputKind;
94
95    fn make_registry() -> GeneratorRegistry {
96        let mut reg = GeneratorRegistry::new();
97        reg.register(Box::new(MockGenerator));
98        reg
99    }
100
101    fn mock_spec() -> ProcGenSpec {
102        ProcGenSpec::parse(
103            r#"
104generator = "mock"
105[meta]
106name = "test"
107version = "1.0.0"
108[seed]
109mode = "fixed"
110value = 42
111"#,
112        )
113        .unwrap()
114    }
115
116    #[test]
117    fn register_and_get() {
118        let reg = make_registry();
119        assert!(reg.get("mock").is_some());
120        assert!(reg.get("nonexistent").is_none());
121    }
122
123    #[test]
124    fn generator_names() {
125        let reg = make_registry();
126        let names = reg.generator_names();
127        assert!(names.contains(&"mock"));
128    }
129
130    #[test]
131    fn generate_from_spec_success() {
132        let reg = make_registry();
133        let spec = mock_spec();
134        let output = reg.generate_from_spec(&spec).unwrap();
135        assert_eq!(output.kind(), OutputKind::Mesh);
136
137        let mesh = output.as_mesh().unwrap();
138        assert_eq!(mesh.vertex_count(), 24);
139        assert_eq!(mesh.triangle_count(), 12);
140        assert!(mesh.validate().is_ok());
141    }
142
143    #[test]
144    fn generate_from_spec_unknown_generator() {
145        let reg = make_registry();
146        let mut spec = mock_spec();
147        spec.generator = "nonexistent".into();
148        let err = reg.generate_from_spec(&spec).unwrap_err();
149        assert!(matches!(err, ProcGenError::UnknownGenerator(ref name) if name == "nonexistent"));
150    }
151
152    #[test]
153    fn generate_from_spec_deterministic() {
154        let reg = make_registry();
155        let spec = mock_spec();
156        let a = reg.generate_from_spec(&spec).unwrap();
157        let b = reg.generate_from_spec(&spec).unwrap();
158        assert_eq!(a, b);
159    }

161    #[test]
162    fn box_dyn_generator_compiles() {

164        let _: Box<dyn Generator> = Box::new(MockGenerator);
165    }

167    /// Every parameter a built-in generator takes says what it does: the
168    /// previewer shows it on hover and the crate README's catalog prints it.
169    #[test]
170    fn every_built_in_parameter_is_described() {
171        let mut reg = GeneratorRegistry::new();
172        crate::register_built_in_generators(&mut reg);
173        let mut missing = Vec::new();
174        for name in reg.generator_names() {
175            let schema = reg.get(name).unwrap().param_schema();
176            for (key, prop) in schema["properties"].as_object().unwrap() {
177                let text = prop["description"].as_str().unwrap_or("");
178                if text.trim().is_empty() {
179                    missing.push(format!("{name}.{key}"));
180                }
181            }
182        }
183        assert!(missing.is_empty(), "undescribed: {missing:?}");
184    }
185
186    #[test]
187    fn default_registry_is_empty() {
188        let reg = GeneratorRegistry::default();
189        assert!(reg.generator_names().is_empty());
190    }
