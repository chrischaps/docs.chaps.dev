1//! Generator registry — maps type names to generator implementations.
2//!
3//! The registry is the central dispatch point: given a [`ProcGenSpec`], it
4//! resolves the generator by name, creates a [`SeededRng`], and delegates
5//! to [`Generator::generate()`].
6
7use std::collections::HashMap;
8
9use crate::generator::Generator;
10use crate::output::GeneratorOutput;
11use crate::rng::SeededRng;
12use crate::spec::ProcGenSpec;
13use crate::{ProcGenError, Result};
14
15/// A registry of procedural generators, keyed by type name.
16pub struct GeneratorRegistry {
17    generators: HashMap<String, Box<dyn Generator>>,
18}
19
20impl GeneratorRegistry {
21    /// Create an empty registry.
22    pub fn new() -> Self {
23        Self {
24            generators: HashMap::new(),
25        }
26    }
27
28    /// Register a generator. Replaces any existing generator with the same
29    /// type name.
30    pub fn register(&mut self, generator: Box<dyn Generator>) {
31        let name = generator.type_name().to_string();
32        self.generators.insert(name, generator);
33    }
34
35    /// Look up a generator by type name.
36    pub fn get(&self, type_name: &str) -> Option<&dyn Generator> {
37        self.generators.get(type_name).map(|g| g.as_ref())
38    }
39
40    /// All registered generator type names.
41    pub fn generator_names(&self) -> Vec<&str> {
42        self.generators.keys().map(|k| k.as_str()).collect()
43    }
44
45    /// Resolve the generator from a spec, create a seeded RNG, and run
46    /// generation.
47    pub fn generate_from_spec(&self, spec: &ProcGenSpec) -> Result<GeneratorOutput> {
48        let generator = self
49            .generators
50            .get(&spec.generator)
51            .ok_or_else(|| ProcGenError::UnknownGenerator(spec.generator.clone()))?;
52
53        let seed = spec.resolve_seed()?;
54        let mut rng = SeededRng::new(seed);
55
56        generator.generate(spec, &mut rng)
57    }
58}
59
60impl Default for GeneratorRegistry {
61    fn default() -> Self {
62        Self::new()
63    }
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use crate::mock::MockGenerator;
70    use crate::output::OutputKind;
71
72    fn make_registry() -> GeneratorRegistry {
73        let mut reg = GeneratorRegistry::new();
74        reg.register(Box::new(MockGenerator));
75        reg
76    }
77
78    fn mock_spec() -> ProcGenSpec {
79        ProcGenSpec::parse(
80            r#"
81generator = "mock"
82[meta]
83name = "test"
84version = "1.0.0"
85[seed]
86mode = "fixed"
87value = 42
88"#,
89        )
90        .unwrap()
91    }
92
93    #[test]
94    fn register_and_get() {
95        let reg = make_registry();
96        assert!(reg.get("mock").is_some());
97        assert!(reg.get("nonexistent").is_none());
98    }
99
100    #[test]
101    fn generator_names() {
102        let reg = make_registry();
103        let names = reg.generator_names();
104        assert!(names.contains(&"mock"));
105    }
106
107    #[test]
108    fn generate_from_spec_success() {
109        let reg = make_registry();
110        let spec = mock_spec();
111        let output = reg.generate_from_spec(&spec).unwrap();
112        assert_eq!(output.kind(), OutputKind::Mesh);
113
114        let mesh = output.as_mesh().unwrap();
115        assert_eq!(mesh.vertex_count(), 24);
116        assert_eq!(mesh.triangle_count(), 12);
117        assert!(mesh.validate().is_ok());
118    }
119
120    #[test]
121    fn generate_from_spec_unknown_generator() {
122        let reg = make_registry();
123        let mut spec = mock_spec();
124        spec.generator = "nonexistent".into();
125        let err = reg.generate_from_spec(&spec).unwrap_err();
126        assert!(matches!(err, ProcGenError::UnknownGenerator(ref name) if name == "nonexistent"));
127    }
128
129    #[test]
130    fn generate_from_spec_deterministic() {
131        let reg = make_registry();
132        let spec = mock_spec();
133        let a = reg.generate_from_spec(&spec).unwrap();
134        let b = reg.generate_from_spec(&spec).unwrap();
135        assert_eq!(a, b);
136    }
