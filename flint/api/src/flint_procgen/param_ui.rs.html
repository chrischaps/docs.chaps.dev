1//! Schema-to-descriptor parser for procedural generator parameter UI.
2//!
3//! Reads a generator's `param_schema()` JSON Schema and produces a list of
4//! structured field descriptors (`ParamFieldSpec`) that a UI layer can consume
5//! directly — keeping JSON Schema interpretation out of the rendering code.
6
7use serde_json::Value;
8
9/// Describes a single parameter field for UI rendering.
10#[derive(Debug, Clone)]
11pub struct ParamFieldSpec {
12    /// The parameter name (matches the TOML key).
13    pub name: String,
14    /// The UI-appropriate type with constraints.
15    pub field_type: ParamFieldType,
16    /// The schema-declared default value, if any.
17    pub default: Option<Value>,
18    /// The schema's `description`: what the parameter does, in a sentence.
19    /// The previewer shows it on hover.
20    pub description: Option<String>,
21    /// When the field is read only for some values of another field
22    /// (`"x-when"` in the schema); `None` means always.
23    pub when: Option<ParamCondition>,
24}
25
26/// The widget type and constraints for a parameter field.
27#[derive(Debug, Clone)]
28pub enum ParamFieldType {
29    /// Floating-point slider/drag.
30    Float { min: Option<f64>, max: Option<f64> },
31    /// Integer slider/drag.
32    Integer { min: Option<i64>, max: Option<i64> },
33    /// Checkbox.
34    Bool,
35    /// Hex color string (e.g. "#573214").
36    HexColor,
37    /// Dropdown with fixed choices.
38    Enum { values: Vec<String> },
39    /// Free-form text input.
40    String,
41    /// Array of strings, optionally constrained to enum choices (multi-checkbox).
42    StringArray { item_enum: Option<Vec<String>> },
43    /// Array of objects (e.g. bones, body_parts, limb_chains).
44    ObjectArray,
45    /// A table (`"type": "object"`), usually of named tables, such as a
46    /// creature's `[params.materials.<name>]`. Shown, not edited.
47    ObjectMap,
48}

50/// Parse a generator's JSON Schema into a list of UI field descriptors.
51///
52/// Expects a schema of the form:
53/// ```json
54/// {
55///   "type": "object",
56///   "properties": {
57///     "field_name": { "type": "number", "minimum": 0.0, "default": 5.0 },
58///     ...
59///   }
60/// }
61/// ```
62pub fn parse_param_schema(schema: &Value) -> Vec<ParamFieldSpec> {
63    let properties = match schema.get("properties").and_then(|p| p.as_object()) {
64        Some(props) => props,
65        None => return Vec::new(),
66    };
67
68    let mut fields: Vec<ParamFieldSpec> = properties
69        .iter()
70        .map(|(name, prop)| {
71            let field_type = classify_field(prop);
72            let default = prop.get("default").cloned();
73            ParamFieldSpec {
74                name: name.clone(),
75                field_type,
76                default,
77                description: prop
78                    .get("description")
79                    .and_then(|d| d.as_str())
80                    .map(str::to_string),
81                when: parse_condition(prop),
82            }
83        })
84        .collect();
85
86    // Sort by name for stable ordering
87    fields.sort_by(|a, b| a.name.cmp(&b.name));
88    fields
89}

91/// Classify a single property schema into a `ParamFieldType`.
92fn classify_field(prop: &Value) -> ParamFieldType {
93    let type_str = prop.get("type").and_then(|t| t.as_str()).unwrap_or("");
94
95    match type_str {
96        "number" => {
97            let min = get_f64(prop, "minimum").or_else(|| get_f64(prop, "exclusiveMinimum"));
98            let max = get_f64(prop, "maximum").or_else(|| get_f64(prop, "exclusiveMaximum"));
99            ParamFieldType::Float { min, max }
100        }
101        "integer" => {
102            let min = get_i64(prop, "minimum").or_else(|| get_i64(prop, "exclusiveMinimum"));
103            let max = get_i64(prop, "maximum").or_else(|| get_i64(prop, "exclusiveMaximum"));
104            ParamFieldType::Integer { min, max }
105        }
106        "boolean" => ParamFieldType::Bool,
107        "string" => {
108            // Check for enum first (dropdown)
109            if let Some(values) = extract_string_enum(prop) {
110                return ParamFieldType::Enum { values };
111            }
112            // Check for hex color pattern
113            if let Some(pattern) = prop.get("pattern").and_then(|p| p.as_str()) {
114                if pattern.contains("[0-9a-fA-F]") {
115                    return ParamFieldType::HexColor;
116                }
117            }
118            ParamFieldType::String
119        }
120        "array" => {
121            // Check if items are objects (e.g. bones, body_parts, limb_chains)
122            if let Some(items) = prop.get("items") {
123                if items.get("type").and_then(|t| t.as_str()) == Some("object") {
124                    return ParamFieldType::ObjectArray;
125                }
126            }
127            let item_enum = prop.get("items").and_then(extract_string_enum);
128            ParamFieldType::StringArray { item_enum }
129        }
130        "object" => ParamFieldType::ObjectMap,
131        _ => ParamFieldType::String,
132    }
133}

135/// Extract an enum array of strings from a schema node.
136fn extract_string_enum(prop: &Value) -> Option<Vec<String>> {
137    prop.get("enum")
138        .and_then(|e| e.as_array())
139        .map(|arr| {
140            arr.iter()
141                .filter_map(|v| v.as_str().map(|s| s.to_string()))
142                .collect()
143        })
144        .filter(|v: &Vec<String>| !v.is_empty())
145}

147fn get_f64(prop: &Value, key: &str) -> Option<f64> {
148    prop.get(key).and_then(|v| v.as_f64())
149}
150
151fn get_i64(prop: &Value, key: &str) -> Option<i64> {
152    prop.get(key).and_then(|v| v.as_i64())
153}

157/// A field that only applies when another field holds one of some values,
158/// declared in the schema as `"x-when": { "pattern": ["voronoi_brick"] }`.
159///
160/// JSON Schema ignores keywords it doesn't know, so validators still accept
161/// the schema; the UI hides the field and [`lint_params`] reports it when the
162/// condition is false.
163#[derive(Debug, Clone, PartialEq)]
164pub struct ParamCondition {
165    /// The controlling field (e.g. `"pattern"`).
166    pub key: String,
167    /// The values of `key` under which this field is read.
168    pub any_of: Vec<String>,
169}
170
171fn parse_condition(prop: &Value) -> Option<ParamCondition> {
172    let when = prop.get("x-when")?.as_object()?;
173    let (key, values) = when.iter().next()?;
174    let any_of = values
175        .as_array()?
176        .iter()
177        .filter_map(|v| v.as_str().map(str::to_string))
178        .collect();
179    Some(ParamCondition {
180        key: key.clone(),
181        any_of,
182    })
183}

185/// The value a string field has in `params`, or its schema default.
186fn effective_string(
187    schema: &Value,
188    params: &toml::map::Map<String, toml::Value>,
189    key: &str,
190) -> Option<String> {
191    if let Some(v) = params.get(key) {
192        return v.as_str().map(str::to_string);
193    }
194    schema
195        .get("properties")?
196        .get(key)?
197        .get("default")?
198        .as_str()
199        .map(str::to_string)
200}

202impl ParamFieldSpec {
203    /// Whether this field is read, given the spec's current params. Fields
204    /// without an `x-when` condition always apply.
205    pub fn applies(&self, schema: &Value, params: &toml::map::Map<String, toml::Value>) -> bool {
206        match &self.when {
207            None => true,
208            Some(c) => {
209                effective_string(schema, params, &c.key).is_some_and(|v| c.any_of.contains(&v))
210            }
211        }
212    }
213}

215/// Something in a spec's `[params]` that the generator will not read.
216#[derive(Debug, Clone, PartialEq)]
217pub enum ParamIssue {
218    /// The key is not in the schema at all (usually a typo). `suggestion` is
219    /// the closest schema key, when one is close. `within` names the table
220    /// the key sits in (its schema's `title`, e.g. "brick_grid op") when that
221    /// isn't the generator's own params.
222    Unknown {
223        key: String,
224        suggestion: Option<String>,
225        within: Option<String>,
226    },
227    /// The key exists but only applies when `controller` is one of `needs`;
228    /// the spec has `controller = actual`, so the value is ignored.
229    Inapplicable {
230        key: String,
231        controller: String,
232        actual: String,
233        needs: Vec<String>,
234    },
235    /// The schema marks the key `"deprecated": true`: it is read (or once
236    /// was) but no longer does anything. `reason` is the key's description.
237    Deprecated { key: String, reason: Option<String> },
238}
239
240impl ParamIssue {
241    /// The params key the issue is about: a path such as `ops[2].colums`
242    /// when it sits inside a table or an array of them.
243    pub fn key(&self) -> &str {
244        match self {
245            Self::Unknown { key, .. }
246            | Self::Inapplicable { key, .. }
247            | Self::Deprecated { key, .. } => key,
248        }
249    }
250}
251
252impl std::fmt::Display for ParamIssue {
253    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
254        match self {
255            Self::Unknown {
256                key,
257                suggestion,
258                within,
259            } => {
260                match within {
261                    Some(w) => write!(f, "`{key}` is not a parameter of the {w}")?,
262                    None => write!(f, "`{key}` is not a parameter of this generator")?,
263                }
264                match suggestion {
265                    Some(s) => write!(f, " (did you mean `{s}`?)"),
266                    None => Ok(()),
267                }
268            }
269            Self::Inapplicable {
270                key,
271                controller,
272                actual,
273                needs,
274            } => write!(
275                f,
276                "`{key}` is ignored: it only applies when {controller} is {}, and this spec has {controller} = \"{actual}\"",
277                needs
278                    .iter()
279                    .map(|n| format!("\"{n}\""))
280                    .collect::<Vec<_>>()
281                    .join(" or ")
282            ),
283            Self::Deprecated {
284                key,
285                reason: Some(r),
286            } => write!(f, "`{key}` is deprecated: {r}"),
287            Self::Deprecated { key, .. } => write!(f, "`{key}` is deprecated"),
288        }
289    }
290}

292/// Report every key in `params` that the generator described by `schema`
293/// will silently ignore: keys missing from the schema (when it sets
294/// `"additionalProperties": false`), keys whose `x-when` condition the
295/// spec doesn't meet, and keys the schema marks `"deprecated": true`.
296/// Returned in key order; empty means every key is read.
297///
298/// Tables inside the params are checked against their own schemas, the
299/// issue's key then being a path (`materials.shell.rougness`, `ops[2].colums`):
300/// - a property with `"properties"` is checked like the params themselves;
301/// - a table of named tables (`"additionalProperties": { ... }`) checks
302///   every entry against that schema;
303/// - an array checks each element against `"items"`, and when `items` is a
304///   `"oneOf"` with a `"discriminator": { "propertyName": "type" }`, against
305///   the branch whose `type` property has the element's value as `const`.
306pub fn lint_params(schema: &Value, params: &toml::Value) -> Vec<ParamIssue> {
307    let mut issues = Vec::new();
308    lint_value(schema, params, "", &mut issues);
309    issues
310}
311
312/// Lint `value` against `schema`, prefixing issue keys with `path`.
313fn lint_value(schema: &Value, value: &toml::Value, path: &str, issues: &mut Vec<ParamIssue>) {
314    match value {
315        toml::Value::Table(table) => {
316            if schema.get("properties").is_some() {
317                lint_table(schema, table, path, issues);
318            } else if let Some(entry) = schema.get("additionalProperties").filter(|e| e.is_object())
319            {
320                let mut names: Vec<&String> = table.keys().collect();
321                names.sort();
322                for name in names {
323                    lint_value(entry, &table[name.as_str()], &join(path, name), issues);
324                }
325            }
326        }
327        toml::Value::Array(items) => {
328            let Some(item_schema) = schema.get("items") else {
329                return;
330            };
331            for (i, item) in items.iter().enumerate() {
332                if let Some(branch) = discriminated_branch(item_schema, item) {
333                    lint_value(branch, item, &format!("{path}[{i}]"), issues);
334                }
335            }
336        }
337        _ => {}
338    }
339}

341/// The schema an array element is checked against: `items` itself, or for a
342/// discriminated `oneOf` the branch matching the element's tag. `None` when
343/// no branch matches (the generator reports an unknown type itself).
344fn discriminated_branch<'a>(items: &'a Value, item: &toml::Value) -> Option<&'a Value> {
345    let Some(branches) = items.get("oneOf").and_then(|b| b.as_array()) else {
346        return Some(items);
347    };
348    let tag = items.get("discriminator")?.get("propertyName")?.as_str()?;
349    let wanted = item.get(tag)?.as_str()?;
350    branches.iter().find(|b| {
351        b.get("properties")
352            .and_then(|p| p.get(tag))
353            .and_then(|t| t.get("const"))
354            .and_then(|c| c.as_str())
355            == Some(wanted)
356    })
357}

359fn join(path: &str, key: &str) -> String {
360    if path.is_empty() {
361        key.to_string()
362    } else {
363        format!("{path}.{key}")
364    }
365}

367fn lint_table(
368    schema: &Value,
369    table: &toml::map::Map<String, toml::Value>,
370    path: &str,
371    issues: &mut Vec<ParamIssue>,
372) {
373    let properties = schema.get("properties").and_then(|p| p.as_object());
374    let closed = schema.get("additionalProperties") == Some(&Value::Bool(false));
375    let within = if path.is_empty() {
376        None
377    } else {
378        schema
379            .get("title")
380            .and_then(|t| t.as_str())
381            .map(str::to_string)
382    };
383
384    let mut keys: Vec<&String> = table.keys().collect();
385    keys.sort();
386    for key in keys {
387        match properties.and_then(|p| p.get(key.as_str())) {
388            None if closed => issues.push(ParamIssue::Unknown {
389                key: join(path, key),
390                suggestion: properties.and_then(|p| closest(key, p.keys())),
391                within: within.clone(),
392            }),
393            None => {}
394            Some(prop) => {
395                if prop.get("deprecated") == Some(&Value::Bool(true)) {
396                    issues.push(ParamIssue::Deprecated {
397                        key: join(path, key),
398                        reason: prop
399                            .get("description")
400                            .and_then(|d| d.as_str())
401                            .map(str::to_string),
402                    });
403                    continue;
404                }
405                if let Some(c) = parse_condition(prop) {
406                    let actual = effective_string(schema, table, &c.key).unwrap_or_default();
407                    if !c.any_of.contains(&actual) {
408                        issues.push(ParamIssue::Inapplicable {
409                            key: join(path, key),
410                            controller: c.key,
411                            actual,
412                            needs: c.any_of,
413                        });
414                        // Nothing inside an ignored key is read either.
415                        continue;
416                    }
417                }
418                lint_value(prop, &table[key.as_str()], &join(path, key), issues);
419            }
420        }
421    }
422}

424/// The candidate within edit distance 2 (or a third of the key's length,
425/// whichever is larger) of `key`, nearest first.
426fn closest<'a>(key: &str, candidates: impl Iterator<Item = &'a String>) -> Option<String> {
427    let limit = (key.chars().count() / 3).max(2);
428    candidates
429        .map(|c| (edit_distance(key, c), c))
430        .filter(|(d, _)| *d <= limit)
431        .min_by(|a, b| a.0.cmp(&b.0).then_with(|| a.1.cmp(b.1)))
432        .map(|(_, c)| c.clone())
433}
434
435fn edit_distance(a: &str, b: &str) -> usize {
436    let b: Vec<char> = b.chars().collect();
437    let mut prev: Vec<usize> = (0..=b.len()).collect();
438    for (i, ca) in a.chars().enumerate() {
439        let mut cur = vec![i + 1; b.len() + 1];
440        for (j, cb) in b.iter().enumerate() {
441            let sub = prev[j] + usize::from(ca != *cb);
442            cur[j + 1] = sub.min(prev[j + 1] + 1).min(cur[j] + 1);
443        }
444        prev = cur;
445    }
446    prev[b.len()]
447}

451/// A Markdown reference for one generator's `params`, built from its
452/// schema: a table per condition (keys every spec may set first, then one
453/// table per `x-when`), each row the key, its type and range, its default
454/// and its `description`. The crate README's catalog is this, for every
455/// built-in generator, and a test keeps the two equal.
456pub fn param_reference_markdown(schema: &Value) -> String {
457    let Some(props) = schema.get("properties").and_then(|p| p.as_object()) else {
458        return String::new();
459    };
460    // Group by condition; BTreeMap keeps the groups and their keys in order.
461    let mut groups: std::collections::BTreeMap<String, Vec<(&String, &Value)>> = Default::default();
462    for (key, prop) in props {
463        let heading = match parse_condition(prop) {
464            None => String::new(),
465            Some(c) => format!(
466                "Only when `{}` is {}",
467                c.key,
468                c.any_of
469                    .iter()
470                    .map(|v| format!("`{v}`"))
471                    .collect::<Vec<_>>()
472                    .join(" or ")
473            ),
474        };
475        groups.entry(heading).or_default().push((key, prop));
476    }
477
478    let mut out = String::new();
479    for (heading, rows) in &groups {
480        if !heading.is_empty() {
481            out.push_str(&format!("\n{heading}:\n\n"));
482        }
483        out.push_str("| Key | Type | Default | What it does |\n|---|---|---|---|\n");
484        for (key, prop) in rows {
485            let mut text = prop
486                .get("description")
487                .and_then(|d| d.as_str())
488                .unwrap_or("")
489                .to_string();
490            if prop.get("deprecated") == Some(&Value::Bool(true)) {
491                text = format!("**Deprecated.** {text}");
492            }
493            out.push_str(&format!(
494                "| `{key}` | {} | {} | {} |\n",
495                type_cell(prop),
496                prop.get("default").map(default_cell).unwrap_or_default(),
497                text.replace('|', "\\|")
498            ));
499        }
500    }
501    out
502}

504/// "number ≥ 0", "integer 1–8", "`a` \| `b`", "hex colour" and so on.
505fn type_cell(prop: &Value) -> String {
506    let num = |k: &str| prop.get(k).and_then(|v| v.as_f64());
507    let fmt = |v: f64| {
508        if v.fract() == 0.0 {
509            format!("{v:.0}")
510        } else {
511            format!("{v}")
512        }
513    };
514    let range = || match (num("minimum"), num("exclusiveMinimum"), num("maximum")) {
515        (Some(lo), _, Some(hi)) => format!(" {}–{}", fmt(lo), fmt(hi)),
516        (_, Some(lo), Some(hi)) => format!(" >{} to {}", fmt(lo), fmt(hi)),
517        (Some(lo), _, None) => format!(" ≥ {}", fmt(lo)),
518        (_, Some(lo), None) => format!(" > {}", fmt(lo)),
519        (None, None, Some(hi)) => format!(" ≤ {}", fmt(hi)),
520        _ => String::new(),
521    };
522    match classify_field(prop) {
523        ParamFieldType::Float { .. } => format!("number{}", range()),
524        ParamFieldType::Integer { .. } => format!("integer{}", range()),
525        ParamFieldType::Bool => "bool".into(),
526        ParamFieldType::HexColor => "hex colour".into(),
527        ParamFieldType::Enum { values } => values
528            .iter()
529            .map(|v| format!("`{v}`"))
530            .collect::<Vec<_>>()
531            .join(" \\| "),
532        ParamFieldType::String => "string".into(),
533        ParamFieldType::StringArray {
534            item_enum: Some(values),
535        } => format!(
536            "list of {}",
537            values
538                .iter()
539                .map(|v| format!("`{v}`"))
540                .collect::<Vec<_>>()
541                .join(", ")
542        ),
543        ParamFieldType::StringArray { item_enum: None } => "list of strings".into(),
544        ParamFieldType::ObjectArray => "array of tables".into(),
545        ParamFieldType::ObjectMap => "table".into(),
546    }
547}
548
549fn default_cell(v: &Value) -> String {
550    match v {
551        Value::String(s) => format!("`\"{s}\"`"),
552        Value::Array(items) => format!(
553            "`[{}]`",
554            items
555                .iter()
556                .map(|i| i.to_string())
557                .collect::<Vec<_>>()
558                .join(", ")
559        ),
560        other => format!("`{other}`"),
561    }
562}

846    #[test]
847    fn when_is_parsed_and_follows_the_controller_or_its_default() {
848        let schema = shape_schema();
849        let fields = parse_param_schema(&schema);
850        let side = fields.iter().find(|f| f.name == "side").unwrap();
851        assert_eq!(
852            side.when,
853            Some(ParamCondition {
854                key: "shape".into(),
855                any_of: vec!["square".into()]
856            })
857        );
858        let radius = fields.iter().find(|f| f.name == "radius").unwrap();
859        let color = fields.iter().find(|f| f.name == "color").unwrap();
860        let empty = toml::map::Map::new();
861        // No `shape` given: the default (circle) decides.
862        assert!(radius.applies(&schema, &empty));
863        assert!(!side.applies(&schema, &empty));
864        let square = params("shape = \"square\"");
865        assert!(side.applies(&schema, square.as_table().unwrap()));
866        assert!(!radius.applies(&schema, square.as_table().unwrap()));
867        assert!(color.applies(&schema, square.as_table().unwrap()));
868    }
