10#[derive(Debug, Clone)]
11pub struct RegisteredAsset {
12    pub hash: String,
13    pub sidecar_path: PathBuf,
14    pub meta: AssetMeta,
15}

17/// Register a generated asset into the current project ([`Project::current`],
18/// found by walking up from the working directory, #179):
19/// - content store: `<root>/.flint/assets`
20/// - sidecars: `<root>/assets/<kind>/*.asset.toml`
21pub fn register_generated_asset(
22    request: &GenerateRequest,
23    result: &GenerateResult,
24) -> Result<RegisteredAsset> {
25    register_generated_asset_in(&Project::current(), request, result)
26}

28/// Register a generated asset into `project`'s store and catalog.
29pub fn register_generated_asset_in(
30    project: &Project,
31    request: &GenerateRequest,
32    result: &GenerateResult,
33) -> Result<RegisteredAsset> {
34    register_generated_asset_with_roots(
35        request,
36        result,
37        &project.store_dir(),
38        &project.catalog_dir(),
39    )
40}
41
42/// Register a generated asset into a specific content store and sidecar root.
43pub fn register_generated_asset_with_roots(
44    request: &GenerateRequest,
45    result: &GenerateResult,
46    store_root: &Path,
47    assets_root: &Path,
48) -> Result<RegisteredAsset> {
49    let output_path = Path::new(&result.output_path);
50    if !output_path.exists() {
51        return Err(FlintError::GenerationError(format!(
52            "Generated file not found: {}",
53            output_path.display()
54        )));
55    }
56
57    let store = ContentStore::new(store_root);
58    let hash = store.store(output_path)?;
59    let hash_str = hash.to_prefixed_hex();
60
61    let meta = AssetMeta {
62        name: request.name.clone(),
63        asset_type: asset_type_for_kind(request.kind),
64        hash: hash_str.clone(),
65        source_path: Some(result.output_path.clone()),
66        format: output_path
67            .extension()
68            .and_then(|e| e.to_str())
69            .map(|s| s.to_string()),
70        properties: build_generation_properties(result),
71        tags: request.tags.clone(),
72    };
73
74    let sidecar_path = write_asset_sidecar_with_root(&meta, assets_root)?;
75
76    Ok(RegisteredAsset {
77        hash: hash_str,
78        sidecar_path,
79        meta,
80    })
81}

83/// Write a sidecar file into the current project's `assets/` catalog.
84pub fn write_asset_sidecar(meta: &AssetMeta) -> Result<PathBuf> {
85    write_asset_sidecar_with_root(meta, &Project::current().catalog_dir())
86}
87
88/// Write a sidecar file under a specific assets root directory.
89pub fn write_asset_sidecar_with_root(meta: &AssetMeta, assets_root: &Path) -> Result<PathBuf> {
90    #[derive(Serialize)]
91    struct Sidecar<'a> {
92        asset: &'a AssetMeta,
93    }
94
95    let subdir = asset_subdirectory(meta.asset_type);
96    let sidecar_dir = assets_root.join(subdir);
97    std::fs::create_dir_all(&sidecar_dir)?;
98
99    let sidecar_path = sidecar_dir.join(format!("{}.asset.toml", meta.name));
100    let toml_str = toml::to_string_pretty(&Sidecar { asset: meta })?;
101    std::fs::write(&sidecar_path, toml_str)?;
102    Ok(sidecar_path)
103}
104
105fn build_generation_properties(result: &GenerateResult) -> HashMap<String, toml::Value> {
106    let mut props = HashMap::new();
107    props.insert(
108        "prompt".to_string(),
109        toml::Value::String(result.prompt_used.clone()),
110    );
111    props.insert(
112        "provider".to_string(),
113        toml::Value::String(result.provider.clone()),
114    );
115    for (k, v) in &result.metadata {
116        props.insert(k.clone(), toml::Value::String(v.clone()));
117    }
118    props
119}

121fn asset_type_for_kind(kind: AssetKind) -> AssetType {
122    match kind {
123        AssetKind::Texture => AssetType::Texture,
124        AssetKind::Model => AssetType::Mesh,
125        AssetKind::Audio => AssetType::Audio,
126    }
127}
128
129fn asset_subdirectory(asset_type: AssetType) -> &'static str {
130    match asset_type {
131        AssetType::Mesh => "meshes",
132        AssetType::Texture => "textures",
133        AssetType::Material => "materials",
134        AssetType::Audio => "audio",
135        AssetType::Script => "scripts",
136    }
137}

176    #[test]
177    fn test_register_generated_asset_with_roots() {
178        let dir = temp_dir();
179        let store_root = dir.join("store");
180        let assets_root = dir.join("assets");
181        let out_dir = dir.join("out");
182        std::fs::create_dir_all(&out_dir).unwrap();
183
184        let output_path = out_dir.join("brick wall.png");
185        let mut f = std::fs::File::create(&output_path).unwrap();
186        f.write_all(b"fake-image").unwrap();
187
188        let request = sample_request(AssetKind::Texture, "brick_wall");
189        let result = GenerateResult {
190            output_path: output_path.to_string_lossy().to_string(),
191            prompt_used: "weathered \"brick\" wall".to_string(),
192            provider: "mock".to_string(),
193            duration_secs: 0.1,
194            content_hash: None,
195            metadata: HashMap::from([("seed".to_string(), "42".to_string())]),
196        };
197
198        let registered =
199            register_generated_asset_with_roots(&request, &result, &store_root, &assets_root)
200                .unwrap();
201
202        assert!(registered.hash.starts_with("sha256:"));
203        assert!(registered.sidecar_path.exists());
204        assert_eq!(registered.meta.name, "brick_wall");
205        assert_eq!(registered.meta.asset_type, AssetType::Texture);
206
207        let sidecar = std::fs::read_to_string(&registered.sidecar_path).unwrap();
208        let parsed: toml::Value = toml::from_str(&sidecar).unwrap();
209        assert_eq!(
210            parsed
211                .get("asset")
212                .and_then(|a| a.get("name"))
213                .and_then(|v| v.as_str()),
214            Some("brick_wall")
215        );
216        assert_eq!(
217            parsed
218                .get("asset")
219                .and_then(|a| a.get("properties"))
220                .and_then(|p| p.get("prompt"))
221                .and_then(|v| v.as_str()),
222            Some("weathered \"brick\" wall")
223        );
224
225        std::fs::remove_dir_all(dir).ok();
226    }
