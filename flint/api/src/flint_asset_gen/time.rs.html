1//! UTC timestamps for provenance records, without a date crate.
2
3/// The current time as an ISO 8601 UTC timestamp: `2026-10-31T14:05:09Z`.
4pub fn now_iso8601() -> String {
5    let secs = std::time::SystemTime::now()
6        .duration_since(std::time::UNIX_EPOCH)
7        .map(|d| d.as_secs())
8        .unwrap_or_default();
9    iso8601_utc(secs)
10}
11
12/// Format seconds since the Unix epoch as an ISO 8601 UTC timestamp.
13pub fn iso8601_utc(unix_secs: u64) -> String {
14    let days = (unix_secs / 86_400) as i64;
15    let rem = unix_secs % 86_400;
16    let (y, m, d) = civil_from_days(days);
17    format!(
18        "{y:04}-{m:02}-{d:02}T{:02}:{:02}:{:02}Z",
19        rem / 3600,
20        (rem % 3600) / 60,
21        rem % 60
22    )
23}
24
25/// Days since 1970-01-01 to a (year, month, day) civil date
26/// (Howard Hinnant's algorithm, proleptic Gregorian).
27fn civil_from_days(z: i64) -> (i64, u32, u32) {
28    let z = z + 719_468;
29    let era = z.div_euclid(146_097);
30    let doe = z.rem_euclid(146_097);
31    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
32    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
33    let mp = (5 * doy + 2) / 153;
34    let d = (doy - (153 * mp + 2) / 5 + 1) as u32;
35    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
36    let y = yoe + era * 400 + i64::from(m <= 2);
37    (y, m, d)
38}

44    #[test]
45    fn known_instants() {
46        assert_eq!(iso8601_utc(0), "1970-01-01T00:00:00Z");
47        // A leap day and the last second of a century leap year.
48        assert_eq!(iso8601_utc(951_782_400), "2000-02-29T00:00:00Z");
49        assert_eq!(iso8601_utc(978_307_199), "2000-12-31T23:59:59Z");
50        assert_eq!(iso8601_utc(1_793_480_737), "2026-10-31T21:05:37Z");
51    }
52
53    #[test]
54    fn now_is_shaped_like_iso8601() {
55        let s = now_iso8601();
56        assert_eq!(s.len(), 20, "{s}");
57        assert!(s.ends_with('Z') && s.as_bytes()[10] == b'T', "{s}");
58    }
