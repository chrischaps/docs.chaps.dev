8/// Project a world-space point to screen coordinates.
9/// Returns None if the point is behind the camera.
10pub fn world_to_screen(
11    camera: &Camera,
12    screen_size: [f32; 2],
13    pos: [f32; 3],
14) -> Option<egui::Pos2> {
15    let vp = camera.view_projection_matrix();
16    let clip_x = vp[0][0] * pos[0] + vp[1][0] * pos[1] + vp[2][0] * pos[2] + vp[3][0];
17    let clip_y = vp[0][1] * pos[0] + vp[1][1] * pos[1] + vp[2][1] * pos[2] + vp[3][1];
18    let clip_w = vp[0][3] * pos[0] + vp[1][3] * pos[1] + vp[2][3] * pos[2] + vp[3][3];
19
20    if clip_w <= 0.001 {
21        return None;
22    }
23
24    let ndc_x = clip_x / clip_w;
25    let ndc_y = clip_y / clip_w;
26
27    Some(egui::pos2(
28        (ndc_x + 1.0) * 0.5 * screen_size[0],
29        (1.0 - ndc_y) * 0.5 * screen_size[1],
30    ))
31}

33/// Compute a world-space ray from a screen pixel coordinate.
34/// Returns (origin, normalized direction).
35pub fn screen_to_world_ray(
36    camera: &Camera,
37    screen_size: [f32; 2],
38    sx: f32,
39    sy: f32,
40) -> ([f32; 3], [f32; 3]) {
41    let inv_vp = camera.inverse_view_projection_matrix();
42
43    let ndc_x = (sx / screen_size[0]) * 2.0 - 1.0;
44    let ndc_y = 1.0 - (sy / screen_size[1]) * 2.0;
45
46    let near = mat4_transform_point(&inv_vp, [ndc_x, ndc_y, -1.0]);
47    let far = mat4_transform_point(&inv_vp, [ndc_x, ndc_y, 1.0]);
48
49    let dx = far[0] - near[0];
50    let dy = far[1] - near[1];
51    let dz = far[2] - near[2];
52    let len = (dx * dx + dy * dy + dz * dz).sqrt();
53
54    let dir = if len > 1e-8 {
55        [dx / len, dy / len, dz / len]
56    } else {
57        [0.0, 0.0, -1.0]
58    };
59
60    (near, dir)
61}

63/// Transform a 3D point by a 4x4 column-major matrix (with perspective divide).
64pub fn mat4_transform_point(m: &[[f32; 4]; 4], p: [f32; 3]) -> [f32; 3] {
65    let x = m[0][0] * p[0] + m[1][0] * p[1] + m[2][0] * p[2] + m[3][0];
66    let y = m[0][1] * p[0] + m[1][1] * p[1] + m[2][1] * p[2] + m[3][1];
67    let z = m[0][2] * p[0] + m[1][2] * p[1] + m[2][2] * p[2] + m[3][2];
68    let w = m[0][3] * p[0] + m[1][3] * p[1] + m[2][3] * p[2] + m[3][3];
69
70    if w.abs() < 1e-10 {
71        [x, y, z]
72    } else {
73        [x / w, y / w, z / w]
74    }
75}

77/// Compute the distance from a ray to a point.
78pub fn ray_point_distance(ray_origin: [f32; 3], ray_dir: [f32; 3], point: [f32; 3]) -> f32 {
79    let v = [
80        point[0] - ray_origin[0],
81        point[1] - ray_origin[1],
82        point[2] - ray_origin[2],
83    ];
84    let t = (v[0] * ray_dir[0] + v[1] * ray_dir[1] + v[2] * ray_dir[2]).max(0.0);
85    let closest = [
86        ray_origin[0] + ray_dir[0] * t,
87        ray_origin[1] + ray_dir[1] * t,
88        ray_origin[2] + ray_dir[2] * t,
89    ];
90    let dx = closest[0] - point[0];
91    let dy = closest[1] - point[1];
92    let dz = closest[2] - point[2];
93    (dx * dx + dy * dy + dz * dz).sqrt()
94}

96/// Compute the view-space depth of a point (for draw ordering).
97pub fn point_depth(camera: &Camera, pos: [f32; 3]) -> f32 {
98    let view = camera.view_matrix();
99    let z = view[0][2] * pos[0] + view[1][2] * pos[1] + view[2][2] * pos[2] + view[3][2];
100    -z
101}

103/// Intersect a ray with a plane defined by normal and distance from origin.
104/// Returns the intersection point, or None if ray is parallel to the plane.
105pub fn ray_plane_intersect(
106    ray_o: [f32; 3],
107    ray_d: [f32; 3],
108    plane_n: [f32; 3],
109    plane_d: f32,
110) -> Option<[f32; 3]> {
111    let denom = plane_n[0] * ray_d[0] + plane_n[1] * ray_d[1] + plane_n[2] * ray_d[2];
112    if denom.abs() < 1e-6 {
113        return None;
114    }
115    let t =
116        (plane_d - plane_n[0] * ray_o[0] - plane_n[1] * ray_o[1] - plane_n[2] * ray_o[2]) / denom;
117    if t < 0.0 {
118        return None;
119    }
120    Some([
121        ray_o[0] + ray_d[0] * t,
122        ray_o[1] + ray_d[1] * t,
123        ray_o[2] + ray_d[2] * t,
124    ])
125}
