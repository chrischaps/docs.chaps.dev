7/// Configuration for a procedural generation AI agent.
8///
9/// The API key is never written out: `Debug` redacts it and serialization
10/// skips it, so a config can be logged or saved beside a project without
11/// leaking the secret. Load the key from the environment instead.
12#[derive(Clone, Deserialize, Serialize, PartialEq)]
13pub struct AgentConfig {
14    /// Provider name (e.g. "anthropic", "mock").
15    pub provider: String,
16    /// Model identifier (e.g. "claude-sonnet-5-5").
17    pub model: String,
18    /// Maximum refinement iterations before stopping.
19    pub max_iterations: u32,
20    /// API key for the provider. Not serialized.
21    #[serde(default, skip_serializing)]
22    pub api_key: String,
23    /// Reasoning effort (`low` ... `max`) for providers that take one;
24    /// `None` uses the model's default.
25    #[serde(default, skip_serializing_if = "Option::is_none")]
26    pub effort: Option<String>,
27}
28
29impl AgentConfig {
30    /// A config for [`crate::MockAgent`]: no key, three iterations.
31    pub fn mock() -> Self {
32        Self {
33            provider: "mock".to_string(),
34            model: "mock".to_string(),
35            max_iterations: 3,
36            api_key: String::new(),
37            effort: None,
38        }
39    }
40
41    /// A config for [`crate::AnthropicAgent`]: [`crate::anthropic::DEFAULT_MODEL`],
42    /// three iterations, and no key (the agent reads `ANTHROPIC_API_KEY`).
43    pub fn anthropic() -> Self {
44        Self {
45            provider: "anthropic".to_string(),
46            model: crate::anthropic::DEFAULT_MODEL.to_string(),
47            max_iterations: 3,
48            api_key: String::new(),
49            effort: None,
50        }
51    }
52
53    /// Check the config is usable: a provider and model are named, at least
54    /// one iteration is allowed, and any provider but `mock` has a key.
55    pub fn validate(&self) -> Result<()> {
56        let err = |m: &str| Err(ProcGenAiError::ConfigError(m.to_string()));
57        if self.provider.trim().is_empty() {
58            return err("provider is empty");
59        }
60        if self.model.trim().is_empty() {
61            return err("model is empty");
62        }
63        if self.max_iterations == 0 {
64            return err("max_iterations must be at least 1");
65        }
66        if self.provider != "mock" && self.api_key.trim().is_empty() {
67            return Err(ProcGenAiError::ConfigError(format!(
68                "provider '{}' needs an api_key",
69                self.provider
70            )));
71        }
72        Ok(())
73    }
74}
75
76impl Default for AgentConfig {
77    fn default() -> Self {
78        Self::mock()
79    }
80}
81
82impl fmt::Debug for AgentConfig {
83    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
84        let key = if self.api_key.is_empty() {
85            "<none>"
86        } else {
87            "<redacted>"
88        };
89        f.debug_struct("AgentConfig")
90            .field("provider", &self.provider)
91            .field("model", &self.model)
92            .field("max_iterations", &self.max_iterations)
93            .field("api_key", &key)
94            .field("effort", &self.effort)
95            .finish()
96    }
97}

120    #[test]
121    fn serialization_drops_the_key() {
122        let text = toml::to_string(&real()).unwrap();
123        assert!(!text.contains("sk-secret"), "{text}");
124        let back: AgentConfig = toml::from_str(&text).unwrap();
125        assert_eq!(back.api_key, "");
126        assert_eq!(back.model, "claude-sonnet-5-5");

129    #[test]
130    fn validation() {
131        assert!(AgentConfig::mock().validate().is_ok());
132        assert!(real().validate().is_ok());
133        let mut keyless = real();
134        keyless.api_key.clear();
135        assert!(keyless.validate().is_err());
136        let mut zero = AgentConfig::mock();
137        zero.max_iterations = 0;
138        assert!(zero.validate().is_err());
