1//! How far one image is from another, cheaply, so a refine can tell whether
2//! a round moved its asset toward the reference.
3//!
4//! Three coarse measures, each scaled to 0..1, and their mean as a score out
5//! of 100 (0 = the same image):
6//!
7//! - **mean colour**: the distance between the two images' average sRGB
8//!   colours, over the largest distance there is (black to white).
9//! - **histogram**: half the L1 distance between 16-bin luminance histograms,
10//!   so how much of one image's tonal spread would have to move to match the
11//!   other's.
12//! - **layout**: the mean absolute difference of the two images' luminance
13//!   at 16x16, which is where the light and dark masses sit (a silhouette,
14//!   a crown's outline, a brick course).
15//!
16//! None of these is perceptual. They are a ruler, not a judge: good enough
17//! to say "closer" or "further" between two rounds of the same refine.
18
19use image::imageops::FilterType;
20use image::{GrayImage, RgbImage};
21
22/// Bins in the luminance histogram.
23const BINS: usize = 16;
24/// Side of the thumbnail the layout term compares.
25const LAYOUT: u32 = 16;
26/// The largest mean-colour distance: black to white in sRGB 0..255.
27const MAX_RGB_DISTANCE: f64 = 441.672_955_930_063_7; // 255 * sqrt(3)
28
29/// How far an image is from a reference. Lower is closer.
30#[derive(Debug, Clone, Copy, PartialEq)]
31pub struct Distance {
32    /// The mean of the three terms below, times 100: 0 is identical, 100
33    /// is as far apart as these measures go.
34    pub score: f64,
35    /// Mean-colour distance in sRGB units (0..441.7).
36    pub mean_colour: f64,
37    /// Luminance-histogram distance, 0..1.
38    pub histogram: f64,
39    /// 16x16 luminance-layout distance, 0..1.
40    pub layout: f64,
41    /// The reference's average colour.
42    pub reference_mean: [u8; 3],
43    /// The measured image's average colour.
44    pub current_mean: [u8; 3],
45}
46
47/// The distance from `current` to `reference`, both encoded images (PNG,
48/// JPEG, ...). `None` when either fails to decode or is empty.
49pub fn distance(reference: &[u8], current: &[u8]) -> Option<Distance> {
50    let r = decode(reference)?;
51    let c = decode(current)?;
52    Some(distance_rgb(&r, &c))
53}
54
55/// [`distance`] for decoded images.
56pub fn distance_rgb(reference: &RgbImage, current: &RgbImage) -> Distance {
57    let (rm, cm) = (mean(reference), mean(current));
58    let mean_colour = rm
59        .iter()
60        .zip(cm)
61        .map(|(a, b)| (a - b).powi(2))
62        .sum::<f64>()
63        .sqrt();
64
65    let (rh, ch) = (histogram(reference), histogram(current));
66    let histogram = rh.iter().zip(ch).map(|(a, b)| (a - b).abs()).sum::<f64>() / 2.0;
67
68    let (rl, cl) = (thumbnail(reference), thumbnail(current));
69    let layout = rl
70        .pixels()
71        .zip(cl.pixels())
72        .map(|(a, b)| (f64::from(a.0[0]) - f64::from(b.0[0])).abs())
73        .sum::<f64>()
74        / f64::from(LAYOUT * LAYOUT)
75        / 255.0;
76
77    let score = (mean_colour / MAX_RGB_DISTANCE + histogram + layout) / 3.0 * 100.0;
78    Distance {
79        score,
80        mean_colour,
81        histogram,
82        layout,
83        reference_mean: rm.map(|v| v.round().clamp(0.0, 255.0) as u8),
84        current_mean: cm.map(|v| v.round().clamp(0.0, 255.0) as u8),
85    }
86}

88/// The average sRGB colour of an encoded image, ignoring alpha.
89pub fn mean_rgb(bytes: &[u8]) -> Option<[u8; 3]> {
90    decode(bytes).map(|img| mean(&img).map(|v| v.round().clamp(0.0, 255.0) as u8))
91}
92
93fn decode(bytes: &[u8]) -> Option<RgbImage> {
94    let img = image::load_from_memory(bytes).ok()?.to_rgb8();
95    (img.width() > 0 && img.height() > 0).then_some(img)
96}
97
98fn mean(img: &RgbImage) -> [f64; 3] {
99    let n = f64::from(img.width()) * f64::from(img.height());
100    let mut sum = [0.0f64; 3];
101    for p in img.pixels() {
102        for (s, c) in sum.iter_mut().zip(p.0) {
103            *s += f64::from(c);
104        }
105    }
106    sum.map(|s| s / n)
107}
108
109/// Rec. 709 weights on the sRGB values: a tonal measure, not a colorimetric
110/// one, which is all the histogram and layout terms need.
111fn luma(p: [u8; 3]) -> f64 {
112    0.2126 * f64::from(p[0]) + 0.7152 * f64::from(p[1]) + 0.0722 * f64::from(p[2])
113}
114
115fn histogram(img: &RgbImage) -> [f64; BINS] {
116    let mut bins = [0.0f64; BINS];
117    for p in img.pixels() {
118        let b = ((luma(p.0) / 256.0) * BINS as f64) as usize;
119        bins[b.min(BINS - 1)] += 1.0;
120    }
121    let n = f64::from(img.width()) * f64::from(img.height());
122    bins.map(|b| b / n)
123}
124
125fn thumbnail(img: &RgbImage) -> GrayImage {
126    let gray = GrayImage::from_fn(img.width(), img.height(), |x, y| {
127        image::Luma([luma(img.get_pixel(x, y).0).round().clamp(0.0, 255.0) as u8])
128    });
129    image::imageops::resize(&gray, LAYOUT, LAYOUT, FilterType::Triangle)
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    fn solid(r: u8, g: u8, b: u8) -> RgbImage {
137        RgbImage::from_pixel(32, 32, image::Rgb([r, g, b]))
138    }
139
140    #[test]
141    fn an_image_is_zero_from_itself() {
142        let img = solid(120, 80, 40);
143        let d = distance_rgb(&img, &img);
144        assert_eq!(d.score, 0.0);
145        assert_eq!(d.current_mean, [120, 80, 40]);
146    }
147
148    #[test]
149    fn black_and_white_are_as_far_apart_as_it_goes() {
150        let d = distance_rgb(&solid(0, 0, 0), &solid(255, 255, 255));
151        assert!((d.mean_colour - MAX_RGB_DISTANCE).abs() < 1e-9);
152        assert!((d.histogram - 1.0).abs() < 1e-9);
153        assert!((d.layout - 1.0).abs() < 1e-9);
154        assert!((d.score - 100.0).abs() < 1e-9);
155    }
156
157    #[test]
158    fn layout_sees_where_the_mass_is_when_colour_does_not() {
159        // Same pixels, mirrored: identical mean and histogram, different layout.
160        let left = RgbImage::from_fn(32, 32, |x, _| {
161            if x < 16 {
162                image::Rgb([0, 0, 0])
163            } else {
164                image::Rgb([255, 255, 255])
165            }
166        });
167        let mut right = left.clone();
168        image::imageops::flip_horizontal_in_place(&mut right);
169        let d = distance_rgb(&left, &right);
170        assert!(d.mean_colour < 1e-9 && d.histogram < 1e-9);
171        assert!(d.layout > 0.9, "{d:?}");
172    }
173
174    #[test]
175    fn closer_colours_score_lower() {
176        let r = solid(40, 110, 50);
177        let near = distance_rgb(&r, &solid(50, 100, 50)).score;
178        let far = distance_rgb(&r, &solid(200, 60, 40)).score;
179        assert!(near < far, "{near} vs {far}");
180    }
181
182    #[test]
183    fn undecodable_bytes_have_no_distance() {
184        assert!(distance(b"nope", b"nope").is_none());
185        assert!(mean_rgb(b"nope").is_none());
186    }
