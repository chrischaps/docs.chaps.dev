1//! [`MockAgent`]: a local, deterministic stand-in for a real AI agent.
2//!
3//! It knows nothing about language beyond a small keyword table, but every
4//! spec it writes is one the generator accepts: params come from the
5//! generator's own `param_schema()` defaults, jitter stays inside the
6//! schema's ranges, and the same prompt always yields the same spec. That
7//! lets the create → interpret → refine pipeline be tested end to end (and
8//! fed to `flint gen`) with no network and no API key.
9

139/// FNV-1a: a stable 64-bit hash, so seeds survive Rust and platform changes.
140pub(crate) fn fnv1a(parts: &[&[u8]]) -> u64 {
141    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
142    for part in parts {
143        for b in part.iter().chain(&[0xff]) {
144            h ^= u64::from(*b);
145            h = h.wrapping_mul(0x0100_0000_01b3);
146        }
147    }
148    h
149}

158/// The registered name for `generator`: exact, or with `_v1` appended
159/// (so `"tree"` finds `tree_v1`).
160pub(crate) fn resolve_name(registry: &GeneratorRegistry, generator: &str) -> Option<String> {
161    let g = generator.trim();
162    if registry.get(g).is_some() {
163        return Some(g.to_string());
164    }
165    let v1 = format!("{g}_v1");
166    registry.get(&v1).is_some().then_some(v1)
167}

169pub(crate) fn unknown_generator(registry: &GeneratorRegistry, generator: &str) -> ProcGenAiError {
170    let mut names = registry.generator_names();
171    names.sort_unstable();
172    ProcGenAiError::SpecError(format!(
173        "unknown generator '{generator}'; known: {}",
174        names.join(", ")
175    ))
176}

178fn words(prompt: &str) -> Vec<String> {
179    prompt
180        .split(|c: char| !c.is_alphanumeric())
181        .filter(|w| !w.is_empty())
182        .map(str::to_lowercase)
183        .collect()

186pub(crate) fn props(schema: &Json) -> Option<&serde_json::Map<String, Json>> {
187    schema.get("properties").and_then(Json::as_object)
188}

190/// Scale every float param that has a schema default by `1 ± amount`,
191/// clamped into its range and rounded to three decimals. Params are visited
192/// in sorted order so the RNG stream, and the result, is reproducible.
193fn jitter_floats(schema: &Json, params: &mut toml::Value, rng: &mut SeededRng, amount: f32) {
194    let (Some(props), Some(table)) = (props(schema), params.as_table_mut()) else {
195        return;
196    };
197    let mut keys: Vec<String> = table.keys().cloned().collect();
198    keys.sort();
199    for key in keys {
200        let Some(prop) = props.get(&key) else {
201            continue;
202        };
203        if prop_type(prop) != "number" {
204            continue;
205        }
206        let Some(toml::Value::Float(v)) = table.get(&key).cloned() else {
207            continue;
208        };
209        let factor = 1.0 + f64::from(rng.next_range(-amount, amount));
210        let bounds = NumBounds::of(prop);
211        table.insert(key, toml::Value::Float(round3(bounds.clamp(v * factor))));
212    }
213}

215/// Apply the keyword table. Returns the words that changed something.
216fn apply_keywords(schema: &Json, params: &mut toml::Value, prompt: &str) -> Vec<String> {
217    let (Some(props), Some(table)) = (props(schema), params.as_table_mut()) else {
218        return Vec::new();
219    };
220    let said = words(prompt);
221    let mut used = Vec::new();
222    for (triggers, param, nudge) in KEYWORDS {
223        let Some(word) = said.iter().find(|w| triggers.contains(&w.as_str())) else {
224            continue;
225        };
226        let Some(prop) = props.get(*param) else {
227            continue;
228        };
229        let bounds = NumBounds::of(prop);
230        let current = table
231            .get(*param)
232            .and_then(|v| v.as_float().or(v.as_integer().map(|i| i as f64)));
233        let value = match (nudge, current) {
234            (Nudge::Scale(f), Some(v)) => toml::Value::Float(round3(bounds.clamp(v * f))),
235            (Nudge::SetF(v), _) => toml::Value::Float(round3(bounds.clamp(*v))),
236            (Nudge::SetS(s), _) => toml::Value::String((*s).to_string()),
237            (Nudge::Scale(_), None) => continue,
238        };
239        table.insert((*param).to_string(), value);
240        if !used.contains(word) {
241            used.push(word.clone());
242        }
243    }
244    used
245}

247pub(crate) fn slug(prompt: &str) -> String {
248    let w = words(prompt);
249    if w.is_empty() {
250        "untitled".to_string()
251    } else {
252        w.into_iter().take(4).collect::<Vec<_>>().join("_")
253    }
254}

256/// `1.2.3` → `1.2.4`; anything else is left alone.
257pub(crate) fn bump_patch(version: &str) -> String {
258    let mut parts: Vec<&str> = version.split('.').collect();
259    if let Some(last) = parts.last_mut().and_then(|p| p.parse::<u64>().ok()) {
260        let bumped = (last + 1).to_string();
261        let n = parts.len();
262        let mut out: Vec<String> = parts.drain(..n - 1).map(str::to_string).collect();
263        out.push(bumped);
264        return out.join(".");
265    }
266    version.to_string()
267}

269/// The spec with its generator name resolved and its params repaired
270/// against the schema. Returns the repaired spec and what was fixed.
271fn repaired(
272    registry: &GeneratorRegistry,
273    spec: &ProcGenSpec,
274) -> Option<(ProcGenSpec, Vec<String>)> {
275    let name = resolve_name(registry, &spec.generator)?;
276    let generator = registry.get(&name)?;
277    let schema = generator.param_schema();
278    let mut fixed = spec.clone();
279    let mut notes: Vec<String> = check_params(&schema, &spec.params)
280        .iter()
281        .map(ToString::to_string)
282        .collect();
283    if name != spec.generator {
284        notes.insert(
285            0,
286            format!(
287                "generator \"{}\" is registered as \"{name}\"; use that name",
288                spec.generator
289            ),
290        );
291        fixed.generator = name;
292    }
293    repair_params(&schema, &mut fixed.params);
294    Some((fixed, notes))
295}

297impl MockAgent {
298    /// The generator a spec names, if it is registered.
299    fn with_generator<T>(
300        &self,
301        generator: &str,
302        f: impl FnOnce(&str, &dyn Generator) -> Result<T>,
303    ) -> Result<T> {
304        let registry = registry();
305        let name = resolve_name(&registry, generator)
306            .ok_or_else(|| unknown_generator(&registry, generator))?;
307        let g = registry.get(&name).expect("resolved names are registered");
308        f(&name, g)
309    }
310}

312impl ProcGenAgent for MockAgent {
313    fn interpret_spec(&self, spec: &ProcGenSpec) -> Result<SpecFeedback> {
314        let registry = registry();
315        let Some((fixed, mut suggestions)) = repaired(&registry, spec) else {
316            return Ok(SpecFeedback {
317                suggestions: vec![unknown_generator(&registry, &spec.generator).to_string()],
318                confidence: 0.0,
319                revised_spec: None,
320            });
321        };
322        let problems = suggestions.len();
323        let mut revised = fixed;
324
325        if spec.seed.mode == SeedMode::Random {
326            suggestions.push(
327                "the seed is random, so every run differs; fix it to make the asset reproducible"
328                    .to_string(),
329            );
330            revised.seed = SeedConfig {
331                mode: SeedMode::Fixed,
332                value: Some(fnv1a(&[spec.meta.name.as_bytes()]) % 100_000),
333                derive_from: None,
334            };
335        }
336        let generator = registry
337            .get(&revised.generator)
338            .expect("repaired names resolve");
339        if generator.output_kind() == OutputKind::Mesh && spec.lod.is_none() {
340            suggestions.push(
341                "no [[lod]] levels: add a few so the mesh stays cheap at a distance".to_string(),
342            );
343        }
344        if suggestions.is_empty() {
345            suggestions.push(format!(
346                "params fit {}'s schema; try a few other seeds to see this spec's siblings",
347                revised.generator
348            ));
349        }
350
351        let confidence = if problems == 0 {
352            0.9
353        } else {
354            (0.9 / (1.0 + problems as f32)).max(0.1)
355        };
356        Ok(SpecFeedback {
357            suggestions,
358            confidence,
359            revised_spec: (revised != *spec).then_some(revised),
360        })
361    }

363    fn create_spec_from_prompt(&self, prompt: &str, generator: &str) -> Result<ProcGenSpec> {
364        self.with_generator(generator, |name, g| {
365            let schema = g.param_schema();
366            let seed = fnv1a(&[name.as_bytes(), prompt.as_bytes()]);
367            let mut rng = SeededRng::new(seed);
368
369            // Start from a template where defaults can't build anything.
370            let mut params = match name {
371                "creature_v1" => ProcGenSpec::parse(CREATURE_TEMPLATE)?.params,
372                _ => toml::Value::Table(Default::default()),
373            };
374            // Write out every scalar default, so the spec says what it makes.
375            if let (Some(props), Some(table)) = (props(&schema), params.as_table_mut()) {
376                for (key, prop) in props {
377                    let scalar =
378                        matches!(prop_type(prop), "number" | "integer" | "boolean" | "string");
379                    if scalar && !table.contains_key(key) {
380                        if let Some(v) = default_of(prop) {
381                            table.insert(key.clone(), v);
382                        }
383                    }
384                }
385            }
386            jitter_floats(&schema, &mut params, &mut rng, 0.05);
387            let used = apply_keywords(&schema, &mut params, prompt);
388            repair_params(&schema, &mut params);
389
390            let mut tags = vec!["ai-generated".to_string(), "mock".to_string()];
391            tags.extend(used);
392            Ok(ProcGenSpec {
393                meta: SpecMeta {
394                    name: slug(prompt),
395                    description: Some(prompt.to_string()),
396                    version: "1.0.0".to_string(),
397                    tags,
398                },
399                generator: name.to_string(),
400                seed: SeedConfig {
401                    mode: SeedMode::Fixed,
402                    value: Some(seed % 100_000),
403                    derive_from: None,
404                },
405                params,
406                lod: None,
407            })
408        })
409    }
410
411    fn refine_spec(&self, spec: &ProcGenSpec, reference: Option<&[u8]>) -> Result<ProcGenSpec> {
412        let registry = registry();
413        let (mut refined, _) = repaired(&registry, spec)
414            .ok_or_else(|| unknown_generator(&registry, &spec.generator))?;
415        let schema = registry
416            .get(&refined.generator)
417            .expect("repaired names resolve")
418            .param_schema();
419
420        // The mock can't see the reference; it only lets it pick the variation.
421        let seed_bytes = spec.seed.value.unwrap_or(0).to_le_bytes();
422        let mut rng = SeededRng::new(fnv1a(&[
423            spec.meta.name.as_bytes(),
424            &seed_bytes,
425            reference.unwrap_or(&[]),
426        ]));
427        jitter_floats(&schema, &mut refined.params, &mut rng, 0.08);
428        refined.meta.version = bump_patch(&refined.meta.version);
429        Ok(refined)
430    }
