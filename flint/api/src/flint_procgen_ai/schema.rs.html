1//! Checking a spec's `params` against its generator's JSON param schema.
2//!
3//! Every [`flint_procgen::Generator`] publishes `param_schema()`. This module
4//! reads the small subset of JSON Schema those schemas use (`type`,
5//! `minimum`/`exclusiveMinimum`, `maximum`/`exclusiveMaximum`, `enum`, hex
6//! colour `pattern`, `required`, `additionalProperties: false`), reports what
7//! a spec gets wrong, and repairs it. Agents use it to tell an honest spec
8//! from a plausible-looking one: a param the generator never reads is a
9//! silent no-op, and a value out of range is an error at generation time.
10
11use std::fmt;

13use serde_json::Value as Json;

15/// A numeric range from a schema property, each end inclusive or exclusive.
16#[derive(Debug, Clone, Copy, Default, PartialEq)]
17pub struct NumBounds {
18    /// Lower bound and whether it is exclusive.
19    pub min: Option<(f64, bool)>,
20    /// Upper bound and whether it is exclusive.
21    pub max: Option<(f64, bool)>,
22}
23
24impl NumBounds {
25    /// Read the bounds of one schema property.
26    pub fn of(prop: &Json) -> Self {
27        let get = |k: &str| prop.get(k).and_then(Json::as_f64);
28        Self {
29            min: get("minimum")
30                .map(|v| (v, false))
31                .or_else(|| get("exclusiveMinimum").map(|v| (v, true))),
32            max: get("maximum")
33                .map(|v| (v, false))
34                .or_else(|| get("exclusiveMaximum").map(|v| (v, true))),
35        }
36    }
37
38    /// Whether `v` lies inside the range.
39    pub fn contains(&self, v: f64) -> bool {
40        let lo = match self.min {
41            Some((m, true)) => v > m,
42            Some((m, false)) => v >= m,
43            None => true,
44        };
45        let hi = match self.max {
46            Some((m, true)) => v < m,
47            Some((m, false)) => v <= m,
48            None => true,
49        };
50        lo && hi
51    }
52
53    /// The nearest value inside the range. An exclusive end is stepped
54    /// inside by 1% of its magnitude (at least 0.001).
55    pub fn clamp(&self, v: f64) -> f64 {
56        let nudge = |m: f64| (m.abs() * 0.01).max(0.001);
57        let mut out = v;
58        if let Some((m, excl)) = self.min {
59            if out < m || (excl && out <= m) {
60                out = if excl { m + nudge(m) } else { m };
61            }
62        }
63        if let Some((m, excl)) = self.max {
64            if out > m || (excl && out >= m) {
65                out = if excl { m - nudge(m) } else { m };
66            }
67        }
68        out
69    }
70}

72/// What is wrong with one param.
73#[derive(Debug, Clone, PartialEq)]
74pub enum IssueKind {
75    /// The generator's schema has no such param, so the generator ignores it.
76    Unknown,
77    /// The value has the wrong TOML type.
78    WrongType {
79        /// The schema's type name (`number`, `integer`, ...).
80        expected: String,
81    },
82    /// A number outside the schema's range.
83    OutOfRange {
84        /// The offending value.
85        value: f64,
86        /// The allowed range.
87        bounds: NumBounds,
88    },
89    /// A string not among the schema's `enum` values.
90    NotInEnum {
91        /// The offending value.
92        value: String,
93        /// The allowed values.
94        allowed: Vec<String>,
95    },
96    /// A string that should be a `#RRGGBB` / `#RRGGBBAA` colour.
97    BadColor {
98        /// The offending value.
99        value: String,
100    },
101    /// A param the schema requires is absent.
102    MissingRequired,
103}
104
105/// One problem found in a spec's params.
106#[derive(Debug, Clone, PartialEq)]
107pub struct ParamIssue {
108    /// The param's name.
109    pub param: String,
110    /// What is wrong with it.
111    pub kind: IssueKind,
112}
113
114impl fmt::Display for ParamIssue {
115    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
116        let p = &self.param;
117        match &self.kind {
118            IssueKind::Unknown => write!(
119                f,
120                "`{p}` is not a parameter of this generator and will be ignored; remove it"
121            ),
122            IssueKind::WrongType { expected } => {
123                write!(f, "`{p}` should be a {expected}")
124            }
125            IssueKind::OutOfRange { value, bounds } => {
126                write!(f, "`{p}` = {value} is outside {}", describe(bounds))?;
127                write!(f, "; use {}", round3(bounds.clamp(*value)))
128            }
129            IssueKind::NotInEnum { value, allowed } => write!(
130                f,
131                "`{p}` = \"{value}\" is not one of: {}",
132                allowed.join(", ")
133            ),
134            IssueKind::BadColor { value } => {
135                write!(f, "`{p}` = \"{value}\" is not a #RRGGBB colour")
136            }
137            IssueKind::MissingRequired => write!(f, "`{p}` is required"),
138        }
139    }
140}

142fn describe(b: &NumBounds) -> String {
143    let lo = match b.min {
144        Some((m, true)) => format!("({m}"),
145        Some((m, false)) => format!("[{m}"),
146        None => "(-inf".to_string(),
147    };
148    let hi = match b.max {
149        Some((m, true)) => format!("{m})"),
150        Some((m, false)) => format!("{m}]"),
151        None => "inf)".to_string(),
152    };
153    format!("{lo}, {hi}")
154}

156/// Round to three decimals, so agent-written specs stay readable.
157pub fn round3(v: f64) -> f64 {
158    (v * 1000.0).round() / 1000.0
159}

161fn properties(schema: &Json) -> Option<&serde_json::Map<String, Json>> {
162    schema.get("properties").and_then(Json::as_object)
163}

165fn is_hex_color(s: &str) -> bool {
166    let h = s.strip_prefix('#').unwrap_or(s);
167    (h.len() == 6 || h.len() == 8) && h.chars().all(|c| c.is_ascii_hexdigit())
168}
169
170fn color_pattern(prop: &Json) -> bool {
171    prop.get("pattern")
172        .and_then(Json::as_str)
173        .is_some_and(|p| p.contains("[0-9a-fA-F]"))
174}
175
176fn string_enum(prop: &Json) -> Option<Vec<String>> {
177    let values: Vec<String> = prop
178        .get("enum")?
179        .as_array()?
180        .iter()
181        .filter_map(|v| v.as_str().map(str::to_string))
182        .collect();
183    (!values.is_empty()).then_some(values)
184}

186/// The schema `type` of one property (`""` when absent).
187pub fn prop_type(prop: &Json) -> &str {
188    prop.get("type").and_then(Json::as_str).unwrap_or("")
189}
190
191fn check_one(prop: &Json, value: &toml::Value) -> Option<IssueKind> {
192    let wrong = || {
193        Some(IssueKind::WrongType {
194            expected: prop_type(prop).to_string(),
195        })
196    };
197    match prop_type(prop) {
198        "number" => {
199            let v = match value {
200                toml::Value::Float(f) => *f,
201                toml::Value::Integer(i) => *i as f64,
202                _ => return wrong(),
203            };
204            let bounds = NumBounds::of(prop);
205            (!bounds.contains(v)).then_some(IssueKind::OutOfRange { value: v, bounds })
206        }
207        "integer" => {
208            let toml::Value::Integer(i) = value else {
209                return wrong();
210            };
211            let bounds = NumBounds::of(prop);
212            let v = *i as f64;
213            (!bounds.contains(v)).then_some(IssueKind::OutOfRange { value: v, bounds })
214        }
215        "boolean" if !value.is_bool() => wrong(),
216        "string" => {
217            let Some(s) = value.as_str() else {
218                return wrong();
219            };
220            if let Some(allowed) = string_enum(prop) {
221                if !allowed.iter().any(|a| a == s) {
222                    return Some(IssueKind::NotInEnum {
223                        value: s.to_string(),
224                        allowed,
225                    });
226                }
227            }
228            if color_pattern(prop) && !is_hex_color(s) {
229                return Some(IssueKind::BadColor {
230                    value: s.to_string(),
231                });
232            }
233            None
234        }
235        "array" if !value.is_array() => wrong(),
236        "object" if !value.is_table() => wrong(),
237        _ => None,
238    }
239}

241/// Every problem in `params` against `schema`, sorted by param name.
242pub fn check_params(schema: &Json, params: &toml::Value) -> Vec<ParamIssue> {
243    let mut issues = Vec::new();
244    let Some(props) = properties(schema) else {
245        return issues;
246    };
247    let closed = schema.get("additionalProperties") == Some(&Json::Bool(false));
248    let empty = toml::map::Map::new();
249    let table = params.as_table().unwrap_or(&empty);
250
251    for (name, value) in table {
252        match props.get(name) {
253            Some(prop) => {
254                if let Some(kind) = check_one(prop, value) {
255                    issues.push(ParamIssue {
256                        param: name.clone(),
257                        kind,
258                    });
259                }
260            }
261            None if closed => issues.push(ParamIssue {
262                param: name.clone(),
263                kind: IssueKind::Unknown,
264            }),
265            None => {}
266        }
267    }
268    if let Some(required) = schema.get("required").and_then(Json::as_array) {
269        for r in required.iter().filter_map(Json::as_str) {
270            if !table.contains_key(r) {
271                issues.push(ParamIssue {
272                    param: r.to_string(),
273                    kind: IssueKind::MissingRequired,
274                });
275            }
276        }
277    }
278    issues.sort_by(|a, b| a.param.cmp(&b.param));
279    issues
280}

282/// Convert a JSON default into TOML. Returns `None` for `null`.
283pub fn json_to_toml(v: &Json) -> Option<toml::Value> {
284    Some(match v {
285        Json::Null => return None,
286        Json::Bool(b) => toml::Value::Boolean(*b),
287        Json::Number(n) => match n.as_i64() {
288            Some(i) => toml::Value::Integer(i),
289            None => toml::Value::Float(n.as_f64()?),
290        },
291        Json::String(s) => toml::Value::String(s.clone()),
292        Json::Array(a) => toml::Value::Array(a.iter().filter_map(json_to_toml).collect()),
293        Json::Object(o) => toml::Value::Table(
294            o.iter()
295                .filter_map(|(k, v)| Some((k.clone(), json_to_toml(v)?)))
296                .collect(),
297        ),
298    })
299}

301/// A schema property's default as TOML, typed to match the property
302/// (`"number"` defaults always come back as floats).
303pub fn default_of(prop: &Json) -> Option<toml::Value> {
304    let v = json_to_toml(prop.get("default")?)?;
305    Some(match (prop_type(prop), v) {
306        ("number", toml::Value::Integer(i)) => toml::Value::Float(i as f64),
307        (_, v) => v,
308    })
309}
310
311/// The value a repair writes for a missing or mistyped param: its default,
312/// else the first enum value, else nothing.
313fn fallback(prop: &Json) -> Option<toml::Value> {
314    default_of(prop).or_else(|| {
315        string_enum(prop)
316            .and_then(|e| e.into_iter().next())
317            .map(toml::Value::String)
318    })
319}

321/// Fix every issue [`check_params`] reports: drop unknown params, clamp
322/// numbers into range, and replace bad values with the schema default.
323/// Returns the number of params changed.
324pub fn repair_params(schema: &Json, params: &mut toml::Value) -> usize {
325    let issues = check_params(schema, params);
326    let Some(props) = properties(schema) else {
327        return 0;
328    };
329    if !params.is_table() {
330        *params = toml::Value::Table(Default::default());
331    }
332    let table = params.as_table_mut().expect("params is a table");
333    let mut changed = 0;
334    for issue in &issues {
335        let prop = props.get(&issue.param);
336        match (&issue.kind, prop) {
337            (IssueKind::Unknown, _) => {
338                table.remove(&issue.param);
339            }
340            (IssueKind::OutOfRange { value, bounds }, Some(prop)) => {
341                let fixed = bounds.clamp(*value);
342                let v = if prop_type(prop) == "integer" {
343                    toml::Value::Integer(fixed.round() as i64)
344                } else {
345                    toml::Value::Float(round3(fixed))
346                };
347                table.insert(issue.param.clone(), v);
348            }
349            (_, Some(prop)) => match fallback(prop) {
350                Some(v) => {
351                    table.insert(issue.param.clone(), v);
352                }
353                None => {
354                    table.remove(&issue.param);
355                }
356            },
357            (_, None) => continue,
358        }
359        changed += 1;
360    }
361    changed
362}
