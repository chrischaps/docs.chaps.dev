1use thiserror::Error;
2
3/// Errors specific to AI-assisted procedural generation.
4#[derive(Debug, Error)]
5pub enum ProcGenAiError {
6    /// Invalid or malformed spec.
7    #[error("spec error: {0}")]
8    SpecError(String),
9
10    /// Agent communication or processing failure.
11    #[error("agent error: {0}")]
12    AgentError(String),
13
14    /// Configuration issue (missing API key, bad model name, etc.).
15    #[error("config error: {0}")]
16    ConfigError(String),
17
18    /// The provider rejected the credentials (HTTP 401 / 403).
19    #[error("authentication failed ({status}): {message}")]
20    Auth {
21        /// HTTP status.
22        status: u16,
23        /// The provider's message.
24        message: String,
25    },
26
27    /// Rate limited (HTTP 429) or overloaded (HTTP 529), after retries.
28    #[error("{} after {attempts} attempt(s){}: {message}", if *status == 529 { "provider overloaded" } else { "rate limited" }, retry_after.map(|s| format!("; retry in {s}s")).unwrap_or_default())]
29    RateLimited {
30        /// HTTP status (429 or 529).
31        status: u16,
32        /// Seconds the provider asked us to wait, if it said.
33        retry_after: Option<u64>,
34        /// How many requests were sent.
35        attempts: u32,
36        /// The provider's message.
37        message: String,
38    },
39
40    /// Any other non-success HTTP response.
41    #[error("API error {status} ({kind}): {message}")]
42    Api {
43        /// HTTP status.
44        status: u16,
45        /// The provider's error type (`invalid_request_error`, ...).
46        kind: String,
47        /// The provider's message.
48        message: String,
49    },
50
51    /// The request never got a response (DNS, TLS, timeout...).
52    #[error("network error: {0}")]
53    Network(String),
54
55    /// The response arrived but could not be used: not JSON, no JSON
56    /// object in the text, cut off at `max_tokens`, or a refusal.
57    #[error("unusable response: {0}")]
58    BadResponse(String),
59
60    /// The next call could take the run past its cost ceiling, so it was
61    /// not sent.
62    #[error("cost ceiling reached: {} spent, and the next call could cost up to {}, past --max-cost {}; nothing more was sent", crate::cost::usd(*spent), crate::cost::usd(*next), crate::cost::usd(*max))]
63    CostCeiling {
64        /// Dollars spent so far this run.
65        spent: f64,
66        /// The next call's estimated worst case (smallest useful reply).
67        next: f64,
68        /// The ceiling.
69        max: f64,
70    },
71
72    /// An error from the procgen subsystem itself.
73    #[error(transparent)]
74    ProcGen(#[from] flint_procgen::ProcGenError),
75}
76
77impl ProcGenAiError {
78    /// Whether sending the same request again might succeed.
79    pub fn is_retryable(&self) -> bool {
80        matches!(self, Self::RateLimited { .. } | Self::Network(_))
81            || matches!(self, Self::Api { status, .. } if *status >= 500)
82    }
83}
84
85pub type Result<T> = std::result::Result<T, ProcGenAiError>;
