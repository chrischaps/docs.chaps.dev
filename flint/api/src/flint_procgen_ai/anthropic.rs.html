1//! [`AnthropicAgent`]: a [`ProcGenAgent`] that asks Claude.
2//!
3//! Every reply goes through the same gate the engine would apply: its params
4//! are checked against the generator's `param_schema()` and then actually
5//! generated. A reply that fails goes back to Claude with the validator's
6//! report, up to `max_iterations` rounds, so what comes out is a spec the
7//! generator builds and not just JSON that parses.

33/// The model used when a config names none.
34pub const DEFAULT_MODEL: &str = "claude-opus-5-5";
35/// The Messages API endpoint.
36pub const API_URL: &str = "https://api.anthropic.com/v1/messages";
37/// The `anthropic-version` header sent with every request.
38pub const API_VERSION: &str = "2026-10-21";
39/// The environment variable read for the key.
40pub const API_KEY_ENV: &str = "ANTHROPIC_API_KEY";
41/// Beta that lets a request declined by a safety classifier be re-run on
42/// another model inside the same call (`fallbacks: "default"`).
43const FALLBACK_BETA: &str = "server-side-fallback-2026-07-01";
44/// Models that take `fallbacks: "default"`.
45const FALLBACK_MODELS: &[&str] = &[
46    "claude-opus-5-5",
47    "claude-opus-5",
48    "claude-fable-5-1",
49    "claude-sonnet-5-5",
50];
51/// The API's limit on one base64 image is 5 MB of encoded data.
52const MAX_IMAGE_BYTES: usize = 5 * 1024 * 1024 * 3 / 4;
53/// Images and textures are generated at most this wide when only checked or
54/// shown to the model: the check is for errors, not pixels.
55const PROBE_SIZE: i64 = 256;

57/// The key to use: the environment's when it is set and non-empty, else
58/// the config's. The environment wins so a key saved in a project file can
59/// be overridden per shell without editing the file.
60pub fn resolve_api_key(config_key: &str, env_key: Option<&str>) -> Option<String> {
61    env_key
62        .map(str::trim)
63        .filter(|k| !k.is_empty())
64        .or_else(|| Some(config_key.trim()).filter(|k| !k.is_empty()))
65        .map(str::to_string)
66}

233    /// One Messages API call, retried on 429 / 529 / 5xx / network errors.
234    fn call(&self, system: &str, messages: &[Json]) -> Result<Reply> {
235        let mut headers = vec![
236            ("x-api-key".to_string(), self.config.api_key.clone()),
237            ("anthropic-version".to_string(), API_VERSION.to_string()),
238            ("content-type".to_string(), "application/json".to_string()),
239        ];
240        if self.uses_fallback() {
241            headers.push(("anthropic-beta".to_string(), FALLBACK_BETA.to_string()));
242        }
243        let input_estimate = estimate_input_tokens(&self.body(system, messages, self.max_tokens));
244        let mut attempt = 0;
245        loop {
246            attempt += 1;
247            // Checked before every send, retries included: a retry is a new
248            // call, and the budget may have moved since the last check.
249            let max_tokens = match &self.meter {
250                Some(m) => m.admit(&self.config.model, input_estimate, self.max_tokens)?,
251                None => self.max_tokens,
252            };
253            let req = HttpRequest {
254                url: self.url.clone(),
255                headers: headers.clone(),
256                body: self.body(system, messages, max_tokens).to_string(),
257            };
258            let result = match self.transport.post(&req) {
259                Ok(resp) => {
260                    self.meter_response(&resp);
261                    parse_response(&resp, attempt)
262                }
263                Err(e) => Err(ProcGenAiError::Network(e)),
264            };
265            match result {
266                Err(e) if e.is_retryable() && attempt <= self.max_retries => {
267                    let wait = match &e {
268                        ProcGenAiError::RateLimited {
269                            retry_after: Some(s),
270                            ..
271                        } => *s,
272                        _ => 1 << attempt,
273                    };
274                    (self.sleep)(Duration::from_secs(wait.min(60)));
275                }
276                other => return other,
277            }
278        }
279    }

281    /// Record a successful response's usage, read from the raw body so a
282    /// reply cut off at `max_tokens` or declined still counts what it cost.
283    fn meter_response(&self, resp: &HttpResponse) {
284        let Some(meter) = &self.meter else { return };
285        if !(200..300).contains(&resp.status) {
286            return;
287        }
288        let Ok(v) = serde_json::from_str::<Json>(&resp.body) else {
289            return;
290        };
291        let Some(usage) = v.get("usage") else { return };
292        let model = v
293            .get("model")
294            .and_then(Json::as_str)
295            .filter(|m| !m.is_empty())
296            .unwrap_or(&self.config.model);
297        meter.record(model, &Usage::of(usage));
298    }

300    /// Ask, validate, and on failure send the validator's report back, up to
301    /// `max_iterations` rounds. Returns the accepted reply object and its
302    /// params (`None` when the reply set `"params": null` and that's
303    /// allowed). `shape` adds task-specific checks on the reply object.
304    ///
305    /// `messages` ends with the user turn to answer, and on return holds the
306    /// whole exchange, ending with the assistant's last reply, so a caller
307    /// can add a turn and converse again.
308    fn converse(
309        &self,
310        ctx: &Ctx,
311        messages: &mut Vec<Json>,
312        params_required: bool,
313        shape: impl Fn(&Map<String, Json>) -> Vec<String>,
314    ) -> Result<(Map<String, Json>, Option<toml::Value>, Report)> {
315        let system = prompt::system_prompt(&ctx.name, ctx.generator());
316        let mut report = Report::default();
317        let mut last: Option<(Map<String, Json>, Option<toml::Value>)> = None;
318
319        for round in 1..=self.config.max_iterations {
320            let reply = self.call(&system, messages)?;
321            messages.push(json!({"role": "assistant", "content": reply.content}));
322
323            let mut issues = Vec::new();
324            match extract_json_object(&reply.text) {
325                Err(e) => issues.push(format!("{e}; reply with a single JSON object")),
326                Ok(obj) => {
327                    issues.extend(shape(&obj));
328                    let params = match obj.get("params") {
329                        None | Some(Json::Null) if !params_required => None,
330                        None | Some(Json::Null) => {
331                            issues.push("the reply has no `params` object".to_string());
332                            None
333                        }
334                        Some(v) => match ctx.params_from_json(v) {
335                            Some(p) => {
336                                issues.extend(ctx.issues(&p));
337                                Some(p)
338                            }
339                            None => {
340                                issues.push("`params` must be a JSON object".to_string());
341                                None
342                            }
343                        },
344                    };
345                    last = Some((obj, params));
346                }
347            }
348            report.rounds.push(Round {
349                issues: issues.clone(),
350                usage: reply.usage,
351                model: reply.model,
352            });
353            if issues.is_empty() {
354                let (obj, params) = last.expect("an accepted round parsed");
355                return Ok((obj, params, report));
356            }
357            if round < self.config.max_iterations {
358                messages.push(json!({
359                    "role": "user",
360                    "content": [{"type": "text", "text": prompt::feedback(round, &issues)}],
361                }));
362            }
363        }
364
365        // Out of rounds. Repair what the schema can fix; refuse what it can't.
366        let remaining = report
367            .rounds
368            .last()
369            .map(|r| r.issues.join("; "))
370            .unwrap_or_default();
371        let rounds = report.rounds.len();
372        let fail = || {
373            ProcGenAiError::AgentError(format!(
374                "no valid reply after {rounds} round(s); last issues: {remaining}"
375            ))
376        };
377        let Some((obj, params)) = last else {
378            return Err(fail());
379        };
380        if !shape(&obj).is_empty() {
381            return Err(fail());
382        }
383        let params = match params {
384            Some(mut p) => {
385                report.repaired = repair_params(&ctx.schema, &mut p);
386                if !ctx.generation_issues(&p).is_empty() {
387                    return Err(fail());
388                }
389                Some(p)
390            }
391            None if params_required => return Err(fail()),
392            None => None,
393        };
394        Ok((obj, params, report))
395    }

397    /// [`ProcGenAgent::create_spec_from_prompt`], with the conversation's
398    /// [`Report`].
399    pub fn create_spec_reported(
400        &self,
401        prompt_text: &str,
402        generator: &str,
403    ) -> Result<(ProcGenSpec, Report)> {
404        let ctx = Ctx::new(generator)?;
405        self.set_meter_context(&slug(prompt_text), "create");
406        let mut messages = vec![user_turn(vec![text_block(&prompt::create_task(
407            prompt_text,
408        ))])];
409        let (obj, params, mut report) = self.converse(&ctx, &mut messages, true, |_| Vec::new())?;
410        let params = params.expect("create requires params");
411
412        let name = obj
413            .get("name")
414            .and_then(Json::as_str)
415            .map(slug)
416            .filter(|s| s != "untitled")
417            .unwrap_or_else(|| slug(prompt_text));
418        let mut tags = vec!["ai-generated".to_string(), "anthropic".to_string()];
419        for t in strings(&obj, "tags").into_iter().take(8) {
420            let t = slug(&t);
421            if !tags.contains(&t) {
422                tags.push(t);
423            }
424        }
425        report.notes = strings(&obj, "rationale");
426        let seed = fnv1a(&[ctx.name.as_bytes(), prompt_text.as_bytes()]) % 100_000;
427        let spec = ProcGenSpec {
428            meta: SpecMeta {
429                name,
430                description: Some(prompt_text.to_string()),
431                version: "1.0.0".to_string(),
432                tags,
433            },
434            generator: ctx.name.clone(),
435            seed: SeedConfig {
436                mode: SeedMode::Fixed,
437                value: Some(seed),
438                derive_from: None,
439            },
440            params,
441            lod: None,
442        };
443        Ok((spec, report))
444    }

446    /// [`ProcGenAgent::interpret_spec`], with the conversation's [`Report`].
447    pub fn interpret_spec_reported(&self, spec: &ProcGenSpec) -> Result<(SpecFeedback, Report)> {
448        let ctx = match Ctx::new(&spec.generator) {
449            Ok(ctx) => ctx,
450            // Nothing to ask Claude about: the spec names no generator.
451            Err(e) => {
452                return Ok((
453                    SpecFeedback {
454                        suggestions: vec![e.to_string()],
455                        confidence: 0.0,
456                        revised_spec: None,
457                    },
458                    Report::default(),
459                ))
460            }
461        };
462        let mut findings = Vec::new();
463        if ctx.name != spec.generator {
464            findings.push(format!(
465                "generator \"{}\" is registered as \"{}\"",
466                spec.generator, ctx.name
467            ));
468        }
469        findings.extend(ctx.issues(&spec.params));
470        if spec.seed.mode == SeedMode::Random {
471            findings.push("the seed is random, so every run builds a different asset".into());
472        }
473
474        self.set_meter_context(&spec.meta.name, "interpret");
475        let mut messages = vec![user_turn(vec![text_block(&prompt::interpret_task(
476            spec, &findings,
477        ))])];
478        let shape = |obj: &Map<String, Json>| {
479            if strings(obj, "suggestions").is_empty() {
480                vec!["`suggestions` must be a non-empty array of strings".to_string()]
481            } else {
482                Vec::new()
483            }
484        };
485        let (obj, params, report) = self.converse(&ctx, &mut messages, false, shape)?;
486
487        let confidence = obj
488            .get("confidence")
489            .and_then(Json::as_f64)
490            .map_or(0.5, |c| c.clamp(0.0, 1.0)) as f32;
491        let mut revised = spec.clone();
492        revised.generator = ctx.name.clone();
493        if let Some(p) = params {
494            revised.params = p;
495        }
496        if spec.seed.mode == SeedMode::Random {
497            revised.seed = SeedConfig {
498                mode: SeedMode::Fixed,
499                value: Some(fnv1a(&[spec.meta.name.as_bytes()]) % 100_000),
500                derive_from: None,
501            };
502        }
503        let changed = revised != *spec;
504        if changed {
505            revised.meta.version = bump_patch(&revised.meta.version);
506        }
507        Ok((
508            SpecFeedback {
509                suggestions: strings(&obj, "suggestions"),
510                confidence,
511                revised_spec: changed.then_some(revised),
512            },
513            report,
514        ))
515    }

517    /// [`ProcGenAgent::refine_spec`], with the conversation's [`Report`].
518    ///
519    /// With a reference, Claude also sees what the spec builds now beside
520    /// it: a texture's own pixels, or whatever the [`Previewer`] draws (a
521    /// mesh render). When both images can be measured the refine is a
522    /// loop: each round's result is pictured and measured
523    /// ([`measure::distance`]), and while the distance falls the new
524    /// picture and numbers go back for another round, up to
525    /// [`Self::refine_rounds`]. The first round that doesn't get closer
526    /// ends it, and the closest spec is returned: the input itself
527    /// (`report.kept == 0`) when no round improved on it.
528    pub fn refine_spec_reported(
529        &self,
530        spec: &ProcGenSpec,
531        reference: Option<&[u8]>,
532    ) -> Result<(ProcGenSpec, Report)> {
533        let ctx = Ctx::new(&spec.generator)?;
534        self.set_meter_context(&spec.meta.name, "refine");
535        let mut turn = Vec::new();
536        if let Some(bytes) = reference {
537            turn.push(image_block(bytes)?);
538        }
539        let (current, shown) = match reference {
540            Some(_) => self.picture(&ctx, spec),
541            None => (None, Current::None),
542        };
543        if let Some((_, block)) = &current {
544            turn.push(block.clone());
545        }
546        let measure_of = |img: &[u8]| reference.and_then(|r| measure::distance(r, img));
547        // Eyes misjudge the overall tone of a small image; numbers don't.
548        let start = current.as_ref().and_then(|(img, _)| measure_of(img));
549        turn.push(text_block(&prompt::refine_task(
550            spec,
551            reference.is_some(),
552            shown,
553            start.as_ref(),
554        )));
555        let mut messages = vec![user_turn(turn)];
556
557        let mut report = Report::default();
558        report.distances.extend(start);
559        let rounds = if start.is_some() {
560            self.refine_rounds.max(1)
561        } else {
562            1
563        };
564        let mut best: Option<(ProcGenSpec, Distance)> = None;
565        for round in 1..=rounds {
566            let (obj, params, r) = self.converse(&ctx, &mut messages, true, |_| Vec::new())?;
567            report.rounds.extend(r.rounds);
568            report.repaired += r.repaired;
569            let mut refined = spec.clone();
570            refined.generator = ctx.name.clone();
571            refined.params = params.expect("refine requires params");
572            refined.meta.version = bump_patch(&refined.meta.version);
573            let changes = strings(&obj, "changes");
574
575            // Unmeasured (no start, or this round's result can't be pictured
576            // or measured): keep what was asked for, as a single shot would.
577            let before = best.as_ref().map(|(_, d)| *d).or(start);
578            let measured = before.and_then(|before| {
579                let (shot, _) = self.picture(&ctx, &refined);
580                let (img, block) = shot?;
581                Some((before, measure_of(&img)?, block))
582            });
583            let Some((before, after, block)) = measured else {
584                report.notes = changes;
585                report.kept = round as usize;
586                return Ok((refined, report));
587            };
588            report.distances.push(after);
589            if after.score >= before.score {
590                if best.is_none() {
591                    report.notes = vec![format!(
592                        "round {round} moved the asset no closer (distance {:.1} -> {:.1}); the spec is returned unchanged",
593                        before.score, after.score
594                    )];
595                }
596                break;
597            }
598            report.notes = changes;
599            report.kept = round as usize;
600            best = Some((refined, after));
601            if round < rounds {
602                messages.push(user_turn(vec![
603                    block,
604                    text_block(&prompt::refine_progress(round, &before, &after, shown)),
605                ]));
606            }
607        }
608        Ok((best.map_or_else(|| spec.clone(), |(s, _)| s), report))
609    }

611    /// What `spec` builds, as an encoded image and its message block: the
612    /// [`Previewer`]'s picture when it gives one, else a texture's own
613    /// pixels. `(None, Current::None)` when neither can show it.
614    fn picture(&self, ctx: &Ctx, spec: &ProcGenSpec) -> (Option<(Vec<u8>, Json)>, Current) {
615        let texture = ctx.generator().output_kind() == OutputKind::Image;
616        let drawn = self
617            .previewer
618            .as_ref()
619            .and_then(|p| p.preview(spec))
620            .map(|img| {
621                (
622                    img,
623                    if texture {
624                        Current::Texture
625                    } else {
626                        Current::Render
627                    },
628                )
629            });
630        let found = drawn.or_else(|| {
631            texture
632                .then(|| ctx.preview_png(&spec.params))
633                .flatten()
634                .map(|img| (img, Current::Texture))
635        });
636        match found.and_then(|(img, kind)| image_block(&img).ok().map(|b| ((img, b), kind))) {
637            Some((pair, kind)) => (Some(pair), kind),
638            None => (None, Current::None),
639        }
640    }
641}

677/// The generator a conversation is about, and how to validate against it.
678struct Ctx {
679    registry: GeneratorRegistry,
680    name: String,
681    schema: Json,
682}
683
684impl Ctx {
685    fn new(generator: &str) -> Result<Self> {
686        let registry = registry();
687        let name = resolve_name(&registry, generator)
688            .ok_or_else(|| unknown_generator(&registry, generator))?;
689        let schema = registry
690            .get(&name)
691            .expect("resolved names are registered")
692            .param_schema();
693        Ok(Self {
694            registry,
695            name,
696            schema,
697        })
698    }
699
700    fn generator(&self) -> &dyn Generator {
701        self.registry
702            .get(&self.name)
703            .expect("resolved names are registered")
704    }
705
706    /// Claude's JSON params as TOML, with whole numbers written for `number`
707    /// params turned into floats (generators read those with `as_float`).
708    fn params_from_json(&self, v: &Json) -> Option<toml::Value> {
709        let obj = v.as_object()?;
710        let props = props(&self.schema);
711        let mut table = toml::map::Map::new();
712        for (k, v) in obj {
713            let Some(mut t) = json_to_toml(v) else {
714                continue;
715            };
716            let is_number = props
717                .and_then(|p| p.get(k))
718                .is_some_and(|p| prop_type(p) == "number");
719            if let (true, toml::Value::Integer(i)) = (is_number, &t) {
720                t = toml::Value::Float(*i as f64);
721            }
722            table.insert(k.clone(), t);
723        }
724        Some(toml::Value::Table(table))
725    }
726
727    /// Schema issues, then (if there are none) what generating them reveals.
728    fn issues(&self, params: &toml::Value) -> Vec<String> {
729        let schema: Vec<String> = check_params(&self.schema, params)
730            .iter()
731            .map(ToString::to_string)
732            .collect();
733        if !schema.is_empty() {
734            return schema;
735        }
736        self.generation_issues(params)
737    }
738
739    /// A spec for these params, small enough to build quickly.
740    fn probe(&self, params: &toml::Value, size: i64) -> ProcGenSpec {
741        let mut params = params.clone();
742        if self.generator().output_kind() == OutputKind::Image {
743            if let Some(t) = params.as_table_mut() {
744                for key in ["width", "height"] {
745                    let w = t.get(key).and_then(toml::Value::as_integer).unwrap_or(size);
746                    t.insert(key.to_string(), toml::Value::Integer(w.min(size)));
747                }
748            }
749        }
750        ProcGenSpec {
751            meta: SpecMeta {
752                name: "probe".to_string(),
753                description: None,
754                version: "1.0.0".to_string(),
755                tags: Vec::new(),
756            },
757            generator: self.name.clone(),
758            seed: SeedConfig {
759                mode: SeedMode::Fixed,
760                value: Some(1),
761                derive_from: None,
762            },
763            params,
764            lod: None,
765        }
766    }
767
768    /// Build the params and report a failure or an empty result.
769    fn generation_issues(&self, params: &toml::Value) -> Vec<String> {
770        let empty = "the generator built an empty mesh (0 vertices); the params describe nothing";
771        match self
772            .registry
773            .generate_from_spec(&self.probe(params, PROBE_SIZE))
774        {
775            Err(e) => vec![format!("the generator rejected the params: {e}")],
776            Ok(GeneratorOutput::Mesh(m)) if m.vertices.is_empty() => vec![empty.to_string()],
777            Ok(GeneratorOutput::MeshWithLods(l))
778                if l.first().is_none_or(|m| m.vertices.is_empty()) =>
779            {
780                vec![empty.to_string()]
781            }
782            Ok(GeneratorOutput::SkinnedMesh(m)) if m.vertices.is_empty() => {
783                vec![empty.to_string()]
784            }
785            Ok(_) => Vec::new(),
786        }
787    }
788
789    /// The first image these params build, as a PNG, for the model to see.
790    fn preview_png(&self, params: &toml::Value) -> Option<Vec<u8>> {
791        let image = match self
792            .registry
793            .generate_from_spec(&self.probe(params, PROBE_SIZE))
794            .ok()?
795        {
796            GeneratorOutput::Image(i) => i,
797            GeneratorOutput::ImageSet(mut s) if !s.is_empty() => s.swap_remove(0),
798            _ => return None,
799        };
800        let rgba = image::RgbaImage::from_raw(image.width, image.height, image.pixels)?;
801        let mut png = std::io::Cursor::new(Vec::new());
802        rgba.write_to(&mut png, image::ImageFormat::Png).ok()?;
803        Some(png.into_inner())
804    }
805}

807fn user_turn(content: Vec<Json>) -> Json {
808    json!({"role": "user", "content": content})
809}
810
811fn text_block(text: &str) -> Json {
812    json!({"type": "text", "text": text})
813}
814
815fn image_block(bytes: &[u8]) -> Result<Json> {
816    let media_type = image_media_type(bytes).ok_or_else(|| {
817        ProcGenAiError::SpecError("the reference image is not PNG, JPEG, GIF or WebP".to_string())
818    })?;
819    if bytes.len() > MAX_IMAGE_BYTES {
820        return Err(ProcGenAiError::SpecError(format!(
821            "the reference image is {:.1} MB; the API takes at most {:.1} MB",
822            bytes.len() as f64 / 1_048_576.0,
823            MAX_IMAGE_BYTES as f64 / 1_048_576.0
824        )));
825    }
826    Ok(json!({
827        "type": "image",
828        "source": {"type": "base64", "media_type": media_type, "data": base64(bytes)},
829    }))
830}
831
832/// A string field, or each string in an array field.
833fn strings(obj: &Map<String, Json>, key: &str) -> Vec<String> {
834    match obj.get(key) {
835        Some(Json::String(s)) if !s.trim().is_empty() => vec![s.trim().to_string()],
836        Some(Json::Array(a)) => a
837            .iter()
838            .filter_map(Json::as_str)
839            .map(str::trim)
840            .filter(|s| !s.is_empty())
841            .map(|s| shorten(s, 600))
842            .collect(),
843        _ => Vec::new(),
844    }
845}
