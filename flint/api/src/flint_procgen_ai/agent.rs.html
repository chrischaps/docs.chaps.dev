5/// Feedback from an AI agent after analyzing a spec.
6#[derive(Debug, Clone)]
7pub struct SpecFeedback {
8    /// Suggested improvements or observations.
9    pub suggestions: Vec<String>,
10    /// Agent's confidence in the spec quality (0.0 = low, 1.0 = high).
11    pub confidence: f32,
12    /// An optional revised spec incorporating the suggestions.
13    pub revised_spec: Option<ProcGenSpec>,
14}
15
16/// An AI agent that can create, interpret, and refine procedural generation specs.
17///
18/// Object-safe (`Box<dyn ProcGenAgent>` works). Implementations must be `Send + Sync`
19/// to support future async/threaded usage.
20pub trait ProcGenAgent: Send + Sync {
21    /// Analyze a spec and return feedback with suggestions.
22    fn interpret_spec(&self, spec: &ProcGenSpec) -> Result<SpecFeedback>;
23
24    /// Create a new spec from a natural-language prompt and target generator type.
25    fn create_spec_from_prompt(&self, prompt: &str, generator: &str) -> Result<ProcGenSpec>;
26
27    /// Refine an existing spec, optionally using a reference image for guidance.
28    fn refine_spec(&self, spec: &ProcGenSpec, reference: Option<&[u8]>) -> Result<ProcGenSpec>;
29}
