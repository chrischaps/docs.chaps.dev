7/// Feedback from an AI agent after analyzing a spec.
8#[derive(Debug, Clone)]
9pub struct SpecFeedback {
10    /// Suggested improvements or observations.
11    pub suggestions: Vec<String>,
12    /// Agent's confidence in the spec quality (0.0 = low, 1.0 = high).
13    pub confidence: f32,
14    /// An optional revised spec incorporating the suggestions.
15    pub revised_spec: Option<ProcGenSpec>,
16}
17
18/// One request/response round of a conversation.
19#[derive(Debug, Clone, Default)]
20pub struct Round {
21    /// What the validator found in this round's reply; empty when accepted.
22    pub issues: Vec<String>,
23    /// Tokens this round cost.
24    pub usage: Usage,
25    /// The model that answered.
26    pub model: String,
27}
28
29/// How a conversation went. Agents that make no calls return an empty one.
30#[derive(Debug, Clone, Default)]
31pub struct Report {
32    /// Each round, in order.
33    pub rounds: Vec<Round>,
34    /// Params the schema repair had to fix after the last round (0 when
35    /// the agent converged on its own).
36    pub repaired: usize,
37    /// The model's own account of its choices (`rationale` / `changes`).
38    /// For a measured refine, the changes of the round that was kept.
39    pub notes: Vec<String>,
40    /// A refine's measured distance to the reference, per refine round:
41    /// `[0]` is the spec as it came in, `[i]` what round `i` built. Empty
42    /// when nothing could be measured (no reference, or no picture of
43    /// what the spec builds).
44    pub distances: Vec<Distance>,
45    /// Which refine round's spec was returned: 0 for the spec as it came in
46    /// (no round moved it closer), `i` for round `i`. A refine that isn't
47    /// measured always keeps its one round. 0 for create and interpret.
48    pub kept: usize,
49}
50
51impl Report {
52    /// Whether the last round's reply passed validation unaided.
53    pub fn converged(&self) -> bool {
54        self.rounds.last().is_some_and(|r| r.issues.is_empty())
55    }
56
57    /// Tokens across all rounds.
58    pub fn usage(&self) -> Usage {
59        self.rounds
60            .iter()
61            .fold(Usage::default(), |a, r| a.plus(r.usage))
62    }
63}

65/// An AI agent that can create, interpret, and refine procedural generation specs.
66///
67/// Object-safe (`Box<dyn ProcGenAgent>` works). Implementations must be `Send + Sync`
68/// to support future async/threaded usage.
69pub trait ProcGenAgent: Send + Sync {
70    /// Analyze a spec and return feedback with suggestions.
71    fn interpret_spec(&self, spec: &ProcGenSpec) -> Result<SpecFeedback>;
72
73    /// Create a new spec from a natural-language prompt and target generator type.
74    fn create_spec_from_prompt(&self, prompt: &str, generator: &str) -> Result<ProcGenSpec>;
75
76    /// Refine an existing spec, optionally using a reference image for guidance.
77    fn refine_spec(&self, spec: &ProcGenSpec, reference: Option<&[u8]>) -> Result<ProcGenSpec>;
78
79    /// [`Self::create_spec_from_prompt`] with how the conversation went.
80    /// The default reports nothing; agents that talk to a model override it.
81    fn create_spec_with_report(
82        &self,
83        prompt: &str,
84        generator: &str,
85    ) -> Result<(ProcGenSpec, Report)> {
86        Ok((
87            self.create_spec_from_prompt(prompt, generator)?,
88            Report::default(),
89        ))
90    }
91
92    /// [`Self::interpret_spec`] with how the conversation went.
93    fn interpret_spec_with_report(&self, spec: &ProcGenSpec) -> Result<(SpecFeedback, Report)> {
94        Ok((self.interpret_spec(spec)?, Report::default()))
95    }
96
97    /// [`Self::refine_spec`] with how the conversation went.
98    fn refine_spec_with_report(
99        &self,
100        spec: &ProcGenSpec,
101        reference: Option<&[u8]>,
102    ) -> Result<(ProcGenSpec, Report)> {
103        Ok((self.refine_spec(spec, reference)?, Report::default()))
104    }
105}
