1//! [`Previewer`]: how a caller lets [`crate::AnthropicAgent`] see what a spec
2//! builds.
3//!
4//! This crate is tool-time and depends on `flint-procgen` only, so it can
5//! show Claude a texture (the generator's pixels are the asset) but cannot
6//! draw a mesh. A caller that can render, such as `flint-cli` with the
7//! headless model renderer, implements this trait and hands it to
8//! [`crate::AnthropicAgent::with_previewer`].
9
10use flint_procgen::ProcGenSpec;
11
12/// Renders a spec to an encoded image (PNG or JPEG) for the model to look at
13/// and for the refine loop to measure against the reference.
14///
15/// Return `None` when the spec can't be previewed (a generator this
16/// previewer doesn't draw, a GPU that isn't there). The agent then falls
17/// back to what it can show by itself, which for a mesh is nothing.
18///
19/// Every closure `Fn(&ProcGenSpec) -> Option<Vec<u8>>` is a previewer.
20pub trait Previewer: Send + Sync {
21    /// The image `spec` builds, encoded.
22    fn preview(&self, spec: &ProcGenSpec) -> Option<Vec<u8>>;
23}
24
25impl<F> Previewer for F
26where
27    F: Fn(&ProcGenSpec) -> Option<Vec<u8>> + Send + Sync,
28{
29    fn preview(&self, spec: &ProcGenSpec) -> Option<Vec<u8>> {
30        self(spec)
31    }
32}
