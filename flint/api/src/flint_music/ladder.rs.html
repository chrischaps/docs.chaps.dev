1//! The disintegration ladder: ordered rungs with coherence thresholds and
2//! per-rung parameters, mirrored across audio and visuals (ADR 0015).
3//!
4//! The ladder is data, not code: `config/ladder.toml` in the game repo,
5//! hot-reloadable mid-session like the coherence config. Each tick the
6//! harness feeds the coherence scalar to [`Ladder::observe`]; the resolved
7//! [`LadderParams`] for the current rung is the *single source of truth* —
8//! the audio driver applies its audio half as mixer tweens and the visual
9//! frame carries its visual half, which is what keeps the world coming
10//! apart (and reassembling) as one thing.
11//!
12//! Rung parameters are absolute, not cumulative: each rung states the full
13//! degraded state, so a deeper rung's numbers include the shallower rungs'
14//! degradation. Hysteresis lives in the thresholds (`enter_below` <
15//! `exit_above`), so a noisy coherence value at a boundary never flickers
16//! the world.
17//!
18//! Protected buses: the ladder never touches `foundation` (the world's
19//! anchor — "last to go" means it never goes in the prototype) or the motif
20//! buses (`home_theme`/`child_motif` — the narrative lines reintegration
21//! leads with) via gain or dropout. The low-pass applies to every non-motif
22//! bus, foundation included — filtering the whole world is the woozy intent;
23//! silencing its pulse is not.
24

42#[derive(Debug, Clone, PartialEq)]
43pub struct RungAudio {
44    /// Low-pass cutoff applied to every non-motif bus.
45    pub lpf_hz: f64,
46    /// Per-bus gain trims in dB (degradable buses only).
47    pub thin_db: BTreeMap<String, f32>,
48    /// Buses silenced outright at this rung (degradable buses only).
49    pub drop: Vec<String>,
50    /// Detune warble LFO depth (semitones, peak) on non-motif buses.
51    pub warble_depth_semitones: f64,
52    /// Warble LFO rate in Hz.
53    pub warble_rate_hz: f64,
54    /// Buses whose composed degraded alternate takes over at this rung
55    /// (ADR 0032; degradable buses only, and only where the manifest
56    /// authored one — elsewhere it is a silent no-op).
57    pub alternates: Vec<String>,
58}

96#[derive(Debug, Clone, Copy, PartialEq)]
97pub struct FullFail {
98    /// Coherence below this starts (or continues) the full-fail hold. Must
99    /// sit below the deepest rung's `enter_below`; per ADR 0010 only
100    /// sustained pulse misses reach values down there. Under sustained
101    /// neglect the value saw-tooths (miss impulses down, the tracking
102    /// integrator back up), so the hold has its own hysteresis:
103    pub enter_below: f64,
104    /// The hold cancels only when coherence rises above this (> enter_below)
105    /// — a sawtooth peak poking over `enter_below` does not save the world.
106    pub exit_above: f64,
107    /// Debounce: how long the hold must run before reintegration fires.
108    pub hold_ms: f64,
109}

111#[derive(Debug, Clone, PartialEq)]
112pub struct LadderConfig {
113    pub rungs: Vec<Rung>,
114    pub full_fail: FullFail,
115    /// Authored seam envelope (ms): the fade that carries the world out of
116    /// the old timeline at reintegration. An envelope, not a cut.
117    pub seam_fade_ms: f64,
118    /// Rewind gesture length in beats (at the tempo where the fail happens):
119    /// the whole world spins down like a record played backwards (a
120    /// playback-rate ramp on every stem, mirrored visually) for exactly this
121    /// long, ending at the seam — which still lands on a bar line, so the
122    /// gesture starts on a beat. 0 = no spin-down.
123    pub seam_rewind_beats: f64,
124    /// Where the spin-down lands, in semitones (negative; −30 ≈ 18% speed —
125    /// audibly "stopped" by the time the seam fade takes the tail).
126    pub seam_rewind_drop_st: f64,
127    /// Pickup spin-up: in the last beats before the seam, the lead bus plays
128    /// the re-entry material as a cue winding up from the spin-down's rate
129    /// to full speed, arriving exactly on the re-entry downbeat — a musical
130    /// pickup that telegraphs when to come back in. Beats are measured at
131    /// the re-entry tempo. 0 disables.
132    pub seam_pickup_beats: f64,
133    /// Lead-in: the ensemble re-enters this many beats BEFORE the checkpoint
134    /// downbeat, playing the pickup into the phrase at full lock — "3, 4,
135    /// go" — so the player gets prep time before content resumes (P5, Chris,
136    /// 2026-08-26). Beats at the re-entry tempo; judgment windows inside the
137    /// lead-in re-open. 0 (default) = enter exactly on the checkpoint
138    /// downbeat, byte-identical to the pre-lead-in behavior. Clamped so it
139    /// never reaches before suite sample 0.
140    pub seam_lead_in_beats: f64,
141    /// The ladder arms when coherence first reaches this (sessions open at a
142    /// neutral value inside rung territory; the world can only come apart
143    /// after it has first cohered). Measured on real engaged play: coherence
144    /// reaches 0.80 within a learning session's first half-minute.
145    pub arm_above: f64,
146    /// Crossfade length (ms) for bus↔alternate handovers (ADR 0032;
147    /// `[alternates] xfade_ms`).
148    pub alternate_xfade_ms: f64,
149}

258    pub fn parse(text: &str) -> Result<Self> {
259        let (root, _) =
260            config_toml::parse_versioned("ladder config", text, VersionPolicy::DefaultZeroStrict)?;
261        let d = Self::default().full_fail;
262        let ff = root.get("full_fail");
263        let ff_f = |key: &str, default: f64| {
264            ff.and_then(|t| t.get(key))
265                .and_then(toml_f64)
266                .unwrap_or(default)
267        };
268        let enter_below = ff_f("enter_below", d.enter_below);
269        let full_fail = FullFail {
270            enter_below,
271            exit_above: ff_f("exit_above", (enter_below + 0.10).min(1.0)),
272            hold_ms: ff_f("hold_ms", d.hold_ms),
273        };
274
275        let mut rungs = Vec::new();
276        if let Some(arr) = root.get("rungs").and_then(|v| v.as_array()) {
277            for (i, r) in arr.iter().enumerate() {
278                let field = |key: &str| r.get(key).and_then(toml_f64);
279                let req = |key: &str| {
280                    field(key).ok_or_else(|| {
281                        FlintError::ValidationError(format!(
282                            "ladder config: rung {i} missing `{key}`"
283                        ))
284                    })
285                };
286                let audio_t = r.get("audio");
287                let af = |key: &str, default: f64| {
288                    audio_t
289                        .and_then(|t| t.get(key))
290                        .and_then(toml_f64)
291                        .unwrap_or(default)
292                };
293                let mut thin_db = BTreeMap::new();
294                if let Some(tbl) = audio_t
295                    .and_then(|t| t.get("thin_db"))
296                    .and_then(|v| v.as_table())
297                {
298                    for (bus, v) in tbl {
299                        let db = toml_f64(v).ok_or_else(|| {
300                            FlintError::ValidationError(format!(
301                                "ladder config: rung {i} thin_db.{bus} not a number"
302                            ))
303                        })?;
304                        thin_db.insert(bus.clone(), db as f32);
305                    }
306                }
307                let drop: Vec<String> = audio_t
308                    .and_then(|t| t.get("drop"))
309                    .and_then(|v| v.as_array())
310                    .map(|a| {
311                        a.iter()
312                            .filter_map(|v| v.as_str().map(String::from))
313                            .collect()
314                    })
315                    .unwrap_or_default();
316                let alternates: Vec<String> = audio_t
317                    .and_then(|t| t.get("alternates"))
318                    .and_then(|v| v.as_array())
319                    .map(|a| {
320                        a.iter()
321                            .filter_map(|v| v.as_str().map(String::from))
322                            .collect()
323                    })
324                    .unwrap_or_default();
325                let visual_t = r.get("visual");
326                let vf = |key: &str| {
327                    visual_t
328                        .and_then(|t| t.get(key))
329                        .and_then(toml_f64)
330                        .unwrap_or(0.0) as f32
331                };
332                rungs.push(Rung {
333                    name: r
334                        .get("name")
335                        .and_then(|v| v.as_str())
336                        .map(String::from)
337                        .unwrap_or_else(|| format!("rung{}", i + 1)),
338                    enter_below: req("enter_below")?,
339                    exit_above: req("exit_above")?,
340                    ramp_ms: field("ramp_ms").unwrap_or(350.0),
341                    audio: RungAudio {
342                        lpf_hz: af("lpf_hz", LPF_OPEN_HZ),
343                        thin_db,
344                        drop,
345                        warble_depth_semitones: af("warble_depth_semitones", 0.0),
346                        warble_rate_hz: af("warble_rate_hz", 0.0),
347                        alternates,
348                    },
349                    visual: RungVisual {
350                        desaturate: vf("desaturate"),
351                        blur: vf("blur"),
352                        chromatic: vf("chromatic"),
353                    },
354                });
355            }
356        }
357        if rungs.is_empty() {
358            rungs = Self::default().rungs;
359        }
360        let d_all = Self::default();
361        let seam_t = root.get("seam");
362        let seam_fade_ms = seam_t
363            .and_then(|t| t.get("fade_ms"))
364            .and_then(toml_f64)
365            .unwrap_or(d_all.seam_fade_ms);
366        let seam_rewind_beats = seam_t
367            .and_then(|t| t.get("rewind_beats"))
368            .and_then(toml_f64)
369            .unwrap_or(d_all.seam_rewind_beats);
370        let seam_rewind_drop_st = seam_t
371            .and_then(|t| t.get("rewind_drop_semitones"))
372            .and_then(toml_f64)
373            .unwrap_or(d_all.seam_rewind_drop_st);
374        let seam_pickup_beats = seam_t
375            .and_then(|t| t.get("pickup_beats"))
376            .and_then(toml_f64)
377            .unwrap_or(d_all.seam_pickup_beats);
378        let seam_lead_in_beats = seam_t
379            .and_then(|t| t.get("lead_in_beats"))
380            .and_then(toml_f64)
381            .unwrap_or(d_all.seam_lead_in_beats);
382        let arm_above = root
383            .get("arm_above")
384            .and_then(toml_f64)
385            .unwrap_or_else(|| Self::default().arm_above);
386        let alternate_xfade_ms = root
387            .get("alternates")
388            .and_then(|t| t.get("xfade_ms"))
389            .and_then(toml_f64)
390            .unwrap_or(d_all.alternate_xfade_ms);
391        let cfg = Self {
392            rungs,
393            full_fail,
394            seam_fade_ms,
395            seam_rewind_beats,
396            seam_rewind_drop_st,
397            seam_pickup_beats,
398            seam_lead_in_beats,
399            arm_above,
400            alternate_xfade_ms,
401        };
402        cfg.validate()?;
403        Ok(cfg)
404    }

406    fn validate(&self) -> Result<()> {
407        let err = |msg: String| Err(FlintError::ValidationError(format!("ladder config: {msg}")));
408        let mut prev_enter = f64::INFINITY;
409        for (i, r) in self.rungs.iter().enumerate() {
410            if !(0.0..=1.0).contains(&r.enter_below) || !(0.0..=1.0).contains(&r.exit_above) {
411                return err(format!("rung {i} thresholds must be in 0..1"));
412            }
413            if r.exit_above <= r.enter_below {
414                return err(format!(
415                    "rung {i} needs hysteresis: exit_above {} <= enter_below {}",
416                    r.exit_above, r.enter_below
417                ));
418            }
419            if r.enter_below >= prev_enter {
420                return err(format!("rung {i} enter_below must strictly decrease"));
421            }
422            prev_enter = r.enter_below;
423            if !r.ramp_ms.is_finite() || r.ramp_ms < 0.0 {
424                return err(format!("rung {i} ramp_ms {} out of range", r.ramp_ms));
425            }
426            if !(1.0..=LPF_OPEN_HZ).contains(&r.audio.lpf_hz) {
427                return err(format!("rung {i} lpf_hz {} out of range", r.audio.lpf_hz));
428            }
429            for bus in r
430                .audio
431                .thin_db
432                .keys()
433                .chain(r.audio.drop.iter())
434                .chain(r.audio.alternates.iter())
435            {
436                if !DEGRADABLE_BUSES.contains(&bus.as_str()) {
437                    return err(format!(
438                        "rung {i} touches protected bus '{bus}' (degradable: {DEGRADABLE_BUSES:?})"
439                    ));
440                }
441            }
442            if r.audio.warble_depth_semitones < 0.0 || r.audio.warble_depth_semitones > 1.0 {
443                return err(format!(
444                    "rung {i} warble depth out of range (0..1 semitones)"
445                ));
446            }
447            if r.audio.warble_rate_hz < 0.0 || r.audio.warble_rate_hz > 8.0 {
448                return err(format!("rung {i} warble rate out of range (0..8 Hz)"));
449            }
450        }
451        let deepest = self.rungs.last().map(|r| r.enter_below).unwrap_or(0.0);
452        if self.full_fail.enter_below >= deepest {
453            return err(format!(
454                "full_fail.enter_below {} must sit below the deepest rung's enter_below {deepest}",
455                self.full_fail.enter_below
456            ));
457        }
458        if self.full_fail.exit_above <= self.full_fail.enter_below {
459            return err(format!(
460                "full_fail needs hysteresis: exit_above {} <= enter_below {}",
461                self.full_fail.exit_above, self.full_fail.enter_below
462            ));
463        }
464        if !self.full_fail.hold_ms.is_finite() || self.full_fail.hold_ms < 0.0 {
465            return err(format!("full_fail.hold_ms {}", self.full_fail.hold_ms));
466        }
467        if !(5.0..=200.0).contains(&self.seam_fade_ms) {
468            return err(format!(
469                "seam.fade_ms {} out of range (5..200)",
470                self.seam_fade_ms
471            ));
472        }
473        if !(0.0..=16.0).contains(&self.seam_rewind_beats) {
474            return err(format!(
475                "seam.rewind_beats {} out of range (0..16)",
476                self.seam_rewind_beats
477            ));
478        }
479        if !(-60.0..=0.0).contains(&self.seam_rewind_drop_st) {
480            return err(format!(
481                "seam.rewind_drop_semitones {} out of range (-60..0)",
482                self.seam_rewind_drop_st
483            ));
484        }
485        if !(0.0..=8.0).contains(&self.seam_pickup_beats) {
486            return err(format!(
487                "seam.pickup_beats {} out of range (0..8)",
488                self.seam_pickup_beats
489            ));
490        }
491        if !(0.0..=8.0).contains(&self.seam_lead_in_beats) {
492            return err(format!(
493                "seam.lead_in_beats {} out of range (0..8)",
494                self.seam_lead_in_beats
495            ));
496        }
497        if !(0.0..=1.0).contains(&self.arm_above) || self.arm_above <= self.full_fail.enter_below {
498            return err(format!(
499                "arm_above {} must be in 0..1 and above full_fail.enter_below {}",
500                self.arm_above, self.full_fail.enter_below
501            ));
502        }
503        if !(10.0..=4_000.0).contains(&self.alternate_xfade_ms) {
504            return err(format!(
505                "alternates.xfade_ms {} out of range (10..4000)",
506                self.alternate_xfade_ms
507            ));
508        }
509        Ok(())
510    }

512    /// Snapshot for log headers — session reproducibility depends on knowing
513    /// exactly which ladder was live.
514    pub fn to_json(&self) -> serde_json::Value {
515        let mut v = serde_json::json!({
516            "full_fail": {
517                "enter_below": self.full_fail.enter_below,
518                "exit_above": self.full_fail.exit_above,
519                "hold_ms": self.full_fail.hold_ms,
520            },
521            "seam": {
522                "fade_ms": self.seam_fade_ms,
523                "rewind_beats": self.seam_rewind_beats,
524                "rewind_drop_semitones": self.seam_rewind_drop_st,
525                "pickup_beats": self.seam_pickup_beats,
526                "lead_in_beats": self.seam_lead_in_beats,
527            },
528            "arm_above": self.arm_above,
529            "rungs": self.rungs.iter().map(|r| {
530                let mut rung = serde_json::json!({
531                    "name": r.name,
532                    "enter_below": r.enter_below,
533                    "exit_above": r.exit_above,
534                    "ramp_ms": r.ramp_ms,
535                    "audio": {
536                        "lpf_hz": r.audio.lpf_hz,
537                        "thin_db": r.audio.thin_db,
538                        "drop": r.audio.drop,
539                        "warble_depth_semitones": r.audio.warble_depth_semitones,
540                        "warble_rate_hz": r.audio.warble_rate_hz,
541                    },
542                    "visual": {
543                        "desaturate": r.visual.desaturate,
544                        "blur": r.visual.blur,
545                        "chromatic": r.visual.chromatic,
546                    },
547                });
548                // Emitted only when authored, so alternate-free configs keep
549                // byte-identical headers (the W2 additive-serialization rule).
550                if !r.audio.alternates.is_empty() {
551                    rung["audio"]["alternates"] = serde_json::json!(r.audio.alternates);
552                }
553                rung
554            }).collect::<Vec<_>>(),
555        });
556        if self.alternate_xfade_ms != Self::default().alternate_xfade_ms
557            || self.rungs.iter().any(|r| !r.audio.alternates.is_empty())
558        {
559            v["alternates"] = serde_json::json!({ "xfade_ms": self.alternate_xfade_ms });
560        }
561        v
562    }

565/// The resolved parameter set for the current ladder level — the one struct
566/// both the audio driver and the visual frame read.
567#[derive(Debug, Clone, PartialEq)]
568pub struct LadderParams {
569    /// 0 = clean; 1..=rungs.len() = that rung (1-based).
570    pub level: usize,
571    pub lpf_hz: f64,
572    /// Per-bus gain trim in dB; dropped buses appear as [`DROP_DB`].
573    pub gain_trim_db: BTreeMap<String, f32>,
574    pub warble_depth_semitones: f64,
575    pub warble_rate_hz: f64,
576    pub ramp_ms: f64,
577    /// Per-bus alternate crossfade weight (0 = bus, 1 = alternate; ADR 0032).
578    /// Absent bus = 0. Applied only while `SeqPhase::Playing` — per-sound
579    /// volume is the reintegration entry-envelope channel during reassembly.
580    pub alternate_mix: BTreeMap<String, f64>,
581    pub visual: RungVisual,
582}
583
584impl LadderParams {
585    pub fn clean() -> Self {
586        Self {
587            level: 0,
588            lpf_hz: LPF_OPEN_HZ,
589            gain_trim_db: BTreeMap::new(),
590            warble_depth_semitones: 0.0,
591            warble_rate_hz: 0.0,
592            ramp_ms: 350.0,
593            alternate_mix: BTreeMap::new(),
594            visual: RungVisual::default(),
595        }
596    }
597
598    fn from_rung(level: usize, r: &Rung) -> Self {
599        let mut gain_trim_db = r.audio.thin_db.clone();
600        for bus in &r.audio.drop {
601            gain_trim_db.insert(bus.clone(), DROP_DB);
602        }
603        let alternate_mix = r
604            .audio
605            .alternates
606            .iter()
607            .map(|b| (b.clone(), 1.0))
608            .collect();
609        Self {
610            level,
611            lpf_hz: r.audio.lpf_hz,
612            gain_trim_db,
613            warble_depth_semitones: r.audio.warble_depth_semitones,
614            warble_rate_hz: r.audio.warble_rate_hz,
615            ramp_ms: r.ramp_ms,
616            alternate_mix,
617            visual: r.visual,
618        }
619    }
620
621    /// Interpolate between two param sets (reassembly runs the ladder in
622    /// reverse by lerping from the full-fail state back to the target).
623    /// `t` = 0 → `self`, 1 → `other`. LPF interpolates in log-Hz (perceptual);
624    /// gains and visuals linearly; warble rate holds while depth fades.
625    pub fn lerp(&self, other: &Self, t: f64) -> Self {
626        let t = t.clamp(0.0, 1.0);
627        let tf = t as f32;
628        let lerp_f = |a: f64, b: f64| a + (b - a) * t;
629        let mut gain_trim_db = BTreeMap::new();
630        for bus in self.gain_trim_db.keys().chain(other.gain_trim_db.keys()) {
631            let a = *self.gain_trim_db.get(bus).unwrap_or(&0.0);
632            let b = *other.gain_trim_db.get(bus).unwrap_or(&0.0);
633            gain_trim_db.insert(bus.clone(), a + (b - a) * tf);
634        }
635        let mut alternate_mix = BTreeMap::new();
636        for bus in self.alternate_mix.keys().chain(other.alternate_mix.keys()) {
637            let a = *self.alternate_mix.get(bus).unwrap_or(&0.0);
638            let b = *other.alternate_mix.get(bus).unwrap_or(&0.0);
639            alternate_mix.insert(bus.clone(), a + (b - a) * t);
640        }
641        Self {
642            level: if t < 1.0 { self.level } else { other.level },
643            lpf_hz: (lerp_f(self.lpf_hz.ln(), other.lpf_hz.ln())).exp(),
644            gain_trim_db,
645            warble_depth_semitones: lerp_f(
646                self.warble_depth_semitones,
647                other.warble_depth_semitones,
648            ),
649            warble_rate_hz: if t < 0.5 {
650                self.warble_rate_hz
651            } else {
652                other.warble_rate_hz
653            },
654            ramp_ms: lerp_f(self.ramp_ms, other.ramp_ms),
655            alternate_mix,
656            visual: RungVisual {
657                desaturate: self.visual.desaturate
658                    + (other.visual.desaturate - self.visual.desaturate) * tf,
659                blur: self.visual.blur + (other.visual.blur - self.visual.blur) * tf,
660                chromatic: self.visual.chromatic
661                    + (other.visual.chromatic - self.visual.chromatic) * tf,
662            },
663        }
664    }
665}

667/// The rung state machine: feed it coherence every tick, read the resolved
668/// params. Hysteresis means the level only moves when coherence crosses
669/// `enter_below` going down or `exit_above` going up.
670pub struct Ladder {
671    cfg: LadderConfig,
672    level: usize,
673    armed: bool,
674}
675
676impl Ladder {
677    pub fn new(cfg: LadderConfig) -> Self {
678        Self {
679            cfg,
680            level: 0,
681            armed: false,
682        }
683    }
684
685    pub fn config(&self) -> &LadderConfig {
686        &self.cfg
687    }
688
689    /// Swap the config mid-session; the current level carries over (clamped),
690    /// so tuning never resets the world's state.
691    pub fn reconfigure(&mut self, cfg: LadderConfig) {
692        self.level = self.level.min(cfg.rungs.len());
693        self.cfg = cfg;
694    }
695
696    /// 0 = clean, 1..=n = rung index (1-based).
697    pub fn level(&self) -> usize {
698        self.level
699    }
700
701    pub fn rung(&self) -> Option<&Rung> {
702        (self.level > 0).then(|| &self.cfg.rungs[self.level - 1])
703    }
704
705    /// Whether the ladder has armed: sessions open at a neutral coherence
706    /// (0.5 by default, inside rung territory), so the world would boot
707    /// mid-disintegration without this. The ladder stays clean until
708    /// coherence first clears the top rung's `exit_above` — the world can
709    /// only come apart after it has first cohered.
710    pub fn armed(&self) -> bool {
711        self.armed
712    }
713
714    /// Update the level from the current coherence value. Returns the level.
715    pub fn observe(&mut self, coherence: f64) -> usize {
716        if !self.armed {
717            if coherence >= self.cfg.arm_above {
718                self.armed = true;
719            } else {
720                return 0;
721            }
722        }
723        while self.level < self.cfg.rungs.len()
724            && coherence < self.cfg.rungs[self.level].enter_below
725        {
726            self.level += 1;
727        }
728        while self.level > 0 && coherence > self.cfg.rungs[self.level - 1].exit_above {
729            self.level -= 1;
730        }
731        self.level
732    }
733
734    /// Resolved params for the current level.
735    pub fn params(&self) -> LadderParams {
736        match self.rung() {
737            None => LadderParams::clean(),
738            Some(r) => LadderParams::from_rung(self.level, r),
739        }
740    }
741
742    /// Params at the deepest rung — the state reassembly lerps back from.
743    pub fn full_fail_params(&self) -> LadderParams {
744        match self.cfg.rungs.last() {
745            None => LadderParams::clean(),
746            Some(r) => LadderParams::from_rung(self.cfg.rungs.len(), r),
747        }
748    }
749}

751/// Applies [`LadderParams`] to the mixer as tweens, idempotently: a command
752/// is issued only when a bus's target actually changes, so calling `apply`
753/// every tick is free while the world holds still. The warble is a slow
754/// symmetric LFO evaluated from suite time (deterministic — same time in,
755/// same detune out) and smoothed by short tweens.
756///
757/// The driver owns *track* gain (ladder trims). Reintegration entry
758/// envelopes live on *per-sound* volume (`StemBus::set_sound_volume`), so
759/// the two never fight.
760pub struct LadderDriver {
761    last_lpf: BTreeMap<String, f64>,
762    last_gain: BTreeMap<String, f32>,
763    last_detune: BTreeMap<String, f64>,
764    /// Last commanded alternate crossfade weight per bus, including the span
765    /// gate (so crossing a span edge re-issues even at a held rung).
766    last_alt_mix: BTreeMap<String, f64>,
767}
768
769/// Smoothing tween for per-tick warble updates (the LFO is the shape; this
770/// just bridges between tick evaluations without zipper noise).
771const WARBLE_SMOOTH_MS: f64 = 50.0;
772/// Gain/detune changes smaller than this don't re-issue commands.
773const GAIN_EPS: f32 = 0.01;
774const DETUNE_EPS: f64 = 0.001;
775const ALT_MIX_EPS: f64 = 0.005;
776
777impl Default for LadderDriver {
778    fn default() -> Self {
779        Self::new()
780    }
781}
782
783impl LadderDriver {
784    pub fn new() -> Self {
785        Self {
786            last_lpf: BTreeMap::new(),
787            last_gain: BTreeMap::new(),
788            last_detune: BTreeMap::new(),
789            last_alt_mix: BTreeMap::new(),
790        }
791    }
792
793    /// Forget everything previously issued, so the next `apply` restates the
794    /// full target state. Call after a seam: re-played stems reset detune,
795    /// and entry gains change out from under the shadow bookkeeping.
796    pub fn reset(&mut self) {
797        self.last_lpf.clear();
798        self.last_gain.clear();
799        self.last_detune.clear();
800        self.last_alt_mix.clear();
801    }
802
803    /// Drive the mixer toward `params` plus the gradient's additive
804    /// `offsets` (ADR 0024 — summed into the targets before the epsilon
805    /// diff, so this stays the mixer's one writer and a recovered gradient
806    /// re-asserts clean values instead of fighting shadow state; pass
807    /// `GradientOffsets::default()` when no gradient is in play).
808    /// `now_seconds` is suite time (for the warble LFO phase); `ramp_ms` is
809    /// the tween length for gain/LPF moves — normally `params.ramp_ms`, but
810    /// reassembly passes tick-scale ramps while it lerps params itself.
811    /// `alternates`: `Some((now_sample, xfade_ms))` applies the rung's
812    /// bus↔alternate crossfades (span-gated at that sample); `None` skips
813    /// them entirely — reassembly must pass `None`, because per-sound volume
814    /// is the reintegration entry-envelope channel there (ADR 0032).
815    pub fn apply(
816        &mut self,
817        params: &LadderParams,
818        offsets: &GradientOffsets,
819        now_seconds: f64,
820        ramp_ms: f64,
821        alternates: Option<(i64, f64)>,
822        mixer: &mut BusMixer,
823    ) {
824        let tween = |ms: f64| Tween {
825            duration: Duration::from_secs_f64(ms.max(0.0) / 1000.0),
826            ..Default::default()
827        };
828        let warble = if params.warble_depth_semitones > 0.0 && params.warble_rate_hz > 0.0 {
829            params.warble_depth_semitones
830                * (std::f64::consts::TAU * params.warble_rate_hz * now_seconds).sin()
831        } else {
832            0.0
833        };
834        for name in crate::BUSES {
835            let is_motif = MOTIF_BUSES.contains(&name);
836            let is_degradable = DEGRADABLE_BUSES.contains(&name);
837            let Some(bus) = mixer.bus_mut(name) else {
838                continue;
839            };
840
841            // LPF: every non-motif bus follows the rung cutoff.
842            if !is_motif {
843                let target = params.lpf_hz;
844                if self.last_lpf.get(name) != Some(&target) {
845                    bus.set_lpf(target, tween(ramp_ms));
846                    self.last_lpf.insert(name.into(), target);
847                }
848            }
849
850            // Track gain = ladder trim + gradient offset (0 on protected
851            // buses by construction — both validators enforce it).
852            let target = params.gain_trim_db.get(name).copied().unwrap_or(0.0)
853                + offsets.gain_db.get(name).copied().unwrap_or(0.0);
854            if (self.last_gain.get(name).copied().unwrap_or(0.0) - target).abs() > GAIN_EPS
855                || !self.last_gain.contains_key(name)
856            {
857                bus.set_gain(target, tween(ramp_ms));
858                self.last_gain.insert(name.into(), target);
859            }
860
861            // Warble: degradable buses only. Detuning the foundation would
862            // un-anchor the pulse, and motif lines stay in tune — the world
863            // warbles around them. (Nonzero detune breaks sample-lock while
864            // active — mixer docs; the seam re-play re-locks.)
865            if is_degradable && bus.state.playing {
866                let target = warble + offsets.detune.get(name).copied().unwrap_or(0.0);
867                let last = self.last_detune.get(name).copied().unwrap_or(0.0);
868                if (last - target).abs() > DETUNE_EPS {
869                    bus.set_detune(target, tween(WARBLE_SMOOTH_MS.min(ramp_ms.max(1.0))));
870                    self.last_detune.insert(name.into(), target);
871                }
872            }
873
874            // Composed alternates (ADR 0032): crossfade toward the rung's
875            // authored replacement, span-gated — the *effective* mix is
876            // shadowed so crossing a span edge at a held rung re-issues
877            // (otherwise the bus would stay ducked against authored silence).
878            if let Some((now_sample, xfade_ms)) = alternates {
879                if let Some((from, to)) = bus.alternate_span() {
880                    let requested = params.alternate_mix.get(name).copied().unwrap_or(0.0);
881                    let effective = if (from..to).contains(&now_sample) {
882                        requested
883                    } else {
884                        0.0
885                    };
886                    let last = self.last_alt_mix.get(name).copied().unwrap_or(0.0);
887                    if (last - effective).abs() > ALT_MIX_EPS
888                        || !self.last_alt_mix.contains_key(name)
889                    {
890                        bus.set_alternate_mix(effective, now_sample, tween(xfade_ms));
891                        self.last_alt_mix.insert(name.into(), effective);
892                    }
893                }
894            }
895        }
896    }
897}
