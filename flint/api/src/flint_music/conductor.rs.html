1//! The Conductor: the single query API for musical time.
2//!
3//! Everything that needs to know "where are we in the music" asks the
4//! Conductor and nothing else. It owns the tempo map and section list from the
5//! manifest and converts between sample time, seconds, and bar/beat/tick,
6//! including across tempo and meter changes. It also applies latency
7//! compensation: `now` answers for what the ear hears, not what the DAC was
8//! just handed.
9//!
10//! The Conductor holds no audio handles and never mutates — the playback
11//! clock counts samples (see `session`), and all musical interpretation is
12//! pure arithmetic here. One source of truth, no drift.
13
14use crate::manifest::{Section, SuiteManifest};
15use crate::tempo::ms_to_samples;
16use crate::TempoMap;
17
18/// Pulses per quarter note for the `tick` readout field.
19pub const TICKS_PER_BEAT: i64 = 960;
20
21/// A grid to schedule or align against.
22#[derive(Debug, Clone, Copy, PartialEq)]
23pub enum Grid {
24    Beat,
25    Bar,
26    /// Every `n` beats since the governing anchor (e.g. 0.5 for eighths).
27    Beats(f64),
28}
29
30/// A fully resolved moment in musical time.
31#[derive(Debug, Clone, Copy, PartialEq)]
32pub struct MusicalPosition {
33    /// Suite sample position (negative during pre-roll).
34    pub sample: i64,
35    /// Suite seconds (sample / sample_rate).
36    pub seconds: f64,
37    /// Beats from suite sample 0, accumulated across tempo changes.
38    pub beat: f64,
39    /// Bar index from 0; anchors always start a new bar.
40    pub bar: i64,
41    /// Beat within the current bar, from 0.
42    pub beat_in_bar: f64,
43    /// Ticks (at [`TICKS_PER_BEAT`]) within the current beat.
44    pub tick: i64,
45}
46
47pub struct Conductor {
48    tempo: TempoMap,
49    sections: Vec<Section>,
50    latency_samples: i64,
51}
52
53impl Conductor {
54    /// Build from a validated manifest. `latency_ms` is the measured output
55    /// latency (from `logs/latency/`); `None` means zero compensation, which
56    /// is exact for offline rendering.
57    pub fn new(manifest: &SuiteManifest, latency_ms: Option<f64>) -> Self {
58        let latency_samples = latency_ms
59            .map(|ms| ms_to_samples(ms, manifest.sample_rate as f64))
60            .unwrap_or(0);
61        Self {
62            tempo: TempoMap::new(manifest.tempo.clone(), manifest.sample_rate),
63            sections: manifest.sections.clone(),
64            latency_samples,
65        }
66    }
67
68    pub fn tempo(&self) -> &TempoMap {
69        &self.tempo
70    }
71
72    pub fn latency_samples(&self) -> i64 {
73        self.latency_samples
74    }
75
76    /// The musical position at an absolute suite sample.
77    pub fn position_at_sample(&self, sample: i64) -> MusicalPosition {
78        let beat = self.tempo.beats_at_sample(sample).unwrap_or(0.0);
79        let (bar, beat_in_bar) = self.tempo.bar_beat_at_sample(sample).unwrap_or((0, 0.0));
80        let tick =
81            (beat_in_bar.rem_euclid(1.0) * TICKS_PER_BEAT as f64).round() as i64 % TICKS_PER_BEAT;
82        MusicalPosition {
83            sample,
84            seconds: sample as f64 / self.tempo.sample_rate() as f64,
85            beat,
86            bar,
87            beat_in_bar,
88            tick,
89        }
90    }
91
92    /// What the ear hears now, given the playback clock's sample count:
93    /// the clock position pulled back by the measured output latency.
94    pub fn now(&self, clock_sample: i64) -> MusicalPosition {
95        self.position_at_sample(clock_sample - self.latency_samples)
96    }
97
98    /// Nearest whole sample of a beats-from-start position.
99    pub fn sample_at_beat(&self, beat: f64) -> i64 {
100        self.tempo.sample_at_beat(beat).unwrap_or(0.0).round() as i64
101    }
102
103    /// Nearest whole sample of `beat_in_bar` beats into bar `bar`.
104    pub fn sample_at_bar(&self, bar: i64, beat_in_bar: f64) -> i64 {
105        self.tempo
106            .sample_at_bar(bar, beat_in_bar)
107            .unwrap_or(0.0)
108            .round() as i64
109    }
110
111    /// The first grid sample strictly after `from`. Anchors count as grid
112    /// points for every grid (they start a new bar).
113    ///
114    /// Strictly-after is load-bearing (callers step the grid by feeding each
115    /// result back in — a non-advancing return would loop them forever), and
116    /// fractional grids can violate it naively: at e.g. 84 BPM / 48 kHz a
117    /// beat is 34285.71 samples, and when the ideal next-grid point rounds
118    /// *down* onto `from`, `bar_beat_at_sample(from)` reads fractionally
119    /// below the grid index (a half-sample beats `next_in_bar`'s 1e-9 beat
120    /// epsilon) and re-derives the same point. One re-probe from `from + 1`
121    /// lands cleanly past it. Degenerate maps (no tempo data) still return
122    /// `from` — callers advancing a cursor must treat `<= from` as "grid
123    /// exhausted", never as a point.
124    pub fn next_grid_sample(&self, from: i64, grid: Grid) -> i64 {
125        let mut probe = from;
126        for _ in 0..2 {
127            let (bar, beat_in_bar) = self.tempo.bar_beat_at_sample(probe).unwrap_or((0, 0.0));
128            let next = match grid {
129                Grid::Bar => self.tempo.sample_at_bar(bar + 1, 0.0),
130                Grid::Beat => self.next_in_bar(bar, beat_in_bar, 1.0),
131                Grid::Beats(n) if n > 0.0 => self.next_in_bar(bar, beat_in_bar, n),
132                Grid::Beats(_) => None,
133            };
134            match next.map(|s| s.round() as i64) {
135                Some(s) if s > from => return s,
136                Some(_) => probe = from + 1,
137                None => return from,
138            }
139        }
140        from
141    }
142
143    /// Next multiple of `step` beats within the bar, or the next bar line if
144    /// the bar runs out first (which also covers anchors cutting bars short).
145    fn next_in_bar(&self, bar: i64, beat_in_bar: f64, step: f64) -> Option<f64> {
146        let next_beat = ((beat_in_bar / step + 1e-9).floor() + 1.0) * step;
147        let candidate = self.tempo.sample_at_bar(bar, next_beat)?;
148        let bar_end = self.tempo.sample_at_bar(bar + 1, 0.0)?;
149        Some(candidate.min(bar_end))
150    }
151
152    /// The section containing `sample` (the last one starting at or before it).
153    pub fn section_at_sample(&self, sample: i64) -> Option<&Section> {
154        self.sections
155            .iter()
156            .take_while(|s| s.start_sample <= sample)
157            .last()
158    }
159
160    pub fn section_by_name(&self, name: &str) -> Option<&Section> {
161        self.sections.iter().find(|s| s.name == name)
162    }
163
164    /// The reintegration checkpoint for a full-fail at `sample`: the latest
165    /// manifest re-entry section starting at or before it, falling back to
166    /// the earliest re-entry section when the fail lands before any (the
167    /// validator guarantees `re_entry_sections` is non-empty and resolvable).
168    pub fn previous_re_entry(
169        &self,
170        sample: i64,
171        reintegration: &crate::manifest::Reintegration,
172    ) -> Option<&Section> {
173        let mut entries: Vec<&Section> = reintegration
174            .re_entry_sections
175            .iter()
176            .filter_map(|name| self.section_by_name(name))
177            .collect();
178        entries.sort_by_key(|s| s.start_sample);
179        entries
180            .iter()
181            .rev()
182            .find(|s| s.start_sample <= sample)
183            .or(entries.first())
184            .copied()
185    }
186}

237    #[test]
238    fn position_readout_across_tempo_change() {
239        let c = Conductor::new(&manifest(), None);
240        let p = c.position_at_sample(960_000 + 32_000);
241        assert_eq!(p.beat, 41.0);
242        assert_eq!(p.bar, 10);
243        assert!((p.beat_in_bar - 1.0).abs() < 1e-9);
244        assert_eq!(p.tick, 0);
245        assert!((p.seconds - (960_000.0 + 32_000.0) / 48_000.0).abs() < 1e-12);
246    }

248    #[test]
249    fn latency_compensation_shifts_now() {
250        let c = Conductor::new(&manifest(), Some(500.0)); // 24_000 samples
251        assert_eq!(c.latency_samples(), 24_000);
252        let p = c.now(24_000);
253        assert_eq!(p.sample, 0);
254        // Early in pre-roll the ear-position is negative.
255        assert!(c.now(0).sample < 0 && c.now(0).beat < 0.0);
256    }

258    #[test]
259    fn next_grid_monotone_and_lands_on_grid() {
260        let c = Conductor::new(&manifest(), None);
261        let mut s = 1; // just after the downbeat
262        for _ in 0..50 {
263            let n = c.next_grid_sample(s, Grid::Beat);
264            assert!(n > s, "grid must advance: {s} -> {n}");
265            let bib = c.position_at_sample(n).beat_in_bar;
266            assert!((bib - bib.round()).abs() < 1e-6, "off-beat landing: {bib}");
267            s = n;
268        }
269        // Beat grid crosses the tempo change cleanly.
270        assert!(s > 960_000);
271        // Bar grid: from mid-bar 9 the next bar is the anchor itself.
272        let from = c.sample_at_bar(9, 2.0);
273        assert_eq!(c.next_grid_sample(from, Grid::Bar), 960_000);
274    }

276    #[test]
277    fn next_grid_advances_on_fractional_beat_lengths() {
278        // 84 BPM at 48 kHz: 34285.714… samples per beat — the prototype
279        // track's real tempo. Rounding can land the ideal grid point back on
280        // `from` (a half-sample beats next_in_bar's 1e-9 beat epsilon),
281        // which must never stall the strictly-after contract: haptics (ADR
282        // 0026) steps this grid by feeding results back in, and a stall
283        // there was an unbounded allocation before the re-probe fix.
284        let mut m = manifest();
285        m.tempo = vec![TempoAnchor {
286            sample: 0,
287            bpm: 84.0,
288            beats_per_bar: 4,
289            beat_unit: 4,
290        }];
291        let c = Conductor::new(&m, None);
292        let mut s = 0;
293        for _ in 0..2000 {
294            let n = c.next_grid_sample(s, Grid::Beat);
295            assert!(n > s, "fractional grid must still advance: {s} -> {n}");
296            assert!(
297                n - s <= 34_287,
298                "advance must stay one beat, not skip: {s} -> {n}"
299            );
300            s = n;
301        }
302    }
