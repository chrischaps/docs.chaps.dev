1//! Suite manifest parsing.
2//!
3//! Parsing is shape-tolerant on purpose: it accepts any bus names, any section
4//! ordering, any version number, and reports *semantic* problems as coded
5//! [`Issue`](crate::validate::Issue)s from the validator rather than parse
6//! errors. Only structurally unreadable input (missing tables, wrong types)
7//! fails the parse.
8
9use crate::config_toml::{self, VersionPolicy};
10use flint_core::{FlintError, Result};
11use std::collections::BTreeMap;
12use std::path::Path;

18    pub beats_per_bar: i64,

50#[derive(Debug, Clone)]
51pub struct SuiteManifest {
52    pub schema_version: i64,
53    pub id: String,
54    pub title: String,
55    pub sample_rate: u32,
56    pub tempo: Vec<TempoAnchor>,
57    pub sections: Vec<Section>,
58    pub reintegration: Reintegration,
59    pub buses: BTreeMap<String, BusDecl>,
60    pub degraded_alternates: Vec<DegradedAlternate>,
61}
62
63impl SuiteManifest {
64    pub fn load(path: &Path) -> Result<Self> {
65        let text = std::fs::read_to_string(path)?;
66        Self::parse(&text)
67    }
68
69    pub fn parse(text: &str) -> Result<Self> {
70        let (root, schema_version) =
71            config_toml::parse_versioned(NAME, text, VersionPolicy::Required)?;
72
73        let suite = table(&root, "suite")?;
74        let id = string(suite, "id")?;
75        let title = string(suite, "title")?;
76
77        let audio = table(&root, "audio")?;
78        // Checked, not `as`: a negative or >u32::MAX value would silently
79        // wrap into a plausible-looking rate.
80        let sample_rate = u32::try_from(int_in(audio, "sample_rate")?)
81            .map_err(|_| bad("audio.sample_rate (out of range for u32)"))?;
82
83        let mut tempo = Vec::new();
84        for (i, v) in array(&root, "tempo")?.iter().enumerate() {
85            let t = v.as_table().ok_or_else(|| bad(&format!("tempo[{i}]")))?;
86            let sig = t
87                .get("time_signature")
88                .and_then(|v| v.as_array())
89                .ok_or_else(|| bad(&format!("tempo[{i}].time_signature")))?;
90            if sig.len() != 2 {
91                return Err(bad(&format!(
92                    "tempo[{i}].time_signature (need [beats, unit])"
93                )));
94            }
95            tempo.push(TempoAnchor {
96                sample: int_in(t, "sample")?,
97                bpm: float_in(t, "bpm")?,
98                beats_per_bar: sig[0]
99                    .as_integer()
100                    .ok_or_else(|| bad(&format!("tempo[{i}].time_signature[0]")))?,
101                beat_unit: sig[1]
102                    .as_integer()
103                    .ok_or_else(|| bad(&format!("tempo[{i}].time_signature[1]")))?,
104            });
105        }
106
107        let mut sections = Vec::new();
108        for (i, v) in array(&root, "sections")?.iter().enumerate() {
109            let t = v.as_table().ok_or_else(|| bad(&format!("sections[{i}]")))?;
110            sections.push(Section {
111                name: string(t, "name")?,
112                start_sample: int_in(t, "start_sample")?,
113                pulse_window_ms: float_in(t, "pulse_window_ms")?,
114            });
115        }
116
117        let reint = table(&root, "reintegration")?;
118        let re_entry_sections = reint
119            .get("re_entry_sections")
120            .and_then(|v| v.as_array())
121            .ok_or_else(|| bad("reintegration.re_entry_sections"))?
122            .iter()
123            .map(|v| {
124                v.as_str()
125                    .map(String::from)
126                    .ok_or_else(|| bad("reintegration.re_entry_sections entry"))
127            })
128            .collect::<Result<Vec<_>>>()?;
129        let reintegration = Reintegration {
130            re_entry_sections,
131            lead_bus: string(reint, "lead_bus")?,
132            reassembly_bars: int_in(reint, "reassembly_bars")?,
133        };
134
135        let mut buses = BTreeMap::new();
136        for (name, v) in table(&root, "buses")?.iter() {
137            let t = v.as_table().ok_or_else(|| bad(&format!("buses.{name}")))?;
138            buses.insert(
139                name.clone(),
140                BusDecl {
141                    file: t.get("file").and_then(|v| v.as_str()).map(String::from),
142                    silent: t.get("silent").and_then(|v| v.as_bool()).unwrap_or(false),
143                },
144            );
145        }
146
147        let mut degraded_alternates = Vec::new();
148        if let Some(arr) = root.get("degraded_alternates").and_then(|v| v.as_array()) {
149            for (i, v) in arr.iter().enumerate() {
150                let t = v
151                    .as_table()
152                    .ok_or_else(|| bad(&format!("degraded_alternates[{i}]")))?;
153                degraded_alternates.push(DegradedAlternate {
154                    bus: string(t, "bus")?,
155                    file: string(t, "file")?,
156                    from_sample: int_in(t, "from_sample")?,
157                    to_sample: int_in(t, "to_sample")?,
158                });
159            }
160        }
161
162        Ok(SuiteManifest {
163            schema_version,
164            id,
165            title,
166            sample_rate,
167            tempo,
168            sections,
169            reintegration,
170            buses,
171            degraded_alternates,
172        })
173    }
174}

176// -- small extraction helpers (FlintError::ParseError on shape problems) ------
177
178const NAME: &str = "suite manifest";
179
180fn bad(what: &str) -> FlintError {
181    config_toml::bad(NAME, what)
182}
183
184fn table<'a>(v: &'a toml::Value, key: &str) -> Result<&'a toml::value::Table> {
185    config_toml::table(NAME, v, key)
186}
187
188fn array<'a>(v: &'a toml::Value, key: &str) -> Result<&'a Vec<toml::Value>> {
189    config_toml::array(NAME, v, key)
190}
191
192fn int_in(t: &toml::value::Table, key: &str) -> Result<i64> {
193    config_toml::int_in(NAME, t, key)
194}
195
196fn float_in(t: &toml::value::Table, key: &str) -> Result<f64> {
197    config_toml::float_in(NAME, t, key, key)
198}
199
200fn string(t: &toml::value::Table, key: &str) -> Result<String> {
201    config_toml::string_in(NAME, t, key, key)
