1//! The coherence scalar: the single 0..1 value downstream systems see.
2//!
3//! Model (ADR 0010): a leaky integrator with asymmetric, bar-denominated
4//! time constants. The continuous tracking signal sets a per-step target
5//! ("fit"); the value eases toward it with `alpha = 1 - exp(-Δbars/τ)`,
6//! rising with `rise_bars` and falling with `fall_bars`. Discrete pulse
7//! outcomes enter as bounded impulses: a judged hit nudges up by how clean
8//! it was, a miss nudges down by `miss_penalty`, a spurious pulse by
9//! `spurious_penalty` (0 by default — flow, not evaluation). Everything is
10//! plain f64 arithmetic in a fixed order: same records in, same value out.
11//!
12//! Every knob lives in `CoherenceConfig`, loaded from TOML and reloadable
13//! mid-session (the value survives a reload). Tuning these is a primary
14//! goal of the feel prototype; the raw judgment logs exist precisely so
15//! alternatives can be evaluated offline.
16

22#[derive(Debug, Clone, Copy, PartialEq)]
23pub struct CoherenceConfig {
24    /// Blend weight of the continuous tracking signal.
25    pub w_lean: f64,
26    /// Blend weight of discrete pulse outcomes.
27    pub w_pulse: f64,
28    /// Blend weights of the W2 continuous channels. **Default 0.0 = inert**:
29    /// a zero-weight non-lean Track record is skipped entirely (no fit, no
30    /// time advance), so an unmodified config produces bit-identical values
31    /// on any chart.
32    pub w_sway: f64,
33    pub w_pressure_l: f64,
34    pub w_pressure_r: f64,
35    /// Press depth error treated as fully wrong (quality factor on the hit
36    /// impulse; only press records carry one).
37    pub press_depth_err_full: f64,
38    /// Flick direction error (degrees) treated as fully wrong.
39    pub flick_dir_err_full_deg: f64,
40    /// Tracking error treated as fully wrong (Euclidean, lean plane).
41    pub track_err_full: f64,
42    /// Exponent shaping the tracking penalty curve.
43    pub track_curve: f64,
44    /// |timing error| treated as fully wrong for a judged pulse.
45    pub pulse_err_full_ms: f64,
46    /// Impulse magnitude of a miss (in penalty units).
47    pub miss_penalty: f64,
48    /// Impulse magnitude of a spurious pulse (default 0: free).
49    pub spurious_penalty: f64,
50    /// Scale of all pulse impulses on the 0..1 value.
51    pub impulse_gain: f64,
52    /// Starting value of a session.
53    pub initial: f64,
54    /// Time constant (bars) when the value is rising.
55    pub rise_bars: f64,
56    /// Time constant (bars) when the value is falling.
57    pub fall_bars: f64,
58}

83impl CoherenceConfig {
84    pub fn load(path: &Path) -> Result<Self> {
85        let text = std::fs::read_to_string(path)?;
86        Self::parse(&text)
87    }
88
89    pub fn parse(text: &str) -> Result<Self> {
90        let (root, _) = config_toml::parse_versioned(
91            "coherence config",
92            text,
93            VersionPolicy::DefaultZeroStrict,
94        )?;
95        let f = |table: &str, key: &str, default: f64| -> f64 {
96            config_toml::section_f64(&root, table, key, default)
97        };
98        let d = Self::default();
99        let cfg = Self {
100            w_lean: f("weights", "lean", d.w_lean),
101            w_pulse: f("weights", "pulse", d.w_pulse),
102            w_sway: f("weights", "sway", d.w_sway),
103            w_pressure_l: f("weights", "pressure_l", d.w_pressure_l),
104            w_pressure_r: f("weights", "pressure_r", d.w_pressure_r),
105            press_depth_err_full: f("press", "depth_err_full", d.press_depth_err_full),
106            flick_dir_err_full_deg: f("flick", "dir_err_full_deg", d.flick_dir_err_full_deg),
107            track_err_full: f("tracking", "err_full", d.track_err_full),
108            track_curve: f("tracking", "curve", d.track_curve),
109            pulse_err_full_ms: f("pulse", "err_full_ms", d.pulse_err_full_ms),
110            miss_penalty: f("pulse", "miss_penalty", d.miss_penalty),
111            spurious_penalty: f("pulse", "spurious_penalty", d.spurious_penalty),
112            impulse_gain: f("pulse", "impulse_gain", d.impulse_gain),
113            initial: f("smoothing", "initial", d.initial),
114            rise_bars: f("smoothing", "rise_bars", d.rise_bars),
115            fall_bars: f("smoothing", "fall_bars", d.fall_bars),
116        };
117        for (name, v, lo) in [
118            ("weights.lean", cfg.w_lean, 0.0),
119            ("weights.pulse", cfg.w_pulse, 0.0),
120            ("weights.sway", cfg.w_sway, 0.0),
121            ("weights.pressure_l", cfg.w_pressure_l, 0.0),
122            ("weights.pressure_r", cfg.w_pressure_r, 0.0),
123            (
124                "press.depth_err_full",
125                cfg.press_depth_err_full,
126                f64::MIN_POSITIVE,
127            ),
128            (
129                "flick.dir_err_full_deg",
130                cfg.flick_dir_err_full_deg,
131                f64::MIN_POSITIVE,
132            ),
133            ("tracking.err_full", cfg.track_err_full, f64::MIN_POSITIVE),
134            (
135                "pulse.err_full_ms",
136                cfg.pulse_err_full_ms,
137                f64::MIN_POSITIVE,
138            ),
139            ("smoothing.rise_bars", cfg.rise_bars, f64::MIN_POSITIVE),
140            ("smoothing.fall_bars", cfg.fall_bars, f64::MIN_POSITIVE),
141        ] {
142            if !v.is_finite() || v < lo {
143                return Err(FlintError::ValidationError(format!(
144                    "coherence config: `{name}` = {v} out of range"
145                )));
146            }
147        }
148        Ok(cfg)
149    }
150}

152impl CoherenceConfig {
153    /// Rebuild from a header snapshot (inverse of [`Self::to_json`]);
154    /// absent keys keep defaults, so old snapshots stay readable.
155    pub fn from_json(v: &serde_json::Value) -> Self {
156        let d = Self::default();
157        let f = |table: &str, key: &str, default: f64| {
158            v.get(table)
159                .and_then(|t| t.get(key))
160                .and_then(|x| x.as_f64())
161                .unwrap_or(default)
162        };
163        Self {
164            w_lean: f("weights", "lean", d.w_lean),
165            w_pulse: f("weights", "pulse", d.w_pulse),
166            w_sway: f("weights", "sway", d.w_sway),
167            w_pressure_l: f("weights", "pressure_l", d.w_pressure_l),
168            w_pressure_r: f("weights", "pressure_r", d.w_pressure_r),
169            press_depth_err_full: f("press", "depth_err_full", d.press_depth_err_full),
170            flick_dir_err_full_deg: f("flick", "dir_err_full_deg", d.flick_dir_err_full_deg),
171            track_err_full: f("tracking", "err_full", d.track_err_full),
172            track_curve: f("tracking", "curve", d.track_curve),
173            pulse_err_full_ms: f("pulse", "err_full_ms", d.pulse_err_full_ms),
174            miss_penalty: f("pulse", "miss_penalty", d.miss_penalty),
175            spurious_penalty: f("pulse", "spurious_penalty", d.spurious_penalty),
176            impulse_gain: f("pulse", "impulse_gain", d.impulse_gain),
177            initial: f("smoothing", "initial", d.initial),
178            rise_bars: f("smoothing", "rise_bars", d.rise_bars),
179            fall_bars: f("smoothing", "fall_bars", d.fall_bars),
180        }
181    }
182
183    /// Snapshot for log headers and reload records — reproducibility of a
184    /// session depends on knowing exactly which knobs were live. The W2
185    /// keys are emitted only when they differ from defaults, so pre-W2
186    /// sessions and lean-only configs produce byte-identical headers.
187    pub fn to_json(&self) -> serde_json::Value {
188        let d = Self::default();
189        let mut v = serde_json::json!({
190            "weights": { "lean": self.w_lean, "pulse": self.w_pulse },
191            "tracking": { "err_full": self.track_err_full, "curve": self.track_curve },
192            "pulse": {
193                "err_full_ms": self.pulse_err_full_ms,
194                "miss_penalty": self.miss_penalty,
195                "spurious_penalty": self.spurious_penalty,
196                "impulse_gain": self.impulse_gain,
197            },
198            "smoothing": {
199                "initial": self.initial,
200                "rise_bars": self.rise_bars,
201                "fall_bars": self.fall_bars,
202            },
203        });
204        if self.w_sway != d.w_sway {
205            v["weights"]["sway"] = serde_json::json!(self.w_sway);
206        }
207        if self.w_pressure_l != d.w_pressure_l {
208            v["weights"]["pressure_l"] = serde_json::json!(self.w_pressure_l);
209        }
210        if self.w_pressure_r != d.w_pressure_r {
211            v["weights"]["pressure_r"] = serde_json::json!(self.w_pressure_r);
212        }
213        if self.press_depth_err_full != d.press_depth_err_full {
214            v["press"] = serde_json::json!({ "depth_err_full": self.press_depth_err_full });
215        }
216        if self.flick_dir_err_full_deg != d.flick_dir_err_full_deg {
217            v["flick"] = serde_json::json!({ "dir_err_full_deg": self.flick_dir_err_full_deg });
218        }
219        v
220    }
221}

223pub struct Coherence {
224    value: f64,
225    cfg: CoherenceConfig,
226}
227
228impl Coherence {
229    pub fn new(cfg: CoherenceConfig) -> Self {
230        Self {
231            value: cfg.initial.clamp(0.0, 1.0),
232            cfg,
233        }
234    }
235
236    pub fn value(&self) -> f64 {
237        self.value
238    }
239
240    /// Swap the config mid-session; the current value survives, so tuning
241    /// never resets the run.
242    pub fn reconfigure(&mut self, cfg: CoherenceConfig) {
243        self.cfg = cfg;
244    }
245
246    /// Fold one batch of judgment records into the scalar. `beats_per_bar`
247    /// is the current meter (bar-denominated time constants); `grid_beats`
248    /// is the judgment grid step, so each track record advances time by
249    /// `grid_beats / beats_per_bar` bars.
250    pub fn step(&mut self, records: &[JudgmentRecord], grid_beats: f64, beats_per_bar: f64) -> f64 {
251        let c = &self.cfg;
252        for rec in records {
253            match rec {
254                JudgmentRecord::Track { channel, err, .. } => {
255                    // Lean keeps its historical semantics exactly; the W2
256                    // channels are skipped at weight 0 (inert by default —
257                    // fit=1 at w=0 would push the value *up*).
258                    let weight = match channel.as_str() {
259                        "sway" => c.w_sway,
260                        "pressure_l" => c.w_pressure_l,
261                        "pressure_r" => c.w_pressure_r,
262                        _ => c.w_lean,
263                    };
264                    if channel != "lean" && weight == 0.0 {
265                        continue;
266                    }
267                    let pen = (err / c.track_err_full).clamp(0.0, 1.0).powf(c.track_curve);
268                    let fit = 1.0 - weight * pen;
269                    let tau = if fit >= self.value {
270                        c.rise_bars
271                    } else {
272                        c.fall_bars
273                    };
274                    let step_bars = grid_beats / beats_per_bar.max(f64::MIN_POSITIVE);
275                    let alpha = 1.0 - (-step_bars / tau).exp();
276                    self.value += (fit - self.value) * alpha;
277                }
278                JudgmentRecord::Pulse {
279                    err_ms,
280                    depth_err,
281                    dir_err,
282                    ..
283                } => {
284                    let pen = (err_ms.abs() / c.pulse_err_full_ms).clamp(0.0, 1.0);
285                    let mut quality = 1.0 - pen;
286                    if let Some(d) = depth_err {
287                        quality *= 1.0 - (d / c.press_depth_err_full).clamp(0.0, 1.0);
288                    }
289                    if let Some(d) = dir_err {
290                        quality *= 1.0 - (d / c.flick_dir_err_full_deg).clamp(0.0, 1.0);
291                    }
292                    self.value += c.w_pulse * c.impulse_gain * quality;
293                }
294                JudgmentRecord::Miss { .. } => {
295                    self.value -= c.w_pulse * c.impulse_gain * c.miss_penalty;
296                }
297                JudgmentRecord::Spurious { .. } => {
298                    self.value -= c.w_pulse * c.impulse_gain * c.spurious_penalty;
299                }
300            }
301            self.value = self.value.clamp(0.0, 1.0);
302        }
303        self.value
304    }
305}

332    fn run_bars(c: &mut Coherence, bars: usize, err: f64) {
333        for _ in 0..bars * 32 {
334            c.step(&[track(err)], 0.125, 4.0);

339    fn perfect_play_rises_toward_one() {
340        let mut c = Coherence::new(CoherenceConfig::default());
341        let mut last = c.value();
342        for _ in 0..4 {

350    #[test]
351    fn neglect_decays_with_the_configured_half_life() {
352        let cfg = CoherenceConfig::default();
353        let mut c = Coherence::new(cfg);
354        run_bars(&mut c, 8, 0.0); // settle near 1.0
355        let start = c.value();
356        assert!(start > 0.99);
357
358        // Full tracking error: fit floor = 1 - w_lean = 0.4.
359        let floor = 1.0 - cfg.w_lean;
360        run_bars(&mut c, 2, 10.0); // fall_bars = 2 → one time constant
361        let expected = floor + (start - floor) * (-1.0f64).exp();
362        assert!(
363            (c.value() - expected).abs() < 0.02,
364            "after one τ: {} vs expected {expected}",
365            c.value()
366        );
367        // Smooth: no step exceeds one grid-step alpha bound.
368        let alpha = 1.0 - (-0.125 / 4.0 / cfg.fall_bars).exp();
369        let mut prev = c.value();
370        for _ in 0..32 {
371            c.step(&[track(10.0)], 0.125, 4.0);
372            assert!((prev - c.value()) <= alpha * 1.0 + 1e-12);
373            prev = c.value();
374        }
375    }

377    #[test]
378    fn single_miss_is_nearly_invisible() {
379        let mut c = Coherence::new(CoherenceConfig::default());
380        run_bars(&mut c, 8, 0.0);
381        let before = c.value();
382        c.step(&[miss()], 0.125, 4.0);
383        let dip = before - c.value();
384        // GDD: single mistakes nearly invisible. One miss at defaults is a
385        // 0.06 dip that the next bar of good play erases.
386        assert!(dip > 0.0 && dip < 0.1, "dip {dip}");
387        // rise_bars = 1: one bar recovers ~63% of the dip, two bars ~86%.
388        run_bars(&mut c, 2, 0.0);
389        assert!(c.value() > before - 0.01, "recovered to {}", c.value());
390    }
