1//! Programmatic ears: pure analysis functions over rendered audio buffers.
2//!
3//! These are the assertions the offline render harness makes available —
4//! per-tone envelopes (Goertzel), level-transition location, and click
5//! detection — so seam and scheduling checks are automated tests instead of
6//! listening tasks. All functions take interleaved stereo f32 as produced by
7//! [`crate::offline::render_offline`], where index 0 = suite sample 0.
8
9/// Power of a single frequency over `[from, to)` (suite samples), via the
10/// Goertzel algorithm on the mono sum. Cheap: one pass, no FFT.
11pub fn goertzel_power(
12    interleaved: &[f32],
13    sample_rate: u32,
14    freq: f64,
15    from: usize,
16    to: usize,
17) -> f64 {
18    let frames = interleaved.len() / 2;
19    let (from, to) = (from.min(frames), to.min(frames));
20    if to <= from {
21        return 0.0;
22    }
23    let n = to - from;
24    let w = std::f64::consts::TAU * freq / sample_rate as f64;
25    let coeff = 2.0 * w.cos();
26    let (mut s1, mut s2) = (0.0f64, 0.0);
27    for i in from..to {
28        let x = (interleaved[i * 2] + interleaved[i * 2 + 1]) as f64 * 0.5;
29        let s0 = x + coeff * s1 - s2;
30        s2 = s1;
31        s1 = s0;
32    }
33    let power = s1 * s1 + s2 * s2 - coeff * s1 * s2;
34    power / (n as f64 * n as f64 / 4.0) // normalize ~ amplitude^2
35}

37/// Per-window power of one frequency along the whole buffer: element `i`
38/// covers suite samples `[i*window, (i+1)*window)`.
39pub fn tone_envelope(interleaved: &[f32], sample_rate: u32, freq: f64, window: usize) -> Vec<f64> {
40    let frames = interleaved.len() / 2;
41    (0..frames / window)
42        .map(|i| goertzel_power(interleaved, sample_rate, freq, i * window, (i + 1) * window))
43        .collect()
44}

46/// Locate level transitions of one tone: suite samples where its windowed
47/// power first crosses below (drop) or above (rise) the midpoint between the
48/// local levels on either side. Resolution is one window.
49pub fn find_tone_transitions(envelope: &[f64], window: usize) -> Vec<(i64, TransitionKind)> {
50    // Reference level: median-ish via max of the envelope.
51    let peak = envelope.iter().cloned().fold(0.0f64, f64::max);
52    if peak <= 0.0 {
53        return vec![];
54    }
55    let hi = peak * 0.5; // above = "on"
56    let lo = peak * 0.01; // below = "off" (-20 dB)
57    let mut out = Vec::new();
58    let mut state_on = envelope.first().map(|&e| e > hi).unwrap_or(false);
59    for (i, &e) in envelope.iter().enumerate() {
60        if state_on && e < lo {
61            state_on = false;
62            out.push(((i * window) as i64, TransitionKind::Drop));
63        } else if !state_on && e > hi {
64            state_on = true;
65            out.push(((i * window) as i64, TransitionKind::Rise));
66        }
67    }
68    out
69}

71#[derive(Debug, Clone, Copy, PartialEq, Eq)]
72pub enum TransitionKind {
73    Drop,
74    Rise,
75}

77/// Max per-channel sample-to-sample step — the cheap click/discontinuity
78/// detector. Scripted changes always ramp (>= a few ms), so any step above
79/// the content's natural slew is a genuine discontinuity.
80pub fn max_step(interleaved: &[f32]) -> f32 {
81    let mut max = 0.0f32;
82    for ch in 0..2 {
83        let mut prev: Option<f32> = None;
84        for frame in interleaved.chunks_exact(2) {
85            let x = frame[ch];
86            if let Some(p) = prev {
87                max = max.max((x - p).abs());
88            }
89            prev = Some(x);
90        }
91    }
92    max
93}
94
95/// Peak absolute sample value.
96pub fn peak(interleaved: &[f32]) -> f32 {
97    interleaved.iter().fold(0.0f32, |m, &x| m.max(x.abs()))
98}

100#[cfg(test)]
101mod tests {
102    use super::*;

104    #[test]
105    fn goertzel_finds_its_tone() {
106        let sr = 48_000u32;
107        let mut buf = Vec::new();
108        for i in 0..sr as usize {
109            let t = i as f64 / sr as f64;
110            let v = (std::f64::consts::TAU * 1000.0 * t).sin() as f32 * 0.5;
111            buf.push(v);
112            buf.push(v);
113        }
114        let on = goertzel_power(&buf, sr, 1000.0, 0, 48_000);
115        let off = goertzel_power(&buf, sr, 3000.0, 0, 48_000);
116        assert!(on > 0.2, "tone power {on}");
117        assert!(off < on / 1000.0, "off-tone power {off} vs {on}");
118    }
119
120    #[test]
121    fn transitions_locate_a_gated_tone() {
122        let sr = 48_000u32;
123        let window = 64usize;
124        // 1 kHz tone, silent in the middle third [16000, 32000).
125        let mut buf = Vec::new();
126        for i in 0..48_000usize {
127            let t = i as f64 / sr as f64;
128            let amp = if (16_000..32_000).contains(&i) {
129                0.0
130            } else {
131                0.5
132            };
133            let v = (std::f64::consts::TAU * 1000.0 * t).sin() as f32 * amp as f32;
134            buf.push(v);
135            buf.push(v);
136        }
137        let env = tone_envelope(&buf, sr, 1000.0, window);
138        let tr = find_tone_transitions(&env, window);
139        assert_eq!(tr.len(), 2, "{tr:?}");
140        assert_eq!(tr[0].1, TransitionKind::Drop);
141        assert!(
142            (tr[0].0 - 16_000).unsigned_abs() as usize <= 2 * window,
143            "{tr:?}"
144        );
145        assert_eq!(tr[1].1, TransitionKind::Rise);
146        assert!(
147            (tr[1].0 - 32_000).unsigned_abs() as usize <= 2 * window,
148            "{tr:?}"
149        );
150    }
151
152    #[test]
153    fn max_step_catches_a_click() {
154        let mut buf = vec![0.0f32; 2000];
155        assert!(max_step(&buf) == 0.0);
156        buf[1000] = 0.9; // hard discontinuity on the left channel
157        assert!(max_step(&buf) >= 0.9);
158    }
