1//! Bridge between wall-clock instants and the suite's sample clock.
2//!
3//! The input capture thread stamps events with `Instant`s; judgment needs
4//! suite sample positions. kira's clock is readable only on the session
5//! thread, and its shared counter advances once per audio callback, so a
6//! single paired read is quantized to the device buffer (2–20+ ms). The
7//! bridge therefore collects paired `(Instant, clock_sample)` observations
8//! from the session loop and fits a line through them: the regression
9//! averages the per-chunk sawtooth to sub-millisecond error and its slope
10//! absorbs audio-vs-monotonic drift.
11//!
12//! Threading: the session loop calls [`ClockBridge::observe`]; any thread
13//! holding a clone may call [`ClockBridge::sample_at`]. Lock contention is
14//! negligible (two short critical sections per millisecond at most).
15
16use std::collections::VecDeque;
17use std::sync::{Arc, Mutex};
18use std::time::Instant;
19
20/// Ring capacity. With [`MIN_PAIR_SPACING`] the ring always spans ~2 s no
21/// matter how fast the session loop calls `observe`.
22const MAX_PAIRS: usize = 64;
23/// Minimum seconds between accepted observations — a fast loop is decimated
24/// rather than shrinking the fit window (slope needs time-span, not density).
25const MIN_PAIR_SPACING: f64 = 0.03;
26/// Minimum pairs before the bridge answers queries.
27const WARMUP_PAIRS: usize = 8;
28/// Minimum time span the ring must cover before fitting — slope from a
29/// shorter baseline is dominated by chunk-quantization noise.
30const MIN_FIT_SPAN: f64 = 1.0;
31/// A clock that reports the same sample for longer than this is not running
32/// yet (device warm-up), not merely chunk-quantized; restart the ring.
33const STALL_SECONDS: f64 = 0.1;
34/// Refit the line every this many observations.
35const REFIT_EVERY: usize = 16;
36/// Allowed slope deviation from the nominal sample rate (drift tolerance).
37const SLOPE_TOLERANCE: f64 = 0.005;
38
39#[derive(Debug, Clone, Copy)]
40struct LinModel {
41    /// Samples per second of wall time (≈ sample rate).
42    slope: f64,
43    /// Sample position at the bridge epoch.
44    intercept: f64,
45}
46
47#[derive(Debug, Clone, Copy, PartialEq)]
48pub struct BridgeStats {
49    pub pairs: usize,
50    /// RMS of fit residuals, in milliseconds of clock time.
51    pub residual_rms_ms: f64,
52    /// Fitted samples-per-second (should sit near the sample rate).
53    pub slope_hz: f64,
54}
55
56struct BridgeInner {
57    epoch: Instant,
58    sample_rate: f64,
59    pairs: VecDeque<(f64, i64)>, // (seconds since epoch, clock sample)
60    model: Option<LinModel>,
61    since_fit: usize,
62}
63
64/// Cloneable handle; all clones share one model.
65#[derive(Clone)]
66pub struct ClockBridge {
67    inner: Arc<Mutex<BridgeInner>>,
68}
69
70/// Lock the bridge state, recovering from poisoning (ADR 0039): the inner
71/// state is a self-repairing observation buffer (a bad fit refits within
72/// REFIT_EVERY ticks), so the last-written state is safe to reuse after a
73/// panicking writer — propagating would panic the 1 kHz capture thread and
74/// the per-frame session loop on every later call. Logs once per process.
75fn lock_or_recover<T>(m: &std::sync::Mutex<T>) -> std::sync::MutexGuard<'_, T> {
76    m.lock().unwrap_or_else(|poisoned| {
77        static LOGGED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
78        if !LOGGED.swap(true, std::sync::atomic::Ordering::Relaxed) {
79            tracing::error!("clock-bridge mutex poisoned; recovering with last state (ADR 0039)");
80        }
81        poisoned.into_inner()
82    })
83}

85impl ClockBridge {
86    pub fn new(sample_rate: u32) -> Self {
87        Self {
88            inner: Arc::new(Mutex::new(BridgeInner {
89                epoch: Instant::now(),
90                sample_rate: sample_rate as f64,
91                pairs: VecDeque::with_capacity(MAX_PAIRS),
92                model: None,
93                since_fit: 0,
94            })),
95        }
96    }
97
98    /// Record a paired observation. Call from the thread that can read the
99    /// session clock, once per loop tick, with `Instant::now()` taken as
100    /// close to the `clock_sample()` read as possible.
101    pub fn observe(&self, at: Instant, clock_sample: i64) {
102        let mut inner = lock_or_recover(&self.inner);
103        let t = at.duration_since(inner.epoch).as_secs_f64();
104        if let Some(&(last_t, last_s)) = inner.pairs.back() {
105            if t - last_t < MIN_PAIR_SPACING {
106                return;
107            }
108            // Startup stall: the device hasn't begun consuming yet, so the
109            // frozen pairs would poison the first fit. Start over from here.
110            if clock_sample == last_s && t - last_t > STALL_SECONDS {
111                inner.pairs.clear();
112                inner.model = None;
113            }
114        }
115        if inner.pairs.len() == MAX_PAIRS {
116            inner.pairs.pop_front();
117        }
118        inner.pairs.push_back((t, clock_sample));
119        inner.since_fit += 1;
120        let due = inner.model.is_none() || inner.since_fit >= REFIT_EVERY;
121        let span = inner
122            .pairs
123            .back()
124            .zip(inner.pairs.front())
125            .map(|((tb, _), (tf, _))| tb - tf)
126            .unwrap_or(0.0);
127        if due && inner.pairs.len() >= WARMUP_PAIRS && span >= MIN_FIT_SPAN {
128            inner.refit();
129            inner.since_fit = 0;
130        }
131    }
132
133    /// The suite clock sample at a wall-clock instant. `None` until warmed
134    /// up. Extrapolates freely — input events arrive after the newest pair.
135    pub fn sample_at(&self, at: Instant) -> Option<i64> {
136        let inner = lock_or_recover(&self.inner);
137        let model = inner.model?;
138        let t = at.duration_since(inner.epoch).as_secs_f64();
139        Some((model.slope * t + model.intercept).round() as i64)
140    }
141
142    pub fn stats(&self) -> BridgeStats {
143        let inner = lock_or_recover(&self.inner);
144        let (residual_rms_ms, slope_hz) = match inner.model {
145            Some(m) => {
146                let n = inner.pairs.len() as f64;
147                let ss: f64 = inner
148                    .pairs
149                    .iter()
150                    .map(|&(t, s)| {
151                        let e = s as f64 - (m.slope * t + m.intercept);
152                        e * e
153                    })
154                    .sum();
155                (((ss / n).sqrt() / inner.sample_rate) * 1000.0, m.slope)
156            }
157            None => (f64::NAN, f64::NAN),
158        };
159        BridgeStats {
160            pairs: inner.pairs.len(),
161            residual_rms_ms,
162            slope_hz,
163        }
164    }
165}

167impl BridgeInner {
168    fn refit(&mut self) {
169        let n = self.pairs.len() as f64;
170        let (mut st, mut ss) = (0.0, 0.0);
171        for &(t, s) in &self.pairs {
172            st += t;
173            ss += s as f64;
174        }
175        let (tm, sm) = (st / n, ss / n);
176        let (mut num, mut den) = (0.0, 0.0);
177        for &(t, s) in &self.pairs {
178            num += (t - tm) * (s as f64 - sm);
179            den += (t - tm) * (t - tm);
180        }
181        if den <= 0.0 {
182            return; // all observations at one instant; keep the old model
183        }
184        let mut slope = num / den;
185        let (lo, hi) = (
186            self.sample_rate * (1.0 - SLOPE_TOLERANCE),
187            self.sample_rate * (1.0 + SLOPE_TOLERANCE),
188        );
189        if !(lo..=hi).contains(&slope) {
190            // Early fits over a window still containing device spin-up are
191            // expected to be off; only a full ring makes this worth a warn.
192            if self.pairs.len() == MAX_PAIRS {
193                tracing::warn!(
194                    slope,
195                    nominal = self.sample_rate,
196                    "clock bridge slope outside tolerance; clamping"
197                );
198            } else {
199                tracing::debug!(slope, "early clock-bridge fit clamped (spin-up)");
200            }
201            slope = slope.clamp(lo, hi);
202        }
203        // Intercept re-derived from the means so a clamped slope still
204        // centers the line on the data.
205        self.model = Some(LinModel {
206            slope,
207            intercept: sm - slope * tm,
208        });
209    }
210}

217    const RATE: f64 = 48_000.0;
218
219    /// Feed pairs simulating a clock that advances once per `chunk` samples
220    /// (kira's per-callback update), observed every `every` seconds.
221    fn feed(bridge: &ClockBridge, base: Instant, secs: f64, every: f64, chunk: i64, drift: f64) {
222        let mut t = 0.0;
223        while t < secs {
224            let true_sample = (t * RATE * (1.0 + drift)) as i64;
225            let quantized = (true_sample / chunk) * chunk;
226            bridge.observe(base + Duration::from_secs_f64(t), quantized);
227            t += every;
228        }
229    }

231    #[test]
232    fn warms_up_then_answers() {
233        let bridge = ClockBridge::new(48_000);
234        let base = Instant::now();
235        for i in 0..WARMUP_PAIRS {
236            let t = i as f64 * 0.15; // 8 pairs span 1.05 s >= MIN_FIT_SPAN
237            assert!(bridge.sample_at(base).is_none() || i + 1 >= WARMUP_PAIRS);
238            bridge.observe(base + Duration::from_secs_f64(t), (t * RATE) as i64);
239        }
240        assert!(bridge.sample_at(base).is_some());
241    }

243    #[test]
244    fn averages_chunk_quantization_below_one_ms() {
245        let bridge = ClockBridge::new(48_000);
246        let base = Instant::now();
247        // 512-sample chunks (~10.7 ms) observed at 500 Hz for 2 s.
248        feed(&bridge, base, 2.0, 0.002, 512, 0.0);
249        // Quantization floors the readings, so the fit sits ~half a chunk
250        // low of the true clock — a constant judgment bias well under the
251        // pulse windows, and the *spread* is what matters. Assert both.
252        let t_query = 2.5;
253        let est = bridge
254            .sample_at(base + Duration::from_secs_f64(t_query))
255            .unwrap();
256        let truth = (t_query * RATE) as i64;
257        let bias_ms = (truth - est) as f64 / RATE * 1000.0;
258        assert!(
259            bias_ms.abs() < 512.0 / RATE * 1000.0,
260            "bias {bias_ms:.2} ms exceeds one chunk"
261        );
262        let stats = bridge.stats();
263        assert!(
264            stats.residual_rms_ms < 4.0,
265            "residual {:.2} ms too large",
266            stats.residual_rms_ms
267        );
268        assert!((stats.slope_hz - RATE).abs() < RATE * SLOPE_TOLERANCE);
269    }

271    #[test]
272    fn tracks_real_drift_within_tolerance() {
273        let bridge = ClockBridge::new(48_000);
274        let base = Instant::now();
275        // +0.2% fast audio clock, small chunks.
276        feed(&bridge, base, 2.0, 0.002, 128, 0.002);
277        let stats = bridge.stats();
278        assert!(
279            (stats.slope_hz - RATE * 1.002).abs() < 20.0,
280            "slope {} missed drifted rate",
281            stats.slope_hz
282        );
283        // Extrapolate 0.5 s past the last pair: error stays sub-millisecond
284        // because the slope carries the drift.
285        let t_query = 2.5;
286        let est = bridge
287            .sample_at(base + Duration::from_secs_f64(t_query))
288            .unwrap();
289        let truth = (t_query * RATE * 1.002) as i64;
290        assert!(
291            ((truth - est).abs() as f64) < RATE / 1000.0 + 128.0,
292            "extrapolation error {} samples",
293            truth - est
294        );
295    }

297    #[test]
298    fn absurd_slope_is_clamped() {
299        let bridge = ClockBridge::new(48_000);
300        let base = Instant::now();
301        // Clock racing at 2x: slope clamps to +0.5%.
302        for i in 0..MAX_PAIRS {
303            let t = i as f64 * 0.04;
304            bridge.observe(base + Duration::from_secs_f64(t), (t * RATE * 2.0) as i64);
305        }
306        let stats = bridge.stats();
307        assert!((stats.slope_hz - RATE * (1.0 + SLOPE_TOLERANCE)).abs() < 1.0);
308    }
