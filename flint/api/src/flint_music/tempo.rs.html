1//! Tempo-map time conversion: sample position <-> musical time (beats, bars).
2//!
3//! This is the arithmetic core the Conductor will be built on: an ordered list
4//! of `(sample, bpm, time_signature)` anchors, where each anchor starts a new
5//! bar. Meter changes are a first-class design feature, so everything works
6//! per-segment rather than assuming one global BPM.
7
8use crate::manifest::TempoAnchor;
9
10/// Milliseconds → whole samples, rounded — the one spelling of the crate's
11/// most-repeated conversion. Callers pass `sample_rate as f64`; keep the
12/// argument order (ms, rate) everywhere.
13pub fn ms_to_samples(ms: f64, sample_rate: f64) -> i64 {
14    (ms / 1000.0 * sample_rate).round() as i64
15}
16
17/// Snap `x` to the nearest integer when within `tol`, else leave it alone.
18/// Bar/beat boundaries land on fractional samples, so every place that floors
19/// or ceils a bar count must first snap through the half-sample tolerance.
20fn snap(x: f64, tol: f64) -> f64 {
21    if (x - x.round()).abs() <= tol {
22        x.round()
23    } else {
24        x
25    }
26}

28pub struct TempoMap {
29    anchors: Vec<TempoAnchor>,
30    sample_rate: u32,
31}
32
33impl TempoMap {
34    /// Anchors must be validated (origin at 0, strictly increasing) before
35    /// conversion results mean anything; construction itself is permissive.
36    pub fn new(anchors: Vec<TempoAnchor>, sample_rate: u32) -> Self {
37        Self {
38            anchors,
39            sample_rate,
40        }
41    }
42
43    pub fn sample_rate(&self) -> u32 {
44        self.sample_rate
45    }
46
47    /// Samples per beat under a given anchor.
48    fn samples_per_beat(&self, anchor: &TempoAnchor) -> f64 {
49        self.sample_rate as f64 * 60.0 / anchor.bpm
50    }
51
52    /// The anchor governing a sample position (the last anchor at or before it).
53    pub fn anchor_at(&self, sample: i64) -> Option<&TempoAnchor> {
54        self.anchors
55            .iter()
56            .take_while(|a| a.sample <= sample)
57            .last()
58    }
59
60    /// Beats elapsed since the governing anchor at `sample`.
61    pub fn beats_since_anchor(&self, sample: i64) -> Option<f64> {
62        let anchor = self.anchor_at(sample)?;
63        Some((sample - anchor.sample) as f64 / self.samples_per_beat(anchor))
64    }
65
66    /// Beats from suite start (sample 0) to `sample`, accumulating across
67    /// anchors. Negative samples (pre-roll) extrapolate the origin anchor
68    /// backwards. Returns None if the map is empty or has no origin anchor.
69    pub fn beats_at_sample(&self, sample: i64) -> Option<f64> {
70        if self.anchors.first().map(|a| a.sample)? != 0 {
71            return None;
72        }
73        if sample < 0 {
74            return Some(sample as f64 / self.samples_per_beat(self.anchors.first()?));
75        }
76        let mut beats = 0.0;
77        for (i, a) in self.anchors.iter().enumerate() {
78            let seg_end = self
79                .anchors
80                .get(i + 1)
81                .map(|n| n.sample.min(sample))
82                .unwrap_or(sample);
83            if seg_end <= a.sample {
84                break;
85            }
86            beats += (seg_end - a.sample) as f64 / self.samples_per_beat(a);
87            if seg_end == sample {
88                break;
89            }
90        }
91        Some(beats)
92    }
93
94    /// Exact inverse of [`beats_at_sample`](Self::beats_at_sample): the
95    /// (fractional) sample position of a beats-from-start value. Negative
96    /// beats extrapolate the origin anchor backwards.
97    pub fn sample_at_beat(&self, beats: f64) -> Option<f64> {
98        if self.anchors.first().map(|a| a.sample)? != 0 {
99            return None;
100        }
101        if beats < 0.0 {
102            return Some(beats * self.samples_per_beat(self.anchors.first()?));
103        }
104        let mut acc = 0.0;
105        for (i, a) in self.anchors.iter().enumerate() {
106            let spb = self.samples_per_beat(a);
107            let seg_beats = match self.anchors.get(i + 1) {
108                Some(next) => (next.sample - a.sample) as f64 / spb,
109                None => f64::INFINITY,
110            };
111            if beats <= acc + seg_beats {
112                return Some(a.sample as f64 + (beats - acc) * spb);
113            }
114            acc += seg_beats;
115        }
116        None
117    }
118
119    /// Bar index (from 0) and beat-within-bar at `sample`. Each anchor starts
120    /// a new bar, so a partial bar cut short by an anchor still counts as one
121    /// bar. Bar arithmetic snaps through the half-sample tolerance so bars
122    /// landing on fractional samples (e.g. 84 BPM at 48 kHz) resolve cleanly.
123    pub fn bar_beat_at_sample(&self, sample: i64) -> Option<(i64, f64)> {
124        if self.anchors.first().map(|a| a.sample)? != 0 {
125            return None;
126        }
127        if sample < 0 {
128            let a = self.anchors.first()?;
129            let beats = sample as f64 / self.samples_per_beat(a);
130            let bars = (beats / a.beats_per_bar as f64).floor();
131            return Some((bars as i64, beats - bars * a.beats_per_bar as f64));
132        }
133        let mut bars_before: i64 = 0;
134        for (i, a) in self.anchors.iter().enumerate() {
135            let spb = self.samples_per_beat(a);
136            let bpb = a.beats_per_bar.max(1) as f64;
137            let seg_end = self.anchors.get(i + 1).map(|n| n.sample);
138            let in_segment = match seg_end {
139                Some(end) => sample < end,
140                None => true,
141            };
142            if in_segment {
143                let beats = (sample - a.sample) as f64 / spb;
144                let bars = snap(beats / bpb, 0.5 / (spb * bpb));
145                let bar_in_seg = bars.floor();
146                // Snapped onto a bar line: exactly 0. Otherwise subtract
147                // rather than un-divide (beats/bpb*bpb is lossy).
148                let beat_in_bar = if bars == bar_in_seg {
149                    0.0
150                } else {
151                    (beats - bar_in_seg * bpb).max(0.0)
152                };
153                return Some((bars_before + bar_in_seg as i64, beat_in_bar));
154            }
155            let end = seg_end.expect("in_segment is false only with a next anchor");
156            let seg_bars = snap((end - a.sample) as f64 / (spb * bpb), 0.5 / (spb * bpb));
157            bars_before += seg_bars.ceil() as i64;
158        }
159        None
160    }
161
162    /// The (fractional) sample position of a bar line, `bar` counted from 0
163    /// with anchors starting new bars (the inverse of
164    /// [`bar_beat_at_sample`](Self::bar_beat_at_sample) at beat 0).
165    pub fn sample_at_bar(&self, bar: i64, beat_in_bar: f64) -> Option<f64> {
166        if self.anchors.first().map(|a| a.sample)? != 0 {
167            return None;
168        }
169        if bar < 0 {
170            let a = self.anchors.first()?;
171            let spb = self.samples_per_beat(a);
172            return Some((bar as f64 * a.beats_per_bar as f64 + beat_in_bar) * spb);
173        }
174        let mut bars_before: i64 = 0;
175        for (i, a) in self.anchors.iter().enumerate() {
176            let spb = self.samples_per_beat(a);
177            let bpb = a.beats_per_bar.max(1) as f64;
178            let seg_bars = match self.anchors.get(i + 1) {
179                Some(next) => snap(
180                    (next.sample - a.sample) as f64 / (spb * bpb),
181                    0.5 / (spb * bpb),
182                )
183                .ceil() as i64,
184                None => i64::MAX,
185            };
186            if bar - bars_before < seg_bars {
187                return Some(
188                    a.sample as f64 + ((bar - bars_before) as f64 * bpb + beat_in_bar) * spb,
189                );
190            }
191            bars_before += seg_bars;
192        }
193        None
194    }
195
196    /// Whether `sample` lands on a bar line, within half a sample of tolerance.
197    /// Each anchor is defined to start a new bar.
198    pub fn is_bar_line(&self, sample: i64) -> bool {
199        let Some(anchor) = self.anchor_at(sample) else {
200            return false;
201        };
202        if anchor.beats_per_bar <= 0 {
203            return false;
204        }
205        let spb = self.samples_per_beat(anchor);
206        let beats = (sample - anchor.sample) as f64 / spb;
207        let bars = beats / anchor.beats_per_bar as f64;
208        // tolerance: half a sample, expressed in bars
209        let tol = 0.5 / (spb * anchor.beats_per_bar as f64);
210        (bars - bars.round()).abs() <= tol
211    }
212}

218    fn map() -> TempoMap {
219        // 48 kHz: 120 BPM 4/4 until sample 960_000 (= 40 beats = 10 bars),
220        // then 90 BPM 3/4.
221        TempoMap::new(
222            vec![
223                TempoAnchor {
224                    sample: 0,
225                    bpm: 120.0,
226                    beats_per_bar: 4,
227                    beat_unit: 4,
228                },
229                TempoAnchor {
230                    sample: 960_000,
231                    bpm: 90.0,
232                    beats_per_bar: 3,
233                    beat_unit: 4,
234                },
235            ],
236            48_000,
237        )
238    }

240    #[test]
241    fn beats_accumulate_across_meter_change() {
242        let m = map();
243        assert_eq!(m.beats_at_sample(0), Some(0.0));
244        // 120 BPM @ 48kHz -> 24_000 samples per beat
245        assert_eq!(m.beats_at_sample(24_000), Some(1.0));
246        assert_eq!(m.beats_at_sample(960_000), Some(40.0));
247        // 90 BPM -> 32_000 samples per beat
248        assert_eq!(m.beats_at_sample(960_000 + 32_000), Some(41.0));
249    }
250
251    #[test]
252    fn sample_at_beat_inverts_beats_at_sample() {
253        let m = map();
254        for sample in [0i64, 24_000, 500_000, 960_000, 960_000 + 32_000, 2_000_000] {
255            let beats = m.beats_at_sample(sample).unwrap();
256            let back = m.sample_at_beat(beats).unwrap();
257            assert!(
258                (back - sample as f64).abs() < 0.5,
259                "round-trip {sample} -> {beats} -> {back}"
260            );
261        }
262        // Negative (pre-roll) extrapolation, both directions.
263        assert_eq!(m.beats_at_sample(-24_000), Some(-1.0));
264        assert_eq!(m.sample_at_beat(-1.0), Some(-24_000.0));
265    }

267    #[test]
268    fn bar_numbering_accumulates_across_meter_change() {
269        let m = map();
270        assert_eq!(m.bar_beat_at_sample(0), Some((0, 0.0)));
271        // 4/4 @ 120 BPM: bar = 96_000 samples.
272        assert_eq!(m.bar_beat_at_sample(96_000), Some((1, 0.0)));
273        let (bar, beat) = m.bar_beat_at_sample(96_000 + 24_000).unwrap();
274        assert_eq!(bar, 1);
275        assert!((beat - 1.0).abs() < 1e-9);
276        // Anchor at 960_000 = 10 full 4/4 bars, starts bar 10 in 3/4.
277        assert_eq!(m.bar_beat_at_sample(960_000), Some((10, 0.0)));
278        // One 3/4 bar @ 90 BPM = 96_000 samples.
279        assert_eq!(m.bar_beat_at_sample(960_000 + 96_000), Some((11, 0.0)));
280    }

321    #[test]
322    fn fractional_bars_snap_through_tolerance() {
323        // The prototype tempo: 84 BPM 4/4 @ 48 kHz -> bar = 960000/7 samples,
324        // never integral. Section starts are rounded; grid math must agree.
325        let m = TempoMap::new(
326            vec![TempoAnchor {
327                sample: 0,
328                bpm: 84.0,
329                beats_per_bar: 4,
330                beat_unit: 4,
331            }],
332            48_000,
333        );
334        let bar_len = 960_000.0 / 7.0;
335        for bar in [1i64, 7, 16, 100] {
336            let rounded = (bar as f64 * bar_len).round() as i64;
337            assert!(m.is_bar_line(rounded), "bar {bar} at rounded {rounded}");
338            assert_eq!(m.bar_beat_at_sample(rounded), Some((bar, 0.0)));
339            assert!((m.sample_at_bar(bar, 0.0).unwrap() - bar as f64 * bar_len).abs() < 0.5);
340        }
341    }
