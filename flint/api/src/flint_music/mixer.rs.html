1//! The six-bus stem mixer: runtime gain, low-pass, and detune per bus.
2//!
3//! Every bus in [`crate::BUSES`] gets a kira sub-track and filter — including
4//! silent ones, which simply have no sound playing. Silent buses stay
5//! addressable so later phases (reintegration entries, degraded alternates)
6//! can bring material in without rebuilding the mixer.
7//!
8//! kira handles are write-only, so each bus keeps shadow state of the last
9//! *commanded* values for introspection. During a tween the audible value lags
10//! the shadow value by design; the readout reports intent, not the DSP state.
11
12use flint_core::{FlintError, Result};
13use kira::backend::Backend;
14use kira::effect::filter::{FilterBuilder, FilterHandle, FilterMode};
15use kira::sound::static_sound::{StaticSoundData, StaticSoundHandle};
16use kira::sound::PlaybackPosition;
17use kira::track::{TrackBuilder, TrackHandle};
18use kira::{AudioManager, Decibels, PlaybackRate, StartTime, Tween};

20/// Cutoff meaning "filter open" — matches the spike and Milestone 0.
21pub const LPF_OPEN_HZ: f64 = 20_000.0;
22
23/// Last commanded values for one bus (shadow state; see module docs).
24#[derive(Debug, Clone, Copy)]
25pub struct BusState {
26    pub gain_db: f32,
27    pub lpf_hz: f64,
28    pub detune_semitones: f64,
29    /// Whether this bus has a stem loaded (silent buses: false).
30    pub playing: bool,
31    /// Last commanded bus↔alternate crossfade weight (0 = bus, 1 =
32    /// alternate; ADR 0032). Always 0 on buses without an alternate.
33    pub alternate_mix: f64,
34}
35
36impl Default for BusState {
37    fn default() -> Self {
38        Self {
39            gain_db: 0.0,
40            lpf_hz: LPF_OPEN_HZ,
41            detune_semitones: 0.0,
42            playing: false,
43            alternate_mix: 0.0,
44        }
45    }
46}

48/// Volume floor for the muted half of a bus↔alternate crossfade — matches
49/// the ladder's [`crate::ladder::DROP_DB`] silence convention.
50const ALT_MUTED_DB: f32 = -80.0;
51
52/// Linear crossfade weight → dB (floor at [`ALT_MUTED_DB`]).
53fn mix_db(weight: f64) -> f32 {
54    if weight <= 1e-4 {
55        ALT_MUTED_DB
56    } else {
57        (20.0 * weight.log10()).max(ALT_MUTED_DB as f64) as f32
58    }
59}

61/// A composed degraded alternate riding its bus's track (ADR 0032): a
62/// full-length stem, authored silent outside its manifest span, started at
63/// suite start muted — sample-locked by construction (the shared-tick
64/// start), crossfaded in per-sound volume when a ladder rung engages.
65struct AlternateVoice {
66    sound: StaticSoundHandle,
67    data: StaticSoundData,
68    from_sample: i64,
69    to_sample: i64,
70}

72pub struct StemBus {
73    pub name: String,
74    track: TrackHandle,
75    filter: FilterHandle,
76    sound: Option<StaticSoundHandle>,
77    /// The stem's sound data, retained so reintegration can re-play it at a
78    /// new position (Arc-backed; the clone is cheap). `None` for silent buses.
79    data: Option<StaticSoundData>,
80    alternate: Option<AlternateVoice>,
81    pub state: BusState,
82}
83
84impl StemBus {
85    /// Set bus gain in dB. `tween.start_time` may be a future clock time —
86    /// that is the scheduling hook every grid-aligned event uses.
87    pub fn set_gain(&mut self, db: f32, tween: Tween) {
88        self.track.set_volume(Decibels(db), tween);
89        self.state.gain_db = db;
90    }
91
92    /// Immediate gain set with the default tween — for callers (CLI
93    /// harnesses) that shouldn't need kira types for a plain fade.
94    pub fn set_gain_now(&mut self, db: f32) {
95        self.set_gain(db, Tween::default());
96    }
97
98    /// Set the low-pass cutoff in Hz ([`LPF_OPEN_HZ`] = effectively open).
99    pub fn set_lpf(&mut self, hz: f64, tween: Tween) {
100        self.filter.set_cutoff(hz, tween);
101        self.state.lpf_hz = hz;
102    }
103
104    /// Detune in semitones via playback rate (per-sound — kira has no
105    /// per-track pitch; see ADR 0001). Warble only: any nonzero detune breaks
106    /// sample-lock against the other stems while active, so offline alignment
107    /// asserts only hold at detune 0. No-op with a warning on silent buses.
108    /// An alternate voice warbles identically — whichever half of the
109    /// crossfade is audible, the rung's wobble is on it.
110    pub fn set_detune(&mut self, semitones: f64, tween: Tween) {
111        match &mut self.sound {
112            Some(sound) => {
113                let rate = PlaybackRate(2f64.powf(semitones / 12.0));
114                sound.set_playback_rate(rate, tween);
115                if let Some(alt) = &mut self.alternate {
116                    alt.sound.set_playback_rate(rate, tween);
117                }
118                self.state.detune_semitones = semitones;
119            }
120            None => tracing::warn!("set_detune on silent bus '{}' ignored", self.name),
121        }
122    }
123
124    /// Crossfade between this bus's stem and its composed alternate
125    /// (0 = bus, 1 = alternate; ADR 0032), via per-sound volumes so the
126    /// track's ladder trim and LPF apply to both halves equally. Span-gated:
127    /// outside the alternate's authored span the mix is forced to 0 (the
128    /// alternate is silent there by authoring, and the bus must not duck
129    /// against nothing). No-op when the bus has no alternate.
130    ///
131    /// Per-sound volume is also the reintegration entry-envelope channel —
132    /// callers (the ladder driver) must not write it during reassembly.
133    pub fn set_alternate_mix(&mut self, mix: f64, now_sample: i64, tween: Tween) {
134        let Some(alt) = &mut self.alternate else {
135            return;
136        };
137        let in_span = (alt.from_sample..alt.to_sample).contains(&now_sample);
138        let m = if in_span { mix.clamp(0.0, 1.0) } else { 0.0 };
139        if let Some(sound) = &mut self.sound {
140            sound.set_volume(Decibels(mix_db(1.0 - m)), tween);
141        }
142        alt.sound.set_volume(Decibels(mix_db(m)), tween);
143        self.state.alternate_mix = m;
144    }
145
146    /// Whether this bus carries a composed alternate.
147    pub fn has_alternate(&self) -> bool {
148        self.alternate.is_some()
149    }
150
151    /// The alternate's authored span `[from, to)`, if one is attached.
152    pub fn alternate_span(&self) -> Option<(i64, i64)> {
153        self.alternate
154            .as_ref()
155            .map(|a| (a.from_sample, a.to_sample))
156    }
157
158    pub fn sound(&self) -> Option<&StaticSoundHandle> {
159        self.sound.as_ref()
160    }
161
162    /// Fade this bus's playing sound to silence and stop it. The tween's
163    /// `start_time`/`duration` are the authored seam envelope — this is how
164    /// reintegration leaves the old timeline without a hard cut.
165    pub fn stop(&mut self, fade: Tween) {
166        if let Some(sound) = &mut self.sound {
167            sound.stop(fade);
168            self.state.playing = false;
169        }
170        if let Some(alt) = &mut self.alternate {
171            alt.sound.stop(fade);
172        }
173    }
174
175    /// Play this bus's retained stem again from `start_position_samples`
176    /// (a position within the stem = a suite sample, since stems start at
177    /// suite sample 0), scheduled at `start_at`, with an initial per-sound
178    /// volume of `entry_db` (the reintegration entry level — 0 for the lead
179    /// bus, deep negative for buses that ramp back in). Replaces the old
180    /// handle; detune resets to 0, which re-establishes sample-lock.
181    /// Track-level gain and LPF carry over untouched. No-op on silent buses.
182    pub fn replay_at(
183        &mut self,
184        start_position_samples: u64,
185        start_at: StartTime,
186        entry_db: f32,
187    ) -> Result<()> {
188        let Some(data) = &self.data else {
189            return Ok(());
190        };
191        let handle = self
192            .track
193            .play(
194                data.start_position(PlaybackPosition::Samples(start_position_samples as usize))
195                    .start_time(start_at)
196                    .volume(Decibels(entry_db)),
197            )
198            .map_err(|e| FlintError::AudioError(format!("bus '{}' replay: {e}", self.name)))?;
199        self.sound = Some(handle);
200        self.state.playing = true;
201        self.state.detune_semitones = 0.0;
202        // The alternate re-plays from the same tick, muted — the crossfade
203        // state resets to bus-active; the world re-enters clean (ADR 0032).
204        if let Some(alt) = &mut self.alternate {
205            let handle = self
206                .track
207                .play(
208                    alt.data
209                        .start_position(PlaybackPosition::Samples(start_position_samples as usize))
210                        .start_time(start_at)
211                        .volume(Decibels(ALT_MUTED_DB)),
212                )
213                .map_err(|e| {
214                    FlintError::AudioError(format!("bus '{}' alternate replay: {e}", self.name))
215                })?;
216            alt.sound = handle;
217            self.state.alternate_mix = 0.0;
218        }
219        Ok(())
220    }
221
222    /// Set the *per-sound* volume (independent of the track gain the ladder
223    /// drives). The reintegration reassembly ramp lives here: one clock-timed
224    /// tween per bus from its entry level to full, so entry envelopes never
225    /// fight ladder trims. No-op on silent buses.
226    pub fn set_sound_volume(&mut self, db: f32, tween: Tween) {
227        if let Some(sound) = &mut self.sound {
228            sound.set_volume(Decibels(db), tween);
229        }
230    }
231
232    /// Play an extra one-shot of this bus's retained stem as a *cue* — a
233    /// gesture outside the sample-locked timeline (the reintegration pickup
234    /// spin-up). The caller owns the returned handle and commands its
235    /// tweens/stop; the bus's own `sound` slot is untouched. `Ok(None)` on
236    /// silent buses.
237    pub fn play_cue(
238        &mut self,
239        start_position_samples: u64,
240        start_at: StartTime,
241        volume_db: f32,
242        rate_semitones: f64,
243    ) -> Result<Option<StaticSoundHandle>> {
244        let Some(data) = &self.data else {
245            return Ok(None);
246        };
247        let handle = self
248            .track
249            .play(
250                data.start_position(PlaybackPosition::Samples(start_position_samples as usize))
251                    .start_time(start_at)
252                    .volume(Decibels(volume_db))
253                    .playback_rate(PlaybackRate(2f64.powf(rate_semitones / 12.0))),
254            )
255            .map_err(|e| FlintError::AudioError(format!("bus '{}' cue: {e}", self.name)))?;
256        Ok(Some(handle))
257    }
258}

260/// All six buses, in canonical [`crate::BUSES`] order.
261pub struct BusMixer {
262    buses: Vec<StemBus>,
263}
264
265impl BusMixer {
266    /// Build one sub-track + filter per bus and schedule each provided stem
267    /// to start at `start_at` (one shared clock time = sample-locked start).
268    /// `stems` pairs each bus name with its sound data, `None` for silent.
269    pub fn build<B: Backend>(
270        manager: &mut AudioManager<B>,
271        stems: Vec<(String, Option<StaticSoundData>)>,
272        start_at: StartTime,
273    ) -> Result<Self> {
274        let mut buses = Vec::new();
275        for (name, data) in stems {
276            let mut builder = TrackBuilder::new();
277            let filter = builder.add_effect(
278                FilterBuilder::new()
279                    .mode(FilterMode::LowPass)
280                    .cutoff(LPF_OPEN_HZ),
281            );
282            let mut track = manager
283                .add_sub_track(builder)
284                .map_err(|e| FlintError::AudioError(format!("bus '{name}': {e}")))?;
285            let sound = match &data {
286                Some(data) => Some(
287                    track
288                        .play(data.start_time(start_at))
289                        .map_err(|e| FlintError::AudioError(format!("bus '{name}': {e}")))?,
290                ),
291                None => None,
292            };
293            let state = BusState {
294                playing: sound.is_some(),
295                ..Default::default()
296            };
297            buses.push(StemBus {
298                name,
299                track,
300                filter,
301                sound,
302                data,
303                alternate: None,
304                state,
305            });
306        }
307        Ok(Self { buses })
308    }
309
310    /// Attach a composed degraded alternate to a bus (ADR 0032): the
311    /// full-length stem starts muted at the same shared clock time as
312    /// everything else (sample-locked by construction), on the bus's own
313    /// track so ladder LPF/trim hit both crossfade halves. Call between
314    /// [`BusMixer::build`] and the clock start.
315    pub fn attach_alternate(
316        &mut self,
317        bus_name: &str,
318        data: StaticSoundData,
319        from_sample: i64,
320        to_sample: i64,
321        start_at: StartTime,
322    ) -> Result<()> {
323        let bus = self.bus_mut(bus_name).ok_or_else(|| {
324            FlintError::AudioError(format!("alternate for unknown bus '{bus_name}'"))
325        })?;
326        if bus.alternate.is_some() {
327            return Err(FlintError::AudioError(format!(
328                "bus '{bus_name}' already has an alternate (budget is one per bus)"
329            )));
330        }
331        let sound = bus
332            .track
333            .play(data.start_time(start_at).volume(Decibels(ALT_MUTED_DB)))
334            .map_err(|e| FlintError::AudioError(format!("bus '{bus_name}' alternate: {e}")))?;
335        bus.alternate = Some(AlternateVoice {
336            sound,
337            data,
338            from_sample,
339            to_sample,
340        });
341        Ok(())
342    }
343
344    pub fn bus_mut(&mut self, name: &str) -> Option<&mut StemBus> {
345        self.buses.iter_mut().find(|b| b.name == name)
346    }
347
348    pub fn buses(&self) -> impl Iterator<Item = &StemBus> {
349        self.buses.iter()
350    }
351
352    pub fn states(&self) -> impl Iterator<Item = (&str, &BusState)> {
353        self.buses.iter().map(|b| (b.name.as_str(), &b.state))
354    }
355
356    /// Whether every loaded stem has finished playing.
357    pub fn all_stopped(&self) -> bool {
358        self.buses
359            .iter()
360            .filter_map(|b| b.sound.as_ref())
361            .all(|s| s.state() == kira::sound::PlaybackState::Stopped)
362    }
363
364    /// Fade every playing stem to silence and stop it (see [`StemBus::stop`]).
365    pub fn stop_all(&mut self, fade: Tween) {
366        for bus in &mut self.buses {
367            bus.stop(fade);
368        }
369    }
370
371    /// Re-play every non-silent stem from `start_position_samples`, all
372    /// scheduled at the same `start_at` — one shared clock time is what makes
373    /// the re-entry sample-locked, exactly like the initial start.
374    /// `entry_db` gives each bus its entry level by name.
375    pub fn replay_all_at(
376        &mut self,
377        start_position_samples: u64,
378        start_at: StartTime,
379        entry_db: &dyn Fn(&str) -> f32,
380    ) -> Result<()> {
381        for bus in &mut self.buses {
382            let db = entry_db(&bus.name);
383            bus.replay_at(start_position_samples, start_at, db)?;
384        }
385        Ok(())
386    }

388    /// Max pairwise stem position disagreement, in seconds. Positions are
389    /// polled back-to-back; expect ~0 while all stems are playing.
390    pub fn max_stem_skew(&self) -> f64 {
391        let positions: Vec<f64> = self
392            .buses
393            .iter()
394            .flat_map(|b| b.sound.iter().chain(b.alternate.as_ref().map(|a| &a.sound)))
395            .map(|s| s.position())
396            .collect();
397        let mut max = 0.0f64;
398        for i in 0..positions.len() {
399            for j in (i + 1)..positions.len() {
400                max = max.max((positions[i] - positions[j]).abs());
401            }
402        }
403        max
404    }
