12/// The suite's static structure for the debug timeline map: everything the
13/// manifest pins that a strip-chart overlay wants to draw. Built once at
14/// session open ([`TimelineMap::build`]); dev surface only, never the
15/// player-facing world (the no-gamified-UI rule).
16#[derive(Debug, Clone, PartialEq, Default)]
17pub struct TimelineMap {
18    pub sample_rate: u32,
19    /// Suite-axis extent of the map: authored content end (last judgment
20    /// window close), rounded up to the next bar line.
21    pub end_sample: i64,
22    pub sections: Vec<TimelineSection>,
23    /// Every bar line within `end_sample`, precomputed so the overlay needs
24    /// no tempo math (correct across tempo/meter changes by construction).
25    pub bars: Vec<BarMark>,
26    /// The manifest's tempo anchors, verbatim (index 0 is the opening
27    /// tempo/meter, not a "change").
28    pub tempo_marks: Vec<TempoMark>,
29}

31#[derive(Debug, Clone, PartialEq)]
32pub struct TimelineSection {
33    pub name: String,
34    pub start_sample: i64,
35    /// The next section's start (the map's `end_sample` for the last one) —
36    /// sections carry no authored end.
37    pub end_sample: i64,
38    /// Listed in the manifest's `reintegration.re_entry_sections`.
39    pub re_entry: bool,
40}

42#[derive(Debug, Clone, Copy, PartialEq)]
43pub struct BarMark {
44    /// Bar index, counted from 0 (the tempo map's convention).
45    pub bar: i64,
46    pub sample: i64,
47}

49#[derive(Debug, Clone, Copy, PartialEq)]
50pub struct TempoMark {
51    pub sample: i64,
52    pub bpm: f64,
53    pub beats_per_bar: i64,
54    pub beat_unit: i64,
55}
56
57impl TimelineMap {
58    /// Snapshot the manifest's structure. `judge_end_sample` is the judge's
59    /// authored-content end ([`Judge::end_sample`]); the map extends to at
60    /// least the last section start, rounded up to a bar line.
61    pub fn build(manifest: &SuiteManifest, conductor: &Conductor, judge_end_sample: i64) -> Self {
62        let last_section_start = manifest
63            .sections
64            .iter()
65            .map(|s| s.start_sample)
66            .max()
67            .unwrap_or(0);
68        let raw_end = judge_end_sample.max(last_section_start).max(0);
69        // Round up to the next bar line so the strip ends on musical ground.
70        let end_bar = conductor.position_at_sample(raw_end).bar + 1;
71        let end_sample = conductor.sample_at_bar(end_bar, 0.0);
72
73        let mut sections: Vec<TimelineSection> = manifest
74            .sections
75            .iter()
76            .map(|s| TimelineSection {
77                name: s.name.clone(),
78                start_sample: s.start_sample,
79                end_sample, // patched below from the next section's start
80                re_entry: manifest.reintegration.re_entry_sections.contains(&s.name),
81            })
82            .collect();
83        for i in 0..sections.len().saturating_sub(1) {
84            sections[i].end_sample = sections[i + 1].start_sample;
85        }
86
87        let mut bars = Vec::new();
88        // Hard cap defends against a degenerate tempo map, not real content.
89        for bar in 0..10_000 {
90            let sample = conductor.sample_at_bar(bar, 0.0);
91            if sample > end_sample {
92                break;
93            }
94            bars.push(BarMark { bar, sample });
95        }
96
97        Self {
98            sample_rate: manifest.sample_rate,
99            end_sample,
100            sections,
101            bars,
102            tempo_marks: manifest
103                .tempo
104                .iter()
105                .map(|a| TempoMark {
106                    sample: a.sample,
107                    bpm: a.bpm,
108                    beats_per_bar: a.beats_per_bar,
109                    beat_unit: a.beat_unit,
110                })
111                .collect(),
112        }
113    }
114}

116/// A reintegration-sequencer event retained for the debug timeline (this-run
117/// history). Suite-sample addressed; pushed by [`ChartCore::step_seq`] as
118/// events fire, read only by [`ChartSession::timeline_frame`] — no
119/// deterministic path touches it.
120#[derive(Debug, Clone, Copy, PartialEq, Eq)]
121pub enum SeamMarkKind {
122    /// Full-fail latched: the rewind gesture begins here.
123    FullFail,
124    /// The seam committed: playback re-entered at `re_entry_sample`.
125    Seam,
126    /// Reassembly ramp finished.
127    ReassemblyComplete,
128}
129
130#[derive(Debug, Clone, Copy, PartialEq)]
131pub struct SeamMark {
132    pub kind: SeamMarkKind,
133    /// Where on the suite axis the event happened (for `Seam` this IS the
134    /// re-entry point — post-jump suite position).
135    pub suite_sample: i64,
136    pub re_entry_sample: i64,
137}

139/// One judged pulse retained for the debug timeline (this-run history).
140#[derive(Debug, Clone, Copy, PartialEq)]
141pub struct PulseMark {
142    pub sample: i64,
143    /// Timing error in ms (hits only; 0.0 for miss/spurious).
144    pub err_ms: f64,
145    pub kind: PulseKind,
146}

148/// Per-frame debug-timeline snapshot: the playhead plus this-run history.
149/// Pairs with the static [`TimelineMap`].
150#[derive(Debug, Clone, PartialEq, Default)]
151pub struct TimelineFrame {
152    /// Latency-compensated suite sample (negative during preroll).
153    pub playhead_sample: i64,
154    /// The monotonic raw clock (never rewinds).
155    pub raw_clock_sample: i64,
156    /// `raw − suite`; changes once per committed seam.
157    pub timeline_offset: i64,
158    pub preroll: bool,
159    pub seams: Vec<SeamMark>,
160    pub pulses: Vec<PulseMark>,
161}
