1//! Shared TOML plumbing for every config/content format this crate parses.
2//!
3//! One place for the read → parse → `schema_version` contract that the seven
4//! per-format parsers previously each re-implemented, plus the
5//! "missing or malformed" accessor helpers. Error text is unchanged from the
6//! per-module originals (each helper takes the format's display name).
7//!
8//! Two `schema_version` policies exist on purpose — content formats (chart,
9//! event script, suite manifest) REQUIRE the key and leave value checks to
10//! their validators (golden fixtures pin those error codes), while config
11//! files (coherence/gradient/haptics/ladder) default an absent key to 0 and
12//! reject non-zero at parse. Unifying them is a behavior change; propose it,
13//! don't slip it in here.
14
15use flint_core::toml_util::toml_f64;
16use flint_core::{FlintError, Result};
17
18/// How a format treats `schema_version`.
19pub(crate) enum VersionPolicy {
20    /// Key must be present (content formats); the value is returned untouched.
21    Required,
22    /// Absent means 0; any non-zero value is rejected here (config files).
23    DefaultZeroStrict,
24}
25
26/// Parse TOML text and resolve `schema_version` under the format's policy.
27/// Every error carries `name` as its prefix, matching the historical text.
28pub(crate) fn parse_versioned(
29    name: &str,
30    text: &str,
31    policy: VersionPolicy,
32) -> Result<(toml::Value, i64)> {
33    let root: toml::Value = text
34        .parse()
35        .map_err(|e| FlintError::TomlParseError(format!("{name}: {e}")))?;
36    let version = root.get("schema_version").and_then(|v| v.as_integer());
37    let version = match policy {
38        VersionPolicy::Required => version.ok_or_else(|| bad(name, "schema_version"))?,
39        VersionPolicy::DefaultZeroStrict => {
40            let v = match version {
41                Some(v) => v,
42                None => {
43                    // ADR 0040: absent still parses as 0 (backward compatible
44                    // with every committed config) but is no longer silent —
45                    // an unversioned or typo'd key used to be untraceable.
46                    tracing::warn!(
47                        "{name}: no `schema_version` key — assuming 0 (declare `schema_version = 0`)"
48                    );
49                    0
50                }
51            };
52            if v != 0 {
53                return Err(FlintError::ValidationError(format!(
54                    "{name}: unknown schema_version {v}"
55                )));
56            }
57            v
58        }
59    };
60    Ok((root, version))
61}

63/// The shared shape-error: `"{name}: missing or malformed `{what}`"`.
64pub(crate) fn bad(name: &str, what: &str) -> FlintError {
65    FlintError::ParseError(format!("{name}: missing or malformed `{what}`"))
66}

68// -- accessors over a root value ----------------------------------------------
69
70pub(crate) fn table<'a>(
71    name: &str,
72    v: &'a toml::Value,
73    key: &str,
74) -> Result<&'a toml::value::Table> {
75    v.get(key)
76        .and_then(|v| v.as_table())
77        .ok_or_else(|| bad(name, key))
78}
79
80pub(crate) fn array<'a>(name: &str, v: &'a toml::Value, key: &str) -> Result<&'a Vec<toml::Value>> {
81    v.get(key)
82        .and_then(|v| v.as_array())
83        .ok_or_else(|| bad(name, key))
84}
85
86/// Array-of-tables, absent or non-array treated as empty (the chart idiom).
87pub(crate) fn tables<'a>(root: &'a toml::Value, key: &str) -> Vec<&'a toml::value::Table> {
88    root.get(key)
89        .and_then(|v| v.as_array())
90        .map(|arr| arr.iter().filter_map(|v| v.as_table()).collect())
91        .unwrap_or_default()
92}

94// -- accessors inside a table -------------------------------------------------
95// `ctx` is what the error names; callers pass either the key itself or a
96// fuller path like `curves[3].beat`.
97
98pub(crate) fn int_in(name: &str, t: &toml::value::Table, key: &str) -> Result<i64> {
99    t.get(key)
100        .and_then(|v| v.as_integer())
101        .ok_or_else(|| bad(name, key))
102}
103
104pub(crate) fn float_in(name: &str, t: &toml::value::Table, key: &str, ctx: &str) -> Result<f64> {
105    t.get(key).and_then(toml_f64).ok_or_else(|| bad(name, ctx))
106}
107
108pub(crate) fn string_in(
109    name: &str,
110    t: &toml::value::Table,
111    key: &str,
112    ctx: &str,
113) -> Result<String> {
114    t.get(key)
115        .and_then(|v| v.as_str())
116        .map(String::from)
117        .ok_or_else(|| bad(name, ctx))
118}
119
120/// Array of f64 (any element malformed fails the whole array).
121pub(crate) fn floats(name: &str, v: Option<&toml::Value>, ctx: &str) -> Result<Vec<f64>> {
122    v.and_then(|v| v.as_array())
123        .map(|arr| {
124            arr.iter()
125                .map(|e| toml_f64(e).ok_or_else(|| bad(name, ctx)))
126                .collect()
127        })
128        .unwrap_or_else(|| Err(bad(name, ctx)))
129}

131/// `root.[table].key` as f64 with a default — the config-file read idiom.
132pub(crate) fn section_f64(root: &toml::Value, table: &str, key: &str, default: f64) -> f64 {
133    root.get(table)
134        .and_then(|t| t.get(key))
135        .and_then(toml_f64)
136        .unwrap_or(default)
