1//! Lookahead scheduler: events queued in musical time, landing on the grid.
2//!
3//! kira's clock-timed `Tween`s do the sample-accurate landing; the scheduler's
4//! job is to hold events beyond a lookahead horizon, resolve musical time to
5//! a sample position **once, at push** (via the Conductor — a resolved event
6//! never re-interprets time), and issue the kira commands as events enter the
7//! horizon. Section transitions and reintegration entries in later phases are
8//! consumers of this same queue.
9
10use crate::conductor::Conductor;
11use crate::mixer::BusMixer;
12use kira::clock::{ClockHandle, ClockTime};
13use kira::{StartTime, Tween};
14use std::cmp::Reverse;
15use std::collections::BinaryHeap;
16use std::time::Duration;
17
18/// When an event should land, in any of the Conductor's time vocabularies.
19#[derive(Debug, Clone, Copy, PartialEq)]
20pub enum EventTime {
21    Sample(i64),
22    Beat(f64),
23    BarBeat { bar: i64, beat: f64 },
24}
25
26/// What lands.
27#[derive(Debug, Clone, PartialEq)]
28pub enum BusAction {
29    SetGain {
30        db: f32,
31        ramp_ms: f64,
32    },
33    SetLpf {
34        hz: f64,
35        ramp_ms: f64,
36    },
37    SetDetune {
38        semitones: f64,
39        ramp_ms: f64,
40    },
41    /// No audio effect; returned from `pump` for logging / section hooks.
42    Marker(String),
43}
44
45#[derive(Debug, Clone, PartialEq)]
46pub struct ScheduledEvent {
47    pub at: EventTime,
48    /// Target bus; `None` is only meaningful for `Marker`.
49    pub bus: Option<String>,
50    pub action: BusAction,
51}
52
53/// An event the scheduler has issued (its tween is queued in kira, or it was
54/// a marker), reported back from `pump`.
55#[derive(Debug, Clone, PartialEq)]
56pub struct FiredEvent {
57    pub sample: i64,
58    pub bus: Option<String>,
59    pub action: BusAction,
60    /// True if the event's sample was already behind the play head when
61    /// issued; it fired immediately instead of landing on the grid.
62    pub late: bool,
63}
64
65#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
66struct QueueKey {
67    sample: i64,
68    /// Push order breaks ties deterministically.
69    seq: u64,
70}
71
72pub struct Scheduler {
73    horizon_samples: i64,
74    next_seq: u64,
75    pending: BinaryHeap<Reverse<(QueueKey, EventKeyed)>>,
76}
77
78/// BinaryHeap needs Ord; events themselves aren't Ord, so they ride along
79/// behind the key and are never compared (QueueKey is unique via seq).
80#[derive(Debug, Clone)]
81struct EventKeyed(ScheduledEvent);
82
83impl PartialEq for EventKeyed {
84    fn eq(&self, _: &Self) -> bool {
85        true
86    }
87}
88impl Eq for EventKeyed {}
89impl PartialOrd for EventKeyed {
90    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
91        Some(self.cmp(other))
92    }
93}
94impl Ord for EventKeyed {
95    fn cmp(&self, _: &Self) -> std::cmp::Ordering {
96        std::cmp::Ordering::Equal
97    }
98}
99
100impl Scheduler {
101    /// `horizon_samples`: how far ahead of the play head events are handed to
102    /// kira. Must comfortably exceed the caller's pump interval plus output
103    /// latency; the realtime default is 500 ms of samples.
104    pub fn new(horizon_samples: i64) -> Self {
105        Self {
106            horizon_samples,
107            next_seq: 0,
108            pending: BinaryHeap::new(),
109        }
110    }
111
112    pub fn default_horizon(sample_rate: u32) -> i64 {
113        sample_rate as i64 / 2
114    }
115
116    /// Queue an event, resolving its musical time to a sample now.
117    pub fn push(&mut self, ev: ScheduledEvent, conductor: &Conductor) {
118        let sample = match ev.at {
119            EventTime::Sample(s) => s,
120            EventTime::Beat(b) => conductor.sample_at_beat(b),
121            EventTime::BarBeat { bar, beat } => conductor.sample_at_bar(bar, beat),
122        };
123        let key = QueueKey {
124            sample,
125            seq: self.next_seq,
126        };
127        self.next_seq += 1;
128        self.pending.push(Reverse((key, EventKeyed(ev))));
129    }
130
131    pub fn pending_len(&self) -> usize {
132        self.pending.len()
133    }
134
135    /// Drop every event not yet handed to kira. Reintegration calls this at
136    /// full-fail: pending events were resolved against the old timeline and
137    /// would land at wrong musical positions after the seam. Returns how many
138    /// were discarded. Events already inside the horizon are kira's — they
139    /// cannot be withdrawn, which is one reason ladder moves don't use the
140    /// scheduler.
141    pub fn flush(&mut self) -> usize {
142        let n = self.pending.len();
143        self.pending.clear();
144        n
145    }
146
147    /// Issue every pending event within `now_sample + horizon` as a
148    /// clock-timed kira command (or immediately, with a warning, if its time
149    /// has already passed). Returns what was issued, in order.
150    /// `tick_base` maps suite samples to clock ticks (`tick = sample +
151    /// tick_base`): the pre-roll length, plus the session's timeline offset
152    /// once reintegration seams exist.
153    pub fn pump(
154        &mut self,
155        now_sample: i64,
156        clock: &ClockHandle,
157        tick_base: i64,
158        mixer: &mut BusMixer,
159    ) -> Vec<FiredEvent> {
160        let mut fired = Vec::new();
161        while let Some(Reverse((key, _))) = self.pending.peek() {
162            if key.sample > now_sample + self.horizon_samples {
163                break;
164            }
165            let Reverse((key, EventKeyed(ev))) = self.pending.pop().expect("peeked");
166            // Also immediate if the target tick would precede clock start
167            // (an event scheduled before the pre-roll window can reach).
168            let late = key.sample < now_sample || key.sample + tick_base < 0;
169            let start_time = if late {
170                tracing::warn!(
171                    "event at sample {} issued late (play head at {}); firing immediately",
172                    key.sample,
173                    now_sample
174                );
175                StartTime::Immediate
176            } else {
177                // Suite sample -> clock tick via the caller's base (pre-roll
178                // plus any reintegration timeline offset).
179                StartTime::ClockTime(ClockTime::from_ticks_u64(
180                    clock,
181                    (key.sample + tick_base) as u64,
182                ))
183            };
184            let tween = |ramp_ms: f64| Tween {
185                start_time,
186                duration: Duration::from_secs_f64(ramp_ms.max(0.0) / 1000.0),
187                ..Default::default()
188            };
189            match (&ev.action, ev.bus.as_deref()) {
190                (BusAction::Marker(_), _) => {}
191                (action, Some(bus_name)) => match mixer.bus_mut(bus_name) {
192                    Some(bus) => match action {
193                        BusAction::SetGain { db, ramp_ms } => bus.set_gain(*db, tween(*ramp_ms)),
194                        BusAction::SetLpf { hz, ramp_ms } => bus.set_lpf(*hz, tween(*ramp_ms)),
195                        BusAction::SetDetune { semitones, ramp_ms } => {
196                            bus.set_detune(*semitones, tween(*ramp_ms))
197                        }
198                        BusAction::Marker(_) => unreachable!(),
199                    },
200                    None => {
201                        tracing::warn!("scheduled event targets unknown bus '{bus_name}'; dropped")
202                    }
203                },
204                (_, None) => {
205                    tracing::warn!("scheduled bus action has no bus; dropped");
206                }
207            }
208            fired.push(FiredEvent {
209                sample: key.sample,
210                bus: ev.bus,
211                action: ev.action,
212                late,
213            });
214        }
215        fired
216    }
