1//! Event scripts: a small TOML format describing a scripted session — bus
2//! changes and markers at musical times — for the offline render harness and
3//! automated Milestone tests.
4//!
5//! This is a tool/test format, not part of the pinned suite schema, but it is
6//! versioned all the same. Times use the console readout's vocabulary:
7//! `bar:N` and `bar:N:B` are 1-based (bar 1 beat 1 = suite start), while
8//! `beat:F` (beats from start) and `sample:N` are absolute 0-based axes.
9//!
10//! ```toml
11//! schema_version = 0
12//!
13//! [[events]]
14//! at = "bar:9"
15//! bus = "home_theme"
16//! action = "set_gain"   # set_gain | set_lpf | set_detune | marker
17//! db = -60.0            # set_gain: db; set_lpf: hz; set_detune: semitones;
18//! ramp_ms = 10.0        # marker: label
19//! ```
20
21use crate::config_toml::{self, VersionPolicy};
22use crate::scheduler::{BusAction, EventTime, ScheduledEvent};
23use flint_core::toml_util::toml_f64;
24use flint_core::{FlintError, Result};
25use std::path::Path;
26
27pub const EVENT_SCHEMA_VERSION: i64 = 0;
28
29#[derive(Debug, Clone)]
30pub struct EventScript {
31    pub schema_version: i64,
32    pub events: Vec<ScheduledEvent>,
33}
34
35impl EventScript {
36    pub fn load(path: &Path) -> Result<Self> {
37        let text = std::fs::read_to_string(path)?;
38        Self::parse(&text)
39    }
40
41    pub fn parse(text: &str) -> Result<Self> {
42        let (root, schema_version) =
43            config_toml::parse_versioned(NAME, text, VersionPolicy::Required)?;
44        if schema_version != EVENT_SCHEMA_VERSION {
45            return Err(FlintError::ParseError(format!(
46                "event script: schema_version {schema_version} not supported (want {EVENT_SCHEMA_VERSION})"
47            )));
48        }
49
50        let mut events = Vec::new();
51        if let Some(arr) = root.get("events").and_then(|v| v.as_array()) {
52            for (i, v) in arr.iter().enumerate() {
53                let t = v.as_table().ok_or_else(|| bad(&format!("events[{i}]")))?;
54                events.push(parse_event(t, i)?);
55            }
56        }
57        Ok(Self {
58            schema_version,
59            events,
60        })
61    }
62}
63
64fn parse_event(t: &toml::value::Table, i: usize) -> Result<ScheduledEvent> {
65    let at_str = t
66        .get("at")
67        .and_then(|v| v.as_str())
68        .ok_or_else(|| bad(&format!("events[{i}].at")))?;
69    let at = parse_time(at_str).ok_or_else(|| bad(&format!("events[{i}].at ('{at_str}')")))?;
70
71    let action_str = t
72        .get("action")
73        .and_then(|v| v.as_str())
74        .ok_or_else(|| bad(&format!("events[{i}].action")))?;
75    let ramp_ms = t.get("ramp_ms").and_then(toml_f64).unwrap_or(0.0);
76    let num = |key: &str| {
77        t.get(key)
78            .and_then(toml_f64)
79            .ok_or_else(|| bad(&format!("events[{i}].{key}")))
80    };
81    let action = match action_str {
82        "set_gain" => BusAction::SetGain {
83            db: num("db")? as f32,
84            ramp_ms,
85        },
86        "set_lpf" => BusAction::SetLpf {
87            hz: num("hz")?,
88            ramp_ms,
89        },
90        "set_detune" => BusAction::SetDetune {
91            semitones: num("semitones")?,
92            ramp_ms,
93        },
94        "marker" => BusAction::Marker(
95            t.get("label")
96                .and_then(|v| v.as_str())
97                .unwrap_or("")
98                .to_string(),
99        ),
100        other => {
101            return Err(FlintError::ParseError(format!(
102                "event script: events[{i}].action '{other}' unknown"
103            )))
104        }
105    };
106
107    let bus = t.get("bus").and_then(|v| v.as_str()).map(String::from);
108    if bus.is_none() && !matches!(action, BusAction::Marker(_)) {
109        return Err(bad(&format!("events[{i}].bus (required for {action_str})")));
110    }
111    Ok(ScheduledEvent { at, bus, action })
112}

114/// `bar:N` / `bar:N:B` (1-based, matching the console readout) or
115/// `beat:F` / `sample:N` (absolute, 0-based).
116fn parse_time(s: &str) -> Option<EventTime> {
117    let mut parts = s.split(':');
118    match parts.next()? {
119        "bar" => {
120            let bar: i64 = parts.next()?.parse().ok()?;
121            let beat: f64 = match parts.next() {
122                Some(b) => b.parse().ok()?,
123                None => 1.0,
124            };
125            if bar < 1 || beat < 1.0 || parts.next().is_some() {
126                return None;
127            }
128            Some(EventTime::BarBeat {
129                bar: bar - 1,
130                beat: beat - 1.0,
131            })
132        }
133        "beat" => {
134            let b: f64 = parts.next()?.parse().ok()?;
135            parts.next().is_none().then_some(EventTime::Beat(b))
136        }
137        "sample" => {
138            let n: i64 = parts.next()?.parse().ok()?;
139            parts.next().is_none().then_some(EventTime::Sample(n))
140        }
141        _ => None,
142    }
143}

151#[cfg(test)]
152mod tests {
153    use super::*;
154
155    #[test]
156    fn parses_all_forms() {
157        let s = EventScript::parse(
158            r#"
159schema_version = 0
160
161[[events]]
162at = "bar:9"
163bus = "home_theme"
164action = "set_gain"
165db = -60.0
166ramp_ms = 10.0
167
168[[events]]
169at = "bar:13:2.5"
170bus = "home_theme"
171action = "set_gain"
172db = 0
173
174[[events]]
175at = "beat:16.5"
176bus = "texture"
177action = "set_lpf"
178hz = 400.0
179
180[[events]]
181at = "sample:48000"
182action = "marker"
183label = "checkpoint"
184"#,
185        )
186        .unwrap();
187        assert_eq!(s.events.len(), 4);
188        assert_eq!(s.events[0].at, EventTime::BarBeat { bar: 8, beat: 0.0 });
189        assert_eq!(s.events[1].at, EventTime::BarBeat { bar: 12, beat: 1.5 });
190        // Integer TOML value for a float field must coerce (toml_util rule).
191        assert!(matches!(
192            s.events[1].action,
193            BusAction::SetGain { db, .. } if db == 0.0
194        ));
195        assert_eq!(s.events[2].at, EventTime::Beat(16.5));
196        assert_eq!(s.events[3].at, EventTime::Sample(48_000));
197        assert!(matches!(&s.events[3].action, BusAction::Marker(l) if l == "checkpoint"));
198    }

200    #[test]
201    fn rejects_bad_input() {
202        assert!(EventScript::parse("schema_version = 1").is_err());
203        let bad_bus = r#"
204schema_version = 0
205[[events]]
206at = "bar:1"
207action = "set_gain"
208db = 0.0
209"#;
210        assert!(EventScript::parse(bad_bus).is_err(), "bus required");
211        let bad_time = r#"
212schema_version = 0
213[[events]]
214at = "bar:0"
215bus = "texture"
216action = "set_lpf"
217hz = 400.0
218"#;
219        assert!(EventScript::parse(bad_time).is_err(), "bars are 1-based");
220    }
