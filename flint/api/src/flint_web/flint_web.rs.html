1//! Flint in a browser (#94; async startup #99, bundles #98, docs/design/web.md).
2//!
3//! The web twin of `main.rs`. Arguments come from the page's query string
4//! instead of clap, and the player starts with winit's `spawn_app` instead of
5//! the blocking `run_app`. `PlayerApp` then asks for its GPU device without
6//! blocking (`player_app/web_startup.rs`) and finishes starting up when it
7//! lands.
8//!
9//! A game arrives as a bundle (`flint bundle`, `flint_core::bundle`): the
10//! page fetches it with a progress bar, installs its files as the vfs, and
11//! loads the scene the manifest names, exactly as the desktop reads a game
12//! folder. Its late bundles (audio) are fetched once the player is handed to
13//! the browser; when one lands the vfs grows and the player loads the sounds
14//! it couldn't find at start.
15//!
16//! Query string (all optional):
17//!   game=<id>       a game on play.chaps.dev (#103): the bundle at
18//!                   games/<id>/<id>.flintpack, which the site's deploy packs
19//!   bundle=<url>    a .flintpack, relative to the page (wins over game)
20//!   scene=<path>    with bundle: a scene inside it instead of the manifest's;
21//!                   without: a scene TOML fetched from <url> (default: the
22//!                   first-light demo, which needs no other files)
23//!   msaa=1|4        scene-pass MSAA (ADR 0058)
24//!   scale=<f>|auto  the 3D scene's render scale (#122). By default it is
25//!                   drawn at no more than DPR 1.5 (`WEB_DPR_CAP`) and the
26//!                   HUD at the full DPR; a number from 0.25 to 1 draws at
27//!                   that scale instead, whatever the DPR (`scale=1` is
28//!                   native); `auto` keeps the cap and steps under it when
29//!                   frames run over budget for seconds
30//!   quality=k=v,... quality settings over msaa and scale (FlintKart#55), as
31//!                   flint-player's --quality: `quality=shadow_cascades=1,ssao=0`
32//!                   (comma-separated; scripts change them with set_quality)
33//!   width=, height= fix the canvas to this many CSS pixels instead of the page
34//!   frame_step=<s>  step game time exactly s per frame (`--frame-step`)
35//!   hold_at=<s>     stop game time at s seconds, for a screenshot of an
36//!                   exact moment (with frame_step; `PlayerApp::hold_at`)
37//!   perf=1|cpu      the perf readout in a corner (#120): frame times over
38//!                   the last 10 s, each phase's CPU time, each pass's GPU
39//!                   time where the adapter has timestamp-query (`cpu`
40//!                   leaves GPU timing off), and a line to copy
41//!   fps=30|60|auto|native  which refreshes are drawn (#140). A rate is
42//!                   drawn as a whole number of refreshes (30 is every
43//!                   other at 60 Hz, every fourth at 120); `auto` draws
44//!                   every refresh while the frames hold it and drops to
45//!                   the rate they do hold when they don't; `native` draws
46//!                   every refresh unpaced. Default: `auto` where the
47//!                   primary pointer is a finger, `native` elsewhere
48//!   loop=poll       run winit's loop flat out (`ControlFlow::Poll`), as
49//!                   before #121, to compare against; the default draws once
50//!                   per `requestAnimationFrame`
51//!
52//! Build and serve from `crates/flint-player`:
53//!   trunk serve            # http://127.0.0.1:8080
54//!   trunk build --release  # dist/, wasm-opt -Oz applied
55

104        touch: flint_runtime::TouchControlsMode,

111    impl WebArgs {
112        fn from_query() -> Self {
113            let params = web_sys::window()
114                .and_then(|w| w.location().search().ok())
115                .and_then(|q| web_sys::UrlSearchParams::new_with_str(&q).ok());
116            let get = |key: &str| params.as_ref().and_then(|p| p.get(key));
117            let num = |key: &str| get(key).and_then(|v| v.parse::<u32>().ok());
118            let secs = |key: &str| {
119                get(key)
120                    .and_then(|v| v.parse::<f64>().ok())
121                    .filter(|s| s.is_finite() && *s >= 0.0)
122            };
123            Self {
124                bundle: get("bundle")
125                    .filter(|b| !b.is_empty())
126                    .or_else(|| get("game").as_deref().and_then(game_bundle)),
127                scene: get("scene").filter(|s| !s.is_empty()),
128                msaa: num("msaa").filter(|&m| m == 1 || m == 4).unwrap_or(1),
129                render_scale: get("scale")
130                    .and_then(|v| flint_player::RenderScaleSetting::web().parse(&v))
131                    .unwrap_or_else(flint_player::RenderScaleSetting::web),
132                size: num("width").zip(num("height")),
133                frame_step: secs("frame_step").filter(|s| *s > 0.0),
134                hold_at: secs("hold_at"),
135                perf: get("perf").and_then(|v| match v.as_str() {
136                    "cpu" => Some(false),
137                    "0" => None,
138                    _ => Some(true),
139                }),
140                poll: get("loop").as_deref() == Some("poll"),
141                old_mix: get("mix").as_deref() == Some("old"),
142                touch: get("touch")
143                    .and_then(|v| flint_runtime::TouchControlsMode::parse(&v))
144                    .unwrap_or_default(),
145                fps: get("fps")
146                    .and_then(|v| flint_player::FrameRateSetting::parse(&v))
147                    .unwrap_or_else(|| {
148                        if flint_player::web_coarse_pointer() {
149                            flint_player::FrameRateSetting::Auto
150                        } else {
151                            flint_player::FrameRateSetting::Native
152                        }
153                    }),
154                quality: get("quality").unwrap_or_default(),
155            }
156        }
157    }

159    /// Where play.chaps.dev keeps a game's bundle (`tools/play-site/build.py`
160    /// packs each one there). An id is a bundle name: lowercase letters,
161    /// digits, `-` and `_`, so a query can't point outside `games/`.
162    fn game_bundle(id: &str) -> Option<String> {
163        let ok = !id.is_empty()
164            && id
165                .bytes()
166                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-' || b == b'_');
167        ok.then(|| format!("games/{id}/{id}.flintpack"))
168    }

171        console_error_panic_hook::set_once();
172        tracing_wasm::set_as_global_default_with_config(
173            tracing_wasm::WASMLayerConfigBuilder::new()
174                .set_max_level(tracing::Level::INFO)
175                .build(),
176        );
177        wasm_bindgen_futures::spawn_local(async {
178            if let Err(e) = start().await {
179                status(&format!("couldn't start: {e}"));
180                web_sys::console::error_1(&format!("[flint-web] {e}").into());
181            }
182        });

194    async fn start() -> Result<(), String> {
195        let args = WebArgs::from_query();
196        if let Some((w, h)) = args.size {
197            fix_canvas_size(w, h);
198        }
199
200        let mut late = Vec::new();
201        let loaded = match &args.bundle {
202            Some(url) => {
203                let (loaded, source, parts) = load_bundle(url, args.scene.as_deref()).await?;
204                late.push((source, parts, absolute_url(url)?));
205                loaded
206            }
207            None => load_scene_url(args.scene.as_deref().unwrap_or(DEFAULT_SCENE)).await?,
208        };
209        tracing::info!(
210            "[flint-web] scene {:?}: {} entities",
211            loaded.scene_file.scene.name,
212            loaded.world.entity_count()
213        );
214        // The page may have named the game already (play.chaps.dev's
215        // doorway, from games.json); the scene's own name is the fallback.
216        let title = page_text("gate-title").unwrap_or_else(|| loaded.scene_file.scene.name.clone());
217        let line = page_text("gate-line").or_else(|| loaded.scene_file.scene.description.clone());
218        status(&title);
219
220        let event_loop = EventLoop::new().map_err(|e| format!("event loop: {e}"))?;
221        // Draw once per requestAnimationFrame: `about_to_wait` asks for a
222        // redraw, winit waits for the browser's next frame, and the main
223        // thread sleeps in between. Poll queued a new task as soon as the
224        // last one ended, so the thread never went idle; Chrome still ran
225        // its frames, but Firefox starved its own refresh driver and showed
226        // one frame in five, with stalls of seconds and broken sound (#121,
227        // docs/design/web.md).
228        event_loop.set_control_flow(if args.poll {
229            ControlFlow::Poll
230        } else {
231            ControlFlow::Wait
232        });
233        let mut app = PlayerApp::new(
234            loaded.world,
235            loaded.scene_path,
236            false,
237            None,
238            loaded.scene_file.scene.input_config.clone(),
239        );
240        app.msaa_sample_count = args.msaa;
241        app.render_scale = args.render_scale;
242        app.quality_overrides = args.quality;
243        app.input.touch_controls_mut().set_mode(args.touch);
244        app.clock.frame_step = args.frame_step;
245        app.hold_at = args.hold_at;
246        app.set_frame_rate(args.fps);
247        if let Some(gpu) = args.perf {
248            app.enable_perf(gpu);
249        }
250        app.audio.set_old_mix(args.old_mix);
251        app.apply_scene_settings(&loaded.scene_file);
252        app.set_schema_paths(loaded.schemas);
253        app.set_event_proxy(event_loop.create_proxy());
254        // Returns at once: the browser drives the loop from here on.
255        event_loop.spawn_app(app);
256        open_gate(&title, line.as_deref().unwrap_or(""));
257
258        for (source, parts, base) in late {
259            fetch_late_bundles(source, parts, base, title.clone());
260        }
261        Ok(())
262    }

264    /// Fetch the main bundle, make it the file system, and load its scene.
265    /// Returns the scene, the installed files (late bundles add to them) and
266    /// the late bundles still to fetch.
267    async fn load_bundle(
268        url: &str,
269        scene_override: Option<&str>,
270    ) -> Result<(Loaded, MemorySource, Vec<LateBundle>), String> {
271        let bytes = fetch_bytes(url, |got, total| {
272            progress(url_name(url), got, total);
273        })
274        .await?;
275        let bundle = Bundle::parse(bytes).map_err(|e| format!("{url}: {e}"))?;
276        let manifest = bundle.manifest.clone();
277        tracing::info!(
278            "[flint-web] bundle {}: {} files, {:.2} MB",
279            manifest.name,
280            manifest.files.len(),
281            bundle.len() as f64 / 1e6
282        );
283        let source = bundle.to_memory_source();
284        drop(bundle);
285        vfs::install(Arc::new(source.clone()));
286
287        let scene_path = scene_override
288            .map(str::to_string)
289            .or(manifest.scene.clone())
290            .ok_or_else(|| format!("{url} names no scene; add ?scene="))?;
291
292        // The engine's schemas are compiled in; the bundle's own (its engine
293        // subtree's, then the game's) load over them, as `--schemas` does.
294        let mut registry = engine_registry()?;
295        let mut schemas = manifest.schemas.clone();
296        for dir in flint_schema::discover_schema_dirs(&scene_path) {
297            let dir = vfs::normalize(&dir);
298            if !schemas.contains(&dir) {
299                schemas.push(dir);
300            }
301        }
302        for dir in &schemas {
303            if vfs::is_dir(dir) {
304                registry
305                    .load_directory(dir)
306                    .map_err(|e| format!("schemas {dir}: {e}"))?;
307            }
308        }
309
310        status(&format!("loading {scene_path}"));
311        let (world, scene_file) = flint_scene::load_scene(&scene_path, &registry)
312            .map_err(|e| format!("{scene_path}: {e}"))?;
313        Ok((
314            Loaded {
315                world,
316                scene_file,
317                scene_path,
318                schemas,
319            },
320            source,
321            manifest.late,
322        ))
323    }

325    /// No bundle: one scene TOML by URL, made of the engine's own geometry.
326    async fn load_scene_url(url: &str) -> Result<Loaded, String> {
327        let registry = engine_registry()?;
328        status(&format!("loading {url}"));
329        let text = fetch_text(url).await?;
330        let (world, scene_file) =
331            flint_scene::load_scene_string(&text, &registry).map_err(|e| format!("{url}: {e}"))?;
332        Ok(Loaded {
333            world,
334            scene_file,
335            scene_path: url.to_string(),
336            schemas: Vec::new(),
337        })
338    }

340    /// Fetch the late bundles one after another, each growing the vfs when
341    /// it lands. The player notices (`vfs::generation`) and loads the audio.
342    fn fetch_late_bundles(
343        mut source: MemorySource,
344        parts: Vec<LateBundle>,
345        base: String,
346        title: String,
347    ) {
348        if parts.is_empty() {
349            hide_progress();
350            return;
351        }
352        wasm_bindgen_futures::spawn_local(async move {
353            for part in parts {
354                let url = match web_sys::Url::new_with_base(&part.url, &base) {
355                    Ok(u) => u.href(),
356                    Err(_) => part.url.clone(),
357                };
358                let label = part.part.clone();
359                let result = fetch_bytes(&url, |got, total| {
360                    progress(&label, got, total.or(Some(part.size as usize)));
361                })
362                .await
363                .and_then(|bytes| Bundle::parse(bytes).map_err(|e| e.to_string()));
364                match result {
365                    Ok(bundle) => {
366                        tracing::info!(
367                            "[flint-web] late bundle {}: {} files, {:.2} MB",
368                            bundle.manifest.part,
369                            bundle.manifest.files.len(),
370                            bundle.len() as f64 / 1e6
371                        );
372                        bundle.insert_into(&mut source);
373                        vfs::install(Arc::new(source.clone()));
374                    }
375                    Err(e) => {
376                        web_sys::console::warn_1(&format!("[flint-web] {url}: {e}").into());
377                    }
378                }
379            }
380            status(&title);
381            hide_progress();
382        });
383    }

400    async fn fetch_response(url: &str) -> Result<web_sys::Response, String> {
401        let js = |e: wasm_bindgen::JsValue| format!("{url}: {e:?}");
402        let window = web_sys::window().ok_or("no window")?;
403        let response: web_sys::Response = JsFuture::from(window.fetch_with_str(url))
404            .await
405            .map_err(js)?
406            .dyn_into()
407            .map_err(js)?;
408        if !response.ok() {
409            return Err(format!("{url}: HTTP {}", response.status()));
410        }
411        Ok(response)
412    }

424    /// Fetch a bundle chunk by chunk, reporting `(bytes so far, whole size)`.
425    /// The size comes from the bundle's own header once it has arrived, not
426    /// Content-Length, which is the compressed size when the host gzips.
427    async fn fetch_bytes(
428        url: &str,
429        mut report: impl FnMut(usize, Option<usize>),
430    ) -> Result<Vec<u8>, String> {
431        let js = |e: wasm_bindgen::JsValue| format!("{url}: {e:?}");
432        let response = fetch_response(url).await?;
433        let body = response.body().ok_or_else(|| format!("{url}: no body"))?;
434        let reader: web_sys::ReadableStreamDefaultReader = body.get_reader().unchecked_into();
435        let mut bytes = Vec::new();
436        let mut total = None;
437        report(0, None);
438        loop {
439            let chunk = JsFuture::from(reader.read()).await.map_err(js)?;
440            let done = js_sys::Reflect::get(&chunk, &"done".into())
441                .map(|v| v.is_truthy())
442                .unwrap_or(true);
443            if done {
444                break;
445            }
446            let value: js_sys::Uint8Array = js_sys::Reflect::get(&chunk, &"value".into())
447                .map_err(js)?
448                .unchecked_into();
449            let start = bytes.len();
450            bytes.resize(start + value.length() as usize, 0);
451            value.copy_to(&mut bytes[start..]);
452            if total.is_none() {
453                total = bundle::total_len(&bytes).map_err(|e| format!("{url}: {e}"))?;
454                if let Some(t) = total {
455                    bytes.reserve(t.saturating_sub(bytes.len()));
456                }
457            }
458            report(bytes.len(), total);
459        }
460        Ok(bytes)
461    }

464    fn absolute_url(url: &str) -> Result<String, String> {

466            .and_then(|w| w.location().href().ok())
467            .ok_or("no page URL")?;
468        web_sys::Url::new_with_base(url, &page)
469            .map(|u| u.href())
470            .map_err(|e| format!("{url}: {e:?}"))

500    /// The bar under the wordmark and its line: "float.flintpack 1.8 / 3.7 MB".
501    fn progress(label: &str, got: usize, total: Option<usize>) {
502        let mb = |b: usize| b as f64 / 1e6;
503        let text = match total {
504            Some(t) => format!("{label}  {:.1} / {:.1} MB", mb(got), mb(t)),
505            None => format!("{label}  {:.1} MB", mb(got)),
506        };
507        status(&text);
508        if let Some(bar) = element("progress") {
509            let _ = bar.set_attribute("data-active", "");
510            let fraction = total.map(|t| got as f64 / t.max(1) as f64).unwrap_or(0.0);
511            let _ = bar.set_attribute("style", &format!("--p: {:.4}", fraction.min(1.0)));
512        }
513    }

515    /// The Begin gate (#102). The world is already running behind it,
516    /// silent. Its click is the gesture a browser asks for: it locks the
517    /// pointer and starts the sound (`flint_player::web_begin`), and the
518    /// veil lifts. Afterwards a line under the world says how to take the
519    /// pointer back whenever Escape has freed it.
520    fn open_gate(title: &str, line: &str) {
521        use std::cell::Cell;
522        use std::rc::Rc;
523        use wasm_bindgen::closure::Closure;
524
525        let (Some(gate), Some(button)) = (element("gate"), element("begin")) else {
526            flint_player::web_begin();
527            return;
528        };
529        if let Some(el) = element("gate-title") {
530            el.set_text_content(Some(title));
531        }
532        if let Some(el) = element("gate-line") {
533            el.set_text_content(Some(line));
534        }
535        let _ = gate.set_attribute("data-ready", "");
536        if let Ok(button) = button.clone().dyn_into::<web_sys::HtmlElement>() {
537            let _ = button.focus();
538        }
539
540        let begun = Rc::new(Cell::new(false));
541        let on_begin = {
542            let begun = begun.clone();
543            Closure::<dyn FnMut()>::new(move || {
544                if begun.replace(true) {
545                    return;
546                }
547                flint_player::web_begin();
548                let _ = gate.set_attribute("data-gone", "");
549                if let Some(canvas) = element(flint_player::WEB_CANVAS_ID)
550                    .and_then(|c| c.dyn_into::<web_sys::HtmlElement>().ok())
551                {
552                    let _ = canvas.focus();
553                }
554            })
555        };
556        let _ = button.add_event_listener_with_callback("click", on_begin.as_ref().unchecked_ref());
557        on_begin.forget();
558
559        let Some(document) = web_sys::window().and_then(|w| w.document()) else {
560            return;
561        };
562        let on_lock = {
563            let document = document.clone();
564            Closure::<dyn FnMut()>::new(move || {
565                let Some(hint) = element("relook") else {
566                    return;
567                };
568                if document.pointer_lock_element().is_some() || !begun.get() {
569                    let _ = hint.remove_attribute("data-on");
570                } else {
571                    let _ = hint.set_attribute("data-on", "");
572                }
573            })
574        };
575        let _ = document.add_event_listener_with_callback(
576            "pointerlockchange",
577            on_lock.as_ref().unchecked_ref(),
578        );
579        on_lock.forget();
580    }

582    fn hide_progress() {

584            let _ = bar.remove_attribute("data-active");
585        }
