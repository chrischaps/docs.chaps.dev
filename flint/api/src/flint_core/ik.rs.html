1//! Analytic two-bone inverse kinematics (ADR 0070).
2//!
3//! The chain is root → mid → tip with fixed segment lengths `a` (root→mid) and
4//! `b` (mid→tip). Given a target for the tip and a pole point that says which
5//! side the mid joint bends toward, the solution is closed-form: the law of
6//! cosines fixes the angle at the root, and the bend plane is spanned by the
7//! root→target line and the pole. Pure geometry, no world or entities; the
8//! animation crate's `ik_pass` turns the result into node rotations.
9
10use crate::Vec3;
11
12/// Joint positions produced by [`solve_two_bone`].
13#[derive(Debug, Clone, Copy, PartialEq)]
14pub struct TwoBoneSolution {
15    /// The mid joint (elbow / knee).
16    pub mid: Vec3,
17    /// Where the tip lands: the target when reachable, else the closest point
18    /// on the root→target line with the chain fully extended (or fully folded).
19    pub tip: Vec3,
20}
21
22/// Solve a two-bone chain. Returns `None` when the target sits on the root or
23/// the pole is colinear with the root→target line, so the bend plane is
24/// undefined; callers keep the previous pose in that case.
25pub fn solve_two_bone(
26    root: Vec3,
27    target: Vec3,
28    pole: Vec3,
29    upper_len: f32,
30    fore_len: f32,
31) -> Option<TwoBoneSolution> {
32    const EPS: f32 = 1e-3;
33    let a = upper_len;
34    let b = fore_len;
35    if a <= 0.0 || b <= 0.0 {
36        return None;
37    }
38    let to_target = target - root;
39    let d = to_target.length();
40    if d < 1e-6 {
41        return None;
42    }
43    let t_hat = to_target * (1.0 / d);
44
45    // Keep a hair inside the reachable annulus so the mid direction stays well
46    // defined at full extension and full fold.
47    let d_c = d.clamp((a - b).abs() * (1.0 + EPS), (a + b) * (1.0 - EPS));
48
49    let to_pole = pole - root;
50    let n = t_hat.cross(&to_pole);
51    if n.length() < 1e-6 {
52        return None;
53    }
54    let n = n.normalized();
55    // In-plane direction perpendicular to the target line, on the pole's side.
56    let u = n.cross(&t_hat);
57
58    let cos_alpha = ((a * a + d_c * d_c - b * b) / (2.0 * a * d_c)).clamp(-1.0, 1.0);
59    let alpha = cos_alpha.acos();
60    let mid = root + t_hat * (a * alpha.cos()) + u * (a * alpha.sin());
61    let tip = root + t_hat * d_c;
62    Some(TwoBoneSolution { mid, tip })
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68
69    fn close(a: f32, b: f32) -> bool {
70        (a - b).abs() < 1e-4
71    }
72
73    fn dist(a: Vec3, b: Vec3) -> f32 {
74        (a - b).length()
75    }
76
77    #[test]
78    fn reachable_target_lands_exactly_on_the_pole_side() {
79        let root = Vec3::new(0.0, 1.0, 0.0);
80        let target = Vec3::new(0.3, 0.9, -0.2);
81        let pole = Vec3::new(0.5, 0.0, 0.0);
82        let s = solve_two_bone(root, target, pole, 0.25, 0.2).unwrap();
83        assert!(dist(s.tip, target) < 1e-5, "{:?}", s.tip);
84        assert!(close(dist(root, s.mid), 0.25));
85        assert!(close(dist(s.mid, s.tip), 0.2));
86        // The mid joint sits on the pole's side of the root→target line.
87        let t_hat = (target - root).normalized();
88        let perp = (s.mid - root) - t_hat * (s.mid - root).dot(&t_hat);
89        let pole_perp = (pole - root) - t_hat * (pole - root).dot(&t_hat);
90        assert!(perp.dot(&pole_perp) > 0.0);
91    }
92
93    #[test]
94    fn flipping_the_pole_flips_the_bend() {
95        let root = Vec3::ZERO;
96        let target = Vec3::new(0.3, 0.0, 0.0);
97        let up = solve_two_bone(root, target, Vec3::new(0.0, 1.0, 0.0), 0.25, 0.25).unwrap();
98        let down = solve_two_bone(root, target, Vec3::new(0.0, -1.0, 0.0), 0.25, 0.25).unwrap();
99        assert!(up.mid.y > 0.1 && down.mid.y < -0.1, "{:?} {:?}", up.mid, down.mid);
100        assert!(close(up.mid.y, -down.mid.y));
101    }
102
103    #[test]
104    fn overreach_straightens_toward_the_target() {
105        let root = Vec3::ZERO;
106        let target = Vec3::new(2.0, 0.0, 0.0);
107        let s = solve_two_bone(root, target, Vec3::new(0.0, 1.0, 0.0), 0.25, 0.25).unwrap();
108        assert!(s.tip.x < 0.5 && s.tip.x > 0.49, "{:?}", s.tip);
109        assert!(close(s.tip.y, 0.0) && close(s.tip.z, 0.0));
110        assert!(s.mid.y.abs() < 0.02 && close(dist(root, s.mid), 0.25));
111    }
112
113    #[test]
114    fn degenerate_inputs_return_none() {
115        let root = Vec3::ZERO;
116        let target = Vec3::new(0.3, 0.0, 0.0);
117        // Pole on the target line: no bend plane.
118        assert!(solve_two_bone(root, target, Vec3::new(1.0, 0.0, 0.0), 0.25, 0.25).is_none());
119        // Target on the root.
120        assert!(solve_two_bone(root, root, Vec3::new(0.0, 1.0, 0.0), 0.25, 0.25).is_none());
121        assert!(solve_two_bone(root, target, Vec3::new(0.0, 1.0, 0.0), 0.0, 0.25).is_none());
122    }
