1//! A read-only file layer for the runtime.
2//!
3//! Every file the engine reads while a game runs (scenes, schemas, scripts,
4//! models, textures, audio, charts) goes through these functions instead of
5//! `std::fs`. On desktop the source is [`DiskSource`] and nothing changes:
6//! each call is the `std::fs` call it replaced. A browser has no file system,
7//! so the web build installs a [`MemorySource`] filled from a fetched bundle
8//! before the player starts, and the same loaders read from it.
9//!
10//! ```
11//! use flint_core::vfs::{self, MemorySource};
12//! use std::sync::Arc;
13//!
14//! let mut bundle = MemorySource::new();
15//! bundle.insert("levels/demo.scene.toml", "[scene]\nname = \"demo\"\n");
16//! vfs::scoped(Arc::new(bundle), || {
17//!     assert!(vfs::is_dir("levels"));
18//!     assert!(vfs::read_to_string("./levels/demo.scene.toml").unwrap().contains("demo"));
19//! });
20//! ```
21//!
22//! **Paths** are the ones the runtime already uses: relative to the working
23//! directory (the game root), with `..` for prefab and schema references and
24//! either slash on Windows. [`MemorySource`] keys them by [`normalize`]:
25//! forward slashes, `.` dropped and `..` folded, so `levels/../models\a.glb`
26//! and `models/a.glb` are the same file. [`read_dir`] returns each entry as
27//! the directory path the caller passed joined with the entry's name, the way
28//! `std::fs::read_dir(..).path()` does.
29//!
30//! **Writes are out of scope.** Scene saving, save
31//! games and replay recordings still write with `std::fs`, natively only.
32//! Input-config overrides go through `flint_player::user_files` (#100).
33//! Hot reload also stays on the disk: it polls `std::fs::metadata` for
34//! modification times, which a memory source has none of, so on the web it
35//! simply never fires.
36//!
37//! **Which source.** [`install`] sets the process-wide source (the web
38//! player, once, at startup). [`scoped`] overrides it for the current thread
39//! while a closure runs, which is what tests use so they don't leak a bundle
40//! into other tests running in parallel.
41//!
42//! **Android** (`flint-android/src/asset_extractor.rs`) copies the APK's
43//! assets into internal storage at first launch and then reads them from
44//! disk. It could instead install a source backed by `AAssetManager` here and
45//! skip the copy; nothing in this module stops that, but it is not done yet.
46

54/// What a path names, if anything.
55#[derive(Debug, Clone, Copy, PartialEq, Eq)]
56pub enum EntryKind {
57    File,
58    Dir,
59}
60
61/// Somewhere files can be read from.
62pub trait FileSource: Send + Sync {
63    /// The whole file.
64    fn read(&self, path: &Path) -> io::Result<Vec<u8>>;
65
66    /// The entries directly inside a directory, each as `path.join(name)`.
67    fn read_dir(&self, path: &Path) -> io::Result<Vec<PathBuf>>;
68
69    /// Whether `path` is a file, a directory or nothing.
70    fn kind(&self, path: &Path) -> Option<EntryKind>;
71}
72
73/// The local disk, through `std::fs`. The default source.
74#[derive(Debug, Default, Clone, Copy)]
75pub struct DiskSource;
76
77// The one place the runtime may touch `std::fs` for reading.
78#[allow(clippy::disallowed_methods)]
79impl FileSource for DiskSource {
80    fn read(&self, path: &Path) -> io::Result<Vec<u8>> {
81        std::fs::read(path)
82    }
83
84    fn read_dir(&self, path: &Path) -> io::Result<Vec<PathBuf>> {
85        // An entry that can't be read is skipped, as the loops this replaced
86        // (`entries.flatten()`) did.
87        Ok(std::fs::read_dir(path)?
88            .filter_map(|entry| entry.ok().map(|e| e.path()))
89            .collect())
90    }
91
92    fn kind(&self, path: &Path) -> Option<EntryKind> {
93        let meta = std::fs::metadata(path).ok()?;
94        Some(if meta.is_dir() {
95            EntryKind::Dir
96        } else {
97            EntryKind::File
98        })
99    }
100}

102/// Files held in memory, keyed by [`normalize`]d path. Directories exist
103/// wherever a file's path passes through them.
104#[derive(Debug, Default, Clone)]
105pub struct MemorySource {
106    files: BTreeMap<String, Arc<[u8]>>,
107}
108
109impl MemorySource {
110    pub fn new() -> Self {
111        Self::default()
112    }
113
114    /// Add (or replace) a file.
115    pub fn insert(&mut self, path: impl AsRef<Path>, bytes: impl Into<Vec<u8>>) {
116        self.files
117            .insert(normalize(path.as_ref()), Arc::from(bytes.into()));
118    }
119
120    /// How many files it holds.
121    pub fn len(&self) -> usize {
122        self.files.len()
123    }
124
125    pub fn is_empty(&self) -> bool {
126        self.files.is_empty()
127    }
128
129    /// Every file's normalized path, in order.
130    pub fn paths(&self) -> impl Iterator<Item = &str> {
131        self.files.keys().map(String::as_str)
132    }
133
134    /// The files under `dir/`, as `(name after the prefix, bytes)`.
135    fn under<'a>(&'a self, dir: &str) -> impl Iterator<Item = (&'a str, &'a Arc<[u8]>)> + 'a {
136        let prefix = if dir.is_empty() {
137            String::new()
138        } else {
139            format!("{dir}/")
140        };
141        let skip = prefix.len();
142        self.files
143            .range(prefix.clone()..)
144            .take_while(move |(k, _)| k.starts_with(&prefix))
145            .map(move |(k, v)| (&k[skip..], v))
146            .filter(|(rest, _)| !rest.is_empty())
147    }
148}
149
150impl<P: AsRef<Path>, B: Into<Vec<u8>>> FromIterator<(P, B)> for MemorySource {
151    fn from_iter<I: IntoIterator<Item = (P, B)>>(iter: I) -> Self {
152        let mut source = Self::new();
153        for (path, bytes) in iter {
154            source.insert(path, bytes);
155        }
156        source
157    }
158}

160impl FileSource for MemorySource {
161    fn read(&self, path: &Path) -> io::Result<Vec<u8>> {
162        self.files
163            .get(&normalize(path))
164            .map(|bytes| bytes.to_vec())
165            .ok_or_else(|| not_found(path))
166    }
167
168    fn read_dir(&self, path: &Path) -> io::Result<Vec<PathBuf>> {
169        let dir = normalize(path);
170        if self.files.contains_key(&dir) {
171            return Err(io::Error::other(format!(
172                "{} is a file, not a directory",
173                path.display()
174            )));
175        }
176        let mut names: Vec<&str> = self
177            .under(&dir)
178            .map(|(rest, _)| rest.split('/').next().unwrap_or(rest))
179            .collect();
180        names.dedup();
181        if names.is_empty() {
182            return Err(not_found(path));
183        }
184        Ok(names.into_iter().map(|name| path.join(name)).collect())
185    }
186
187    fn kind(&self, path: &Path) -> Option<EntryKind> {
188        let key = normalize(path);
189        if self.files.contains_key(&key) {
190            Some(EntryKind::File)
191        } else if self.under(&key).next().is_some() {
192            Some(EntryKind::Dir)
193        } else {
194            None
195        }
196    }
197}
198
199fn not_found(path: &Path) -> io::Error {
200    io::Error::new(
201        io::ErrorKind::NotFound,
202        format!("{} is not in the file source", path.display()),
203    )
204}

206/// The key a [`MemorySource`] files a path under: forward slashes, no `.`
207/// components, each `..` folded into the component before it (a leading `..`
208/// that has nothing to fold stays), no trailing slash. An absolute path keeps
209/// its leading `/` (or drive prefix). The empty string is the root.
210pub fn normalize(path: &Path) -> String {
211    let text = path.to_string_lossy().replace('\\', "/");
212    let absolute = text.starts_with('/');
213    let mut parts: Vec<&str> = Vec::new();
214    for part in text.split('/') {
215        match part {
216            "" | "." => {}
217            ".." => match parts.last() {
218                Some(&last) if last != ".." => {
219                    parts.pop();
220                }
221                _ if absolute => {}
222                _ => parts.push(".."),
223            },
224            other => parts.push(other),
225        }
226    }
227    let joined = parts.join("/");
228    if absolute {
229        format!("/{joined}")
230    } else {
231        joined
232    }
233}

237fn global() -> &'static RwLock<Arc<dyn FileSource>> {
238    static GLOBAL: OnceLock<RwLock<Arc<dyn FileSource>>> = OnceLock::new();
239    GLOBAL.get_or_init(|| RwLock::new(Arc::new(DiskSource)))
240}

242thread_local! {
243    static SCOPED: RefCell<Vec<Arc<dyn FileSource>>> = const { RefCell::new(Vec::new()) };
244}

246/// Make `source` the process-wide source, replacing the disk (or whatever was
247/// installed before).
248pub fn install(source: Arc<dyn FileSource>) {
249    *global().write().unwrap_or_else(|e| e.into_inner()) = source;
250    GENERATION.fetch_add(1, Ordering::Release);
251}
252
253static GENERATION: AtomicU64 = AtomicU64::new(0);
254
255/// How many times [`install`] has run. It changes whenever the process-wide
256/// files may have: the web player installs a grown source when a late bundle
257/// (audio) arrives, and the player then loads what it couldn't find before.
258pub fn generation() -> u64 {
259    GENERATION.load(Ordering::Acquire)
260}

262/// Run `f` with `source` as this thread's source, then restore the previous
263/// one (even if `f` panics). Other threads are unaffected.
264pub fn scoped<R>(source: Arc<dyn FileSource>, f: impl FnOnce() -> R) -> R {
265    struct Pop;
266    impl Drop for Pop {
267        fn drop(&mut self) {
268            SCOPED.with(|s| s.borrow_mut().pop());
269        }
270    }
271    SCOPED.with(|s| s.borrow_mut().push(source));
272    let _pop = Pop;
273    f()
274}
275
276/// The source reads go to right now on this thread.
277pub fn current() -> Arc<dyn FileSource> {
278    SCOPED
279        .with(|s| s.borrow().last().cloned())
280        .unwrap_or_else(|| global().read().unwrap_or_else(|e| e.into_inner()).clone())
281}

285/// Read a whole file (`std::fs::read`).
286pub fn read(path: impl AsRef<Path>) -> io::Result<Vec<u8>> {
287    current().read(path.as_ref())
288}
289
290/// Read a whole file as UTF-8 (`std::fs::read_to_string`).
291pub fn read_to_string(path: impl AsRef<Path>) -> io::Result<String> {
292    let path = path.as_ref();
293    String::from_utf8(read(path)?).map_err(|_| {
294        io::Error::new(
295            io::ErrorKind::InvalidData,
296            format!("{} is not valid UTF-8", path.display()),
297        )
298    })
299}

301/// The entries directly inside a directory, each as `dir.join(name)`
302/// (`std::fs::read_dir` and `DirEntry::path`). The disk lists them in its own
303/// order, a memory source alphabetically; sort if order matters.
304pub fn read_dir(dir: impl AsRef<Path>) -> io::Result<Vec<PathBuf>> {
305    current().read_dir(dir.as_ref())
306}
307
308/// Whether anything is at `path` (`Path::exists`).
309pub fn exists(path: impl AsRef<Path>) -> bool {
310    current().kind(path.as_ref()).is_some()
311}
312
313/// Whether `path` is a file (`Path::is_file`).
314pub fn is_file(path: impl AsRef<Path>) -> bool {
315    current().kind(path.as_ref()) == Some(EntryKind::File)
316}
317
318/// Whether `path` is a directory (`Path::is_dir`).
319pub fn is_dir(path: impl AsRef<Path>) -> bool {
320    current().kind(path.as_ref()) == Some(EntryKind::Dir)
321}

343        let n = |s: &str| normalize(Path::new(s));
344        assert_eq!(n("levels/demo.scene.toml"), "levels/demo.scene.toml");
345        assert_eq!(n("./levels//demo.scene.toml"), "levels/demo.scene.toml");
346        assert_eq!(

351        assert_eq!(n("../.."), "../..");
