1//! Text written in the world, drawn with a depth test (ADR 0104).
2//!
3//! A `WorldText` is a line of text at a world point. Scripts queue them with
4//! `draw_world_text`; the player lays them out in the HUD's fonts and draws
5//! them over the finished image, before the HUD, hidden wherever the scene's
6//! depth is nearer. A shop's name on its sign stays behind the roof in front
7//! of it, as `draw_text_opts` at a projected point can't.
8//!
9//! Without a plane the text is a billboard: it faces the screen at the size
10//! asked, in logical points, like HUD text, and takes the point's depth.
11//! On a [`TextPlane`] it lies in that plane, as if painted there, and every
12//! length (its size, letter spacing, stroke and shadow) is in world units.
13
14/// One depth-tested line of text in the world.
15#[derive(Debug, Clone, PartialEq)]
16pub struct WorldText {
17    /// The anchor, in world units. `align` and `valign` place the text
18    /// round it.
19    pub at: [f32; 3],
20    pub text: String,
21    /// The font size: logical points for a billboard, world units (the em)
22    /// on a plane.
23    pub size: f32,
24    /// Display-space (sRGB) RGBA, the same space as the HUD's colours.
25    pub color: [f32; 4],
26    /// Horizontal placement about `at`: 0 left edge, 1 centre, 2 right edge.
27    pub align: u8,
28    /// Vertical placement about `at`: 0 top, 1 middle, 2 bottom.
29    pub valign: u8,
30    /// Outline colour and width, as `draw_text_opts`'s.
31    pub stroke: Option<([f32; 4], f32)>,
32    /// A family registered from the game's `fonts/`, or the default.
33    pub font: Option<String>,
34    /// Extra space between letters.
35    pub spacing: f32,
36    /// Shadow colour and its offset (right, down).
37    pub shadow: Option<([f32; 4], f32, f32)>,
38    /// The plane it's written on, or `None` for a billboard.
39    pub plane: Option<TextPlane>,
40}
41
42/// The plane a [`WorldText`] lies in.
43#[derive(Debug, Clone, Copy, PartialEq)]
44pub struct TextPlane {
45    /// The way the written face looks. The text reads from this side and
46    /// isn't drawn from behind, like paint on one face of a board.
47    pub normal: [f32; 3],
48    /// Which way is up for the letters. Only its part across the plane
49    /// counts, so world up works for any wall or sign.
50    pub up: [f32; 3],
51}
52
53impl TextPlane {
54    /// The letters' right and up directions, unit length, in the plane:
55    /// right is the reader's right, facing the written face. `None` if the
56    /// normal is zero or `up` lies along it.
57    pub fn basis(&self) -> Option<([f32; 3], [f32; 3])> {
58        let n = normalize(self.normal)?;
59        let right = normalize(cross(self.up, n))?;
60        Some((right, cross(n, right)))
61    }
62
63    /// Whether `eye` is on the written side of the plane through `at`.
64    pub fn faces(&self, at: [f32; 3], eye: [f32; 3]) -> bool {
65        let to_eye = [eye[0] - at[0], eye[1] - at[1], eye[2] - at[2]];
66        dot(to_eye, self.normal) > 0.0
67    }
68}
69
70fn dot(a: [f32; 3], b: [f32; 3]) -> f32 {
71    a[0] * b[0] + a[1] * b[1] + a[2] * b[2]
72}
73
74fn cross(a: [f32; 3], b: [f32; 3]) -> [f32; 3] {
75    [
76        a[1] * b[2] - a[2] * b[1],
77        a[2] * b[0] - a[0] * b[2],
78        a[0] * b[1] - a[1] * b[0],
79    ]
80}
81
82fn normalize(v: [f32; 3]) -> Option<[f32; 3]> {
83    let len = dot(v, v).sqrt();
84    (len > 1e-6).then(|| [v[0] / len, v[1] / len, v[2] / len])
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90
91    fn close(a: [f32; 3], b: [f32; 3]) -> bool {
92        (0..3).all(|i| (a[i] - b[i]).abs() < 1e-5)
93    }
94
95    #[test]
96    fn a_wall_facing_south_reads_east_and_up() {
97        // Facing +z, a reader stands at +z looking toward -z: their right
98        // is +x.
99        let plane = TextPlane { normal: [0.0, 0.0, 1.0], up: [0.0, 1.0, 0.0] };
100        let (right, up) = plane.basis().unwrap();
101        assert!(close(right, [1.0, 0.0, 0.0]), "{right:?}");
102        assert!(close(up, [0.0, 1.0, 0.0]), "{up:?}");
103    }
104
105    #[test]
106    fn up_is_squared_to_a_leaning_plane() {
107        let plane = TextPlane { normal: [0.0, 1.0, 1.0], up: [0.0, 1.0, 0.0] };
108        let (right, up) = plane.basis().unwrap();
109        assert!(close(right, [1.0, 0.0, 0.0]), "{right:?}");
110        let h = std::f32::consts::FRAC_1_SQRT_2;
111        assert!(close(up, [0.0, h, -h]), "{up:?}");
112    }
113
114    #[test]
115    fn up_along_the_normal_has_no_basis() {
116        let flat = TextPlane { normal: [0.0, 1.0, 0.0], up: [0.0, 1.0, 0.0] };
117        assert!(flat.basis().is_none());
118        let floor = TextPlane { normal: [0.0, 1.0, 0.0], up: [0.0, 0.0, -1.0] };
119        let (right, up) = floor.basis().unwrap();
120        assert!(close(right, [1.0, 0.0, 0.0]), "{right:?}");
121        assert!(close(up, [0.0, 0.0, -1.0]), "{up:?}");
122    }
123
124    #[test]
125    fn it_faces_only_the_side_its_normal_points_to() {
126        let plane = TextPlane { normal: [0.0, 0.0, 1.0], up: [0.0, 1.0, 0.0] };
127        assert!(plane.faces([0.0, 0.0, 0.0], [3.0, 1.0, 5.0]));
128        assert!(!plane.faces([0.0, 0.0, 0.0], [3.0, 1.0, -5.0]));
129        assert!(!plane.faces([0.0, 0.0, 0.0], [3.0, 1.0, 0.0]));
130    }
