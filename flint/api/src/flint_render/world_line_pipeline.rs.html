1//! Depth-tested world-space lines (ADR 0083).
2//!
3//! Scripts queue `WorldLine`s with `draw_world_line`. Each frame the renderer
4//! projects them with the camera it renders with and expands every segment
5//! into a screen-space quad of the requested width. It then draws them onto
6//! the finished, tonemapped image, before the HUD. The scene's resolved depth
7//! hides whatever lies behind geometry. The lines don't write depth, so they
8//! never hide each other or anything drawn later.
9//!
10//! The width is in logical points, as the HUD's lines are, and the edges are
11//! anti-aliased in the fragment shader, since the pass runs after MSAA and
12//! FXAA. Colours are display-space (sRGB), as the HUD's are, so a ring keeps
13//! the colour it had as a HUD line. Fog, bloom and tonemapping don't touch it.
14
15use bytemuck::{Pod, Zeroable};
16use flint_core::world_line::WorldLine;
17
18/// How far each end is pulled toward the eye, as a fraction of its distance.
19/// A line laid on a surface would otherwise fight that surface for depth. At
20/// 100 m this is 20 cm, which keeps a ring on the ground visible and still
21/// hides it behind anything that really stands in front.
22pub const DEPTH_SLACK: f32 = 0.002;
23
24/// Anti-aliasing feather, in pixels, added to each side of a line.
25const FEATHER_PX: f32 = 1.0;
26
27/// One vertex of a line's quad. Layout must match `VsIn` in
28/// `world_line_shader.wgsl`.
29#[repr(C)]
30#[derive(Copy, Clone, Debug, Pod, Zeroable, PartialEq)]
31pub struct WorldLineVertex {
32    /// NDC position with `w = 1`. NDC depth is affine across the screen, so
33    /// the quad interpolates the segment's true depth.
34    pub position: [f32; 4],
35    /// RGBA in the target's space (linear for an sRGB target).
36    pub color: [f32; 4],
37    /// `x`: signed distance from the centre line in pixels; `y`: half the
38    /// line's width in pixels.
39    pub edge: [f32; 2],
40}
41
42impl WorldLineVertex {
43    const ATTRIBUTES: [wgpu::VertexAttribute; 3] =
44        wgpu::vertex_attr_array![0 => Float32x4, 1 => Float32x4, 2 => Float32x2];
45
46    fn desc() -> wgpu::VertexBufferLayout<'static> {
47        wgpu::VertexBufferLayout {
48            array_stride: std::mem::size_of::<Self>() as wgpu::BufferAddress,
49            step_mode: wgpu::VertexStepMode::Vertex,
50            attributes: &Self::ATTRIBUTES,
51        }
52    }
53}

55/// The camera and target a frame's lines are built for.
56#[derive(Debug, Clone, Copy)]
57pub struct LineView {
58    /// Column-major view-projection (`m[col][row]`), wgpu clip space.
59    pub view_proj: [[f32; 4]; 4],
60    /// Camera position in world space.
61    pub eye: [f32; 3],
62    /// Target size in physical pixels.
63    pub viewport: [f32; 2],
64    /// Physical pixels per logical point.
65    pub px_per_point: f32,
66    /// Convert display-space colours to linear (an sRGB target re-encodes).
67    pub linear_output: bool,
68}
69
70/// Expand `lines` into a triangle list, six vertices a segment, appended to
71/// `out`. A segment wholly behind the near plane, or shorter than a
72/// hundredth of a pixel on screen, adds nothing. One crossing the near plane
73/// is cut there.
74pub fn build_vertices(lines: &[WorldLine], view: &LineView, out: &mut Vec<WorldLineVertex>) {
75    let [w, h] = view.viewport;
76    if w <= 0.0 || h <= 0.0 {
77        return;
78    }
79    for line in lines {
80        let half = line.thickness.max(0.0) * view.px_per_point * 0.5;
81        if half <= 0.0 || line.color[3] <= 0.0 {
82            continue;
83        }
84        let Some((p, q)) = clip_segment(
85            project(&view.view_proj, toward(line.a, view.eye)),
86            project(&view.view_proj, toward(line.b, view.eye)),
87        ) else {
88            continue;
89        };
90        let (pn, qn) = (ndc(p), ndc(q));
91        // Pixel space, y up (NDC's direction; only lengths matter here).
92        let pa = [(pn[0] * 0.5 + 0.5) * w, (pn[1] * 0.5 + 0.5) * h];
93        let pb = [(qn[0] * 0.5 + 0.5) * w, (qn[1] * 0.5 + 0.5) * h];
94        let d = [pb[0] - pa[0], pb[1] - pa[1]];
95        let len = (d[0] * d[0] + d[1] * d[1]).sqrt();
96        if len < 0.01 {
97            continue;
98        }
99        let u = [d[0] / len, d[1] / len];
100        let n = [-u[1], u[0]];
101        let e = half + FEATHER_PX;
102        // Butt caps, half a pixel long, so a polyline's joints meet.
103        let cap = 0.5;
104        let to_ndc = [2.0 / w, 2.0 / h];
105        let corner = |at: [f32; 3], along: f32, side: f32| -> [f32; 4] {
106            [
107                at[0] + (u[0] * along + n[0] * side) * to_ndc[0],
108                at[1] + (u[1] * along + n[1] * side) * to_ndc[1],
109                at[2],
110                1.0,
111            ]
112        };
113        let color = if view.linear_output {
114            [
115                srgb_to_linear(line.color[0]),
116                srgb_to_linear(line.color[1]),
117                srgb_to_linear(line.color[2]),
118                line.color[3],
119            ]
120        } else {
121            line.color
122        };
123        let v = |position: [f32; 4], side: f32| WorldLineVertex {
124            position,
125            color,
126            edge: [side, half],
127        };
128        let a0 = v(corner(pn, -cap, -e), -e);
129        let a1 = v(corner(pn, -cap, e), e);
130        let b0 = v(corner(qn, cap, -e), -e);
131        let b1 = v(corner(qn, cap, e), e);
132        out.extend_from_slice(&[a0, b0, b1, a0, b1, a1]);
133    }
134}

136/// `p` moved toward `eye` by `DEPTH_SLACK` of the distance between them.
137fn toward(p: [f32; 3], eye: [f32; 3]) -> [f32; 3] {
138    [
139        p[0] + (eye[0] - p[0]) * DEPTH_SLACK,
140        p[1] + (eye[1] - p[1]) * DEPTH_SLACK,
141        p[2] + (eye[2] - p[2]) * DEPTH_SLACK,
142    ]
143}
144
145/// World point to clip space with a column-major view-projection.
146fn project(m: &[[f32; 4]; 4], p: [f32; 3]) -> [f32; 4] {
147    let mut out = [0.0f32; 4];
148    for (row, o) in out.iter_mut().enumerate() {
149        *o = m[0][row] * p[0] + m[1][row] * p[1] + m[2][row] * p[2] + m[3][row];
150    }
151    out
152}
153
154/// Cut a clip-space segment at the near plane (`z = 0` in wgpu clip space,
155/// for perspective and orthographic cameras alike).
156fn clip_segment(p: [f32; 4], q: [f32; 4]) -> Option<([f32; 4], [f32; 4])> {
157    match (p[2] >= 0.0, q[2] >= 0.0) {
158        (true, true) => Some((p, q)),
159        (false, false) => None,
160        (p_in, _) => {
161            let t = p[2] / (p[2] - q[2]);
162            let mut m = [0.0f32; 4];
163            for i in 0..4 {
164                m[i] = p[i] + (q[i] - p[i]) * t;
165            }
166            m[2] = 0.0;
167            if m[3] <= 0.0 {
168                return None;
169            }
170            Some(if p_in { (p, m) } else { (m, q) })
171        }
172    }
173}
174
175fn ndc(c: [f32; 4]) -> [f32; 3] {
176    [c[0] / c[3], c[1] / c[3], c[2] / c[3]]
177}
178
179fn srgb_to_linear(c: f32) -> f32 {
180    if c <= 0.04045 {
181        c / 12.92
182    } else {
183        ((c + 0.055) / 1.055).powf(2.4)
184    }
185}

187/// The pipeline and a vertex buffer that grows to the busiest frame.
188pub struct WorldLinePipeline {
189    pipeline: wgpu::RenderPipeline,
190    buffer: Option<wgpu::Buffer>,
191    capacity: usize,
192    vertices: Vec<WorldLineVertex>,
193}
194
195impl WorldLinePipeline {
196    /// `format` is the final target's (the surface's); the pass is single
197    /// sample against a `Depth32Float` depth view.
198    pub fn new(device: &wgpu::Device, format: wgpu::TextureFormat) -> Self {
199        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
200            label: Some("World Line Shader"),
201            source: wgpu::ShaderSource::Wgsl(include_str!("world_line_shader.wgsl").into()),
202        });
203        let layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
204            label: Some("World Line Pipeline Layout"),
205            bind_group_layouts: &[],
206            push_constant_ranges: &[],
207        });
208        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
209            label: Some("World Line Pipeline"),
210            layout: Some(&layout),
211            vertex: wgpu::VertexState {
212                module: &shader,
213                entry_point: Some("vs_world_line"),
214                buffers: &[WorldLineVertex::desc()],
215                compilation_options: Default::default(),
216            },
217            fragment: Some(wgpu::FragmentState {
218                module: &shader,
219                entry_point: Some("fs_world_line"),
220                targets: &[Some(wgpu::ColorTargetState {
221                    format,
222                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
223                    write_mask: wgpu::ColorWrites::ALL,
224                })],
225                compilation_options: Default::default(),
226            }),
227            primitive: wgpu::PrimitiveState {
228                topology: wgpu::PrimitiveTopology::TriangleList,
229                cull_mode: None,
230                ..Default::default()
231            },
232            depth_stencil: Some(wgpu::DepthStencilState {
233                format: wgpu::TextureFormat::Depth32Float,
234                depth_write_enabled: false,
235                depth_compare: wgpu::CompareFunction::LessEqual,
236                stencil: wgpu::StencilState::default(),
237                bias: wgpu::DepthBiasState::default(),
238            }),
239            multisample: wgpu::MultisampleState::default(),
240            multiview: None,
241            cache: None,
242        });
243        Self {
244            pipeline,
245            buffer: None,
246            capacity: 0,
247            vertices: Vec::new(),
248        }
249    }
250
251    /// Build and draw `lines` onto `target`, tested against `depth`. Records
252    /// nothing when no segment survives projection.
253    #[allow(clippy::too_many_arguments)]
254    pub fn draw(
255        &mut self,
256        device: &wgpu::Device,
257        queue: &wgpu::Queue,
258        encoder: &mut wgpu::CommandEncoder,
259        target: &wgpu::TextureView,
260        depth: &wgpu::TextureView,
261        lines: &[WorldLine],
262        view: &LineView,
263    ) {
264        self.vertices.clear();
265        build_vertices(lines, view, &mut self.vertices);
266        if self.vertices.is_empty() {
267            return;
268        }
269        if self.capacity < self.vertices.len() {
270            let capacity = self.vertices.len().next_power_of_two();
271            self.buffer = Some(device.create_buffer(&wgpu::BufferDescriptor {
272                label: Some("World Line Vertices"),
273                size: (capacity * std::mem::size_of::<WorldLineVertex>()) as u64,
274                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
275                mapped_at_creation: false,
276            }));
277            self.capacity = capacity;
278        }
279        let buffer = self.buffer.as_ref().expect("allocated above");
280        queue.write_buffer(buffer, 0, bytemuck::cast_slice(&self.vertices));
281
282        let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
283            label: Some("World Lines"),
284            color_attachments: &[Some(wgpu::RenderPassColorAttachment {
285                view: target,
286                resolve_target: None,
287                ops: wgpu::Operations {
288                    load: wgpu::LoadOp::Load,
289                    store: wgpu::StoreOp::Store,
290                },
291            })],
292            depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
293                view: depth,
294                depth_ops: Some(wgpu::Operations {
295                    load: wgpu::LoadOp::Load,
296                    store: wgpu::StoreOp::Store,
297                }),
298                stencil_ops: None,
299            }),
300            timestamp_writes: None,
301            occlusion_query_set: None,
302        });
303        pass.set_pipeline(&self.pipeline);
304        pass.set_vertex_buffer(0, buffer.slice(..));
305        pass.draw(0..self.vertices.len() as u32, 0..1);
306    }
307}

313    /// A camera at the origin looking down -z: 90° vertical fov, square
314    /// viewport, near 1, far 100 (wgpu depth 0..1). Column-major.
315    fn perspective() -> [[f32; 4]; 4] {
316        let (n, f) = (1.0f32, 100.0f32);
317        [
318            [1.0, 0.0, 0.0, 0.0],
319            [0.0, 1.0, 0.0, 0.0],
320            [0.0, 0.0, f / (n - f), -1.0],
321            [0.0, 0.0, n * f / (n - f), 0.0],
322        ]
323    }
324
325    fn view() -> LineView {
326        LineView {
327            view_proj: perspective(),
328            eye: [0.0; 3],
329            viewport: [200.0, 200.0],
330            px_per_point: 1.0,
331            linear_output: false,
332        }
333    }

335    fn line(a: [f32; 3], b: [f32; 3], thickness: f32) -> WorldLine {
336        WorldLine {
337            a,
338            b,
339            color: [1.0, 0.5, 0.25, 0.8],
340            thickness,
341        }
342    }

344    fn build(lines: &[WorldLine], v: &LineView) -> Vec<WorldLineVertex> {
345        let mut out = Vec::new();
346        build_vertices(lines, v, &mut out);
347        out
348    }

350    #[test]
351    fn a_segment_is_two_triangles_as_wide_as_asked_plus_the_feather() {
352        // Horizontal across the view at z = -10: x from -5 to 5 lands at
353        // NDC -0.5..0.5, so pixels 50..150 on a 200 px viewport.
354        let out = build(&[line([-5.0, 0.0, -10.0], [5.0, 0.0, -10.0], 4.0)], &view());
355        assert_eq!(out.len(), 6);
356        let ys: Vec<f32> = out.iter().map(|v| v.position[1] * 100.0).collect();
357        let span = ys.iter().cloned().fold(f32::MIN, f32::max)
358            - ys.iter().cloned().fold(f32::MAX, f32::min);
359        // 4 px wide plus a 1 px feather each side.
360        assert!((span - 6.0).abs() < 1e-3, "span {span}");
361        for v in &out {
362            assert_eq!(v.edge[1], 2.0);
363            assert_eq!(v.edge[0].abs(), 3.0);
364            assert_eq!(v.position[3], 1.0);
365            assert_eq!(v.color, [1.0, 0.5, 0.25, 0.8]);
366        }
367    }
