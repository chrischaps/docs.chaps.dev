1//! Textures the stylize chain draws into: each stage's ping-pong pair and
2//! the styles' own intermediates. They live in the post chain's resources,
3//! which a resize rebuilds, so every one of them (and every bind group
4//! cached over one) goes with the old size. A render-scale step keeps the
5//! surface's size, so it keeps the set (#270): display-stage targets, and
6//! the history in them, carry on, and HDR-stage ones are remade at the
7//! scene's new size the next time they're asked for.
8
9use std::collections::HashMap;
10use std::sync::{Arc, Mutex};
11
12/// A texture and its default view.
13pub struct StyleTexture {
14    pub texture: wgpu::Texture,
15    pub view: wgpu::TextureView,
16    pub width: u32,
17    pub height: u32,
18}
19
20/// Textures by name and format, made on first use.
21#[derive(Default)]
22pub struct StyleTargets {
23    textures: Mutex<HashMap<(&'static str, wgpu::TextureFormat), Arc<StyleTexture>>>,
24}
25
26impl StyleTargets {
27    /// The texture `name` in `format`, `width` x `height`. A name asked
28    /// for at a second size is made again: an HDR-stage target after a
29    /// render-scale step (the bind groups over the old one went with the
30    /// resources it was kept from), or a caller drawing one chain at two
31    /// sizes, which nothing does (so it's logged).
32    pub fn get(
33        &self,
34        device: &wgpu::Device,
35        name: &'static str,
36        format: wgpu::TextureFormat,
37        width: u32,
38        height: u32,
39    ) -> Arc<StyleTexture> {
40        let (width, height) = (width.max(1), height.max(1));
41        let mut textures = self.textures.lock().unwrap_or_else(|e| e.into_inner());
42        if let Some(t) = textures.get(&(name, format)) {
43            if (t.width, t.height) == (width, height) {
44                return t.clone();
45            }
46            tracing::debug!(
47                "style target {name} asked for at {width}x{height}, made at {}x{}",
48                t.width,
49                t.height
50            );
51        }
52        let texture = device.create_texture(&wgpu::TextureDescriptor {
53            label: Some(name),
54            size: wgpu::Extent3d {
55                width,
56                height,
57                depth_or_array_layers: 1,
58            },
59            mip_level_count: 1,
60            sample_count: 1,
61            dimension: wgpu::TextureDimension::D2,
62            format,
63            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
64            view_formats: &[],
65        });
66        let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
67        let made = Arc::new(StyleTexture {
68            texture,
69            view,
70            width,
71            height,
72        });
73        textures.insert((name, format), made.clone());
74        made
75    }
76
77    /// How many textures it holds.
78    pub fn len(&self) -> usize {
79        self.textures.lock().map_or(0, |t| t.len())
80    }
81
82    pub fn is_empty(&self) -> bool {
83        self.len() == 0
84    }
85
86    /// What it holds: how many textures, their bytes, and how many of them
87    /// are `width` x `height` in `format` (the display stage's full-size
88    /// textures, #302). Textures stay until a resize, so this is everything
89    /// any configuration since then has asked for.
90    pub fn summary(&self, format: wgpu::TextureFormat, width: u32, height: u32) -> TargetSummary {
91        let Ok(textures) = self.textures.lock() else {
92            return TargetSummary::default();
93        };
94        let mut summary = TargetSummary::default();
95        for ((_, f), t) in textures.iter() {
96            summary.count += 1;
97            let texel = f.block_copy_size(None).unwrap_or(4);
98            summary.bytes += u64::from(t.width) * u64::from(t.height) * u64::from(texel);
99            if (*f, t.width, t.height) == (format, width, height) {
100                summary.full_size += 1;
101            }
102        }
103        summary
104    }
105}

107/// [`StyleTargets::summary`].
