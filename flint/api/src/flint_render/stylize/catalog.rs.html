1//! What styles exist and what they take, for agents and people (#303):
2//! `flint render --list-styles` prints [`catalog_text`], and with `--json`
3//! the [`catalog`] as JSON. Everything comes from the registry's
4//! descriptors, so a new style (built-in or a `styles/<name>.shader.toml`
5//! once [`custom::load_dir`](super::custom::load_dir) has run) appears with
6//! no change here.
7//!
8//! Kuwahara is the one exception while it keeps its settings in
9//! `kuwahara_*` fields on `[post_process]` and its descriptor has no
10//! params: [`KUWAHARA_PARAMS`] gives them here, with the F4 menu's ranges,
11//! so it lists like any other style. Once the descriptor declares its own
12//! (#298 moves them into the style table) those are listed instead, and
13//! this fallback can go.
14

47/// One registered style, as listed.
48#[derive(Debug, Clone, Serialize)]
49pub struct StyleListing {
50    pub name: String,
51    pub label: String,
52    /// `"hdr"` (linear, before fog/bloom/tonemap) or `"display"` (after
53    /// the composite, at output resolution).
54    pub stage: &'static str,
55    /// `"builtin"`, or `"user"` for a `styles/<name>.shader.toml`.
56    pub source: &'static str,
57    /// Where a scene sets it: `[post_process.styles.<name>]`, or for
58    /// Kuwahara, `[post_process]` with `kuwahara_` before each key.
59    pub table: String,
60    /// The component an entity takes the style with (per-entity styles).
61    pub component: Option<String>,
62    /// Whether it reads the entity style mask.
63    pub uses_mask: bool,
64    pub params: Vec<ParamListing>,
65    pub options: Vec<OptionListing>,
66    /// The colours (#315): a key a scene writes as `[r, g, b]` or
67    /// `"#rrggbb"`, over three of `params` (`tint` over `tint_r/g/b`).
68    pub colors: Vec<ColorListing>,
69}

71/// One colour setting.
72#[derive(Debug, Clone, Serialize)]
73pub struct ColorListing {
74    /// The key a scene writes (`tint`).
75    pub key: String,
76    /// Display values, 0..1 per channel.
77    pub default: [f32; 3],
78    /// The parameters underneath, red, green, blue.
79    pub channels: [String; 3],
80    pub label: String,
81    pub help: String,
82}

84/// One numeric parameter.
85#[derive(Debug, Clone, Serialize)]
86pub struct ParamListing {
87    pub name: String,
88    /// The key a scene writes (`kuwahara_radius` for Kuwahara, else `name`).
89    pub key: String,
90    pub default: f32,
91    /// The F4 menu's range. A value outside it is taken as written.
92    pub min: f32,
93    pub max: f32,
94    /// Whole numbers only (a cell size, a count).
95    pub integer: bool,
96    /// The colour this is a channel of (`tint` for `tint_r`), if any.
97    pub color: Option<String>,
98    pub label: String,
99    pub help: String,
100}

102/// One named choice (a string in the table).
103#[derive(Debug, Clone, Serialize)]
104pub struct OptionListing {
105    pub name: String,
106    pub default: String,
107    /// The values offered; empty means any text (a file or set name).
108    pub choices: Vec<String>,
109    pub label: String,
110    pub help: String,

113/// Every registered style, in registration order.
114pub fn catalog() -> Vec<StyleListing> {
115    registry().iter().map(|d| listing(d)).collect()
116}

118/// One style's listing.
119pub fn listing(d: &StyleDescriptor) -> StyleListing {
120    // The legacy fields, only while the descriptor declares none (#298)
121    let kuwahara = d.name == "kuwahara" && d.params.is_empty();
122    let params: &[StyleParam] = if kuwahara { &KUWAHARA_PARAMS } else { d.params };
123    let help = |name: &str| d.hint(name).map(|h| h.help.to_string()).unwrap_or_default();
124    StyleListing {
125        name: d.name.to_string(),
126        label: d.label.to_string(),
127        stage: match d.stage {
128            StyleStage::Hdr => "hdr",
129            StyleStage::Display => "display",
130        },
131        source: if is_builtin(d.name) {
132            "builtin"
133        } else {
134            "user"
135        },
136        table: if kuwahara {
137            "[post_process] kuwahara_*".to_string()
138        } else {
139            format!("[post_process.styles.{}]", d.name)
140        },
141        component: d.component.map(str::to_string),
142        uses_mask: d.uses_mask,
143        params: params
144            .iter()
145            .map(|p| ParamListing {
146                name: p.name.to_string(),
147                key: if kuwahara {
148                    format!("kuwahara_{}", p.name)
149                } else {
150                    p.name.to_string()
151                },
152                default: p.default,
153                min: p.min,
154                max: p.max,
155                integer: (kuwahara && p.name == "radius")
156                    || d.hint(p.name).is_some_and(|h| h.integer),
157                color: d.color_of(p.name).map(|c| c.base.to_string()),
158                label: d.label_for(p.name),
159                help: help(p.name),
160            })
161            .collect(),
162        options: d
163            .options
164            .iter()
165            .map(|o| OptionListing {
166                name: o.name.to_string(),
167                default: o.default.to_string(),
168                choices: o.choices.iter().map(|c| c.to_string()).collect(),
169                label: d.label_for(o.name),
170                help: help(o.name),
171            })
172            .collect(),
173        colors: d
174            .colors()
175            .into_iter()
176            .map(|c| ColorListing {
177                key: c.base.to_string(),
178                default: c.default_rgb(),
179                channels: c.channels.map(|p| p.name.to_string()),
180                label: d.label_for(c.base),
181                help: help(c.base),
182            })
183            .collect(),
184    }
185}

187/// [`catalog`] for a terminal: one block per style.
188pub fn catalog_text() -> String {
189    let mut out = String::new();
190    for s in catalog() {
191        let mut tags = vec![s.stage.to_string(), s.source.to_string()];
192        if let Some(c) = &s.component {
193            tags.push(format!("per-entity: {c}"));
194        }
195        if s.uses_mask {
196            tags.push("reads mask".into());
197        }
198        out.push_str(&format!("{} ({}) [{}]\n", s.name, s.label, tags.join(", ")));
199        out.push_str(&format!("  {}\n", s.table));
200        if s.params.is_empty() && s.options.is_empty() {
201            out.push_str("  (no settings)\n");
202        }
203        let width = s
204            .params
205            .iter()
206            .map(|p| p.key.len())
207            .chain(s.options.iter().map(|o| o.name.len()))
208            .max()
209            .unwrap_or(0);
210        for o in &s.options {
211            let choices = if o.choices.is_empty() {
212                "<text>".to_string()
213            } else {
214                o.choices.join("|")
215            };
216            out.push_str(&format!(
217                "  {:width$}  \"{}\"  {}{}\n",
218                o.name,
219                o.default,
220                choices,
221                tail(&o.help)
222            ));
223        }
224        for p in &s.params {
225            // A colour's channels are listed as the colour, where red stands
226            if let Some(key) = &p.color {
227                let c = s.colors.iter().find(|c| &c.key == key);
228                if let Some(c) = c.filter(|c| c.channels[0] == p.name) {
229                    let [r, g, b] = c.default.map(num);
230                    out.push_str(&format!(
231                        "  {:width$}  [{r}, {g}, {b}]  colour, or \"#rrggbb\"{}\n",
232                        c.key,
233                        tail(&c.help)
234                    ));
235                }
236                continue;
237            }
238            let int = if p.integer { ", integer" } else { "" };
239            out.push_str(&format!(
240                "  {:width$}  {}  [{} .. {}{}]{}\n",
241                p.key,
242                num(p.default),
243                num(p.min),
244                num(p.max),
245                int,
246                tail(&p.help)
247            ));
248        }
249        out.push('\n');
250    }
251    out.push_str(
252        "Every table also takes `enabled` and `order = <n>` (lower runs first in its stage).\n\
253         Try one with `flint render <scene> --style <name>:<key>=<value>`; `--json` for this as JSON.\n",
254    );
255    out
256}
257
258fn tail(help: &str) -> String {
259    if help.is_empty() {
260        String::new()
261    } else {
262        format!("  — {help}")
263    }
264}

266/// `4` rather than `4.0`, `0.35` as written.
267fn num(n: f32) -> String {
268    if n.fract() == 0.0 && n.abs() < 1e7 {
269        format!("{}", n as i64)
270    } else {
271        format!("{n}")
272    }

279    #[test]
280    fn lists_kuwahara_and_invert_with_stages_and_ranges() {
281        let all = catalog();
282        let kuwahara = all.iter().find(|s| s.name == "kuwahara").unwrap();
283        assert_eq!(kuwahara.stage, "hdr");
284        let radius = kuwahara.params.iter().find(|p| p.name == "radius").unwrap();
285        assert_eq!((radius.min, radius.max, radius.default), (1.0, 8.0, 4.0));
286        // The legacy field, or the table key once #298 lands
287        assert!(radius.key == "kuwahara_radius" || radius.key == "radius");
288        assert!(radius.integer);
289        let invert = all.iter().find(|s| s.name == "invert").unwrap();
290        assert_eq!(invert.stage, "display");
291        assert!(invert.params.iter().any(|p| p.name == "strength"));
292        assert_eq!(invert.table, "[post_process.styles.invert]");
293    }

296    fn json_parses_and_lists_every_style() {

298        let back: serde_json::Value = serde_json::from_str(&json).unwrap();
299        assert_eq!(back.as_array().unwrap().len(), registry().len());
300    }

305        let kuwahara = catalog()

307            .find(|s| s.name == "kuwahara")

315    #[test]
316    fn a_colour_is_listed_once_by_its_key() {
317        let ascii = listing(super::super::find("ascii").unwrap());
318        assert_eq!(ascii.colors.len(), 1);
319        let tint = &ascii.colors[0];
320        assert_eq!(tint.key, "tint");
321        assert_eq!(tint.default, [0.35, 1.0, 0.45]);
322        assert_eq!(
323            tint.channels,
324            ["tint_r", "tint_g", "tint_b"].map(String::from)
325        );
326        let g = ascii.params.iter().find(|p| p.name == "tint_g").unwrap();
327        assert_eq!(g.color.as_deref(), Some("tint"));
328        let text = catalog_text();
329        assert!(text.contains("[0.35, 1, 0.45]  colour"), "{text}");
330        assert!(!text.contains("  tint_r "), "{text}");
331    }
