1//! Pencil sketch (#273): the scene drawn in graphite on paper. Tone
2//! becomes layers of cross-hatching, the scene's depth and creases become
3//! wobbling contour lines, and both sit on a paper with grain.
4//!
5//! ```toml
6//! [post_process.styles.pencil]
7//! spacing = 6.0          # hatch line spacing, points
8//! layers = 3             # hatch layers (0-4); each joins as the tone darkens
9//! direction = "surface"  # surface | screen | tensor
10//! angle = 50.0           # the base angle, degrees
11//! contour = 1.0          # outline strength
12//! wobble = 1.5           # how far lines wander, points
13//! sketchy = 0.6          # a second, offset outline
14//! grain = 0.5            # paper grain, and how much it breaks strokes
15//! paper = [0.95, 0.93, 0.88]   # a 3-array fills paper_r/g/b
16//! lead = [0.16, 0.16, 0.19]
17//! color = 0.0            # 1 draws over the scene's colour instead of paper
18//! seed = 0               # a different hand; the same seed draws the same picture
19//! boil = 0.0             # redraws a second (0 = still)
20//! ```
21//!
22//! It runs at the display stage, on the finished picture at output
23//! resolution, so strokes stay crisp under a render scale; sizes are in
24//! points (times [`StyleFrame::pixels_per_point`]). Two passes: the first
25//! writes each pixel's tone, contour strength and local stroke direction
26//! into a scratch target; the second blurs the direction into a flow and
27//! draws strokes, outlines and paper from it.
28//!
29//! Stroke direction (`direction`): `surface` runs strokes along the
30//! surface's iso-depth lines (round a barrel, across a floor), taken from
31//! the depth buffer; `tensor` runs them along the picture's luminance
32//! edges (its structure tensor); `screen` hatches every surface at
33//! `angle`. Where a surface or tensor direction is unsure (a face turned
34//! to the camera, a flat colour), `angle` shows through.
35//!
36//! Output is a pure function of the picture, the depth, the settings and
37//! `seed` (with `boil`, also the post clock, which `--style-time` sets):
38//! every random choice is an integer hash.
39//!
40//! ## Per entity
41//!
42//! As ASCII (#266): an entity with a `pencil_style` component, or a
43//! `style_mask { style = "pencil" }`, is drawn in pencil while the rest of
44//! the frame passes through, and each distinct set of settings is a row of
45//! the variant table the passes index by mask id. `scope = "entities"`
46//! keeps a scene table as those entities' preset, and `grid = "object"`
47//! pins an entity's hatching to its projected origin, so its strokes
48//! travel with it rather than swimming over it.

275/// The camera's forward axis from its inverse view-projection: from the
276/// centre of the near plane to the centre of the far one.
277pub fn forward_axis(inv_view_proj: &[[f32; 4]; 4]) -> [f32; 3] {
278    let at = |z: f32| {
279        let m = inv_view_proj;
280        let row = |r: usize| m[2][r] * z + m[3][r];
281        let w = row(3);
282        [row(0) / w, row(1) / w, row(2) / w]
283    };
284    let (near, far) = (at(0.0), at(1.0));
285    let d = [far[0] - near[0], far[1] - near[1], far[2] - near[2]];
286    let len = (d[0] * d[0] + d[1] * d[1] + d[2] * d[2]).sqrt();
287    if len.is_finite() && len > 0.0 {
288        d.map(|c| c / len)
289    } else {
290        [0.0, 0.0, -1.0]
291    }
292}

294/// The direction mode's shader number; an unknown name is `surface`.
295pub fn direction_mode(name: &str) -> f32 {
296    DIRECTIONS.iter().position(|m| *m == name).unwrap_or(0) as f32
297}

299/// The seed a frame draws with: `seed`, moved on `boil` times a second of
300/// post time, so a still drawing (boil 0) never changes and a boiling one
301/// changes in steps, the same steps at the same `--style-time`.
302pub fn effective_seed(seed: f32, boil: f32, time: f32) -> f32 {
303    let base = seed.max(0.0).floor();
304    if boil > 0.0 && time.is_finite() {
305        base + (time.max(0.0) * boil).floor()
306    } else {
307        base
308    }
309}

367struct Pencil {
368    prep_pipeline: wgpu::RenderPipeline,
369    draw_pipeline: wgpu::RenderPipeline,
370    uniforms: wgpu::Buffer,
371    uniform_bgl: wgpu::BindGroupLayout,
372    input_bgl: wgpu::BindGroupLayout,
373    prep_bgl: wgpu::BindGroupLayout,
374    geometry_bgl: wgpu::BindGroupLayout,
375    _no_mask: wgpu::Texture,
376    no_mask_view: wgpu::TextureView,
377    /// Whether the too-many-variants warning has been given.
378    warned: Mutex<bool>,
379}
380
381impl Pencil {
382    /// The depth and the mask (or the one-pixel stand-in).
383    fn geometry_bind(&self, frame: &StyleFrame<'_>, with_mask: bool) -> Arc<wgpu::BindGroup> {
384        let mask = frame.mask.filter(|_| with_mask);
385        frame.depth_bind("Pencil Geometry BG", mask.is_some() as u64, || {
386            frame.device.create_bind_group(&wgpu::BindGroupDescriptor {
387                label: Some("Pencil Geometry BG"),
388                layout: &self.geometry_bgl,
389                entries: &[
390                    wgpu::BindGroupEntry {
391                        binding: 0,
392                        resource: wgpu::BindingResource::TextureView(frame.depth),
393                    },
394                    wgpu::BindGroupEntry {
395                        binding: 1,
396                        resource: wgpu::BindingResource::TextureView(
397                            mask.unwrap_or(&self.no_mask_view),
398                        ),
399                    },
400                ],
401            })
402        })
403    }
404
405    /// Row 0 (the whole frame's) and a row for each per-entity variant,
406    /// with the row of every mask id.
407    fn variant_table(&self, frame: &StyleFrame<'_>) -> (Vec<PencilRow>, [u32; MAX_IDS]) {
408        let whole = row(
409            &|n| frame.param(n),
410            &|n| frame.option(n).to_string(),
411            frame.time,
412            [0.0, 0.0],
413            0.0,
414        );
415        let mut rows = vec![whole];
416        let mut slots = [0u32; MAX_IDS];
417        // Ids another style draws are that style's pixels
418        for (id, _) in frame.style_ids.iter() {
419            let Some(v) = frame.style_ids.variant(id) else {
420                continue;
421            };
422            if (id as usize) < MAX_IDS
423                && v.style != STYLE.name
424                && !v.off
425                && super::find(&v.style).is_some()
426            {
427                slots[id as usize] = FOREIGN;
428            }
429        }
430        for (id, v) in frame.variants() {
431            if rows.len() == MAX_VARIANTS || id as usize >= MAX_IDS {
432                let mut warned = self.warned.lock().unwrap_or_else(|e| e.into_inner());
433                if !*warned {
434                    tracing::warn!(
435                        "pencil: more per-entity variants than the {} (or ids past {MAX_IDS}) \
436                         one frame holds; the rest are drawn like the whole frame",
437                        MAX_VARIANTS - 1
438                    );
439                    *warned = true;
440                }
441                continue;
442            }
443            let origin = v
444                .anchor
445                .and_then(|a| frame.project(a))
446                .map_or([0.0, 0.0], |at| [at[0].round(), at[1].round()]);
447            let flags = if v.off { PASS_THROUGH } else { 0.0 };
448            slots[id as usize] = rows.len() as u32;
449            rows.push(row(
450                &|n| frame.variant_param(v, n),
451                &|n| frame.variant_option(v, n).to_string(),
452                frame.time,
453                origin,
454                flags,
455            ));
456        }
457        (rows, slots)
458    }
459}

461impl StylePass for Pencil {
462    fn encode(
463        &self,
464        frame: &StyleFrame<'_>,
465        encoder: &mut wgpu::CommandEncoder,
466        input: &StyleInput<'_>,
467        output: &wgpu::TextureView,
468    ) {
469        let (rows, slots) = self.variant_table(frame);
470        let with_mask = (rows.len() > 1 || slots.contains(&FOREIGN)) && frame.mask.is_some();
471        let mut uniforms = PencilUniforms {
472            inv_view_proj: frame.camera.inv_view_proj,
473            camera_pos: frame.camera.position,
474            pixels_per_point: if frame.pixels_per_point.is_finite() && frame.pixels_per_point > 0.0
475            {
476                frame.pixels_per_point
477            } else {
478                1.0
479            },
480            size: [frame.width as f32, frame.height as f32],
481            srgb: if frame.srgb() { 1.0 } else { 0.0 },
482            _pad0: 0.0,
483            forward: forward_axis(&frame.camera.inv_view_proj),
484            _pad1: 0.0,
485            info: [
486                rows.len() as u32,
487                !(frame.whole_frame() && frame.option("scope") != "entities") as u32,
488                with_mask as u32,
489                0,
490            ],
491            rows: [PencilRow::zeroed(); MAX_VARIANTS],
492            slots: [[0; 4]; MAX_IDS / 4],
493        };
494        uniforms.rows[..rows.len()].copy_from_slice(&rows);
495        for (i, slot) in slots.iter().enumerate() {
496            uniforms.slots[i / 4][i % 4] = *slot;
497        }
498        frame
499            .queue
500            .write_buffer(&self.uniforms, 0, bytemuck::bytes_of(&uniforms));
501
502        let uniform = frame.uniform("Pencil Uniform BG", &self.uniform_bgl, &self.uniforms);
503        let input_bind = frame.sampled_input("Pencil Input BG", &self.input_bgl, input);
504        let geometry = self.geometry_bind(frame, with_mask);
505        let prep = frame.target("pencil.prep", PREP_FORMAT);
506
507        kit::fullscreen_pass(
508            encoder,
509            "Pencil Prep",
510            &prep.view,
511            &self.prep_pipeline,
512            &[&uniform, &input_bind, &geometry],
513        );
514        kit::fullscreen_pass(
515            encoder,
516            "Pencil Pass",
517            output,
518            &self.draw_pipeline,
519            &[
520                &uniform,
521                &input_bind,
522                &geometry,
523                &frame.sampled("Pencil Prep BG", &self.prep_bgl, &prep.view),
524            ],
525        );
526    }
527}

585    /// The `pencil_style` schema and the descriptor agree: every field is a
586    /// parameter, an option, a 3-array of `_r/_g/_b` parameters or
587    /// `enabled`; every parameter and option can be set from it; and no
588    /// field has a default, which the loader would fill in over the
589    /// scene's table (the #266 gotcha).
590    #[test]
591    fn the_component_schema_matches_the_descriptor() {
592        let schema: toml::Table = toml::from_str(include_str!(
593            "../../../../schemas/components/pencil_style.toml"
594        ))
595        .unwrap();
596        let fields = schema["component"]["pencil_style"]["fields"]
597            .as_table()
598            .unwrap();
599        let is_param = |n: &str| STYLE.params.iter().any(|p| p.name == n);
600        let is_option = |n: &str| STYLE.options.iter().any(|o| o.name == n);
601        let mut reached: Vec<String> = Vec::new();
602        for (key, field) in fields {
603            assert!(field.get("default").is_none(), "{key} has a default");
604            if key == "enabled" {
605                continue;
606            }
607            if is_param(key) || is_option(key) {
608                reached.push(key.clone());
609            } else {
610                let channels = ["r", "g", "b"].map(|c| format!("{key}_{c}"));
611                assert!(
612                    channels.iter().all(|c| is_param(c)),
613                    "{key} is no parameter, option or colour of pencil"
614                );
615                assert_eq!(field["type"].as_str(), Some("vec3"), "{key}");
616                reached.extend(channels);
617            }
618        }
619        for p in STYLE.params {
620            assert!(
621                reached.iter().any(|r| r == p.name),
622                "{} not in the schema",
623                p.name
624            );
625        }
626        for o in STYLE.options {
627            // scope is the scene table's, not an entity's
628            if o.name != "scope" {
629                assert!(
630                    reached.iter().any(|r| r == o.name),
631                    "{} not in the schema",
632                    o.name
633                );
634            }
635        }
636    }
