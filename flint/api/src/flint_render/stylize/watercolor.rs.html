1//! Watercolor (#272): the picture as transparent pigment on paper. Colour
2//! bleeds into soft washes, their borders wobble, pigment pools darker
3//! along the edges of each wash and settles into the paper's grain.
4//!
5//! ```toml
6//! [post_process.styles.watercolor]
7//! wash = 4.0          # how far pigment diffuses, in points (the blur radius)
8//! bleed = 0.55        # how much of the colour is the diffused wash
9//! wobble = 2.0        # how far wash borders wander, in points
10//! wobble_scale = 40   # the wander's wavelength, in points
11//! edge = 1.0          # pigment pooling where colour changes (edge darkening)
12//! edge_depth = 0.8    # pooling along silhouettes, from depth
13//! edge_width = 1.5    # how wide the pooled rim is, in points
14//! turbulence = 0.35   # low-frequency blotches of thicker and thinner pigment
15//! granulation = 0.5   # pigment settling into the paper's valleys
16//! grain = 0.35        # the paper's own relief, shading even bare paper
17//! grain_scale = 1.0   # the paper's tooth (bigger = coarser)
18//! pigment = 1.0       # overall density: < 1 dilutes, > 1 deepens
19//! layers = 4          # glazes: tone steps, each edged darker (0 = smooth)
20//! paper_r = 0.97      # the paper's colour (a component's `paper = [r, g, b]`)
21//! scope = "screen"    # or "entities": a preset for watercolor_style entities
22//! ```
23//!
24//! It runs at the display stage, on display-referred colour, because it
25//! paints on paper: the paper and its grain must not be fogged, bloomed or
26//! tonemapped as if they were in the world. Three passes: the wash (a
27//! separable Gaussian of the picture, horizontal then vertical, in two
28//! targets of its own), then the paint pass, which wobbles where it reads
29//! the picture, the wash and the mask, mixes the wash in, darkens pigment
30//! at colour and depth edges, modulates its density with turbulence and
31//! granulation (Bousseau et al.'s `C - (C - C^2)(d - 1)`), and lays the
32//! result on the paper.
33//!
34//! The paper lives in screen space, in points, so its grain stays put as
35//! the camera moves (the paper doesn't travel with the world), and the
36//! same size to the eye on a high-DPI screen.
37//!
38//! ## Per entity
39//!
40//! An entity with a `watercolor_style` component (or a `style_mask` naming
41//! `watercolor`) is painted with no table at all, the rest of the frame
42//! passing through, exactly as ASCII does it (#266); `scope = "entities"`
43//! keeps a table as their preset. The mask is read at the wobbled position,
44//! so a painted entity's silhouette wanders out over its surroundings like
45//! a real wash's. The wash radius is one per frame, the largest any row
46//! asks for; every other setting is per entity.
47//!
48//! ## With Kuwahara
49//!
50//! They chain: Kuwahara runs at the HDR stage and flattens the scene into
51//! brush pools, watercolor then paints those pools on paper. Neither needs
52//! the other, and they share nothing (Kuwahara's structure tensor is an
53//! HDR-stage intermediate; the wash is the only smoothing watercolor does).

199impl Watercolor {
200    fn new(d: &super::StyleDevice) -> Self {
201        let device = d.device;
202        let uniform_bgl = kit::uniform_layout(device, "Watercolor Uniform BGL");
203        let sampled_bgl = kit::sampled_layout(device, "Watercolor Texture BGL");
204        // The paint pass's mask and depth
205        let entity_bgl = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
206            label: Some("Watercolor Entity BGL"),
207            entries: &[
208                wgpu::BindGroupLayoutEntry {
209                    binding: 0,
210                    visibility: wgpu::ShaderStages::FRAGMENT,
211                    ty: wgpu::BindingType::Texture {
212                        sample_type: wgpu::TextureSampleType::Uint,
213                        view_dimension: wgpu::TextureViewDimension::D2,
214                        multisampled: false,
215                    },
216                    count: None,
217                },
218                wgpu::BindGroupLayoutEntry {
219                    binding: 1,
220                    visibility: wgpu::ShaderStages::FRAGMENT,
221                    ty: wgpu::BindingType::Texture {
222                        sample_type: wgpu::TextureSampleType::Depth,
223                        view_dimension: wgpu::TextureViewDimension::D2,
224                        multisampled: false,
225                    },
226                    count: None,
227                },
228            ],
229        });
230        let module = kit::shader(device, "Watercolor Shader", include_str!("watercolor.wgsl"));
231        let wash_module = kit::shader(
232            device,
233            "Watercolor Wash Shader",
234            include_str!("watercolor_wash.wgsl"),
235        );
236        // A one-texel mask of id 0, bound while no entity asks
237        let no_mask = device.create_texture(&wgpu::TextureDescriptor {
238            label: Some("Watercolor No Mask"),
239            size: wgpu::Extent3d {
240                width: 1,
241                height: 1,
242                depth_or_array_layers: 1,
243            },
244            mip_level_count: 1,
245            sample_count: 1,
246            dimension: wgpu::TextureDimension::D2,
247            format: crate::style_mask::STYLE_MASK_FORMAT,
248            usage: wgpu::TextureUsages::TEXTURE_BINDING,
249            view_formats: &[],
250        });
251        let no_mask_view = no_mask.create_view(&Default::default());
252        Self {
253            wash_pipeline: kit::fullscreen_pipeline(
254                device,
255                "Watercolor Wash Pipeline",
256                &wash_module,
257                &[&uniform_bgl, &sampled_bgl],
258                d.format,
259            ),
260            paint_pipeline: kit::fullscreen_pipeline(
261                device,
262                "Watercolor Pipeline",
263                &module,
264                &[&uniform_bgl, &sampled_bgl, &sampled_bgl, &entity_bgl],
265                d.format,
266            ),
267            wash_h: kit::uniform_buffer::<WashUniforms>(device, "Watercolor Wash H Uniforms"),
268            wash_v: kit::uniform_buffer::<WashUniforms>(device, "Watercolor Wash V Uniforms"),
269            uniforms: kit::uniform_buffer::<PaintUniforms>(device, "Watercolor Uniforms"),
270            uniform_bgl,
271            sampled_bgl,
272            entity_bgl,
273            _no_mask: no_mask,
274            no_mask_view,
275            warned: Default::default(),
276        }
277    }

279    /// The mask (or the stand-in) and the depth, bound together.
280    fn entity_bind(&self, frame: &StyleFrame<'_>, with_mask: bool) -> Arc<wgpu::BindGroup> {
281        let mask = frame.mask.filter(|_| with_mask);
282        frame.depth_bind("Watercolor Entity BG", mask.is_some() as u64, || {
283            frame.device.create_bind_group(&wgpu::BindGroupDescriptor {
284                label: Some("Watercolor Entity BG"),
285                layout: &self.entity_bgl,
286                entries: &[
287                    wgpu::BindGroupEntry {
288                        binding: 0,
289                        resource: wgpu::BindingResource::TextureView(
290                            mask.unwrap_or(&self.no_mask_view),
291                        ),
292                    },
293                    wgpu::BindGroupEntry {
294                        binding: 1,
295                        resource: wgpu::BindingResource::TextureView(frame.depth),
296                    },
297                ],
298            })
299        })
300    }
301}

303/// A row from a settings source: the scene's (`param` = the frame's) or a
304/// variant's.
305fn row(param: impl Fn(&str) -> f32, off: bool) -> RowUniform {
306    RowUniform {
307        wobble: param("wobble").clamp(0.0, 16.0),
308        wobble_scale: param("wobble_scale").max(1.0),
309        bleed: param("bleed").clamp(0.0, 1.0),
310        edge: param("edge").max(0.0),
311        edge_depth: param("edge_depth").max(0.0),
312        edge_width: param("edge_width").clamp(0.25, 16.0),
313        turbulence: param("turbulence").clamp(0.0, 1.0),
314        granulation: param("granulation").max(0.0),
315        grain: param("grain").clamp(0.0, 1.0),
316        grain_scale: param("grain_scale").clamp(0.05, 16.0),
317        pigment: param("pigment").clamp(0.0, 4.0),
318        flags: if off { PASS_THROUGH } else { 0 },
319        paper: [
320            param("paper_r").clamp(0.0, 1.0),
321            param("paper_g").clamp(0.0, 1.0),
322            param("paper_b").clamp(0.0, 1.0),
323        ],
324        layers: param("layers").clamp(0.0, 32.0),
325    }
326}

328impl StylePass for Watercolor {
329    fn encode(
330        &self,
331        frame: &StyleFrame<'_>,
332        encoder: &mut wgpu::CommandEncoder,
333        input: &StyleInput<'_>,
334        output: &wgpu::TextureView,
335    ) {
336        // The variant table: row 0 the scene's, then each entity variant
337        let mut rows = vec![row(|n| frame.param(n), false)];
338        let mut slots = [0u32; MAX_IDS];
339        let mut wash = frame.param("wash");
340        for (id, v) in frame.variants() {
341            if rows.len() == MAX_VARIANTS || id as usize >= MAX_IDS {
342                if !self.warned.swap(true, std::sync::atomic::Ordering::Relaxed) {
343                    tracing::warn!(
344                        "watercolor: more per-entity variants than the {} (or ids past \
345                         {MAX_IDS}) one frame holds; the rest are painted like the scene",
346                        MAX_VARIANTS - 1
347                    );
348                }
349                continue;
350            }
351            if !v.off {
352                wash = wash.max(frame.variant_param(v, "wash"));
353            }
354            slots[id as usize] = rows.len() as u32;
355            rows.push(row(|n| frame.variant_param(v, n), v.off));
356        }
357        let with_mask = rows.len() > 1;
358        let passthrough = !(frame.whole_frame() && frame.option("scope") != "entities");
359
360        // Pass 1 and 2: the wash, a separable Gaussian
361        let radius = (wash.max(0.0) * frame.pixels_per_point).min(MAX_WASH_PX);
362        let texel = frame.texel_size();
363        for (buffer, dir) in [(&self.wash_h, [1.0, 0.0]), (&self.wash_v, [0.0, 1.0])] {
364            let u = WashUniforms {
365                texel,
366                dir,
367                radius,
368                _pad: [0.0; 3],
369            };
370            frame.queue.write_buffer(buffer, 0, bytemuck::bytes_of(&u));
371        }
372        let half = frame.target("watercolor.wash.h", frame.format);
373        let washed = frame.target("watercolor.wash", frame.format);
374        kit::fullscreen_pass(
375            encoder,
376            "Watercolor Wash H",
377            &half.view,
378            &self.wash_pipeline,
379            &[
380                &frame.uniform("Watercolor Wash H BG", &self.uniform_bgl, &self.wash_h),
381                &frame.sampled_input("Watercolor Input BG", &self.sampled_bgl, input),
382            ],
383        );
384        kit::fullscreen_pass(
385            encoder,
386            "Watercolor Wash V",
387            &washed.view,
388            &self.wash_pipeline,
389            &[
390                &frame.uniform("Watercolor Wash V BG", &self.uniform_bgl, &self.wash_v),
391                &frame.sampled(
392                    "Watercolor Wash H Texture BG",
393                    &self.sampled_bgl,
394                    &half.view,
395                ),
396            ],
397        );
398
399        // Pass 3: paint
400        let mut uniforms = PaintUniforms {
401            size: [frame.width as f32, frame.height as f32],
402            pixels_per_point: frame.pixels_per_point.max(0.25),
403            srgb: frame.srgb() as u32,
404            row_count: rows.len() as u32,
405            passthrough: passthrough as u32,
406            near: frame.camera.near,
407            far: frame.camera.far,
408            rows: [RowUniform::zeroed(); MAX_VARIANTS],
409            slots: [[0; 4]; MAX_IDS / 4],
410        };
411        uniforms.rows[..rows.len()].copy_from_slice(&rows);
412        for (i, slot) in slots.iter().enumerate() {
413            uniforms.slots[i / 4][i % 4] = *slot;
414        }
415        frame
416            .queue
417            .write_buffer(&self.uniforms, 0, bytemuck::bytes_of(&uniforms));
418        kit::fullscreen_pass(
419            encoder,
420            "Watercolor Pass",
421            output,
422            &self.paint_pipeline,
423            &[
424                &frame.uniform("Watercolor Uniform BG", &self.uniform_bgl, &self.uniforms),
425                &frame.sampled_input("Watercolor Paint Input BG", &self.sampled_bgl, input),
426                &frame.sampled(
427                    "Watercolor Wash Texture BG",
428                    &self.sampled_bgl,
429                    &washed.view,
430                ),
431                &self.entity_bind(frame, with_mask),
432            ],
433        );
434    }
435}

441    #[test]
442    fn uniforms_match_the_shader_layout() {
443        assert_eq!(std::mem::size_of::<RowUniform>(), 64);
444        assert_eq!(std::mem::size_of::<WashUniforms>(), 32);
445        assert_eq!(
446            std::mem::size_of::<PaintUniforms>(),
447            32 + MAX_VARIANTS * 64 + MAX_IDS / 4 * 16
448        );
449    }
450
451    #[test]
452    fn rows_read_the_scene_then_the_variant() {
453        let mut config = crate::postprocess::PostProcessConfig::default();
454        config.styles.set_param("watercolor", "edge", 2.0);
455        let mut v = crate::style_mask::StyleVariant::plain("watercolor");
456        v.params.insert("paper_r".into(), 0.5);
457        let r = row(|n| v.param(&STYLE, &config, n), false);
458        assert_eq!(r.edge, 2.0, "the scene's table");
459        assert_eq!(r.paper, [0.5, 0.94, 0.87], "the entity's, then defaults");
460        assert_eq!(r.flags, 0);
461        assert_eq!(row(|n| STYLE.param(&config, n), true).flags, PASS_THROUGH);
462    }

464    #[test]
465    fn a_paper_triple_fills_the_paper_channels() {
466        let table: toml::Table = toml::from_str("paper = [0.9, 0.8, 0.7]\nwobble = 3").unwrap();
467        let mut v = crate::style_mask::StyleVariant::plain("watercolor");
468        crate::style_mask::read_overrides(&STYLE, &table, &mut v);
469        assert_eq!(v.params.get("paper_g"), Some(&0.8));
470        assert_eq!(v.params.get("wobble"), Some(&3.0));
471    }
