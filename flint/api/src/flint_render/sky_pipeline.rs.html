1//! Procedural sky pipeline — gradient, sun disc, stars, FBM clouds.
2//!
3//! Drawn as a fullscreen triangle at the far plane (same trick as the
4//! texture skybox, which it replaces when a `sky` component is present).
5//! All parameters live in one uniform block interpolated per frame by a
6//! game-side time-of-day controller — sunrise blends to noon to starfield
7//! continuously.
8
9use bytemuck::{Pod, Zeroable};
10use flint_core::toml_util::{toml_color, toml_f32};
11
12/// Uniform block — layout must match `SkyUniforms` in sky_shader.wgsl.
13#[repr(C)]
14#[derive(Copy, Clone, Debug, Pod, Zeroable)]
15pub struct SkyUniformsGpu {
16    pub inv_view_proj: [[f32; 4]; 4],
17    pub sun_dir: [f32; 3],
18    pub sun_disc_size: f32,
19    pub sun_color: [f32; 4],
20    pub zenith_color: [f32; 4],
21    pub horizon_color: [f32; 4],
22    pub haze_color: [f32; 4],
23    pub cloud_tint: [f32; 4],
24    pub sun_glow: f32,
25    pub star_opacity: f32,
26    pub cloud_coverage: f32,
27    pub cloud_density: f32,
28    pub cloud_scale: f32,
29    pub cloud_drift_x: f32,
30    pub cloud_drift_y: f32,
31    pub time: f32,
32}

34/// CPU-side mirror of the `sky` component (minus sun, which is copied from
35/// the scene's first directional light so disc and lighting always agree).
36#[derive(Debug, Clone, PartialEq)]
37pub struct SkyParams {
38    pub zenith_color: [f32; 4],
39    pub horizon_color: [f32; 4],
40    pub haze_color: [f32; 4],
41    pub cloud_tint: [f32; 4],
42    pub sun_disc_size: f32,
43    pub sun_glow: f32,
44    pub star_opacity: f32,
45    pub cloud_coverage: f32,
46    pub cloud_density: f32,
47    pub cloud_scale: f32,
48    pub cloud_drift_x: f32,
49    pub cloud_drift_y: f32,
50    /// Optional hemisphere ambient override (rgb + intensity in alpha).
51    pub ambient_sky: Option<[f32; 4]>,
52    pub ambient_ground: Option<[f32; 4]>,
53}
54
55impl Default for SkyParams {

60            haze_color: [0.85, 0.92, 0.98, 0.6],

64            star_opacity: 0.0,
65            cloud_coverage: 0.35,
66            cloud_density: 0.25,
67            cloud_scale: 0.9,
68            cloud_drift_x: 0.004,
69            cloud_drift_y: 0.0015,
70            ambient_sky: None,
71            ambient_ground: None,

74}

76impl SkyParams {
77    pub fn from_component(value: &toml::Value) -> Self {
78        let d = Self::default();
79        let f = |name: &str, dv: f32| value.get(name).and_then(toml_f32).unwrap_or(dv);
80        let c = |name: &str, dv: [f32; 4]| value.get(name).and_then(toml_color).unwrap_or(dv);
81        Self {
82            zenith_color: c("zenith_color", d.zenith_color),
83            horizon_color: c("horizon_color", d.horizon_color),
84            haze_color: c("haze_color", d.haze_color),
85            cloud_tint: c("cloud_tint", d.cloud_tint),
86            sun_disc_size: f("sun_disc_size", d.sun_disc_size).clamp(0.001, 0.5),
87            sun_glow: f("sun_glow", d.sun_glow).max(0.0),
88            star_opacity: f("star_opacity", d.star_opacity).clamp(0.0, 1.0),
89            cloud_coverage: f("cloud_coverage", d.cloud_coverage).clamp(0.0, 1.0),
90            cloud_density: f("cloud_density", d.cloud_density).clamp(0.0, 1.0),
91            cloud_scale: f("cloud_scale", d.cloud_scale).max(0.01),
92            cloud_drift_x: f("cloud_drift_x", d.cloud_drift_x),
93            cloud_drift_y: f("cloud_drift_y", d.cloud_drift_y),
94            ambient_sky: value.get("ambient_sky").and_then(toml_color),
95            ambient_ground: value.get("ambient_ground").and_then(toml_color),
96        }
97    }
98}

100/// The procedural sky pipeline.
101pub struct SkyPipeline {
102    pub pipeline: wgpu::RenderPipeline,
103    pub uniform_bind_group_layout: wgpu::BindGroupLayout,
104}
105
106impl SkyPipeline {
107    pub fn new(device: &wgpu::Device, format: wgpu::TextureFormat) -> Self {
108        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
109            label: Some("Sky Shader"),
110            source: wgpu::ShaderSource::Wgsl(include_str!("sky_shader.wgsl").into()),
111        });
112
113        let uniform_bind_group_layout =
114            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
115                entries: &[wgpu::BindGroupLayoutEntry {
116                    binding: 0,
117                    visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
118                    ty: wgpu::BindingType::Buffer {
119                        ty: wgpu::BufferBindingType::Uniform,
120                        has_dynamic_offset: false,
121                        min_binding_size: None,
122                    },
123                    count: None,
124                }],
125                label: Some("Sky Uniform Bind Group Layout"),
126            });
127
128        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
129            label: Some("Sky Pipeline Layout"),
130            bind_group_layouts: &[&uniform_bind_group_layout],
131            push_constant_ranges: &[],
132        });
133
134        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
135            label: Some("Sky Render Pipeline"),
136            layout: Some(&pipeline_layout),
137            vertex: wgpu::VertexState {
138                module: &shader,
139                entry_point: Some("vs_sky"),
140                buffers: &[],
141                compilation_options: Default::default(),
142            },
143            fragment: Some(wgpu::FragmentState {
144                module: &shader,
145                entry_point: Some("fs_sky"),
146                targets: &[Some(wgpu::ColorTargetState {
147                    format,
148                    blend: None,
149                    write_mask: wgpu::ColorWrites::ALL,
150                })],
151                compilation_options: Default::default(),
152            }),
153            primitive: wgpu::PrimitiveState {
154                topology: wgpu::PrimitiveTopology::TriangleList,
155                cull_mode: None,
156                ..Default::default()
157            },
158            depth_stencil: Some(wgpu::DepthStencilState {
159                format: wgpu::TextureFormat::Depth32Float,
160                depth_write_enabled: false,
161                depth_compare: wgpu::CompareFunction::LessEqual,
162                stencil: wgpu::StencilState::default(),
163                bias: wgpu::DepthBiasState::default(),
164            }),
165            multisample: wgpu::MultisampleState::default(),
166            multiview: None,
167            cache: None,
168        });
169
170        Self {
171            pipeline,
172            uniform_bind_group_layout,
173        }
174    }
175}
