1/// Camera or light frustum defined by 6 clip planes.
2///
3/// Each plane `[a, b, c, d]` defines the half-space `ax + by + cz + d >= 0`
4/// as visible. Planes are normalized (unit-length normal).
5pub struct Frustum {
6    pub planes: [[f32; 4]; 6], // left, right, bottom, top, near, far
7}
8
9impl Frustum {
10    /// Extract 6 frustum planes from a view-projection matrix (Griggs/Hartmann method).
11    ///
12    /// Works for both perspective and orthographic projections.
13    /// The input matrix should be `projection * view` (or `projection * view * model`
14    /// if testing AABBs in model-local space).
15    pub fn from_view_projection(vp: &[[f32; 4]; 4]) -> Self {
16        // Columns of the VP matrix (vp is column-major: vp[col][row])
17        let c0 = vp[0];
18        let c1 = vp[1];
19        let c2 = vp[2];
20        let c3 = vp[3];
21
22        let mut planes = [[0.0f32; 4]; 6];
23
24        // Extract planes from VP matrix rows (Griggs/Hartmann method).
25        // In column-major layout, row R = [c0[R], c1[R], c2[R], c3[R]].
26        // Left  = row3 + row0
27        planes[0] = [c0[3] + c0[0], c1[3] + c1[0], c2[3] + c2[0], c3[3] + c3[0]];
28        // Right = row3 - row0
29        planes[1] = [c0[3] - c0[0], c1[3] - c1[0], c2[3] - c2[0], c3[3] - c3[0]];
30        // Bottom = row3 + row1
31        planes[2] = [c0[3] + c0[1], c1[3] + c1[1], c2[3] + c2[1], c3[3] + c3[1]];
32        // Top = row3 - row1
33        planes[3] = [c0[3] - c0[1], c1[3] - c1[1], c2[3] - c2[1], c3[3] - c3[1]];
34        // Near = row2 (wgpu [0,1] clip: visible half-space is z_ndc >= 0;
35        // the GL-convention extraction was row3 + row2. ADR 0055)
36        planes[4] = [c0[2], c1[2], c2[2], c3[2]];
37        // Far = row3 - row2
38        planes[5] = [c0[3] - c0[2], c1[3] - c1[2], c2[3] - c2[2], c3[3] - c3[2]];
39
40        // Normalize each plane
41        for plane in &mut planes {
42            let len = (plane[0] * plane[0] + plane[1] * plane[1] + plane[2] * plane[2]).sqrt();
43            if len > 1e-10 {
44                plane[0] /= len;
45                plane[1] /= len;
46                plane[2] /= len;
47                plane[3] /= len;
48            }
49        }
50
51        Self { planes }
52    }
53
54    /// Conservative AABB visibility test (p-vertex method).
55    ///
56    /// Returns `true` if the AABB is potentially visible (inside or intersecting
57    /// the frustum). Returns `false` only when the AABB is fully outside at least
58    /// one frustum plane. No false negatives — partially visible boxes always pass.
59    pub fn aabb_visible(&self, aabb_min: [f32; 3], aabb_max: [f32; 3]) -> bool {
60        for plane in &self.planes {
61            let (a, b, c, d) = (plane[0], plane[1], plane[2], plane[3]);
62
63            // P-vertex: the AABB corner most in the direction of the plane normal
64            let px = if a >= 0.0 { aabb_max[0] } else { aabb_min[0] };
65            let py = if b >= 0.0 { aabb_max[1] } else { aabb_min[1] };
66            let pz = if c >= 0.0 { aabb_max[2] } else { aabb_min[2] };
67
68            // If the p-vertex is behind this plane, the entire AABB is outside
69            if a * px + b * py + c * pz + d < 0.0 {
70                return false;
71            }
72        }
73        true
74    }
75}

81    /// Build a simple perspective VP looking down -Z from origin.
82    /// FOV 90°, aspect 1:1, near 0.1, far 100.
83    /// Uses the same wgpu [0,1] NDC convention as the engine's `Camera::perspective_matrix()`.
84    fn test_perspective_vp() -> [[f32; 4]; 4] {
85        let fov = std::f32::consts::FRAC_PI_2; // 90 degrees
86        let aspect = 1.0;
87        let near = 0.1_f32;
88        let far = 100.0_f32;
89
90        let f = 1.0 / (fov / 2.0).tan();
91        let depth = far - near;
92        // Perspective projection (right-handed, [0,1] depth — matches engine convention)
93        let proj = [
94            [f / aspect, 0.0, 0.0, 0.0],
95            [0.0, f, 0.0, 0.0],
96            [0.0, 0.0, -far / depth, -1.0],
97            [0.0, 0.0, -(far * near) / depth, 0.0],
98        ];
99
100        // View: camera at (0,0,0) looking down -Z (identity — already looking down -Z in RH)
101        let view: [[f32; 4]; 4] = [
102            [1.0, 0.0, 0.0, 0.0],
103            [0.0, 1.0, 0.0, 0.0],
104            [0.0, 0.0, 1.0, 0.0],
105            [0.0, 0.0, 0.0, 1.0],
106        ];
107
108        mat4_mul(&proj, &view)
109    }

111    /// Helper: multiply two 4×4 matrices (column-major, matching flint_core).
112    fn mat4_mul(a: &[[f32; 4]; 4], b: &[[f32; 4]; 4]) -> [[f32; 4]; 4] {
113        let mut out = [[0.0f32; 4]; 4];
114        for col in 0..4 {
115            for row in 0..4 {
116                out[col][row] = a[0][row] * b[col][0]
117                    + a[1][row] * b[col][1]
118                    + a[2][row] * b[col][2]
119                    + a[3][row] * b[col][3];
120            }
121        }
122        out
123    }
