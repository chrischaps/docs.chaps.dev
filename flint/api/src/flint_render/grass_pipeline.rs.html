101/// Number of indices in the shared blade mesh (3 quads × 4 triangles × 3 = 36)
102pub const BLADE_INDEX_COUNT: u32 = 36;

104/// Generate the shared cross-quad blade mesh.
105/// Returns (vertices, indices) for 3 intersecting quads at 60° intervals.
106/// Each quad has 7 vertices (4 segments + pointed tip).
107pub fn generate_blade_mesh() -> (Vec<GrassVertex>, Vec<u16>) {
108    let mut vertices = Vec::with_capacity(21);
109    let mut indices = Vec::with_capacity(36);
110
111    let half_w = 0.5_f32; // Normalized; scaled by blade_width in vertex shader
112
113    for quad_idx in 0..3u32 {
114        let angle = (quad_idx as f32) * std::f32::consts::PI / 3.0; // 0°, 60°, 120°
115        let cos_a = angle.cos();
116        let sin_a = angle.sin();
117
118        let base_vertex = (quad_idx * 7) as u16;
119
120        // 7 vertices per quad: 2 per segment row (4 rows) minus shared tip
121        // Row 0 (base): v=0.0
122        // Row 1: v=0.33
123        // Row 2: v=0.66
124        // Row 3 (tip): v=1.0 (single vertex)
125        let rows: [(f32, f32); 4] = [
126            (0.0, half_w), // base: full width
127            (0.33, half_w * 0.7),
128            (0.66, half_w * 0.35),
129            (1.0, 0.0), // tip: zero width (point)
130        ];
131
132        for (row_idx, &(v, hw)) in rows.iter().enumerate() {
133            if row_idx < 3 {
134                // Two vertices per row (left + right)
135                vertices.push(GrassVertex {
136                    position: [-hw * cos_a, v, -hw * sin_a],
137                    uv: [0.0, v],
138                });
139                vertices.push(GrassVertex {
140                    position: [hw * cos_a, v, hw * sin_a],
141                    uv: [1.0, v],
142                });
143            } else {
144                // Tip: single vertex
145                vertices.push(GrassVertex {
146                    position: [0.0, v, 0.0],
147                    uv: [0.5, v],
148                });
149            }
150        }
151
152        // Indices: 3 rectangular segments + 1 tip triangle = 4 triangles
153        // Segment 0: row0-row1
154        indices.push(base_vertex);
155        indices.push(base_vertex + 1);
156        indices.push(base_vertex + 2);
157        indices.push(base_vertex + 2);
158        indices.push(base_vertex + 1);
159        indices.push(base_vertex + 3);
160
161        // Segment 1: row1-row2
162        indices.push(base_vertex + 2);
163        indices.push(base_vertex + 3);
164        indices.push(base_vertex + 4);
165        indices.push(base_vertex + 4);
166        indices.push(base_vertex + 3);
167        indices.push(base_vertex + 5);
168
169        // Tip triangle: row2-tip
170        // Note: 2 triangles from the 2 row2 vertices to the single tip vertex
171        // But since tip is a point, we get a degenerate second triangle.
172        // Better: one triangle left-right-tip, skip the degenerate
173        // Actually for consistent 12 indices per quad (4 tris), use both:
174        indices.push(base_vertex + 4);
175        indices.push(base_vertex + 5);
176        indices.push(base_vertex + 6);
177        // Backface of tip (same triangle, reversed for double-sided)
178        indices.push(base_vertex + 6);
179        indices.push(base_vertex + 5);
180        indices.push(base_vertex + 4);
181    }
182
183    (vertices, indices)
184}

186/// The grass rendering pipeline (compute + render)
187pub struct GrassPipeline {
188    pub compute_pipeline: wgpu::ComputePipeline,
189    pub render_pipeline: wgpu::RenderPipeline,
190    pub shadow_pipeline: wgpu::RenderPipeline,
191    // Bind group layouts
192    pub compute_uniform_layout: wgpu::BindGroupLayout,
193    pub compute_texture_layout: wgpu::BindGroupLayout,
194    pub compute_storage_layout: wgpu::BindGroupLayout,
195    pub render_grass_layout: wgpu::BindGroupLayout,
196    pub render_instance_layout: wgpu::BindGroupLayout,
197    pub shadow_dummy_layout: wgpu::BindGroupLayout,
198    pub shadow_dummy_bind_group: wgpu::BindGroup,
199    // Shared blade mesh
200    pub blade_vertex_buffer: wgpu::Buffer,
201    pub blade_index_buffer: wgpu::Buffer,
202}
203
204impl GrassPipeline {
205    pub fn new(
206        device: &wgpu::Device,
207        scene_format: wgpu::TextureFormat,
208        transform_bind_group_layout: &wgpu::BindGroupLayout,
209        light_bind_group_layout: &wgpu::BindGroupLayout,
210        sample_count: u32,
211    ) -> Option<Self> {
212        // Compute shader
213        let compute_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
214            label: Some("Grass Compute Shader"),
215            source: wgpu::ShaderSource::Wgsl(include_str!("grass_compute.wgsl").into()),
216        });
217
218        // Render shader
219        let render_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
220            label: Some("Grass Render Shader"),
221            source: wgpu::ShaderSource::Wgsl(include_str!("grass_render.wgsl").into()),
222        });
223
224        // --- Compute bind group layouts ---
225
226        // Group 0: Compute uniforms
227        let compute_uniform_layout =
228            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
229                entries: &[wgpu::BindGroupLayoutEntry {
230                    binding: 0,
231                    visibility: wgpu::ShaderStages::COMPUTE,
232                    ty: wgpu::BindingType::Buffer {
233                        ty: wgpu::BufferBindingType::Uniform,
234                        has_dynamic_offset: false,
235                        min_binding_size: None,
236                    },
237                    count: None,
238                }],
239                label: Some("Grass Compute Uniform Layout"),
240            });
241
242        // Group 1: Heightmap + splat textures
243        let compute_texture_layout =
244            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
245                entries: &[
246                    wgpu::BindGroupLayoutEntry {
247                        binding: 0,
248                        visibility: wgpu::ShaderStages::COMPUTE,
249                        ty: wgpu::BindingType::Texture {
250                            // R32Float is not filterable without Float32Filterable feature
251                            sample_type: wgpu::TextureSampleType::Float { filterable: false },
252                            view_dimension: wgpu::TextureViewDimension::D2,
253                            multisampled: false,
254                        },
255                        count: None,
256                    },
257                    wgpu::BindGroupLayoutEntry {
258                        binding: 1,
259                        visibility: wgpu::ShaderStages::COMPUTE,
260                        ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::NonFiltering),
261                        count: None,
262                    },
263                    wgpu::BindGroupLayoutEntry {
264                        binding: 2,
265                        visibility: wgpu::ShaderStages::COMPUTE,
266                        ty: wgpu::BindingType::Texture {
267                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
268                            view_dimension: wgpu::TextureViewDimension::D2,
269                            multisampled: false,
270                        },
271                        count: None,
272                    },
273                    wgpu::BindGroupLayoutEntry {
274                        binding: 3,
275                        visibility: wgpu::ShaderStages::COMPUTE,
276                        ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
277                        count: None,
278                    },
279                ],
280                label: Some("Grass Compute Texture Layout"),
281            });
282
283        // Group 2: Instance storage buffer (read-write) + atomic counter
284        let compute_storage_layout =
285            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
286                entries: &[
287                    wgpu::BindGroupLayoutEntry {
288                        binding: 0,
289                        visibility: wgpu::ShaderStages::COMPUTE,
290                        ty: wgpu::BindingType::Buffer {
291                            ty: wgpu::BufferBindingType::Storage { read_only: false },
292                            has_dynamic_offset: false,
293                            min_binding_size: None,
294                        },
295                        count: None,
296                    },
297                    wgpu::BindGroupLayoutEntry {
298                        binding: 1,
299                        visibility: wgpu::ShaderStages::COMPUTE,
300                        ty: wgpu::BindingType::Buffer {
301                            ty: wgpu::BufferBindingType::Storage { read_only: false },
302                            has_dynamic_offset: false,
303                            min_binding_size: None,
304                        },
305                        count: None,
306                    },
307                ],
308                label: Some("Grass Compute Storage Layout"),
309            });
310
311        // Compute pipeline
312        let compute_pipeline_layout =
313            device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
314                bind_group_layouts: &[
315                    &compute_uniform_layout,
316                    &compute_texture_layout,
317                    &compute_storage_layout,
318                ],
319                push_constant_ranges: &[],
320                label: Some("Grass Compute Pipeline Layout"),
321            });
322
323        let compute_pipeline = device.create_compute_pipeline(&wgpu::ComputePipelineDescriptor {
324            label: Some("Grass Compute Pipeline"),
325            layout: Some(&compute_pipeline_layout),
326            module: &compute_shader,
327            entry_point: Some("cs_main"),
328            compilation_options: Default::default(),
329            cache: None,
330        });
331
332        // --- Render bind group layouts ---
333
334        // Group 1: Grass render uniforms
335        let render_grass_layout =
336            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
337                entries: &[wgpu::BindGroupLayoutEntry {
338                    binding: 0,
339                    visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
340                    ty: wgpu::BindingType::Buffer {
341                        ty: wgpu::BufferBindingType::Uniform,
342                        has_dynamic_offset: false,
343                        min_binding_size: None,
344                    },
345                    count: None,
346                }],
347                label: Some("Grass Render Uniform Layout"),
348            });
349
350        // Group 3: Instance buffer (read) + entity positions (read)
351        let render_instance_layout =
352            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
353                entries: &[
354                    wgpu::BindGroupLayoutEntry {
355                        binding: 0,
356                        visibility: wgpu::ShaderStages::VERTEX,
357                        ty: wgpu::BindingType::Buffer {
358                            ty: wgpu::BufferBindingType::Storage { read_only: true },
359                            has_dynamic_offset: false,
360                            min_binding_size: None,
361                        },
362                        count: None,
363                    },
364                    wgpu::BindGroupLayoutEntry {
365                        binding: 1,
366                        visibility: wgpu::ShaderStages::VERTEX,
367                        ty: wgpu::BindingType::Buffer {
368                            ty: wgpu::BufferBindingType::Storage { read_only: true },
369                            has_dynamic_offset: false,
370                            min_binding_size: None,
371                        },
372                        count: None,
373                    },
374                ],
375                label: Some("Grass Render Instance Layout"),
376            });
377
378        // Render pipeline
379        let render_pipeline_layout =
380            device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
381                bind_group_layouts: &[
382                    transform_bind_group_layout, // Group 0
383                    &render_grass_layout,        // Group 1
384                    light_bind_group_layout,     // Group 2
385                    &render_instance_layout,     // Group 3
386                ],
387                push_constant_ranges: &[],
388                label: Some("Grass Render Pipeline Layout"),
389            });
390
391        let render_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
392            label: Some("Grass Render Pipeline"),
393            layout: Some(&render_pipeline_layout),
394            vertex: wgpu::VertexState {
395                module: &render_shader,
396                entry_point: Some("vs_main"),
397                buffers: &[GrassVertex::desc()],
398                compilation_options: Default::default(),
399            },
400            fragment: Some(wgpu::FragmentState {
401                module: &render_shader,
402                entry_point: Some("fs_main"),
403                targets: &[Some(wgpu::ColorTargetState {
404                    format: scene_format,
405                    blend: None, // Opaque with alpha test (discard in shader)
406                    write_mask: wgpu::ColorWrites::ALL,
407                })],
408                compilation_options: Default::default(),
409            }),
410            primitive: wgpu::PrimitiveState {
411                topology: wgpu::PrimitiveTopology::TriangleList,
412                cull_mode: None, // Double-sided grass blades
413                ..Default::default()
414            },
415            depth_stencil: Some(wgpu::DepthStencilState {
416                format: wgpu::TextureFormat::Depth32Float,
417                depth_write_enabled: true,
418                depth_compare: wgpu::CompareFunction::Less,
419                stencil: Default::default(),
420                bias: Default::default(),
421            }),
422            multisample: wgpu::MultisampleState {
423                count: sample_count,
424                ..Default::default()
425            },
426            multiview: None,
427            cache: None,
428        });
429
430        // Shadow pipeline (depth-only, uses vs_shadow entry point)
431        // Shadow pipeline uses same 4-group layout as render so @group bindings match,
432        // but group 2 is an EMPTY placeholder instead of the light bind group.
433        // This avoids a texture usage conflict: the light bind group references the
434        // shadow depth texture as a resource, but the shadow pass writes to that same
435        // texture as a depth attachment. Using an empty group at position 2 sidesteps this.
436        let shadow_dummy_layout =
437            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
438                entries: &[],
439                label: Some("Grass Shadow Dummy Layout"),
440            });
441
442        let shadow_dummy_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
443            layout: &shadow_dummy_layout,
444            entries: &[],
445            label: Some("Grass Shadow Dummy Bind Group"),
446        });
447
448        let shadow_pipeline_layout =
449            device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
450                bind_group_layouts: &[
451                    transform_bind_group_layout, // Group 0
452                    &render_grass_layout,        // Group 1
453                    &shadow_dummy_layout, // Group 2: empty placeholder (NOT light bind group)
454                    &render_instance_layout, // Group 3
455                ],
456                push_constant_ranges: &[],
457                label: Some("Grass Shadow Pipeline Layout"),
458            });
459
460        let shadow_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
461            label: Some("Grass Shadow Pipeline"),
462            layout: Some(&shadow_pipeline_layout),
463            vertex: wgpu::VertexState {
464                module: &render_shader,
465                entry_point: Some("vs_shadow"),
466                buffers: &[GrassVertex::desc()],
467                compilation_options: Default::default(),
468            },
469            fragment: None, // Depth-only
470            primitive: wgpu::PrimitiveState {
471                topology: wgpu::PrimitiveTopology::TriangleList,
472                cull_mode: None,
473                ..Default::default()
474            },
475            depth_stencil: Some(wgpu::DepthStencilState {
476                format: wgpu::TextureFormat::Depth32Float,
477                depth_write_enabled: true,
478                depth_compare: wgpu::CompareFunction::Less,
479                stencil: Default::default(),
480                bias: wgpu::DepthBiasState {
481                    constant: 2,
482                    slope_scale: 1.5,
483                    clamp: 0.0,
484                },
485            }),
486            multisample: Default::default(),
487            multiview: None,
488            cache: None,
489        });
490
491        // Generate blade mesh
492        let (blade_verts, blade_indices) = generate_blade_mesh();
493
494        let blade_vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
495            label: Some("Grass Blade Vertex Buffer"),
496            contents: bytemuck::cast_slice(&blade_verts),
497            usage: wgpu::BufferUsages::VERTEX,
498        });
499
500        let blade_index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
501            label: Some("Grass Blade Index Buffer"),
502            contents: bytemuck::cast_slice(&blade_indices),
503            usage: wgpu::BufferUsages::INDEX,
504        });
505
506        Some(Self {
507            compute_pipeline,
508            render_pipeline,
509            shadow_pipeline,
510            compute_uniform_layout,
511            compute_texture_layout,
512            compute_storage_layout,
513            render_grass_layout,
514            render_instance_layout,
515            shadow_dummy_layout,
516            shadow_dummy_bind_group,
517            blade_vertex_buffer,
518            blade_index_buffer,
519        })
520    }
