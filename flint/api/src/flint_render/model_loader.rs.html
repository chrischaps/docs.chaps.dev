1//! Shared model and texture loading from ECS world
2//!
3//! Provides [`load_models_from_world`] and [`load_textures_from_world`] that handle
4//! the full pipeline: entity scanning, path resolution, GLB import, multi-node
5//! expansion, skinned mesh detection, and GPU upload.
6//!
7//! Callers can perform skeletal animation registration and catalog pre-resolution
8//! on top of the returned [`ModelLoadResult`].
9

17/// Metadata about a single loaded model.
18pub struct LoadedModel {
19    pub entity_id: EntityId,
20    pub asset_name: String,
21    /// The full import result — present for skinned models or models with node
22    /// animations that were freshly imported (not already cached).
23    pub import_result: Option<ImportResult>,
24    pub is_skinned: bool,
25    pub was_expanded: bool,
26    /// For animated multi-node models: maps glTF node names to child entity IDs.
27    /// Used by NodeSync to write animation transforms to the correct entities.
28    pub node_map: Option<HashMap<String, EntityId>>,
29}

38/// Configuration for model and texture path resolution.
39pub struct ModelLoadConfig {
40    pub scene_dir: PathBuf,
41    /// Pre-resolved paths for specific asset names (e.g. from catalog lookup).
42    /// Checked before the standard scene_dir/models/ search.
43    pub overrides: HashMap<String, PathBuf>,
44}

46impl ModelLoadConfig {
47    /// Create a config from a scene file path, with no overrides.
48    pub fn from_scene_path(scene_path: &str) -> Self {
49        let scene_dir = Path::new(scene_path)
50            .parent()
51            .unwrap_or_else(|| Path::new("."))
52            .to_path_buf();
53        Self {
54            scene_dir,
55            overrides: HashMap::new(),
56        }
57    }

60/// Convert a quaternion [x, y, z, w] to Euler angles (degrees) in XYZ order.
61pub fn quat_to_euler_xyz(q: [f32; 4]) -> [f32; 3] {
62    let (x, y, z, w) = (q[0], q[1], q[2], q[3]);
63    let sinr_cosp = 2.0 * (w * x + y * z);
64    let cosr_cosp = 1.0 - 2.0 * (x * x + y * y);
65    let roll = sinr_cosp.atan2(cosr_cosp);
66    let sinp = 2.0 * (w * y - z * x);
67    let pitch = if sinp.abs() >= 1.0 {
68        std::f32::consts::FRAC_PI_2.copysign(sinp)
69    } else {
70        sinp.asin()
71    };
72    let siny_cosp = 2.0 * (w * z + x * y);
73    let cosy_cosp = 1.0 - 2.0 * (y * y + z * z);
74    let yaw = siny_cosp.atan2(cosy_cosp);
75    [roll.to_degrees(), pitch.to_degrees(), yaw.to_degrees()]
76}

78/// Resolve the file path for a model asset name.
79fn resolve_model_path(config: &ModelLoadConfig, asset_name: &str) -> Option<PathBuf> {
80    if let Some(path) = config.overrides.get(asset_name) {
81        if path.exists() {
82            return Some(path.clone());
83        }
84    }
85    let p = config.scene_dir.join("models").join(format!("{}.glb", asset_name));
86    if p.exists() {
87        return Some(p);
88    }
89    if let Some(parent) = config.scene_dir.parent() {
90        let p = parent.join("models").join(format!("{}.glb", asset_name));
91        if p.exists() {
92            return Some(p);
93        }
94    }
95    None
96}

118/// Create child entities preserving the glTF parent-child hierarchy.
119///
120/// Stores each node's LOCAL TRS on the entity
121/// (no vertex baking) and sets parent to the node's actual parent entity.
122/// This allows the animation system to write to child transforms and have the
123/// renderer's `get_world_matrix()` parent chain compose them correctly.
124///
125/// Returns a map from glTF node names to their corresponding entity IDs, used
126/// by `NodeSync` to route animation tracks to the right entities.
127fn expand_nodes_animated(
128    world: &mut FlintWorld,
129    import_result: &ImportResult,
130    root_node_indices: &[usize],
131    parent_entity_id: EntityId,
132    parent_entity_name: &str,
133    asset_name: &str,
134    renderer: &mut SceneRenderer,
135    device: &wgpu::Device,
136) -> HashMap<String, EntityId> {
137    let mut node_map: HashMap<String, EntityId> = HashMap::new();
138    // Map from ImportedNode index to spawned EntityId
139    let mut index_to_entity: HashMap<usize, EntityId> = HashMap::new();
140    let default_color = [0.5_f32, 0.5, 0.5, 1.0];
141
142    // Walk the node tree recursively (DFS), creating entities for ALL nodes
143    // (not just mesh-bearing ones) to preserve the full hierarchy.
144    struct StackEntry {
145        node_idx: usize,
146        parent_ecs_id: EntityId,
147        parent_name: String,
148    }
149
150    let mut stack: Vec<StackEntry> = root_node_indices
151        .iter()
152        .rev()
153        .map(|&idx| StackEntry {
154            node_idx: idx,
155            parent_ecs_id: parent_entity_id,
156            parent_name: parent_entity_name.to_string(),
157        })
158        .collect();
159
160    while let Some(entry) = stack.pop() {
161        let node = &import_result.nodes[entry.node_idx];
162        let child_name = format!("{}__{}", entry.parent_name, node.name);
163
164        // Spawn entity for this node
165        let child_id = match world.spawn(&child_name) {
166            Ok(id) => id,
167            Err(e) => {
168                eprintln!("Failed to spawn child entity '{}': {:?}", child_name, e);
169                for &c in node.children.iter().rev() {
170                    stack.push(StackEntry {
171                        node_idx: c,
172                        parent_ecs_id: entry.parent_ecs_id,
173                        parent_name: child_name.clone(),
174                    });
175                }
176                continue;
177            }
178        };
179
180        // Set transform to node's LOCAL TRS (not baked)
181        let transform = toml::Value::Table({
182            let mut t = toml::map::Map::new();
183            t.insert(
184                "position".to_string(),
185                toml::Value::Array(vec![
186                    toml::Value::Float(node.translation[0] as f64),
187                    toml::Value::Float(node.translation[1] as f64),
188                    toml::Value::Float(node.translation[2] as f64),
189                ]),
190            );
191            // Store quaternion to preserve exact rotation (avoid Euler conversion)
192            t.insert(
193                "rotation_quat".to_string(),
194                toml::Value::Array(vec![
195                    toml::Value::Float(node.rotation[0] as f64),
196                    toml::Value::Float(node.rotation[1] as f64),
197                    toml::Value::Float(node.rotation[2] as f64),
198                    toml::Value::Float(node.rotation[3] as f64),
199                ]),
200            );
201            t.insert(
202                "scale".to_string(),
203                toml::Value::Array(vec![
204                    toml::Value::Float(node.scale[0] as f64),
205                    toml::Value::Float(node.scale[1] as f64),
206                    toml::Value::Float(node.scale[2] as f64),
207                ]),
208            );
209            t
210        });
211        let _ = world.set_component(child_id, "transform", transform);
212
213        // Upload mesh without baking (bake_transform: None)
214        if !node.mesh_primitive_indices.is_empty() {
215            let cache_key = format!("{}/{}", asset_name, node.name);
216            renderer.mesh_cache_mut().upload_mesh_subset(
217                device,
218                &cache_key,
219                import_result,
220                &node.mesh_primitive_indices,
221                default_color,
222                None, // No vertex baking — transforms live on the entity
223            );
224
225            let model = toml::Value::Table({
226                let mut m = toml::map::Map::new();
227                m.insert("asset".to_string(), toml::Value::String(cache_key));
228                m
229            });
230            let _ = world.set_component(child_id, "model", model);
231        }
232
233        // Set parent to preserve hierarchy
234        let _ = world.set_parent(child_id, entry.parent_ecs_id);
235
236        // Record in maps
237        node_map.insert(node.name.clone(), child_id);
238        index_to_entity.insert(entry.node_idx, child_id);
239
240        println!("  Expanded node: {} (animated, local TRS)", child_name);
241
242        // Push children for traversal
243        for &c in node.children.iter().rev() {
244            stack.push(StackEntry {
245                node_idx: c,
246                parent_ecs_id: child_id,
247                parent_name: child_name.clone(),
248            });
249        }
250    }
251
252    node_map
253}

255/// Load all models referenced by entities in the world.
256///
257/// Handles path resolution, GLB import, multi-node expansion, skinned mesh
258/// detection, and GPU upload. Returns metadata so callers can perform
259/// skeletal animation registration on top.
260pub fn load_models_from_world(
261    world: &mut FlintWorld,
262    renderer: &mut SceneRenderer,
263    device: &wgpu::Device,
264    queue: &wgpu::Queue,
265    config: &ModelLoadConfig,
266) -> ModelLoadResult {
267    let mut result = ModelLoadResult {
268        models: Vec::new(),
269        skinned_entities: HashMap::new(),
270    };
271
272    // Pass 1: Collect entity-model pairs (can't mutate world while iterating)
273    // Also collect whether each entity has an animator component (for animated expansion)
274    let entity_models: Vec<(EntityId, String, String, bool)> = world
275        .all_entities()
276        .iter()
277        .filter_map(|e| {
278            let components = world.get_components(e.id);
279            let model_asset = components
280                .and_then(|c| c.get("model").cloned())
281                .and_then(|model| model.get("asset").and_then(|v| v.as_str().map(String::from)));
282            let has_animator = components
283                .map(|c| c.get("animator").is_some())
284                .unwrap_or(false);
285            model_asset.map(|asset| (e.id, e.name.clone(), asset, has_animator))
286        })
287        .collect();
288
289    // Cache import results for multi-node GLBs so subsequent entities
290    // referencing the same asset can be expanded into their own children.
291    let mut expansion_cache: HashMap<String, ImportResult> = HashMap::new();
292
293    // Pass 2: Load and expand
294    for (entity_id, entity_name, asset_name, has_animator) in &entity_models {
295        if renderer.mesh_cache().contains(asset_name) {
296            // If this asset was previously expanded (same load call), expand for this entity too
297            if let Some(cached_import) = expansion_cache.get(asset_name.as_str()) {
298                let nmap = expand_nodes_animated(
299                    world,
300                    cached_import,
301                    &cached_import.root_nodes,
302                    *entity_id,
303                    entity_name,
304                    asset_name,
305                    renderer,
306                    device,
307                );
308                if let Some(components) = world.get_components_mut(*entity_id) {
309                    components.remove("model");
310                }
311                result.models.push(LoadedModel {
312                    entity_id: *entity_id,
313                    asset_name: asset_name.clone(),
314                    import_result: None,
315                    is_skinned: false,
316                    was_expanded: true,
317                    node_map: Some(nmap),
318                });
319                continue;
320            }
321
322            // Mesh cached from a previous scene/load — re-import to check
323            // if this is a multi-node model that needs per-entity expansion.
324            if let Some(model_path) = resolve_model_path(config, asset_name) {
325                if let Ok(import_result) = import_gltf(&model_path) {
326                    if import_result.needs_expansion() {
327                        let nmap = expand_nodes_animated(
328                            world,
329                            &import_result,
330                            &import_result.root_nodes,
331                            *entity_id,
332                            entity_name,
333                            asset_name,
334                            renderer,
335                            device,
336                        );
337                        if let Some(components) = world.get_components_mut(*entity_id) {
338                            components.remove("model");
339                        }
340                        result.models.push(LoadedModel {
341                            entity_id: *entity_id,
342                            asset_name: asset_name.clone(),
343                            import_result: None,
344                            is_skinned: false,
345                            was_expanded: true,
346                            node_map: Some(nmap),
347                        });
348                        expansion_cache.insert(asset_name.clone(), import_result);
349                        continue;
350                    }
351                }
352            }
353
354            if renderer.mesh_cache().contains_skinned(asset_name) {
355                result.skinned_entities.insert(*entity_id, asset_name.clone());
356                result.models.push(LoadedModel {
357                    entity_id: *entity_id,
358                    asset_name: asset_name.clone(),
359                    import_result: None,
360                    is_skinned: true,
361                    was_expanded: false,
362                    node_map: None,
363                });
364            }
365            continue;
366        }
367
368        let model_path = match resolve_model_path(config, asset_name) {
369            Some(p) => p,
370            None => {
371                eprintln!("Model file not found: {}", asset_name);
372                continue;
373            }
374        };
375
376        match import_gltf(&model_path) {
377            Ok(import_result) => {
378                let has_skins = !import_result.skeletons.is_empty();
379                let has_skinned_meshes = import_result
380                    .meshes
381                    .iter()
382                    .any(|m| m.joint_indices.is_some());
383                let is_skinned = has_skinned_meshes && has_skins;
384
385                let bounds_info = import_result
386                    .bounds()
387                    .map(|b| format!(", bounds: {}", b))
388                    .unwrap_or_default();
389
390                println!(
391                    "Loaded model: {} ({} meshes, {} nodes{}{})",
392                    asset_name,
393                    import_result.meshes.len(),
394                    import_result.nodes.len(),
395                    if has_skins {
396                        format!(
397                            ", {} skins, {} skeletal clips",
398                            import_result.skeletons.len(),
399                            import_result.skeletal_clips.len()
400                        )
401                    } else {
402                        String::new()
403                    },
404                    bounds_info,
405                );
406
407                let was_expanded;
408                let kept_result;
409                let node_map;
410
411                if is_skinned {
412                    renderer.load_skinned_model(device, queue, asset_name, &import_result);
413                    renderer.load_model(device, queue, asset_name, &import_result);
414                    result.skinned_entities.insert(*entity_id, asset_name.clone());
415                    was_expanded = false;
416                    kept_result = Some(import_result);
417                    node_map = None;
418                } else if import_result.needs_expansion() {
419                    // Use the whole-model upload for any code paths that look
420                    // up the asset by its base name (e.g. bounds queries).
421                    renderer.load_model(device, queue, asset_name, &import_result);
422
423                    // Always use hierarchy-preserving expansion: keep local TRS
424                    // on each child entity so (a) the scene tree shows the real
425                    // glTF node hierarchy, and (b) animation can write to them.
426                    let has_animations = import_result.has_node_animations() && *has_animator;
427                    let nmap = expand_nodes_animated(
428                        world,
429                        &import_result,
430                        &import_result.root_nodes,
431                        *entity_id,
432                        entity_name,
433                        asset_name,
434                        renderer,
435                        device,
436                    );
437
438                    if let Some(components) = world.get_components_mut(*entity_id) {
439                        components.remove("model");
440                    }
441
442                    was_expanded = true;
443                    if has_animations {
444                        // Animated models need the import result for NodeSync
445                        // registration; also cache for subsequent entities.
446                        kept_result = Some(import_result);
447                    } else {
448                        // Cache for subsequent entities referencing the same asset
449                        expansion_cache.insert(asset_name.clone(), import_result);
450                        kept_result = None;
451                    }
452                    node_map = Some(nmap);
453                } else {
454                    renderer.load_model(device, queue, asset_name, &import_result);
455                    was_expanded = false;
456                    kept_result = None;
457                    node_map = None;
458                }
459
460                result.models.push(LoadedModel {
461                    entity_id: *entity_id,
462                    asset_name: asset_name.clone(),
463                    import_result: kept_result,
464                    is_skinned,
465                    was_expanded,
466                    node_map,
467                });
468            }
469            Err(e) => {
470                eprintln!("Failed to load model '{}': {:?}", asset_name, e);
471            }
472        }
473    }
474
475    // Load textures
476    load_textures_from_world(world, renderer, device, queue, config);
477
478    result
479}

481/// Load texture files referenced by material and sprite components.
482pub fn load_textures_from_world(
483    world: &FlintWorld,
484    renderer: &mut SceneRenderer,
485    device: &wgpu::Device,
486    queue: &wgpu::Queue,
487    config: &ModelLoadConfig,
488) {
489    let mut loaded = std::collections::HashSet::new();
490
491    for entity in world.all_entities() {
492        let components = world.get_components(entity.id);
493        let mut tex_names = Vec::new();
494
495        if let Some(comps) = &components {
496            if let Some(material) = comps.get("material") {
497                if let Some(tex) = material.get("texture").and_then(|v| v.as_str()) {
498                    tex_names.push(tex.to_string());
499                }
500            }
501            if let Some(sprite) = comps.get("sprite") {
502                if let Some(tex) = sprite.get("texture").and_then(|v| v.as_str()) {
503                    if !tex.is_empty() {
504                        tex_names.push(tex.to_string());
505                    }
506                }
507            }
508        }
509
510        for tex_name in tex_names {
511            if loaded.contains(&tex_name) {
512                continue;
513            }
514            loaded.insert(tex_name.clone());
515
516            let tex_path = match resolve_texture_path(config, &tex_name) {
517                Some(p) => p,
518                None => {
519                    eprintln!("Texture file not found: {}", tex_name);
520                    continue;
521                }
522            };
523
524            match renderer.load_texture_file(device, queue, &tex_name, &tex_path) {
525                Ok(true) => {
526                    println!("Loaded texture: {}", tex_name);
527                }
528                Ok(false) => {}
529                Err(e) => {
530                    eprintln!("Failed to load texture '{}': {}", tex_name, e);
531                }
532            }
533        }
534    }
