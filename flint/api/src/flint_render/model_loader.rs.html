1//! Shared model and texture loading from ECS world
2//!
3//! Provides [`load_models_from_world`] and [`load_textures_from_world`] that handle
4//! the full pipeline: entity scanning, path resolution, GLB import, multi-node
5//! expansion, skinned mesh detection, and GPU upload.
6//!
7//! Callers can perform skeletal animation registration and catalog pre-resolution
8//! on top of the returned [`ModelLoadResult`].
9

18/// Metadata about a single loaded model.
19pub struct LoadedModel {
20    pub entity_id: EntityId,
21    pub asset_name: String,
22    /// The full import result — present for skinned models or models with node
23    /// animations that were freshly imported (not already cached).
24    pub import_result: Option<ImportResult>,
25    pub is_skinned: bool,
26    pub was_expanded: bool,
27    /// For animated multi-node models: maps glTF node names to child entity IDs.
28    /// Used by NodeSync to write animation transforms to the correct entities.
29    pub node_map: Option<HashMap<String, EntityId>>,
30}

39/// Configuration for model and texture path resolution.
40pub struct ModelLoadConfig {
41    pub scene_dir: PathBuf,
42    /// Pre-resolved paths for specific asset names (e.g. from catalog lookup).
43    /// Checked before the standard scene_dir/models/ search.
44    pub overrides: HashMap<String, PathBuf>,
45}

47impl ModelLoadConfig {
48    /// Create a config from a scene file path, with no overrides.
49    pub fn from_scene_path(scene_path: &str) -> Self {
50        let scene_dir = Path::new(scene_path)
51            .parent()
52            .unwrap_or_else(|| Path::new("."))
53            .to_path_buf();
54        Self {
55            scene_dir,
56            overrides: HashMap::new(),
57        }

61/// Convert a quaternion [x, y, z, w] to Euler angles (degrees) in XYZ order.
62pub fn quat_to_euler_xyz(q: [f32; 4]) -> [f32; 3] {
63    let (x, y, z, w) = (q[0], q[1], q[2], q[3]);
64    let sinr_cosp = 2.0 * (w * x + y * z);
65    let cosr_cosp = 1.0 - 2.0 * (x * x + y * y);
66    let roll = sinr_cosp.atan2(cosr_cosp);
67    let sinp = 2.0 * (w * y - z * x);
68    let pitch = if sinp.abs() >= 1.0 {
69        std::f32::consts::FRAC_PI_2.copysign(sinp)
70    } else {
71        sinp.asin()
72    };
73    let siny_cosp = 2.0 * (w * z + x * y);
74    let cosy_cosp = 1.0 - 2.0 * (y * y + z * z);
75    let yaw = siny_cosp.atan2(cosy_cosp);
76    [roll.to_degrees(), pitch.to_degrees(), yaw.to_degrees()]
77}

79/// Resolve the file path for a model asset name.
80fn resolve_model_path(config: &ModelLoadConfig, asset_name: &str) -> Option<PathBuf> {
81    if let Some(path) = config.overrides.get(asset_name) {
82        if path.exists() {
83            return Some(path.clone());
84        }
85    }
86    // Try both .glb and .gltf in each search directory
87    for ext in &["glb", "gltf"] {
88        let p = config
89            .scene_dir
90            .join("models")
91            .join(format!("{}.{}", asset_name, ext));
92        if p.exists() {
93            return Some(p);
94        }
95    }
96    if let Some(parent) = config.scene_dir.parent() {
97        for ext in &["glb", "gltf"] {
98            let p = parent
99                .join("models")
100                .join(format!("{}.{}", asset_name, ext));
101            if p.exists() {
102                return Some(p);
103            }
104        }
105    }
106    None
107}

109/// Resolve the file path for a texture name.
110fn resolve_texture_path(config: &ModelLoadConfig, tex_name: &str) -> Option<PathBuf> {
111    if let Some(path) = config.overrides.get(tex_name) {
112        if path.exists() {
113            return Some(path.clone());
114        }
115    }
116
117    // Candidate names: the bare name, then with common extensions (for procgen textures)
118    let candidates: Vec<String> = if tex_name.contains('.') {
119        vec![tex_name.to_string()]
120    } else {
121        vec![
122            tex_name.to_string(),
123            format!("{}.png", tex_name),
124            format!("{}_color.png", tex_name),
125        ]
126    };
127
128    // Search directories: scene_dir, scene_dir/textures, parent, parent/textures
129    let mut search_dirs: Vec<PathBuf> = vec![config.scene_dir.clone()];
130    search_dirs.push(config.scene_dir.join("textures"));
131    if let Some(parent) = config.scene_dir.parent() {
132        search_dirs.push(parent.to_path_buf());
133        search_dirs.push(parent.join("textures"));
134    }
135
136    for dir in &search_dirs {
137        for name in &candidates {
138            let p = dir.join(name);
139            if p.exists() {
140                return Some(p);
141            }
142        }
143    }
144
145    None
146}

148/// Create child entities preserving the glTF parent-child hierarchy.
149///
150/// Unlike `expand_nodes_flat()`, this stores each node's LOCAL TRS on the entity
151/// (no vertex baking) and sets parent to the node's actual parent entity.
152/// This allows the animation system to write to child transforms and have the
153/// renderer's `get_world_matrix()` parent chain compose them correctly.
154///
155/// Returns a map from glTF node names to their corresponding entity IDs, used
156/// by `NodeSync` to route animation tracks to the right entities.
157fn expand_nodes_animated(
158    world: &mut FlintWorld,
159    import_result: &ImportResult,
160    root_node_indices: &[usize],
161    parent_entity_id: EntityId,
162    parent_entity_name: &str,
163    asset_name: &str,
164    renderer: &mut SceneRenderer,
165    device: &wgpu::Device,
166) -> HashMap<String, EntityId> {
167    let mut node_map: HashMap<String, EntityId> = HashMap::new();
168    // Map from ImportedNode index to spawned EntityId
169    let mut index_to_entity: HashMap<usize, EntityId> = HashMap::new();
170    let default_color = [0.5_f32, 0.5, 0.5, 1.0];
171
172    // Walk the node tree recursively (DFS), creating entities for ALL nodes
173    // (not just mesh-bearing ones) to preserve the full hierarchy.
174    struct StackEntry {
175        node_idx: usize,
176        parent_ecs_id: EntityId,
177        parent_name: String,
178    }
179
180    let mut stack: Vec<StackEntry> = root_node_indices
181        .iter()
182        .rev()
183        .map(|&idx| StackEntry {
184            node_idx: idx,
185            parent_ecs_id: parent_entity_id,
186            parent_name: parent_entity_name.to_string(),
187        })
188        .collect();
189
190    while let Some(entry) = stack.pop() {
191        let node = &import_result.nodes[entry.node_idx];
192        let child_name = format!("{}__{}", entry.parent_name, node.name);
193
194        // Spawn entity for this node
195        let child_id = match world.spawn(&child_name) {
196            Ok(id) => id,
197            Err(e) => {
198                tracing::warn!("Failed to spawn child entity '{}': {:?}", child_name, e);
199                for &c in node.children.iter().rev() {
200                    stack.push(StackEntry {
201                        node_idx: c,
202                        parent_ecs_id: entry.parent_ecs_id,
203                        parent_name: child_name.clone(),
204                    });
205                }
206                continue;
207            }
208        };
209
210        // Set transform to node's LOCAL TRS (not baked)
211        let transform = toml::Value::Table({
212            let mut t = toml::map::Map::new();
213            t.insert(
214                "position".to_string(),
215                toml::Value::Array(vec![
216                    toml::Value::Float(node.translation[0] as f64),
217                    toml::Value::Float(node.translation[1] as f64),
218                    toml::Value::Float(node.translation[2] as f64),
219                ]),
220            );
221            // Store quaternion to preserve exact rotation (avoid Euler conversion)
222            t.insert(
223                "rotation_quat".to_string(),
224                toml::Value::Array(vec![
225                    toml::Value::Float(node.rotation[0] as f64),
226                    toml::Value::Float(node.rotation[1] as f64),
227                    toml::Value::Float(node.rotation[2] as f64),
228                    toml::Value::Float(node.rotation[3] as f64),
229                ]),
230            );
231            t.insert(
232                "scale".to_string(),
233                toml::Value::Array(vec![
234                    toml::Value::Float(node.scale[0] as f64),
235                    toml::Value::Float(node.scale[1] as f64),
236                    toml::Value::Float(node.scale[2] as f64),
237                ]),
238            );
239            t
240        });
241        let _ = world.set_component(child_id, comp::TRANSFORM, transform);
242
243        // Upload mesh without baking (bake_transform: None)
244        if !node.mesh_primitive_indices.is_empty() {
245            let cache_key = format!("{}/{}", asset_name, node.name);
246            renderer.mesh_cache_mut().upload_mesh_subset(
247                device,
248                &cache_key,
249                import_result,
250                &node.mesh_primitive_indices,
251                default_color,
252                None, // No vertex baking — transforms live on the entity
253            );
254
255            let model = toml::Value::Table({
256                let mut m = toml::map::Map::new();
257                m.insert("asset".to_string(), toml::Value::String(cache_key));
258                m
259            });
260            let _ = world.set_component(child_id, comp::MODEL, model);
261        }
262
263        // Set parent to preserve hierarchy
264        let _ = world.set_parent(child_id, entry.parent_ecs_id);
265
266        // Record in maps
267        node_map.insert(node.name.clone(), child_id);
268        index_to_entity.insert(entry.node_idx, child_id);
269
270        println!("  Expanded node: {} (animated, local TRS)", child_name);
271
272        // Push children for traversal
273        for &c in node.children.iter().rev() {
274            stack.push(StackEntry {
275                node_idx: c,
276                parent_ecs_id: child_id,
277                parent_name: child_name.clone(),
278            });
279        }
280    }
281
282    node_map
283}

285/// Load all models referenced by entities in the world.
286///
287/// Handles path resolution, GLB import, multi-node expansion, skinned mesh
288/// detection, and GPU upload. Returns metadata so callers can perform
289/// skeletal animation registration on top.
290pub fn load_models_from_world(
291    world: &mut FlintWorld,
292    renderer: &mut SceneRenderer,
293    device: &wgpu::Device,
294    queue: &wgpu::Queue,
295    config: &ModelLoadConfig,
296) -> ModelLoadResult {
297    let mut result = ModelLoadResult {
298        models: Vec::new(),
299        skinned_entities: HashMap::new(),
300    };
301
302    // Pass 1: Collect entity-model pairs (can't mutate world while iterating)
303    // Also collect whether each entity has an animator component (for animated expansion)
304    let entity_models: Vec<(EntityId, String, String, bool)> = world
305        .all_entities()
306        .iter()
307        .filter_map(|e| {
308            let components = world.get_components(e.id);
309            let model_asset = components
310                .and_then(|c| c.get(comp::MODEL).cloned())
311                .and_then(|model| {
312                    model
313                        .get("asset")
314                        .and_then(|v| v.as_str().map(String::from))
315                });
316            let has_animator = components
317                .map(|c| c.get(comp::ANIMATOR).is_some())
318                .unwrap_or(false);
319            model_asset.map(|asset| (e.id, e.name.clone(), asset, has_animator))
320        })
321        .collect();
322
323    // Cache import results for multi-node GLBs so subsequent entities
324    // referencing the same asset can be expanded into their own children.
325    let mut expansion_cache: HashMap<String, ImportResult> = HashMap::new();
326
327    // Pass 2: Load and expand
328    for (entity_id, entity_name, asset_name, has_animator) in &entity_models {
329        if renderer.mesh_cache().contains(asset_name) {
330            // If this asset was previously expanded (same load call), expand for this entity too
331            if let Some(cached_import) = expansion_cache.get(asset_name.as_str()) {
332                let nmap = expand_nodes_animated(
333                    world,
334                    cached_import,
335                    &cached_import.root_nodes,
336                    *entity_id,
337                    entity_name,
338                    asset_name,
339                    renderer,
340                    device,
341                );
342                if let Some(components) = world.get_components_mut(*entity_id) {
343                    components.remove(comp::MODEL);
344                }
345                result.models.push(LoadedModel {
346                    entity_id: *entity_id,
347                    asset_name: asset_name.clone(),
348                    import_result: None,
349                    is_skinned: false,
350                    was_expanded: true,
351                    node_map: Some(nmap),
352                });
353                continue;
354            }
355
356            // Mesh cached from a previous scene/load — re-import to check
357            // if this is a multi-node model that needs per-entity expansion.
358            if let Some(model_path) = resolve_model_path(config, asset_name) {
359                if let Ok(import_result) = import_gltf(&model_path) {
360                    if import_result.needs_expansion() {
361                        let nmap = expand_nodes_animated(
362                            world,
363                            &import_result,
364                            &import_result.root_nodes,
365                            *entity_id,
366                            entity_name,
367                            asset_name,
368                            renderer,
369                            device,
370                        );
371                        if let Some(components) = world.get_components_mut(*entity_id) {
372                            components.remove(comp::MODEL);
373                        }
374                        result.models.push(LoadedModel {
375                            entity_id: *entity_id,
376                            asset_name: asset_name.clone(),
377                            import_result: None,
378                            is_skinned: false,
379                            was_expanded: true,
380                            node_map: Some(nmap),
381                        });
382                        expansion_cache.insert(asset_name.clone(), import_result);
383                        continue;
384                    }
385                }
386            }
387
388            if renderer.mesh_cache().contains_skinned(asset_name) {
389                result
390                    .skinned_entities
391                    .insert(*entity_id, asset_name.clone());
392                result.models.push(LoadedModel {
393                    entity_id: *entity_id,
394                    asset_name: asset_name.clone(),
395                    import_result: None,
396                    is_skinned: true,
397                    was_expanded: false,
398                    node_map: None,
399                });
400            }
401            continue;
402        }
403
404        let model_path = match resolve_model_path(config, asset_name) {
405            Some(p) => p,
406            None => {
407                tracing::warn!("Model file not found: {}", asset_name);
408                continue;
409            }
410        };
411
412        match import_gltf(&model_path) {
413            Ok(import_result) => {
414                let has_skins = !import_result.skeletons.is_empty();
415                let has_skinned_meshes = import_result
416                    .meshes
417                    .iter()
418                    .any(|m| m.joint_indices.is_some());
419                let is_skinned = has_skinned_meshes && has_skins;
420
421                let bounds_info = import_result
422                    .bounds()
423                    .map(|b| format!(", bounds: {}", b))
424                    .unwrap_or_default();
425
426                println!(
427                    "Loaded model: {} ({} meshes, {} nodes{}{})",
428                    asset_name,
429                    import_result.meshes.len(),
430                    import_result.nodes.len(),
431                    if has_skins {
432                        format!(
433                            ", {} skins, {} skeletal clips",
434                            import_result.skeletons.len(),
435                            import_result.skeletal_clips.len()
436                        )
437                    } else {
438                        String::new()
439                    },
440                    bounds_info,
441                );
442
443                let was_expanded;
444                let kept_result;
445                let node_map;
446
447                if is_skinned {
448                    renderer.load_skinned_model(device, queue, asset_name, &import_result);
449                    renderer.load_model(device, queue, asset_name, &import_result);
450                    result
451                        .skinned_entities
452                        .insert(*entity_id, asset_name.clone());
453                    was_expanded = false;
454                    kept_result = Some(import_result);
455                    node_map = None;
456                } else if import_result.needs_expansion() {
457                    // Use the whole-model upload for any code paths that look
458                    // up the asset by its base name (e.g. bounds queries).
459                    renderer.load_model(device, queue, asset_name, &import_result);
460
461                    // Always use hierarchy-preserving expansion: keep local TRS
462                    // on each child entity so (a) the scene tree shows the real
463                    // glTF node hierarchy, and (b) animation can write to them.
464                    let has_animations = import_result.has_node_animations() && *has_animator;
465                    let nmap = expand_nodes_animated(
466                        world,
467                        &import_result,
468                        &import_result.root_nodes,
469                        *entity_id,
470                        entity_name,
471                        asset_name,
472                        renderer,
473                        device,
474                    );
475
476                    if let Some(components) = world.get_components_mut(*entity_id) {
477                        components.remove(comp::MODEL);
478                    }
479
480                    was_expanded = true;
481                    if has_animations {
482                        // Animated models need the import result for NodeSync
483                        // registration; also cache for subsequent entities.
484                        kept_result = Some(import_result);
485                    } else {
486                        // Cache for subsequent entities referencing the same asset
487                        expansion_cache.insert(asset_name.clone(), import_result);
488                        kept_result = None;
489                    }
490                    node_map = Some(nmap);
491                } else {
492                    renderer.load_model(device, queue, asset_name, &import_result);
493                    was_expanded = false;
494                    kept_result = None;
495                    node_map = None;
496                }
497
498                result.models.push(LoadedModel {
499                    entity_id: *entity_id,
500                    asset_name: asset_name.clone(),
501                    import_result: kept_result,
502                    is_skinned,
503                    was_expanded,
504                    node_map,
505                });
506            }
507            Err(e) => {
508                tracing::warn!("Failed to load model '{}': {:?}", asset_name, e);
509            }
510        }
511    }
512
513    // Load textures
514    load_textures_from_world(world, renderer, device, queue, config);
515
516    result
517}

519/// Load texture files referenced by material and sprite components.
520pub fn load_textures_from_world(
521    world: &FlintWorld,
522    renderer: &mut SceneRenderer,
523    device: &wgpu::Device,
524    queue: &wgpu::Queue,
525    config: &ModelLoadConfig,
526) {
527    let mut loaded = std::collections::HashSet::new();
528
529    for entity in world.all_entities() {
530        let components = world.get_components(entity.id);
531        let mut tex_names = Vec::new();
532
533        if let Some(comps) = &components {
534            if let Some(material) = comps.get(comp::MATERIAL) {
535                if let Some(tex) = material.get("texture").and_then(|v| v.as_str()) {
536                    tex_names.push(tex.to_string());
537                }
538            }
539            if let Some(sprite) = comps.get(comp::SPRITE) {
540                if let Some(tex) = sprite.get("texture").and_then(|v| v.as_str()) {
541                    if !tex.is_empty() {
542                        tex_names.push(tex.to_string());
543                    }
544                }
545            }
546            // Load font textures referenced by ui_text components
547            if let Some(ui_text) = comps.get(comp::UI_TEXT) {
548                if let Some(font_path_str) = ui_text.get("font").and_then(|v| v.as_str()) {
549                    if !font_path_str.is_empty() {
550                        let font_file = config.scene_dir.join(font_path_str);
551                        if let Some(font) = crate::bitmap_font::BitmapFont::load(&font_file) {
552                            if !font.texture.is_empty() {
553                                tex_names.push(font.texture);
554                            }
555                        }
556                    }
557                }
558            }
559        }
560
561        for tex_name in tex_names {
562            if loaded.contains(&tex_name) {
563                continue;
564            }
565            loaded.insert(tex_name.clone());
566
567            let tex_path = match resolve_texture_path(config, &tex_name) {
568                Some(p) => p,
569                None => {
570                    tracing::warn!("Texture file not found: {}", tex_name);
571                    continue;
572                }
573            };
574
575            match renderer.load_texture_file(device, queue, &tex_name, &tex_path) {
576                Ok(true) => {
577                    println!("Loaded texture: {}", tex_name);
578                }
579                Ok(false) => {}
580                Err(e) => {
581                    tracing::warn!("Failed to load texture '{}': {}", tex_name, e);
582                }
583            }
584
585            // Try loading companion PBR maps (_normal, _roughness) alongside the color texture
586            for suffix in &["_normal", "_roughness"] {
587                let companion_name = format!("{}{}", tex_name, suffix);
588                if loaded.contains(&companion_name) {
589                    continue;
590                }
591                if let Some(companion_path) = resolve_texture_path(config, &companion_name) {
592                    loaded.insert(companion_name.clone());
593                    match renderer.load_texture_file(device, queue, &companion_name, &companion_path) {
594                        Ok(true) => {
595                            println!("Loaded texture: {}", companion_name);
596                        }
597                        Ok(false) => {}
598                        Err(e) => {
599                            tracing::warn!("Failed to load texture '{}': {}", companion_name, e);
600                        }
601                    }
602                }
603            }
604        }
605    }
