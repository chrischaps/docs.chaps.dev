1//! Bitmap font loader and text layout for GPU sprite rendering.
2//!
3//! Grid-based ASCII font atlas: a `.font.toml` file defines the texture,
4//! glyph grid dimensions, and character range. Each character maps to one
5//! [`Sprite2dInstanceGpu`] so entire strings batch into the existing sprite2d
6//! pipeline with zero new shaders.
7
8use crate::sprite2d_pipeline::Sprite2dInstanceGpu;
9use flint_core::toml_util::toml_f32;
10use std::collections::HashMap;
11use std::path::Path;
12
13/// Parsed bitmap font definition.
14#[derive(Clone, Debug)]
15pub struct BitmapFont {
16    /// Texture name (used as key in TextureCache)
17    pub texture: String,
18    /// Glyph cell width in pixels
19    pub glyph_width: u32,
20    /// Glyph cell height in pixels
21    pub glyph_height: u32,
22    /// Number of columns in the atlas grid
23    pub columns: u32,
24    /// ASCII code of the first glyph in the atlas
25    pub first_char: u32,
26    /// Number of glyphs in the atlas
27    pub char_count: u32,
28    /// Horizontal spacing multiplier (1.0 = no extra space)
29    pub spacing: f32,
30    /// Per-character width overrides (glyph_char -> width in pixels), for variable-width
31    pub char_widths: HashMap<char, u32>,
32}
33
34impl BitmapFont {
35    /// Parse a `.font.toml` file. Returns `None` on parse failure.
36    pub fn load(path: &Path) -> Option<Self> {
37        let content = std::fs::read_to_string(path).ok()?;
38        let table: toml::Value = content.parse().ok()?;
39        let font = table.get("font")?;
40
41        let texture = font.get("texture")?.as_str()?.to_string();
42        let glyph_width = font.get("glyph_width")?.as_integer()? as u32;
43        let glyph_height = font.get("glyph_height")?.as_integer()? as u32;
44        let columns = font.get("columns")?.as_integer()? as u32;
45        let first_char = font
46            .get("first_char")
47            .and_then(|v| v.as_integer())
48            .unwrap_or(32) as u32;
49        let char_count = font
50            .get("char_count")
51            .and_then(|v| v.as_integer())
52            .unwrap_or(95) as u32;
53        let spacing = font.get("spacing").and_then(toml_f32).unwrap_or(1.0);
54
55        Some(Self {
56            texture,
57            glyph_width,
58            glyph_height,
59            columns,
60            first_char,
61            char_count,
62            spacing,
63            char_widths: HashMap::new(),
64        })
65    }
66
67    /// UV rect `[u_min, v_min, u_max, v_max]` for a character, normalized 0–1.
68    /// `tex_w`/`tex_h` are the texture dimensions in pixels.
69    pub fn glyph_uv(&self, ch: char, tex_w: u32, tex_h: u32) -> Option<[f32; 4]> {
70        let code = ch as u32;
71        if code < self.first_char || code >= self.first_char + self.char_count {
72            return None;
73        }
74        let index = code - self.first_char;
75        let col = index % self.columns;
76        let row = index / self.columns;
77
78        let u_min = (col * self.glyph_width) as f32 / tex_w as f32;
79        let v_min = (row * self.glyph_height) as f32 / tex_h as f32;
80        let u_max = ((col + 1) * self.glyph_width) as f32 / tex_w as f32;
81        let v_max = ((row + 1) * self.glyph_height) as f32 / tex_h as f32;
82
83        Some([u_min, v_min, u_max, v_max])
84    }
85
86    /// World-unit width of a single glyph at the given scale.
87    fn glyph_advance(&self, scale: f32) -> f32 {
88        let aspect = self.glyph_width as f32 / self.glyph_height as f32;
89        scale * aspect * self.spacing
90    }
91
92    /// Expand a text string into per-glyph sprite instances.
93    ///
94    /// * `x, y` — baseline-left origin in world coordinates
95    /// * `scale` — text height in world units
96    /// * `tint` — RGBA colour
97    /// * `layer` — sprite layer for z-ordering
98    /// * `align` — "left", "center", or "right"
99    /// * `tex_w, tex_h` — font texture dimensions in pixels
100    ///
101    /// Returns `Vec<(texture_name, layer, instance)>` ready to push into
102    /// `sprite2d_collected`.
103    pub fn layout_text(
104        &self,
105        text: &str,
106        x: f32,
107        y: f32,
108        scale: f32,
109        tint: [f32; 4],
110        layer: i32,
111        align: &str,
112        tex_w: u32,
113        tex_h: u32,
114    ) -> Vec<(String, i32, Sprite2dInstanceGpu)> {
115        let advance = self.glyph_advance(scale);
116        let total_width = text.len() as f32 * advance;
117
118        let start_x = match align {
119            "center" => x - total_width * 0.5,
120            "right" => x - total_width,
121            _ => x, // "left" or default
122        };
123
124        let glyph_w = scale * (self.glyph_width as f32 / self.glyph_height as f32);
125        let glyph_h = scale;
126
127        let mut result = Vec::with_capacity(text.len());
128        let mut cursor_x = start_x;
129
130        for ch in text.chars() {
131            if let Some(uv_rect) = self.glyph_uv(ch, tex_w, tex_h) {
132                let instance = Sprite2dInstanceGpu {
133                    pos_layer: [cursor_x + glyph_w * 0.5, y, 0.5, layer as f32 * 0.01],
134                    size: [glyph_w, glyph_h, 0.0, 0.0],
135                    uv_rect,
136                    tint,
137                    flags: [0.0, 0.0, 0.0, 0.0],
138                };
139                result.push((self.texture.clone(), layer, instance));
140            }
141            cursor_x += advance;
142        }
143
144        result
145    }
146}

258/// Compute the screen-space origin for a named anchor point.
259///
260/// * `half_w` = `ortho_height * aspect_ratio / 2.0`
261/// * `half_h` = `ortho_height / 2.0`
262///
263/// Returns `(ax, ay)` relative to camera center.
264pub fn anchor_origin(anchor: &str, half_w: f32, half_h: f32) -> (f32, f32) {
265    match anchor {
266        "top-left" => (-half_w, half_h),
267        "top-center" => (0.0, half_h),
268        "top-right" => (half_w, half_h),
269        "center-left" => (-half_w, 0.0),
270        "center" => (0.0, 0.0),
271        "center-right" => (half_w, 0.0),
272        "bottom-left" => (-half_w, -half_h),
273        "bottom-center" => (0.0, -half_h),
274        "bottom-right" => (half_w, -half_h),
275        _ => (0.0, 0.0), // default to center
276    }
277}

279/// Apply ui_fill clipping to a sprite's UV rect and dimensions.
280///
281/// Returns `(new_uv_rect, new_width, new_height, x_offset, y_offset)`.
282/// The offsets compensate position so the visible portion stays aligned.
283pub fn apply_fill(
284    uv_rect: [f32; 4],
285    width: f32,
286    height: f32,
287    value: f32,
288    direction: &str,
289) -> ([f32; 4], f32, f32, f32, f32) {
290    let value = value.clamp(0.0, 1.0);
291
292    match direction {
293        "left_to_right" => {
294            let new_w = width * value;
295            let u_range = uv_rect[2] - uv_rect[0];
296            let new_u_max = uv_rect[0] + u_range * value;
297            let x_off = -(width - new_w) * 0.5;
298            (
299                [uv_rect[0], uv_rect[1], new_u_max, uv_rect[3]],
300                new_w,
301                height,
302                x_off,
303                0.0,
304            )
305        }
306        "right_to_left" => {
307            let new_w = width * value;
308            let u_range = uv_rect[2] - uv_rect[0];
309            let new_u_min = uv_rect[2] - u_range * value;
310            let x_off = (width - new_w) * 0.5;
311            (
312                [new_u_min, uv_rect[1], uv_rect[2], uv_rect[3]],
313                new_w,
314                height,
315                x_off,
316                0.0,
317            )
318        }
319        "bottom_to_top" => {
320            let new_h = height * value;
321            let v_range = uv_rect[3] - uv_rect[1];
322            let new_v_min = uv_rect[3] - v_range * value;
323            let y_off = -(height - new_h) * 0.5;
324            (
325                [uv_rect[0], new_v_min, uv_rect[2], uv_rect[3]],
326                width,
327                new_h,
328                0.0,
329                y_off,
330            )
331        }
332        "top_to_bottom" => {
333            let new_h = height * value;
334            let v_range = uv_rect[3] - uv_rect[1];
335            let new_v_max = uv_rect[1] + v_range * value;
336            let y_off = (height - new_h) * 0.5;
337            (
338                [uv_rect[0], uv_rect[1], uv_rect[2], new_v_max],
339                width,
340                new_h,
341                0.0,
342                y_off,
343            )
344        }
345        _ => (uv_rect, width, height, 0.0, 0.0),
346    }
347}
