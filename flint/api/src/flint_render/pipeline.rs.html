5/// Depth bias for the line-list overlays (wireframe, skeleton), pulling
6/// them in front of the surface they trace. WebGPU forbids any depth bias
7/// on line topologies and rejects the pipeline, so the web draws them
8/// unbiased.
9pub(crate) fn line_depth_bias() -> wgpu::DepthBiasState {
10    if cfg!(target_arch = "wasm32") {
11        wgpu::DepthBiasState::default()
12    } else {
13        wgpu::DepthBiasState {
14            constant: -2,
15            slope_scale: -1.0,
16            clamp: 0.0,
17        }
18    }
19}

68/// Material uniform buffer data (bind group 1)
69#[repr(C)]
70#[derive(Copy, Clone, Debug, Pod, Zeroable)]
71pub struct MaterialUniforms {
72    pub base_color: [f32; 4],
73    pub metallic: f32,
74    pub roughness: f32,
75    pub use_vertex_color: u32,
76    pub debug_mode: u32,
77    pub enable_tonemapping: u32,
78    pub has_base_color_tex: u32,
79    pub has_normal_map: u32,
80    pub has_metallic_roughness_tex: u32,
81    pub selection_highlight: u32,
82    pub opacity: f32,
83    pub alpha_cutoff: f32,
84    pub texture_scale: f32,
85    /// Emission (ADR 0076). `rgb` is the material's own light, linear and
86    /// already times its strength. `w` scales per-vertex emission, where a
87    /// vertex's alpha says how much of it is surface (see
88    /// `ImportedMaterial::vertex_emission`).
89    pub emissive: [f32; 4],
90    /// 1 when the material has an emissive map (glTF `emissiveTexture`),
91    /// bound at 7 and 8. Its rgb multiplies `emissive.rgb`.
92    pub has_emissive_tex: u32,
93    /// 1 when vertex alpha is the surface share, not opacity (ADR 0090).
94    /// Separate from `emissive.w` so a strength of 0 leaves the light parts
95    /// dark rather than turning them back into lit, see-through surface.
96    pub vertex_emission: u32,
97    /// How much of each per-vertex light part is also lit surface, 0..1
98    /// (ADR 0096: the entity's `material.glow_surface`). Its light is
99    /// unchanged, so with `emissive_scale` a light part's surface and light
100    /// are set apart. 0 is the mesh as uploaded.
101    pub glow_surface: f32,
102    pub _pad_emissive: u32,
103    /// Multiplies every vertex colour, `rgb` only, as surface and as
104    /// per-vertex light (ADR 0102: the entity's `material.vertex_tint_*`).
105    /// White is the mesh as uploaded. Unused without `use_vertex_color`.
106    pub vertex_tint: [f32; 4],
107}

129            glow_surface: 0.0,

135    /// Material from imported PBR parameters
136    pub fn from_pbr(base_color: [f32; 4], metallic: f32, roughness: f32) -> Self {
137        Self {
138            base_color,
139            metallic,
140            roughness,
141            use_vertex_color: 0,
142            debug_mode: 0,
143            enable_tonemapping: 1,
144            has_base_color_tex: 0,
145            has_normal_map: 0,
146            has_metallic_roughness_tex: 0,
147            selection_highlight: 0,
148            opacity: 1.0,
149            alpha_cutoff: 0.5,
150            texture_scale: 1.0,
151            emissive: [0.0; 4],
152            has_emissive_tex: 0,
153            vertex_emission: 0,
154            glow_surface: 0.0,
155            _pad_emissive: 0,
156            vertex_tint: [1.0; 4],
157        }
158    }
159
160    /// Sets the emission from an imported material.
161    pub fn set_emission(&mut self, material: &flint_import::ImportedMaterial) {
162        let [r, g, b] = material.emissive;
163        let s = material.emissive_strength;
164        let per_vertex = material.vertex_emission && material.use_vertex_color;
165        self.emissive = [r * s, g * s, b * s, if per_vertex { s } else { 0.0 }];
166        self.vertex_emission = per_vertex as u32;
167    }
168
169    /// Multiplies all of the emission, the material's own and per-vertex,
170    /// by `scale` (ADR 0090: the entity's `material.emissive_scale`). The
171    /// surface share of each vertex is unchanged, so 0 leaves pure-light
172    /// vertices black and lit-surface vertices lit.
173    pub fn scale_emission(&mut self, scale: f32) {
174        let scale = scale.max(0.0);
175        for c in &mut self.emissive {
176            *c *= scale;
177        }
178    }
179}

184    }

289/// Combined light uniform buffer (bind group 2)
290#[repr(C)]
291#[derive(Copy, Clone, Debug, Pod, Zeroable)]
292pub struct LightUniforms {
293    pub directional_lights: [DirectionalLight; MAX_DIRECTIONAL_LIGHTS],
294    pub point_lights: [PointLight; MAX_POINT_LIGHTS],
295    pub spot_lights: [SpotLight; MAX_SPOT_LIGHTS],
296    pub directional_count: u32,
297    pub point_count: u32,
298    pub spot_count: u32,
299    pub _pad: u32,
300    /// rgb = hemisphere sky color; w encodes the diffuse-wrap knob as
301    /// (1 + wrap) so legacy writes of 1.0 decode to wrap = 0 (ADR 0044).
302    pub ambient_sky: [f32; 4],
303    /// rgb = hemisphere ground color; w encodes the Oren-Nayar diffuse
304    /// blend the same way: (1 + oren), legacy 1.0 = off (ADR 0048).
305    pub ambient_ground: [f32; 4],
306    /// rgb = sheen tint (linear), w = sheen strength; all-zero = off.
307    /// New field (not an encoding trick): `Pod::zeroed()` and every
308    /// pre-existing writer leave it disabled. Mirrored in SIX WGSL structs —
309    /// shader / skinned / terrain / grass_render / volumetric / ocean —
310    /// see the `light_uniforms_layout` test (ADR 0048).
311    pub sheen_color_strength: [f32; 4],
312}

328            color: [1.0, 0.98, 0.95],
329            intensity: 3.0,
330            volumetric_color: [1.0, 0.98, 0.95],
331            angular_size: 0.0,
332        };
333
334        // Fill light (cool, from lower-left-behind)
335        lights.directional_lights[1] = DirectionalLight {
336            direction: [-0.4, -0.3, -0.6],
337            volumetric_intensity: 0.0,
338            color: [0.6, 0.7, 0.9],

358pub struct RenderPipeline {
359    pub pipeline: wgpu::RenderPipeline,
360    pub line_pipeline: wgpu::RenderPipeline,
361    pub overlay_line_pipeline: wgpu::RenderPipeline,
362    pub skeleton_line_pipeline: wgpu::RenderPipeline,
363    pub outline_pipeline: wgpu::RenderPipeline,
364    pub depth_prepass_pipeline: wgpu::RenderPipeline,
365    pub transparent_alpha_pipeline: wgpu::RenderPipeline,
366    /// Back-face pass of the alpha pipeline (drawn first).
367    pub transparent_alpha_back_pipeline: wgpu::RenderPipeline,
368    pub transparent_additive_pipeline: wgpu::RenderPipeline,
369    pub transparent_multiply_pipeline: wgpu::RenderPipeline,
370    pub transform_bind_group_layout: wgpu::BindGroupLayout,
371    pub material_bind_group_layout: wgpu::BindGroupLayout,
372    pub light_bind_group_layout: wgpu::BindGroupLayout,
373    /// The sampler every PBR material's textures share, bound in the light
374    /// group (binding 4) rather than in each material's: linear, with linear
375    /// mips, repeating, as `TextureCache` made every texture's own.
376    pub material_sampler: wgpu::Sampler,
377}
378
379impl RenderPipeline {
380    pub fn new(device: &wgpu::Device, format: wgpu::TextureFormat, sample_count: u32) -> Self {
381        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
382            label: Some("PBR Shader"),
383            source: wgpu::ShaderSource::Wgsl(include_str!("shader.wgsl").into()),
384        });
385
386        // Bind group 0: Transform uniforms (vertex + fragment)
387        let transform_bind_group_layout =
388            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
389                entries: &[wgpu::BindGroupLayoutEntry {
390                    binding: 0,
391                    visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
392                    ty: wgpu::BindingType::Buffer {
393                        ty: wgpu::BufferBindingType::Uniform,
394                        has_dynamic_offset: false,
395                        min_binding_size: None,
396                    },
397                    count: None,
398                }],
399                label: Some("Transform Bind Group Layout"),
400            });
401
402        // Bind group 1: Material uniforms + textures (fragment only). No
403        // samplers: they sample with the light group's `material_sampler`,
404        // so a material bind group spends no sampler descriptors. wgpu 23's
405        // DX12 backend copies every bind group's samplers into a 2048-slot
406        // shader-visible heap with no sharing, and four a material capped a
407        // scene at about 512 distinct looks (Capitol #525, #577). Bindings
408        // keep their numbers (textures at 1, 3, 5, 7).
409        let material_bind_group_layout =
410            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
411                entries: &[
412                    // binding 0: MaterialUniforms
413                    wgpu::BindGroupLayoutEntry {
414                        binding: 0,
415                        visibility: wgpu::ShaderStages::FRAGMENT,
416                        ty: wgpu::BindingType::Buffer {
417                            ty: wgpu::BufferBindingType::Uniform,
418                            has_dynamic_offset: false,
419                            min_binding_size: None,
420                        },
421                        count: None,
422                    },
423                    // binding 1: base_color_texture
424                    wgpu::BindGroupLayoutEntry {
425                        binding: 1,
426                        visibility: wgpu::ShaderStages::FRAGMENT,
427                        ty: wgpu::BindingType::Texture {
428                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
429                            view_dimension: wgpu::TextureViewDimension::D2,
430                            multisampled: false,
431                        },
432                        count: None,
433                    },
434                    // binding 3: normal_map_texture
435                    wgpu::BindGroupLayoutEntry {
436                        binding: 3,
437                        visibility: wgpu::ShaderStages::FRAGMENT,
438                        ty: wgpu::BindingType::Texture {
439                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
440                            view_dimension: wgpu::TextureViewDimension::D2,
441                            multisampled: false,
442                        },
443                        count: None,
444                    },
445                    // binding 5: metallic_roughness_texture
446                    wgpu::BindGroupLayoutEntry {
447                        binding: 5,
448                        visibility: wgpu::ShaderStages::FRAGMENT,
449                        ty: wgpu::BindingType::Texture {
450                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
451                            view_dimension: wgpu::TextureViewDimension::D2,
452                            multisampled: false,
453                        },
454                        count: None,
455                    },
456                    // binding 7: emissive_texture
457                    wgpu::BindGroupLayoutEntry {
458                        binding: 7,
459                        visibility: wgpu::ShaderStages::FRAGMENT,
460                        ty: wgpu::BindingType::Texture {
461                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
462                            view_dimension: wgpu::TextureViewDimension::D2,
463                            multisampled: false,
464                        },
465                        count: None,
466                    },
467                ],
468                label: Some("Material Bind Group Layout"),
469            });
470
471        // Bind group 2: Light uniforms + shadow resources (fragment only)
472        let light_bind_group_layout =
473            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
474                entries: &[
475                    // binding 0: LightUniforms
476                    wgpu::BindGroupLayoutEntry {
477                        binding: 0,
478                        visibility: wgpu::ShaderStages::FRAGMENT,
479                        ty: wgpu::BindingType::Buffer {
480                            ty: wgpu::BufferBindingType::Uniform,
481                            has_dynamic_offset: false,
482                            min_binding_size: None,
483                        },
484                        count: None,
485                    },
486                    // binding 1: shadow_maps (depth texture array)
487                    wgpu::BindGroupLayoutEntry {
488                        binding: 1,
489                        visibility: wgpu::ShaderStages::FRAGMENT,
490                        ty: wgpu::BindingType::Texture {
491                            sample_type: wgpu::TextureSampleType::Depth,
492                            view_dimension: wgpu::TextureViewDimension::D2Array,
493                            multisampled: false,
494                        },
495                        count: None,
496                    },
497                    // binding 2: shadow comparison sampler
498                    wgpu::BindGroupLayoutEntry {
499                        binding: 2,
500                        visibility: wgpu::ShaderStages::FRAGMENT,
501                        ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Comparison),
502                        count: None,
503                    },
504                    // binding 3: ShadowUniforms (cascade matrices + splits)
505                    wgpu::BindGroupLayoutEntry {
506                        binding: 3,
507                        visibility: wgpu::ShaderStages::FRAGMENT,
508                        ty: wgpu::BindingType::Buffer {
509                            ty: wgpu::BufferBindingType::Uniform,
510                            has_dynamic_offset: false,
511                            min_binding_size: None,
512                        },
513                        count: None,
514                    },
515                    // binding 4: material_sampler, the one sampler every
516                    // material's textures share (see bind group 1)
517                    wgpu::BindGroupLayoutEntry {
518                        binding: 4,
519                        visibility: wgpu::ShaderStages::FRAGMENT,
520                        ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
521                        count: None,
522                    },
523                ],
524                label: Some("Light Bind Group Layout"),
525            });
526
527        let material_sampler = device.create_sampler(&wgpu::SamplerDescriptor {
528            label: Some("Material Sampler"),
529            mag_filter: wgpu::FilterMode::Linear,
530            min_filter: wgpu::FilterMode::Linear,
531            mipmap_filter: wgpu::FilterMode::Linear,
532            address_mode_u: wgpu::AddressMode::Repeat,
533            address_mode_v: wgpu::AddressMode::Repeat,
534            // Ground seen edge-on (roads, floors) keeps its detail; needs
535            // all three filters linear, as they are
536            anisotropy_clamp: 8,
537            ..Default::default()
538        });
539
540        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
541            label: Some("PBR Pipeline Layout"),
542            bind_group_layouts: &[
543                &transform_bind_group_layout,
544                &material_bind_group_layout,
545                &light_bind_group_layout,
546            ],
547            push_constant_ranges: &[],
548        });
549
550        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
551            label: Some("PBR Render Pipeline"),
552            layout: Some(&pipeline_layout),
553            vertex: wgpu::VertexState {
554                module: &shader,
555                entry_point: Some("vs_main"),
556                buffers: &[Vertex::desc()],
557                compilation_options: Default::default(),
558            },
559            fragment: Some(wgpu::FragmentState {
560                module: &shader,
561                entry_point: Some("fs_main"),
562                targets: &[Some(wgpu::ColorTargetState {
563                    format,
564                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
565                    write_mask: wgpu::ColorWrites::ALL,
566                })],
567                compilation_options: Default::default(),
568            }),
569            primitive: wgpu::PrimitiveState {
570                topology: wgpu::PrimitiveTopology::TriangleList,
571                strip_index_format: None,
572                front_face: wgpu::FrontFace::Ccw,
573                cull_mode: Some(wgpu::Face::Back),
574                polygon_mode: wgpu::PolygonMode::Fill,
575                unclipped_depth: false,
576                conservative: false,
577            },
578            depth_stencil: Some(wgpu::DepthStencilState {
579                format: wgpu::TextureFormat::Depth32Float,
580                depth_write_enabled: true,
581                depth_compare: wgpu::CompareFunction::Less,
582                stencil: wgpu::StencilState::default(),
583                bias: wgpu::DepthBiasState::default(),
584            }),
585            multisample: wgpu::MultisampleState {
586                count: sample_count,
587                ..Default::default()
588            },
589            multiview: None,
590            cache: None,
591        });
592
593        // Line pipeline for wireframes and grid
594        let line_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
595            label: Some("Line Render Pipeline"),
596            layout: Some(&pipeline_layout),
597            vertex: wgpu::VertexState {
598                module: &shader,
599                entry_point: Some("vs_main"),
600                buffers: &[Vertex::desc()],
601                compilation_options: Default::default(),
602            },
603            fragment: Some(wgpu::FragmentState {
604                module: &shader,
605                entry_point: Some("fs_main"),
606                targets: &[Some(wgpu::ColorTargetState {
607                    format,
608                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
609                    write_mask: wgpu::ColorWrites::ALL,
610                })],
611                compilation_options: Default::default(),
612            }),
613            primitive: wgpu::PrimitiveState {
614                topology: wgpu::PrimitiveTopology::LineList,
615                strip_index_format: None,
616                front_face: wgpu::FrontFace::Ccw,
617                cull_mode: None,
618                polygon_mode: wgpu::PolygonMode::Fill,
619                unclipped_depth: false,
620                conservative: false,
621            },
622            depth_stencil: Some(wgpu::DepthStencilState {
623                format: wgpu::TextureFormat::Depth32Float,
624                depth_write_enabled: true,
