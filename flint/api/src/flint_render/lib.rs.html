4//! rendering entities as colored shapes or imported glTF models with
5//! physically-based Cook-Torrance shading. Supports GPU vertex skinning

184    #[test]
185    fn post_process_uniforms_layout() {
186        // Desaturate reused a pre-existing pad slot (176 bytes); the DoF row
187        // (strength/focus/range/_pad2) and the render-mode rows
188        // (render_mode/mode_mix/mode_time/pad + mode_params vec4) were
189        // appended after inv_view_proj (224); the color-grade rows
190        // (lift+enabled / gamma+pad / gain+pad, ADR 0050) grew it to 272,
191        // with film grain reusing the former _pad slot. Any other size means
192        // the Rust struct and the WGSL mirror have diverged.
193        let size = std::mem::size_of::<crate::postprocess::PostProcessUniforms>();
194        assert_eq!(size % 16, 0, "PostProcessUniforms not 16-byte aligned");
195        assert_eq!(size, 272, "PostProcessUniforms size changed");
196    }

198    #[test]
199    fn light_uniforms_layout() {
200        // LightUniforms is mirrored field-for-field in SIX WGSL structs
201        // (shader / skinned / terrain / grass_render / volumetric / ocean);
202        // bind groups use min_binding_size: None, so a divergence surfaces
203        // as a draw-time wgpu validation error, not a compile error. The
204        // sheen vec4 (ADR 0048) grew the struct from 1264 to 1280 bytes;
205        // PointLight's source_radius row (ADR 0056) grew it to 1536
206        // (16 point lights × 16 extra bytes).
207        let size = std::mem::size_of::<crate::pipeline::LightUniforms>();
208        assert_eq!(size % 16, 0, "LightUniforms not 16-byte aligned");
209        assert_eq!(
210            size, 1536,
211            "LightUniforms size changed — update all six WGSL mirrors"
212        );
213    }

215    #[test]
216    fn light_uniforms_defaults_are_lever_neutral() {
217        use bytemuck::Zeroable;
218        // The clay-look levers must decode to "off" from every pre-existing
219        // writer: wrap rides ambient_sky.w as (1 + wrap), Oren-Nayar rides
220        // ambient_ground.w the same way, sheen is a plain zeroed vec4.
221        let defaults = crate::pipeline::LightUniforms::default_scene_lights();
222        assert_eq!(defaults.ambient_sky[3], 1.0, "wrap must decode to 0");
223        assert_eq!(defaults.ambient_ground[3], 1.0, "oren must decode to 0");
224        assert_eq!(defaults.sheen_color_strength, [0.0; 4], "sheen must be off");
225        let zeroed = crate::pipeline::LightUniforms::zeroed();
226        assert_eq!(zeroed.sheen_color_strength, [0.0; 4]);
227        // Area-light levers (ADR 0056): 0 = punctual → verbatim legacy path.
228        for d in defaults
229            .directional_lights
230            .iter()
231            .chain(zeroed.directional_lights.iter())
232        {
233            assert_eq!(d.angular_size, 0.0, "angular_size must default punctual");
234        }
235        for p in defaults
236            .point_lights
237            .iter()
238            .chain(zeroed.point_lights.iter())
239        {
240            assert_eq!(p.source_radius, 0.0, "point source_radius must default 0");
241        }
242        for sl in defaults.spot_lights.iter().chain(zeroed.spot_lights.iter()) {
243            assert_eq!(sl.source_radius, 0.0, "spot source_radius must default 0");
244        }
245        use crate::pipeline::{DirectionalLight, PointLight, SpotLight};
246        assert_eq!(DirectionalLight::default().angular_size, 0.0);
247        assert_eq!(PointLight::default().source_radius, 0.0);
248        assert_eq!(SpotLight::default().source_radius, 0.0);
249    }
