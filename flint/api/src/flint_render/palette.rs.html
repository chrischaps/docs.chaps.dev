1//! Colour palettes (#276): a short list of ink colours and a background,
2//! read from a `*.palette.toml`, for any style that wants a fixed set of
3//! colours (ASCII's `palette` colour mode first).
4//!
5//! ```toml
6//! # palettes/phosphor.palette.toml
7//! match = "luminance"          # nearest (default) | luminance
8//! background = "#020a04"       # the ground behind the inks (default black)
9//! colors = ["#1f6e33", "#33b04d", "#4dff6e", "#b4ffc4"]
10//! ```
11//!
12//! `nearest` gives each cell the ink closest to its own colour (a CGA
13//! screen); `luminance` lists the inks darkest first and picks one by the
14//! cell's brightness alone (a one-colour phosphor tube). Colours are
15//! display (sRGB) values, `#rrggbb` or `#rgb`, at most [`MAX_COLORS`].
16//!
17//! A style names a palette as `builtin` names ([`BUILTIN_PALETTES`]), a
18//! bare name looked up as `<dir>/palettes/<name>.palette.toml` in the
19//! scene's folder then its parent, or a `.palette.toml` path. Every read
20//! goes through the vfs.
21

29/// The most inks a palette holds (what the styles' uniforms make room for).
30pub const MAX_COLORS: usize = 32;
31
32/// File suffix of a palette.
33pub const PALETTE_SUFFIX: &str = ".palette.toml";
34
35/// The palettes built into the engine, by name.
36pub const BUILTIN_PALETTES: &[&str] = &["cga", "cga-warm", "cga16", "phosphor", "amber", "mono"];

38/// How a palette's ink is chosen for a colour.
39#[derive(Clone, Copy, Debug, PartialEq, Eq, Default, Deserialize)]
40#[serde(rename_all = "snake_case")]
41pub enum PaletteMatch {
42    /// The ink nearest the colour.
43    #[default]
44    Nearest,
45    /// The inks are a ladder, darkest first; brightness picks the rung.
46    Luminance,
47}
48
49impl PaletteMatch {
50    /// The shader's number for it.
51    pub fn index(self) -> u32 {
52        match self {
53            PaletteMatch::Nearest => 0,
54            PaletteMatch::Luminance => 1,
55        }
56    }
57}

59/// A palette did not parse or could not be found.
60#[derive(Clone, Debug, PartialEq)]
61pub enum PaletteError {
62    Read(String),
63    Parse(String),
64}
65
66impl fmt::Display for PaletteError {
67    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
68        match self {
69            PaletteError::Read(e) => write!(f, "{e}"),
70            PaletteError::Parse(e) => write!(f, "{e}"),
71        }
72    }
73}

75impl std::error::Error for PaletteError {}

87/// A named set of inks and the ground behind them, in display values 0-1.
88#[derive(Clone, Debug, PartialEq)]
89pub struct Palette {
90    name: String,
91    colors: Vec<[f32; 3]>,
92    background: [f32; 3],
93    matching: PaletteMatch,
94}
95
96impl Palette {
97    /// A palette from display-space colours. Errors on none or too many.
98    pub fn new(
99        name: impl Into<String>,
100        colors: Vec<[f32; 3]>,
101        background: [f32; 3],
102        matching: PaletteMatch,
103    ) -> Result<Self, PaletteError> {
104        let name = name.into();
105        if colors.is_empty() {
106            return Err(PaletteError::Parse(format!(
107                "palette `{name}` has no colors"
108            )));
109        }
110        if colors.len() > MAX_COLORS {
111            return Err(PaletteError::Parse(format!(
112                "palette `{name}` has {} colors; at most {MAX_COLORS}",
113                colors.len()
114            )));
115        }
116        Ok(Self {
117            name,
118            colors,
119            background,
120            matching,
121        })
122    }
123
124    /// Parse a `*.palette.toml`.
125    pub fn from_toml(name: &str, text: &str) -> Result<Self, PaletteError> {
126        let file: PaletteFile = toml::from_str(text)
127            .map_err(|e| PaletteError::Parse(format!("palette `{name}`: {e}")))?;
128        let colors = file
129            .colors
130            .iter()
131            .map(|c| {
132                parse_hex(c).map_err(|e| PaletteError::Parse(format!("palette `{name}`: {e}")))
133            })
134            .collect::<Result<Vec<_>, _>>()?;
135        let background = match &file.background {
136            Some(c) => {
137                parse_hex(c).map_err(|e| PaletteError::Parse(format!("palette `{name}`: {e}")))?
138            }
139            None => [0.0; 3],
140        };
141        Self::new(name, colors, background, file.matching)
142    }
143
144    /// Read a `*.palette.toml` through the vfs.
145    pub fn load(path: &Path) -> Result<Self, PaletteError> {
146        let text = vfs::read_to_string(path)
147            .map_err(|e| PaletteError::Read(format!("{}: {e}", path.display())))?;
148        let name = path
149            .file_name()
150            .and_then(|n| n.to_str())
151            .map(|n| n.trim_end_matches(PALETTE_SUFFIX).to_string())
152            .unwrap_or_default();
153        Self::from_toml(&name, &text)
154    }
155
156    /// A built-in palette by name.
157    pub fn builtin(name: &str) -> Option<Arc<Palette>> {
158        static BUILTINS: OnceLock<HashMap<&'static str, Arc<Palette>>> = OnceLock::new();
159        BUILTINS
160            .get_or_init(|| {
161                BUILTIN_PALETTES
162                    .iter()
163                    .map(|n| {
164                        let text = builtin_text(n).expect("every built-in has a file");
165                        let p = Palette::from_toml(n, text).expect("built-in palettes parse");
166                        (*n, Arc::new(p))
167                    })
168                    .collect()
169            })
170            .get(name)
171            .cloned()
172    }
173
174    pub fn name(&self) -> &str {
175        &self.name
176    }
177
178    /// The inks, in file order.
179    pub fn colors(&self) -> &[[f32; 3]] {
180        &self.colors
181    }
182
183    pub fn background(&self) -> [f32; 3] {
184        self.background
185    }
186
187    pub fn matching(&self) -> PaletteMatch {
188        self.matching
189    }
190
191    /// The ink nearest `color` (display values; plain RGB distance).
192    pub fn nearest(&self, color: [f32; 3]) -> [f32; 3] {
193        let d = |c: &[f32; 3]| {
194            (0..3)
195                .map(|i| (c[i] - color[i]) * (c[i] - color[i]))
196                .sum::<f32>()
197        };
198        *self
199            .colors
200            .iter()
201            .min_by(|a, b| d(a).total_cmp(&d(b)))
202            .expect("a palette has a colour")
203    }
204
205    /// The rung of a luminance ladder for tone `t` (0-1).
206    pub fn rung(&self, t: f32) -> [f32; 3] {
207        let n = self.colors.len();
208        let i = ((t.clamp(0.0, 1.0) * n as f32) as usize).min(n - 1);
209        self.colors[i]
210    }
211
212    /// The ink for a cell of full-value `hue` and tone `t`, as the shader
213    /// picks it.
214    pub fn ink(&self, hue: [f32; 3], t: f32) -> [f32; 3] {
215        match self.matching {
216            PaletteMatch::Nearest => self.nearest(hue),
217            PaletteMatch::Luminance => self.rung(t),
218        }
219    }
220
221    /// The inks as uniform slots (rgb + 0), padded to [`MAX_COLORS`].
222    pub fn uniform_colors(&self) -> [[f32; 4]; MAX_COLORS] {
223        let mut out = [[0.0; 4]; MAX_COLORS];
224        for (slot, c) in out.iter_mut().zip(&self.colors) {
225            *slot = [c[0], c[1], c[2], 0.0];
226        }
227        out
228    }
229}

231/// `#rrggbb`, `rrggbb`, `#rgb` or `rgb` as display values 0-1.
232pub fn parse_hex(text: &str) -> Result<[f32; 3], String> {
233    let hex = text.trim().trim_start_matches('#');
234    let digits: Vec<u32> = hex
235        .chars()
236        .map(|c| c.to_digit(16))
237        .collect::<Option<_>>()
238        .ok_or_else(|| format!("`{text}` is not a hex colour"))?;
239    let bytes = match digits.len() {
240        6 => [
241            digits[0] * 16 + digits[1],
242            digits[2] * 16 + digits[3],
243            digits[4] * 16 + digits[5],
244        ],
245        3 => [digits[0] * 17, digits[1] * 17, digits[2] * 17],
246        _ => return Err(format!("`{text}` is not #rrggbb or #rgb")),
247    };
248    Ok(bytes.map(|b| b as f32 / 255.0))
249}
250
251fn builtin_text(name: &str) -> Option<&'static str> {
252    Some(match name {
253        // CGA mode 4, palette 1 at high intensity: the cyan-magenta screen
254        "cga" => {
255            r##"colors = ["#55ffff", "#ff55ff", "#ffffff"]
256background = "#000000""##
257        }
258        // CGA mode 4, palette 0 at high intensity: green, red, yellow
259        "cga-warm" => {
260            r##"colors = ["#55ff55", "#ff5555", "#ffff55"]
261background = "#000000""##
262        }
263        // The sixteen text-mode colours, black as the ground
264        "cga16" => {
265            r##"colors = ["#0000aa", "#00aa00", "#00aaaa", "#aa0000", "#aa00aa", "#aa5500", "#aaaaaa",
266          "#555555", "#5555ff", "#55ff55", "#55ffff", "#ff5555", "#ff55ff", "#ffff55", "#ffffff"]
267background = "#000000""##
268        }
269        // A green phosphor tube: one hue, four brightnesses
270        "phosphor" => {
271            r##"match = "luminance"
272colors = ["#1f7a38", "#33b850", "#5cff7a", "#c8ffd2"]
273background = "#03100a""##
274        }
275        // An amber monitor
276        "amber" => {
277            r##"match = "luminance"
278colors = ["#8a4a00", "#cc7a00", "#ffb000", "#ffe0a0"]
279background = "#120800""##
280        }
281        // Greys on black
282        "mono" => {
283            r##"match = "luminance"
284colors = ["#5a5a5a", "#999999", "#d0d0d0", "#ffffff"]
285background = "#000000""##
286        }
287        _ => return None,
288    })
289}

291/// Palettes by the name a style gives, loaded once each. Failures are
292/// kept, so a bad name warns once rather than every frame.
293#[derive(Default)]
294pub struct PaletteCache {
295    search: Vec<PathBuf>,
296    palettes: HashMap<String, Result<Arc<Palette>, PaletteError>>,
297}
298
299impl PaletteCache {
300    /// A cache that resolves names against `search`, in order.
301    pub fn new(search: Vec<PathBuf>) -> Self {
302        Self {
303            search,
304            palettes: HashMap::new(),
305        }
306    }
307
308    /// The directories names resolve against.
309    pub fn search(&self) -> &[PathBuf] {
310        &self.search
311    }
312
313    /// The palette `name` names, loading it the first time.
314    pub fn get(&mut self, name: &str) -> Result<Arc<Palette>, PaletteError> {
315        if let Some(found) = self.palettes.get(name) {
316            return found.clone();
317        }
318        let loaded = self.resolve(name);
319        if let Err(e) = &loaded {
320            tracing::warn!("palette `{name}`: {e}");
321        }
322        self.palettes.insert(name.to_string(), loaded.clone());
323        loaded
324    }
325
326    /// Forget every palette, so the next [`Self::get`] reads files again.
327    pub fn clear(&mut self) {
328        self.palettes.clear();
329    }
330
331    fn resolve(&self, name: &str) -> Result<Arc<Palette>, PaletteError> {
332        let candidates: Vec<PathBuf> = if name.ends_with(PALETTE_SUFFIX) {
333            let mut c: Vec<PathBuf> = self.search.iter().map(|d| d.join(name)).collect();
334            c.push(PathBuf::from(name));
335            c
336        } else {
337            self.search
338                .iter()
339                .map(|d| d.join("palettes").join(format!("{name}{PALETTE_SUFFIX}")))
340                .collect()
341        };
342        // A file beside the scene wins over a built-in of the same name
343        if let Some(path) = candidates.iter().find(|p| vfs::is_file(p)) {
344            return Palette::load(path).map(Arc::new);
345        }
346        if let Some(p) = Palette::builtin(name) {
347            return Ok(p);
348        }
349        Err(PaletteError::Read(format!(
350            "no palette `{name}` (built-in: {}; looked for {})",
351            BUILTIN_PALETTES.join(", "),
352            candidates
353                .iter()
354                .map(|p| p.display().to_string())
355                .collect::<Vec<_>>()
356                .join(", ")
357        )))
358    }
359}

365    #[test]
366    fn every_builtin_parses() {
367        for name in BUILTIN_PALETTES {
368            let p = Palette::builtin(name).unwrap();
369            assert_eq!(p.name(), *name);

375    #[test]
376    fn hex_parses_long_and_short() {
377        assert_eq!(parse_hex("#ffffff").unwrap(), [1.0, 1.0, 1.0]);
378        assert_eq!(parse_hex("000").unwrap(), [0.0, 0.0, 0.0]);
379        assert_eq!(parse_hex("#f80").unwrap(), [1.0, 136.0 / 255.0, 0.0]);
380        assert!(parse_hex("#ff").is_err());
381        assert!(parse_hex("#gggggg").is_err());
382    }

384    #[test]
385    fn a_file_parses_with_its_defaults() {
386        let p = Palette::from_toml("two", r##"colors = ["#ff0000", "#0000ff"]"##).unwrap();
387        assert_eq!(p.matching(), PaletteMatch::Nearest);
388        assert_eq!(p.background(), [0.0; 3]);
389        assert_eq!(p.colors().len(), 2);
390        assert!(Palette::from_toml("none", "colors = []").is_err());
391        assert!(Palette::from_toml("typo", r##"colours = ["#fff"]"##).is_err());
392        let many = format!("colors = [{}]", vec!["\"#fff\""; MAX_COLORS + 1].join(","));
393        assert!(Palette::from_toml("many", &many).is_err());
394    }

396    #[test]
397    fn cga_sends_each_hue_to_its_nearest_ink() {
398        let cga = Palette::builtin("cga").unwrap();
399        let cyan = [0x55 as f32 / 255.0, 1.0, 1.0];
400        let magenta = [1.0, 0x55 as f32 / 255.0, 1.0];
401        assert_eq!(cga.ink([0.0, 1.0, 1.0], 0.5), cyan);
402        assert_eq!(cga.ink([0.0, 0.8, 1.0], 0.5), cyan);
403        assert_eq!(cga.ink([1.0, 0.0, 0.9], 0.5), magenta);
404        assert_eq!(cga.ink([1.0, 1.0, 1.0], 0.5), [1.0; 3]);
405        // A full-value red lands on magenta, never on the black ground
406        assert_eq!(cga.ink([1.0, 0.0, 0.0], 0.1), magenta);
407    }
