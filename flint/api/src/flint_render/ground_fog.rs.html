1//! Ground fog (ADR 0106): fog that lies on a floor of its own, read from a
2//! map over a world rectangle, so a mist can fill hollows at many heights.
3//! The height fog (ADR 0101) is one level layer over the whole scene: low
4//! enough to fill a high valley it buries the plain, and at the plain it
5//! misses every valley above it.
6//!
7//! Each texel of a [`GroundFog`] holds a floor (world Y) and an amount (0
8//! to 1). Over a texel the fog's density is `amount × ground_fog_density`
9//! at its floor, falling by e every `1 / ground_fog_falloff` metres above
10//! it, and holding at the floor's under it. The composite pass marches
11//! each view ray through the box the map covers, from the lowest floor up
12//! to where the fog over the highest has thinned to nothing. Its steps
13//! lengthen with distance from the eye, so the drifts near it are seen
14//! fine wherever the march begins. The map is read at each step's ends, at
15//! the mip level the step spans, and along a step the height over the
16//! floor runs straight from one end's to the other's, so on a slope the
17//! floor follows the ground. The falloff is integrated along each step
18//! exactly, so a ray that dips into a thin layer between two reads still
19//! meets it.
20//!
21//! The look (colour, density, falloff, steps) is on
22//! [`PostProcessConfig`](crate::PostProcessConfig) and can change every
23//! frame; the map is set once with `SceneRenderer::set_ground_fog`, and
24//! again when the ground changes. The fog isn't lit: the game gives it the
25//! colour the hour's light would.
26
27/// A ground fog's map over a world rectangle, for
28/// `SceneRenderer::set_ground_fog`.
29#[derive(Clone, Copy, Debug)]
30pub struct GroundFog<'a> {
31    /// The rectangle's minimum corner, world X and Z.
32    pub origin: [f32; 2],
33    /// The rectangle's extent in world units, X and Z.
34    pub size: [f32; 2],
35    /// Texels across X and along Z.
36    pub width: u32,
37    pub depth: u32,
38    /// `[floor, amount]` per texel, `width × depth` of them, row by row
39    /// toward +Z. Each is the value at the middle of its texel's cell, and
40    /// the fog blends the four round a point. The floor is a world Y; give
41    /// a texel with no fog the ground's height too, since its floor still
42    /// shapes the fog at its edges. The amount scales the density, 0 to 1.
43    pub texels: &'a [[f32; 2]],
44}
45
46impl GroundFog<'_> {
47    /// Errs when the texels don't fill the stated size, the rectangle is
48    /// empty, or a value isn't finite.
49    pub fn validate(&self) -> Result<(), String> {
50        if self.width == 0 || self.depth == 0 {
51            return Err("ground fog: the map has no texels".into());
52        }
53        let n = self.width as usize * self.depth as usize;
54        if self.texels.len() != n {
55            return Err(format!(
56                "ground fog: {} texels for a {}×{} map",
57                self.texels.len(),
58                self.width,
59                self.depth
60            ));
61        }
62        if !(self.size[0] > 0.0 && self.size[1] > 0.0) {
63            return Err(format!("ground fog: size {:?} is empty", self.size));
64        }
65        if !self.origin.iter().all(|v| v.is_finite())
66            || !self.size.iter().all(|v| v.is_finite())
67            || !self.texels.iter().flatten().all(|v| v.is_finite())
68        {
69            return Err("ground fog: a value isn't finite".into());
70        }
71        Ok(())
72    }
73}

75/// A map made ready for the GPU: its mip chain, with each floor stored over
76/// `floor_base` so half floats keep it fine.
77#[derive(Debug, Clone, PartialEq)]
78pub(crate) struct FogMips {
79    /// `(width, depth, [floor - floor_base, amount])` per level, finest first.
80    pub levels: Vec<(u32, u32, Vec<[f32; 2]>)>,
81    /// The lowest floor: the bottom of the box the pass marches.
82    pub floor_base: f32,
83    /// The highest floor under any fog.
84    pub floor_top: f32,
85}
86
87/// The mip chain of `fog`, down to one texel. Each level is the last's size
88/// halved, rounding down as the GPU does, so an odd last row or column
89/// joins the texel beside it. A texel's amount is the mean of those under
90/// it and its floor the mean of theirs weighted by their amount, so a
91/// coarse level keeps the fog on the floors it lies on. `None` when no
92/// texel has any fog.
93pub(crate) fn mip_chain(fog: &GroundFog) -> Option<FogMips> {
94    let fogged = fog.texels.iter().filter(|t| t[1] > 0.0);
95    let floor_top = fogged.map(|t| t[0]).fold(f32::NEG_INFINITY, f32::max);
96    if floor_top == f32::NEG_INFINITY {
97        return None;
98    }
99    let floor_base = fog
100        .texels
101        .iter()
102        .map(|t| t[0])
103        .fold(f32::INFINITY, f32::min);
104    let first: Vec<[f32; 2]> = fog
105        .texels
106        .iter()
107        .map(|&[floor, amount]| [floor - floor_base, amount.clamp(0.0, 1.0)])
108        .collect();
109    let mut levels = vec![(fog.width, fog.depth, first)];
110    loop {
111        let (w, d, last) = levels.last().unwrap();
112        if *w == 1 && *d == 1 {
113            break;
114        }
115        let (w, d) = (*w as usize, *d as usize);
116        let (nw, nd) = ((w / 2).max(1), (d / 2).max(1));
117        // The fine texels under coarse texel `c` of `n`, of `fine`.
118        let under =
119            |c: usize, n: usize, fine: usize| 2 * c..if c + 1 == n { fine } else { 2 * c + 2 };
120        let mut next = Vec::with_capacity(nw * nd);
121        for b in 0..nd {
122            for a in 0..nw {
123                let (mut amount, mut weighted, mut floors, mut count) = (0.0, 0.0, 0.0, 0.0);
124                for z in under(b, nd, d) {
125                    for x in under(a, nw, w) {
126                        let [floor, amt] = last[z * w + x];
127                        amount += amt;
128                        weighted += amt * floor;
129                        floors += floor;
130                        count += 1.0;
131                    }
132                }
133                let floor = if amount > 0.0 {
134                    weighted / amount
135                } else {
136                    floors / count
137                };
138                next.push([floor, amount / count]);
139            }
140        }
141        levels.push((nw as u32, nd as u32, next));
142    }
143    Some(FogMips {
144        levels,
145        floor_base,
146        floor_top,
147    })
148}

150/// A ground fog on the GPU, as the composite pass reads it. Made by
151/// `SceneRenderer::set_ground_fog`.
152pub struct GroundFogGpu {
153    _texture: wgpu::Texture,
154    pub(crate) view: wgpu::TextureView,
155    pub(crate) origin: [f32; 2],
156    pub(crate) size: [f32; 2],
157    /// World units a finest texel spans (the larger of X and Z).
158    pub(crate) texel: f32,
159    pub(crate) mip_count: u32,
160    pub(crate) floor_base: f32,
161    pub(crate) floor_top: f32,
162    /// Its [`gpu_id`](crate::bind_cache::gpu_id), for the composite's kept
163    /// bind group (#139).
164    pub(crate) id: u64,
165}
166
167/// The texture format: two half floats, filterable on every backend.
168pub(crate) const FORMAT: wgpu::TextureFormat = wgpu::TextureFormat::Rg16Float;
169
170impl GroundFogGpu {
171    /// Upload `fog`. `Ok(None)` when it holds no fog at all.
172    pub(crate) fn new(
173        device: &wgpu::Device,
174        queue: &wgpu::Queue,
175        fog: &GroundFog,
176    ) -> Result<Option<Self>, String> {
177        fog.validate()?;
178        let Some(mips) = mip_chain(fog) else {
179            return Ok(None);
180        };
181        let texture = device.create_texture(&wgpu::TextureDescriptor {
182            label: Some("Ground Fog Map"),
183            size: wgpu::Extent3d {
184                width: fog.width,
185                height: fog.depth,
186                depth_or_array_layers: 1,
187            },
188            mip_level_count: mips.levels.len() as u32,
189            sample_count: 1,
190            dimension: wgpu::TextureDimension::D2,
191            format: FORMAT,
192            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
193            view_formats: &[],
194        });
195        for (level, (w, d, texels)) in mips.levels.iter().enumerate() {
196            let bytes: Vec<u8> = texels
197                .iter()
198                .flat_map(|t| t.iter().flat_map(|&v| f16_bits(v).to_le_bytes()))
199                .collect();
200            queue.write_texture(
201                wgpu::ImageCopyTexture {
202                    texture: &texture,
203                    mip_level: level as u32,
204                    origin: wgpu::Origin3d::ZERO,
205                    aspect: wgpu::TextureAspect::All,
206                },
207                &bytes,
208                wgpu::ImageDataLayout {
209                    offset: 0,
210                    bytes_per_row: Some(w * 4),
211                    rows_per_image: Some(*d),
212                },
213                wgpu::Extent3d {
214                    width: *w,
215                    height: *d,
216                    depth_or_array_layers: 1,
217                },
218            );
219        }
220        Ok(Some(Self {
221            view: texture.create_view(&wgpu::TextureViewDescriptor::default()),
222            _texture: texture,
223            origin: fog.origin,
224            size: fog.size,
225            texel: (fog.size[0] / fog.width as f32).max(fog.size[1] / fog.depth as f32),
226            mip_count: mips.levels.len() as u32,
227            floor_base: mips.floor_base,
228            floor_top: mips.floor_top,
229            id: crate::bind_cache::gpu_id(),
230        }))
231    }
232}

234/// `v` as an IEEE half float's bits, rounded to nearest even, saturating
235/// to infinity past the half range.
236pub(crate) fn f16_bits(v: f32) -> u16 {
237    let bits = v.to_bits();
238    let sign = ((bits >> 16) & 0x8000) as u16;
239    let exp = ((bits >> 23) & 0xff) as i32;
240    let man = bits & 0x7f_ffff;
241    if exp == 0xff {
242        return sign | 0x7c00 | if man != 0 { 0x200 } else { 0 };
243    }
244    let e = exp - 127 + 15;
245    if e >= 0x1f {
246        return sign | 0x7c00;
247    }
248    if e <= 0 {
249        // Subnormal, or too small for one.
250        if e < -10 {
251            return sign;
252        }
253        let man = man | 0x80_0000;
254        let shift = (14 - e) as u32;
255        let half = 1u32 << (shift - 1);
256        let rest = man & ((1 << shift) - 1);
257        let mut out = man >> shift;
258        if rest > half || (rest == half && out & 1 == 1) {
259            out += 1;
260        }
261        return sign | out as u16;
262    }
263    let mut out = ((e as u32) << 10) | (man >> 13);
264    let rest = man & 0x1fff;
265    if rest > 0x1000 || (rest == 0x1000 && out & 1 == 1) {
266        // A carry into the exponent is the right rounding too.
267        out += 1;
268    }
269    sign | out as u16
270}

272/// The optical depth a step of length `len` gathers through fog of
273/// density `density` at its floor, falling by e every `1 / k` above it,
274/// from `ya` to `yb` over the floor. Under the floor the density holds at
275/// the floor's. Exact while the height over the floor runs straight; the WGSL
276/// `fog_step` in `composite_shader.wgsl` is this, line for line.
277#[cfg(test)]
278pub(crate) fn fog_step(density: f32, k: f32, ya: f32, yb: f32, len: f32) -> f32 {
279    // Above the floor: len' × e^-k·lo × (1 - e^-x) / x over the part
280    // above, x = k × (hi - lo). Under it, the part under at full density.
281    let (lo, hi) = if ya < yb { (ya, yb) } else { (yb, ya) };
282    let span = hi - lo;
283    if span < 1e-4 {
284        return density * len * (-k * lo.max(0.0)).exp();
285    }
286    let under = ((0.0 - lo) / span).clamp(0.0, 1.0);
287    let lo = lo.max(0.0);
288    let x = k * (hi - lo);
289    let share = if x > 1e-3 {
290        (1.0 - (-x).exp()) / x
291    } else {
292        1.0 - 0.5 * x
293    };
294    density * len * (under + (1.0 - under) * (-k * lo).exp() * share)
295}

297/// Where step `i` of `n` starts on a march from `t0` to `t1` along the
298/// ray: each step longer than the last in proportion to its distance from
299/// the eye (counted from a metre behind it). The WGSL `fog_step_at` is this.
300#[cfg(test)]
301pub(crate) fn step_at(i: f32, n: f32, t0: f32, t1: f32) -> f32 {
302    (t0 + 1.0) * ((t1 + 1.0) / (t0 + 1.0)).powf(i / n) - 1.0
303}

309    fn numeric(density: f32, k: f32, ya: f32, yb: f32, len: f32) -> f32 {
310        let steps = 20_000;
311        (0..steps)
312            .map(|s| {
313                let y = ya + (yb - ya) * (s as f32 + 0.5) / steps as f32;
314                density * (-k * y.max(0.0)).exp()
315            })
316            .sum::<f32>()
317            * len
318            / steps as f32
319    }

323        for (ya, yb) in [
324            (2.0, 2.0),
325            (0.5, 6.0),
326            (9.0, 0.2),

343    #[test]
344    fn steps_lengthen_with_distance_and_cover_the_march() {
345        let n = 24.0;
346        // From the eye out: fine near it, coarse far off.
347        let (t0, t1) = (0.0, 1000.0);
348        assert!((step_at(0.0, n, t0, t1) - t0).abs() < 1e-4);
349        assert!((step_at(n, n, t0, t1) - t1).abs() < 1e-2);
350        let first = step_at(1.0, n, t0, t1) - step_at(0.0, n, t0, t1);
351        let last = step_at(n, n, t0, t1) - step_at(n - 1.0, n, t0, t1);
352        assert!(first < 0.5 && last > 100.0, "{first} {last}");
353        // Entering the fog from far above: about even, so the steps at the
354        // ground, where the mist lies, are as fine as those at the top.
355        let (t0, t1) = (300.0, 800.0);
356        let first = step_at(1.0, n, t0, t1) - step_at(0.0, n, t0, t1);
357        let last = step_at(n, n, t0, t1) - step_at(n - 1.0, n, t0, t1);
358        assert!(last < 3.0 * first && last < 40.0, "{first} {last}");
359    }

362    fn the_mips_keep_the_fog_on_its_floors() {
363        // A 4×2 map: fog on a floor at 10 beside clear ground at 40.
364        let texels = [
365            [10.0, 1.0],
366            [40.0, 0.0],
367            [10.0, 0.5],
368            [12.0, 0.0],
369            [10.0, 1.0],
370            [40.0, 0.0],
371            [10.0, 0.0],
372            [12.0, 0.0],
373        ];
374        let fog = GroundFog {
375            origin: [0.0, 0.0],
376            size: [40.0, 20.0],
377            width: 4,
378            depth: 2,
379            texels: &texels,
380        };
381        fog.validate().unwrap();
382        let mips = mip_chain(&fog).unwrap();
383        assert_eq!(mips.floor_base, 10.0);
384        assert_eq!(mips.floor_top, 10.0);
385        let sizes: Vec<_> = mips.levels.iter().map(|(w, d, _)| (*w, *d)).collect();
386        assert_eq!(sizes, [(4, 2), (2, 1), (1, 1)]);
387        // Half fogged, all of it on the floor at 10, not halfway up to 40.
388        assert_eq!(mips.levels[1].2[0], [0.0, 0.5]);
389        // An eighth fogged, on the fogged texel's floor.
390        assert_eq!(mips.levels[1].2[1], [0.0, 0.125]);
391        assert_eq!(mips.levels[2].2[0], [0.0, 0.3125]);
392        // No fog anywhere: nothing to upload.
393        let clear = [[3.0, 0.0]; 4];
394        let fog = GroundFog {
395            texels: &clear,
396            width: 2,
397            depth: 2,
398            ..fog
399        };
400        assert!(mip_chain(&fog).is_none());
401        let short = GroundFog {
402            texels: &clear[..3],
403            ..fog
404        };
405        assert!(short.validate().is_err());
406    }

410        // As many levels as wgpu allows, each `size >> level`, at least 1.
411        for (w, d) in [(5, 3), (555, 555), (7, 1), (1, 9)] {
412            let texels = vec![[0.0, 1.0]; (w * d) as usize];
413            let fog = GroundFog {
414                origin: [0.0; 2],

416                width: w,
417                depth: d,
418                texels: &texels,
419            };
420            let mips = mip_chain(&fog).unwrap();
421            assert_eq!(mips.levels.len() as u32, w.max(d).ilog2() + 1, "{w}×{d}");
422            for (l, (lw, ld, t)) in mips.levels.iter().enumerate() {
423                assert_eq!((*lw, *ld), ((w >> l).max(1), (d >> l).max(1)));

442    #[test]
443    fn half_floats_round_to_nearest() {
444        for (v, bits) in [
445            (0.0, 0x0000),
446            (1.0, 0x3c00),
447            (-2.0, 0xc000),
448            (0.5, 0x3800),
449            (65504.0, 0x7bff),
450            (1e6, 0x7c00),
451            (6.1035156e-5, 0x0400),
452            (5.9604645e-8, 0x0001),
453            (1.0 + 1.0 / 2048.0, 0x3c00),
454            (1.0 + 3.0 / 2048.0, 0x3c02),
455        ] {
456            assert_eq!(f16_bits(v), bits, "{v}");
457        }
458        // A floor 37.3 m over the base keeps better than 2 cm.
459        let bits = f16_bits(37.3);
460        let e = ((bits >> 10) & 0x1f) as i32 - 15;
461        let back = (1.0 + (bits & 0x3ff) as f32 / 1024.0) * 2f32.powi(e);
462        assert!((back - 37.3).abs() < 0.02, "{back}");
463    }
