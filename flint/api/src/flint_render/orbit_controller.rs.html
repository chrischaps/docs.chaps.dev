1//! Shared orbit camera controller for mouse + keyboard input.
2//!
3//! Used by `gen-preview`, `preview`, and `serve`/viewer to provide a
4//! consistent orbit-camera experience with left-drag orbit, right-drag pan,
5//! scroll zoom, and WASD/QE keyboard controls.
6
7use std::collections::HashSet;
8
9use winit::event::{ElementState, MouseButton, MouseScrollDelta, WindowEvent};
10use winit::keyboard::{KeyCode, PhysicalKey};
11
12use crate::Camera;
13
14/// Orbit camera controller handling mouse drag and keyboard held-key input.
15pub struct OrbitCameraController {
16    // Mouse state
17    pub left_pressed: bool,
18    pub right_pressed: bool,
19    pub last_mouse_pos: Option<(f64, f64)>,
20
21    // Keyboard held-key tracking (only WASD/QE)
22    held_keys: HashSet<KeyCode>,
23
24    // Sensitivity
25    pub orbit_sensitivity: f32,
26    pub zoom_scroll_factor: f32,
27    pub key_orbit_speed: f32,
28    pub key_zoom_speed: f32,
29
30    // Auto-orbit
31    pub auto_orbit: bool,
32    pub auto_orbit_speed: f32,
33}
34
35impl Default for OrbitCameraController {
36    fn default() -> Self {
37        Self::new()
38    }
39}
40
41impl OrbitCameraController {
42    pub fn new() -> Self {
43        Self {
44            left_pressed: false,
45            right_pressed: false,
46            last_mouse_pos: None,
47            held_keys: HashSet::new(),
48            orbit_sensitivity: 0.005,
49            zoom_scroll_factor: 0.1,
50            key_orbit_speed: 2.0,
51            key_zoom_speed: 2.0,
52            auto_orbit: false,
53            auto_orbit_speed: 0.5,
54        }
55    }
56
57    /// Process a winit `WindowEvent`, updating mouse/key state and applying
58    /// immediate camera transforms (drag orbit, drag pan, scroll zoom).
59    ///
60    /// Returns `true` if the event was consumed by the controller.
61    /// `egui_wants_input` should be `true` when egui's pointer is over its UI.
62    pub fn handle_event(
63        &mut self,
64        event: &WindowEvent,
65        camera: &mut Camera,
66        egui_wants_input: bool,
67    ) -> bool {
68        match event {
69            WindowEvent::MouseInput { state, button, .. } => {
70                if egui_wants_input {
71                    return false;
72                }
73                match button {
74                    MouseButton::Left => {
75                        self.left_pressed = *state == ElementState::Pressed;
76                        if self.left_pressed {
77                            self.auto_orbit = false;
78                        }
79                    }
80                    MouseButton::Right => {
81                        self.right_pressed = *state == ElementState::Pressed;
82                    }
83                    _ => return false,
84                }
85                if *state == ElementState::Released {
86                    self.last_mouse_pos = None;
87                }
88                true
89            }
90
91            WindowEvent::CursorMoved { position, .. } => {
92                if egui_wants_input {
93                    return false;
94                }
95                if let Some((lx, ly)) = self.last_mouse_pos {
96                    let dx = (position.x - lx) as f32;
97                    let dy = (position.y - ly) as f32;
98
99                    if self.left_pressed {
100                        // Orbit
101                        camera.yaw -= dx * self.orbit_sensitivity;
102                        camera.pitch += dy * self.orbit_sensitivity;
103                        camera.pitch = camera.pitch.clamp(-1.4, 1.4);
104                        camera.update_orbit();
105                    } else if self.right_pressed {
106                        // Pan using Camera::pan() for correct right/up vectors
107                        let pan_factor = camera.distance * 0.002;
108                        camera.pan(-dx * pan_factor, dy * pan_factor);
109                    }
110                }
111                self.last_mouse_pos = Some((position.x, position.y));
112                self.left_pressed || self.right_pressed
113            }
114
115            WindowEvent::MouseWheel { delta, .. } => {
116                if egui_wants_input {
117                    return false;
118                }
119                let scroll = match delta {
120                    MouseScrollDelta::LineDelta(_, y) => *y,
121                    MouseScrollDelta::PixelDelta(pos) => pos.y as f32 * 0.01,
122                };
123                camera.distance *= 1.0 - scroll * self.zoom_scroll_factor;
124                camera.distance = camera.distance.max(0.1);
125                camera.update_orbit();
126                true
127            }
128
129            _ => false,
130        }
131    }

133    /// Track WASD/QE held state from a `KeyEvent`.
134    /// Call this from your `KeyboardInput` handler so the controller knows
135    /// which orbit keys are held. Does not consume the event.
136    pub fn handle_key_event(&mut self, event: &winit::event::KeyEvent) {
137        if let PhysicalKey::Code(code) = event.physical_key {
138            match code {
139                KeyCode::KeyW
140                | KeyCode::KeyA
141                | KeyCode::KeyS
142                | KeyCode::KeyD
143                | KeyCode::KeyQ
144                | KeyCode::KeyE => {
145                    if event.state == ElementState::Pressed {
146                        self.held_keys.insert(code);
147                        self.auto_orbit = false;
148                    } else {
149                        self.held_keys.remove(&code);
150                    }
151                }
152                _ => {}
153            }
154        }
155    }
156
157    /// Apply held-key orbit/zoom and auto-orbit each frame. Call once per
158    /// frame with the frame delta in seconds.
159    pub fn update(&self, camera: &mut Camera, dt: f32) {
160        let mut yaw_delta = 0.0f32;
161        let mut pitch_delta = 0.0f32;
162        let mut zoom_factor = 1.0f32;
163
164        // Auto-orbit: continuous yaw rotation
165        if self.auto_orbit {
166            yaw_delta += self.auto_orbit_speed * dt;
167        }
168
169        if self.held_keys.contains(&KeyCode::KeyA) {
170            yaw_delta += self.key_orbit_speed * dt;
171        }
172        if self.held_keys.contains(&KeyCode::KeyD) {
173            yaw_delta -= self.key_orbit_speed * dt;
174        }
175        if self.held_keys.contains(&KeyCode::KeyW) {
176            pitch_delta += self.key_orbit_speed * dt;
177        }
178        if self.held_keys.contains(&KeyCode::KeyS) {
179            pitch_delta -= self.key_orbit_speed * dt;
180        }
181        if self.held_keys.contains(&KeyCode::KeyQ) {
182            zoom_factor *= 1.0 + self.key_zoom_speed * dt;
183        }
184        if self.held_keys.contains(&KeyCode::KeyE) {
185            zoom_factor *= 1.0 - self.key_zoom_speed * dt;
186        }
187
188        if yaw_delta != 0.0 || pitch_delta != 0.0 || zoom_factor != 1.0 {
189            camera.yaw += yaw_delta;
190            camera.pitch = (camera.pitch + pitch_delta).clamp(-1.4, 1.4);
191            camera.distance = (camera.distance * zoom_factor).max(0.1);
192            camera.update_orbit();
193        }
194    }
195
196    /// Multiply the auto-orbit speed by `factor`, clamped to 0.1–5.0 rad/s.
197    pub fn adjust_auto_orbit_speed(&mut self, factor: f32) {
198        self.auto_orbit_speed = (self.auto_orbit_speed * factor).clamp(0.1, 5.0);
199    }
200
201    /// Returns true if any orbit key is currently held.
202    pub fn has_held_keys(&self) -> bool {
203        !self.held_keys.is_empty()
204    }
205
206    /// Check if a specific key is currently held.
207    pub fn is_key_held(&self, key: KeyCode) -> bool {
208        self.held_keys.contains(&key)
209    }
