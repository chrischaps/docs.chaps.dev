6pub(super) fn identity_matrix() -> [[f32; 4]; 4] {
7    [

9        [0.0, 1.0, 0.0, 0.0],
10        [0.0, 0.0, 1.0, 0.0],

15/// Compute the inverse-transpose of a 4x4 model matrix (for correct normal transformation).
16/// Only the upper 3x3 matters for normals; we embed it in a 4x4 for GPU upload.
17pub(super) fn mat4_inv_transpose(m: &[[f32; 4]; 4]) -> [[f32; 4]; 4] {
18    // Extract upper-left 3x3
19    let a = m[0][0];
20    let b = m[1][0];
21    let c = m[2][0];
22    let d = m[0][1];
23    let e = m[1][1];
24    let f = m[2][1];
25    let g = m[0][2];
26    let h = m[1][2];
27    let i = m[2][2];
28
29    let det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g);
30
31    if det.abs() < 1e-10 {
32        return identity_matrix();
33    }
34
35    let inv_det = 1.0 / det;
36
37    // Cofactor matrix: cof(i,j) / det gives the inverse-transpose entries.
38    // In column-major storage m[col][row], column c needs [cof(0,c), cof(1,c), cof(2,c)] / det.
39    //
40    // Row 0 cofactors:
41    let cof00 = (e * i - f * h) * inv_det;
42    let cof01 = (f * g - d * i) * inv_det;
43    let cof02 = (d * h - e * g) * inv_det;
44    // Row 1 cofactors:
45    let cof10 = (c * h - b * i) * inv_det;
46    let cof11 = (a * i - c * g) * inv_det;
47    let cof12 = (b * g - a * h) * inv_det;
48    // Row 2 cofactors:
49    let cof20 = (b * f - c * e) * inv_det;
50    let cof21 = (c * d - a * f) * inv_det;
51    let cof22 = (a * e - b * d) * inv_det;
52
53    // Column-major: column j = [cof(0,j), cof(1,j), cof(2,j)]
54    [
55        [cof00, cof10, cof20, 0.0],
56        [cof01, cof11, cof21, 0.0],
57        [cof02, cof12, cof22, 0.0],
58        [0.0, 0.0, 0.0, 1.0],
59    ]
60}

62/// Extract both the size and center offset from bounds.
63pub(super) fn extract_bounds_info(bounds: &toml::Value) -> Option<([f32; 3], [f32; 3])> {
64    let min = bounds.get("min")?;
65    let max = bounds.get("max")?;
66
67    let min_arr = toml_vec3(min)?;
68    let max_arr = toml_vec3(max)?;
69
70    let size = [
71        max_arr[0] - min_arr[0],
72        max_arr[1] - min_arr[1],
73        max_arr[2] - min_arr[2],
74    ];
75
76    let center = [
77        (min_arr[0] + max_arr[0]) / 2.0,
78        (min_arr[1] + max_arr[1]) / 2.0,
79        (min_arr[2] + max_arr[2]) / 2.0,
80    ];
81
82    Some((size, center))
83}

85/// Parse a blend mode string from TOML into the BlendMode enum
86pub(super) fn parse_blend_mode(s: &str) -> BlendMode {
87    match s {
88        "additive" => BlendMode::Additive,
89        "multiply" => BlendMode::Multiply,
90        _ => BlendMode::Alpha,
91    }
