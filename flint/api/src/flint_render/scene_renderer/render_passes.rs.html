18/// Which slice of the main pass to render. `All` is the classic single-pass
19/// path; Pre/PostOcean split around the refraction grab (opaque scene is
20/// snapshotted between them so the ocean can sample it).
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub(super) enum RenderPhase {
23    All,
24    PreOcean,
25    PostOcean,
26}
27
28impl RenderPhase {
29    fn draws_opaque(self) -> bool {
30        self != RenderPhase::PostOcean
31    }
32    fn draws_ocean_and_after(self) -> bool {
33        self != RenderPhase::PreOcean
34    }
35}

38    /// Render depth from light perspective for cascaded shadow mapping.
39    pub(super) fn render_shadow_pass(
40        &mut self,
41        device: &wgpu::Device,
42        queue: &wgpu::Queue,
43        camera: &Camera,
44        camera_pos: [f32; 3],
45    ) {
46        let shadow_pass = match &mut self.shadow_pass {
47            Some(sp) => sp,
48            None => return,
49        };
50
51        if !shadow_pass.enabled || self.light_uniforms.directional_count == 0 {
52            return;
53        }
54
55        // Use the first directional light for shadows
56        let light = &self.light_uniforms.directional_lights[0];
57
58        // Update cascade matrices
59        let camera_inv = camera.inverse_view_projection_matrix();
60        shadow_pass.update_cascades(light.direction, camera_pos, camera_inv, 0.1, 200.0, 100.0);
61
62        // Write shadow uniforms
63        queue.write_buffer(
64            &shadow_pass.shadow_uniforms_buffer,
65            0,
66            bytemuck::cast_slice(&[*shadow_pass.shadow_uniforms()]),
67        );
68
69        // Render each cascade
70        for cascade in 0..CASCADE_COUNT {
71            let cascade_vp = shadow_pass.shadow_uniforms().cascade_view_proj[cascade];
72
73            let mut encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
74                label: Some(&format!("Shadow Cascade {} Encoder", cascade)),
75            });
76
77            {
78                let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
79                    label: Some(&format!("Shadow Cascade {} Pass", cascade)),
80                    color_attachments: &[],
81                    depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
82                        view: &shadow_pass.cascade_views[cascade],
83                        depth_ops: Some(wgpu::Operations {
84                            load: wgpu::LoadOp::Clear(1.0),
85                            store: wgpu::StoreOp::Store,
86                        }),
87                        stencil_ops: None,
88                    }),
89                    timestamp_writes: None,
90                    occlusion_query_set: None,
91                });
92
93                pass.set_pipeline(&shadow_pass.shadow_pipeline);
94
95                // Render each solid entity
96                for draw in &self.entity_draws {
97                    if draw.is_wireframe {
98                        continue;
99                    }
100
101                    let shadow_uniforms = ShadowDrawUniforms {
102                        light_view_proj: cascade_vp,
103                        model: draw.model,
104                    };
105
106                    let shadow_buffer =
107                        device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
108                            label: Some("Shadow Draw Uniform"),
109                            contents: bytemuck::cast_slice(&[shadow_uniforms]),
110                            usage: wgpu::BufferUsages::UNIFORM,
111                        });
112
113                    let shadow_bind = device.create_bind_group(&wgpu::BindGroupDescriptor {
114                        layout: &shadow_pass.shadow_bind_group_layout,
115                        entries: &[wgpu::BindGroupEntry {
116                            binding: 0,
117                            resource: shadow_buffer.as_entire_binding(),
118                        }],
119                        label: Some("Shadow Draw Bind Group"),
120                    });
121
122                    pass.set_bind_group(0, &shadow_bind, &[]);
123                    pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
124                    pass.set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
125                    pass.draw_indexed(0..draw.index_count, 0, 0..1);
126                }
127
128                // Render terrain chunks into shadow map (frustum cull per cascade)
129                let cascade_frustum = crate::frustum::Frustum::from_view_projection(&cascade_vp);
130                for draw in &self.terrain_draws {
131                    if !cascade_frustum.aabb_visible(draw.aabb_min, draw.aabb_max) {
132                        continue;
133                    }
134                    let shadow_uniforms = ShadowDrawUniforms {
135                        light_view_proj: cascade_vp,
136                        model: draw.model,
137                    };
138
139                    let shadow_buffer =
140                        device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
141                            label: Some("Terrain Shadow Draw Uniform"),
142                            contents: bytemuck::cast_slice(&[shadow_uniforms]),
143                            usage: wgpu::BufferUsages::UNIFORM,
144                        });
145
146                    let shadow_bind = device.create_bind_group(&wgpu::BindGroupDescriptor {
147                        layout: &shadow_pass.shadow_bind_group_layout,
148                        entries: &[wgpu::BindGroupEntry {
149                            binding: 0,
150                            resource: shadow_buffer.as_entire_binding(),
151                        }],
152                        label: Some("Terrain Shadow Draw Bind Group"),
153                    });
154
155                    pass.set_bind_group(0, &shadow_bind, &[]);
156                    pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
157                    pass.set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
158                    pass.draw_indexed(0..draw.index_count, 0, 0..1);
159                }
160
161                // Render grass into shadow cascade (nearest 2 only)
162                if cascade < 2 {
163                    if let (Some(gp), Some(render_bg), Some(instance_bg)) = (
164                        &self.grass_pipeline,
165                        &self.grass_render_uniform_bind_group,
166                        &self.grass_render_instance_bind_group,
167                    ) {
168                        if self.grass_instance_count > 0 {
169                            // Grass shadow pipeline uses TransformUniforms (same layout as PBR)
170                            let grass_shadow_uniforms = TransformUniforms {
171                                view_proj: cascade_vp,
172                                model: identity_matrix(),
173                                model_inv_transpose: identity_matrix(),
174                                camera_pos: [0.0; 3],
175                                _pad: 0.0,
176                            };
177                            let grass_shadow_buffer =
178                                device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
179                                    label: Some("Grass Shadow Transform"),
180                                    contents: bytemuck::cast_slice(&[grass_shadow_uniforms]),
181                                    usage: wgpu::BufferUsages::UNIFORM,
182                                });
183                            let grass_shadow_bind =
184                                device.create_bind_group(&wgpu::BindGroupDescriptor {
185                                    layout: &self.pipeline.transform_bind_group_layout,
186                                    entries: &[wgpu::BindGroupEntry {
187                                        binding: 0,
188                                        resource: grass_shadow_buffer.as_entire_binding(),
189                                    }],
190                                    label: Some("Grass Shadow Transform Bind Group"),
191                                });
192
193                            pass.set_pipeline(&gp.shadow_pipeline);
194                            pass.set_bind_group(0, &grass_shadow_bind, &[]);
195                            pass.set_bind_group(1, render_bg, &[]);
196                            pass.set_bind_group(2, &gp.shadow_dummy_bind_group, &[]);
197                            pass.set_bind_group(3, instance_bg, &[]);
198                            pass.set_vertex_buffer(0, gp.blade_vertex_buffer.slice(..));
199                            pass.set_index_buffer(
200                                gp.blade_index_buffer.slice(..),
201                                wgpu::IndexFormat::Uint16,
202                            );
203                            pass.draw_indexed(
204                                0..BLADE_INDEX_COUNT,
205                                0,
206                                0..self.grass_instance_count,
207                            );
208                        }
209                    }
210                }
211
212                // Render transparent entities into shadow map (skip very transparent)
213                for draw in &self.transparent_draws {
214                    if draw.is_wireframe {
215                        continue;
216                    }
217
218                    let shadow_uniforms = ShadowDrawUniforms {
219                        light_view_proj: cascade_vp,
220                        model: draw.model,
221                    };
222
223                    let shadow_buffer =
224                        device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
225                            label: Some("Transparent Shadow Draw Uniform"),
226                            contents: bytemuck::cast_slice(&[shadow_uniforms]),
227                            usage: wgpu::BufferUsages::UNIFORM,
228                        });
229
230                    let shadow_bind = device.create_bind_group(&wgpu::BindGroupDescriptor {
231                        layout: &shadow_pass.shadow_bind_group_layout,
232                        entries: &[wgpu::BindGroupEntry {
233                            binding: 0,
234                            resource: shadow_buffer.as_entire_binding(),
235                        }],
236                        label: Some("Transparent Shadow Draw Bind Group"),
237                    });
238
239                    pass.set_bind_group(0, &shadow_bind, &[]);
240                    pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
241                    pass.set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
242                    pass.draw_indexed(0..draw.index_count, 0, 0..1);
243                }
244
245                // Render skinned entities into shadow map
246                pass.set_pipeline(&shadow_pass.skinned_shadow_pipeline);
247
248                for draw in &self.skinned_entity_draws {
249                    let shadow_uniforms = ShadowDrawUniforms {
250                        light_view_proj: cascade_vp,
251                        model: draw.model,
252                    };
253
254                    let shadow_buffer =
255                        device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
256                            label: Some("Skinned Shadow Draw Uniform"),
257                            contents: bytemuck::cast_slice(&[shadow_uniforms]),
258                            usage: wgpu::BufferUsages::UNIFORM,
259                        });
260
261                    let shadow_bind = device.create_bind_group(&wgpu::BindGroupDescriptor {
262                        layout: &shadow_pass.shadow_bind_group_layout,
263                        entries: &[wgpu::BindGroupEntry {
264                            binding: 0,
265                            resource: shadow_buffer.as_entire_binding(),
266                        }],
267                        label: Some("Skinned Shadow Draw Bind Group"),
268                    });
269
270                    pass.set_bind_group(0, &shadow_bind, &[]);
271                    pass.set_bind_group(1, &draw.bone_bind_group, &[]);
272                    pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
273                    pass.set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
274                    pass.draw_indexed(0..draw.index_count, 0, 0..1);
275                }
276            }
277
278            queue.submit(std::iter::once(encoder.finish()));
279        }
280    }

282    /// Write per-frame uniforms to all GPU buffers and sort transparent draw calls.
283    pub(super) fn update_per_frame_uniforms(
284        &mut self,
285        queue: &wgpu::Queue,
286        view_proj: [[f32; 4]; 4],
287        camera_pos: [f32; 3],
288        debug_mode_u32: u32,
289        tonemapping_u32: u32,
290        camera: &Camera,
291    ) {
292        // Update grid transform
293        if let Some(grid) = &self.grid_draw {
294            let uniforms = TransformUniforms {
295                view_proj,
296                model: identity_matrix(),
297                model_inv_transpose: identity_matrix(),
298                camera_pos,
299                _pad: 0.0,
300            };
301            queue.write_buffer(&grid.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
302        }
303
304        // Update entity transforms and write debug_mode + tonemapping into each material buffer
305        for draw in &self.entity_draws {
306            let uniforms = TransformUniforms {
307                view_proj,
308                model: draw.model,
309                model_inv_transpose: draw.model_inv_transpose,
310                camera_pos,
311                _pad: 0.0,
312            };
313            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
314
315            // Write debug_mode at byte offset 28 in the material buffer
316            // Layout: base_color(16) + metallic(4) + roughness(4) + use_vertex_color(4) = 28
317            queue.write_buffer(
318                &draw.material_buffer,
319                28,
320                bytemuck::cast_slice(&[debug_mode_u32]),
321            );
322
323            // Write enable_tonemapping at byte offset 32
324            // Layout: ... + debug_mode(4) = 32
325            queue.write_buffer(
326                &draw.material_buffer,
327                32,
328                bytemuck::cast_slice(&[tonemapping_u32]),
329            );
330        }
331
332        // Update skinned entity transforms
333        for draw in &self.skinned_entity_draws {
334            let uniforms = TransformUniforms {
335                view_proj,
336                model: draw.model,
337                model_inv_transpose: draw.model_inv_transpose,
338                camera_pos,
339                _pad: 0.0,
340            };
341            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
342
343            queue.write_buffer(
344                &draw.material_buffer,
345                28,
346                bytemuck::cast_slice(&[debug_mode_u32]),
347            );
348            queue.write_buffer(
349                &draw.material_buffer,
350                32,
351                bytemuck::cast_slice(&[tonemapping_u32]),
352            );
353        }
354
355        // Update transparent entity transforms
356        for draw in &self.transparent_draws {
357            let uniforms = TransformUniforms {
358                view_proj,
359                model: draw.model,
360                model_inv_transpose: draw.model_inv_transpose,
361                camera_pos,
362                _pad: 0.0,
363            };
364            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
365
366            queue.write_buffer(
367                &draw.material_buffer,
368                28,
369                bytemuck::cast_slice(&[debug_mode_u32]),
370            );
371            queue.write_buffer(
372                &draw.material_buffer,
373                32,
374                bytemuck::cast_slice(&[tonemapping_u32]),
375            );
376        }
377
378        // Update transparent skinned entity transforms
379        for draw in &self.transparent_skinned_draws {
380            let uniforms = TransformUniforms {
381                view_proj,
382                model: draw.model,
383                model_inv_transpose: draw.model_inv_transpose,
384                camera_pos,
385                _pad: 0.0,
386            };
387            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
388
389            queue.write_buffer(
390                &draw.material_buffer,
391                28,
392                bytemuck::cast_slice(&[debug_mode_u32]),
393            );
394            queue.write_buffer(
395                &draw.material_buffer,
396                32,
397                bytemuck::cast_slice(&[tonemapping_u32]),
398            );
399        }
400
401        // Update terrain chunk transforms
402        for draw in &self.terrain_draws {
403            let uniforms = TransformUniforms {
404                view_proj,
405                model: draw.model,
406                model_inv_transpose: draw.model_inv_transpose,
407                camera_pos,
408                _pad: 0.0,
409            };
410            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
411        }
412
413        // Update terrain material buffer (tonemapping flag)
414        if let Some(buf) = &self.terrain_material_buffer {
415            // enable_tonemapping is at byte offset 12 (texture_tile(4) + metallic(4) + roughness(4))
416            queue.write_buffer(buf, 12, bytemuck::cast_slice(&[tonemapping_u32]));
417        }
418
419        // Update ocean uniforms (wave phases in f64, grid follows the camera)
420        if self.ocean_active {
421            if let (Some(spectrum), Some(ubuf), Some(tbuf)) = (
422                &self.ocean_spectrum,
423                &self.ocean_uniform_buffer,
424                &self.ocean_transform_buffer,
425            ) {
426                let transform = TransformUniforms {
427                    view_proj,
428                    model: identity_matrix(),
429                    model_inv_transpose: identity_matrix(),
430                    camera_pos,
431                    _pad: 0.0,
432                };
433                queue.write_buffer(tbuf, 0, bytemuck::cast_slice(&[transform]));
434
435                let v = &self.ocean_visuals;
436                // Snap the grid to inner-cell multiples so wave sampling
437                // positions stay world-anchored while the mesh follows us.
438                let q = v.snap_quantum();
439                let grid_offset = [
440                    (camera_pos[0] / q).floor() * q,
441                    (camera_pos[2] / q).floor() * q,
442                ];
443                let uniforms = crate::ocean_pipeline::OceanUniformsGpu {
444                    waves: spectrum.to_gpu(self.ocean_time),
445                    deep_color: v.deep_color,
446                    shallow_color: v.shallow_color,
447                    foam_color: v.foam_color,
448                    sss_color: v.sss_color,
449                    grid_offset,
450                    num_waves: spectrum.waves.len() as u32,
451                    enable_tonemapping: tonemapping_u32,
452                    grid_scale: v.grid_scale,
453                    fade_start: v.fade_start,
454                    fade_end: v.fade_end,
455                    foam_threshold: v.foam_threshold,
456                    foam_noise_scale: v.foam_noise_scale,
457                    ramp_steps: v.ramp_steps,
458                    specular_strength: v.specular_strength,
459                    time: (self.ocean_time % 100_000.0) as f32,
460                    absorption_color: v.absorption_color,
461                    turbidity: v.turbidity,
462                    refraction_strength: v.refraction_strength,
463                    camera_near: self.ocean_camera_near_far.0,
464                    camera_far: self.ocean_camera_near_far.1,
465                    grab_enabled: u32::from(self.ocean_grab_this_frame),
466                    fog_color: {
467                        let c = self.postprocess_config.fog_color;
468                        let strength = if self.postprocess_config.fog_enabled { 1.0 } else { 0.0 };
469                        [c[0], c[1], c[2], strength]
470                    },
471                    sky_zenith: self.sky_params.zenith_color,
472                    sky_horizon: self.sky_params.horizon_color,
473                    sky_haze: self.sky_params.haze_color,
474                    // Reflections need a procedural sky to reflect; scenes
475                    // with a texture skybox (or none) keep the old look.
476                    sky_reflection_strength: if self.sky_active {
477                        v.sky_reflection_strength
478                    } else {
479                        0.0
480                    },
481                    // Contact foam is off (strength 0) without a hull.
482                    splash_strength: if self.ocean_contact.is_some() {
483                        v.splash_strength
484                    } else {
485                        0.0
486                    },
487                    splash_width: v.splash_width,
488                    splash_flicker_speed: v.splash_flicker_speed,
489                    hull_a: self
490                        .ocean_contact
491                        .map(|(a, _)| a)
492                        .unwrap_or([0.0, 0.0, 1.0, 0.0]),
493                    hull_b: {
494                        let h = self.ocean_contact.map(|(_, h)| h).unwrap_or([1.0, 1.0]);
495                        [h[0], h[1], v.splash_baseline, v.splash_noise_scale]
496                    },
497                    hull_c: {
498                        let hv = self.ocean_contact_vel;
499                        [hv[0], hv[1], hv[2], v.splash_response]
500                    },
501                    band_wobble: v.band_wobble,
502                    band_dither: v.band_dither,
503                    band_dither_scale: v.band_dither_scale,
504                    rain_ripple: v.rain_ripple,
505                    foam_glow: [
506                        v.foam_glow_color[0],
507                        v.foam_glow_color[1],
508                        v.foam_glow_color[2],
509                        v.foam_glow,
510                    ],
511                    // Reality tear rides the shared post-process fields: the
512                    // remap happens in-shader AFTER the TOD palette lands in
513                    // these uniforms, so it can never race time_of_day.rhai.
514                    mode: self.postprocess_config.render_mode,
515                    mode_mix: self.postprocess_config.mode_mix,
516                    _pad_mode: [0.0; 2],
517                };
518                queue.write_buffer(ubuf, 0, bytemuck::cast_slice(&[uniforms]));
519            }
520        }
521
522        // Sort transparent draws back-to-front (descending distance from camera)
523        {
524            let cam = camera_pos;
525            for draw in &mut self.transparent_draws {
526                let dx = draw.model[3][0] - cam[0];
527                let dy = draw.model[3][1] - cam[1];
528                let dz = draw.model[3][2] - cam[2];
529                draw.sort_depth = dx * dx + dy * dy + dz * dz;
530            }
531            self.transparent_draws.sort_by(|a, b| {
532                b.sort_depth
533                    .partial_cmp(&a.sort_depth)
534                    .unwrap_or(std::cmp::Ordering::Equal)
535            });
536
537            for draw in &mut self.transparent_skinned_draws {
538                let dx = draw.model[3][0] - cam[0];
539                let dy = draw.model[3][1] - cam[1];
540                let dz = draw.model[3][2] - cam[2];
541                draw.sort_depth = dx * dx + dy * dy + dz * dz;
542            }
543            self.transparent_skinned_draws.sort_by(|a, b| {
544                b.sort_depth
545                    .partial_cmp(&a.sort_depth)
546                    .unwrap_or(std::cmp::Ordering::Equal)
547            });
548        }
549
550        // Update wireframe overlay transforms
551        for draw in &self.wireframe_overlay_draws {
552            let uniforms = TransformUniforms {
553                view_proj,
554                model: draw.model,
555                model_inv_transpose: draw.model_inv_transpose,
556                camera_pos,
557                _pad: 0.0,
558            };
559            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
560        }
561
562        // Update billboard uniforms with camera vectors
563        {
564            let cam_right = camera.right_vector();
565            let cam_up = camera.up_vector();
566            let billboard_uniforms = BillboardUniforms {
567                view_proj,
568                camera_right: cam_right,
569                _pad0: 0.0,
570                camera_up: cam_up,
571                _pad1: 0.0,
572            };
573            for draw in &self.billboard_draws {
574                queue.write_buffer(
575                    &draw.billboard_buffer,
576                    0,
577                    bytemuck::cast_slice(&[billboard_uniforms]),
578                );
579            }
580        }
581
582        // Update particle uniforms with camera vectors
583        if let Some(pp) = &self.particle_pipeline {
584            if !self.particle_draws.is_empty() {
585                let cam_right = camera.right_vector();
586                let cam_up = camera.up_vector();
587                let particle_uniforms = ParticleUniforms {
588                    view_proj,
589                    camera_right: cam_right,
590                    _pad0: 0.0,
591                    camera_up: cam_up,
592                    _pad1: 0.0,
593                };
594                queue.write_buffer(
595                    &pp.uniform_buffer,
596                    0,
597                    bytemuck::cast_slice(&[particle_uniforms]),
598                );
599            }
600        }
601
602        // Update sprite2d uniforms with view_proj
603        if let Some(sp) = &self.sprite2d_pipeline {
604            if !self.sprite2d_batches.is_empty() {
605                let sprite2d_uniforms = Sprite2dUniforms { view_proj };
606                queue.write_buffer(
607                    &sp.uniform_buffer,
608                    0,
609                    bytemuck::cast_slice(&[sprite2d_uniforms]),
610                );
611            }
612        }
613
614        // Update normal arrow transforms
615        for draw in &self.normal_arrow_draws {
616            let uniforms = TransformUniforms {
617                view_proj,
618                model: draw.model,
619                model_inv_transpose: draw.model_inv_transpose,
620                camera_pos,
621                _pad: 0.0,
622            };
623            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
624        }
625
626        // Update skeleton overlay transforms
627        for draw in &self.skeleton_overlay_draws {
628            let uniforms = TransformUniforms {
629                view_proj,
630                model: draw.model,
631                model_inv_transpose: draw.model_inv_transpose,
632                camera_pos,
633                _pad: 0.0,
634            };
635            queue.write_buffer(&draw.transform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
636        }
637    }

639    /// Execute the main render pass: skybox → grid → terrain → entities → sprites → particles → debug.
640    pub(super) fn render_main_pass<'a>(
641        &'a self,
642        render_pass: &mut wgpu::RenderPass<'a>,
643        wireframe_only: bool,
644        queue: &wgpu::Queue,
645        camera: &Camera,
646        phase: RenderPhase,
647    ) {
648        // Bind lights once for the entire pass (group 2 is shared)
649        render_pass.set_bind_group(2, &self.light_bind_group, &[]);
650
651        // Render sky at the far plane: the procedural sky (driven by the
652        // `sky` component) replaces the texture skybox when present.
653        let mut sky_drawn = false;
654        if self.sky_active && phase.draws_opaque() {
655            if let (Some(sp), Some(ub), Some(ubg)) = (
656                &self.sky_pipeline,
657                &self.sky_uniform_buffer,
658                &self.sky_uniform_bind_group,
659            ) {
660                let view = camera.view_matrix();
661                let view_rot_only =
662                    [view[0], view[1], view[2], [0.0, 0.0, 0.0, 1.0]];
663                let proj = camera.projection_matrix();
664                let vp = mat4_mul(&proj, &view_rot_only);
665                let inv_vp = mat4_inverse(&vp);
666
667                // Sun disc = the scene's first directional light, so the
668                // drawn sun and the lighting can never disagree.
669                let sun = &self.light_uniforms.directional_lights[0];
670                let p = &self.sky_params;
671                let uniforms = crate::sky_pipeline::SkyUniformsGpu {
672                    inv_view_proj: inv_vp,
673                    sun_dir: sun.direction,
674                    sun_disc_size: p.sun_disc_size,
675                    sun_color: [
676                        sun.color[0] * sun.intensity,
677                        sun.color[1] * sun.intensity,
678                        sun.color[2] * sun.intensity,
679                        1.0,
680                    ],
681                    zenith_color: p.zenith_color,
682                    horizon_color: p.horizon_color,
683                    haze_color: p.haze_color,
684                    cloud_tint: p.cloud_tint,
685                    sun_glow: p.sun_glow,
686                    star_opacity: p.star_opacity,
687                    cloud_coverage: p.cloud_coverage,
688                    cloud_density: p.cloud_density,
689                    cloud_scale: p.cloud_scale,
690                    cloud_drift_x: p.cloud_drift_x,
691                    cloud_drift_y: p.cloud_drift_y,
692                    time: (self.ocean_time % 100_000.0) as f32,
693                };
694                queue.write_buffer(ub, 0, bytemuck::cast_slice(&[uniforms]));
695
696                render_pass.set_pipeline(&sp.pipeline);
697                render_pass.set_bind_group(0, ubg, &[]);
698                render_pass.draw(0..3, 0..1);
699                sky_drawn = true;
700            }
701        }
702
703        // Render texture skybox (before everything else, at the far plane)
704        if !sky_drawn && phase.draws_opaque() {
705            if let (Some(sp), Some(ub), Some(ubg), Some(tbg)) = (
706                &self.skybox_pipeline,
707                &self.skybox_uniform_buffer,
708                &self.skybox_uniform_bind_group,
709                &self.skybox_texture_bind_group,
710            ) {
711                // Build view matrix with translation stripped (rotation only)
712                let view = camera.view_matrix();
713                let view_rot_only = [
714                    view[0],
715                    view[1],
716                    view[2],
717                    [0.0, 0.0, 0.0, 1.0], // zero out translation column
718                ];
719                let proj = camera.projection_matrix();
720                let vp = mat4_mul(&proj, &view_rot_only);
721                let inv_vp = mat4_inverse(&vp);
722
723                queue.write_buffer(
724                    ub,
725                    0,
726                    bytemuck::cast_slice(&[SkyboxUniforms {
727                        inv_view_proj: inv_vp,
728                    }]),
729                );
730
731                render_pass.set_pipeline(&sp.pipeline);
732                render_pass.set_bind_group(0, ubg, &[]);
733                render_pass.set_bind_group(1, tbg, &[]);
734                render_pass.draw(0..3, 0..1);
735            }
736        }
737
738        // Render grid
739        if phase.draws_opaque() {
740        if let Some(grid) = &self.grid_draw {
741            render_pass.set_pipeline(&self.pipeline.line_pipeline);
742            render_pass.set_bind_group(0, &grid.transform_bind_group, &[]);
743            render_pass.set_bind_group(1, &grid.material_bind_group, &[]);
744            render_pass.set_vertex_buffer(0, grid.vertex_buffer.slice(..));
745            render_pass.set_index_buffer(grid.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
746            render_pass.draw_indexed(0..grid.index_count, 0, 0..1);
747        }
748        }
749
750        // In WireframeOnly mode: depth prepass masks outline interior,
751        // then outline, then wireframe lines on top.
752        if wireframe_only {
753            self.render_wireframe_only_pass(render_pass);
754        } else {
755            self.render_normal_pass(render_pass, phase);
756        }
757
758        // Normal arrows pass (always uses line pipeline)
759        if self.debug_state.show_normals && phase.draws_ocean_and_after() {
760            render_pass.set_pipeline(&self.pipeline.line_pipeline);
761            for draw in &self.normal_arrow_draws {
762                render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
763                render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
764                render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
765                render_pass
766                    .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
767                render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
768            }
769        }
770
771        // Skeleton overlay pass (bone lines, always uses line pipeline)
772        if self.debug_state.show_skeleton && phase.draws_ocean_and_after() {
773            render_pass.set_pipeline(&self.pipeline.line_pipeline);
774            for draw in &self.skeleton_overlay_draws {
775                render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
776                render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
777                render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
778                render_pass
779                    .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
780                render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
781            }
782        }
783    }

785    /// Wireframe-only mode rendering: depth prepass + outline + wireframe lines.
786    fn render_wireframe_only_pass<'a>(&'a self, render_pass: &mut wgpu::RenderPass<'a>) {
787        if let Some(sel_id) = self.selected_entity {
788            // Step 1: Depth prepass — write front-face depth (no color)
789            // so the outline interior is blocked by the depth buffer.
790            render_pass.set_pipeline(&self.pipeline.depth_prepass_pipeline);
791            for draw in self
792                .entity_draws
793                .iter()
794                .chain(self.transparent_draws.iter())
795            {
796                if draw.entity_id == Some(sel_id) && !draw.is_wireframe {
797                    render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
798                    render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
799                    render_pass
800                        .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
801                    render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
802                }
803            }
804
805            if let Some(sp) = &self.skinned_pipeline {
806                render_pass.set_pipeline(&sp.depth_prepass_pipeline);
807                for draw in self
808                    .skinned_entity_draws
809                    .iter()
810                    .chain(self.transparent_skinned_draws.iter())
811                {
812                    if draw.entity_id == Some(sel_id) {
813                        render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
814                        render_pass.set_bind_group(3, &draw.bone_bind_group, &[]);
815                        render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
816                        render_pass.set_index_buffer(
817                            draw.index_buffer.slice(..),
818                            wgpu::IndexFormat::Uint32,
819                        );
820                        render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
821                    }
822                }
823            }
824
825            // Step 2: Outline — back faces pushed out, only visible at silhouette
826            render_pass.set_pipeline(&self.pipeline.outline_pipeline);
827            for draw in self
828                .entity_draws
829                .iter()
830                .chain(self.transparent_draws.iter())
831            {
832                if draw.entity_id == Some(sel_id) && !draw.is_wireframe {
833                    render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
834                    render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
835                    render_pass
836                        .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
837                    render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
838                }
839            }
840
841            if let Some(sp) = &self.skinned_pipeline {
842                render_pass.set_pipeline(&sp.outline_pipeline);
843                for draw in self
844                    .skinned_entity_draws
845                    .iter()
846                    .chain(self.transparent_skinned_draws.iter())
847                {
848                    if draw.entity_id == Some(sel_id) {
849                        render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
850                        render_pass.set_bind_group(3, &draw.bone_bind_group, &[]);
851                        render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
852                        render_pass.set_index_buffer(
853                            draw.index_buffer.slice(..),
854                            wgpu::IndexFormat::Uint32,
855                        );
856                        render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
857                    }
858                }
859            }
860        }
861
862        // Step 3: Wireframe lines — use overlay pipeline (LessEqual + bias)
863        // so lines draw on top of the depth prepass.
864        render_pass.set_pipeline(&self.pipeline.overlay_line_pipeline);
865        for draw in &self.wireframe_overlay_draws {
866            render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
867            render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
868            render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
869            render_pass.set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
870            render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
871        }
872
873        // Also render entities that are already wireframe (rooms etc.)
874        for draw in &self.entity_draws {
875            if draw.is_wireframe {
876                render_pass.set_pipeline(&self.pipeline.overlay_line_pipeline);
877                render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
878                render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
879                render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
880                render_pass
881                    .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
882                render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
883            }
884        }
885    }

887    /// Normal mode rendering: terrain → outlines → entities → skinned → billboards → sprites → transparent → particles → wireframe.
888    fn render_normal_pass<'a>(&'a self, render_pass: &mut wgpu::RenderPass<'a>, phase: RenderPhase) {
889        // Terrain rendering (early in pass — fills depth buffer for occlusion)
890        if phase.draws_opaque() {
891        if let (Some(tp), Some(mat_bg)) =
892            (&self.terrain_pipeline, &self.terrain_material_bind_group)
893        {
894            if !self.terrain_draws.is_empty() {
895                render_pass.set_pipeline(&tp.pipeline);
896                render_pass.set_bind_group(1, mat_bg, &[]);
897                render_pass.set_bind_group(2, &self.light_bind_group, &[]);
898
899                for draw in &self.terrain_draws {
900                    if let Some(ref frustum) = self.camera_frustum {
901                        if !frustum.aabb_visible(draw.aabb_min, draw.aabb_max) {
902                            continue;
903                        }
904                    }
905                    render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
906                    render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
907                    render_pass
908                        .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
909                    render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
910                }
911            }
912        }
913
914        // Grass rendering (after terrain -- also writes depth)
915        if let (Some(gp), Some(render_bg), Some(instance_bg)) = (
916            &self.grass_pipeline,
917            &self.grass_render_uniform_bind_group,
918            &self.grass_render_instance_bind_group,
919        ) {
920            if self.grass_instance_count > 0 {
921                // Explicitly set bind group 0 (transform) — cannot rely on terrain having set it
922                if let Some(first_terrain) = self.terrain_draws.first() {
923                    render_pass.set_bind_group(0, &first_terrain.transform_bind_group, &[]);
924                }
925                render_pass.set_pipeline(&gp.render_pipeline);
926                render_pass.set_bind_group(1, render_bg, &[]);
927                render_pass.set_bind_group(2, &self.light_bind_group, &[]);
928                render_pass.set_bind_group(3, instance_bg, &[]);
929                render_pass.set_vertex_buffer(0, gp.blade_vertex_buffer.slice(..));
930                render_pass
931                    .set_index_buffer(gp.blade_index_buffer.slice(..), wgpu::IndexFormat::Uint16);
932                render_pass.draw_indexed(0..BLADE_INDEX_COUNT, 0, 0..self.grass_instance_count);
933            }
934        }
935        } // end opaque phase (terrain + grass)
936
937        // Ocean rendering (after all opaque — writes depth so fog applies;
938        // group 3 holds the grab-pass snapshots, or dummies when disabled)
939        if self.ocean_active && phase.draws_ocean_and_after() {
940            let grab_bg = if self.ocean_grab_this_frame {
941                self.ocean_grab_bind_group.as_ref()
942            } else {
943                self.ocean_grab_dummy_bind_group.as_ref()
944            };
945            if let (Some(op), Some(ocean_bg), Some(transform_bg), Some(grab_bg)) = (
946                &self.ocean_pipeline,
947                &self.ocean_uniform_bind_group,
948                &self.ocean_transform_bind_group,
949                grab_bg,
950            ) {
951                render_pass.set_pipeline(&op.pipeline);
952                render_pass.set_bind_group(0, transform_bg, &[]);
953                render_pass.set_bind_group(1, ocean_bg, &[]);
954                render_pass.set_bind_group(2, &self.light_bind_group, &[]);
955                render_pass.set_bind_group(3, grab_bg, &[]);
956                render_pass.set_vertex_buffer(0, op.grid_vertex_buffer.slice(..));
957                render_pass
958                    .set_index_buffer(op.grid_index_buffer.slice(..), wgpu::IndexFormat::Uint32);
959                render_pass.draw_indexed(0..op.grid_index_count, 0, 0..1);
960            }
961        }
962
963        if phase.draws_opaque() {
964        // Outline pass for selected standard entities (before normal rendering)
965        if let Some(sel_id) = self.selected_entity {
966            render_pass.set_pipeline(&self.pipeline.outline_pipeline);
967            for draw in self
968                .entity_draws
969                .iter()
970                .chain(self.transparent_draws.iter())
971            {
972                if draw.entity_id == Some(sel_id) && !draw.is_wireframe {
973                    render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
974                    render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
975                    render_pass
976                        .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
977                    render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
978                }
979            }
980        }
981
982        // Normal entity rendering
983        for draw in &self.entity_draws {
984            if draw.is_wireframe {
985                render_pass.set_pipeline(&self.pipeline.line_pipeline);
986            } else {
987                render_pass.set_pipeline(&self.pipeline.pipeline);
988            }
989            render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
990            render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
991            render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
992            render_pass.set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
993            render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
994        }
995
996        // Outline pass for selected skinned entities
997        if let Some(sel_id) = self.selected_entity {
998            if let Some(sp) = &self.skinned_pipeline {
999                render_pass.set_pipeline(&sp.outline_pipeline);
1000                for draw in self
1001                    .skinned_entity_draws
1002                    .iter()
1003                    .chain(self.transparent_skinned_draws.iter())
1004                {
1005                    if draw.entity_id == Some(sel_id) {
1006                        render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
1007                        render_pass.set_bind_group(3, &draw.bone_bind_group, &[]);
1008                        render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
1009                        render_pass.set_index_buffer(
1010                            draw.index_buffer.slice(..),
1011                            wgpu::IndexFormat::Uint32,
1012                        );
1013                        render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
1014                    }
1015                }
1016            }
1017        }
1018
1019        // Render skinned entities
1020        if let Some(sp) = &self.skinned_pipeline {
1021            if !self.skinned_entity_draws.is_empty() {
1022                render_pass.set_pipeline(&sp.pipeline);
1023                for draw in &self.skinned_entity_draws {
1024                    render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
1025                    render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
1026                    // Light bind group 2 is already set
1027                    render_pass.set_bind_group(3, &draw.bone_bind_group, &[]);
1028                    render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
1029                    render_pass
1030                        .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1031                    render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
1032                }
1033            }
1034        }
1035
1036        // Outline pass for selected billboards
1037        if let Some(sel_id) = self.selected_entity {
1038            if let Some(bp) = &self.billboard_pipeline {
1039                render_pass.set_pipeline(&bp.outline_pipeline);
1040                for draw in &self.billboard_draws {
1041                    if draw.entity_id == Some(sel_id) {
1042                        render_pass.set_bind_group(0, &draw.billboard_bind_group, &[]);
1043                        render_pass.set_bind_group(1, &draw.texture_bind_group, &[]);
1044                        render_pass.set_index_buffer(
1045                            bp.quad_index_buffer.slice(..),
1046                            wgpu::IndexFormat::Uint32,
1047                        );
1048                        render_pass.draw_indexed(0..6, 0, 0..1);
1049                    }
1050                }
1051            }
1052        }
1053
1054        // Billboard sprites (after geometry, uses depth test)
1055        if let Some(bp) = &self.billboard_pipeline {
1056            if !self.billboard_draws.is_empty() {
1057                render_pass.set_pipeline(&bp.pipeline);
1058                for draw in &self.billboard_draws {
1059                    render_pass.set_bind_group(0, &draw.billboard_bind_group, &[]);
1060                    render_pass.set_bind_group(1, &draw.texture_bind_group, &[]);
1061                    render_pass.set_index_buffer(
1062                        bp.quad_index_buffer.slice(..),
1063                        wgpu::IndexFormat::Uint32,
1064                    );
1065                    render_pass.draw_indexed(0..6, 0, 0..1);
1066                }
1067            }
1068        }
1069        } // end opaque phase (outlines + entities + skinned + billboards)
1070
1071        if !phase.draws_ocean_and_after() {
1072            return;
1073        }
1074
1075        // 2D sprites (after billboards, instanced batched rendering)
1076        if let Some(sp) = &self.sprite2d_pipeline {
1077            if !self.sprite2d_batches.is_empty() {
1078                render_pass.set_pipeline(&sp.pipeline);
1079                render_pass
1080                    .set_index_buffer(sp.quad_index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1081                render_pass.set_bind_group(0, &sp.uniform_bind_group, &[]);
1082                for batch in &self.sprite2d_batches {
1083                    render_pass.set_bind_group(1, &batch.instance_bind_group, &[]);
1084                    render_pass.set_bind_group(2, &batch.texture_bind_group, &[]);
1085                    render_pass.draw_indexed(0..6, 0, 0..batch.instance_count);
1086                }
1087            }
1088        }
1089
1090        // Transparent PBR entities (sorted back-to-front, depth write OFF)
1091        if !self.transparent_draws.is_empty() {
1092            // Re-bind lights for group 2 (may have been displaced by billboard binds)
1093            render_pass.set_bind_group(2, &self.light_bind_group, &[]);
1094
1095            for draw in &self.transparent_draws {
1096                if draw.is_wireframe {
1097                    continue;
1098                }
1099                let pipeline = match draw.blend_mode {
1100                    crate::pipeline::BlendMode::Alpha => &self.pipeline.transparent_alpha_pipeline,
1101                    crate::pipeline::BlendMode::Additive => {
1102                        &self.pipeline.transparent_additive_pipeline
1103                    }
1104                    crate::pipeline::BlendMode::Multiply => {
1105                        &self.pipeline.transparent_multiply_pipeline
1106                    }
1107                };
1108                render_pass.set_pipeline(pipeline);
1109                render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
1110                render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
1111                render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
1112                render_pass
1113                    .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1114                render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
1115            }
1116        }
1117
1118        // Transparent skinned entities (sorted back-to-front, depth write OFF)
1119        if let Some(sp) = &self.skinned_pipeline {
1120            if !self.transparent_skinned_draws.is_empty() {
1121                for draw in &self.transparent_skinned_draws {
1122                    let pipeline = match draw.blend_mode {
1123                        crate::pipeline::BlendMode::Alpha => &sp.transparent_alpha_pipeline,
1124                        crate::pipeline::BlendMode::Additive => &sp.transparent_additive_pipeline,
1125                        crate::pipeline::BlendMode::Multiply => &sp.transparent_multiply_pipeline,
1126                    };
1127                    render_pass.set_pipeline(pipeline);
1128                    render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
1129                    render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
1130                    render_pass.set_bind_group(2, &self.light_bind_group, &[]);
1131                    render_pass.set_bind_group(3, &draw.bone_bind_group, &[]);
1132                    render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
1133                    render_pass
1134                        .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1135                    render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
1136                }
1137            }
1138        }
1139
1140        // Particle systems (after billboards + transparent, before wireframe overlay)
1141        // Alpha-blended particles first, then additive
1142        if let Some(pp) = &self.particle_pipeline {
1143            if !self.particle_draws.is_empty() {
1144                render_pass
1145                    .set_index_buffer(pp.quad_index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1146                render_pass.set_bind_group(0, &pp.uniform_bind_group, &[]);
1147
1148                // Alpha pass
1149                render_pass.set_pipeline(&pp.alpha_pipeline);
1150                for draw in &self.particle_draws {
1151                    if draw.additive {
1152                        continue;
1153                    }
1154                    render_pass.set_bind_group(1, &draw.instance_bind_group, &[]);
1155                    render_pass.set_bind_group(2, &draw.texture_bind_group, &[]);
1156                    render_pass.draw_indexed(0..6, 0, 0..draw.instance_count);
1157                }
1158
1159                // Additive pass
1160                render_pass.set_pipeline(&pp.additive_pipeline);
1161                for draw in &self.particle_draws {
1162                    if !draw.additive {
1163                        continue;
1164                    }
1165                    render_pass.set_bind_group(1, &draw.instance_bind_group, &[]);
1166                    render_pass.set_bind_group(2, &draw.texture_bind_group, &[]);
1167                    render_pass.draw_indexed(0..6, 0, 0..draw.instance_count);
1168                }
1169            }
1170        }
1171
1172        // Wireframe overlay pass (on top of solid geometry)
1173        if self.debug_state.mode == DebugMode::WireframeOverlay {
1174            render_pass.set_pipeline(&self.pipeline.overlay_line_pipeline);
1175            for draw in &self.wireframe_overlay_draws {
1176                render_pass.set_bind_group(0, &draw.transform_bind_group, &[]);
1177                render_pass.set_bind_group(1, &draw.material_bind_group, &[]);
1178                render_pass.set_vertex_buffer(0, draw.vertex_buffer.slice(..));
1179                render_pass
1180                    .set_index_buffer(draw.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1181                render_pass.draw_indexed(0..draw.index_count, 0, 0..1);
1182            }
1183        }
1184    }

1186    /// Run post-processing: SSAO, bloom, composite HDR → sRGB.
1187    pub(super) fn render_postprocess(
1188        &self,
1189        device: &wgpu::Device,
1190        queue: &wgpu::Queue,
1191        depth_view: &wgpu::TextureView,
1192        target_view: &wgpu::TextureView,
1193        camera: &Camera,
1194    ) {
1195        let pp = match &self.postprocess_pipeline {
1196            Some(pp) => pp,
1197            None => return,
1198        };
1199        let resources = match &self.postprocess_resources {
1200            Some(r) => r,
1201            None => return,
1202        };
1203
1204        // Run Kuwahara filter if enabled
1205        if self.postprocess_config.enabled && self.postprocess_config.kuwahara_enabled {
1206            pp.run_kuwahara(device, queue, resources, &self.postprocess_config);
1207        }
1208
1209        // Run SSAO if enabled
1210        if self.postprocess_config.enabled && self.postprocess_config.ssao_enabled {
1211            pp.run_ssao(
1212                device,
1213                queue,
1214                resources,
1215                &self.postprocess_config,
1216                depth_view,
1217                camera,
1218            );
1219        }
1220
1221        // Run volumetric (god rays) if enabled and shadows are available
1222        if self.postprocess_config.enabled && self.postprocess_config.volumetric_enabled {
1223            if let Some(shadow_pass) = &self.shadow_pass {
1224                if shadow_pass.enabled {
1225                    pp.run_volumetric(
1226                        device,
1227                        queue,
1228                        resources,
1229                        &self.postprocess_config,
1230                        depth_view,
1231                        camera,
1232                        &shadow_pass.shadow_view,
1233                        &shadow_pass.shadow_sampler,
1234                        &shadow_pass.shadow_uniforms_buffer,
1235                        &self.light_buffer,
1236                    );
1237                }
1238            }
1239        }
1240
1241        // Run bloom if post-processing effects are enabled
1242        if self.postprocess_config.enabled
1243            && self.postprocess_config.bloom_enabled
1244            && resources.bloom_mip_count > 0
1245        {
1246            pp.run_bloom(device, queue, resources, &self.postprocess_config);
1247        }
1248
1249        // Composite: HDR + bloom + SSAO + volumetric + fog → tonemapped sRGB surface
1250        // (always runs — this is what converts Rgba16Float → surface format)
1251        pp.composite(
1252            device,
1253            queue,
1254            resources,
1255            &self.postprocess_config,
1256            target_view,
1257            depth_view,
1258            camera,
1259            // Same f64 phase wrap as the ocean upload — keeps mode animation
1260            // precise over long sessions. Headless render leaves this 0.0.
1261            (self.ocean_time % 100_000.0) as f32,
1262        );
1263    }

1265    /// Dispatch the grass compute shader to scatter instances.
1266    /// Call before render_main_pass, after update_per_frame_uniforms.
1267    pub fn dispatch_grass_compute(
1268        &mut self,
1269        _device: &wgpu::Device,
1270        queue: &wgpu::Queue,
1271        encoder: &mut wgpu::CommandEncoder,
1272        camera: &Camera,
1273        time: f32,
1274    ) {
1275        let config = match &self.grass_config {
1276            Some(c) if c.enabled => c.clone(),
1277            _ => return,
1278        };
1279
1280        let grass_pipeline = match &self.grass_pipeline {
1281            Some(p) => p,
1282            None => return,
1283        };
1284
1285        let compute_uniform_bg = match &self.grass_compute_uniform_bind_group {
1286            Some(bg) => bg,
1287            None => return,
1288        };
1289        let compute_texture_bg = match &self.grass_compute_texture_bind_group {
1290            Some(bg) => bg,
1291            None => return,
1292        };
1293        let compute_storage_bg = match &self.grass_compute_storage_bind_group {
1294            Some(bg) => bg,
1295            None => return,
1296        };
1297
1298        // Reset atomic counter to 0
1299        if let Some(counter_buf) = &self.grass_counter_buffer {
1300            queue.write_buffer(counter_buf, 0, &[0u8; 4]);
1301        }
1302
1303        // Update compute uniforms
1304        let uniforms = GrassComputeUniforms {
1305            camera_pos: [camera.position.x, camera.position.y, camera.position.z],
1306            time,
1307            terrain_offset: self.grass_terrain_offset,
1308            density: config.density,
1309            terrain_width: self.grass_terrain_width,
1310            terrain_depth: self.grass_terrain_depth,
1311            height_scale: self.grass_terrain_height_scale,
1312            max_distance: config.max_distance,
1313            fade_start: config.fade_start,
1314            density_threshold: config.density_threshold,
1315            density_layer: config.density_layer,
1316            blade_height: config.blade_height,
1317            height_variation: config.height_variation,
1318            _pad0: 0.0,
1319            _pad1: 0.0,
1320            _pad2: 0.0,
1321        };
1322
1323        if let Some(buf) = &self.grass_compute_uniform_buffer {
1324            queue.write_buffer(buf, 0, bytemuck::cast_slice(&[uniforms]));
1325        }
1326
1327        // Update render uniforms (wind, colors, etc.)
1328        let wind_dir = {
1329            let d = config.wind_direction;
1330            let len = (d[0] * d[0] + d[2] * d[2]).sqrt().max(0.001);
1331            [d[0] / len, d[1], d[2] / len]
1332        };
1333
1334        let render_uniforms = GrassRenderUniforms {
1335            wind_direction: wind_dir,
1336            wind_speed: config.wind_speed,
1337            wind_strength: config.wind_strength,
1338            time,
1339            bend_radius: config.bend_radius,
1340            bend_strength: config.bend_strength,
1341            color_base: config.color_base,
1342            blade_width: config.blade_width,
1343            color_tip: config.color_tip,
1344            blade_height: config.blade_height,
1345            color_dry: config.color_dry,
1346            dry_amount: config.dry_amount,
1347            entity_count: self.grass_entity_count,
1348            _pad0: 0.0,
1349            _pad1: 0.0,
1350            _pad2: 0.0,
1351        };
1352
1353        if let Some(buf) = &self.grass_render_uniform_buffer {
1354            queue.write_buffer(buf, 0, bytemuck::cast_slice(&[render_uniforms]));
1355        }
1356
1357        // Compute dispatch
1358        let spacing = 1.0 / config.density.sqrt();
1359        let grid_x = (self.grass_terrain_width / spacing).ceil() as u32;
1360        let grid_z = (self.grass_terrain_depth / spacing).ceil() as u32;
1361        let workgroups_x = (grid_x + 7) / 8;
1362        let workgroups_z = (grid_z + 7) / 8;
1363
1364        {
1365            let mut pass = encoder.begin_compute_pass(&wgpu::ComputePassDescriptor {
1366                label: Some("Grass Compute Pass"),
1367                timestamp_writes: None,
1368            });
1369            pass.set_pipeline(&grass_pipeline.compute_pipeline);
1370            pass.set_bind_group(0, compute_uniform_bg, &[]);
1371            pass.set_bind_group(1, compute_texture_bg, &[]);
1372            pass.set_bind_group(2, compute_storage_bg, &[]);
1373            pass.dispatch_workgroups(workgroups_x, workgroups_z, 1);
1374        }
1375
1376        // Copy counter to staging buffer for CPU readback
1377        if let (Some(counter), Some(staging)) =
1378            (&self.grass_counter_buffer, &self.grass_staging_buffer)
1379        {
1380            encoder.copy_buffer_to_buffer(counter, 0, staging, 0, 4);
1381        }
1382    }

1384    /// Read back the grass instance count from the staging buffer.
1385    /// Call after submitting the command buffer from the previous frame.
1386    pub fn read_grass_instance_count(&mut self, device: &wgpu::Device) {
1387        let staging = match &self.grass_staging_buffer {
1388            Some(s) => s,
1389            None => return,
1390        };
1391
1392        let slice = staging.slice(..);
1393        let (tx, rx) = std::sync::mpsc::channel();
1394        slice.map_async(wgpu::MapMode::Read, move |result| {
1395            let _ = tx.send(result);
1396        });
1397        device.poll(wgpu::Maintain::Wait);
1398
1399        if let Ok(Ok(())) = rx.recv() {
1400            let data = slice.get_mapped_range();
1401            let count = u32::from_le_bytes([data[0], data[1], data[2], data[3]]);
1402            self.grass_instance_count = count.min(self.grass_max_instances);
1403            drop(data);
1404            staging.unmap();
1405        }
1406    }
