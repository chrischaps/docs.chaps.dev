1//! Particle upload bridge and editor helpers (ADR 0068).
2//!
3//! Player, viewer, particle editor and `flint render` all feed particles to
4//! the renderer through [`SceneRenderer::update_particles_from`], so the
5//! `flint-particles` → GPU mapping lives in exactly one place.

19    /// Upload every packed emitter of a [`ParticleSync`]. Call after
20    /// `ParticleSystem::pack`.
21    pub fn update_particles_from(
22        &mut self,
23        device: &wgpu::Device,
24        queue: &wgpu::Queue,
25        sync: &ParticleSync,
26    ) {
27        let draws: Vec<ParticleDrawData<'_>> = sync
28            .draw_data()
29            .into_iter()
30            .map(|d| ParticleDrawData {
31                instances: d.instances,
32                texture: d.texture,
33                blend: d.blend_mode,
34                sort_key: d.sort_key,
35            })
36            .collect();
37        self.update_particles(device, queue, &draws);
38    }

40    /// Low-level upload: concatenate all emitters' instances into the shared
41    /// storage buffer (one `write_buffer`) and build sorted draw calls.
42    pub fn update_particles(
43        &mut self,
44        device: &wgpu::Device,
45        queue: &wgpu::Queue,
46        draws: &[ParticleDrawData<'_>],
47    ) {
48        self.particle_draws.clear();
49        if self.particle_pipeline.is_none() {
50            return;
51        }
52        self.ensure_soft_disc_texture(device, queue);
53
54        let total: usize = draws.iter().map(|d| d.instances.len()).sum();
55        if total == 0 {
56            return;
57        }
58
59        let mut scratch = std::mem::take(&mut self.particle_upload_scratch);
60        scratch.clear();
61        scratch.reserve(total);
62
63        let mut calls = Vec::with_capacity(draws.len());
64        for data in draws {
65            if data.instances.is_empty() {
66                continue;
67            }
68            let first_instance = scratch.len() as u32;
69            scratch.extend_from_slice(data.instances);
70            let texture_bind_group = self.particle_texture_bind_group(device, data.texture);
71            calls.push(ParticleDrawCall {
72                first_instance,
73                instance_count: data.instances.len() as u32,
74                texture_bind_group,
75                blend: data.blend,
76                sort_key: data.sort_key,
77            });
78        }
79
80        if let Some(pp) = &mut self.particle_pipeline {
81            pp.ensure_capacity(device, scratch.len() as u32);
82            pp.write_instances(queue, &scratch);
83        }
84        sort_particle_draws(&mut calls);
85        self.particle_draws = calls;
86        self.particle_upload_scratch = scratch;
87    }

89    /// Cached texture bind group for a particle sprite (white when the name
90    /// is empty or not loaded).
91    fn particle_texture_bind_group(
92        &mut self,
93        device: &wgpu::Device,
94        texture: &str,
95    ) -> Arc<wgpu::BindGroup> {
96        let key = if !texture.is_empty()
97            && self
98                .texture_cache
99                .as_ref()
100                .is_some_and(|tc| tc.contains(texture))
101        {
102            texture
103        } else {
104            ""
105        };
106        if let Some(bg) = self.particle_texture_bind_groups.get(key) {
107            return bg.clone();
108        }
109        let pp = self
110            .particle_pipeline
111            .as_ref()
112            .expect("particle pipeline present");
113        let tc = self
114            .texture_cache
115            .as_ref()
116            .expect("TextureCache required for particle rendering");
117        let tex = if key.is_empty() {
118            tc.get(SOFT_DISC_TEXTURE).unwrap_or(&tc.default_white)
119        } else {
120            tc.get(key).expect("checked above")
121        };
122        let bg = Arc::new(pp.create_texture_bind_group(
123            device,
124            &tex.view,
125            &tex.sampler,
126            &format!("Particle Texture Bind Group ({key})"),
127        ));
128        self.particle_texture_bind_groups
129            .insert(key.to_string(), bg.clone());
130        bg
131    }

133    /// Upload the procedural soft-disc sprite used when an emitter names no
134    /// texture: a radial falloff so untextured particles read as glows,
135    /// not squares (ADR 0068). Idempotent.
136    fn ensure_soft_disc_texture(&mut self, device: &wgpu::Device, queue: &wgpu::Queue) {
137        let already = self
138            .texture_cache
139            .as_ref()
140            .is_some_and(|tc| tc.contains(SOFT_DISC_TEXTURE));
141        if already {
142            return;
143        }
144        const N: u32 = 64;
145        let mut data = Vec::with_capacity((N * N * 4) as usize);
146        for y in 0..N {
147            for x in 0..N {
148                let u = (x as f32 + 0.5) / N as f32 * 2.0 - 1.0;
149                let v = (y as f32 + 0.5) / N as f32 * 2.0 - 1.0;
150                let r = (u * u + v * v).sqrt();
151                // Smooth falloff: solid core, feathered edge.
152                let a = (1.0 - ((r - 0.35) / 0.65).clamp(0.0, 1.0)).powf(1.6);
153                data.extend_from_slice(&[255, 255, 255, (a * 255.0).round() as u8]);
154            }
155        }
156        if let Err(e) = self.load_texture_rgba(device, queue, SOFT_DISC_TEXTURE, N, N, &data, false)
157        {
158            tracing::warn!("soft disc particle texture upload failed: {e}");
159        }
160    }

167    /// Editor/gizmo overlay: a line-list mesh drawn on top of the scene
168    /// regardless of debug mode (unlike the skeleton overlay).
169    pub fn set_debug_overlay(&mut self, device: &wgpu::Device, mesh: &Mesh) {
170        self.debug_overlay_draws.clear();
171        if mesh.indices.is_empty() {
172            return;
173        }
174        let tex_cache = self.texture_cache.as_ref().expect("texture cache");
175        let draw: DrawCall = Self::create_draw_call(
176            device,
177            &self.pipeline,
178            mesh,
179            true,
180            TransformUniforms::new(),
181            MaterialUniforms::procedural(),
182            tex_cache,
183        );
184        self.debug_overlay_draws.push(draw);
185    }
186
187    pub fn clear_debug_overlay(&mut self) {
188        self.debug_overlay_draws.clear();
189    }

205    /// Whether, once `render` returns, the context's depth view holds the
206    /// frame's scene depth, single sample, for overlays drawn against it
207    /// such as world text (ADR 0104). It doesn't with MSAA and no HDR
208    /// chain, where the multisampled depth is never resolved.
209    pub fn frame_depth_ready(&self) -> bool {
210        let has_postprocess =
211            self.postprocess_pipeline.is_some() && self.postprocess_resources.is_some();
212        !self.device_lost && (has_postprocess || self.sample_count == 1)
213    }

226/// Load every texture referenced by the live particle instances and
227/// registered effects into the renderer's texture cache. Names are cache
228/// keys verbatim; files are searched in `search_dirs` in order.
229pub fn load_particle_textures(
230    renderer: &mut SceneRenderer,
231    device: &wgpu::Device,
232    queue: &wgpu::Queue,
233    sync: &ParticleSync,
234    search_dirs: &[PathBuf],
235) {
236    let mut any_new = false;
237    for tex_name in sync.texture_names() {
238        let mut found = false;
239        for dir in search_dirs {
240            let path = dir.join(&tex_name);
241            if flint_core::vfs::exists(&path) {
242                match renderer.load_texture_file(device, queue, &tex_name, &path) {
243                    Ok(true) => {
244                        println!("Loaded particle texture: {tex_name}");
245                        any_new = true;
246                    }
247                    Ok(false) => {} // already cached
248                    Err(e) => {
249                        tracing::warn!("Failed to load particle texture '{}': {}", tex_name, e)
250                    }
251                }
252                found = true;
253                break;
254            }
255        }
256        if !found {
257            tracing::warn!("Particle texture not found: {}", tex_name);
258        }
259    }
260    if any_new {
261        renderer.invalidate_particle_textures();
262    }
