1//! GPU-instanced 2D sprite render pipeline
2//!
3//! Renders flat XY-plane quads via instanced draw calls using a storage buffer.
4//! Follows the particle pipeline pattern: no vertex buffers, quads generated
5//! from `vertex_index` in the shader. One draw call per texture batch.
6//!
7//! Key differences from billboard pipeline:
8//! - Sprites are axis-aligned on the XY plane (not camera-facing)
9//! - No `camera_right`/`camera_up` needed — just `view_proj`
10//! - Storage buffer instancing for efficient batched rendering
11//! - Layer-based Z-offset for draw ordering (depth write disabled)
12
13use bytemuck::{Pod, Zeroable};
14use wgpu::util::DeviceExt;
15
16/// GPU instance data for a single 2D sprite — 80 bytes, 16-byte aligned (5 x vec4).
17#[repr(C)]
18#[derive(Copy, Clone, Debug, Pod, Zeroable)]
19pub struct Sprite2dInstanceGpu {
20    /// xy = world position, z = anchor_x (0-1), w = layer z-offset
21    pub pos_layer: [f32; 4],
22    /// x = width, y = height, z = anchor_y (0-1), w = unused
23    pub size: [f32; 4],
24    /// UV rect: [u_min, v_min, u_max, v_max]
25    pub uv_rect: [f32; 4],
26    /// RGBA tint color multiplier
27    pub tint: [f32; 4],
28    /// x = flip_x (0/1), y = flip_y (0/1), z = unused, w = unused
29    pub flags: [f32; 4],
30}
31
32/// Camera uniforms for the 2D sprite pipeline — just view_proj, no billboard vectors.
33#[repr(C)]
34#[derive(Copy, Clone, Debug, Pod, Zeroable)]
35pub struct Sprite2dUniforms {
36    pub view_proj: [[f32; 4]; 4],
37}
38
39/// A single 2D sprite batch draw call (one per texture group).
40pub struct Sprite2dBatch {
41    pub instance_buffer: wgpu::Buffer,
42    pub instance_count: u32,
43    pub texture_bind_group: wgpu::BindGroup,
44    pub instance_bind_group: wgpu::BindGroup,
45}
46
47/// The 2D sprite rendering pipeline.
48pub struct Sprite2dPipeline {
49    pub pipeline: wgpu::RenderPipeline,
50    pub uniform_bind_group_layout: wgpu::BindGroupLayout,
51    pub instance_bind_group_layout: wgpu::BindGroupLayout,
52    pub texture_bind_group_layout: wgpu::BindGroupLayout,
53    pub quad_index_buffer: wgpu::Buffer,
54    pub uniform_buffer: wgpu::Buffer,
55    pub uniform_bind_group: wgpu::BindGroup,
56}
57
58impl Sprite2dPipeline {
59    pub fn new(device: &wgpu::Device, format: wgpu::TextureFormat, sample_count: u32) -> Self {
60        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
61            label: Some("Sprite2D Shader"),
62            source: wgpu::ShaderSource::Wgsl(include_str!("sprite2d_shader.wgsl").into()),
63        });
64
65        // Group 0: Sprite2dUniforms (view_proj)
66        let uniform_bind_group_layout =
67            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
68                entries: &[wgpu::BindGroupLayoutEntry {
69                    binding: 0,
70                    visibility: wgpu::ShaderStages::VERTEX,
71                    ty: wgpu::BindingType::Buffer {
72                        ty: wgpu::BufferBindingType::Uniform,
73                        has_dynamic_offset: false,
74                        min_binding_size: None,
75                    },
76                    count: None,
77                }],
78                label: Some("Sprite2D Uniform Bind Group Layout"),
79            });
80
81        // Group 1: Instance storage buffer (read-only)
82        let instance_bind_group_layout =
83            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
84                entries: &[wgpu::BindGroupLayoutEntry {
85                    binding: 0,
86                    visibility: wgpu::ShaderStages::VERTEX,
87                    ty: wgpu::BindingType::Buffer {
88                        ty: wgpu::BufferBindingType::Storage { read_only: true },
89                        has_dynamic_offset: false,
90                        min_binding_size: None,
91                    },
92                    count: None,
93                }],
94                label: Some("Sprite2D Instance Bind Group Layout"),
95            });
96
97        // Group 2: Texture + sampler
98        let texture_bind_group_layout =
99            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
100                entries: &[
101                    wgpu::BindGroupLayoutEntry {
102                        binding: 0,
103                        visibility: wgpu::ShaderStages::FRAGMENT,
104                        ty: wgpu::BindingType::Texture {
105                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
106                            view_dimension: wgpu::TextureViewDimension::D2,
107                            multisampled: false,
108                        },
109                        count: None,
110                    },
111                    wgpu::BindGroupLayoutEntry {
112                        binding: 1,
113                        visibility: wgpu::ShaderStages::FRAGMENT,
114                        ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
115                        count: None,
116                    },
117                ],
118                label: Some("Sprite2D Texture Bind Group Layout"),
119            });
120
121        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
122            label: Some("Sprite2D Pipeline Layout"),
123            bind_group_layouts: &[
124                &uniform_bind_group_layout,
125                &instance_bind_group_layout,
126                &texture_bind_group_layout,
127            ],
128            push_constant_ranges: &[],
129        });
130
131        // Alpha blending, depth test enabled but depth write DISABLED
132        // (2D sprites rely on layer-based draw order, not depth buffer writes)
133        let depth_stencil = wgpu::DepthStencilState {
134            format: wgpu::TextureFormat::Depth32Float,
135            depth_write_enabled: false,
136            depth_compare: wgpu::CompareFunction::LessEqual,
137            stencil: wgpu::StencilState::default(),
138            bias: wgpu::DepthBiasState::default(),
139        };
140
141        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
142            label: Some("Sprite2D Pipeline"),
143            layout: Some(&pipeline_layout),
144            vertex: wgpu::VertexState {
145                module: &shader,
146                entry_point: Some("vs_sprite2d"),
147                buffers: &[],
148                compilation_options: Default::default(),
149            },
150            fragment: Some(wgpu::FragmentState {
151                module: &shader,
152                entry_point: Some("fs_sprite2d"),
153                targets: &[Some(wgpu::ColorTargetState {
154                    format,
155                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
156                    write_mask: wgpu::ColorWrites::ALL,
157                })],
158                compilation_options: Default::default(),
159            }),
160            primitive: wgpu::PrimitiveState {
161                topology: wgpu::PrimitiveTopology::TriangleList,
162                strip_index_format: None,
163                front_face: wgpu::FrontFace::Ccw,
164                cull_mode: None,
165                polygon_mode: wgpu::PolygonMode::Fill,
166                unclipped_depth: false,
167                conservative: false,
168            },
169            depth_stencil: Some(depth_stencil),
170            multisample: wgpu::MultisampleState {
171                count: sample_count,
172                ..Default::default()
173            },
174            multiview: None,
175            cache: None,
176        });
177
178        // Shared quad index buffer (same as particle pipeline)
179        let quad_indices: [u32; 6] = [0, 1, 2, 2, 1, 3];
180        let quad_index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
181            label: Some("Sprite2D Quad Index Buffer"),
182            contents: bytemuck::cast_slice(&quad_indices),
183            usage: wgpu::BufferUsages::INDEX,
184        });
185
186        // Persistent uniform buffer for view_proj
187        let uniform_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
188            label: Some("Sprite2D Uniform Buffer"),
189            contents: bytemuck::cast_slice(&[Sprite2dUniforms {
190                view_proj: [[0.0; 4]; 4],
191            }]),
192            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
193        });
194
195        let uniform_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
196            layout: &uniform_bind_group_layout,
197            entries: &[wgpu::BindGroupEntry {
198                binding: 0,
199                resource: uniform_buffer.as_entire_binding(),
200            }],
201            label: Some("Sprite2D Uniform Bind Group"),
202        });
203
204        Self {
205            pipeline,
206            uniform_bind_group_layout,
207            instance_bind_group_layout,
208            texture_bind_group_layout,
209            quad_index_buffer,
210            uniform_buffer,
211            uniform_bind_group,
212        }
213    }
214}
