10/// A single GPU-resident mesh primitive with its material data
11pub struct GpuMesh {
12    /// Shared with every draw call that renders this mesh: static geometry
13    /// is never written after upload, so entities hold `Arc` clones instead
14    /// of re-uploading a private copy every frame.
15    pub vertex_buffer: Arc<wgpu::Buffer>,
16    pub index_buffer: Arc<wgpu::Buffer>,
17    pub index_count: u32,
18    pub material: ImportedMaterial,
19    /// Local-space bounds of the vertices, for frustum culling.
20    pub aabb_min: [f32; 3],
21    pub aabb_max: [f32; 3],
22    /// The camera distances this mesh draws at (its glTF `extras.flint_lod`);
23    /// `None` draws at every distance.
24    pub lod: Option<flint_import::LodBand>,
25    // Keep raw data for creating per-entity copies
26    vertex_data: Vec<u8>,
27    index_data: Vec<u8>,
28}

30/// Local-space bounds of `vertices`; all zeros when there are none.
31pub fn vertex_bounds(vertices: &[Vertex]) -> ([f32; 3], [f32; 3]) {
32    if vertices.is_empty() {
33        return ([0.0; 3], [0.0; 3]);
34    }
35    let mut min = [f32::INFINITY; 3];
36    let mut max = [f32::NEG_INFINITY; 3];
37    for v in vertices {
38        for i in 0..3 {
39            min[i] = min[i].min(v.position[i]);
40            max[i] = max[i].max(v.position[i]);
41        }
42    }
43    (min, max)
44}

115    /// Upload an imported model's meshes to the GPU
116    pub fn upload_imported(
117        &mut self,
118        device: &wgpu::Device,
119        name: &str,
120        import_result: &ImportResult,
121        default_color: [f32; 4],
122    ) {
123        let default_material = ImportedMaterial {
124            name: "default".to_string(),
125            base_color: default_color,
126            metallic: 0.0,
127            roughness: 0.5,
128            base_color_texture: None,
129            normal_texture: None,
130            metallic_roughness_texture: None,
131            use_vertex_color: false,
132            alpha_mode: flint_import::AlphaMode::Opaque,
133            alpha_cutoff: 0.5,
134            ..Default::default()
135        };
136
137        let gpu_meshes: Vec<GpuMesh> = import_result
138            .meshes
139            .iter()
140            .map(|mesh| {
141                let material = mesh
142                    .material_index
143                    .and_then(|i| import_result.materials.get(i))
144                    .cloned()
145                    .unwrap_or_else(|| default_material.clone());
146
147                let vertex_count = mesh.positions.len();
148                let vertices: Vec<Vertex> = (0..vertex_count)
149                    .map(|i| {
150                        let position = mesh.positions[i];
151                        let normal = if i < mesh.normals.len() {
152                            mesh.normals[i]
153                        } else {
154                            [0.0, 1.0, 0.0]
155                        };
156                        let uv = if i < mesh.uvs.len() {
157                            mesh.uvs[i]
158                        } else {
159                            [0.0, 0.0]
160                        };
161
162                        Vertex {
163                            position,
164                            normal,
165                            color: material.base_color,
166                            uv,
167                        }
168                    })
169                    .collect();
170
171                let vertex_data = bytemuck::cast_slice(&vertices).to_vec();
172                let index_data = bytemuck::cast_slice(&mesh.indices).to_vec();
173
174                let vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
175                    label: Some(&format!("{} Vertex Buffer", name)),
176                    contents: &vertex_data,
177                    usage: wgpu::BufferUsages::VERTEX,
178                });
179
180                let index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
181                    label: Some(&format!("{} Index Buffer", name)),
182                    contents: &index_data,
183                    usage: wgpu::BufferUsages::INDEX,
184                });
185
186                let (aabb_min, aabb_max) = vertex_bounds(&vertices);
187                GpuMesh {
188                    vertex_buffer: Arc::new(vertex_buffer),
189                    index_buffer: Arc::new(index_buffer),
190                    index_count: mesh.indices.len() as u32,
191                    material,
192                    aabb_min,
193                    aabb_max,
194                    lod: mesh.lod,
195                    vertex_data,
196                    index_data,
197                }
198            })
199            .collect();
200
201        self.procedural.remove(name);
202        self.meshes.insert(name.to_string(), gpu_meshes);
203    }

205    /// Upload a subset of an imported model's meshes to the GPU, identified by mesh indices.
206    /// Used to upload individual node's primitives under a node-specific cache key.
207    /// If `bake_transform` is provided, vertex positions and normals are transformed
208    /// by the given 4x4 column-major matrix before upload (used to flatten hierarchies
209    /// with non-uniform scale during multi-node GLB expansion).
210    pub fn upload_mesh_subset(
211        &mut self,
212        device: &wgpu::Device,
213        name: &str,
214        import_result: &ImportResult,
215        mesh_indices: &[usize],
216        default_color: [f32; 4],
217        bake_transform: Option<&[[f32; 4]; 4]>,
218    ) {
219        // Compute the normal matrix (inverse-transpose of upper-left 3x3) if baking
220        let normal_mat = bake_transform.map(|m| {
221            let a = m[0][0];
222            let b = m[1][0];
223            let c = m[2][0];
224            let d = m[0][1];
225            let e = m[1][1];
226            let f = m[2][1];
227            let g = m[0][2];
228            let h = m[1][2];
229            let i = m[2][2];
230            let det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g);
231            let inv_det = if det.abs() > 1e-10 { 1.0 / det } else { 1.0 };
232            // Cofactor matrix rows (= inverse-transpose columns)
233            [
234                [
235                    (e * i - f * h) * inv_det,
236                    (f * g - d * i) * inv_det,
237                    (d * h - e * g) * inv_det,
238                ],
239                [
240                    (c * h - b * i) * inv_det,
241                    (a * i - c * g) * inv_det,
242                    (b * g - a * h) * inv_det,
243                ],
244                [
245                    (b * f - c * e) * inv_det,
246                    (c * d - a * f) * inv_det,
247                    (a * e - b * d) * inv_det,
248                ],
249            ]
250        });
251
252        let default_material = ImportedMaterial {
253            name: "default".to_string(),
254            base_color: default_color,
255            metallic: 0.0,
256            roughness: 0.5,
257            base_color_texture: None,
258            normal_texture: None,
259            metallic_roughness_texture: None,
260            use_vertex_color: false,
261            alpha_mode: flint_import::AlphaMode::Opaque,
262            alpha_cutoff: 0.5,
263            ..Default::default()
264        };
265
266        let gpu_meshes: Vec<GpuMesh> = mesh_indices
267            .iter()
268            .filter_map(|&idx| import_result.meshes.get(idx))
269            .map(|mesh| {
270                let material = mesh
271                    .material_index
272                    .and_then(|i| import_result.materials.get(i))
273                    .cloned()
274                    .unwrap_or_else(|| default_material.clone());
275
276                let vertex_count = mesh.positions.len();
277                let vertices: Vec<Vertex> = (0..vertex_count)
278                    .map(|i| {
279                        let mut position = mesh.positions[i];
280                        let mut normal = if i < mesh.normals.len() {
281                            mesh.normals[i]
282                        } else {
283                            [0.0, 1.0, 0.0]
284                        };
285                        let uv = if i < mesh.uvs.len() {
286                            mesh.uvs[i]
287                        } else {
288                            [0.0, 0.0]
289                        };
290
291                        // Bake transform into vertex data if provided
292                        if let Some(m) = bake_transform {
293                            let [px, py, pz] = position;
294                            position = [
295                                m[0][0] * px + m[1][0] * py + m[2][0] * pz + m[3][0],
296                                m[0][1] * px + m[1][1] * py + m[2][1] * pz + m[3][1],
297                                m[0][2] * px + m[1][2] * py + m[2][2] * pz + m[3][2],
298                            ];
299                        }
300                        if let Some(nm) = &normal_mat {
301                            let [nx, ny, nz] = normal;
302                            let tnx = nm[0][0] * nx + nm[1][0] * ny + nm[2][0] * nz;
303                            let tny = nm[0][1] * nx + nm[1][1] * ny + nm[2][1] * nz;
304                            let tnz = nm[0][2] * nx + nm[1][2] * ny + nm[2][2] * nz;
305                            let len = (tnx * tnx + tny * tny + tnz * tnz).sqrt();
306                            if len > 1e-6 {
307                                normal = [tnx / len, tny / len, tnz / len];
308                            }
309                        }
310
311                        Vertex {
312                            position,
313                            normal,
314                            color: material.base_color,
315                            uv,
316                        }
317                    })
318                    .collect();
319
320                let vertex_data = bytemuck::cast_slice(&vertices).to_vec();
321                let index_data = bytemuck::cast_slice(&mesh.indices).to_vec();
322
323                let vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
324                    label: Some(&format!("{} Vertex Buffer", name)),
325                    contents: &vertex_data,
326                    usage: wgpu::BufferUsages::VERTEX,
327                });
328
329                let index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
330                    label: Some(&format!("{} Index Buffer", name)),
331                    contents: &index_data,
332                    usage: wgpu::BufferUsages::INDEX,
333                });
334
335                let (aabb_min, aabb_max) = vertex_bounds(&vertices);
336                GpuMesh {
337                    vertex_buffer: Arc::new(vertex_buffer),
338                    index_buffer: Arc::new(index_buffer),
339                    index_count: mesh.indices.len() as u32,
340                    material,
341                    aabb_min,
342                    aabb_max,
343                    lod: mesh.lod,
344                    vertex_data,
345                    index_data,
346                }
347            })
348            .collect();
349
350        self.procedural.remove(name);
351        self.meshes.insert(name.to_string(), gpu_meshes);
352    }

369    /// Upload skinned meshes from an imported model to the GPU
370    pub fn upload_skinned(
371        &mut self,
372        device: &wgpu::Device,
373        name: &str,
374        import_result: &ImportResult,
375        default_color: [f32; 4],
376        bone_bind_group_layout: &wgpu::BindGroupLayout,
377    ) {
378        let default_material = ImportedMaterial {
379            name: "default".to_string(),
380            base_color: default_color,
381            metallic: 0.0,
382            roughness: 0.5,
383            base_color_texture: None,
384            normal_texture: None,
385            metallic_roughness_texture: None,
386            use_vertex_color: false,
387            alpha_mode: flint_import::AlphaMode::Opaque,
388            alpha_cutoff: 0.5,
389            ..Default::default()
390        };
391
392        let identity_bones: Vec<[[f32; 4]; 4]> = vec![
393            [
394                [1.0, 0.0, 0.0, 0.0],
395                [0.0, 1.0, 0.0, 0.0],
396                [0.0, 0.0, 1.0, 0.0],
397                [0.0, 0.0, 0.0, 1.0],
398            ];
399            MAX_BONES
400        ];
401        // Create bone storage buffer initialized to identity
402        let bone_buffer = Arc::new(
403            device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
404                label: Some(&format!("{} Bone Buffer", name)),
405                contents: bytemuck::cast_slice(&identity_bones),
406                usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
407            }),
408        );
409
410        let gpu_skinned: Vec<GpuSkinnedMesh> = import_result
411            .meshes
412            .iter()
413            .filter(|mesh| mesh.joint_indices.is_some() && mesh.joint_weights.is_some())
414            .map(|mesh| {
415                let material = mesh
416                    .material_index
417                    .and_then(|i| import_result.materials.get(i))
418                    .cloned()
419                    .unwrap_or_else(|| default_material.clone());
420
421                let skin_index = mesh.skin_index.unwrap_or(0);
422                let vertex_count = mesh.positions.len();
423                let joints = mesh.joint_indices.as_ref().unwrap();
424                let weights = mesh.joint_weights.as_ref().unwrap();
425
426                let vertices: Vec<SkinnedVertex> = (0..vertex_count)
427                    .map(|i| {
428                        let position = mesh.positions[i];
429                        let normal = if i < mesh.normals.len() {
430                            mesh.normals[i]
431                        } else {
432                            [0.0, 1.0, 0.0]
433                        };
434                        let uv = if i < mesh.uvs.len() {
435                            mesh.uvs[i]
436                        } else {
437                            [0.0, 0.0]
438                        };
439                        let ji = if i < joints.len() {
440                            [
441                                joints[i][0] as u32,
442                                joints[i][1] as u32,
443                                joints[i][2] as u32,
444                                joints[i][3] as u32,
445                            ]
446                        } else {
447                            [0, 0, 0, 0]
448                        };
449                        let jw = if i < weights.len() {
450                            weights[i]
451                        } else {
452                            [1.0, 0.0, 0.0, 0.0]
453                        };
454
455                        SkinnedVertex {
456                            position,
457                            normal,
458                            color: material.base_color,
459                            uv,
460                            joint_indices: ji,
461                            joint_weights: jw,
462                        }
463                    })
464                    .collect();
465
466                let index_data = bytemuck::cast_slice(&mesh.indices).to_vec();
467
468                let vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
469                    label: Some(&format!("{} Skinned Vertex Buffer", name)),
470                    contents: bytemuck::cast_slice(&vertices),
471                    usage: wgpu::BufferUsages::VERTEX,
472                });
473
474                let index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
475                    label: Some(&format!("{} Skinned Index Buffer", name)),
476                    contents: &index_data,
477                    usage: wgpu::BufferUsages::INDEX,
478                });
479
480                let bone_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
481                    layout: bone_bind_group_layout,
482                    entries: &[wgpu::BindGroupEntry {
483                        binding: 0,
484                        resource: bone_buffer.as_entire_binding(),
485                    }],
486                    label: Some(&format!("{} Bone Bind Group", name)),
487                });
488
489                GpuSkinnedMesh {
490                    vertex_buffer: Arc::new(vertex_buffer),
491                    index_buffer: Arc::new(index_buffer),
492                    index_count: mesh.indices.len() as u32,
493                    material,
494                    bone_buffer: bone_buffer.clone(),
495                    bone_bind_group,
496                    skin_index,
497                    index_data,
498                }
499            })
500            .collect();
501
502        if !gpu_skinned.is_empty() {
503            self.skinned_meshes.insert(name.to_string(), gpu_skinned);
504        }
505    }

534    /// Upload a procedural mesh made of several parts, each with its own
535    /// material (a tree's bark and leaves), all under `name`, replacing any
536    /// mesh already cached there. Draws the way a multi-primitive glTF does.
537    pub fn upload_procedural_parts(
538        &mut self,
539        device: &wgpu::Device,
540        name: &str,
541        parts: Vec<(Vec<Vertex>, Vec<u32>, ImportedMaterial)>,
542    ) {
543        let meshes = parts
544            .into_iter()
545            .map(|(vertices, indices, material)| {
546                Self::procedural_gpu_mesh(device, name, &vertices, &indices, material)
547            })
548            .collect();
549        // Replacing drops the old buffers here; draw lists still holding
550        // them let go at the next `update_from_world`.
551        self.procedural.insert(name.to_string());
552        self.meshes.insert(name.to_string(), meshes);
553    }

560    fn procedural_gpu_mesh(
561        device: &wgpu::Device,
562        name: &str,
563        vertices: &[Vertex],
564        indices: &[u32],
565        material: ImportedMaterial,
566    ) -> GpuMesh {
567        let vertex_data = bytemuck::cast_slice(vertices).to_vec();
568        let index_data = bytemuck::cast_slice(indices).to_vec();
569
570        let vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
571            label: Some(&format!("{} Procedural Vertex Buffer", name)),
572            contents: &vertex_data,
573            usage: wgpu::BufferUsages::VERTEX,
574        });
575
576        let index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
577            label: Some(&format!("{} Procedural Index Buffer", name)),
578            contents: &index_data,
579            usage: wgpu::BufferUsages::INDEX,
580        });
581
582        let (aabb_min, aabb_max) = vertex_bounds(vertices);
583        GpuMesh {
584            vertex_buffer: Arc::new(vertex_buffer),
585            index_buffer: Arc::new(index_buffer),
586            index_count: indices.len() as u32,
587            material,
588            aabb_min,
589            aabb_max,
590            lod: None,
591            vertex_data,
592            index_data,
593        }
594    }

596    /// Remove a mesh (static and skinned) by name. Returns whether it
597    /// existed. Its GPU buffers are freed once the current frame's draw
598    /// lists, which still hold them, are rebuilt.
599    pub fn remove(&mut self, name: &str) -> bool {
600        self.procedural.remove(name);
601        let had_static = self.meshes.remove(name).is_some();
602        let had_skinned = self.skinned_meshes.remove(name).is_some();
603        had_static || had_skinned
604    }
605
606    /// Number of named meshes (static and skinned) in the cache.
607    pub fn len(&self) -> usize {
608        self.meshes.len() + self.skinned_meshes.len()
609    }
610
611    /// Whether the cache holds no meshes.
612    pub fn is_empty(&self) -> bool {
613        self.meshes.is_empty() && self.skinned_meshes.is_empty()
614    }
615
616    /// Check if a model has skinned meshes cached
617    pub fn contains_skinned(&self, name: &str) -> bool {
618        self.skinned_meshes.contains_key(name)
619    }
620
621    /// Clear all cached meshes (static and skinned)
622    pub fn clear(&mut self) {
623        self.meshes.clear();
624        self.procedural.clear();
625        self.skinned_meshes.clear();
626    }

633    #[test]
634    fn vertex_bounds_span_every_position() {
635        let v = |p: [f32; 3]| Vertex {
636            position: p,
637            normal: [0.0, 1.0, 0.0],
638            color: [1.0; 4],
639            uv: [0.0; 2],
640        };
641        let (min, max) = vertex_bounds(&[v([1.0, -2.0, 3.0]), v([-4.0, 5.0, 0.5])]);
