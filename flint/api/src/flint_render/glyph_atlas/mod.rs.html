1//! Glyph atlas: monospaced glyphs rasterized into one coverage texture, for
2//! the ASCII styles and any later text-based look (#264, epic #260).
3//!
4//! A [`GlyphAtlas`] is a grid of equal cells, one glyph each, in an 8-bit
5//! coverage image (0 paper, 255 ink) with a one-pixel gutter round every cell
6//! so a filtered sample never bleeds into a neighbour. It comes from one of
7//! three sources:
8//!
9//! - the built-in 8x8 set ([`GlyphAtlas::builtin`]), drawn by hand for the
10//!   engine in `builtin_8x8.txt` and free of any font licence;
11//! - a TTF/OTF rasterized at a pixel height ([`GlyphAtlas::from_ttf`]);
12//! - a bitmap grid image, one glyph per cell ([`GlyphAtlas::from_grid_image`]).
13//!
14//! A [`GlyphSet`] adds what a style needs on top: named luminance ramps and
15//! the edge glyphs, read from a `*.glyphs.toml` (see [`set`]). Everything is
16//! read through [`flint_core::vfs`], so a game bundle on the web serves it the
17//! same as the disk. [`GpuGlyphAtlas`] uploads an atlas as an `R8Unorm`
18//! texture.

30/// Cells per atlas row.
31pub const ATLAS_COLUMNS: u32 = 16;
32/// Paper round every cell, in pixels.
33pub const CELL_PADDING: u32 = 1;

35/// Why a glyph atlas or set could not be built.
36#[derive(Debug, Clone, PartialEq, Eq)]
37pub enum GlyphError {
38    /// A file could not be read through the vfs.
39    Read(String),
40    /// A `*.glyphs.toml` (or the built-in font text) did not parse.
41    Parse(String),
42    /// The font bytes are not a font ab_glyph can read.
43    Font(String),
44    /// Glyphs a ramp or edge set names that the font does not have.
45    Missing(Vec<char>),
46}
47
48impl fmt::Display for GlyphError {
49    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
50        match self {
51            GlyphError::Read(m) => write!(f, "glyphs: {m}"),
52            GlyphError::Parse(m) => write!(f, "glyphs: {m}"),
53            GlyphError::Font(m) => write!(f, "glyphs: font: {m}"),
54            GlyphError::Missing(chars) => {
55                let list: Vec<String> = chars.iter().map(|c| format!("{c:?}")).collect();
56                write!(f, "glyphs: the font has no glyph for {}", list.join(", "))
57            }
58        }
59    }
60}

69/// Monospaced glyphs packed into one coverage image.
70#[derive(Clone)]
71pub struct GlyphAtlas {
72    cell: [u32; 2],
73    columns: u32,
74    width: u32,
75    height: u32,
76    /// Coverage, row-major, `width * height` bytes.
77    pixels: Vec<u8>,
78    glyphs: Vec<char>,
79    index: HashMap<char, u32>,
80    /// Mean ink per glyph, 0..1, in `glyphs` order.
81    coverage: Vec<f32>,
82}

94impl GlyphAtlas {
95    /// An empty atlas of `glyphs` (deduplicated, in the order given) with
96    /// cells of `cell` pixels; [`Self::paint`] fills them.
97    fn blank(cell: [u32; 2], glyphs: Vec<char>) -> Self {
98        let mut seen = BTreeSet::new();
99        let glyphs: Vec<char> = glyphs.into_iter().filter(|c| seen.insert(*c)).collect();
100        let columns = ATLAS_COLUMNS.min(glyphs.len().max(1) as u32);
101        let rows = (glyphs.len() as u32).div_ceil(columns).max(1);
102        let width = CELL_PADDING + columns * (cell[0] + CELL_PADDING);
103        let height = CELL_PADDING + rows * (cell[1] + CELL_PADDING);
104        let index = glyphs
105            .iter()
106            .enumerate()
107            .map(|(i, c)| (*c, i as u32))
108            .collect();
109        Self {
110            cell,
111            columns,
112            width,
113            height,
114            pixels: vec![0; (width * height) as usize],
115            coverage: vec![0.0; glyphs.len()],
116            glyphs,
117            index,
118        }
119    }
120
121    /// Write one pixel of glyph `i`'s cell, clipped to the cell.
122    fn paint(&mut self, i: u32, x: i32, y: i32, value: u8) {
123        if x < 0 || y < 0 || x as u32 >= self.cell[0] || y as u32 >= self.cell[1] {
124            return;
125        }
126        let [ox, oy] = self.cell_origin(i);
127        let at = ((oy + y as u32) * self.width + ox + x as u32) as usize;
128        self.pixels[at] = self.pixels[at].max(value);
129    }
130
131    /// Measure every glyph's ink once the cells are painted.
132    fn measure(mut self) -> Self {
133        let area = (self.cell[0] * self.cell[1]) as f32;
134        for i in 0..self.glyphs.len() as u32 {
135            let [ox, oy] = self.cell_origin(i);
136            let mut sum = 0u32;
137            for y in oy..oy + self.cell[1] {
138                let row = (y * self.width) as usize;
139                for x in ox..ox + self.cell[0] {
140                    sum += self.pixels[row + x as usize] as u32;
141                }
142            }
143            self.coverage[i as usize] = sum as f32 / (255.0 * area);
144        }
145        self
146    }

148    /// The built-in 8x8 set: printable ASCII, the box-drawing edge glyphs
149    /// `─│╱╲` and the shade blocks `░▒▓█`.
150    /// Built once per process.
151    pub fn builtin() -> std::sync::Arc<GlyphAtlas> {
152        static BUILTIN: std::sync::OnceLock<std::sync::Arc<GlyphAtlas>> =
153            std::sync::OnceLock::new();
154        BUILTIN
155            .get_or_init(|| {
156                std::sync::Arc::new(
157                    Self::from_pixel_text(include_str!("builtin_8x8.txt"), [8, 8])
158                        .expect("the built-in glyph text parses"),
159                )
160            })
161            .clone()
162    }

164    /// Parse the hand-drawn format of `builtin_8x8.txt`: `== c` starts a
165    /// glyph (`== space` for `' '`), then one line per pixel row of `#` and
166    /// `.`. Rows narrower than the cell are centred in it.
167    pub fn from_pixel_text(text: &str, cell: [u32; 2]) -> Result<Self, GlyphError> {
168        let mut drawn: Vec<(char, Vec<&str>)> = Vec::new();
169        for (n, line) in text.lines().enumerate() {
170            let line = line.trim_end();
171            if line.is_empty() || line.starts_with(";;") {
172                continue;
173            }
174            if let Some(name) = line.strip_prefix("== ") {
175                let ch = match name {
176                    "space" => ' ',
177                    _ => {
178                        let mut chars = name.chars();
179                        match (chars.next(), chars.next()) {
180                            (Some(c), None) => c,
181                            _ => {
182                                return Err(GlyphError::Parse(format!(
183                                    "line {}: `== {name}` names more than one character",
184                                    n + 1
185                                )))
186                            }
187                        }
188                    }
189                };
190                drawn.push((ch, Vec::new()));
191            } else if let Some((ch, rows)) = drawn.last_mut() {
192                if line.chars().any(|c| c != '#' && c != '.') {
193                    return Err(GlyphError::Parse(format!(
194                        "line {}: glyph {ch:?} row {line:?} is not `#` and `.`",
195                        n + 1
196                    )));
197                }
198                rows.push(line);
199            } else {
200                return Err(GlyphError::Parse(format!(
201                    "line {}: pixels before any `== c` header",
202                    n + 1
203                )));
204            }
205        }
206
207        let mut atlas = Self::blank(cell, drawn.iter().map(|(c, _)| *c).collect());
208        for (ch, rows) in &drawn {
209            if rows.len() as u32 != cell[1] {
210                return Err(GlyphError::Parse(format!(
211                    "glyph {ch:?} has {} rows, the cell {}",
212                    rows.len(),
213                    cell[1]
214                )));
215            }
216            let i = atlas.index[ch];
217            for (y, row) in rows.iter().enumerate() {
218                let w = row.len() as u32;
219                if w > cell[0] {
220                    return Err(GlyphError::Parse(format!(
221                        "glyph {ch:?} row {row:?} is wider than the cell ({})",
222                        cell[0]
223                    )));
224                }
225                let offset = ((cell[0] - w) / 2) as i32;
226                for (x, px) in row.bytes().enumerate() {
227                    if px == b'#' {
228                        atlas.paint(i, offset + x as i32, y as i32, 255);
229                    }
230                }
231            }
232        }
233        Ok(atlas.measure())
234    }

236    /// Rasterize `chars` (plus `' '`) from a TTF/OTF whose line is `px_height`
237    /// pixels tall. The cell is the line height by the widest advance; each
238    /// glyph sits on the font's baseline, centred in its cell, so a
239    /// proportional font still packs (a monospaced one fills its cells).
240    /// Characters the font lacks are an error listing them.
241    pub fn from_ttf(
242        bytes: &[u8],
243        px_height: f32,
244        chars: impl IntoIterator<Item = char>,
245    ) -> Result<Self, GlyphError> {
246        if !(px_height.is_finite() && px_height >= 4.0) {
247            return Err(GlyphError::Parse(format!(
248                "font size {px_height} is under the 4 px minimum"
249            )));
250        }
251        let font = FontRef::try_from_slice(bytes).map_err(|e| GlyphError::Font(e.to_string()))?;
252        let scaled = font.as_scaled(PxScale::from(px_height));
253
254        let mut wanted: Vec<char> = vec![' '];
255        wanted.extend(chars);
256        let missing: BTreeSet<char> = wanted
257            .iter()
258            .copied()
259            .filter(|c| *c != ' ' && font.glyph_id(*c).0 == 0)
260            .collect();
261        if !missing.is_empty() {
262            return Err(GlyphError::Missing(missing.into_iter().collect()));
263        }
264
265        let ascent = scaled.ascent();
266        let line = ascent - scaled.descent();
267        let advance = wanted
268            .iter()
269            .map(|c| scaled.h_advance(font.glyph_id(*c)))
270            .fold(0.0f32, f32::max);
271        let cell = [advance.ceil().max(1.0) as u32, line.ceil().max(1.0) as u32];
272
273        let mut atlas = Self::blank(cell, wanted);
274        for i in 0..atlas.glyphs.len() as u32 {
275            let ch = atlas.glyphs[i as usize];
276            let id = font.glyph_id(ch);
277            let x = (cell[0] as f32 - scaled.h_advance(id)) * 0.5;
278            let glyph = id.with_scale_and_position(
279                PxScale::from(px_height),
280                ab_glyph::point(x.round(), ascent.round()),
281            );
282            if let Some(outline) = font.outline_glyph(glyph) {
283                let bounds = outline.px_bounds();
284                let (bx, by) = (bounds.min.x as i32, bounds.min.y as i32);
285                let mut ink = Vec::new();
286                outline.draw(|gx, gy, c| {
287                    ink.push((bx + gx as i32, by + gy as i32, (c * 255.0).round() as u8));
288                });
289                for (px, py, v) in ink {
290                    atlas.paint(i, px, py, v);
291                }
292            }
293        }
294        Ok(atlas.measure())
295    }

297    /// Cut glyphs from a grid image whose cells are `cell` pixels, read left to
298    /// right, top to bottom: cell `n` holds `chars[n]`. Ink is luminance times
299    /// alpha (white on transparent or white on black), or `invert` for dark
300    /// ink on light paper.
301    pub fn from_grid_image(
302        image: &image::DynamicImage,
303        cell: [u32; 2],
304        chars: impl IntoIterator<Item = char>,
305        invert: bool,
306    ) -> Result<Self, GlyphError> {
307        if cell[0] == 0 || cell[1] == 0 {
308            return Err(GlyphError::Parse("a grid cell needs a size".into()));
309        }
310        let rgba = image.to_rgba8();
311        let grid_columns = rgba.width() / cell[0];
312        let grid_rows = rgba.height() / cell[1];
313        let chars: Vec<char> = chars.into_iter().collect();
314        let capacity = (grid_columns * grid_rows) as usize;
315        if chars.len() > capacity {
316            return Err(GlyphError::Missing(chars[capacity..].to_vec()));
317        }
318        let mut atlas = Self::blank(cell, chars.clone());
319        for (n, ch) in chars.iter().enumerate() {
320            let i = atlas.index[ch];
321            let (gx, gy) = (
322                (n as u32 % grid_columns) * cell[0],
323                (n as u32 / grid_columns) * cell[1],
324            );
325            for y in 0..cell[1] {
326                for x in 0..cell[0] {
327                    let p = rgba.get_pixel(gx + x, gy + y).0;
328                    let luma = (p[0] as u32 * 54 + p[1] as u32 * 183 + p[2] as u32 * 19) / 256;
329                    let luma = if invert { 255 - luma } else { luma };
330                    let v = (luma * p[3] as u32 / 255) as u8;
331                    atlas.paint(i, x as i32, y as i32, v);
332                }
333            }
334        }
335        Ok(atlas.measure())
336    }

373    /// Top-left pixel of glyph `i`'s cell (inside its gutter).
374    pub fn cell_origin(&self, i: u32) -> [u32; 2] {
375        let (col, row) = (i % self.columns, i / self.columns);
376        [
377            CELL_PADDING + col * (self.cell[0] + CELL_PADDING),
378            CELL_PADDING + row * (self.cell[1] + CELL_PADDING),
379        ]
380    }

382    /// `[u_min, v_min, u_max, v_max]` of `ch`'s cell, 0..1.
383    pub fn uv_rect(&self, ch: char) -> Option<[f32; 4]> {
384        let [x, y] = self.cell_origin(self.index_of(ch)?);
385        let (w, h) = (self.width as f32, self.height as f32);
386        Some([
387            x as f32 / w,
388            y as f32 / h,
389            (x + self.cell[0]) as f32 / w,
390            (y + self.cell[1]) as f32 / h,
391        ])
392    }

394    /// Mean ink of `ch`'s cell, 0 (none) to 1 (solid).
395    pub fn coverage(&self, ch: char) -> Option<f32> {
396        Some(self.coverage[self.index_of(ch)? as usize])
397    }

399    /// Coverage of one pixel of `ch`'s cell, 0..255; `None` off the cell.
400    pub fn sample(&self, ch: char, x: u32, y: u32) -> Option<u8> {
401        if x >= self.cell[0] || y >= self.cell[1] {
402            return None;
403        }
404        let [ox, oy] = self.cell_origin(self.index_of(ch)?);
405        Some(self.pixels[((oy + y) * self.width + ox + x) as usize])
406    }
