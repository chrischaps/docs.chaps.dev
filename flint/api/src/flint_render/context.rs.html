8pub enum RenderError {
9    #[error("Failed to create surface: {0}")]
10    SurfaceCreation(String),
11    #[error("Failed to get adapter")]
12    AdapterNotFound,
13    #[error("Failed to create device: {0}")]
14    DeviceCreation(String),
15    #[error("Surface error: {0}")]
16    SurfaceError(String),
17    #[error("Failed to read render buffer: {0}")]
18    BufferReadFailed(String),

21/// wgpu render context containing device, queue, and surface
22pub struct RenderContext {
23    pub surface: wgpu::Surface<'static>,
24    pub device: wgpu::Device,
25    pub queue: wgpu::Queue,
26    pub config: wgpu::SurfaceConfiguration,
27    pub size: winit::dpi::PhysicalSize<u32>,
28    pub depth_texture: wgpu::Texture,
29    pub depth_view: wgpu::TextureView,
30    // Retained for surface recreation on Android resume
31    instance: wgpu::Instance,
32    pub adapter: wgpu::Adapter,
33    /// True when the swapchain was configured with `COPY_SRC`, so the
34    /// presented frame can be read back via [`Self::read_surface_rgba`].
35    pub surface_copyable: bool,
36    /// Format for the HUD (egui) pass view: the swapchain format without its
37    /// sRGB suffix, so egui's own gamma handling applies and alpha blends in
38    /// display space.
39    pub hud_format: wgpu::TextureFormat,
40    /// Format the scene's last pass writes the swapchain in: always sRGB
41    /// when the surface can give an sRGB view. A desktop surface is sRGB
42    /// itself; a browser canvas offers only `Bgra8Unorm`/`Rgba8Unorm`, so
43    /// it is configured plain with an sRGB view format (#102) and the frame
44    /// is drawn through that view (see [`Self::frame_view`]).
45    pub view_format: wgpu::TextureFormat,
46}

48impl RenderContext {
49    /// Create a new render context for a window
50    pub async fn new(window: Arc<Window>) -> Result<Self, RenderError> {
51        let size = window.inner_size();
52
53        let instance = crate::gpu_select::instance();
54
55        let surface = instance
56            .create_surface(window.clone())
57            .map_err(|e| RenderError::SurfaceCreation(e.to_string()))?;
58
59        let adapter = crate::gpu_select::request_adapter(&instance, Some(&surface))
60            .await
61            .ok_or(RenderError::AdapterNotFound)?;
62
63        #[cfg(target_os = "android")]
64        let limits = {
65            let mut lim = wgpu::Limits::downlevel_defaults();
66            // Use the adapter's actual max texture size so the surface can
67            // match the device's full screen resolution (often > 2048px).
68            let adapter_limits = adapter.limits();
69            lim.max_texture_dimension_2d = adapter_limits.max_texture_dimension_2d;
70            lim
71        };
72        #[cfg(not(target_os = "android"))]
73        let limits = wgpu::Limits::default();
74
75        let (device, queue) = adapter
76            .request_device(
77                &wgpu::DeviceDescriptor {
78                    label: Some("Flint Device"),
79                    // Timestamps where the adapter has them, so the
80                    // renderer's GPU timing can be turned on (ADR 0108).
81                    required_features: adapter.features() & wgpu::Features::TIMESTAMP_QUERY,
82                    required_limits: limits,
83                    memory_hints: Default::default(),
84                },
85                None,
86            )
87            .await
88            .map_err(|e| RenderError::DeviceCreation(e.to_string()))?;
89
90        let surface_caps = surface.get_capabilities(&adapter);
91        let surface_format = surface_caps
92            .formats
93            .iter()
94            .find(|f| f.is_srgb())
95            .copied()
96            .unwrap_or(surface_caps.formats[0]);
97        // Without an sRGB surface (WebGPU canvases), draw through an sRGB
98        // view of the plain one, so the encode happens on write exactly as
99        // it does on the desktop. Without it the web frame came out darker
100        // and more saturated (docs/design/web.md).
101        let view_format = surface_format.add_srgb_suffix();
102
103        // Ask for COPY_SRC on the swapchain when the surface allows it so the
104        // player can screenshot the presented frame (HUD included). Some
105        // backends (notably GL/Android) only permit RENDER_ATTACHMENT.
106        let surface_copyable = surface_caps.usages.contains(wgpu::TextureUsages::COPY_SRC);
107        let mut usage = wgpu::TextureUsages::RENDER_ATTACHMENT;
108        if surface_copyable {
109            usage |= wgpu::TextureUsages::COPY_SRC;
110        } else {
111            tracing::warn!(
112                "surface does not support COPY_SRC; --screenshot is unavailable on this adapter"
113            );
114        }
115
116        // FLINT_VSYNC picks the present mode: unset/"1" = vsync (Fifo),
117        // "0"/"off"/"immediate" = unlocked (real frame times for headless
118        // perf runs), "mailbox" = uncapped rendering with the newest frame
119        // shown at each refresh (no tearing, no Fifo stall). On this Optimus
120        // laptop Fifo has been seen pacing at ~7 Hz against a virtual
121        // display; mailbox sidesteps it.
122        let present_mode = match std::env::var("FLINT_VSYNC")
123            .unwrap_or_default()
124            .to_ascii_lowercase()
125            .as_str()
126        {
127            "0" | "off" | "immediate" => wgpu::PresentMode::AutoNoVsync,
128            "mailbox" => wgpu::PresentMode::Mailbox,
129            _ => wgpu::PresentMode::AutoVsync,
130        };
131        tracing::info!(
132            "surface present modes: {:?}; using {:?}",
133            surface_caps.present_modes,
134            present_mode
135        );
136
137        let config = wgpu::SurfaceConfiguration {
138            usage,
139            format: surface_format,
140            width: size.width.max(1),
141            height: size.height.max(1),
142            present_mode,
143            alpha_mode: surface_caps.alpha_modes[0],
144            // A non-sRGB view of the swapchain for the HUD pass: egui expects to
145            // write sRGB-encoded values itself, so blending a translucent panel
146            // against the scene happens in display space, not linear space.
147            view_formats: if surface_format.remove_srgb_suffix() != surface_format {
148                vec![surface_format.remove_srgb_suffix()]
149            } else if view_format != surface_format {
150                vec![view_format]
151            } else {
152                vec![]
153            },
154            desired_maximum_frame_latency: frame_latency(),
155        };
156        tracing::info!(
157            "surface frame latency: {}",
158            config.desired_maximum_frame_latency
159        );
160        surface.configure(&device, &config);
161
162        let (depth_texture, depth_view) = create_depth_texture(&device, &config);
163
164        Ok(Self {
165            surface,
166            device,
167            queue,
168            config,
169            size,
170            depth_texture,
171            depth_view,
172            instance,
173            adapter,
174            surface_copyable,
175            hud_format: surface_format.remove_srgb_suffix(),
176            view_format,
177        })
178    }

189    /// Read a surface-sized texture back as tightly packed RGBA8 bytes.
190    ///
191    /// Blocks on the GPU (`Maintain::Wait`). The texture must have been
192    /// created with `COPY_SRC` and be in the surface format; Bgra formats are
193    /// swizzled to RGBA on the way out. Mirrors `HeadlessContext::read_pixels`.
194    pub fn read_surface_rgba(&self, texture: &wgpu::Texture) -> Result<Vec<u8>, RenderError> {
195        if !self.surface_copyable {
196            return Err(RenderError::BufferReadFailed(
197                "surface was not configured with COPY_SRC".into(),
198            ));
199        }
200        let width = texture.width();
201        let height = texture.height();
202        let bytes_per_pixel = 4u32;
203        let unpadded_bytes_per_row = width * bytes_per_pixel;
204        let align = wgpu::COPY_BYTES_PER_ROW_ALIGNMENT;
205        let padded_bytes_per_row = unpadded_bytes_per_row.div_ceil(align) * align;
206
207        let staging = self.device.create_buffer(&wgpu::BufferDescriptor {
208            label: Some("Surface Readback Buffer"),
209            size: (padded_bytes_per_row * height) as u64,
210            usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
211            mapped_at_creation: false,
212        });
213
214        let mut encoder = self
215            .device
216            .create_command_encoder(&wgpu::CommandEncoderDescriptor {
217                label: Some("Surface Readback Encoder"),
218            });
219        encoder.copy_texture_to_buffer(
220            wgpu::ImageCopyTexture {
221                texture,
222                mip_level: 0,
223                origin: wgpu::Origin3d::ZERO,
224                aspect: wgpu::TextureAspect::All,
225            },
226            wgpu::ImageCopyBuffer {
227                buffer: &staging,
228                layout: wgpu::ImageDataLayout {
229                    offset: 0,
230                    bytes_per_row: Some(padded_bytes_per_row),
231                    rows_per_image: Some(height),
232                },
233            },
234            wgpu::Extent3d {
235                width,
236                height,
237                depth_or_array_layers: 1,
238            },
239        );
240        self.queue.submit(std::iter::once(encoder.finish()));
241
242        let slice = staging.slice(..);
243        let (tx, rx) = std::sync::mpsc::channel();
244        slice.map_async(wgpu::MapMode::Read, move |result| {
245            let _ = tx.send(result);
246        });
247        self.device.poll(wgpu::Maintain::Wait);
248        rx.recv()
249            .map_err(|e| RenderError::BufferReadFailed(e.to_string()))?
250            .map_err(|e| RenderError::BufferReadFailed(e.to_string()))?;
251
252        let data = slice.get_mapped_range();
253        let mut pixels = Vec::with_capacity((width * height * bytes_per_pixel) as usize);
254        for row in 0..height {
255            let start = (row * padded_bytes_per_row) as usize;
256            pixels.extend_from_slice(&data[start..start + unpadded_bytes_per_row as usize]);
257        }
258        drop(data);
259        staging.unmap();
260
261        let is_bgra = matches!(
262            texture.format(),
263            wgpu::TextureFormat::Bgra8Unorm | wgpu::TextureFormat::Bgra8UnormSrgb
264        );
265        if is_bgra {
266            for px in pixels.chunks_exact_mut(4) {
267                px.swap(0, 2);
268            }
269        }
270        // The swapchain alpha channel is undefined for opaque surfaces.
271        for px in pixels.chunks_exact_mut(4) {
272            px[3] = 255;
273        }
274        Ok(pixels)
275    }

277    /// Recreate the surface from a new window handle.
278    ///
279    /// On Android, the native surface is destroyed when the app is paused and a
280    /// new one is provided when resumed. This method creates a fresh surface,
281    /// reconfigures it with the existing device/format settings, and rebuilds
282    /// the depth texture.
283    pub fn recreate_surface(&mut self, window: Arc<Window>) -> Result<(), RenderError> {
284        let size = window.inner_size();
285
286        let surface = self
287            .instance
288            .create_surface(window)
289            .map_err(|e| RenderError::SurfaceCreation(e.to_string()))?;
290
291        // Reconfigure with current format but new dimensions
292        self.config.width = size.width.max(1);
293        self.config.height = size.height.max(1);
294        surface.configure(&self.device, &self.config);
295
296        let (depth_texture, depth_view) = create_depth_texture(&self.device, &self.config);
297
298        self.surface = surface;
299        self.size = size;
300        self.depth_texture = depth_texture;
301        self.depth_view = depth_view;
302
303        Ok(())
304    }

306    /// Resize the surface
307    pub fn resize(&mut self, new_size: winit::dpi::PhysicalSize<u32>) {
308        if new_size.width > 0 && new_size.height > 0 {
309            self.size = new_size;
310            self.config.width = new_size.width;
311            self.config.height = new_size.height;
312            self.surface.configure(&self.device, &self.config);
313
314            let (depth_texture, depth_view) = create_depth_texture(&self.device, &self.config);
315            self.depth_texture = depth_texture;
316            self.depth_view = depth_view;
317        }
318    }

320    /// True when every presented frame waits for a vblank (Fifo), so frame
321    /// times can be paced to the display's refresh (`GameClock`).
322    pub fn presents_on_vblank(&self) -> bool {
323        matches!(
324            self.config.present_mode,
325            wgpu::PresentMode::AutoVsync | wgpu::PresentMode::Fifo | wgpu::PresentMode::FifoRelaxed
326        )
327    }

335/// Frames the CPU may queue ahead of the display. `FLINT_FRAME_LATENCY`
336/// overrides the default of 2 (1 to 3).
337fn frame_latency() -> u32 {
338    std::env::var("FLINT_FRAME_LATENCY")
339        .ok()
340        .and_then(|v| v.trim().parse::<u32>().ok())
341        .map(|n| n.clamp(1, 3))
342        .unwrap_or(2)
343}

345fn create_depth_texture(
346    device: &wgpu::Device,
347    config: &wgpu::SurfaceConfiguration,
348) -> (wgpu::Texture, wgpu::TextureView) {
349    let size = wgpu::Extent3d {
350        width: config.width.max(1),
351        height: config.height.max(1),
352        depth_or_array_layers: 1,
353    };
354
355    let desc = wgpu::TextureDescriptor {
356        label: Some("Depth Texture"),
357        size,
358        mip_level_count: 1,
359        sample_count: 1,
360        dimension: wgpu::TextureDimension::D2,
361        format: wgpu::TextureFormat::Depth32Float,
362        usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
363        view_formats: &[],
364    };
365
366    let texture = device.create_texture(&desc);
367    let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
368
369    (texture, view)
