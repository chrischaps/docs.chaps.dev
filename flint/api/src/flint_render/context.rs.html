7#[derive(Debug, Error)]
8pub enum RenderError {
9    #[error("Failed to create surface: {0}")]
10    SurfaceCreation(String),
11    #[error("Failed to get adapter")]
12    AdapterNotFound,
13    #[error("Failed to create device: {0}")]
14    DeviceCreation(String),
15    #[error("Surface error: {0}")]
16    SurfaceError(String),
17    #[error("Failed to read render buffer: {0}")]
18    BufferReadFailed(String),
19}

21/// wgpu render context containing device, queue, and surface
22pub struct RenderContext {
23    pub surface: wgpu::Surface<'static>,
24    pub device: wgpu::Device,
25    pub queue: wgpu::Queue,
26    pub config: wgpu::SurfaceConfiguration,
27    pub size: winit::dpi::PhysicalSize<u32>,
28    pub depth_texture: wgpu::Texture,
29    pub depth_view: wgpu::TextureView,
30    // Retained for surface recreation on Android resume
31    instance: wgpu::Instance,
32    pub adapter: wgpu::Adapter,
33}
34
35impl RenderContext {
36    /// Create a new render context for a window
37    pub async fn new(window: Arc<Window>) -> Result<Self, RenderError> {
38        let size = window.inner_size();
39
40        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor {
41            backends: wgpu::Backends::all(),
42            ..Default::default()
43        });
44
45        let surface = instance
46            .create_surface(window.clone())
47            .map_err(|e| RenderError::SurfaceCreation(e.to_string()))?;
48
49        let adapter = instance
50            .request_adapter(&wgpu::RequestAdapterOptions {
51                power_preference: wgpu::PowerPreference::default(),
52                compatible_surface: Some(&surface),
53                force_fallback_adapter: false,
54            })
55            .await
56            .ok_or(RenderError::AdapterNotFound)?;
57
58        #[cfg(target_os = "android")]
59        let limits = {
60            let mut lim = wgpu::Limits::downlevel_defaults();
61            // Use the adapter's actual max texture size so the surface can
62            // match the device's full screen resolution (often > 2048px).
63            let adapter_limits = adapter.limits();
64            lim.max_texture_dimension_2d = adapter_limits.max_texture_dimension_2d;
65            lim
66        };
67        #[cfg(not(target_os = "android"))]
68        let limits = wgpu::Limits::default();
69
70        let (device, queue) = adapter
71            .request_device(
72                &wgpu::DeviceDescriptor {
73                    label: Some("Flint Device"),
74                    required_features: wgpu::Features::empty(),
75                    required_limits: limits,
76                    memory_hints: Default::default(),
77                },
78                None,
79            )
80            .await
81            .map_err(|e| RenderError::DeviceCreation(e.to_string()))?;
82
83        let surface_caps = surface.get_capabilities(&adapter);
84        let surface_format = surface_caps
85            .formats
86            .iter()
87            .find(|f| f.is_srgb())
88            .copied()
89            .unwrap_or(surface_caps.formats[0]);
90
91        let config = wgpu::SurfaceConfiguration {
92            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
93            format: surface_format,
94            width: size.width.max(1),
95            height: size.height.max(1),
96            present_mode: wgpu::PresentMode::AutoVsync,
97            alpha_mode: surface_caps.alpha_modes[0],
98            view_formats: vec![],
99            desired_maximum_frame_latency: 2,
100        };
101        surface.configure(&device, &config);
102
103        let (depth_texture, depth_view) = create_depth_texture(&device, &config);
104
105        Ok(Self {
106            surface,
107            device,
108            queue,
109            config,
110            size,
111            depth_texture,
112            depth_view,
113            instance,
114            adapter,
115        })
116    }
117
118    /// Recreate the surface from a new window handle.
119    ///
120    /// On Android, the native surface is destroyed when the app is paused and a
121    /// new one is provided when resumed. This method creates a fresh surface,
122    /// reconfigures it with the existing device/format settings, and rebuilds
123    /// the depth texture.
124    pub fn recreate_surface(&mut self, window: Arc<Window>) -> Result<(), RenderError> {
125        let size = window.inner_size();
126
127        let surface = self
128            .instance
129            .create_surface(window)
130            .map_err(|e| RenderError::SurfaceCreation(e.to_string()))?;
131
132        // Reconfigure with current format but new dimensions
133        self.config.width = size.width.max(1);
134        self.config.height = size.height.max(1);
135        surface.configure(&self.device, &self.config);
136
137        let (depth_texture, depth_view) = create_depth_texture(&self.device, &self.config);
138
139        self.surface = surface;
140        self.size = size;
141        self.depth_texture = depth_texture;
142        self.depth_view = depth_view;
143
144        Ok(())
145    }
146
147    /// Resize the surface
148    pub fn resize(&mut self, new_size: winit::dpi::PhysicalSize<u32>) {
149        if new_size.width > 0 && new_size.height > 0 {
150            self.size = new_size;
151            self.config.width = new_size.width;
152            self.config.height = new_size.height;
153            self.surface.configure(&self.device, &self.config);
154
155            let (depth_texture, depth_view) = create_depth_texture(&self.device, &self.config);
156            self.depth_texture = depth_texture;
157            self.depth_view = depth_view;
158        }
159    }
160
161    /// Get aspect ratio
162    pub fn aspect_ratio(&self) -> f32 {
163        self.size.width as f32 / self.size.height as f32
164    }
165}

167fn create_depth_texture(
168    device: &wgpu::Device,
169    config: &wgpu::SurfaceConfiguration,
170) -> (wgpu::Texture, wgpu::TextureView) {
171    let size = wgpu::Extent3d {
172        width: config.width.max(1),
173        height: config.height.max(1),
174        depth_or_array_layers: 1,
175    };
176
177    let desc = wgpu::TextureDescriptor {
178        label: Some("Depth Texture"),
179        size,
180        mip_level_count: 1,
181        sample_count: 1,
182        dimension: wgpu::TextureDimension::D2,
183        format: wgpu::TextureFormat::Depth32Float,
184        usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
185        view_formats: &[],
186    };
187
188    let texture = device.create_texture(&desc);
189    let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
190
191    (texture, view)
192}
