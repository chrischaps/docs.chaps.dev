6/// Camera operating mode
7#[derive(Debug, Clone, Copy, PartialEq, Default)]
8pub enum CameraMode {
9    /// Orbit around a target point (default for scene viewer)
10    #[default]
11    Orbit,
12    /// First-person view at a position looking in yaw/pitch direction
13    FirstPerson,
14}

16/// A 3D camera supporting orbit and first-person modes
17pub struct Camera {
18    /// Camera position
19    pub position: Vec3,
20    /// Target point the camera looks at
21    pub target: Vec3,
22    /// Up vector
23    pub up: Vec3,
24    /// Field of view in degrees
25    pub fov: f32,
26    /// Near clipping plane
27    pub near: f32,
28    /// Far clipping plane
29    pub far: f32,
30    /// Aspect ratio (width / height)
31    pub aspect: f32,
32
33    // Orbit control state
34    /// Distance from target
35    pub distance: f32,
36    /// Horizontal angle in radians
37    pub yaw: f32,
38    /// Vertical angle in radians
39    pub pitch: f32,
40
41    /// Camera operating mode
42    pub mode: CameraMode,
43
44    /// Use orthographic projection (true) or perspective (false)
45    pub orthographic: bool,
46
47    /// Explicit orthographic half-height in world units.
48    /// When > 0, `orthographic_matrix()` uses this directly instead of
49    /// deriving from `distance * tan(fov/2)`. Set to 0.0 to use the legacy formula.
50    pub ortho_height: f32,
51
52    /// Clip-space (NDC) offset applied after projection, so the look-at point
53    /// can land off the window centre — editors use it to centre the target
54    /// in the part of the window their side panels leave visible. `[0, 0]`
55    /// is a plain projection.
56    pub ndc_offset: [f32; 2],
57}

85    /// Get camera position as an array for GPU upload
86    pub fn position_array(&self) -> [f32; 3] {
87        [self.position.x, self.position.y, self.position.z]
88    }

90    /// Update position based on orbit parameters
91    pub fn update_orbit(&mut self) {
92        let x = self.distance * self.pitch.cos() * self.yaw.sin();
93        let y = self.distance * self.pitch.sin();
94        let z = self.distance * self.pitch.cos() * self.yaw.cos();
95
96        self.position = Vec3::new(self.target.x + x, self.target.y + y, self.target.z + z);
97    }
98
99    /// Orbit horizontally (rotate around target)
100    pub fn orbit_horizontal(&mut self, delta: f32) {
101        self.yaw += delta;
102        self.update_orbit();
103    }
104
105    /// Orbit vertically (tilt up/down)
106    pub fn orbit_vertical(&mut self, delta: f32) {
107        self.pitch += delta;
108        // Clamp pitch to avoid gimbal lock (1.56 ≈ 89.4° allows near-orthographic top/bottom views)
109        self.pitch = self.pitch.clamp(-1.56, 1.56);
110        self.update_orbit();
111    }
112
113    /// Zoom in/out
114    pub fn zoom(&mut self, delta: f32) {
115        self.distance = (self.distance - delta).max(1.0).min(500.0);
116        self.update_orbit();
117    }
118
119    /// Update for first-person mode: set position directly, compute target from yaw/pitch
120    pub fn update_first_person(&mut self, position: Vec3, yaw: f32, pitch: f32) {
121        self.mode = CameraMode::FirstPerson;
122        self.position = position;
123        self.yaw = yaw;
124        self.pitch = pitch;
125
126        // Compute forward direction from yaw/pitch
127        let forward_x = pitch.cos() * yaw.sin();
128        let forward_y = pitch.sin();
129        let forward_z = pitch.cos() * yaw.cos();
130
131        self.target = Vec3::new(
132            position.x + forward_x,
133            position.y + forward_y,
134            position.z + forward_z,
135        );
136    }
137
138    /// Pan the camera (move target)
139    pub fn pan(&mut self, dx: f32, dy: f32) {
140        // Calculate right and up vectors relative to camera
141        let (_, right, up) = self.basis();
142
143        self.target = self.target + right * dx + up * dy;
144        self.update_orbit();
145    }
146
147    /// The view basis: `(forward, right, up)`, unit length.
148    ///
149    /// `right = forward × up`. Looking straight along `up` (a top-down camera)
150    /// leaves that cross product zero, so the basis falls back to north (-Z)
151    /// as the top of the screen: from above, east (+X) is on the right, as on
152    /// a map. Without the fallback the view matrix collapses and nothing
153    /// renders.
154    fn basis(&self) -> (Vec3, Vec3, Vec3) {
155        let f = (self.target - self.position).normalized();
156        let mut s = f.cross(&self.up);
157        if s.dot(&s) < 1e-12 {
158            s = f.cross(&Vec3::new(0.0, 0.0, -1.0));
159            if s.dot(&s) < 1e-12 {
160                // `up` itself lies along Z, so -Z is no help either.
161                s = f.cross(&Vec3::new(1.0, 0.0, 0.0));
162            }
163        }
164        let s = s.normalized();
165        let u = s.cross(&f);
166        (f, s, u)
167    }
168
169    /// Get the view matrix (4x4, column-major)
170    pub fn view_matrix(&self) -> [[f32; 4]; 4] {
171        let (f, s, u) = self.basis();
172
173        [
174            [s.x, u.x, -f.x, 0.0],
175            [s.y, u.y, -f.y, 0.0],
176            [s.z, u.z, -f.z, 0.0],
177            [
178                -s.dot(&self.position),
179                -u.dot(&self.position),
180                f.dot(&self.position),
181                1.0,
182            ],
183        ]
184    }
185
186    /// Get the projection matrix (4x4, column-major)
187    pub fn projection_matrix(&self) -> [[f32; 4]; 4] {
188        let mut m = if self.orthographic {
189            self.orthographic_matrix()
190        } else {
191            self.perspective_matrix()
192        };
193        // clip.x += ox * clip.w (likewise y): a pure screen-space shift that
194        // is valid for both projections because it rides the w row.
195        let [ox, oy] = self.ndc_offset;
196        if ox != 0.0 || oy != 0.0 {
197            for col in m.iter_mut() {
198                col[0] += ox * col[3];
199                col[1] += oy * col[3];
200            }
201        }
202        m
203    }
204
205    fn perspective_matrix(&self) -> [[f32; 4]; 4] {
206        let fov_rad = self.fov.to_radians();
207        let f = 1.0 / (fov_rad / 2.0).tan();
208
209        let depth = self.far - self.near;
210
211        // wgpu [0,1] depth convention: z_view=-near → 0, z_view=-far → 1
212        // (matches orthographic_matrix; ADR 0055 — the former GL [-1,1] rows
213        // clipped geometry nearer than 2×near and halved linearized depth)
214        [
215            [f / self.aspect, 0.0, 0.0, 0.0],
216            [0.0, f, 0.0, 0.0],
217            [0.0, 0.0, -self.far / depth, -1.0],
218            [0.0, 0.0, -(self.far * self.near) / depth, 0.0],
219        ]
220    }
221
222    fn orthographic_matrix(&self) -> [[f32; 4]; 4] {
223        // When ortho_height is set, use it directly as half-height; otherwise
224        // size the volume so objects at `distance` appear the same size as in perspective.
225        let half_h = if self.ortho_height > 0.0 {
226            self.ortho_height
227        } else {
228            self.distance * (self.fov.to_radians() / 2.0).tan()
229        };
230        let half_w = half_h * self.aspect;
231        let depth = self.far - self.near;
232
233        // Column-major: m[col][row]
234        // Maps depth to [0, 1] (wgpu convention): z_view=-near → 0, z_view=-far → 1
235        [
236            [1.0 / half_w, 0.0, 0.0, 0.0],
237            [0.0, 1.0 / half_h, 0.0, 0.0],
238            [0.0, 0.0, -1.0 / depth, 0.0],
239            [0.0, 0.0, -self.near / depth, 1.0],
240        ]
241    }

243    /// Get combined view-projection matrix
244    pub fn view_projection_matrix(&self) -> [[f32; 4]; 4] {
245        let view = self.view_matrix();
246        let proj = self.projection_matrix();
247        mat4_mul(&proj, &view)
248    }
249
250    /// Get camera right vector (world space)
251    pub fn right_vector(&self) -> [f32; 3] {
252        let (_, s, _) = self.basis();
253        [s.x, s.y, s.z]
254    }
255
256    /// Get camera up vector (world space, perpendicular to both forward and right)
257    pub fn up_vector(&self) -> [f32; 3] {
258        let (_, _, u) = self.basis();
259        [u.x, u.y, u.z]
260    }
261
262    /// Get camera forward direction (world space)
263    pub fn forward_vector(&self) -> [f32; 3] {
264        let f = (self.target - self.position).normalized();
265        [f.x, f.y, f.z]
266    }

268    /// Get inverse of the combined view-projection matrix (for unprojecting)
269    pub fn inverse_view_projection_matrix(&self) -> [[f32; 4]; 4] {
270        let vp = self.view_projection_matrix();
271        mat4_inverse(&vp)
272    }

274    /// Get inverse of the projection matrix (for reconstructing view-space positions from depth)
275    pub fn inverse_projection_matrix(&self) -> [[f32; 4]; 4] {
276        let proj = self.projection_matrix();
277        mat4_inverse(&proj)
278    }

280    /// Unproject a screen-space NDC point into a world-space ray.
281    /// `ndc_x` and `ndc_y` are in [-1, 1] (left-to-right, bottom-to-top).
282    /// Returns `(origin, direction)` as `[f32; 3]` arrays: the origin is on
283    /// the near plane and the direction is unit length. Works for perspective
284    /// and orthographic cameras, and honours `ndc_offset`.
285    pub fn unproject_ray(&self, ndc_x: f32, ndc_y: f32) -> ([f32; 3], [f32; 3]) {
286        flint_core::projection::unproject_ray(&self.view_projection_matrix(), [ndc_x, ndc_y])
287            .unwrap_or((self.position_array(), [0.0, -1.0, 0.0]))
288    }
289
290    /// The point on the horizontal plane `y = plane_y` under a screen point,
291    /// as `[x, z]`, or `None` where the view doesn't reach the plane (above
292    /// the horizon, a level orthographic view, a plane behind the camera).
293    ///
294    /// `(x, y)` has its origin at the viewport's top-left with y down, in the
295    /// same unit as `width`/`height`: logical points with the logical size,
296    /// or physical pixels with the physical size. Either works on a HiDPI
297    /// display, but don't mix them.
298    pub fn screen_to_ground(
299        &self,
300        x: f32,
301        y: f32,
302        width: f32,
303        height: f32,
304        plane_y: f32,
305    ) -> Option<[f32; 2]> {
306        flint_core::projection::screen_to_ground(
307            &self.view_projection_matrix(),
308            x,
309            y,
310            width,
311            height,
312            plane_y,
313        )
314    }

317/// Compute the inverse of a 4x4 column-major matrix using cofactor expansion
318pub(crate) fn mat4_inverse(m: &[[f32; 4]; 4]) -> [[f32; 4]; 4] {
319    // Flatten column-major to indexable
320    let s = |col: usize, row: usize| -> f32 { m[col][row] };
321
322    let c00 = s(2, 2) * s(3, 3) - s(3, 2) * s(2, 3);
323    let c02 = s(1, 2) * s(3, 3) - s(3, 2) * s(1, 3);
324    let c03 = s(1, 2) * s(2, 3) - s(2, 2) * s(1, 3);
325
326    let c04 = s(2, 1) * s(3, 3) - s(3, 1) * s(2, 3);
327    let c06 = s(1, 1) * s(3, 3) - s(3, 1) * s(1, 3);
328    let c07 = s(1, 1) * s(2, 3) - s(2, 1) * s(1, 3);
329
330    let c08 = s(2, 1) * s(3, 2) - s(3, 1) * s(2, 2);
331    let c10 = s(1, 1) * s(3, 2) - s(3, 1) * s(1, 2);
332    let c11 = s(1, 1) * s(2, 2) - s(2, 1) * s(1, 2);
333
334    let c12 = s(2, 0) * s(3, 3) - s(3, 0) * s(2, 3);
335    let c14 = s(1, 0) * s(3, 3) - s(3, 0) * s(1, 3);
336    let c15 = s(1, 0) * s(2, 3) - s(2, 0) * s(1, 3);
337
338    let c16 = s(2, 0) * s(3, 2) - s(3, 0) * s(2, 2);
339    let c18 = s(1, 0) * s(3, 2) - s(3, 0) * s(1, 2);
340    let c19 = s(1, 0) * s(2, 2) - s(2, 0) * s(1, 2);
341
342    let c20 = s(2, 0) * s(3, 1) - s(3, 0) * s(2, 1);
343    let c22 = s(1, 0) * s(3, 1) - s(3, 0) * s(1, 1);
344    let c23 = s(1, 0) * s(2, 1) - s(2, 0) * s(1, 1);
345
346    let f0 = [c00, c00, c02, c03];
347    let f1 = [c04, c04, c06, c07];
348    let f2 = [c08, c08, c10, c11];
349    let f3 = [c12, c12, c14, c15];
350    let f4 = [c16, c16, c18, c19];
351    let f5 = [c20, c20, c22, c23];
352
353    let v0 = [s(1, 0), s(0, 0), s(0, 0), s(0, 0)];
354    let v1 = [s(1, 1), s(0, 1), s(0, 1), s(0, 1)];
355    let v2 = [s(1, 2), s(0, 2), s(0, 2), s(0, 2)];
356    let v3 = [s(1, 3), s(0, 3), s(0, 3), s(0, 3)];
357
358    let mut inv = [[0.0f32; 4]; 4];
359    let sign_a = [1.0, -1.0, 1.0, -1.0];
360    let sign_b = [-1.0, 1.0, -1.0, 1.0];
361
362    for i in 0..4 {
363        inv[0][i] = sign_a[i] * (v1[i] * f0[i] - v2[i] * f1[i] + v3[i] * f2[i]);
364        inv[1][i] = sign_b[i] * (v0[i] * f0[i] - v2[i] * f3[i] + v3[i] * f4[i]);
365        inv[2][i] = sign_a[i] * (v0[i] * f1[i] - v1[i] * f3[i] + v3[i] * f5[i]);
366        inv[3][i] = sign_b[i] * (v0[i] * f2[i] - v1[i] * f4[i] + v2[i] * f5[i]);
367    }
368
369    let det = s(0, 0) * inv[0][0] + s(1, 0) * inv[0][1] + s(2, 0) * inv[0][2] + s(3, 0) * inv[0][3];
370
371    if det.abs() < 1e-10 {
372        return [
373            [1.0, 0.0, 0.0, 0.0],
374            [0.0, 1.0, 0.0, 0.0],
375            [0.0, 0.0, 1.0, 0.0],
376            [0.0, 0.0, 0.0, 1.0],
377        ];
378    }
379
380    let inv_det = 1.0 / det;
381    for col in &mut inv {
382        for val in col.iter_mut() {
383            *val *= inv_det;
384        }
385    }
386    inv
387}

396    /// World point to screen (top-left origin, y down), as the renderer
397    /// would draw it.
398    fn project(cam: &Camera, p: [f32; 3]) -> [f32; 2] {
399        let m = cam.view_projection_matrix();
400        let mut c = [0.0f32; 4];
401        for (row, o) in c.iter_mut().enumerate() {
402            *o = m[0][row] * p[0] + m[1][row] * p[1] + m[2][row] * p[2] + m[3][row];
403        }
404        assert!(c[3] > 0.0, "{p:?} is behind the camera");
405        let ndc = [c[0] / c[3], c[1] / c[3]];
406        [(ndc[0] * 0.5 + 0.5) * W, (0.5 - ndc[1] * 0.5) * H]
407    }

441    /// Ground points spread over what the camera sees: project each one to
442    /// the screen, then pick it back off the ground.
443    fn assert_round_trips(cam: &Camera, plane_y: f32, label: &str) {
444        let t = cam.target;
445        let mut checked = 0;
446        for i in -4..=4 {
447            for j in -4..=4 {
448                let p = [t.x + i as f32 * 9.0, plane_y, t.z + j as f32 * 9.0];
449                let [sx, sy] = project(cam, p);
450                if !(0.0..=W).contains(&sx) || !(0.0..=H).contains(&sy) {
451                    continue;
452                }
453                let hit = cam
454                    .screen_to_ground(sx, sy, W, H, plane_y)
455                    .unwrap_or_else(|| panic!("{label}: no hit for {p:?} at ({sx}, {sy})"));
456                let err = ((hit[0] - p[0]).powi(2) + (hit[1] - p[2]).powi(2)).sqrt();
457                assert!(
458                    err < 2e-3,
459                    "{label}: {p:?} came back as {hit:?} ({err} m off)"
460                );
461                checked += 1;
462            }
463        }
464        assert!(checked >= 20, "{label}: only {checked} points on screen");
465    }
