2//! Gerstner ocean rendering pipeline.
3//!
4//! One static grid mesh — normalized [-1,1]² coordinates, radially warped in
5//! the vertex shader so the center is dense and the rim reaches kilometers —
6//! follows the camera via a snapped `grid_offset` uniform. Vertices displace
7//! with the wave array generated by `flint_core::ocean::WaveSpectrum` (the
8//! CPU/GPU parity keystone: the GPU only sums what the CPU generated).
9//!
10//! Bind groups: 0 = transforms (shared), 1 = OceanUniforms, 2 = lights +
11//! shadows (shared). Group 3 stays free for the M4 refraction grab-pass.
12

15use flint_core::toml_util::{toml_color, toml_f32};

217    /// The world-space snap quantum for grid_offset: one inner grid cell.
218    /// Snapping the offset to cell multiples keeps wave sampling positions
219    /// world-anchored (no "swimming" as the camera moves).
220    pub fn snap_quantum(&self) -> f32 {
221        // Inner cell size ≈ grid_scale * (2 / cells) / 1.02 at the center.
222        self.grid_scale * 2.0 / OCEAN_GRID_CELLS as f32
223    }
224}

226/// Ocean grid vertex: just normalized 2D coordinates; everything else is
227/// derived in the shader.
228#[repr(C)]

230pub struct OceanVertex {
231    pub pos: [f32; 2],
232}

234impl OceanVertex {
235    const ATTRIBS: [wgpu::VertexAttribute; 1] = wgpu::vertex_attr_array![0 => Float32x2];
236
237    pub fn desc() -> wgpu::VertexBufferLayout<'static> {
238        wgpu::VertexBufferLayout {
239            array_stride: std::mem::size_of::<OceanVertex>() as wgpu::BufferAddress,
240            step_mode: wgpu::VertexStepMode::Vertex,
241            attributes: &Self::ATTRIBS,
242        }
243    }

246/// Generate the normalized grid: (cells+1)² vertices spanning [-1, 1]².
247pub fn generate_ocean_grid(cells: u32) -> (Vec<OceanVertex>, Vec<u32>) {
248    let n = cells + 1;
249    let mut vertices = Vec::with_capacity((n * n) as usize);
250    for j in 0..n {
251        for i in 0..n {
252            let x = i as f32 / cells as f32 * 2.0 - 1.0;
253            let z = j as f32 / cells as f32 * 2.0 - 1.0;
254            vertices.push(OceanVertex { pos: [x, z] });
255        }
256    }
257    let mut indices = Vec::with_capacity((cells * cells * 6) as usize);
258    for j in 0..cells {
259        for i in 0..cells {
260            let a = j * n + i;
261            let b = a + 1;
262            let c = a + n;
263            let d = c + 1;
264            indices.extend_from_slice(&[a, c, b, b, c, d]);
265        }
266    }
267    (vertices, indices)
268}

284/// Create a 1x1 texture usable as a grab-pass placeholder binding.

286    device: &wgpu::Device,

294            height: 1,

296        },
297        mip_level_count: 1,
298        sample_count: 1,
299        dimension: wgpu::TextureDimension::D2,

306impl OceanPipeline {
307    pub fn new(
308        device: &wgpu::Device,
309        scene_format: wgpu::TextureFormat,
310        transform_layout: &wgpu::BindGroupLayout,
311        light_layout: &wgpu::BindGroupLayout,
312    ) -> Self {
313        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
314            label: Some("Ocean Shader"),
315            source: wgpu::ShaderSource::Wgsl(include_str!("ocean_shader.wgsl").into()),
316        });
317
318        let ocean_bind_group_layout =
319            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
320                entries: &[wgpu::BindGroupLayoutEntry {
321                    binding: 0,
322                    visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
323                    ty: wgpu::BindingType::Buffer {
324                        ty: wgpu::BufferBindingType::Uniform,
325                        has_dynamic_offset: false,
326                        min_binding_size: None,
327                    },
328                    count: None,
329                }],
330                label: Some("Ocean Bind Group Layout"),
331            });
332
333        // Group 3: grabbed scene color + depth (textureLoad — no samplers).
334        let non_filtering_tex = |binding: u32| wgpu::BindGroupLayoutEntry {
335            binding,
336            visibility: wgpu::ShaderStages::FRAGMENT,
337            ty: wgpu::BindingType::Texture {
338                sample_type: wgpu::TextureSampleType::Float { filterable: false },
339                view_dimension: wgpu::TextureViewDimension::D2,
340                multisampled: false,
341            },
342            count: None,
343        };
344        let grab_bind_group_layout =
345            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
346                entries: &[non_filtering_tex(0), non_filtering_tex(1)],
347                label: Some("Ocean Grab Bind Group Layout"),
348            });
349
350        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
351            bind_group_layouts: &[
352                transform_layout,         // group 0
353                &ocean_bind_group_layout, // group 1
354                light_layout,             // group 2
355                &grab_bind_group_layout,  // group 3
356            ],
357            push_constant_ranges: &[],
358            label: Some("Ocean Pipeline Layout"),
359        });
360
361        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
362            label: Some("Ocean Pipeline"),
363            layout: Some(&pipeline_layout),
364            vertex: wgpu::VertexState {
365                module: &shader,
366                entry_point: Some("vs_main"),
367                buffers: &[OceanVertex::desc()],
368                compilation_options: Default::default(),
369            },
370            fragment: Some(wgpu::FragmentState {
371                module: &shader,
372                entry_point: Some("fs_main"),
373                targets: &[Some(wgpu::ColorTargetState {
374                    format: scene_format,
375                    blend: None, // opaque; depth-written so composite fog applies
376                    write_mask: wgpu::ColorWrites::ALL,
377                })],
378                compilation_options: Default::default(),
379            }),
380            primitive: wgpu::PrimitiveState {
381                topology: wgpu::PrimitiveTopology::TriangleList,
382                // Steep displaced triangles can face away briefly; draw both sides.
383                cull_mode: None,
384                ..Default::default()
385            },
386            depth_stencil: Some(wgpu::DepthStencilState {
387                format: wgpu::TextureFormat::Depth32Float,
388                depth_write_enabled: true,
389                depth_compare: wgpu::CompareFunction::Less,
390                stencil: Default::default(),
391                bias: Default::default(),
392            }),
393            multisample: Default::default(),
394            multiview: None,
395            cache: None,
396        });
397
398        // ── Grab blit pipeline: opaque scene color+depth → sampleable copies ──
399        let blit_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
400            label: Some("Grab Blit Shader"),
401            source: wgpu::ShaderSource::Wgsl(include_str!("grab_blit_shader.wgsl").into()),
402        });
403        let blit_bind_group_layout =
404            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
405                entries: &[
406                    non_filtering_tex(0),
407                    wgpu::BindGroupLayoutEntry {
408                        binding: 1,
409                        visibility: wgpu::ShaderStages::FRAGMENT,
410                        ty: wgpu::BindingType::Texture {
411                            sample_type: wgpu::TextureSampleType::Depth,
412                            view_dimension: wgpu::TextureViewDimension::D2,
413                            multisampled: false,
414                        },
415                        count: None,
416                    },
417                ],
418                label: Some("Grab Blit Bind Group Layout"),
419            });
420        let blit_pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
421            bind_group_layouts: &[&blit_bind_group_layout],
422            push_constant_ranges: &[],
423            label: Some("Grab Blit Pipeline Layout"),
424        });
425        let blit_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
426            label: Some("Grab Blit Pipeline"),
427            layout: Some(&blit_pipeline_layout),
428            vertex: wgpu::VertexState {
429                module: &blit_shader,
430                entry_point: Some("vs_blit"),
431                buffers: &[],
432                compilation_options: Default::default(),
433            },
434            fragment: Some(wgpu::FragmentState {
435                module: &blit_shader,
436                entry_point: Some("fs_blit"),
437                targets: &[
438                    Some(wgpu::ColorTargetState {
439                        format: crate::postprocess::HDR_FORMAT,
440                        blend: None,
441                        write_mask: wgpu::ColorWrites::ALL,
442                    }),
443                    Some(wgpu::ColorTargetState {
444                        format: wgpu::TextureFormat::R32Float,
445                        blend: None,
446                        write_mask: wgpu::ColorWrites::ALL,
447                    }),
448                ],
449                compilation_options: Default::default(),
450            }),
451            primitive: wgpu::PrimitiveState {
452                topology: wgpu::PrimitiveTopology::TriangleList,
453                cull_mode: None,
454                ..Default::default()
455            },
456            depth_stencil: None,
457            multisample: Default::default(),
458            multiview: None,
459            cache: None,
460        });
461
462        let (vertices, indices) = generate_ocean_grid(OCEAN_GRID_CELLS);
463        let grid_vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
464            label: Some("Ocean Grid Vertex Buffer"),
465            contents: bytemuck::cast_slice(&vertices),
466            usage: wgpu::BufferUsages::VERTEX,
467        });
468        let grid_index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
469            label: Some("Ocean Grid Index Buffer"),
470            contents: bytemuck::cast_slice(&indices),
471            usage: wgpu::BufferUsages::INDEX,
472        });
473
474        Self {
475            pipeline,
476            ocean_bind_group_layout,
477            grab_bind_group_layout,
478            blit_pipeline,
479            blit_bind_group_layout,
480            grid_vertex_buffer,
481            grid_index_buffer,
482            grid_index_count: indices.len() as u32,
483        }
484    }
485}

487#[cfg(test)]
488mod tests {
489    use super::*;
490
491    #[test]
492    fn uniform_struct_matches_wgsl_layout() {
493        // waves(512) + 4 colors(64) + grid_offset(8) + 2×u32(8) + 8×f32(32)
494        // + absorption vec3+turbidity(16) + refr/near/far/grab(16)
495        // + fog_color(16) + sky snapshot 3×vec4 + strength(64, incl. 3
496        // splash scalars) + contact hull 3×vec4(48) + band edge 4×f32(16)
497        // + foam_glow vec4(16) + mode u32 + mode_mix + pad2 (16) = 832
498        assert_eq!(std::mem::size_of::<OceanUniformsGpu>(), 832);
499        assert_eq!(832 % 16, 0, "uniform size must be 16-byte aligned");
500    }
501
502    #[test]
503    fn grid_indices_reference_valid_vertices() {
504        let (verts, indices) = generate_ocean_grid(8);
505        assert_eq!(verts.len(), 81);
506        assert_eq!(indices.len(), 8 * 8 * 6);
507        assert!(indices.iter().all(|&i| (i as usize) < verts.len()));
508    }
