1//! The renderer's side of runtime quality settings (FlintKart#55): masking
2//! a scene's post-process config by a tier's caps, and describing the GPU.
3//! `SceneRenderer::set_quality` applies a whole [`QualitySettings`].
4
5pub use flint_core::quality::{GpuInfo, QualityError, QualitySettings, QualityValue};

7use crate::postprocess::PostProcessConfig;

9/// `authored` with `quality`'s caps applied: an effect the caps turn off is
10/// off, one they leave on stays as the scene authored it.
11pub fn mask_post_process(
12    authored: &PostProcessConfig,
13    quality: &QualitySettings,
14) -> PostProcessConfig {
15    let mut c = authored.clone();
16    c.enabled &= quality.post_process;
17    c.bloom_enabled &= quality.bloom;
18    c.ssao_enabled &= quality.ssao;
19    c.ssao_samples = c.ssao_samples.min(quality.ssao_samples);
20    c.fog_enabled &= quality.fog;
21    c.fog_height_enabled &= quality.fog;
22    if !quality.fog {
23        c.ground_fog_density = 0.0;
24    }
25    c.volumetric_enabled &= quality.volumetric;
26    c
27}

29/// The vendor's name from its PCI id.
30fn vendor_name(id: u32) -> &'static str {
31    match id {
32        0x10de => "nvidia",
33        0x1002 | 0x1022 => "amd",
34        0x8086 => "intel",
35        0x13b5 => "arm",
36        0x5143 => "qualcomm",
37        0x106b => "apple",
38        0x1010 => "imagination",
39        0x1414 => "microsoft",
40        _ => "unknown",
41    }
42}

44/// The platform this build runs on.
45pub fn platform() -> &'static str {
46    if cfg!(target_arch = "wasm32") {
47        "web"
48    } else if cfg!(target_os = "android") {
49        "android"
50    } else if cfg!(target_os = "ios") {
51        "ios"
52    } else if cfg!(target_os = "windows") {
53        "windows"
54    } else if cfg!(target_os = "macos") {
55        "macos"
56    } else {
57        "linux"
58    }
59}

61/// The class a game picks a starting tier from; see [`GpuInfo::class`].
62pub fn gpu_class(device_type: wgpu::DeviceType, vendor: &str, platform: &str) -> &'static str {
63    use wgpu::DeviceType as D;
64    let mobile_vendor = matches!(vendor, "arm" | "qualcomm" | "imagination");
65    match device_type {
66        D::Cpu => "software",
67        _ if platform == "android" || platform == "ios" || mobile_vendor => "mobile",
68        D::DiscreteGpu => "desktop",
69        D::IntegratedGpu => "integrated",
70        // A browser reports what the page may know, often nothing.
71        D::VirtualGpu | D::Other => "unknown",
72    }
73}

75/// What `adapter` is; see [`GpuInfo`].
76pub fn gpu_info(adapter: &wgpu::Adapter) -> GpuInfo {
77    let info = adapter.get_info();
78    let vendor = vendor_name(info.vendor).to_string();
79    let platform = platform();
80    let device_type = match info.device_type {
81        wgpu::DeviceType::DiscreteGpu => "discrete",
82        wgpu::DeviceType::IntegratedGpu => "integrated",
83        wgpu::DeviceType::VirtualGpu => "virtual",
84        wgpu::DeviceType::Cpu => "cpu",
85        wgpu::DeviceType::Other => "other",
86    };
87    let backend = match info.backend {
88        wgpu::Backend::Vulkan => "vulkan",
89        wgpu::Backend::Dx12 => "dx12",
90        wgpu::Backend::Metal => "metal",
91        wgpu::Backend::Gl => "gl",
92        wgpu::Backend::BrowserWebGpu => "webgpu",
93        wgpu::Backend::Empty => "none",
94    };
95    let driver = [info.driver.as_str(), info.driver_info.as_str()]
96        .iter()
97        .filter(|s| !s.is_empty())
98        .copied()
99        .collect::<Vec<_>>()
100        .join(" ");
101    GpuInfo {
102        name: info.name.clone(),
103        class: gpu_class(info.device_type, &vendor, platform).to_string(),
104        vendor,
105        device_type: device_type.to_string(),
106        backend: backend.to_string(),
107        driver,
108        platform: platform.to_string(),
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    #[test]
117    fn caps_turn_effects_off_but_never_on() {
118        let authored = PostProcessConfig {
119            bloom_enabled: true,
120            ssao_enabled: false,
121            ssao_samples: 32,
122            fog_enabled: true,
123            ground_fog_density: 0.3,
124            ..PostProcessConfig::default()
125        };
126
127        let all_on = QualitySettings::default();
128        let same = mask_post_process(&authored, &all_on);
129        assert!(same.bloom_enabled);
130        assert!(
131            !same.ssao_enabled,
132            "a cap can't add SSAO the scene left out"
133        );
134        assert_eq!(same.ssao_samples, 32);
135        assert_eq!(same.ground_fog_density, 0.3);
136
137        let low = QualitySettings {
138            bloom: false,
139            ssao: true,
140            ssao_samples: 16,
141            fog: false,
142            ..QualitySettings::default()
143        };
144        let masked = mask_post_process(&authored, &low);
145        assert!(!masked.bloom_enabled);
146        assert!(!masked.fog_enabled);
147        assert_eq!(masked.ground_fog_density, 0.0);
148        assert_eq!(masked.ssao_samples, 16);
149        // The authored config is untouched, so lifting the caps restores it.
150        assert!(authored.bloom_enabled);
151    }
