1//! GPU-instanced particle render pipeline
2//!
3//! Renders camera-facing quads via instanced draw calls.
4//! Two pipelines: alpha blend and additive blend.
5//! Instance data comes from a storage buffer (supports 10,000+ particles).
6
7use bytemuck::{Pod, Zeroable};
8use wgpu::util::DeviceExt;
9
10/// GPU instance data for a single particle — matches WGSL struct layout.
11/// 48 bytes, 16-byte aligned (3 x vec4).
12#[repr(C)]
13#[derive(Copy, Clone, Debug, Pod, Zeroable)]
14pub struct ParticleInstanceGpu {
15    pub pos_size: [f32; 4],       // xyz = position, w = size
16    pub color: [f32; 4],          // rgba
17    pub rotation_frame: [f32; 4], // x = rotation, y = frame, z = frames_x, w = frames_y
18}
19
20/// Data for one emitter's particle draw, provided by the particle system
21pub struct ParticleDrawData<'a> {
22    pub instances: &'a [ParticleInstanceGpu],
23    pub texture: &'a str,
24    pub additive: bool,
25}
26
27/// Camera uniforms shared across all particle draws in a frame
28#[repr(C)]
29#[derive(Copy, Clone, Debug, Pod, Zeroable)]
30pub struct ParticleUniforms {
31    pub view_proj: [[f32; 4]; 4],
32    pub camera_right: [f32; 3],
33    pub _pad0: f32,
34    pub camera_up: [f32; 3],
35    pub _pad1: f32,
36}
37
38/// A particle draw call (one per emitter with alive particles)
39pub struct ParticleDrawCall {
40    pub instance_buffer: wgpu::Buffer,
41    pub instance_count: u32,
42    pub texture_bind_group: wgpu::BindGroup,
43    pub instance_bind_group: wgpu::BindGroup,
44    pub additive: bool,
45}
46
47/// The particle rendering pipeline (alpha + additive variants)
48pub struct ParticlePipeline {
49    pub alpha_pipeline: wgpu::RenderPipeline,
50    pub additive_pipeline: wgpu::RenderPipeline,
51    pub uniform_bind_group_layout: wgpu::BindGroupLayout,
52    pub instance_bind_group_layout: wgpu::BindGroupLayout,
53    pub texture_bind_group_layout: wgpu::BindGroupLayout,
54    pub quad_index_buffer: wgpu::Buffer,
55    pub uniform_buffer: wgpu::Buffer,
56    pub uniform_bind_group: wgpu::BindGroup,
57}
58
59impl ParticlePipeline {
60    pub fn new(device: &wgpu::Device, format: wgpu::TextureFormat) -> Self {
61        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
62            label: Some("Particle Shader"),
63            source: wgpu::ShaderSource::Wgsl(include_str!("particle_shader.wgsl").into()),
64        });
65
66        // Group 0: ParticleUniforms (camera data)
67        let uniform_bind_group_layout =
68            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
69                entries: &[wgpu::BindGroupLayoutEntry {
70                    binding: 0,
71                    visibility: wgpu::ShaderStages::VERTEX,
72                    ty: wgpu::BindingType::Buffer {
73                        ty: wgpu::BufferBindingType::Uniform,
74                        has_dynamic_offset: false,
75                        min_binding_size: None,
76                    },
77                    count: None,
78                }],
79                label: Some("Particle Uniform Bind Group Layout"),
80            });
81
82        // Group 1: Instance storage buffer (read-only)
83        let instance_bind_group_layout =
84            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
85                entries: &[wgpu::BindGroupLayoutEntry {
86                    binding: 0,
87                    visibility: wgpu::ShaderStages::VERTEX,
88                    ty: wgpu::BindingType::Buffer {
89                        ty: wgpu::BufferBindingType::Storage { read_only: true },
90                        has_dynamic_offset: false,
91                        min_binding_size: None,
92                    },
93                    count: None,
94                }],
95                label: Some("Particle Instance Bind Group Layout"),
96            });
97
98        // Group 2: Texture + sampler
99        let texture_bind_group_layout =
100            device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
101                entries: &[
102                    wgpu::BindGroupLayoutEntry {
103                        binding: 0,
104                        visibility: wgpu::ShaderStages::FRAGMENT,
105                        ty: wgpu::BindingType::Texture {
106                            sample_type: wgpu::TextureSampleType::Float { filterable: true },
107                            view_dimension: wgpu::TextureViewDimension::D2,
108                            multisampled: false,
109                        },
110                        count: None,
111                    },
112                    wgpu::BindGroupLayoutEntry {
113                        binding: 1,
114                        visibility: wgpu::ShaderStages::FRAGMENT,
115                        ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
116                        count: None,
117                    },
118                ],
119                label: Some("Particle Texture Bind Group Layout"),
120            });
121
122        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
123            label: Some("Particle Pipeline Layout"),
124            bind_group_layouts: &[
125                &uniform_bind_group_layout,
126                &instance_bind_group_layout,
127                &texture_bind_group_layout,
128            ],
129            push_constant_ranges: &[],
130        });
131
132        // Shared depth stencil: depth test enabled, depth write DISABLED (translucent)
133        let depth_stencil = wgpu::DepthStencilState {
134            format: wgpu::TextureFormat::Depth32Float,
135            depth_write_enabled: false,
136            depth_compare: wgpu::CompareFunction::Less,
137            stencil: wgpu::StencilState::default(),
138            bias: wgpu::DepthBiasState::default(),
139        };
140
141        // Alpha blend pipeline
142        let alpha_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
143            label: Some("Particle Alpha Pipeline"),
144            layout: Some(&pipeline_layout),
145            vertex: wgpu::VertexState {
146                module: &shader,
147                entry_point: Some("vs_particle"),
148                buffers: &[],
149                compilation_options: Default::default(),
150            },
151            fragment: Some(wgpu::FragmentState {
152                module: &shader,
153                entry_point: Some("fs_particle"),
154                targets: &[Some(wgpu::ColorTargetState {
155                    format,
156                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
157                    write_mask: wgpu::ColorWrites::ALL,
158                })],
159                compilation_options: Default::default(),
160            }),
161            primitive: wgpu::PrimitiveState {
162                topology: wgpu::PrimitiveTopology::TriangleList,
163                strip_index_format: None,
164                front_face: wgpu::FrontFace::Ccw,
165                cull_mode: None,
166                polygon_mode: wgpu::PolygonMode::Fill,
167                unclipped_depth: false,
168                conservative: false,
169            },
170            depth_stencil: Some(depth_stencil.clone()),
171            multisample: wgpu::MultisampleState::default(),
172            multiview: None,
173            cache: None,
174        });
175
176        // Additive blend pipeline (src_alpha + One)
177        let additive_blend = wgpu::BlendState {
178            color: wgpu::BlendComponent {
179                src_factor: wgpu::BlendFactor::SrcAlpha,
180                dst_factor: wgpu::BlendFactor::One,
181                operation: wgpu::BlendOperation::Add,
182            },
183            alpha: wgpu::BlendComponent {
184                src_factor: wgpu::BlendFactor::One,
185                dst_factor: wgpu::BlendFactor::One,
186                operation: wgpu::BlendOperation::Add,
187            },
188        };
189
190        let additive_pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
191            label: Some("Particle Additive Pipeline"),
192            layout: Some(&pipeline_layout),
193            vertex: wgpu::VertexState {
194                module: &shader,
195                entry_point: Some("vs_particle"),
196                buffers: &[],
197                compilation_options: Default::default(),
198            },
199            fragment: Some(wgpu::FragmentState {
200                module: &shader,
201                entry_point: Some("fs_particle"),
202                targets: &[Some(wgpu::ColorTargetState {
203                    format,
204                    blend: Some(additive_blend),
205                    write_mask: wgpu::ColorWrites::ALL,
206                })],
207                compilation_options: Default::default(),
208            }),
209            primitive: wgpu::PrimitiveState {
210                topology: wgpu::PrimitiveTopology::TriangleList,
211                strip_index_format: None,
212                front_face: wgpu::FrontFace::Ccw,
213                cull_mode: None,
214                polygon_mode: wgpu::PolygonMode::Fill,
215                unclipped_depth: false,
216                conservative: false,
217            },
218            depth_stencil: Some(depth_stencil),
219            multisample: wgpu::MultisampleState::default(),
220            multiview: None,
221            cache: None,
222        });
223
224        // Shared quad index buffer
225        let quad_indices: [u32; 6] = [0, 1, 2, 2, 1, 3];
226        let quad_index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
227            label: Some("Particle Quad Index Buffer"),
228            contents: bytemuck::cast_slice(&quad_indices),
229            usage: wgpu::BufferUsages::INDEX,
230        });
231
232        // Persistent uniform buffer for camera data
233        let uniform_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
234            label: Some("Particle Uniform Buffer"),
235            contents: bytemuck::cast_slice(&[ParticleUniforms {
236                view_proj: [[0.0; 4]; 4],
237                camera_right: [1.0, 0.0, 0.0],
238                _pad0: 0.0,
239                camera_up: [0.0, 1.0, 0.0],
240                _pad1: 0.0,
241            }]),
242            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
243        });
244
245        let uniform_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
246            layout: &uniform_bind_group_layout,
247            entries: &[wgpu::BindGroupEntry {
248                binding: 0,
249                resource: uniform_buffer.as_entire_binding(),
250            }],
251            label: Some("Particle Uniform Bind Group"),
252        });
253
254        Self {
255            alpha_pipeline,
256            additive_pipeline,
257            uniform_bind_group_layout,
258            instance_bind_group_layout,
259            texture_bind_group_layout,
260            quad_index_buffer,
261            uniform_buffer,
262            uniform_bind_group,
263        }
264    }
265}
