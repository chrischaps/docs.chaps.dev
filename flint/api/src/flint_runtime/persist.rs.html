1//! Persistent Data Store — key-value storage that survives scene transitions.
2//!
3//! Stores data as `toml::Value` for consistency with the ECS dynamic component
4//! system. Data can be saved to / loaded from TOML files for cross-session persistence.
5
6use std::collections::HashMap;
7use std::path::Path;
8
9/// A key-value store that persists across scene transitions.
10///
11/// Values are stored as [`toml::Value`] to match the engine's dynamic component
12/// system. The store can be serialized to / deserialized from TOML files.
13#[derive(Default)]
14pub struct PersistentStore {
15    data: HashMap<String, toml::Value>,
16}
17
18impl PersistentStore {
19    pub fn new() -> Self {
20        Self {
21            data: HashMap::new(),
22        }
23    }
24
25    /// Set a value by key. Overwrites any existing value.
26    pub fn set(&mut self, key: &str, value: toml::Value) {
27        self.data.insert(key.to_string(), value);
28    }
29
30    /// Get a value by key.
31    pub fn get(&self, key: &str) -> Option<&toml::Value> {
32        self.data.get(key)
33    }
34
35    /// Check if a key exists.
36    pub fn has(&self, key: &str) -> bool {
37        self.data.contains_key(key)
38    }
39
40    /// Remove a key, returning the old value if it existed.
41    pub fn remove(&mut self, key: &str) -> Option<toml::Value> {
42        self.data.remove(key)
43    }
44
45    /// Remove all entries.
46    pub fn clear(&mut self) {
47        self.data.clear();
48    }
49
50    /// Return all keys.
51    pub fn keys(&self) -> Vec<&str> {
52        self.data.keys().map(|k| k.as_str()).collect()
53    }
54
55    /// Save the store to a TOML file.
56    pub fn save_to_file(&self, path: &Path) -> Result<(), String> {
57        // Build a TOML table from the data
58        let mut table = toml::map::Map::new();
59        for (k, v) in &self.data {
60            table.insert(k.clone(), v.clone());
61        }
62        let content =
63            toml::to_string_pretty(&table).map_err(|e| format!("serialize error: {e}"))?;
64        std::fs::write(path, content).map_err(|e| format!("write error: {e}"))
65    }
66
67    /// Load the store from a TOML file, replacing all current data.
68    pub fn load_from_file(&mut self, path: &Path) -> Result<(), String> {
69        let content = std::fs::read_to_string(path).map_err(|e| format!("read error: {e}"))?;
70        let table: toml::map::Map<String, toml::Value> =
71            toml::from_str(&content).map_err(|e| format!("parse error: {e}"))?;
72        self.data.clear();
73        for (k, v) in table {
74            self.data.insert(k, v);
75        }
76        Ok(())
77    }
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83
84    #[test]
85    fn set_and_get() {
86        let mut store = PersistentStore::new();
87        store.set("score", toml::Value::Integer(42));
88        assert_eq!(store.get("score"), Some(&toml::Value::Integer(42)));
89    }
90
91    #[test]
92    fn has_and_remove() {
93        let mut store = PersistentStore::new();
94        store.set("name", toml::Value::String("Alice".into()));
95        assert!(store.has("name"));
96        assert!(!store.has("missing"));
97
98        let removed = store.remove("name");
99        assert_eq!(removed, Some(toml::Value::String("Alice".into())));
100        assert!(!store.has("name"));
101    }
102
103    #[test]
104    fn clear() {
105        let mut store = PersistentStore::new();
106        store.set("a", toml::Value::Integer(1));
107        store.set("b", toml::Value::Integer(2));
108        assert_eq!(store.keys().len(), 2);
109
110        store.clear();
111        assert_eq!(store.keys().len(), 0);
112    }
113
114    #[test]
115    fn keys() {
116        let mut store = PersistentStore::new();
117        store.set("x", toml::Value::Boolean(true));
118        store.set("y", toml::Value::Boolean(false));
119        let mut keys = store.keys();
120        keys.sort();
121        assert_eq!(keys, vec!["x", "y"]);
122    }
123
124    #[test]
125    fn overwrite() {
126        let mut store = PersistentStore::new();
127        store.set("val", toml::Value::Integer(1));
128        store.set("val", toml::Value::Integer(2));
129        assert_eq!(store.get("val"), Some(&toml::Value::Integer(2)));
130    }
131
132    #[test]
133    fn save_and_load() {
134        let dir = std::env::temp_dir().join("flint_persist_test");
135        let _ = std::fs::create_dir_all(&dir);
136        let path = dir.join("test_store.toml");
137
138        let mut store = PersistentStore::new();
139        store.set("score", toml::Value::Integer(100));
140        store.set("name", toml::Value::String("Player".into()));
141        store.save_to_file(&path).expect("save failed");
142
143        let mut loaded = PersistentStore::new();
144        loaded.load_from_file(&path).expect("load failed");
145        assert_eq!(loaded.get("score"), Some(&toml::Value::Integer(100)));
146        assert_eq!(
147            loaded.get("name"),
148            Some(&toml::Value::String("Player".into()))
149        );
150
151        let _ = std::fs::remove_file(&path);
152    }
153
154    #[test]
155    fn load_replaces_existing() {
156        let dir = std::env::temp_dir().join("flint_persist_test2");
157        let _ = std::fs::create_dir_all(&dir);
158        let path = dir.join("test_replace.toml");
159
160        let mut store = PersistentStore::new();
161        store.set("only_in_file", toml::Value::Boolean(true));
162        store.save_to_file(&path).expect("save failed");
163
164        let mut store2 = PersistentStore::new();
165        store2.set("old_key", toml::Value::Integer(0));
166        store2.load_from_file(&path).expect("load failed");
167
168        assert!(store2.has("only_in_file"));
169        assert!(!store2.has("old_key")); // replaced
170
171        let _ = std::fs::remove_file(&path);
172    }
