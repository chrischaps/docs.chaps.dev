1//! Animation sequences: timestamped animator events in a `*.sequence.toml`
2//!
3//! A sequence is a list of `[[events]]` sorted by `time`. Each event is one
4//! of the things a script can already do to an `animator` component —
5//! crossfade the base clip, set or fade a layer, change speed — plus named
6//! cues that scripts observe. The runner applies steps as plain ECS field
7//! writes, so the same file drives the `flint edit` previewer and the
8//! player identically, and seeking is a deterministic replay from `t = 0`.
9//!
10//! ```toml
11//! name = "showcase"
12//! loop = false
13//!
14//! [[events]]
15//! time = 1.5
16//! kind = "blend"
17//! clip = "WalkCycle"
18//! duration = 0.4
19//!
20//! [[events]]
21//! time = 2.5
22//! kind = "layer"
23//! index = 0
24//! clip = "StarCower"
25//! mask = "head"
26//! weight = 1.0
27//! fade = 0.3
28//!
29//! [[events]]
30//! time = 6.0
31//! kind = "cue"
32//! name = "done"
33//! ```

44/// Smallest crossfade the skeletal tier will run: `blend_duration <= 0`
45/// disables blending entirely, so a "hard cut" is a one-frame fade.
46pub const MIN_BLEND: f64 = 0.001;

48/// Ordered, timestamped animator events
49#[derive(Debug, Clone, Deserialize)]
50pub struct AnimSequence {
51    pub name: String,
52    /// Explicit length; defaults to the last event plus its transition
53    #[serde(default)]
54    pub duration: Option<f64>,
55    #[serde(default, rename = "loop")]
56    pub looping: bool,
57    #[serde(default)]
58    pub events: Vec<SequenceEvent>,
59}

61#[derive(Debug, Clone, Deserialize)]
62pub struct SequenceEvent {
63    pub time: f64,
64    #[serde(flatten)]
65    pub step: SequenceStep,
66}

68/// One thing to do to the animator at an event's time
69#[derive(Debug, Clone, Deserialize)]
70#[serde(tag = "kind", rename_all = "lowercase")]
71pub enum SequenceStep {
72    /// Crossfade the base clip (`duration = 0` → hard cut)
73    Blend {
74        clip: String,
75        #[serde(default)]
76        duration: f64,
77    },
78    /// Set a layer slot. Omitted fields keep their current value; `fade`
79    /// ramps the weight over that many seconds.
80    Layer {
81        index: usize,
82        #[serde(default)]
83        clip: Option<String>,
84        #[serde(default)]
85        weight: Option<f64>,
86        #[serde(default)]
87        fade: f64,
88        #[serde(default)]
89        mode: Option<String>,
90        #[serde(default)]
91        mask: Option<String>,
92    },
93    /// Base playback speed
94    Speed { value: f64 },
95    /// Named marker delivered to the entity's script (`on_sequence_cue`)
96    Cue { name: String },
97}
98
99impl SequenceStep {
100    /// Seconds the step keeps changing the pose after its event time
101    pub fn transition_len(&self) -> f64 {
102        match self {
103            SequenceStep::Blend { duration, .. } => duration.max(0.0),
104            SequenceStep::Layer { fade, .. } => fade.max(0.0),
105            _ => 0.0,
106        }
107    }
108
109    /// Short label for timelines / logs
110    pub fn label(&self) -> String {
111        match self {
112            SequenceStep::Blend { clip, duration } => {
113                if *duration > 0.0 {
114                    format!("blend → {clip} ({duration:.2}s)")
115                } else {
116                    format!("cut → {clip}")
117                }
118            }
119            SequenceStep::Layer {
120                index,
121                clip,
122                weight,
123                fade,
124                ..
125            } => {
126                let mut s = format!("L{index}");
127                if let Some(c) = clip {
128                    s.push_str(&format!(" {c}"));
129                }
130                if let Some(w) = weight {
131                    s.push_str(&format!(" w={w:.2}"));
132                }
133                if *fade > 0.0 {
134                    s.push_str(&format!(" fade {fade:.2}s"));
135                }
136                s
137            }
138            SequenceStep::Speed { value } => format!("speed ×{value:.2}"),
139            SequenceStep::Cue { name } => format!("cue '{name}'"),
140        }
141    }
142
143    pub fn kind(&self) -> &'static str {
144        match self {
145            SequenceStep::Blend { .. } => "blend",
146            SequenceStep::Layer { .. } => "layer",
147            SequenceStep::Speed { .. } => "speed",
148            SequenceStep::Cue { .. } => "cue",
149        }
150    }
151}

153impl AnimSequence {
154    /// Explicit `duration`, else the last event's time plus its transition
155    pub fn resolved_duration(&self) -> f64 {
156        self.duration.unwrap_or_else(|| {
157            self.events
158                .iter()
159                .map(|e| e.time + e.step.transition_len())
160                .fold(0.0, f64::max)
161        })
162    }
163
164    fn validate(mut self, origin: &str) -> Result<Self> {
165        if self.name.is_empty() {
166            return Err(FlintError::AnimationError(format!(
167                "{origin}: sequence has no name"
168            )));
169        }
170        if let Some(e) = self.events.iter().find(|e| e.time < 0.0) {
171            return Err(FlintError::AnimationError(format!(
172                "{origin}: event at negative time {}",
173                e.time
174            )));
175        }
176        if let Some(d) = self.duration {
177            if d <= 0.0 {
178                return Err(FlintError::AnimationError(format!(
179                    "{origin}: non-positive duration {d}"
180                )));
181            }
182        }
183        if self.looping && self.resolved_duration() <= 0.0 {
184            return Err(FlintError::AnimationError(format!(
185                "{origin}: looping sequence has zero duration"
186            )));
187        }
188        // Stable sort keeps same-time events in authored order
189        self.events.sort_by(|a, b| a.time.total_cmp(&b.time));
190        Ok(self)
191    }
192}

194/// Load a `*.sequence.toml`
195pub fn load_sequence_from_file(path: &Path) -> Result<AnimSequence> {
196    let content = std::fs::read_to_string(path).map_err(|e| {
197        FlintError::AnimationError(format!("Failed to read {}: {}", path.display(), e))
198    })?;
199    load_sequence_from_str(&content, &path.display().to_string())
200}

202/// Parse a sequence from TOML text; `origin` names it in errors
203pub fn load_sequence_from_str(content: &str, origin: &str) -> Result<AnimSequence> {
204    let seq: AnimSequence = toml::from_str(content)
205        .map_err(|e| FlintError::AnimationError(format!("Failed to parse {origin}: {e}")))?;
206    seq.validate(origin)
207}

219/// Per-entity playback of one sequence
220#[derive(Debug, Clone)]
221pub struct SequenceRuntime {
222    pub sequence: Arc<AnimSequence>,
223    pub time: f64,
224    pub playing: bool,
225    /// Runtime loop override (previewer checkbox); `None` = file's value
226    pub loop_override: Option<bool>,
227    fired: Vec<bool>,
228}
229
230impl SequenceRuntime {
231    fn new(sequence: Arc<AnimSequence>) -> Self {
232        let n = sequence.events.len();
233        Self {
234            sequence,
235            time: 0.0,
236            playing: true,
237            loop_override: None,
238            fired: vec![false; n],
239        }
240    }
241
242    pub fn name(&self) -> &str {
243        &self.sequence.name
244    }
245
246    pub fn duration(&self) -> f64 {
247        self.sequence.resolved_duration()
248    }
249
250    pub fn looping(&self) -> bool {
251        self.loop_override.unwrap_or(self.sequence.looping)
252    }
253
254    /// Whether event `i` has fired since the last (re)start / loop wrap
255    pub fn fired(&self, i: usize) -> bool {
256        self.fired.get(i).copied().unwrap_or(false)
257    }
258
259    pub fn fired_count(&self) -> usize {
260        self.fired.iter().filter(|f| **f).count()
261    }
262
263    /// Index of the most recently fired event, if any
264    pub fn last_fired(&self) -> Option<usize> {
265        self.fired.iter().rposition(|f| *f)
266    }
267
268    /// Index of the next unfired event, if any
269    pub fn next_unfired(&self) -> Option<usize> {
270        self.fired.iter().position(|f| !*f)
271    }
272
273    fn restart(&mut self) {
274        self.time = 0.0;
275        self.playing = true;
276        self.fired.iter_mut().for_each(|f| *f = false);
277    }
278}

280/// Registry of sequences plus the per-entity runtimes driving them.
281///
282/// Runs *before* the skeletal tier each frame so its ECS writes are picked
283/// up by `SkeletalSync::sync_from_world` in the same update. Scripts start
284/// and stop sequences through the `animator.sequence` field (see
285/// [`Self::sync_from_world`]); the previewer calls [`Self::play`] directly.
286#[derive(Default)]

292    fired_cues: Vec<SequenceCueEvent>,
293}

375    /// Follow the `animator.sequence` field: a new non-empty value starts
376    /// that sequence, an edge to `""` stops the active one. The field is
377    /// the scripting surface (`play_sequence` / `stop_sequence` just write
378    /// it) and lets a scene autoplay a sequence by authoring it.
379    ///
380    /// Only *edges* of the ECS value count — `last_seen` tracks what the
381    /// field said last frame, never what [`Self::play`] started — so a
382    /// runtime started directly (the previewer) survives an empty field.
383    pub fn sync_from_world(&mut self, world: &FlintWorld) {
384        let ids: Vec<EntityId> = world
385            .entities_with_component(comp::ANIMATOR)
386            .iter()
387            .copied()
388            .collect();
389        for entity_id in ids {
390            let wanted = world
391                .get_components(entity_id)
392                .and_then(|c| c.get(comp::ANIMATOR))
393                .and_then(|a| a.get("sequence"))
394                .and_then(|v| v.as_str())
395                .unwrap_or("")
396                .to_string();
397            let seen = self.last_seen.get(&entity_id).cloned().unwrap_or_default();
398            if wanted == seen {
399                continue;
400            }
401            if wanted.is_empty() {
402                self.stop(&entity_id);
403            } else if !self.play(entity_id, &wanted) {
404                println!(
405                    "WARNING: entity {:?} asked for unknown sequence '{}'",
406                    entity_id, wanted
407                );
408            }
409            self.last_seen.insert(entity_id, wanted);
410        }
411        // Forget entities that vanished
412        self.active
413            .retain(|id, _| world.get_components(*id).is_some());
414        self.last_seen
415            .retain(|id, _| world.get_components(*id).is_some());
416    }

418    /// Advance every playing runtime by `dt`, applying passed events to the
419    /// ECS. Events fire exactly once per pass (inclusive of `time == t`).
420    pub fn advance(&mut self, world: &mut FlintWorld, dt: f64) {
421        let ids: Vec<EntityId> = self.active.keys().copied().collect();
422        for entity_id in ids {
423            let Some(rt) = self.active.get_mut(&entity_id) else {
424                continue;
425            };
426            if !rt.playing {
427                continue;
428            }
429            rt.time += dt;
430            let seq = rt.sequence.clone();
431            let duration = seq.resolved_duration();
432
433            if rt.looping() && duration > 0.0 {
434                // Fire the tail of the current pass (everything up to
435                // `duration`), then wrap and fire from zero. Repeat while
436                // the frame still overruns, so a large `dt` neither skips
437                // events nor delays the next pass by a frame.
438                loop {
439                    let limit = rt.time.min(duration);
440                    for (i, ev) in seq.events.iter().enumerate() {
441                        if rt.fired[i] {
442                            continue;
443                        }
444                        if ev.time > limit {
445                            break;
446                        }
447                        rt.fired[i] = true;
448                        Self::apply_step(world, entity_id, &seq.name, ev, &mut self.fired_cues);
449                    }
450                    if rt.time < duration {
451                        break;
452                    }
453                    rt.time -= duration;
454                    rt.fired.iter_mut().for_each(|f| *f = false);
455                }
456            } else {
457                for (i, ev) in seq.events.iter().enumerate() {
458                    if rt.fired[i] {
459                        continue;
460                    }
461                    if ev.time > rt.time {
462                        break;
463                    }
464                    rt.fired[i] = true;
465                    Self::apply_step(world, entity_id, &seq.name, ev, &mut self.fired_cues);
466                }
467
468                if duration > 0.0 && rt.time >= duration {
469                    rt.time = duration;
470                    rt.playing = false;
471                    // Retire the request so re-playing the same name is an
472                    // edge again (mirrors `blend_target` being cleared).
473                    if self
474                        .last_seen
475                        .get(&entity_id)
476                        .is_some_and(|s| !s.is_empty())
477                    {
478                        if let Some(comps) = world.get_components_mut(entity_id) {
479                            comps.set_field(
480                                comp::ANIMATOR,
481                                "sequence",
482                                toml::Value::String(String::new()),
483                            );
484                        }
485                        self.last_seen.insert(entity_id, String::new());
486                    }
487                } else if duration <= 0.0 && !rt.looping() {
488                    rt.time = 0.0;
489                    rt.playing = false;
490                    if self
491                        .last_seen
492                        .get(&entity_id)
493                        .is_some_and(|s| !s.is_empty())
494                    {
495                        if let Some(comps) = world.get_components_mut(entity_id) {
496                            comps.set_field(
497                                comp::ANIMATOR,
498                                "sequence",
499                                toml::Value::String(String::new()),
500                            );
501                        }
502                        self.last_seen.insert(entity_id, String::new());
503                    }
504                }
505            }
506        }
507    }

509    /// Apply one event as animator field writes — exactly what the Rhai
510    /// `blend_to` / `set_anim_layer*` / `set_anim_speed` calls do.
511    fn apply_step(
512        world: &mut FlintWorld,
513        entity_id: EntityId,
514        sequence: &str,
515        ev: &SequenceEvent,
516        cues: &mut Vec<SequenceCueEvent>,
517    ) {
518        let Some(comps) = world.get_components_mut(entity_id) else {
519            return;
520        };
521        match &ev.step {
522            SequenceStep::Blend { clip, duration } => {
523                // Only `blend_target` changes a tracked skeletal base clip;
524                // `clip` is latched at state creation. Set both so an
525                // untracked entity also starts on the right clip.
526                let has_clip = comps
527                    .get(comp::ANIMATOR)
528                    .and_then(|a| a.get("clip"))
529                    .and_then(|v| v.as_str())
530                    .is_some_and(|c| !c.is_empty());
531                if !has_clip {
532                    comps.set_field(comp::ANIMATOR, "clip", toml::Value::String(clip.clone()));
533                }
534                comps.set_field(comp::ANIMATOR, "playing", toml::Value::Boolean(true));
535                comps.set_field(
536                    comp::ANIMATOR,
537                    "blend_target",
538                    toml::Value::String(clip.clone()),
539                );
540                comps.set_field(
541                    comp::ANIMATOR,
542                    "blend_duration",
543                    toml::Value::Float(duration.max(MIN_BLEND)),
544                );
545            }
546            SequenceStep::Layer {
547                index,
548                clip,
549                weight,
550                fade,
551                mode,
552                mask,
553            } => {
554                layer_edit::edit_layer_table(comps, *index, |t| {
555                    // A slot that was inactive has no meaningful weight;
556                    // a fade onto it should start from silence, not 1.0.
557                    let was_inactive = t
558                        .get("clip")
559                        .and_then(|v| v.as_str())
560                        .map_or(true, |c| c.is_empty());
561                    if was_inactive && *fade > 0.0 && !t.contains_key("weight") {
562                        t.insert("weight".into(), toml::Value::Float(0.0));
563                    }
564                    if let Some(c) = clip {
565                        t.insert("clip".into(), toml::Value::String(c.clone()));
566                    }
567                    if let Some(m) = mode {
568                        t.insert("mode".into(), toml::Value::String(m.clone()));
569                    }
570                    if let Some(m) = mask {
571                        t.insert("mask".into(), toml::Value::String(m.clone()));
572                    }
573                    if let Some(w) = weight {
574                        layer_edit::fade_weight(t, *w, *fade);
575                    }
576                });
577            }
578            SequenceStep::Speed { value } => {
579                comps.set_field(comp::ANIMATOR, "speed", toml::Value::Float(*value));
580            }
581            SequenceStep::Cue { name } => cues.push(SequenceCueEvent {
582                entity_id,
583                sequence: sequence.to_string(),
584                cue: name.clone(),
585                time: ev.time,
586            }),
587        }
588    }

741        assert_eq!(sync.state(&id).unwrap().fired_count(), 1);
742
743        // Walk to exactly 1.5 in 0.1 steps: event at time == t fires
744        for _ in 0..15 {
745            sync.advance(&mut world, 0.1);
746        }
747        let a = animator(&world, id);
748        assert_eq!(a["blend_target"].as_str(), Some("WalkCycle"));
749        assert_eq!(sync.state(&id).unwrap().fired_count(), 2);
750
751        // Past the layer event: fade fields written, no double fire
752        for _ in 0..12 {
753            sync.advance(&mut world, 0.1);
754        }
755        let a = animator(&world, id);
756        let layer = &a["layers"].as_array().unwrap()[0];
757        assert_eq!(layer["clip"].as_str(), Some("StarCower"));
758        assert_eq!(layer["mask"].as_str(), Some("head"));
759        assert_eq!(toml_f64(&layer["fade_target"]), Some(1.0));
760        assert!((toml_f64(&layer["fade_duration"]).unwrap() - 0.3).abs() < 1e-9);
761        assert_eq!(sync.state(&id).unwrap().fired_count(), 3);
762
763        // Run out: cue drained once, sequence stops and clears its request
764        for _ in 0..40 {
765            sync.advance(&mut world, 0.1);
766        }
767        let cues = sync.drain_cues();
768        assert_eq!(cues.len(), 1);
769        assert_eq!(cues[0].cue, "done");
770        assert!(sync.drain_cues().is_empty());
771        let rt = sync.state(&id).unwrap();
772        assert!(!rt.playing);
773        assert_eq!(rt.fired_count(), 6);
774        assert_eq!(toml_f64(&animator(&world, id)["speed"]), Some(1.3));
775    }
