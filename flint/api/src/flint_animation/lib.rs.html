1//! Animation system for Flint engine
2//!
3//! Provides four tiers of animation:
4//! - **Tier 1**: Property tweens — animate any TOML component field via keyframes
5//! - **Tier 2**: Skeletal animation — glTF skin/joint hierarchies with GPU skinning
6//! - **Tier 3**: Node animation — per-node transform animation for scene graphs
7//! - **Tier 4**: Sprite sheet animation — frame-based sprite animation from `.sprite.toml` clips

44/// Top-level animation system integrating clip playback with the ECS world.
45///
46/// Supports property tweens (Tier 1), skeletal animation (Tier 2),
47/// node animation (Tier 3), and sprite sheet animation (Tier 4).
48/// Implements `RuntimeSystem`, bridges TOML components via the various Sync types.
49#[derive(Default)]
50pub struct AnimationSystem {
51    pub(crate) player: AnimationPlayer,
52    pub(crate) sync: AnimationSync,
53    pub(crate) skeletal_sync: SkeletalSync,
54    pub(crate) node_sync: NodeSync,
55    pub(crate) sprite_sync: SpriteAnimSync,
56}

192impl RuntimeSystem for AnimationSystem {
193    fn initialize(&mut self, world: &mut FlintWorld) -> Result<()> {
194        self.sync.sync_from_world(world, &self.player);
195        self.skeletal_sync.sync_from_world(world);
196        self.node_sync.sync_from_world(world);
197        self.sprite_sync.sync_from_world(world);
198        println!(
199            "Animation system initialized ({} property clips, {} skeletal clips, {} node clips, {} sprite clips, {} property entities, {} skeletal entities, {} node entities, {} sprite entities)",
200            self.player.clip_count(),
201            self.skeletal_sync.clip_count(),
202            self.node_sync.clip_count(),
203            self.sprite_sync.clip_count(),
204            self.sync.active_count(),
205            self.skeletal_sync.active_count(),
206            self.node_sync.active_count(),
207            self.sprite_sync.active_count()
208        );
209        Ok(())
210    }
211
212    fn fixed_update(&mut self, _world: &mut FlintWorld, _dt: f64) -> Result<()> {
213        // Animation interpolates smoothly in variable update — no-op here
214        Ok(())
215    }
216
217    fn update(&mut self, world: &mut FlintWorld, dt: f64) -> Result<()> {
218        // Tier 1: Property animation
219        self.sync.sync_from_world(world, &self.player);
220        self.sync.advance_and_write(world, &self.player, dt);
221
222        // Tier 2: Skeletal animation
223        self.skeletal_sync.sync_from_world(world);
224        self.skeletal_sync.advance_and_compute(dt);
225
226        // Tier 3: Node transform animation
227        self.node_sync.sync_from_world(world);
228        self.node_sync.advance_and_apply(world, dt);
229
230        // Tier 4: Sprite sheet animation
231        self.sprite_sync.sync_from_world(world);
232        self.sprite_sync.advance_and_write(world, dt);
233
234        Ok(())
235    }
236
237    fn shutdown(&mut self) -> Result<()> {
238        println!("Animation system shut down");
239        Ok(())
240    }
241
242    fn name(&self) -> &str {
243        "animation"
244    }
