1//! Bridges ECS entities to node-level animation playback
2//!
3//! Manages per-entity node animation state, clip playback, and transform writes.
4//! Analogous to `SkeletalSync` but writes to child entity transforms instead of
5//! computing bone matrices for GPU skinning.

17/// Per-entity node animation playback state (alias for the shared [`ClipPlaybackState`])

20/// Registry of node clips and per-entity playback state
21#[derive(Default)]
22pub struct NodeSync {
23    clips: HashMap<String, NodeClip>,
24    states: HashMap<EntityId, NodePlaybackState>,
25    /// Per root entity: node_name → child EntityId
26    node_maps: HashMap<EntityId, HashMap<String, EntityId>>,
27}

70    /// Scan the world for entities with `animator` (but no `skeleton`) that are
71    /// registered in `node_maps`. Creates playback states for newly discovered entities.
72    /// Detects clip/speed/blend changes for existing entities.
73    pub fn sync_from_world(&mut self, world: &FlintWorld) {
74        for &entity_id in world.entities_with_component(comp::ANIMATOR) {
75            // Must be registered in our node_maps
76            if !self.node_maps.contains_key(&entity_id) {
77                continue;
78            }
79
80            let Some(components) = world.get_components(entity_id) else {
81                continue;
82            };
83
84            // Must have animator but NOT skeleton (skeletal handled by SkeletalSync)
85            let Some(animator) = components.get(comp::ANIMATOR) else {
86                continue;
87            };
88            if components.get(comp::SKELETON).is_some() {
89                continue;
90            }
91
92            // If already tracked, check for clip changes and blend_target changes
93            if let Some(state) = self.states.get_mut(&entity_id) {
94                // Check if clip name changed (e.g. script called play_clip)
95                let ecs_clip = animator
96                    .get("clip")
97                    .and_then(|v| v.as_str())
98                    .unwrap_or("")
99                    .to_string();
100                let ecs_playing = animator
101                    .get("playing")
102                    .and_then(|v| v.as_bool())
103                    .unwrap_or(false);
104                let ecs_speed = animator.get("speed").and_then(toml_f64).unwrap_or(1.0);
105                let ecs_looping = animator
106                    .get("loop")
107                    .and_then(|v| v.as_bool())
108                    .unwrap_or(true);
109
110                // Clip changed — reset playback
111                if !ecs_clip.is_empty()
112                    && ecs_clip != state.clip_name
113                    && self.clips.contains_key(&ecs_clip)
114                {
115                    state.clip_name = ecs_clip;
116                    state.time = 0.0;
117                    state.playing = ecs_playing;
118                    state.speed = ecs_speed;
119                    state.looping = ecs_looping;
120                    state.blend_target.clear();
121                    state.blend_elapsed = 0.0;
122                    continue;
123                }
124
125                // Sync playing/speed/loop state
126                state.playing = ecs_playing;
127                state.speed = ecs_speed;
128                state.looping = ecs_looping;
129
130                // Check for blend_target changes
131                let ecs_blend_target = animator
132                    .get("blend_target")
133                    .and_then(|v| v.as_str())
134                    .unwrap_or("")
135                    .to_string();
136                let ecs_blend_duration = animator
137                    .get("blend_duration")
138                    .and_then(toml_f32)
139                    .unwrap_or(0.3);
140
141                if !ecs_blend_target.is_empty()
142                    && ecs_blend_target != state.blend_target
143                    && self.clips.contains_key(&ecs_blend_target)
144                {
145                    state.blend_target = ecs_blend_target;
146                    state.blend_duration = ecs_blend_duration;
147                    state.blend_elapsed = 0.0;
148                }
149                continue;
150            }
151
152            // New entity — create playback state
153            let clip_name = animator
154                .get("clip")
155                .and_then(|v| v.as_str())
156                .unwrap_or("")
157                .to_string();
158
159            if clip_name.is_empty() || !self.clips.contains_key(&clip_name) {
160                continue;
161            }
162
163            let speed = animator.get("speed").and_then(toml_f64).unwrap_or(1.0);
164
165            let looping = animator
166                .get("loop")
167                .and_then(|v| v.as_bool())
168                .unwrap_or(true);
169
170            let autoplay = animator
171                .get("autoplay")
172                .and_then(|v| v.as_bool())
173                .unwrap_or(false);
174
175            let playing = animator
176                .get("playing")
177                .and_then(|v| v.as_bool())
178                .unwrap_or(false)
179                || autoplay;
180
181            let mut state = NodePlaybackState::new(clip_name, speed, looping, playing);
182
183            state.blend_target = animator
184                .get("blend_target")
185                .and_then(|v| v.as_str())
186                .unwrap_or("")
187                .to_string();
188            state.blend_duration = animator
189                .get("blend_duration")
190                .and_then(toml_f32)
191                .unwrap_or(0.3);
192
193            self.states.insert(entity_id, state);
194        }
195    }

202    /// Set the playback time for a given entity (used for timeline scrubbing)

214    /// Advance all node animation playbacks and write transforms to child entities.
215    pub fn advance_and_apply(&mut self, world: &mut FlintWorld, dt: f64) {
216        let entity_ids: Vec<EntityId> = self.states.keys().copied().collect();
217
218        for entity_id in entity_ids {
219            let state = self.states.get_mut(&entity_id).unwrap();
220            let Some(clip) = self.clips.get(&state.clip_name) else {
221                continue;
222            };
223            let Some(node_map) = self.node_maps.get(&entity_id) else {
224                continue;
225            };
226
227            // Advance time
228            if state.playing {
229                state.time += dt * state.speed;
230
231                if state.looping {
232                    if clip.duration > 0.0 {
233                        if state.time >= clip.duration {
234                            state.time %= clip.duration;
235                        } else if state.time < 0.0 {
236                            state.time = clip.duration - (-state.time % clip.duration);
237                        }
238                    }
239                } else if state.time >= clip.duration {
240                    state.time = clip.duration;
241                    state.playing = false;
242                } else if state.time < 0.0 {
243                    state.time = 0.0;
244                    state.playing = false;
245                }
246            }
247
248            // Sample and apply each node track
249            for track in &clip.node_tracks {
250                let Some(&child_entity_id) = node_map.get(&track.node_name) else {
251                    continue;
252                };
253
254                // Convert NodeTrack to JointTrack for sampling (reuse existing sampler)
255                let joint_track = JointTrack {
256                    joint_index: 0, // unused for sampling
257                    property: track.property.clone(),
258                    interpolation: track.interpolation.clone(),
259                    keyframes: track.keyframes.clone(),
260                };
261
262                let value = sample_joint_track(&joint_track, state.time);
263
264                // Write to child entity's transform
265                apply_sampled_value(world, child_entity_id, &track.property, &value);
266            }
267        }
268    }

271/// Write a sampled animation value to an entity's transform component.
272///
273/// Uses `world.set_field()` which creates the component if needed and
274/// maintains the component index.
275fn apply_sampled_value(
276    world: &mut FlintWorld,
277    entity_id: EntityId,
278    property: &JointProperty,
279    value: &[f32],
280) {
281    match property {
282        JointProperty::Translation => {
283            if value.len() >= 3 {
284                let _ = world.set_field(
285                    entity_id,
286                    comp::TRANSFORM,
287                    "position",
288                    toml::Value::Array(vec![
289                        toml::Value::Float(value[0] as f64),
290                        toml::Value::Float(value[1] as f64),
291                        toml::Value::Float(value[2] as f64),
292                    ]),
293                );
294            }
295        }
296        JointProperty::Rotation => {
297            if value.len() >= 4 {
298                let _ = world.set_field(
299                    entity_id,
300                    comp::TRANSFORM,
301                    "rotation_quat",
302                    toml::Value::Array(vec![
303                        toml::Value::Float(value[0] as f64),
304                        toml::Value::Float(value[1] as f64),
305                        toml::Value::Float(value[2] as f64),
306                        toml::Value::Float(value[3] as f64),
307                    ]),
308                );
309            }
310        }
311        JointProperty::Scale => {
312            if value.len() >= 3 {
313                let _ = world.set_field(
314                    entity_id,
315                    comp::TRANSFORM,
316                    "scale",
317                    toml::Value::Array(vec![
318                        toml::Value::Float(value[0] as f64),
319                        toml::Value::Float(value[1] as f64),
320                        toml::Value::Float(value[2] as f64),
321                    ]),
322                );
323            }
324        }
325    }
