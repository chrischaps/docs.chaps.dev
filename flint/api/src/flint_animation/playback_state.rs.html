1//! Unified playback state for skeletal and node animation clips
2
3/// Per-entity clip playback state shared by skeletal and node animation.
4///
5/// Tracks current time, speed, looping, and crossfade blending parameters.
6#[derive(Debug, Clone)]
7pub struct ClipPlaybackState {
8    pub clip_name: String,
9    pub time: f64,
10    pub speed: f64,
11    pub looping: bool,
12    pub playing: bool,
13    /// Clip name to crossfade into (empty = no blend)
14    pub blend_target: String,
15    /// Duration of the crossfade in seconds
16    pub blend_duration: f32,
17    /// Time elapsed in the current blend
18    pub blend_elapsed: f32,
19}
20
21impl ClipPlaybackState {
22    pub fn new(clip_name: String, speed: f64, looping: bool, playing: bool) -> Self {
23        Self {
24            clip_name,
25            time: 0.0,
26            speed,
27            looping,
28            playing,
29            blend_target: String::new(),
30            blend_duration: 0.3,
31            blend_elapsed: 0.0,
32        }
33    }
34}
