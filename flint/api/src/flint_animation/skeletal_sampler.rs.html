6/// Sample a joint track at a given time.
7///
8/// Returns the interpolated value as a `Vec<f32>` (3 for translation/scale, 4 for rotation).
9pub fn sample_joint_track(track: &JointTrack, time: f64) -> Vec<f32> {
10    let keyframes = &track.keyframes;
11    let is_rotation = track.property == JointProperty::Rotation;
12    let components = if is_rotation { 4 } else { 3 };
13
14    if keyframes.is_empty() {
15        return if is_rotation {
16            vec![0.0, 0.0, 0.0, 1.0] // identity quaternion
17        } else {
18            vec![0.0; 3]
19        };
20    }
21
22    // Before first keyframe — clamp
23    if time <= keyframes[0].time {
24        return keyframes[0].value.clone();
25    }
26
27    // After last keyframe — clamp
28    let last = &keyframes[keyframes.len() - 1];
29    if time >= last.time {
30        return last.value.clone();
31    }
32
33    // Binary search for the interval
34    let idx = match keyframes.binary_search_by(|kf| kf.time.partial_cmp(&time).unwrap()) {
35        Ok(i) => return keyframes[i].value.clone(),
36        Err(i) => i,
37    };
38
39    let prev = &keyframes[idx - 1];
40    let next = &keyframes[idx];
41
42    let span = next.time - prev.time;
43    if span <= 0.0 {
44        return prev.value.clone();
45    }
46    let t = ((time - prev.time) / span) as f32;
47
48    match track.interpolation {
49        Interpolation::Step => prev.value.clone(),
50        Interpolation::Linear => {
51            if is_rotation {
52                quat_slerp(&prev.value, &next.value, t)
53            } else {
54                lerp_vec(&prev.value, &next.value, t, components)
55            }
56        }
57        Interpolation::CubicSpline => {
58            // Tangents are only present when the importer saw a CUBICSPLINE sampler;
59            // degrade gracefully to linear if they're missing.
60            if prev.out_tangent.len() < components || next.in_tangent.len() < components {
61                return if is_rotation {
62                    quat_slerp(&prev.value, &next.value, t)
63                } else {
64                    lerp_vec(&prev.value, &next.value, t, components)
65                };
66            }
67            let mut v = hermite_vec(
68                &prev.value,
69                &prev.out_tangent,
70                &next.value,
71                &next.in_tangent,
72                t,
73                span as f32,
74                components,
75            );
76            if is_rotation {
77                normalize_quat(&mut v);
78            }
79            v
80        }
81    }
82}

84/// Cubic Hermite interpolation as specified by glTF for CUBICSPLINE samplers.
85///
86/// `span` is the keyframe interval in seconds; glTF tangents are per-second slopes
87/// so they are scaled by the interval. Quaternion results must be normalized by the caller.
88fn hermite_vec(
89    p0: &[f32],
90    out_tangent: &[f32],
91    p1: &[f32],
92    in_tangent: &[f32],
93    t: f32,
94    span: f32,
95    count: usize,
96) -> Vec<f32> {
97    let t2 = t * t;
98    let t3 = t2 * t;
99    let h00 = 2.0 * t3 - 3.0 * t2 + 1.0;
100    let h10 = t3 - 2.0 * t2 + t;
101    let h01 = -2.0 * t3 + 3.0 * t2;
102    let h11 = t3 - t2;
103    (0..count)
104        .map(|i| {
105            h00 * p0[i] + h10 * span * out_tangent[i] + h01 * p1[i] + h11 * span * in_tangent[i]
106        })
107        .collect()
108}
109
110fn normalize_quat(q: &mut [f32]) {
111    let len = q.iter().map(|c| c * c).sum::<f32>().sqrt();
112    if len > 1e-6 {
113        for c in q.iter_mut() {
114            *c /= len;
115        }
116    } else {
117        q.copy_from_slice(&[0.0, 0.0, 0.0, 1.0]);
118    }
119}

121/// Component-wise linear interpolation for a Vec<f32>
122fn lerp_vec(a: &[f32], b: &[f32], t: f32, count: usize) -> Vec<f32> {
123    (0..count)
124        .map(|i| {
125            let av = a.get(i).copied().unwrap_or(0.0);
126            let bv = b.get(i).copied().unwrap_or(0.0);
127            av + (bv - av) * t
128        })
129        .collect()
130}

132/// Quaternion spherical linear interpolation (slerp) with shortest-path correction.
133///
134/// Input quaternions are xyzw format. The result is normalized.
135pub fn quat_slerp(a: &[f32], b: &[f32], t: f32) -> Vec<f32> {
136    if a.len() < 4 || b.len() < 4 {
137        return vec![0.0, 0.0, 0.0, 1.0];
138    }
139
140    let (ax, ay, az, aw) = (a[0], a[1], a[2], a[3]);
141    let (mut bx, mut by, mut bz, mut bw) = (b[0], b[1], b[2], b[3]);
142
143    // Compute dot product
144    let mut dot = ax * bx + ay * by + az * bz + aw * bw;
145
146    // Shortest path: if dot < 0, negate b
147    if dot < 0.0 {
148        bx = -bx;
149        by = -by;
150        bz = -bz;
151        bw = -bw;
152        dot = -dot;
153    }
154
155    // If quaternions are very close, use linear interpolation to avoid division by zero
156    let (scale_a, scale_b) = if dot > 0.9995 {
157        (1.0 - t, t)
158    } else {
159        let theta = dot.acos();
160        let sin_theta = theta.sin();
161        (
162            ((1.0 - t) * theta).sin() / sin_theta,
163            (t * theta).sin() / sin_theta,
164        )
165    };
166
167    let rx = scale_a * ax + scale_b * bx;
168    let ry = scale_a * ay + scale_b * by;
169    let rz = scale_a * az + scale_b * bz;
170    let rw = scale_a * aw + scale_b * bw;
171
172    // Normalize
173    let len = (rx * rx + ry * ry + rz * rz + rw * rw).sqrt();
174    if len < 1e-10 {
175        return vec![0.0, 0.0, 0.0, 1.0];
176    }
177    vec![rx / len, ry / len, rz / len, rw / len]
178}

218    #[test]
219    fn slerp_shortest_path() {
220        // When quaternions represent the same rotation but with opposite signs,
221        // slerp should take the shortest path (negate and go the short way)
222        let a = vec![0.0, 0.0, 0.0, 1.0];
223        let neg_a = vec![0.0, 0.0, 0.0, -1.0]; // same rotation, opposite sign
224
225        let r = quat_slerp(&a, &neg_a, 0.5);
226        let len = (r[0] * r[0] + r[1] * r[1] + r[2] * r[2] + r[3] * r[3]).sqrt();
227        assert!((len - 1.0).abs() < 1e-5);
228        // Should be close to identity since they represent the same rotation
229        assert!((r[3].abs() - 1.0).abs() < 1e-4);
