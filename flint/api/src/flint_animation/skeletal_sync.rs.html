21/// `last_writer` value for a joint nothing animated this frame (rest pose).
22pub const WRITER_REST: u8 = u8::MAX;
23/// `last_writer` value for a joint written by the base clip / crossfade.
24pub const WRITER_BASE: u8 = 0;

26/// Per-joint bookkeeping of how the last pose was composed — the data the
27/// previewer colours its skeleton overlay with.
28#[derive(Debug, Clone, Default)]
29pub struct LayerContribution {
30    /// Per joint: [`WRITER_BASE`], `n` = layer `n-1` wrote it last, or
31    /// [`WRITER_REST`] if nothing keyed it.
32    pub last_writer: Vec<u8>,
33    /// `[layer][joint]` effective weight applied (0 when masked, un-keyed,
34    /// muted or inactive).
35    pub weights: Vec<Vec<f32>>,
36    /// `[layer][joint]` whether the layer's clip keys the joint at all.
37    pub keyed: Vec<Vec<bool>>,
38    /// `[layer][joint]` resolved mask (all `true` when no mask is set).
39    pub mask_bits: Vec<Vec<bool>>,
40    /// Mask strings the `mask_bits` were resolved from (cache key).
41    mask_keys: Vec<String>,
42}
43
44impl LayerContribution {
45    fn reset(&mut self, layer_count: usize, joint_count: usize) {
46        self.last_writer.clear();
47        self.last_writer.resize(joint_count, WRITER_REST);
48        self.weights.resize_with(layer_count, Vec::new);
49        self.keyed.resize_with(layer_count, Vec::new);
50        self.mask_bits.resize_with(layer_count, Vec::new);
51        self.mask_keys.resize_with(layer_count, String::new);
52        self.weights.truncate(layer_count);
53        self.keyed.truncate(layer_count);
54        self.mask_bits.truncate(layer_count);
55        self.mask_keys.truncate(layer_count);
56        for li in 0..layer_count {
57            self.weights[li].clear();
58            self.weights[li].resize(joint_count, 0.0);
59            self.keyed[li].clear();
60            self.keyed[li].resize(joint_count, false);
61        }
62    }
63
64    /// Weight layer `li` applied to `joint`, 0 if out of range.
65    pub fn weight(&self, li: usize, joint: usize) -> f32 {
66        self.weights
67            .get(li)
68            .and_then(|w| w.get(joint))
69            .copied()
70            .unwrap_or(0.0)
71    }
72
73    pub fn is_keyed(&self, li: usize, joint: usize) -> bool {
74        self.keyed
75            .get(li)
76            .and_then(|k| k.get(joint))
77            .copied()
78            .unwrap_or(false)
79    }
80
81    pub fn in_mask(&self, li: usize, joint: usize) -> bool {
82        self.mask_bits
83            .get(li)
84            .and_then(|m| m.get(joint))
85            .copied()
86            .unwrap_or(false)
87    }
88}

90/// Registry of skeletal clips and per-entity skeleton state
91#[derive(Default)]
92pub struct SkeletalSync {
93    clips: HashMap<String, SkeletalClip>,
94    skeletons: HashMap<EntityId, Skeleton>,
95    states: HashMap<EntityId, SkeletalPlaybackState>,
96    /// Entities whose crossfade finished this frame. The ECS `blend_target`
97    /// field must be cleared for them (see [`Self::write_back`]) — otherwise
98    /// the next `sync_from_world` sees a target that no longer matches the
99    /// (now-cleared) runtime target and re-arms the same crossfade forever.
100    completed_blends: Vec<EntityId>,
101    /// Per-joint composition bookkeeping from the last `advance_and_compute`.
102    contributions: HashMap<EntityId, LayerContribution>,
103    /// Runtime-only per-layer mute flags (previewer solo/mute). Kept out of
104    /// `ClipPlaybackState` so `sync_from_world` never clobbers them.
105    mutes: HashMap<EntityId, Vec<bool>>,
106    /// In-flight layer weight ramps, `[entity][layer]`. Like `mutes`, kept
107    /// outside the playback state: `sync_from_world` copies every layer's
108    /// weight from the ECS each frame, so the ramp is *driven* by the ECS
109    /// (`fade_target` + `fade_duration`) and the ramped weight is written
110    /// back each frame (see [`Self::write_back`]) — the same contract as
111    /// `blend_target`.
112    fades: HashMap<EntityId, Vec<Option<LayerFade>>>,
113    /// `(entity, layer)` fades that finished this frame; `write_back`
114    /// zeroes their ECS `fade_duration` so the next sync doesn't re-arm.
115    completed_fades: Vec<(EntityId, usize)>,
116}

118/// One layer weight ramp
119#[derive(Debug, Clone, Copy, PartialEq)]
120struct LayerFade {
121    from: f32,
122    to: f32,
123    duration: f32,
124    elapsed: f32,
125}
126
127impl LayerFade {
128    fn weight(&self) -> f32 {
129        let t = (self.elapsed / self.duration).clamp(0.0, 1.0);
130        self.from + (self.to - self.from) * t
131    }
132    fn done(&self) -> bool {
133        self.elapsed >= self.duration
134    }
135}

154    /// Parse the `animator.layers` array, falling back to the legacy
155    /// `layer_clip` / `layer_weight` pair as a single additive layer.
156    pub fn read_layers(animator: &toml::Value) -> Vec<AnimLayer> {
157        if let Some(arr) = animator.get("layers").and_then(|v| v.as_array()) {
158            if !arr.is_empty() {
159                return arr.iter().map(AnimLayer::from_toml).collect();
160            }
161        }
162        let legacy = animator
163            .get("layer_clip")
164            .and_then(|v| v.as_str())
165            .unwrap_or("");
166        if legacy.is_empty() {
167            return Vec::new();
168        }
169        let weight = animator
170            .get("layer_weight")
171            .and_then(toml_f32)
172            .unwrap_or(1.0);
173        vec![AnimLayer::new(legacy, weight)]
174    }
175
176    /// Copy the ECS layer list onto the runtime state. Weight/mode/mask/
177    /// speed are live dials; a clip change restarts that layer's clock.
178    ///
179    /// A layer with `fade_duration > 0` starts (or continues) a weight
180    /// ramp in `fades` instead of copying its weight: the ramp owns the
181    /// weight until it completes. Any other weight write cancels the ramp.
182    fn apply_ecs_layers(
183        state: &mut SkeletalPlaybackState,
184        animator: &toml::Value,
185        fades: &mut Vec<Option<LayerFade>>,
186    ) {
187        let ecs = Self::read_layers(animator);
188        state.layers.truncate(ecs.len());
189        fades.resize(ecs.len(), None);
190        for (i, incoming) in ecs.into_iter().enumerate() {
191            let fading = incoming.fade_duration > 0.0;
192            if let Some(cur) = state.layers.get_mut(i) {
193                if cur.clip != incoming.clip {
194                    cur.clip = incoming.clip;
195                    cur.time = 0.0;
196                }
197                cur.mode = incoming.mode;
198                cur.mask = incoming.mask;
199                cur.speed = incoming.speed;
200                cur.fade_target = incoming.fade_target;
201                cur.fade_duration = incoming.fade_duration;
202                if fading {
203                    let same = fades[i].is_some_and(|f| {
204                        f.to == incoming.fade_target && f.duration == incoming.fade_duration
205                    });
206                    if !same {
207                        fades[i] = Some(LayerFade {
208                            from: cur.weight,
209                            to: incoming.fade_target,
210                            duration: incoming.fade_duration,
211                            elapsed: 0.0,
212                        });
213                    }
214                } else {
215                    cur.weight = incoming.weight;
216                    fades[i] = None;
217                }
218            } else {
219                if fading {
220                    fades[i] = Some(LayerFade {
221                        from: incoming.weight,
222                        to: incoming.fade_target,
223                        duration: incoming.fade_duration,
224                        elapsed: 0.0,
225                    });
226                }
227                state.layers.push(incoming);
228            }
229        }
230        // Legacy mirror
231        state.layer_clip = state
232            .layers
233            .first()
234            .map(|l| l.clip.clone())
235            .unwrap_or_default();
236        state.layer_weight = state.layers.first().map(|l| l.weight).unwrap_or(1.0);
237        state.layer_time = state.layers.first().map(|l| l.time).unwrap_or(0.0);
238    }

277    /// Scan the world for entities with `animator` + `skeleton` components.
278    /// Creates playback states for newly discovered entities.
279    /// Updates blend_target/blend_duration for existing entities if changed in ECS.
280    pub fn sync_from_world(&mut self, world: &FlintWorld) {
281        for entity_id in world.entities_with_components(&[comp::ANIMATOR, comp::SKELETON]) {
282            let Some(components) = world.get_components(entity_id) else {
283                continue;
284            };
285
286            let Some(animator) = components.get(comp::ANIMATOR) else {
287                continue;
288            };
289            // Must also have a skeleton registered for this entity
290            if !self.skeletons.contains_key(&entity_id) {
291                continue;
292            }
293
294            // If already tracked, check for blend_target changes from ECS
295            if let Some(state) = self.states.get_mut(&entity_id) {
296                let ecs_blend_target = animator
297                    .get("blend_target")
298                    .and_then(|v| v.as_str())
299                    .unwrap_or("")
300                    .to_string();
301                let ecs_blend_duration = animator
302                    .get("blend_duration")
303                    .and_then(toml_f32)
304                    .unwrap_or(0.3);
305
306                // Start a new crossfade if ECS sets a new blend_target
307                if !ecs_blend_target.is_empty()
308                    && ecs_blend_target != state.blend_target
309                    && self.clips.contains_key(&ecs_blend_target)
310                {
311                    state.blend_target = ecs_blend_target;
312                    state.blend_duration = ecs_blend_duration;
313                    state.blend_elapsed = 0.0;
314                }
315
316                // Speed and loop are live dials too (previewer Speed
317                // field, `set_anim_speed` from scripts).
318                state.speed = animator.get("speed").and_then(toml_f64).unwrap_or(1.0);
319                state.looping = animator
320                    .get("loop")
321                    .and_then(|v| v.as_bool())
322                    .unwrap_or(true);
323
324                // Layers follow the ECS every frame (weights are live
325                // dials; changing a clip restarts that layer's clock).
326                let fades = self.fades.entry(entity_id).or_default();
327                Self::apply_ecs_layers(state, animator, fades);
328                continue;
329            }
330
331            let clip_name = animator
332                .get("clip")
333                .and_then(|v| v.as_str())
334                .unwrap_or("")
335                .to_string();
336
337            if clip_name.is_empty() || !self.clips.contains_key(&clip_name) {
338                continue;
339            }
340
341            let speed = animator.get("speed").and_then(toml_f64).unwrap_or(1.0);
342
343            let looping = animator
344                .get("loop")
345                .and_then(|v| v.as_bool())
346                .unwrap_or(true);
347
348            let autoplay = animator
349                .get("autoplay")
350                .and_then(|v| v.as_bool())
351                .unwrap_or(false);
352
353            let playing = animator
354                .get("playing")
355                .and_then(|v| v.as_bool())
356                .unwrap_or(false)
357                || autoplay;
358
359            // A resolved clip that is not playing holds its first frame
360            // forever. Both flags default to false, and TOML makes it easy to
361            // strand them on a neighbouring component table (a key after the
362            // next [table] header belongs to THAT table) — which reads as
363            // "the animation is stuck on one frame" with nothing else to
364            // explain it. Say so rather than render a statue in silence.
365            if !playing {
366                println!(
367                    "WARNING: entity {:?} has animator clip '{}' but playing=false and \
368                     autoplay=false — it will hold its first frame. Check that `playing`/\
369                     `autoplay` are under [entities.<name>.animator].",
370                    entity_id, clip_name
371                );
372            }
373
374            let mut state = SkeletalPlaybackState::new(clip_name, speed, looping, playing);
375
376            // Read initial blend fields
377            state.blend_target = animator
378                .get("blend_target")
379                .and_then(|v| v.as_str())
380                .unwrap_or("")
381                .to_string();
382            state.blend_duration = animator
383                .get("blend_duration")
384                .and_then(toml_f32)
385                .unwrap_or(0.3);
386            let fades = self.fades.entry(entity_id).or_default();
387            Self::apply_ecs_layers(&mut state, animator, fades);
388
389            self.states.insert(entity_id, state);
390        }
391    }

393    /// Advance all skeletal playbacks and compute bone matrices.
394    /// Handles crossfade blending when `blend_target` is set.
395    pub fn advance_and_compute(&mut self, dt: f64) {
396        let entity_ids: Vec<EntityId> = self.states.keys().copied().collect();
397
398        for entity_id in entity_ids {
399            let state = self.states.get_mut(&entity_id).unwrap();
400            let Some(clip) = self.clips.get(&state.clip_name) else {
401                continue;
402            };
403            let Some(skeleton) = self.skeletons.get_mut(&entity_id) else {
404                continue;
405            };
406
407            // Advance time
408            if state.playing {
409                state.time += dt * state.speed;
410
411                if state.looping {
412                    if clip.duration > 0.0 {
413                        if state.time >= clip.duration {
414                            state.time %= clip.duration;
415                        } else if state.time < 0.0 {
416                            state.time = clip.duration - (-state.time % clip.duration);
417                        }
418                    }
419                } else if state.time >= clip.duration {
420                    state.time = clip.duration;
421                    state.playing = false;
422                } else if state.time < 0.0 {
423                    state.time = 0.0;
424                    state.playing = false;
425                }
426            }
427
428            // Start from the rest pose every frame. Clips are sparse (an
429            // exporter drops tracks that hold the rest value), so a joint
430            // the current clip doesn't key must read as "rest" — not as
431            // whatever the *previous* clip left in the persistent buffer.
432            // Otherwise walk → idle leaves the legs frozen mid-stride.
433            if skeleton.rest_poses.len() == skeleton.local_poses.len() {
434                skeleton.local_poses.clone_from(&skeleton.rest_poses);
435            }
436
437            // Sample current clip into local_poses
438            Self::sample_clip_into_poses(clip, state.time, &mut skeleton.local_poses);
439
440            // Handle crossfade blending
441            let is_blending = !state.blend_target.is_empty() && state.blend_duration > 0.0;
442
443            if is_blending {
444                let target_clip_name = state.blend_target.clone();
445                if let Some(target_clip) = self.clips.get(&target_clip_name) {
446                    state.blend_elapsed += dt as f32;
447                    let blend_weight = (state.blend_elapsed / state.blend_duration).min(1.0);
448
449                    // Sample the target clip over the REST pose (same sparse-
450                    // clip rule as above) so joints the target doesn't key
451                    // blend toward rest instead of sticking at the old clip.
452                    let joint_count = skeleton.joint_count();
453                    let mut target_poses: Vec<JointPose> =
454                        if skeleton.rest_poses.len() == joint_count {
455                            skeleton.rest_poses.clone()
456                        } else {
457                            skeleton.local_poses.clone()
458                        };
459                    // Compute target clip's time (starts from 0 for the blend-in clip)
460                    let target_time = state.blend_elapsed as f64 * state.speed;
461                    Self::sample_clip_into_poses(target_clip, target_time, &mut target_poses);
462
463                    // Blend current poses (already in skeleton.local_poses) with target
464                    let current_poses: Vec<JointPose> = skeleton.local_poses.clone();
465                    blend_poses(
466                        &current_poses,
467                        &target_poses,
468                        blend_weight,
469                        &mut skeleton.local_poses,
470                    );
471
472                    // Check if blend is complete
473                    if blend_weight >= 1.0 {
474                        // Transition to target clip
475                        state.clip_name = target_clip_name;
476                        state.time = target_time;
477                        state.blend_target.clear();
478                        state.blend_elapsed = 0.0;
479                        self.completed_blends.push(entity_id);
480                    }
481                } else {
482                    // Target clip not found, clear blend
483                    state.blend_target.clear();
484                    state.blend_elapsed = 0.0;
485                    self.completed_blends.push(entity_id);
486                }
487            }
488
489            // ── Animation layers ────────────────────────────────────
490            // Each loops on its own clock and is composed AFTER base +
491            // blend (so layers survive crossfades), in array order.
492            //   Additive: keyed joints contribute delta-from-REST × weight.
493            //   Override: keyed joints blend toward the sampled pose by
494            //             weight.
495            // Both honour an optional subtree mask. Un-keyed joints are
496            // untouched (composing identity would corrupt joints whose
497            // rest rotation is non-identity). The pre-layer pose is
498            // RESTORED after matrix computation: local_poses persists
499            // across frames and base clips only overwrite the joints they
500            // key, so leaving the deltas in would compound them every
501            // frame (feet were windmilling at 120°+).
502            let joint_count = skeleton.joint_count();
503            let pre_layer: Vec<JointPose> = skeleton.local_poses.clone();
504
505            let contrib = self.contributions.entry(entity_id).or_default();
506            contrib.reset(state.layers.len(), joint_count);
507            for track in &clip.joint_tracks {
508                if let Some(w) = contrib.last_writer.get_mut(track.joint_index) {
509                    *w = WRITER_BASE;
510                }
511            }
512            if let Some(target) = self.clips.get(&state.blend_target) {
513                for track in &target.joint_tracks {
514                    if let Some(w) = contrib.last_writer.get_mut(track.joint_index) {
515                        *w = WRITER_BASE;
516                    }
517                }
518            }
519            let mutes = self.mutes.get(&entity_id);
520            let mut fades = self.fades.get_mut(&entity_id);
521
522            for (li, layer) in state.layers.iter_mut().enumerate() {
523                // Advance an in-flight weight ramp. Done before the mute /
524                // zero-weight early-outs so a fade-in from 0 actually starts.
525                if let Some(fade) = fades.as_deref_mut().and_then(|f| f.get_mut(li)) {
526                    if let Some(f) = fade {
527                        f.elapsed += dt as f32;
528                        layer.weight = f.weight();
529                        if f.done() {
530                            layer.weight = f.to;
531                            layer.fade_duration = 0.0;
532                            *fade = None;
533                            self.completed_fades.push((entity_id, li));
534                        }
535                    }
536                }
537                // Resolve mask (cached by mask string) and keyed set even
538                // for inactive layers so the previewer can show them.
539                if contrib.mask_keys[li] != layer.mask || contrib.mask_bits[li].len() != joint_count
540                {
541                    contrib.mask_bits[li] = match layer.mask.as_str() {
542                        "" => vec![true; joint_count],
543                        root => skeleton
544                            .subtree_mask(root)
545                            .unwrap_or_else(|| vec![false; joint_count]),
546                    };
547                    contrib.mask_keys[li] = layer.mask.clone();
548                }
549                let Some(layer_clip) = self.clips.get(&layer.clip) else {
550                    continue;
551                };
552                for track in &layer_clip.joint_tracks {
553                    if let Some(k) = contrib.keyed[li].get_mut(track.joint_index) {
554                        *k = true;
555                    }
556                }
557                let muted = mutes.and_then(|m| m.get(li)).copied().unwrap_or(false);
558                if muted || layer.weight <= 0.001 {
559                    continue;
560                }
561
562                layer.time += dt * state.speed * layer.speed;
563                if layer_clip.duration > 0.0 {
564                    layer.time %= layer_clip.duration;
565                    if layer.time < 0.0 {
566                        layer.time += layer_clip.duration;
567                    }
568                }
569                let w = layer.weight.clamp(0.0, 1.0);
570                let mask = &contrib.mask_bits[li];
571
572                match layer.mode {
573                    LayerMode::Additive => {
574                        for track in &layer_clip.joint_tracks {
575                            let idx = track.joint_index;
576                            if idx >= joint_count || !mask[idx] {
577                                continue;
578                            }
579                            let value = sample_joint_track(track, layer.time);
580                            let pose = &mut skeleton.local_poses[idx];
581                            let rest = &skeleton.rest_poses[idx];
582                            match track.property {
583                                JointProperty::Rotation => {
584                                    if value.len() < 4 {
585                                        continue;
586                                    }
587                                    let sampled = [value[0], value[1], value[2], value[3]];
588                                    // delta = rest⁻¹ * sampled, faded toward
589                                    // identity by weight, applied on the base.
590                                    let delta = quat_mul(&quat_conjugate(&rest.rotation), &sampled);
591                                    let faded = quat_slerp(&[0.0, 0.0, 0.0, 1.0], &delta, w);
592                                    pose.rotation =
593                                        quat_normalize(&quat_mul(&pose.rotation, &faded));
594                                }
595                                JointProperty::Translation => {
596                                    if value.len() < 3 {
597                                        continue;
598                                    }
599                                    for (c, v) in value.iter().take(3).enumerate() {
600                                        pose.translation[c] += (v - rest.translation[c]) * w;
601                                    }
602                                }
603                                JointProperty::Scale => {
604                                    if value.len() < 3 {
605                                        continue;
606                                    }
607                                    for (c, v) in value.iter().take(3).enumerate() {
608                                        let denom =
609                                            rest.scale[c].abs().max(1e-6).copysign(rest.scale[c]);
610                                        let ratio = v / denom;
611                                        pose.scale[c] *= 1.0 + (ratio - 1.0) * w;
612                                    }
613                                }
614                            }
615                            contrib.weights[li][idx] = w;
616                            contrib.last_writer[idx] = (li + 1) as u8;
617                        }
618                    }
619                    LayerMode::Override => {
620                        // Sample onto a copy of the current pose so
621                        // un-keyed properties of a keyed joint stay put.
622                        let mut sampled = skeleton.local_poses.clone();
623                        Self::sample_clip_into_poses(layer_clip, layer.time, &mut sampled);
624                        for idx in 0..joint_count {
625                            if !mask[idx] || !contrib.keyed[li][idx] {
626                                continue;
627                            }
628                            skeleton.local_poses[idx] =
629                                blend_joint(&skeleton.local_poses[idx], &sampled[idx], w);
630                            contrib.weights[li][idx] = w;
631                            contrib.last_writer[idx] = (li + 1) as u8;
632                        }
633                    }
634                }
635            }
636
637            // Legacy mirror of layer 0's clock
638            state.layer_time = state.layers.first().map(|l| l.time).unwrap_or(0.0);
639
640            // Compute final bone matrices
641            skeleton.compute_bone_matrices();
642
643            // Un-apply the layers from the persistent pose buffer (see
644            // the accumulation note above).
645            skeleton.local_poses = pre_layer;
646        }
647    }

688    /// Retire finished crossfades in the ECS.
689    ///
690    /// `blend_to` sets the `blend_target` field and nothing else ever cleared
691    /// it, while `advance_and_compute` cleared only the runtime mirror. The
692    /// next `sync_from_world` therefore saw a non-empty ECS target that no
693    /// longer matched the empty runtime one and started the SAME crossfade
694    /// again — a self-sustaining loop with a period of `blend_duration`, so
695    /// every clip permanently replayed only its first `blend_duration`
696    /// seconds (0.3 s default ≈ 3-4 restarts/second). Clearing the field
697    /// here closes the loop and keeps `blend_to(e, same_clip, t)` meaningful
698    /// as an explicit restart.
699    pub fn write_back(&mut self, world: &mut FlintWorld) {
700        for entity_id in self.completed_blends.drain(..) {
701            let Some(components) = world.get_components_mut(entity_id) else {
702                continue;
703            };
704            components.set_field(
705                comp::ANIMATOR,
706                "blend_target",
707                toml::Value::String(String::new()),
708            );
709        }
710
711        // Layer fades: publish the ramped weight every frame (so the next
712        // sync copies the same value and UI sliders visibly move), and
713        // retire finished ramps by zeroing `fade_duration`.
714        let completed = std::mem::take(&mut self.completed_fades);
715        for (entity_id, fades) in &self.fades {
716            let active: Vec<usize> = fades
717                .iter()
718                .enumerate()
719                .filter_map(|(i, f)| f.is_some().then_some(i))
720                .collect();
721            let done: Vec<usize> = completed
722                .iter()
723                .filter(|(e, _)| e == entity_id)
724                .map(|(_, i)| *i)
725                .collect();
726            if active.is_empty() && done.is_empty() {
727                continue;
728            }
729            let Some(state) = self.states.get(entity_id) else {
730                continue;
731            };
732            let Some(components) = world.get_components_mut(*entity_id) else {
733                continue;
734            };
735            let Some(layers) = components
736                .get_mut(comp::ANIMATOR)
737                .and_then(|a| a.get_mut("layers"))
738                .and_then(|l| l.as_array_mut())
739            else {
740                continue;
741            };
742            for li in active.iter().chain(done.iter()) {
743                let (Some(layer), Some(table)) = (
744                    state.layers.get(*li),
745                    layers.get_mut(*li).and_then(|v| v.as_table_mut()),
746                ) else {
747                    continue;
748                };
749                table.insert("weight".into(), toml::Value::Float(layer.weight as f64));
750                if done.contains(li) {
751                    table.insert("fade_duration".into(), toml::Value::Float(0.0));
752                }
753            }
754        }
755    }

765    /// Sample a clip's joint tracks into a pose array
766    fn sample_clip_into_poses(clip: &SkeletalClip, time: f64, poses: &mut [JointPose]) {
767        for track in &clip.joint_tracks {
768            let value = sample_joint_track(track, time);
769            let idx = track.joint_index;
770            if idx >= poses.len() {
771                continue;
772            }
773
774            match track.property {
775                JointProperty::Translation => {
776                    if value.len() >= 3 {
777                        poses[idx].translation = [value[0], value[1], value[2]];
778                    }
779                }
780                JointProperty::Rotation => {
781                    if value.len() >= 4 {
782                        poses[idx].rotation = [value[0], value[1], value[2], value[3]];
783                    }
784                }
785                JointProperty::Scale => {
786                    if value.len() >= 3 {
787                        poses[idx].scale = [value[0], value[1], value[2]];
788                    }
789                }
790            }
791        }
792    }

1222    fn step(sync: &mut SkeletalSync, world: &mut FlintWorld, frames: usize) {
1223        for _ in 0..frames {
1224            sync.sync_from_world(world);
1225            sync.advance_and_compute(1.0 / 60.0);
1226            sync.write_back(world);
1227        }
1228    }

1310            .push(hold_clip("_", 1, 1.0).joint_tracks.remove(0));
1311        let mut over = hold_clip("over", 0, 10.0);
1312        over.joint_tracks
1313            .push(hold_clip("_", 1, 10.0).joint_tracks.remove(0));
1314        sync.add_clip(base);
1315        sync.add_clip(over);
1316        let (mut world, eid) = world_with_animator("base", "", 0.3);
1317        sync.add_skeleton(eid, two_joint_skeleton());
1318
1319        set_layers(&mut world, eid, &[("over", 1.0, "override", "child")]);
1320        step(&mut sync, &mut world, 1);
1321        let root = sync.skeletons[&eid].global_matrices[0][3][0];
1322        let child_local = sync.skeletons[&eid].global_matrices[1][3][0] - root;
1323        assert!((root - 1.0).abs() < 1e-5, "root outside mask stays base");
1324        assert!((child_local - 10.0).abs() < 1e-5, "child fully overridden");
1325
1326        set_layers(&mut world, eid, &[("over", 0.5, "override", "child")]);
1327        step(&mut sync, &mut world, 1);
1328        let root = sync.skeletons[&eid].global_matrices[0][3][0];
1329        let child_local = sync.skeletons[&eid].global_matrices[1][3][0] - root;
1330        assert!((child_local - 5.5).abs() < 1e-5, "midpoint of 1 and 10");
1331
1332        let c = sync.layer_contribution(&eid).unwrap();
1333        assert_eq!(c.last_writer, vec![WRITER_BASE, 1]);
1334        assert!(c.in_mask(0, 1) && !c.in_mask(0, 0));
1335        assert!(c.is_keyed(0, 0) && c.is_keyed(0, 1));
1336        assert!((c.weight(0, 1) - 0.5).abs() < 1e-6 && c.weight(0, 0) == 0.0);
