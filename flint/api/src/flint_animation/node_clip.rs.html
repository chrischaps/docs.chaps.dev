1//! Node-level animation clip — per-node keyframe tracks for transform animation
2//!
3//! Mirrors the skeletal clip types but targets named scene-graph nodes instead of
4//! skeleton joints. Used for glTF node animations where entire objects move/rotate
5//! via their transforms (as opposed to vertex skinning).
6
7use crate::clip::Interpolation;
8use crate::skeletal_clip::{JointKeyframe, JointProperty};
9
10/// A single track targeting one node's property (translation, rotation, or scale)
11#[derive(Debug, Clone)]
12pub struct NodeTrack {
13    pub node_name: String,
14    pub property: JointProperty,
15    pub interpolation: Interpolation,
16    pub keyframes: Vec<JointKeyframe>,
17}
18
19/// A complete node-level animation clip with per-node tracks
20#[derive(Debug, Clone)]
21pub struct NodeClip {
22    pub name: String,
23    pub duration: f64,
24    pub node_tracks: Vec<NodeTrack>,
25}
26
27impl NodeClip {
28    /// Convert from imported glTF node animation data
29    pub fn from_imported(imported: &flint_import::ImportedNodeClip) -> Self {
30        let node_tracks = imported
31            .channels
32            .iter()
33            .map(|ch| {
34                let property = match ch.property {
35                    flint_import::JointProperty::Translation => JointProperty::Translation,
36                    flint_import::JointProperty::Rotation => JointProperty::Rotation,
37                    flint_import::JointProperty::Scale => JointProperty::Scale,
38                };
39
40                let interpolation = match ch.interpolation.as_str() {
41                    "STEP" => Interpolation::Step,
42                    "CUBICSPLINE" => Interpolation::CubicSpline,
43                    _ => Interpolation::Linear,
44                };
45
46                let keyframes = ch
47                    .keyframes
48                    .iter()
49                    .map(|kf| JointKeyframe {
50                        time: kf.time as f64,
51                        value: kf.value.clone(),
52                    })
53                    .collect();
54
55                NodeTrack {
56                    node_name: ch.node_name.clone(),
57                    property,
58                    interpolation,
59                    keyframes,
60                }
61            })
62            .collect();
63
64        Self {
65            name: imported.name.clone(),
66            duration: imported.duration as f64,
67            node_tracks,
68        }
69    }
70}
