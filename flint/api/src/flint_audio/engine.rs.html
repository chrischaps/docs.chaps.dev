1//! Audio engine wrapping Kira's AudioManager
2//!
3//! Handles sound loading, listener management, and spatial track creation.
4//! Degrades gracefully when no audio device is available.

17/// Mixer bus a sound routes through. Both buses are children of the main
18/// track, so the master low-pass filter still applies to everything.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
20pub enum Bus {
21    #[default]
22    Sfx,
23    Music,
24}
25
26impl Bus {
27    /// Parse an `audio_source.bus` field value ("music"/"sfx"); unknown → Sfx
28    pub fn from_name(s: &str) -> Self {
29        match s {
30            "music" => Bus::Music,
31            _ => Bus::Sfx,
32        }
33    }
34}

36/// Wraps Kira's AudioManager with sound caching and listener management
37pub struct AudioEngine {
38    manager: Option<AudioManager<DefaultBackend>>,
39    listener: Option<kira::listener::ListenerHandle>,
40    sound_cache: HashMap<String, StaticSoundData>,
41    master_volume: f64,
42    filter_handle: Option<FilterHandle>,
43    /// Music mixer bus (sub-track of main)
44    music_bus: Option<TrackHandle>,
45    /// SFX mixer bus (sub-track of main); all spatial tracks hang off this
46    /// or the music bus rather than the main track directly
47    sfx_bus: Option<TrackHandle>,
48    /// Keeps spatial track + sound handles alive until one-shot sounds finish
49    oneshot_tracks: Vec<(SpatialTrackHandle, StaticSoundHandle)>,
50}
51
52impl Default for AudioEngine {
53    fn default() -> Self {
54        Self::new()
55    }
56}
57
58impl AudioEngine {
59    pub fn new() -> Self {
60        // Set up main track with a low-pass filter (fully open at 20 kHz by default)
61        let mut settings = kira::AudioManagerSettings::<DefaultBackend>::default();
62        let filter_handle = settings.main_track_builder.add_effect(
63            FilterBuilder::new()
64                .mode(FilterMode::LowPass)
65                .cutoff(20000.0),
66        );
67
68        // Try to create the audio manager; gracefully fail if no device
69        let mut manager = AudioManager::<DefaultBackend>::new(settings)
70            .map_err(|e| tracing::warn!("No audio device available ({e}), running silent"))
71            .ok();
72
73        // Create the music and SFX buses as children of the main track
74        let (music_bus, sfx_bus) = match &mut manager {
75            Some(m) => (
76                m.add_sub_track(TrackBuilder::new())
77                    .map_err(|e| tracing::warn!("Failed to create music bus: {e}"))
78                    .ok(),
79                m.add_sub_track(TrackBuilder::new())
80                    .map_err(|e| tracing::warn!("Failed to create sfx bus: {e}"))
81                    .ok(),
82            ),
83            None => (None, None),
84        };
85
86        let has_audio = manager.is_some();
87        Self {
88            manager,
89            listener: None,
90            sound_cache: HashMap::new(),
91            master_volume: 1.0,
92            filter_handle: if has_audio { Some(filter_handle) } else { None },
93            music_bus,
94            sfx_bus,
95            oneshot_tracks: Vec::new(),
96        }
97    }

99    /// Set a mixer bus volume (linear amplitude, 0.0 = mute, 1.0 = unity).
100    /// Applies multiplicatively on top of per-sound volumes, so script-driven
101    /// fades keep working underneath.
102    pub fn set_bus_volume(&mut self, bus: Bus, volume: f64) {
103        let handle = match bus {
104            Bus::Music => &mut self.music_bus,
105            Bus::Sfx => &mut self.sfx_bus,
106        };
107        if let Some(handle) = handle {
108            let tween = Tween {
109                duration: Duration::from_millis(16),
110                ..Default::default()
111            };
112            handle.set_volume(amplitude_to_db(volume), tween);
113        }
114    }

128    /// Load a sound file into the cache
129    pub fn load_sound(&mut self, name: &str, path: &Path) -> Result<()> {
130        if self.sound_cache.contains_key(name) {
131            return Ok(());
132        }
133
134        let sound_data = StaticSoundData::from_file(path).map_err(|e| {
135            flint_core::FlintError::AudioError(format!(
136                "Failed to load '{}': {}",
137                path.display(),
138                e
139            ))
140        })?;
141
142        self.sound_cache.insert(name.to_string(), sound_data);
143        Ok(())
144    }

146    /// Create the listener at the given position (typically the player/camera)
147    pub fn create_listener(&mut self, position: Vec3) -> Result<()> {
148        let manager = match &mut self.manager {
149            Some(m) => m,
150            None => return Ok(()),
151        };
152
153        let pos = to_glam_vec3(position);
154        let orientation = glam::Quat::IDENTITY;
155
156        let handle = manager.add_listener(pos, orientation).map_err(|e| {
157            flint_core::FlintError::AudioError(format!("Failed to create listener: {e}"))
158        })?;
159
160        self.listener = Some(handle);
161        Ok(())
162    }
163
164    /// Update listener position and orientation from yaw/pitch angles
165    pub fn update_listener(&mut self, position: Vec3, yaw: f32, pitch: f32) {
166        let Some(listener) = &mut self.listener else {
167            return;
168        };
169
170        let pos = to_glam_vec3(position);
171        let orientation = glam::Quat::from_euler(glam::EulerRot::YXZ, -yaw, -pitch, 0.0);
172
173        let tween = Tween {
174            duration: Duration::ZERO,
175            ..Default::default()
176        };
177
178        listener.set_position(pos, tween);
179        listener.set_orientation(orientation, tween);
180    }

182    /// Create a spatial track for an entity's audio source on the given bus
183    pub fn create_spatial_track(
184        &mut self,
185        position: Vec3,
186        min_distance: f32,
187        max_distance: f32,
188        bus: Bus,
189    ) -> Result<SpatialTrackHandle> {
190        if self.manager.is_none() {
191            return Err(flint_core::FlintError::AudioError("No audio device".into()));
192        }
193
194        let listener_id = match &self.listener {
195            Some(l) => l.id(),
196            None => {
197                return Err(flint_core::FlintError::AudioError(
198                    "No listener created".into(),
199                ));
200            }
201        };
202
203        let pos = to_glam_vec3(position);
204
205        let builder = SpatialTrackBuilder::new()
206            .distances(SpatialTrackDistances {
207                min_distance,
208                max_distance,
209            })
210            .attenuation_function(Some(Easing::OutPowf(2.0)));
211
212        let bus_handle = match bus {
213            Bus::Music => &mut self.music_bus,
214            Bus::Sfx => &mut self.sfx_bus,
215        };
216        let bus_handle = bus_handle.as_mut().ok_or_else(|| {
217            flint_core::FlintError::AudioError(format!("Bus {bus:?} unavailable"))
218        })?;
219
220        let handle = bus_handle
221            .add_spatial_sub_track(listener_id, pos, builder)
222            .map_err(|e| {
223                flint_core::FlintError::AudioError(format!("Failed to create spatial track: {e}"))
224            })?;
225
226        Ok(handle)
227    }

229    /// Play a cached sound on a spatial track (looping or one-shot)
230    pub fn play_on_spatial_track(
231        &mut self,
232        sound_name: &str,
233        track: &mut SpatialTrackHandle,
234        volume: f64,
235        pitch: f64,
236        looping: bool,
237    ) -> Result<StaticSoundHandle> {
238        let sound_data = self
239            .sound_cache
240            .get(sound_name)
241            .ok_or_else(|| {
242                flint_core::FlintError::AudioError(format!("Sound not cached: {sound_name}"))
243            })?
244            .clone();
245
246        let mut data = sound_data
247            .volume(amplitude_to_db(volume * self.master_volume))
248            .playback_rate(kira::PlaybackRate(pitch));
249
250        if looping {
251            data = data.loop_region(..);
252        }
253
254        let handle = track.play(data).map_err(|e| {
255            flint_core::FlintError::AudioError(format!("Failed to play '{sound_name}': {e}"))
256        })?;
257
258        Ok(handle)
259    }

261    /// Play a cached sound on a mixer bus (non-spatial, e.g. ambient/music)
262    pub fn play_non_spatial(
263        &mut self,
264        sound_name: &str,
265        volume: f64,
266        pitch: f64,
267        looping: bool,
268        bus: Bus,
269    ) -> Result<StaticSoundHandle> {
270        if self.manager.is_none() {
271            return Err(flint_core::FlintError::AudioError("No audio device".into()));
272        }
273
274        let sound_data = self
275            .sound_cache
276            .get(sound_name)
277            .ok_or_else(|| {
278                flint_core::FlintError::AudioError(format!("Sound not cached: {sound_name}"))
279            })?
280            .clone();
281
282        let mut data = sound_data
283            .volume(amplitude_to_db(volume * self.master_volume))
284            .playback_rate(kira::PlaybackRate(pitch));
285
286        if looping {
287            data = data.loop_region(..);
288        }
289
290        let bus_handle = match bus {
291            Bus::Music => &mut self.music_bus,
292            Bus::Sfx => &mut self.sfx_bus,
293        };
294        let bus_handle = bus_handle.as_mut().ok_or_else(|| {
295            flint_core::FlintError::AudioError(format!("Bus {bus:?} unavailable"))
296        })?;
297
298        let handle = bus_handle.play(data).map_err(|e| {
299            flint_core::FlintError::AudioError(format!("Failed to play '{sound_name}': {e}"))
300        })?;
301
302        Ok(handle)
303    }

305    /// Play a one-shot sound at a 3D position (creates a temporary spatial track).
306    /// The track and sound handles are kept alive until the sound finishes playing.
307    pub fn play_at_position(
308        &mut self,
309        sound_name: &str,
310        position: Vec3,
311        volume: f64,
312        pitch: f64,
313    ) -> Result<()> {
314        let mut track = self.create_spatial_track(position, 1.0, 25.0, Bus::Sfx)?;
315        let handle = self.play_on_spatial_track(sound_name, &mut track, volume, pitch, false)?;
316        self.oneshot_tracks.push((track, handle));
317        Ok(())
318    }
319
320    /// Remove finished one-shot spatial sounds. Call once per frame.
321    pub fn cleanup_finished_oneshots(&mut self) {
322        self.oneshot_tracks
323            .retain(|(_track, sound)| sound.state() != kira::sound::PlaybackState::Stopped);
324    }
325
326    /// Drop all one-shot spatial handles (for scene transitions).
327    pub fn clear_oneshots(&mut self) {
328        self.oneshot_tracks.clear();
329    }

336    /// Set the low-pass filter cutoff frequency (Hz). 20000 = fully open, lower = more muffled.
337    pub fn set_filter_cutoff(&mut self, hz: f32) {
338        if let Some(handle) = &mut self.filter_handle {
339            let tween = Tween {
340                duration: Duration::from_millis(16),
341                ..Default::default()
342            };
343            handle.set_cutoff(hz as f64, tween);
344        }
345    }

348/// Convert Flint Vec3 to glam Vec3
349fn to_glam_vec3(v: Vec3) -> glam::Vec3 {
350    glam::Vec3::new(v.x, v.y, v.z)
351}

353/// Convert linear amplitude (0.0–2.0) to decibels
354pub fn amplitude_to_db(amplitude: f64) -> kira::Decibels {
355    if amplitude <= 0.0 {
356        kira::Decibels(-60.0) // silence
357    } else {
358        kira::Decibels((20.0 * (amplitude as f32).log10()).max(-60.0))
359    }
