1//! Audio engine wrapping Kira's AudioManager
2//!
3//! Handles sound loading, listener management, and spatial track creation.
4//! Degrades gracefully when no audio device is available.
5//!
6//! The mix runs in one of two places (#136). On the desktop, and in the
7//! browser's fallback, kira runs here on its own backend ([`Mixer::Local`]).
8//! In the browser it normally runs in an `AudioWorklet`, off the main
9//! thread: every call below then becomes a command to that mixer, and each
10//! sound crosses to it once, as samples ([`Mixer::Remote`],
11//! `crate::remote`). Handles ([`SoundHandle`], [`SpatialTrack`]) hide which.

28/// `StaticSoundData::from_file`, read through the vfs: the bytes come from
29/// the active file source (the disk, or the web's bundle) and kira decodes
30/// them from memory, which is all its `from_file` does after opening the
31/// file. kira leaves `from_file` out of wasm32 builds anyway.
32fn static_sound_from_file(
33    path: impl AsRef<Path>,
34) -> std::result::Result<StaticSoundData, kira::sound::FromFileError> {
35    let bytes = vfs::read(path)?;
36    StaticSoundData::from_cursor(std::io::Cursor::new(bytes))
37}

51/// How a placed sound fades between its distances when the caller names no
52/// curve. With `r` the listener's fraction of the way from `min` to `max`,
53/// the sound is `-60·r^rolloff` dB: at 2 it holds up most of the way and
54/// drops near `max`; below 1 it falls fast past `min` and tails off slowly,
55/// near an inverse-distance law (0.6 between 32 and 1250 m is within 1.5 dB
56/// of `(32/d)^1.2` out to 300 m).
57pub const DEFAULT_ROLLOFF: f32 = 2.0;

172/// kira, somewhere else (the browser's audio thread), reached through a
173/// [`Link`].
174struct Remote {
175    link: Link,
176    /// Every sound sent, by name, and how long each plays, seconds: the
177    /// mixer says nothing back, so a one-shot's end is worked out here.
178    sounds: HashMap<String, u32>,
179    lengths: HashMap<u32, f64>,
180    listener: bool,
181    /// The worklet behind the link, while it starts and once it runs.
182    #[cfg(target_arch = "wasm32")]
183    stream: Option<crate::worklet::WorkletStream>,
184}
185
186impl Remote {
187    /// Send a sound, under the id it already has if it was sent before.
188    fn send_sound(&mut self, name: &str, data: &StaticSoundData) {
189        let id = match self.sounds.get(name) {
190            Some(&id) => id,
191            None => {
192                let id = self.link.id();
193                self.sounds.insert(name.to_string(), id);
194                id
195            }
196        };
197        self.lengths.insert(
198            id,
199            data.frames.len() as f64 / data.sample_rate.max(1) as f64,
200        );
201        self.link
202            .load(SoundLoad::new(id, data.sample_rate, &data.frames));
203    }
204
205    fn sound(&self, name: &str) -> Result<u32> {
206        self.sounds
207            .get(name)
208            .copied()
209            .ok_or_else(|| flint_core::FlintError::AudioError(format!("Sound not cached: {name}")))
210    }
211
212    /// Whether sounds still need a copy here: until the worklet is known to
213    /// run, the main-thread fallback may need them.
214    fn keeps_cache(&self) -> bool {
215        #[cfg(target_arch = "wasm32")]
216        if let Some(s) = &self.stream {
217            return !s.running();
218        }
219        false
220    }

229/// A placed one-shot that is playing, and the sound it plays.
230struct OneShot {
231    sound: String,
232    handle: SoundHandle,
233    /// Its own kira spatial track (`?mix=old`, mixed here); without one
234    /// it's a panned voice on its bus that [`AudioEngine::pan_oneshots`]
235    /// steers
236    track: Option<SpatialTrack>,
237    position: Vec3,
238    falloff: Falloff,
239    volume: f64,
240    /// When a remote one has played out: the worklet says nothing back
241    ends_at: Option<Instant>,
242    /// Stopped to make room. A local handle says `Playing` until the audio
243    /// thread reads the stop, which may be a buffer away.
244    stolen: bool,
245}
246
247impl OneShot {
248    /// Heard, or about to be: not stolen, stopping or stopped.
249    fn is_sounding(&self, now: Instant) -> bool {
250        use kira::sound::PlaybackState::*;
251        !self.stolen
252            && match self.handle.local_state() {
253                Some(state) => !matches!(state, Stopping | Stopped),
254                None => self.ends_at.is_none_or(|t| now < t),
255            }
256    }
257
258    fn is_finished(&self, now: Instant) -> bool {
259        match self.handle.local_state() {
260            Some(state) => state == kira::sound::PlaybackState::Stopped,
261            None => self.ends_at.is_some_and(|t| now >= t),
262        }
263    }
264}

366    /// Open the audio device, if it isn't open, and say whether there is
367    /// sound. In the browser, call it from the click (or key) handler that
368    /// begins the game: it opens the worklet's `AudioContext` inside the
369    /// gesture, and the worklet's mixer starts a moment later, with the
370    /// calls made meanwhile waiting for it. Failing leaves the engine
371    /// silent, as before, and a later call tries again.
372    pub fn start(&mut self) -> bool {
373        if !matches!(self.mixer, Mixer::Silent) {
374            return true;
375        }
376        #[cfg(target_arch = "wasm32")]
377        if self.web_mixer == WebMixer::Worklet {
378            if let Some((stream, link)) =
379                crate::worklet::WorkletStream::open(self.meter.clone(), self.buffer_frames)
380            {
381                self.start_remote(link);
382                if let Mixer::Remote(r) = &mut self.mixer {
383                    r.stream = Some(stream);
384                }
385                return true;
386            }
387            tracing::warn!("[flint-audio] no AudioWorklet here: mixing on the main thread");
388        }
389        self.start_local()
390    }

392    /// Mix with kira here, on [`AudioBackend`].
393    fn start_local(&mut self) -> bool {
394        // Main track with a low-pass filter, at the cutoff last asked for
395        // (fully open at 20 kHz by default)
396        #[allow(unused_mut)]
397        let mut settings = kira::AudioManagerSettings::<AudioBackend>::default();
398        #[cfg(target_arch = "wasm32")]
399        {
400            settings.backend_settings.meter = self.meter.clone();
401            settings.backend_settings.buffer_frames = self.buffer_frames;
402        }
403        let filter = settings.main_track_builder.add_effect(
404            FilterBuilder::new()
405                .mode(FilterMode::LowPass)
406                .cutoff(self.filter_cutoff),
407        );
408
409        // Try to create the audio manager; gracefully fail if no device
410        let Some(mut manager) = AudioManager::<AudioBackend>::new(settings)
411            .map_err(|e| tracing::warn!("No audio device available ({e}), running silent"))
412            .ok()
413        else {
414            return false;
415        };
416
417        // Create the music and SFX buses as children of the main track
418        let music_bus = manager
419            .add_sub_track(TrackBuilder::new())
420            .map_err(|e| tracing::warn!("Failed to create music bus: {e}"))
421            .ok();
422        let sfx_bus = manager
423            .add_sub_track(TrackBuilder::new())
424            .map_err(|e| tracing::warn!("Failed to create sfx bus: {e}"))
425            .ok();
426        self.mixer = Mixer::Local(Box::new(Local {
427            manager,
428            filter,
429            music_bus,
430            sfx_bus,
431            listener: None,
432        }));
433        self.apply_bus_volumes();
434        true
435    }

437    /// Mix somewhere else: every call from now on is written to `link`, and
438    /// the sounds cached so far are sent along it. The browser's worklet is
439    /// one end; a test can take the outbox itself.
440    pub fn start_remote(&mut self, link: Link) {
441        let mut remote = Remote {
442            link,
443            sounds: HashMap::new(),
444            lengths: HashMap::new(),
445            listener: false,
446            #[cfg(target_arch = "wasm32")]
447            stream: None,
448        };
449        let mut names: Vec<&String> = self.sound_cache.keys().collect();
450        names.sort();
451        for name in names {
452            remote.send_sound(name, &self.sound_cache[name]);
453        }
454        remote.link.send(Command::FilterCutoff {
455            hz: self.filter_cutoff as f32,
456            tween: 0.0,
457        });
458        self.mixer = Mixer::Remote(remote);
459        self.apply_bus_volumes();
460        self.let_go_of_sent_sounds();
461    }

463    fn apply_bus_volumes(&mut self) {
464        for bus in [Bus::Music, Bus::Sfx] {
465            if self.held || self.bus_volumes[bus as usize] != 1.0 {
466                self.send_bus_volume(bus, 0.016);
467            }
468        }
469    }

472    fn let_go_of_sent_sounds(&mut self) {
473        if let Mixer::Remote(r) = &self.mixer {
474            if !r.keeps_cache() && !self.sound_cache.is_empty() {
475                self.sound_cache.clear();

477        }

480    /// Once a frame: follow the mixer. In the browser this notices the
481    /// worklet running (and lets go of the main thread's copies of the
482    /// sounds it holds) or failing, in which case the main thread takes
483    /// over and [`take_restarted`](Self::take_restarted) says so.
484    pub fn poll(&mut self) {
485        #[cfg(target_arch = "wasm32")]
486        {
487            let failed = match &self.mixer {
488                Mixer::Remote(Remote {
489                    stream: Some(s), ..
490                }) => s.failed(),
491                _ => false,
492            };
493            if failed {
494                // The worklet's one-shots went with it.
495                self.oneshots.clear();
496                self.mixer = Mixer::Silent;
497                self.web_mixer = WebMixer::MainThread;
498                if self.start_local() {
499                    self.restarted = true;
500                }
501                return;
502            }
503        }
504        self.let_go_of_sent_sounds();
505    }

559    /// Hold the mix (FlintKart#27): both buses fade out and stay out, under any
560    /// volume set meanwhile, and come back to it when let go. Sounds keep

563        if held == self.held {
564            return;
565        }
566        self.held = held;

568            self.send_bus_volume(bus, if held { 0.12 } else { 0.25 });

577    /// Give the mixer a bus's volume: the one asked for, or none while held.
578    fn send_bus_volume(&mut self, bus: Bus, tween: f64) {
579        let volume = if self.held {
580            0.0
581        } else {
582            self.bus_volumes[bus as usize]
583        };
584        let db = amplitude_to_db(volume);
585        match &mut self.mixer {
586            Mixer::Silent => {}
587            Mixer::Local(l) => {
588                if let Ok(handle) = l.bus(bus) {
589                    handle.set_volume(db, tween_ms((tween * 1000.0) as u64));
590                }
591            }
592            Mixer::Remote(r) => r.link.send(Command::BusVolume {
593                bus: bus.code(),
594                db: db.0,
595                tween: tween as f32,
596            }),
597        }
598    }

622    /// Load a sound file into the cache
623    pub fn load_sound(&mut self, name: &str, path: &Path) -> Result<()> {
624        if self.has_sound(name) {
625            return Ok(());
626        }

651    /// Queue a sound file to be decoded over the next frames by
652    /// [`decode_queued`](Self::decode_queued), instead of now: a song takes
653    /// 100 to 241 ms to decode in the browser. The file is read now. A sound
654    /// already cached is left alone, and one already queued keeps its place
655    /// unless `need` is sooner.
656    pub fn queue_sound(&mut self, name: &str, path: &Path, need: Need) -> Result<()> {
657        if self.has_sound(name) {
658            return Ok(());
659        }
660        if self.decodes.contains(name) {
661            self.decodes.promote(name, need);
662            return Ok(());
663        }
664        let bytes = vfs::read(path).map_err(|e| {
665            flint_core::FlintError::AudioError(format!("Failed to read '{}': {e}", path.display()))
666        })?;
667        self.decodes.push(name, bytes, need);
668        Ok(())
669    }

671    /// Decode queued sounds for up to `budget`, soonest needed first, and
672    /// cache the ones that finish. Call once a frame; it costs nothing when
673    /// the queue is empty. Returns how many are still queued.
674    pub fn decode_queued(&mut self, budget: Duration) -> usize {
675        if self.decodes.is_empty() {
676            return 0;
677        }
678        for decoded in self.decodes.run(budget) {
679            self.finish_decode(decoded);
680        }
681        if self.decodes.is_empty() {
682            tracing::info!(
683                "Decoded {} queued sounds in {:.0} ms of frame time",
684                self.decodes.decoded,
685                self.decodes.spent.as_secs_f64() * 1000.0
686            );
687        }
688        self.decodes.len()
689    }

701    fn finish_decode(&mut self, decoded: crate::decode::Decoded) {
702        let name = decoded.name;
703        match decoded.result {
704            Ok(data) => {
705                tracing::info!(
706                    "Decoded {name} ({} KB) in {:.1} ms over {} frame(s), longest step {:.1} ms",
707                    decoded.size / 1024,
708                    decoded.spent.as_secs_f64() * 1000.0,
709                    decoded.runs,
710                    decoded.longest.as_secs_f64() * 1000.0

712                self.cache_sound(&name, data);
713            }
714            Err(e) => tracing::warn!("Failed to decode '{name}': {e}"),
715        }

718    /// Make sure a sound about to play is decoded: a small queued one is
719    /// decoded now, a big one jumps the queue and this play is skipped.
720    fn ensure_decoded(&mut self, name: &str) -> Result<()> {
721        if self.has_sound(name) {
722            return Ok(());
723        }
724        match self.decodes.size_of(name) {
725            Some(size) if size <= DECODE_NOW_BYTES => {
726                if let Some(decoded) = self.decodes.decode_now(name) {
727                    self.finish_decode(decoded);
728                }
729                Ok(())
730            }
731            Some(_) => {
732                self.decodes.promote(name, Need::Asked);
733                Err(flint_core::FlintError::AudioError(format!(
734                    "'{name}' skipped: still decoding, moved to the front of the queue"
735                )))
736            }
737            None => Err(flint_core::FlintError::AudioError(format!(
738                "Sound not cached: {name}"
739            ))),
740        }
741    }

754    /// Cache a sound made in memory under `name`, from its samples: one
755    /// channel (played on both sides) or two (left, right), each in -1..1 at
756    /// `sample_rate` Hz. Unequal channels are cut to the shorter. Replaces
757    /// any sound already cached under that name.
758    pub fn load_sound_from_samples(
759        &mut self,
760        name: &str,
761        sample_rate: u32,
762        channels: &[&[f32]],
763    ) -> Result<()> {
764        let frames: Arc<[kira::Frame]> = match channels {
765            [mono] => mono.iter().map(|&s| kira::Frame::from_mono(s)).collect(),
766            [left, right] => left
767                .iter()
768                .zip(right.iter())
769                .map(|(&l, &r)| kira::Frame { left: l, right: r })
770                .collect(),
771            _ => {
772                return Err(flint_core::FlintError::AudioError(format!(
773                    "'{name}': {} channels; a sound has one or two",
774                    channels.len()
775                )))
776            }
777        };
778        if sample_rate == 0 {
779            return Err(flint_core::FlintError::AudioError(format!(
780                "'{name}': sample rate 0"
781            )));
782        }
783        self.cache_sound(
784            name,
785            StaticSoundData {
786                sample_rate,
787                frames,
788                settings: Default::default(),
789                slice: None,
790            },
791        );
792        Ok(())
793    }

881    /// Create a spatial track with its own falloff curve (see
882    /// [`DEFAULT_ROLLOFF`]).
883    pub fn create_spatial_track_with(
884        &mut self,
885        position: Vec3,
886        (min_distance, max_distance): (f32, f32),
887        rolloff: f32,
888        bus: Bus,
889    ) -> Result<SpatialTrack> {
890        let no_listener = || flint_core::FlintError::AudioError("No listener created".into());
891        let pos = to_glam_vec3(position);
892        match &mut self.mixer {
893            Mixer::Silent => Err(flint_core::FlintError::AudioError("No audio device".into())),
894            Mixer::Local(l) => {
895                let listener_id = l.listener.as_ref().ok_or_else(no_listener)?.id();
896                let builder = SpatialTrackBuilder::new()
897                    .distances(SpatialTrackDistances {
898                        min_distance,
899                        max_distance,
900                    })
901                    .attenuation_function(Some(Easing::OutPowf(rolloff.max(0.01) as f64)));
902                let handle = l
903                    .bus(bus)?
904                    .add_spatial_sub_track(listener_id, pos, builder)
905                    .map_err(|e| {
906                        flint_core::FlintError::AudioError(format!(
907                            "Failed to create spatial track: {e}"
908                        ))
909                    })?;
910                Ok(SpatialTrack::Local(Box::new(handle)))
911            }
912            Mixer::Remote(r) => {
913                if !r.listener {
914                    return Err(no_listener());
915                }
916                let id = r.link.id();
917                r.link.send(Command::TrackNew {
918                    track: id,
919                    pos: pos.to_array(),
920                    min: min_distance,
921                    max: max_distance,
922                    rolloff,
923                    bus: bus.code(),
924                });
925                Ok(SpatialTrack::Remote(RemoteTrack {
926                    id,
927                    link: r.link.clone(),
928                }))
929            }
930        }
931    }

1081    /// Play a one-shot at a 3D position with its own falloff, `(min, max)`
1082    /// metres from the listener: full volume within `min`, silent past
1083    /// `max`, along the curve `rolloff` (see [`DEFAULT_ROLLOFF`]). Routed
1084    /// through `bus`.
1085    #[allow(clippy::too_many_arguments)]
1086    pub fn play_oneshot_at(
1087        &mut self,
1088        sound_name: &str,
1089        position: Vec3,
1090        volume: f64,
1091        pitch: f64,
1092        distances: (f32, f32),
1093        rolloff: f32,
1094        bus: Bus,
1095    ) -> Result<()> {
1096        self.ensure_decoded(sound_name)?;
1097        let falloff = Falloff {
1098            min: distances.0,
1099            max: distances.1,
1100            rolloff,
1101        };
1102        if self.old_mix {
1103            return self.play_oneshot_on_track(sound_name, position, volume, pitch, falloff, bus);
1104        }
1105        if !self.is_available() {
1106            return Err(flint_core::FlintError::AudioError("No audio device".into()));
1107        }
1108        // Past `max` it would be mixed in silence, at the full price of a
1109        // voice.
1110        if self.is_out_of_earshot(position, distances.1) {
1111            self.skipped_oneshots += 1;
1112            return Ok(());
1113        }
1114        let (gain, pan) = volume_and_pan(self.ears().gains(to_glam_vec3(position), &falloff));
1115        let handle = self.play_panned(sound_name, volume * gain as f64, pan, pitch, false, bus)?;
1116        let now = Instant::now();
1117        self.steal_beyond_cap(sound_name, now);
1118        let ends_at = match &self.mixer {
1119            Mixer::Remote(r) => r.sound(sound_name).ok().and_then(|id| {
1120                let secs = r.lengths.get(&id)? / pitch.abs().max(1e-3);
1121                Some(now + Duration::from_secs_f64(secs))
1122            }),
1123            _ => None,
1124        };
1125        self.oneshots.push(OneShot {
1126            sound: sound_name.to_string(),
1127            handle,
1128            track: None,
1129            position,
1130            falloff,
1131            volume,
1132            ends_at,
1133            stolen: false,
1134        });
1135        Ok(())
1136    }

1218    /// Fade out the oldest one-shots of `sound` so one more fits under
1219    /// [`MAX_ONESHOTS_PER_SOUND`]. Ones already stopping don't count.
1220    fn steal_beyond_cap(&mut self, sound: &str, now: Instant) {
1221        let mut playing: Vec<&mut OneShot> = self
1222            .oneshots
1223            .iter_mut()
1224            .filter(|o| o.sound == sound && o.is_sounding(now))
1225            .collect();
1226        let excess = (playing.len() + 1).saturating_sub(MAX_ONESHOTS_PER_SOUND);
1227        for oldest in playing.iter_mut().take(excess) {
1228            oldest.handle.stop(Tween {
1229                duration: STEAL_FADE,
1230                ..Default::default()
1231            });
1232            oldest.stolen = true;
1233            let faded = now + STEAL_FADE;
1234            oldest.ends_at = Some(oldest.ends_at.map_or(faded, |t| t.min(faded)));
1235            self.stolen_oneshots += 1;
1236        }
1237    }

1252    pub fn mix_counts(&self) -> Option<MixCounts> {
1253        let mut counts = MixCounts {
1254            skipped_oneshots: self.skipped_oneshots,
1255            stolen_oneshots: self.stolen_oneshots,
1256            ..Default::default()
1257        };
1258        match &self.mixer {
1259            Mixer::Silent => return None,
1260            Mixer::Local(l) => {
1261                for bus in [&l.sfx_bus, &l.music_bus].into_iter().flatten() {
1262                    counts.voices += bus.num_sounds();
1263                    counts.tracks += bus.num_sub_tracks();
1264                }
1265                counts.voices += self.oneshots.iter().filter(|o| o.track.is_some()).count();
1266            }
1267            Mixer::Remote(_) => counts.voices += self.oneshots.len(),
1268        }
1269        Some(counts)
1270    }

1291    pub fn set_filter_cutoff(&mut self, hz: f32) {
