1//! Synchronization between FlintWorld audio_source components and Kira spatial tracks
2//!
3//! Follows the same pattern as PhysicsSync: discovers entities with audio components,
4//! creates Kira tracks, and updates positions each frame.
5//! Also propagates per-frame pitch/volume changes from ECS to Kira sound handles.
6
7use crate::engine::{amplitude_to_db, AudioEngine, Bus};
8use flint_core::components as comp;
9use flint_core::toml_util::{toml_f32, toml_f64};
10use flint_core::{EntityId, Vec3};
11use flint_ecs::FlintWorld;
12use kira::sound::static_sound::StaticSoundHandle;
13use kira::track::SpatialTrackHandle;
14use kira::Tween;
15use std::collections::{HashMap, HashSet};
16use std::time::Duration;

18/// Per-entity audio state: spatial track handle + active sound handles + cached parameters
19struct SpatialEntry {
20    track: SpatialTrackHandle,
21    handles: Vec<StaticSoundHandle>,
22    last_volume: f64,
23    last_pitch: f64,
24}
25
26/// Non-spatial entry: sound handles + cached parameters
27struct NonSpatialEntry {
28    handles: Vec<StaticSoundHandle>,
29    last_volume: f64,
30    last_pitch: f64,
31}
32
33/// Tracks which entities have been synced to Kira and holds their spatial track handles
34#[derive(Default)]
35pub struct AudioSync {
36    /// EntityId → spatial audio entry for 3D-positioned sounds
37    spatial_map: HashMap<EntityId, SpatialEntry>,
38    /// EntityId → non-spatial audio entry (ambient/UI sounds)
39    non_spatial_map: HashMap<EntityId, NonSpatialEntry>,
40    /// Entities that have already been discovered and synced
41    synced_entities: HashSet<EntityId>,
42}
43
44/// Smooth tween duration for parameter changes (avoids clicks)
45const PARAM_TWEEN: Tween = Tween {
46    duration: Duration::from_millis(16),
47    easing: kira::Easing::Linear,
48    start_time: kira::StartTime::Immediate,
49};
50
51impl AudioSync {
52    pub fn new() -> Self {
53        Self {
54            spatial_map: HashMap::new(),
55            non_spatial_map: HashMap::new(),
56            synced_entities: HashSet::new(),
57        }
58    }
59
60    /// Clear all sync state for a scene transition.
61    /// Stops all playing sounds before dropping handles, since Kira sounds
62    /// continue playing even after their handles are dropped.
63    pub fn clear(&mut self) {
64        let stop_tween = Tween {
65            duration: Duration::from_millis(16),
66            ..Default::default()
67        };
68        for entry in self.spatial_map.values_mut() {
69            for handle in &mut entry.handles {
70                let _ = handle.stop(stop_tween);
71            }
72        }
73        for entry in self.non_spatial_map.values_mut() {
74            for handle in &mut entry.handles {
75                let _ = handle.stop(stop_tween);
76            }
77        }
78        self.spatial_map.clear();
79        self.non_spatial_map.clear();
80        self.synced_entities.clear();
81    }

83    /// Discover entities with `audio_source` components and create Kira tracks
84    pub fn sync_new_sources(&mut self, world: &FlintWorld, engine: &mut AudioEngine) {
85        if !engine.is_available() {
86            return;
87        }
88
89        for &entity_id in world.entities_with_component(comp::AUDIO_SOURCE) {
90            if self.synced_entities.contains(&entity_id) {
91                continue;
92            }
93
94            let components = match world.get_components(entity_id) {
95                Some(c) => c,
96                None => continue,
97            };
98
99            let audio_data = match components.get(comp::AUDIO_SOURCE) {
100                Some(v) => v,
101                None => continue,
102            };
103
104            // Read audio_source fields
105            let spatial = audio_data
106                .get("spatial")
107                .and_then(|v| v.as_bool())
108                .unwrap_or(true);
109
110            let autoplay = audio_data
111                .get("autoplay")
112                .and_then(|v| v.as_bool())
113                .unwrap_or(true);
114
115            let looping = audio_data
116                .get("loop")
117                .and_then(|v| v.as_bool())
118                .unwrap_or(false);
119
120            let volume = audio_data.get("volume").and_then(toml_f64).unwrap_or(1.0);
121
122            let pitch = audio_data.get("pitch").and_then(toml_f64).unwrap_or(1.0);
123
124            let file = audio_data
125                .get("file")
126                .and_then(|v| v.as_str())
127                .unwrap_or("");
128
129            let bus = audio_data
130                .get("bus")
131                .and_then(|v| v.as_str())
132                .map(Bus::from_name)
133                .unwrap_or_default();
134
135            if file.is_empty() {
136                self.synced_entities.insert(entity_id);
137                continue;
138            }
139
140            if spatial {
141                let transform = world.get_transform(entity_id).unwrap_or_default();
142
143                let min_distance = audio_data
144                    .get("min_distance")
145                    .and_then(toml_f32)
146                    .unwrap_or(1.0);
147
148                let max_distance = audio_data
149                    .get("max_distance")
150                    .and_then(toml_f32)
151                    .unwrap_or(25.0);
152
153                match engine.create_spatial_track(
154                    transform.position,
155                    min_distance,
156                    max_distance,
157                    bus,
158                ) {
159                    Ok(mut track) => {
160                        let mut handles = Vec::new();
161                        if autoplay && engine.has_sound(file) {
162                            match engine
163                                .play_on_spatial_track(file, &mut track, volume, pitch, looping)
164                            {
165                                Ok(handle) => handles.push(handle),
166                                Err(e) => tracing::warn!("Failed to play '{}': {:?}", file, e),
167                            }
168                        }
169                        self.spatial_map.insert(
170                            entity_id,
171                            SpatialEntry {
172                                track,
173                                handles,
174                                last_volume: volume,
175                                last_pitch: pitch,
176                            },
177                        );
178                    }
179                    Err(e) => {
180                        tracing::warn!("Failed to create spatial track: {:?}", e);
181                    }
182                }
183            } else {
184                // Non-spatial: play directly on the main track
185                let mut handles = Vec::new();
186                if autoplay && engine.has_sound(file) {
187                    match engine.play_non_spatial(file, volume, pitch, looping, bus) {
188                        Ok(handle) => handles.push(handle),
189                        Err(e) => {
190                            tracing::warn!("Failed to play non-spatial '{}': {:?}", file, e)
191                        }
192                    }
193                }
194                self.non_spatial_map.insert(
195                    entity_id,
196                    NonSpatialEntry {
197                        handles,
198                        last_volume: volume,
199                        last_pitch: pitch,
200                    },
201                );
202            }
203
204            self.synced_entities.insert(entity_id);
205        }
206    }

208    /// Update spatial track positions for entities that have moved
209    pub fn update_positions(&mut self, world: &FlintWorld) {
210        let tween = Tween {
211            duration: Duration::ZERO,
212            ..Default::default()
213        };
214
215        for (entity_id, entry) in &mut self.spatial_map {
216            // Use world position to account for transform hierarchy (e.g. sounds parented to kart)
217            let position = match world.get_world_position(*entity_id) {
218                Some(p) => p,
219                None => continue,
220            };
221
222            let pos = to_glam_vec3(position);
223            entry.track.set_position(pos, tween);
224        }
225    }

227    /// Read audio_source pitch/volume from ECS and push changes to Kira sound handles
228    pub fn update_parameters(&mut self, world: &FlintWorld) {
229        // Update spatial entries
230        for (entity_id, entry) in &mut self.spatial_map {
231            let (volume, pitch) = read_audio_params(world, *entity_id);
232
233            let vol_changed = (volume - entry.last_volume).abs() > 0.001;
234            let pitch_changed = (pitch - entry.last_pitch).abs() > 0.001;
235
236            if vol_changed || pitch_changed {
237                for handle in &mut entry.handles {
238                    if vol_changed {
239                        handle.set_volume(amplitude_to_db(volume), PARAM_TWEEN);
240                    }
241                    if pitch_changed {
242                        handle.set_playback_rate(kira::PlaybackRate(pitch), PARAM_TWEEN);
243                    }
244                }
245                entry.last_volume = volume;
246                entry.last_pitch = pitch;
247            }
248        }
249
250        // Update non-spatial entries
251        for (entity_id, entry) in &mut self.non_spatial_map {
252            let (volume, pitch) = read_audio_params(world, *entity_id);
253
254            let vol_changed = (volume - entry.last_volume).abs() > 0.001;
255            let pitch_changed = (pitch - entry.last_pitch).abs() > 0.001;
256
257            if vol_changed || pitch_changed {
258                for handle in &mut entry.handles {
259                    if vol_changed {
260                        handle.set_volume(amplitude_to_db(volume), PARAM_TWEEN);
261                    }
262                    if pitch_changed {
263                        handle.set_playback_rate(kira::PlaybackRate(pitch), PARAM_TWEEN);
264                    }
265                }
266                entry.last_volume = volume;
267                entry.last_pitch = pitch;
268            }
269        }
270    }

272    /// Check if an entity has been synced
273    pub fn is_synced(&self, entity_id: EntityId) -> bool {
274        self.synced_entities.contains(&entity_id)
275    }

277    /// Number of spatial tracks currently active
278    pub fn spatial_track_count(&self) -> usize {
279        self.spatial_map.len()
280    }

283/// Read volume and pitch from an entity's audio_source component
284fn read_audio_params(world: &FlintWorld, entity_id: EntityId) -> (f64, f64) {
285    let components = match world.get_components(entity_id) {
286        Some(c) => c,
287        None => return (1.0, 1.0),
288    };
289
290    let audio_data = match components.get(comp::AUDIO_SOURCE) {
291        Some(v) => v,
292        None => return (1.0, 1.0),
293    };
294
295    let volume = audio_data.get("volume").and_then(toml_f64).unwrap_or(1.0);
296
297    let pitch = audio_data.get("pitch").and_then(toml_f64).unwrap_or(1.0);
298
299    (volume, pitch)
300}

302/// Convert Flint Vec3 to glam Vec3
303fn to_glam_vec3(v: Vec3) -> glam::Vec3 {
304    glam::Vec3::new(v.x, v.y, v.z)

311    #[test]
312    fn test_audio_sync_discovers_entities() {
313        let mut world = FlintWorld::new();
314        let id = world.spawn("fireplace").unwrap();
315
316        // Add transform
317        let transform = toml::Value::Table({
318            let mut t = toml::map::Map::new();
319            t.insert(
320                "position".into(),
321                toml::Value::Array(vec![
322                    toml::Value::Float(-5.0),
323                    toml::Value::Float(1.0),
324                    toml::Value::Float(-3.0),
325                ]),
326            );
327            t
328        });
329        world.set_component(id, "transform", transform).unwrap();
330
331        // Add audio_source
332        let audio = toml::Value::Table({
333            let mut t = toml::map::Map::new();
334            t.insert(
335                "file".into(),
336                toml::Value::String("fire_crackle.ogg".into()),
337            );
338            t.insert("spatial".into(), toml::Value::Boolean(true));
339            t.insert("loop".into(), toml::Value::Boolean(true));
340            t.insert("volume".into(), toml::Value::Float(0.8));
341            t
342        });
343        world.set_component(id, "audio_source", audio).unwrap();
344
345        let sync = AudioSync::new();
346        // Without an engine, we can't sync, but we can verify the sync tracks state
347        assert!(!sync.is_synced(id));
348        assert_eq!(sync.spatial_track_count(), 0);
349    }
