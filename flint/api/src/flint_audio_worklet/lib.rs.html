1//! Flint's mixer in the browser's audio thread (#136).
2//!
3//! The web player mixed on the main thread: cpal's WebAudio host asks for
4//! each buffer from an `onended` callback, so on a phone the mix took about
5//! a third of the frame and every long frame made a buffer late. This crate
6//! is a second, small wasm that runs kira inside an `AudioWorkletProcessor`
7//! instead. It knows nothing of Flint: the page posts it sounds (as samples)
8//! and one batch of [`protocol::Command`]s a frame, and the processor asks
9//! it for 128 frames at a time.
10//!
11//! [`Mixer`] is plain Rust, tested natively; `exports` is the C ABI the
12//! processor's JS calls (`crates/flint-audio/src/worklet.js`). The wasm is
13//! built on its own, with only kira: `trunk` runs
14//! `cargo build -p flint-audio-worklet --target wasm32-unknown-unknown
15//! --profile worklet` before each build (crates/flint-player/Trunk.toml).

40/// A kira backend with no device: it hands the renderer back so the
41/// worklet's `process` can drive it.
42struct Capture {
43    renderer: Arc<Mutex<Option<Renderer>>>,
44}
45
46impl Backend for Capture {
47    type Settings = (u32, Arc<Mutex<Option<Renderer>>>);
48    type Error = ();
49
50    fn setup((rate, renderer): Self::Settings, _: usize) -> Result<(Self, u32), ()> {
51        Ok((Self { renderer }, rate))
52    }
53
54    fn start(&mut self, renderer: Renderer) -> Result<(), ()> {
55        *self.renderer.lock().map_err(|_| ())? = Some(renderer);
56        Ok(())
57    }
58}

60fn tween(secs: f32) -> Tween {
61    Tween {
62        duration: Duration::from_secs_f32(secs.max(0.0)),
63        easing: Easing::Linear,
64        ..Default::default()
65    }

72/// One-shots are swept for finished ones every this many renders (128
73/// frames each, so about every 0.1 s).
74const SWEEP_EVERY: u32 = 32;

76/// kira, set up as `flint_audio::AudioEngine` sets it up on the desktop (a
77/// low-pass on the main track, music and SFX buses under it), driven by
78/// commands instead of calls.
79pub struct Mixer {
80    manager: AudioManager<Capture>,
81    renderer: Renderer,
82    filter: FilterHandle,
83    buses: [TrackHandle; 2],
84    listener: Option<ListenerHandle>,
85    sounds: HashMap<u32, StaticSoundData>,
86    tracks: HashMap<u32, SpatialTrackHandle>,
87    voices: HashMap<u32, StaticSoundHandle>,
88    one_shots: Vec<(SpatialTrackHandle, StaticSoundHandle)>,
89    renders: u32,
90    /// Commands that failed (an unknown id, a full track, an unreadable
91    /// batch). The processor reports the count with its buffer stats.
92    pub errors: u32,
93    interleaved: Vec<f32>,
94    planar: Vec<f32>,
95}

97impl Mixer {
98    pub fn new(sample_rate: u32) -> Option<Self> {
99        let slot = Arc::new(Mutex::new(None));
100        let mut settings = AudioManagerSettings::<Capture> {
101            backend_settings: (sample_rate, slot.clone()),
102            ..Default::default()
103        };
104        let filter = settings.main_track_builder.add_effect(
105            FilterBuilder::new()
106                .mode(FilterMode::LowPass)
107                .cutoff(20_000.0),
108        );
109        let mut manager = AudioManager::new(settings).ok()?;
110        let sfx = manager.add_sub_track(TrackBuilder::new()).ok()?;
111        let music = manager.add_sub_track(TrackBuilder::new()).ok()?;
112        let renderer = slot.lock().ok()?.take()?;
113        Some(Self {
114            manager,
115            renderer,
116            filter,
117            buses: [sfx, music],
118            listener: None,
119            sounds: HashMap::new(),
120            tracks: HashMap::new(),
121            voices: HashMap::new(),
122            one_shots: Vec::new(),
123            renders: 0,
124            errors: 0,
125            interleaved: Vec::new(),
126            planar: Vec::new(),
127        })
128    }

130    /// Keep a sound under `id`, from one channel of samples (played on both
131    /// sides) or two interleaved. Replaces any sound with that id; voices
132    /// already playing keep the old samples.
133    pub fn load_sound(&mut self, id: u32, sample_rate: u32, channels: u32, samples: &[f32]) {
134        let frames: Arc<[Frame]> = match channels {
135            1 => samples.iter().map(|&s| Frame::from_mono(s)).collect(),
136            2 => samples
137                .as_chunks::<2>()
138                .0
139                .iter()
140                .map(|&[left, right]| Frame { left, right })
141                .collect(),
142            _ => {
143                self.errors += 1;
144                return;
145            }
146        };
147        self.sounds.insert(
148            id,
149            StaticSoundData {
150                sample_rate,
151                frames,
152                settings: Default::default(),
153                slice: None,
154            },
155        );
156    }
157
158    /// Apply one batch of commands, in order.
159    pub fn apply(&mut self, batch: &[u8]) {
160        let mut reader = Reader::new(batch);
161        for command in reader.by_ref() {
162            if self.command(command).is_none() {
163                self.errors += 1;
164            }
165        }
166        if reader.error.is_some() {
167            self.errors += 1;
168        }
169    }
170
171    fn sound(&self, id: u32, db: f32, rate: f32, looping: bool) -> Option<StaticSoundData> {
172        let mut data = self
173            .sounds
174            .get(&id)?
175            .volume(Decibels(db))
176            .playback_rate(PlaybackRate(rate as f64));
177        if looping {
178            data = data.loop_region(..);
179        }
180        Some(data)
181    }
182
183    fn bus(&mut self, bus: u8) -> Option<&mut TrackHandle> {
184        self.buses.get_mut(bus as usize)
185    }
186
187    fn spatial_track(
188        &mut self,
189        pos: [f32; 3],
190        (min, max): (f32, f32),
191        rolloff: f32,
192        bus: u8,
193    ) -> Option<SpatialTrackHandle> {
194        let listener = self.listener.as_ref()?.id();
195        let builder = SpatialTrackBuilder::new()
196            .distances(SpatialTrackDistances {
197                min_distance: min,
198                max_distance: max,
199            })
200            .attenuation_function(Some(Easing::OutPowf(rolloff.max(0.01) as f64)));
201        self.bus(bus)?
202            .add_spatial_sub_track(listener, vec3(pos), builder)
203            .ok()
204    }
205
206    fn command(&mut self, command: Command) -> Option<()> {
207        match command {
208            Command::FilterCutoff { hz, tween: t } => {
209                self.filter.set_cutoff(hz as f64, tween(t));
210            }
211            Command::BusVolume { bus, db, tween: t } => {
212                self.bus(bus)?.set_volume(Decibels(db), tween(t));
213            }
214            Command::Listener { pos, quat } => {
215                let orientation = glam::Quat::from_array(quat);
216                match &mut self.listener {
217                    Some(l) => {
218                        l.set_position(vec3(pos), tween(0.0));
219                        l.set_orientation(orientation, tween(0.0));
220                    }
221                    None => {
222                        self.listener =
223                            Some(self.manager.add_listener(vec3(pos), orientation).ok()?);
224                    }
225                }
226            }
227            Command::TrackNew {
228                track,
229                pos,
230                min,
231                max,
232                rolloff,
233                bus,
234            } => {
235                let handle = self.spatial_track(pos, (min, max), rolloff, bus)?;
236                self.tracks.insert(track, handle);
237            }
238            Command::TrackPosition { track, pos } => {
239                self.tracks
240                    .get_mut(&track)?
241                    .set_position(vec3(pos), tween(0.0));
242            }
243            Command::TrackDrop { track } => {
244                self.tracks.remove(&track)?;
245            }
246            Command::Play {
247                voice,
248                sound,
249                target,
250                db,
251                rate,
252                pan,
253                looping,
254            } => {
255                let data = self.sound(sound, db, rate, looping)?.panning(Panning(pan));
256                let handle = match target {
257                    Target::Bus(bus) => self.bus(bus)?.play(data).ok()?,
258                    Target::Track(track) => self.tracks.get_mut(&track)?.play(data).ok()?,
259                };
260                self.voices.insert(voice, handle);
261            }
262            Command::OneShot {
263                sound,
264                pos,
265                db,
266                rate,
267                min,
268                max,
269                rolloff,
270                bus,
271            } => {
272                let data = self.sound(sound, db, rate, false)?;
273                let mut track = self.spatial_track(pos, (min, max), rolloff, bus)?;
274                let handle = track.play(data).ok()?;
275                self.one_shots.push((track, handle));
276            }
277            Command::VoiceVolume {
278                voice,
279                db,
280                tween: t,
281            } => {
282                self.voices
283                    .get_mut(&voice)?
284                    .set_volume(Decibels(db), tween(t));
285            }
286            Command::VoiceRate {
287                voice,
288                rate,
289                tween: t,
290            } => {
291                self.voices
292                    .get_mut(&voice)?
293                    .set_playback_rate(PlaybackRate(rate as f64), tween(t));
294            }
295            Command::VoiceStop { voice, tween: t } => {
296                self.voices.get_mut(&voice)?.stop(tween(t));
297            }
298            Command::VoiceDrop { voice } => {
299                self.voices.remove(&voice)?;
300            }
301            Command::VoicePanning {
302                voice,
303                pan,
304                tween: t,
305            } => {
306                self.voices
307                    .get_mut(&voice)?
308                    .set_panning(Panning(pan), tween(t));
309            }
310            Command::VoicePause { voice, tween: t } => {
311                self.voices.get_mut(&voice)?.pause(tween(t));
312            }
313            Command::VoiceResume { voice, tween: t } => {
314                self.voices.get_mut(&voice)?.resume(tween(t));
315            }
316            Command::ClearOneShots => self.one_shots.clear(),
317        }
318        Some(())
319    }

321    /// Mix the next `frames` frames: left channel, then right.
322    pub fn render(&mut self, frames: usize) -> &[f32] {
323        self.renders = self.renders.wrapping_add(1);
324        if self.renders.is_multiple_of(SWEEP_EVERY) {
325            self.one_shots
326                .retain(|(_, sound)| sound.state() != PlaybackState::Stopped);
327        }
328        self.interleaved.resize(frames * 2, 0.0);
329        self.renderer.on_start_processing();
330        self.renderer.process(&mut self.interleaved, 2);
331        self.planar.resize(frames * 2, 0.0);
332        let (left, right) = self.planar.split_at_mut(frames);
333        for (i, &[l, r]) in self.interleaved.as_chunks::<2>().0.iter().enumerate() {
334            left[i] = l;
335            right[i] = r;
336        }
337        &self.planar
338    }

340    /// Voices the page still holds handles to, and placed one-shots playing.
341    pub fn counts(&self) -> (usize, usize, usize) {
342        (self.voices.len(), self.tracks.len(), self.one_shots.len())
343    }

354    fn batch(commands: &[Command]) -> Vec<u8> {
355        let mut bytes = Vec::new();
356        for c in commands {
357            c.encode(&mut bytes);
358        }
359        bytes
360    }

362    /// A 440 Hz tone, mono, a second long.
363    fn tone() -> Vec<f32> {
364        (0..44_100)
365            .map(|i| (i as f32 * 440.0 * std::f32::consts::TAU / 44_100.0).sin() * 0.5)
366            .collect()
367    }

369    /// RMS of each side over `quanta` renders.
370    fn loudness(m: &mut Mixer, quanta: usize) -> (f32, f32) {
371        let (mut l, mut r) = (0.0, 0.0);
372        for _ in 0..quanta {
373            let out = m.render(QUANTUM);
374            l += out[..QUANTUM].iter().map(|s| s * s).sum::<f32>();
375            r += out[QUANTUM..].iter().map(|s| s * s).sum::<f32>();
376        }
377        let n = (quanta * QUANTUM) as f32;
378        ((l / n).sqrt(), (r / n).sqrt())
379    }

391    #[test]
392    fn a_looping_sound_on_a_bus_plays_fades_and_stops() {
393        let mut m = mixer_with_tone();
394        assert_eq!(loudness(&mut m, 4), (0.0, 0.0), "silent before a play");
395        m.apply(&batch(&[Command::Play {
396            voice: 9,
397            sound: 1,
398            target: Target::Bus(BUS_MUSIC),
399            db: 0.0,
400            rate: 1.0,
401            pan: 0.0,
402            looping: true,
403        }]));
404        let (l, r) = loudness(&mut m, 40);
405        assert!(l > 0.3 && (l - r).abs() < 1e-4, "{l} {r}");
406        // Past the tone's one second, it loops.
407        let (l, _) = loudness(&mut m, 400);
408        assert!(l > 0.3, "{l}");
409
410        m.apply(&batch(&[Command::BusVolume {
411            bus: BUS_MUSIC,
412            db: -20.0,
413            tween: 0.0,
414        }]));
415        loudness(&mut m, 2);
416        let (quiet, _) = loudness(&mut m, 40);
417        assert!(
418            (quiet / l - 0.1).abs() < 0.02,
419            "-20 dB is a tenth: {quiet} vs {l}"
420        );
421
422        m.apply(&batch(&[Command::VoiceStop {
423            voice: 9,
424            tween: 0.0,
425        }]));
426        loudness(&mut m, 4);
427        assert_eq!(loudness(&mut m, 4), (0.0, 0.0));
428        assert_eq!(m.errors, 0);
429    }

432    fn a_spatial_track_pans_to_its_side_and_falls_silent_past_max() {
433        let mut m = mixer_with_tone();
434        let track = |track, x: f32| Command::TrackNew {
435            track,
436            pos: [x, 0.0, 0.0],
437            min: 1.0,
438            max: 25.0,
439            rolloff: 2.0,
440            bus: BUS_SFX,
441        };
442        let play = |voice, track| Command::Play {
443            voice,
444            sound: 1,
445            target: Target::Track(track),
446            db: 0.0,
447            rate: 1.0,
448            pan: 0.0,
449            looping: true,
450        };
451        m.apply(&batch(&[track(1, 3.0), play(1, 1)]));
452        let (l, r) = loudness(&mut m, 40);
453        assert!(r > l * 1.5, "+x is on the right: {l} {r}");
454        m.apply(&batch(&[Command::TrackPosition {
455            track: 1,
456            pos: [-3.0, 0.0, 0.0],
457        }]));
458        loudness(&mut m, 4);
459        let (l, r) = loudness(&mut m, 40);
460        assert!(l > r * 1.5, "moved to -x: {l} {r}");
461
462        m.apply(&batch(&[Command::TrackPosition {
463            track: 1,
464            pos: [0.0, 0.0, 40.0],
465        }]));
466        loudness(&mut m, 4);
467        let (l, r) = loudness(&mut m, 40);
468        assert!(l < 1e-3 && r < 1e-3, "past max_distance: {l} {r}");
469
470        // Dropping the track takes its sound with it.
471        m.apply(&batch(&[
472            Command::TrackPosition {
