1//! What the main thread tells the worklet's mixer (#136).
2//!
3//! The page and its `AudioWorklet` share no memory (GitHub Pages can't send
4//! the headers `SharedArrayBuffer` needs), so the engine's calls become
5//! [`Command`]s, written into one byte batch a frame and posted to the
6//! worklet, which applies them to its own kira. Sounds cross separately,
7//! once each, as samples. Every id is made on the main thread, so a call
8//! returns its handle at once and never waits for an answer.
9//!
10//! This file has no dependencies: `flint-audio` includes it with `#[path]`
11//! to write batches, and the worklet crate reads them, so both sides are
12//! always built from the same encoding.
13
14/// The mixer bus a sound or a spatial track hangs off.
15pub const BUS_SFX: u8 = 0;
16pub const BUS_MUSIC: u8 = 1;
17
18/// Where a voice plays: a bus, or a spatial track made earlier.
19#[derive(Debug, Clone, Copy, PartialEq)]
20pub enum Target {
21    Bus(u8),
22    Track(u32),
23}
24
25/// One engine call. Tween lengths are seconds, linear.
26#[derive(Debug, Clone, PartialEq)]
27pub enum Command {
28    /// The main track's low-pass cutoff, Hz.
29    FilterCutoff {
30        hz: f32,
31        tween: f32,
32    },
33    BusVolume {
34        bus: u8,
35        db: f32,
36        tween: f32,
37    },
38    /// Place the listener, making it if there is none yet.
39    Listener {
40        pos: [f32; 3],
41        quat: [f32; 4],
42    },
43    TrackNew {
44        track: u32,
45        pos: [f32; 3],
46        min: f32,
47        max: f32,
48        rolloff: f32,
49        bus: u8,
50    },
51    TrackPosition {
52        track: u32,
53        pos: [f32; 3],
54    },
55    /// The track's handle was dropped: the track goes, with its sounds.
56    TrackDrop {
57        track: u32,
58    },
59    /// `pan` is kira's `Panning`: -1 left, 1 right, at constant power. A
60    /// placed sound plays on a bus this way, steered once a frame (#138).
61    Play {
62        voice: u32,
63        sound: u32,
64        target: Target,
65        db: f32,
66        rate: f32,
67        pan: f32,
68        looping: bool,
69    },
70    /// A placed one-shot. The worklet keeps its track until it ends.
71    OneShot {
72        sound: u32,
73        pos: [f32; 3],
74        db: f32,
75        rate: f32,
76        min: f32,
77        max: f32,
78        rolloff: f32,
79        bus: u8,
80    },
81    VoiceVolume {
82        voice: u32,
83        db: f32,
84        tween: f32,
85    },
86    VoiceRate {
87        voice: u32,
88        rate: f32,
89        tween: f32,
90    },
91    VoiceStop {
92        voice: u32,
93        tween: f32,
94    },
95    VoicePanning {
96        voice: u32,
97        pan: f32,
98        tween: f32,
99    },
100    /// Paused because nobody can hear it (#138); resumed where it stopped.
101    VoicePause {
102        voice: u32,
103        tween: f32,
104    },
105    VoiceResume {
106        voice: u32,
107        tween: f32,
108    },
109    /// The voice's handle was dropped; the sound plays on to its end.
110    VoiceDrop {
111        voice: u32,
112    },
113    ClearOneShots,
114}

116mod op {
117    pub const FILTER_CUTOFF: u8 = 1;
118    pub const BUS_VOLUME: u8 = 2;

121    pub const TRACK_POSITION: u8 = 5;

123    pub const PLAY_ON_BUS: u8 = 7;
124    pub const PLAY_ON_TRACK: u8 = 8;
125    pub const ONE_SHOT: u8 = 9;
126    pub const VOICE_VOLUME: u8 = 10;
127    pub const VOICE_RATE: u8 = 11;
128    pub const VOICE_STOP: u8 = 12;
129    pub const VOICE_DROP: u8 = 13;
130    pub const CLEAR_ONE_SHOTS: u8 = 14;
131    pub const VOICE_PANNING: u8 = 15;
132    pub const VOICE_PAUSE: u8 = 16;
133    pub const VOICE_RESUME: u8 = 17;
134}

136struct Writer<'a>(&'a mut Vec<u8>);
137
138impl Writer<'_> {
139    fn u8(&mut self, v: u8) -> &mut Self {
140        self.0.push(v);
141        self
142    }
143    fn u32(&mut self, v: u32) -> &mut Self {
144        self.0.extend_from_slice(&v.to_le_bytes());
145        self
146    }
147    fn f32(&mut self, v: f32) -> &mut Self {
148        self.0.extend_from_slice(&v.to_le_bytes());
149        self
150    }
151    fn f32s(&mut self, v: &[f32]) -> &mut Self {
152        for &x in v {
153            self.f32(x);
154        }
155        self
156    }
157}
158
159impl Command {
160    /// Append this command to a batch.
161    pub fn encode(&self, out: &mut Vec<u8>) {
162        let mut w = Writer(out);
163        match *self {
164            Command::FilterCutoff { hz, tween } => {
165                w.u8(op::FILTER_CUTOFF).f32(hz).f32(tween);
166            }
167            Command::BusVolume { bus, db, tween } => {
168                w.u8(op::BUS_VOLUME).u8(bus).f32(db).f32(tween);
169            }
170            Command::Listener { pos, quat } => {
171                w.u8(op::LISTENER).f32s(&pos).f32s(&quat);
172            }
173            Command::TrackNew {
174                track,
175                pos,
176                min,
177                max,
178                rolloff,
179                bus,
180            } => {
181                w.u8(op::TRACK_NEW)
182                    .u32(track)
183                    .f32s(&pos)
184                    .f32(min)
185                    .f32(max)
186                    .f32(rolloff)
187                    .u8(bus);
188            }
189            Command::TrackPosition { track, pos } => {
190                w.u8(op::TRACK_POSITION).u32(track).f32s(&pos);
191            }
192            Command::TrackDrop { track } => {
193                w.u8(op::TRACK_DROP).u32(track);
194            }
195            Command::Play {
196                voice,
197                sound,
198                target,
199                db,
200                rate,
201                pan,
202                looping,
203            } => {
204                match target {
205                    Target::Bus(bus) => w.u8(op::PLAY_ON_BUS).u8(bus),
206                    Target::Track(track) => w.u8(op::PLAY_ON_TRACK).u32(track),
207                };
208                w.u32(voice)
209                    .u32(sound)
210                    .f32(db)
211                    .f32(rate)
212                    .f32(pan)
213                    .u8(looping as u8);
214            }
215            Command::OneShot {
216                sound,
217                pos,
218                db,
219                rate,
220                min,
221                max,
222                rolloff,
223                bus,
224            } => {
225                w.u8(op::ONE_SHOT)
226                    .u32(sound)
227                    .f32s(&pos)
228                    .f32(db)
229                    .f32(rate)
230                    .f32(min)
231                    .f32(max)
232                    .f32(rolloff)
233                    .u8(bus);
234            }
235            Command::VoiceVolume { voice, db, tween } => {
236                w.u8(op::VOICE_VOLUME).u32(voice).f32(db).f32(tween);
237            }
238            Command::VoiceRate { voice, rate, tween } => {
239                w.u8(op::VOICE_RATE).u32(voice).f32(rate).f32(tween);
240            }
241            Command::VoiceStop { voice, tween } => {
242                w.u8(op::VOICE_STOP).u32(voice).f32(tween);
243            }
244            Command::VoiceDrop { voice } => {
245                w.u8(op::VOICE_DROP).u32(voice);
246            }
247            Command::VoicePanning { voice, pan, tween } => {
248                w.u8(op::VOICE_PANNING).u32(voice).f32(pan).f32(tween);
249            }
250            Command::VoicePause { voice, tween } => {
251                w.u8(op::VOICE_PAUSE).u32(voice).f32(tween);
252            }
253            Command::VoiceResume { voice, tween } => {
254                w.u8(op::VOICE_RESUME).u32(voice).f32(tween);
255            }
256            Command::ClearOneShots => {
257                w.u8(op::CLEAR_ONE_SHOTS);
258            }
259        }
260    }
261}

263/// Reads the commands of one batch in order. It stops at the first byte it
264/// can't read (a truncated or unknown command) and keeps it in
265/// [`error`](Self::error).
266pub struct Reader<'a> {
267    bytes: &'a [u8],
268    at: usize,
269    pub error: Option<usize>,
270}
271
272impl<'a> Reader<'a> {
273    pub fn new(bytes: &'a [u8]) -> Self {
274        Self {
275            bytes,
276            at: 0,
277            error: None,
278        }
279    }
280
281    fn take<const N: usize>(&mut self) -> Option<[u8; N]> {
282        let end = self.at.checked_add(N)?;
283        let b = self.bytes.get(self.at..end)?.try_into().ok()?;
284        self.at = end;
285        Some(b)
286    }
287    fn u8(&mut self) -> Option<u8> {
288        self.take::<1>().map(|b| b[0])
289    }
290    fn u32(&mut self) -> Option<u32> {
291        self.take().map(u32::from_le_bytes)
292    }
293    fn f32(&mut self) -> Option<f32> {
294        self.take().map(f32::from_le_bytes)
295    }
296    fn vec3(&mut self) -> Option<[f32; 3]> {
297        Some([self.f32()?, self.f32()?, self.f32()?])
298    }
299
300    fn command(&mut self) -> Option<Command> {
301        Some(match self.u8()? {
302            op::FILTER_CUTOFF => Command::FilterCutoff {
303                hz: self.f32()?,
304                tween: self.f32()?,
305            },
306            op::BUS_VOLUME => Command::BusVolume {
307                bus: self.u8()?,
308                db: self.f32()?,
309                tween: self.f32()?,
310            },
311            op::LISTENER => Command::Listener {
312                pos: self.vec3()?,
313                quat: [self.f32()?, self.f32()?, self.f32()?, self.f32()?],
314            },
315            op::TRACK_NEW => Command::TrackNew {
316                track: self.u32()?,
317                pos: self.vec3()?,
318                min: self.f32()?,
319                max: self.f32()?,
320                rolloff: self.f32()?,
321                bus: self.u8()?,
322            },
323            op::TRACK_POSITION => Command::TrackPosition {
324                track: self.u32()?,
325                pos: self.vec3()?,
326            },
327            op::TRACK_DROP => Command::TrackDrop { track: self.u32()? },
328            code @ (op::PLAY_ON_BUS | op::PLAY_ON_TRACK) => {
329                let target = if code == op::PLAY_ON_BUS {
330                    Target::Bus(self.u8()?)
331                } else {
332                    Target::Track(self.u32()?)
333                };
334                Command::Play {
335                    voice: self.u32()?,
336                    sound: self.u32()?,
337                    target,
338                    db: self.f32()?,
339                    rate: self.f32()?,
340                    pan: self.f32()?,
341                    looping: self.u8()? != 0,
342                }
343            }
344            op::ONE_SHOT => Command::OneShot {
345                sound: self.u32()?,
346                pos: self.vec3()?,
347                db: self.f32()?,
348                rate: self.f32()?,
349                min: self.f32()?,
350                max: self.f32()?,
351                rolloff: self.f32()?,
352                bus: self.u8()?,
353            },
354            op::VOICE_VOLUME => Command::VoiceVolume {
355                voice: self.u32()?,
356                db: self.f32()?,
357                tween: self.f32()?,
358            },
359            op::VOICE_RATE => Command::VoiceRate {
360                voice: self.u32()?,
361                rate: self.f32()?,
362                tween: self.f32()?,
363            },
364            op::VOICE_STOP => Command::VoiceStop {
365                voice: self.u32()?,
366                tween: self.f32()?,
367            },
368            op::VOICE_DROP => Command::VoiceDrop { voice: self.u32()? },
369            op::VOICE_PANNING => Command::VoicePanning {
370                voice: self.u32()?,
371                pan: self.f32()?,
372                tween: self.f32()?,
373            },
374            op::VOICE_PAUSE => Command::VoicePause {
375                voice: self.u32()?,
376                tween: self.f32()?,
377            },
378            op::VOICE_RESUME => Command::VoiceResume {
379                voice: self.u32()?,
380                tween: self.f32()?,
381            },
382            op::CLEAR_ONE_SHOTS => Command::ClearOneShots,
383            _ => return None,
384        })
385    }
386}
387
388impl Iterator for Reader<'_> {
389    type Item = Command;
390
391    fn next(&mut self) -> Option<Command> {
392        if self.at >= self.bytes.len() || self.error.is_some() {
393            return None;
394        }
395        let start = self.at;
396        let command = self.command();
397        if command.is_none() {
398            self.error = Some(start);
399        }
400        command
401    }
402}

496    #[test]
497    fn a_batch_reads_back_as_written() {
498        let commands = every_command();
499        let mut bytes = Vec::new();
500        for c in &commands {
501            c.encode(&mut bytes);
502        }
503        let mut reader = Reader::new(&bytes);
504        let read: Vec<Command> = reader.by_ref().collect();
505        assert_eq!(read, commands);
506        assert_eq!(reader.error, None);
507    }

509    #[test]
510    fn a_cut_or_unknown_command_stops_the_batch_where_it_starts() {
511        let mut bytes = Vec::new();
512        Command::ClearOneShots.encode(&mut bytes);
513        let good = bytes.len();
514        Command::TrackDrop { track: 1 }.encode(&mut bytes);
515        bytes.pop();
516        let mut reader = Reader::new(&bytes);
517        assert_eq!(reader.by_ref().count(), 1);
518        assert_eq!(reader.error, Some(good));
519
520        let mut reader = Reader::new(&[200, 1, 2]);
521        assert_eq!(reader.by_ref().count(), 0);
522        assert_eq!(reader.error, Some(0));
523    }
