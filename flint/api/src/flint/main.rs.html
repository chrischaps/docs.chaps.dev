13#[derive(Parser)]
14#[command(name = "flint")]
15#[command(about = "CLI-first game engine for AI-assisted development", long_about = None)]
16#[command(version)]
17struct Cli {
18    #[command(subcommand)]
19    command: Commands,
20}

22#[derive(Subcommand)]
23enum Commands {
24    /// Initialize a new Flint project
25    Init {
26        /// Project name/directory
27        name: String,
28    },
29
30    /// Entity operations

34    /// Scene operations

36    Scene(scene::SceneCommands),

38    /// Query entities

40        /// Query string (e.g., "entities where archetype == 'door'")
41        query: String,
42
43        /// Path to scene file
44        #[arg(long)]

46
47        /// Output format (json or toml)
48        #[arg(long, default_value = "json")]
49        format: String,
50    },
51
52    /// Show schema information
53    Schema {

55        name: String,

147fn main() -> Result<()> {
148    tracing_subscriber::fmt()
149        .with_env_filter(
150            tracing_subscriber::EnvFilter::try_from_default_env()
151                .unwrap_or_else(|_| tracing_subscriber::EnvFilter::new("warn,rfd=off")),
152        )
153        .init();
154
155    let cli = Cli::parse();
156
157    match cli.command {
158        Commands::Init { name } => init::run(&name),
159        Commands::Entity(cmd) => entity::run(cmd),
160        Commands::Scene(cmd) => scene::run(cmd),
161        Commands::Query {
162            query,
163            scene,
164            format,
165        } => query::run(&query, scene.as_deref(), &format),
166        Commands::Schema { name, schemas } => schema::run(
167            &name,
168            &schemas.iter().map(|s| s.as_str()).collect::<Vec<_>>(),
169        ),
170        Commands::Validate(args) => validate::run(args),
171        Commands::ValidateSuite(args) => validate_suite::run(args),
172        Commands::PlaySuite(args) => play_suite::run(args),
173        Commands::Calibrate(args) => calibrate::run(args),
174        Commands::PlayChart(args) => play_chart::run(args),
175        Commands::SpikeRumble(args) => spike_rumble::run(args),
176        Commands::ReplayChart(args) => replay_chart::run(args),
177        Commands::RenderSuite(args) => render_suite::run(args),
178        Commands::Play(args) => play::run(args),
179        Commands::Prefab(cmd) => prefab::run(cmd),
180        Commands::Asset(cmd) => asset::run(cmd),
181        Commands::Edit(args) => edit_router::run(args),
182        Commands::SplineEdit(args) => spline_edit::run(args),
183        Commands::Preview(args) => preview::run(args),
184        Commands::Serve {
185            scene,
186            watch,
187            schemas,
188            no_inspector,
189        } => {
190            // Serve uses first schemas path (viewer doesn't need multi-dir yet)
191            let schemas_path = schemas.first().map(|s| s.as_str()).unwrap_or("schemas");
192            flint_viewer::app::run(&scene, watch, schemas_path, !no_inspector, false)
193        }
194        Commands::Gen(args) => gen::run(args),
195        Commands::GenPreview(args) => gen_preview::run(args),
196        Commands::TerrainEdit(args) => terrain_edit::run(args),
197        Commands::TexEdit(args) => tex_edit::run(args),
198        Commands::Render(args) => render::run(args),
199    }
200}
