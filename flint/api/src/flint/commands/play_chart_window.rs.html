1//! Windowed front end for `flint play-chart` — a bare wgpu window over the
2//! same [`ChartSession`] the console loop drives.
3//!
4//! Exists for two reasons. Practical: gamepad-to-keyboard mappers (Legion
5//! Space, Steam desktop layouts) inject keystrokes at the OS level, and a
6//! focused game window absorbs them harmlessly instead of letting them type
7//! into the terminal mid-session. Design: it carries the first wordless
8//! visual cues — the GDD forbids meters/icons/grades, so everything here is
9//! field and glow, nothing is a readout:
10//!
11//! - a warm glow drifts where the chart wants the lean (the "somewhere to
12//!   sway toward" that is inaudible in a stem-only build),
13//! - a cooler, smaller glow echoes the player's actual lean — togetherness
14//!   reads as the two lights merging, not as a score,
15//! - the whole field breathes with the beat and bar from the Conductor,
16//! - a pulse press blooms a brief expanding ring from the player's light,
17//! - low coherence drains saturation (foreshadowing the disintegration
18//!   language), high coherence lets the palette warm.
19//!
20//! Debug numbers stay on stdout, exactly as in console mode. Keys: `r`
21//! reloads the coherence config, `q`/Escape quits.
22

35pub fn run_windowed(session: ChartSession) -> Result<()> {
36    let event_loop = EventLoop::new().context("creating event loop")?;
37    event_loop.set_control_flow(ControlFlow::Poll);
38    let mut app = App {
39        session: Some(session),
40        gfx: None,
41        start: Instant::now(),
42        error: None,
43    };
44    event_loop.run_app(&mut app).context("event loop")?;
45    if let Some(err) = app.error.take() {
46        return Err(err);
47    }
48    match app.session.take() {
49        Some(session) => {
50            for n in session.finish().context("finishing chart session")? {
51                println!("{n}");
52            }
53            Ok(())
54        }
55        None => Ok(()),
56    }
57}

59struct App {
60    session: Option<ChartSession>,
61    gfx: Option<Gfx>,
62    start: Instant,
63    error: Option<anyhow::Error>,
64}
65
66impl App {
67    /// Run one session tick; on finish or error, leave the loop.
68    fn drive(&mut self, event_loop: &ActiveEventLoop) {
69        let Some(session) = &mut self.session else {
70            return;
71        };
72        match session.tick() {
73            Ok(out) => {
74                for n in &out.notices {
75                    println!("{n}");
76                }
77                if out.state == Tick::Finished {
78                    event_loop.exit();
79                }
80            }
81            Err(e) => {
82                self.error = Some(anyhow::Error::new(e).context("ticking chart session"));
83                event_loop.exit();
84            }
85        }
86    }
87}
88
89impl ApplicationHandler for App {
90    fn resumed(&mut self, event_loop: &ActiveEventLoop) {
91        if self.gfx.is_some() {
92            return;
93        }
94        let attrs = Window::default_attributes()
95            .with_title("star child — feel harness")
96            .with_inner_size(winit::dpi::LogicalSize::new(960.0, 960.0));
97        let window = match event_loop.create_window(attrs) {
98            Ok(w) => Arc::new(w),
99            Err(e) => {
100                self.error = Some(anyhow::anyhow!("creating window: {e}"));
101                event_loop.exit();
102                return;
103            }
104        };
105        match pollster::block_on(RenderContext::new(window.clone())) {
106            Ok(ctx) => self.gfx = Some(Gfx::new(window, ctx)),
107            Err(e) => {
108                self.error = Some(anyhow::anyhow!("initializing wgpu: {e}"));
109                event_loop.exit();
110            }
111        }
112    }
113
114    fn window_event(&mut self, event_loop: &ActiveEventLoop, _id: WindowId, event: WindowEvent) {
115        match event {
116            WindowEvent::CloseRequested => event_loop.exit(),
117            WindowEvent::Resized(size) => {
118                if let Some(gfx) = &mut self.gfx {
119                    gfx.ctx.resize(size);
120                }
121            }
122            WindowEvent::KeyboardInput { event, .. } => {
123                if event.state != ElementState::Pressed || event.repeat {
124                    return;
125                }
126                match event.physical_key {
127                    PhysicalKey::Code(KeyCode::Escape) | PhysicalKey::Code(KeyCode::KeyQ) => {
128                        println!("quit requested");
129                        event_loop.exit();
130                    }
131                    PhysicalKey::Code(KeyCode::KeyR) => {
132                        if let Some(session) = &mut self.session {
133                            match session.reload_config() {
134                                Ok(notices) => {
135                                    for n in notices {
136                                        println!("{n}");
137                                    }
138                                }
139                                Err(e) => {
140                                    self.error = Some(
141                                        anyhow::Error::new(e).context("reloading session configs"),
142                                    );
143                                    event_loop.exit();
144                                }
145                            }
146                        }
147                    }
148                    // Anything else is likely mapper spam; swallow it here —
149                    // that containment is half the point of this window.
150                    _ => {}
151                }
152            }
153            WindowEvent::RedrawRequested => {
154                let (Some(gfx), Some(session)) = (&mut self.gfx, &self.session) else {
155                    return;
156                };
157                let frame = session.visual_frame();
158                if let Err(e) = gfx.render(&frame, self.start.elapsed().as_secs_f32()) {
159                    self.error = Some(e);
160                    event_loop.exit();
161                }
162            }
163            _ => {}
164        }
165    }
166
167    fn about_to_wait(&mut self, event_loop: &ActiveEventLoop) {
168        // Vsync paces this loop (~monitor rate): present blocks, so ticking
169        // here drains queues often enough. The audio clock owns time.
170        self.drive(event_loop);
171        if let Some(gfx) = &self.gfx {
172            gfx.window.request_redraw();
173        }
174    }
175}

177/// 20 f32s, matching `Uni` in the WGSL below (five 16-byte rows).
178const UNIFORM_SIZE: u64 = 80;

180struct Gfx {
181    window: Arc<Window>,
182    ctx: RenderContext,
183    pipeline: wgpu::RenderPipeline,
184    uniforms: wgpu::Buffer,
185    bind_group: wgpu::BindGroup,
186}
187
188impl Gfx {
189    fn new(window: Arc<Window>, ctx: RenderContext) -> Self {
190        let device = &ctx.device;
191        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
192            label: Some("play-chart-window"),
193            source: wgpu::ShaderSource::Wgsl(SHADER.into()),
194        });
195        let uniforms = device.create_buffer(&wgpu::BufferDescriptor {
196            label: Some("play-chart-uniforms"),
197            size: UNIFORM_SIZE,
198            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
199            mapped_at_creation: false,
200        });
201        let bgl = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
202            label: Some("play-chart-bgl"),
203            entries: &[wgpu::BindGroupLayoutEntry {
204                binding: 0,
205                visibility: wgpu::ShaderStages::FRAGMENT,
206                ty: wgpu::BindingType::Buffer {
207                    ty: wgpu::BufferBindingType::Uniform,
208                    has_dynamic_offset: false,
209                    min_binding_size: wgpu::BufferSize::new(UNIFORM_SIZE),
210                },
211                count: None,
212            }],
213        });
214        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
215            label: Some("play-chart-bg"),
216            layout: &bgl,
217            entries: &[wgpu::BindGroupEntry {
218                binding: 0,
219                resource: uniforms.as_entire_binding(),
220            }],
221        });
222        let layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
223            label: Some("play-chart-pl"),
224            bind_group_layouts: &[&bgl],
225            push_constant_ranges: &[],
226        });
227        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
228            label: Some("play-chart-pipeline"),
229            layout: Some(&layout),
230            vertex: wgpu::VertexState {
231                module: &shader,
232                entry_point: Some("vs_main"),
233                buffers: &[],
234                compilation_options: Default::default(),
235            },
236            fragment: Some(wgpu::FragmentState {
237                module: &shader,
238                entry_point: Some("fs_main"),
239                targets: &[Some(wgpu::ColorTargetState {
240                    format: ctx.config.format,
241                    blend: Some(wgpu::BlendState::REPLACE),
242                    write_mask: wgpu::ColorWrites::ALL,
243                })],
244                compilation_options: Default::default(),
245            }),
246            primitive: wgpu::PrimitiveState::default(),
247            depth_stencil: None,
248            multisample: wgpu::MultisampleState::default(),
249            multiview: None,
250            cache: None,
251        });
252        Self {
253            window,
254            ctx,
255            pipeline,
256            uniforms,
257            bind_group,
258        }
259    }
260
261    fn render(
262        &mut self,
263        frame: &flint_music::chart_session::VisualFrame,
264        time_s: f32,
265    ) -> Result<()> {
266        let aspect = self.ctx.aspect_ratio();
267        let data: [f32; 20] = [
268            frame.lean[0],
269            frame.lean[1],
270            frame.target[0],
271            frame.target[1],
272            frame.beat_phase,
273            frame.bar_phase,
274            frame.coherence,
275            frame.pulse_age_s.min(1e3),
276            aspect,
277            if frame.preroll { 1.0 } else { 0.0 },
278            time_s,
279            frame.desaturate,
280            frame.blur,
281            frame.chromatic,
282            frame.reassembly,
283            if frame.no_input { 1.0 } else { 0.0 },
284            frame.rewind,
285            0.0,
286            0.0,
287            0.0,
288        ];
289        self.ctx
290            .queue
291            .write_buffer(&self.uniforms, 0, bytemuck::cast_slice(&data));
292
293        let surface = match self.ctx.surface.get_current_texture() {
294            Ok(s) => s,
295            Err(wgpu::SurfaceError::Lost | wgpu::SurfaceError::Outdated) => {
296                self.ctx.resize(self.ctx.size);
297                return Ok(()); // retry next frame
298            }
299            Err(e) => return Err(anyhow::anyhow!("acquiring surface: {e}")),
300        };
301        let view = surface
302            .texture
303            .create_view(&wgpu::TextureViewDescriptor::default());
304        let mut encoder = self
305            .ctx
306            .device
307            .create_command_encoder(&wgpu::CommandEncoderDescriptor {
308                label: Some("play-chart-encoder"),
309            });
310        {
311            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
312                label: Some("play-chart-pass"),
313                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
314                    view: &view,
315                    resolve_target: None,
316                    ops: wgpu::Operations {
317                        load: wgpu::LoadOp::Clear(wgpu::Color::BLACK),
318                        store: wgpu::StoreOp::Store,
319                    },
320                })],
321                depth_stencil_attachment: None,
322                timestamp_writes: None,
323                occlusion_query_set: None,
324            });
325            pass.set_pipeline(&self.pipeline);
326            pass.set_bind_group(0, &self.bind_group, &[]);
327            pass.draw(0..3, 0..1);
328        }
329        self.ctx.queue.submit([encoder.finish()]);
330        surface.present();
331        Ok(())
332    }
333}

335const SHADER: &str = r#"
336struct Uni {
337    lean: vec2<f32>,
338    tgt: vec2<f32>,
339    // beat_phase, bar_phase, coherence, pulse_age_s
340    misc0: vec4<f32>,
341    // aspect, preroll, time_s, desaturate
342    misc1: vec4<f32>,
343    // blur, chromatic, reassembly, no_input
344    misc2: vec4<f32>,
345    // rewind, _pad, _pad, _pad
346    misc3: vec4<f32>,
347};
348@group(0) @binding(0) var<uniform> u: Uni;
349
350struct VsOut {
351    @builtin(position) pos: vec4<f32>,
352    // Clip-space xy passed through: y is up, matching stick space.
353    @location(0) p: vec2<f32>,
354};
355
356@vertex
357fn vs_main(@builtin(vertex_index) vi: u32) -> VsOut {
358    var corners = array<vec2<f32>, 3>(
359        vec2(-1.0, -1.0), vec2(3.0, -1.0), vec2(-1.0, 3.0),
360    );
361    var out: VsOut;
362    out.pos = vec4(corners[vi], 0.0, 1.0);
363    out.p = corners[vi];
364    return out;
365}
366
367// The whole luminous field at one sample point. Factored out so chromatic
368// offset can evaluate it per channel at shifted positions — the image is
369// procedural, so "aberration" is re-rendering, not texture sampling.
370fn field(q: vec2<f32>) -> vec3<f32> {
371    let beat_phase = u.misc0.x;
372    let bar_phase = u.misc0.y;
373    let pulse_age = u.misc0.w;
374    let t = u.misc1.z;
375    let blur = u.misc2.x;
376
377    // Defocus: rung blur widens and dims every glow — light losing its
378    // edges, not an image losing pixels.
379    let soft = 1.0 + blur * 3.0;
380    let dim = 1.0 / (1.0 + blur * 0.8);
381
382    // Deep indigo ground with a slow vertical drift.
383    var col = mix(
384        vec3(0.030, 0.028, 0.070),
385        vec3(0.075, 0.050, 0.125),
386        0.5 + 0.5 * q.y + 0.06 * sin(t * 0.11),
387    );
388
389    // Breathing: a soft swell decaying through each beat, deeper each bar.
390    col += vec3(0.045) * exp(-5.0 * beat_phase) + vec3(0.05) * exp(-3.0 * bar_phase);
391
392    // The chart's lean target: a warm light to sway toward.
393    let dt = distance(q, u.tgt * 0.6);
394    col += vec3(0.85, 0.60, 0.36) * 0.34 * dim * exp(-dt * dt / (0.085 * soft));
395
396    // The player's own lean: a cooler, closer light. Togetherness reads as
397    // the two lights merging and brightening — never as a number.
398    let dp = distance(q, u.lean * 0.6);
399    col += vec3(0.34, 0.52, 0.85) * 0.30 * dim * exp(-dp * dp / (0.032 * soft));
400
401    // Pulse: a ring blooms out from the player's light and fades.
402    if (pulse_age < 1.4) {
403        let r = pulse_age * 0.85;
404        let d = distance(q, u.lean * 0.6) - r;
405        let ring = exp(-d * d / (0.0016 * soft)) * exp(-2.6 * pulse_age);
406        col += vec3(0.90, 0.84, 0.68) * 0.45 * dim * ring;
407    }
408    return col;
409}

411@fragment
412fn fs_main(in: VsOut) -> @location(0) vec4<f32> {
413    let coherence = u.misc0.z;
414    let aspect = u.misc1.x;
415    let preroll = u.misc1.y;
416    let t = u.misc1.z;
417    let desat = u.misc1.w;
418    let blur = u.misc2.x;
419    let chroma = u.misc2.y;
420    let reassembly = u.misc2.z;
421
422    var p = vec2(in.p.x * aspect, in.p.y);
423
424    // Rewind interlude: the whole field winds backwards — an accelerating
425    // counter-rotation with a gentle inward pull, the visual half of the
426    // record spinning back to the re-entry point.
427    let rewind = u.misc3.x;
428    if (rewind > 0.0) {
429        let spin = -rewind * rewind * 2.4;
430        let c = cos(spin);
431        let s = sin(spin);
432        p = vec2(p.x * c - p.y * s, p.x * s + p.y * c) * (1.0 + 0.18 * rewind);
433    }
434
435    // Woozy: the whole field swims a little, always — more as focus goes.
436    let wob = vec2(
437        sin(t * 0.31 + p.y * 1.7),
438        cos(t * 0.23 + p.x * 1.9),
439    ) * (0.025 + 0.035 * blur);
440    let q = p + wob;
441
442    // Chromatic wobble: color planes drift apart radially, breathing slowly
443    // so it reads as instability, not a filter.
444    var col: vec3<f32>;
445    let ca = chroma * 0.05 * (0.7 + 0.3 * sin(t * 0.9));
446    if (ca > 0.0005) {
447        col = vec3(
448            field(q * (1.0 + ca)).r,
449            field(q).g,
450            field(q * (1.0 - ca)).b,
451        );
452    } else {
453        col = field(q);
454    }
455
456    // Coherence drains or restores the world's color — the disintegration
457    // language in miniature. Never fully grey: the field stays alive.
458    var grey = vec3(dot(col, vec3(0.299, 0.587, 0.114)));
459    col = mix(grey * 0.72, col, 0.30 + 0.70 * coherence);
460
461    // Ladder desaturation stacks on top: the rungs pull further toward ash.
462    grey = vec3(dot(col, vec3(0.299, 0.587, 0.114)));
463    col = mix(col, grey * 0.62, desat);
464
465    // Reassembly: while the world re-gathers (progress < 1), a warm veil —
466    // home's light — suffuses everything and recedes as the field returns.
467    if (reassembly < 1.0) {
468        let gather = 1.0 - reassembly;
469        col += vec3(0.30, 0.20, 0.10) * gather * (0.55 + 0.15 * sin(t * 1.7));
470    }
471
472    // During the rewind the world also dims and cools toward the seam —
473    // breath out before the breath in. In the final stretch (the pickup),
474    // a warm glow gathers at the target: anticipation of the downbeat.
475    if (rewind > 0.0) {
476        col *= 1.0 - 0.45 * rewind;
477        col += vec3(0.04, 0.05, 0.10) * rewind;
478        let pickup = smoothstep(0.55, 1.0, rewind);
479        let dt2 = distance(q, u.tgt * 0.6);
480        col += vec3(0.85, 0.60, 0.36) * 0.4 * pickup * exp(-dt2 * dt2 / 0.12);
481    }
482
483    // Pre-roll: lights low until the suite begins.
484    col *= mix(1.0, 0.35, preroll);
485
486    // Debug cue (harness only): capture alive but silent — the controller
487    // is delivering nothing. A slow red breath at the frame edge, impossible
488    // to mistake for a mood.
489    let no_input = u.misc2.w;
490    if (no_input > 0.5) {
491        let edge = smoothstep(0.75, 1.0, max(abs(in.p.x), abs(in.p.y)));
492        col = mix(col, vec3(0.55, 0.08, 0.06), edge * (0.55 + 0.35 * sin(t * 2.2)));
493    }
494
495    return vec4(col, 1.0);
496}
