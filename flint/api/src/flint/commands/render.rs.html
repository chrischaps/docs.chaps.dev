440/// `--list-styles [--json]`: the registry as text or JSON, with the user
441/// styles beside the scene when one is given.
442fn list_styles(args: &RenderArgs) -> Result<()> {
443    use flint_render::stylize::{catalog, custom};
444    if !args.scene.is_empty() {
445        let scene_dir = Path::new(&args.scene).parent().unwrap_or(Path::new("."));
446        for message in custom::load_dir(scene_dir) {
447            eprintln!("warning: {message}");
448        }
449    }
450    if args.json {
451        println!("{}", serde_json::to_string_pretty(&catalog::catalog())?);
452    } else {
453        print!("{}", catalog::catalog_text());
454    }
455    Ok(())
456}

462pub fn run(args: RenderArgs) -> Result<()> {
463    if args.list_gpus {
464        for a in flint_render::list_adapters() {
465            println!("{a}");
466        }
467        return Ok(());
468    }
469    if args.list_styles {
470        return list_styles(&args);
471    }
472    if args.look.as_deref() == Some("help") {
473        print!("{}", super::look::help(&args.scene));
474        return Ok(());
475    }
476    if args.scene.is_empty() {
477        anyhow::bail!("a scene is required (flint render <scene.toml>)");
478    }
479    // User-authored styles (#274): registered before anything reads a
480    // `[post_process.styles]` table, a look or a --style that names one
481    let scene_dir = Path::new(&args.scene)
482        .parent()
483        .unwrap_or(Path::new("."))
484        .to_path_buf();
485    for message in flint_render::stylize::custom::load_dir(&scene_dir) {
486        eprintln!("warning: {message}");
487    }
488    args.style
489        .check()
490        .map_err(|e| anyhow::anyhow!("--style: {e}"))?;
491    let look = args
492        .look
493        .as_deref()
494        .map(|name| super::look::find(name, &args.scene))
495        .transpose()?;
496    let (width, height) = match args.sweep {
497        Some(_) => (args.width.unwrap_or(640), args.height.unwrap_or(360)),
498        None => (args.width.unwrap_or(1920), args.height.unwrap_or(1080)),
499    };
500    flint_render::set_gpu_override(args.gpu.clone());
501    // Merge explicit schemas with auto-discovered dirs from scene path
502    let mut all_schemas = args.schemas.clone();
503    for dir in flint_schema::discover_schema_dirs(&args.scene) {
504        let s = dir.to_string_lossy().into_owned();
505        if !all_schemas.contains(&s) {
506            all_schemas.push(s);
507        }
508    }
509
510    // Load schemas from all directories
511    let existing: Vec<&str> = all_schemas
512        .iter()
513        .map(|s| s.as_str())
514        .filter(|p| Path::new(p).exists())
515        .collect();
516    let registry = if !existing.is_empty() {
517        SchemaRegistry::load_from_directories(&existing).context("Failed to load schemas")?
518    } else {
519        println!("Warning: No schemas directories found");
520        SchemaRegistry::new()
521    };
522
523    // Load scene
524    let (mut world, scene_file) =
525        load_scene(&args.scene, &registry).context("Failed to load scene")?;
526    println!("Loaded scene: {}", scene_file.scene.name);
527    println!("Entities: {}", world.entity_count());
528
529    // Create headless context
530    // Every column's post-processing, worked out (and checked) before any
531    // GPU work: the scene's block, under the look, under the column.
532    let scene_pp = scene_file.post_process.as_ref();
533    let base_pp = super::look::base_def(scene_pp, look.as_ref())?;
534    let mut configs = Vec::new();
535    for i in 0..args.sweep.as_ref().map_or(1, |s| s.values.len()) {
536        let def = match &args.sweep {
537            Some(sweep) => sweep.column_def(i, scene_pp, look.as_ref(), &args.scene)?,
538            None => base_pp.clone(),
539        };
540        let mut config = post_process_config(&args, def.as_ref())?;
541        if let Some(spec) = args.sweep.as_ref().and_then(|s| s.column_style(i)) {
542            flint_render::stylize::apply_style_spec(&mut config, &spec)
543                .map_err(|e| anyhow::anyhow!("--sweep: {e}"))?;
544        }
545        configs.push(config);
546    }
547    if let Some(look) = &look {
548        println!("Look: {} ({})", look.name, look.source);
549    }
550
551    let ctx = pollster::block_on(HeadlessContext::new(width, height))
552        .context("Failed to create headless render context")?;
553
554    // Configure camera — scene-level first, then CLI overrides take precedence
555    let mut camera = Camera::new();
556    camera.aspect = ctx.aspect_ratio();
557
558    // Apply scene-level camera configuration
559    let mut scene_set_position = false;
560    if let Some(cam_def) = &scene_file.camera {
561        if cam_def.projection == "orthographic" {
562            camera.orthographic = true;
563            if cam_def.ortho_height > 0.0 {
564                camera.ortho_height = cam_def.ortho_height;
565            }
566        }
567        if let Some(pos) = cam_def.position {
568            camera.position = Vec3::new(pos[0], pos[1], pos[2]);
569            scene_set_position = true;
570        }
571        if let Some(target) = cam_def.target {
572            camera.target = Vec3::new(target[0], target[1], target[2]);
573        }
574        if let Some(fov) = cam_def.fov {
575            camera.fov = fov;
576        }
577        if let Some(near) = cam_def.near {
578            camera.near = near;
579        }
580        if let Some(far) = cam_def.far {
581            camera.far = far;
582        }
583    }
584
585    // Derive orbit parameters from position/target so update_orbit() is consistent
586    if scene_set_position {
587        let dir = camera.position - camera.target;
588        camera.distance = dir.length();
589        if camera.distance > 0.001 {
590            let n = dir * (1.0 / camera.distance);
591            camera.pitch = n.y.asin();
592            camera.yaw = n.x.atan2(n.z);
593        }
594    }
595
596    // CLI overrides take precedence
597    if let Some(d) = args.distance {
598        camera.distance = d;
599    }
600    if let Some(y) = args.yaw {
601        camera.yaw = y.to_radians();
602    }
603    if let Some(p) = args.pitch {
604        camera.pitch = p.to_radians();
605    }
606    if let Some(t) = args.target {
607        camera.target = Vec3::new(t[0], t[1], t[2]);
608    }
609    if let Some(f) = args.fov {
610        camera.fov = f;
611    }
612    camera.update_orbit();
613
614    // Create scene renderer
615    let mut renderer = SceneRenderer::new_headless(
616        &ctx.device,
617        &ctx.queue,
618        ctx.format,
619        ctx.width,
620        ctx.height,
621        RendererConfig {
622            show_grid: !args.no_grid,
623            sample_count: args.msaa,
624        },
625    );
626
627    // Scene-authored hemisphere ambient + diffuse wrap (absent = renderer default)
628    if let Some(env) = &scene_file.environment {
629        if env.ambient_sky.is_some() || env.ambient_ground.is_some() {
630            renderer.set_ambient(
631                env.ambient_sky
632                    .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_SKY),
633                env.ambient_ground
634                    .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_GROUND),
635            );
636        }
637        if let Some(wrap) = env.diffuse_wrap {
638            renderer.set_diffuse_wrap(wrap);
639        }
640        if let Some(oren) = env.oren_nayar {
641            renderer.set_oren_nayar(oren);
642        }
643        if let Some(strength) = env.sheen_strength {
644            renderer.set_sheen(env.sheen_color.unwrap_or([1.0; 3]), strength);
645        }
646    }
647    // CLI overrides win over scene-authored values
648    if let Some(oren) = args.oren_nayar {
649        renderer.set_oren_nayar(oren);
650    }
651    if let Some(strength) = args.sheen_strength {
652        renderer.set_sheen(args.sheen_color.unwrap_or([1.0; 3]), strength);
653    }
654
655    // Load models and textures from the scene
656    let config = ModelLoadConfig::from_scene_path(&args.scene);
657    model_loader::load_models_from_world(
658        &mut world,
659        &mut renderer,
660        &ctx.device,
661        &ctx.queue,
662        &config,
663    );
664
665    // Generate procedural geometry from spline + spline_mesh entities
666    spline_gen::load_splines(
667        &args.scene,
668        &mut world,
669        &mut renderer,
670        None,
671        &ctx.device,
672        &ctx.queue,
673    );
674
675    // Apply debug state
676    if let Some(mode_str) = &args.debug_mode {
677        let mode = match mode_str.as_str() {
678            "wireframe-overlay" => DebugMode::WireframeOverlay,
679            "wireframe" => DebugMode::WireframeOnly,
680            "normals" => DebugMode::Normals,
681            "depth" => DebugMode::Depth,
682            "uv" => DebugMode::UvChecker,
683            "unlit" => DebugMode::Unlit,
684            "metalrough" => DebugMode::MetallicRoughness,
685            "style" => DebugMode::StyleMask,
686            _ => DebugMode::Pbr,
687        };
688        if mode == DebugMode::StyleMask && args.no_postprocess {
689            eprintln!(
690                "warning: --debug-mode style draws through post-processing, \
691                 which --no-postprocess turns off; the image shows no mask"
692            );
693        }
694        renderer.set_debug_mode(mode);
695    }
696    if args.wireframe_overlay {
697        renderer.set_debug_mode(DebugMode::WireframeOverlay);
698    }
699    if args.show_normals {
700        renderer.toggle_normal_arrows();
701    }
702    if args.no_tonemapping {
703        renderer.set_tonemapping(false);
704    }
705    if args.no_shadows {
706        renderer.set_shadows(false);
707    }
708    // The renderer constructs at 2048; only recreate the pass when the flag
709    // actually deviates. Historically the flag defaulted to 1024 but the
710    // != 1024 guard meant an explicit 1024 was silently ignored — with the
711    // texel size now uploaded per-resolution (ADR 0049) the flag is real.
712    if args.shadow_resolution != 2048 {
713        renderer.set_shadow_resolution(&ctx.device, args.shadow_resolution);
714    }
715
716    // Post-processing: the first column's (the only one without --sweep)
717    renderer.set_post_process_config(configs[0].clone());
718    renderer.ensure_style_resources(&ctx.device);
719    renderer.ensure_fxaa_resources(&ctx.device);
720
721    if let Some(scale) = args.render_scale {
722        renderer.set_render_scale(&ctx.device, scale);
723    }
724
725    // Quality settings over the flags above: the renderer's own state is the
726    // starting point, so `--quality` names only what it changes.
727    if let Some(quality) = &args.quality {
728        let mut q = flint_render::QualitySettings {
729            render_scale: renderer.render_scale(),
730            msaa: renderer.msaa(),
731            shadows: renderer.shadows_enabled(),
732            shadow_resolution: renderer.shadow_resolution(),
733            ..Default::default()
734        };
735        if let Some(e) = q.apply_str(quality).first() {
736            anyhow::bail!("--quality: {e}");
737        }
738        let applied = renderer.set_quality(&ctx.device, &q);
739        println!("Quality: {}", applied.to_query());
740    }
741    if let Some(px) = args.pixels_per_point {
742        renderer.pixels_per_point = px;
743    }
744    // Post time for grain/mode animation and animated styles. Headless
745    // renders leave this 0.0 so PNG snapshots stay deterministic;
746    // --style-time (or the older --grain-time) pins an explicit frame.
747    if let Some(t) = args.style.style_time.or(args.grain_time) {
748        renderer.ocean_time = t as f64;
749    }
750
751    // Load terrain (if any)
752    load_terrain_for_render(&world, &args.scene, &ctx.device, &ctx.queue, &mut renderer);
753
754    // Set scene_dir for font/texture resolution and viewport params for screen anchoring
755    renderer.scene_dir = Path::new(&args.scene).parent().map(|p| p.to_path_buf());
756    renderer.ortho_height = camera.ortho_height;
757    renderer.aspect_ratio = camera.aspect;
758
759    // Two-bone IK (ADR 0070): pose authored chains before the single frame.
760    flint_animation::IkPass::new().run(&mut world);
761
762    renderer.update_from_world(&world, &ctx.device);
763
764    // Particles (ADR 0068): fixed-step, fixed-seed simulation so a snapshot
765    // at a given --particle-time is reproducible run to run.
766    if let Some(seconds) = args.particle_time {
767        use flint_runtime::RuntimeSystem;
768        let mut particles = flint_particles::ParticleSystem::new();
769        flint_particles::load_particle_effects_from_world(&args.scene, &mut particles);
770        particles
771            .initialize(&mut world)
772            .context("Failed to initialize particles")?;
773        let dirs = flint_particles::texture_search_dirs(&args.scene);
774        flint_render::load_particle_textures(
775            &mut renderer,
776            &ctx.device,
777            &ctx.queue,
778            &particles.sync,
779            &dirs,
780        );
781        particles.simulate_to(&world, seconds.max(0.0), 1.0 / 60.0);
782        particles.pack(Some(camera.position_array()));
783        renderer.update_particles_from(&ctx.device, &ctx.queue, &particles.sync);
784        println!(
785            "Particles: {} alive across {} emitter(s) at t = {:.2}s",
786            particles.sync.total_alive(),
787            particles.sync.emitter_count(),
788            seconds
789        );
790    }
791
792    // GPU timing: the same frame many times, each pass's time smoothed
793    if let Some(frames) = args.gpu_time {
794        if !renderer.set_gpu_timing(&ctx.device, &ctx.queue, true) {
795            anyhow::bail!("--gpu-time: this adapter has no TIMESTAMP_QUERY");
796        }
797        for _ in 0..frames.max(1) {
798            let mut encoder = ctx
799                .device
800                .create_command_encoder(&wgpu::CommandEncoderDescriptor { label: None });
801            renderer.encode_to(
802                &ctx.device,
803                &ctx.queue,
804                &mut encoder,
805                &ctx.depth_view,
806                &camera,
807                &ctx.color_view,
808            );
809            renderer.encode_gpu_timings(&ctx.device, &mut encoder);
810            ctx.queue.submit(std::iter::once(encoder.finish()));
811            renderer.frame_submitted();
812            ctx.device.poll(wgpu::Maintain::Wait);
813        }
814        let stats = renderer.collect_stats();
815        let (sw, sh) = renderer.scene_size();
816        println!(
817            "GPU time over {} frames at {}x{} (scene {}x{}):",
818            frames, width, height, sw, sh
819        );
820        for pass in &stats.gpu_passes {
821            println!("  {:<12} {:>7.3} ms", pass.label, pass.ms);
822        }
823        println!("  {:<12} {:>7.3} ms", "frame", stats.gpu_frame_ms);
824    }
825
826    // Render each column (one without --sweep), reading each back
827    let mut frames = Vec::with_capacity(configs.len());
828    for (i, config) in configs.iter().enumerate() {
829        if i > 0 {
830            renderer.set_post_process_config(config.clone());
831            renderer.ensure_style_resources(&ctx.device);
832            renderer.ensure_fxaa_resources(&ctx.device);
833        }
834        renderer.render_to(
835            &ctx.device,
836            &ctx.queue,
837            &ctx.depth_view,
838            &camera,
839            &ctx.color_view,
840        );
841        let pixels =
842            pollster::block_on(ctx.read_pixels()).context("Failed to read rendered pixels")?;
843        let mut img = image::RgbaImage::from_raw(width, height, pixels)
844            .context("Failed to create image from pixel data")?;
845        // The style mask view says which hue is which style (#309): in the
846        // terminal once, and on every frame's corner
847        if renderer.debug_state().mode == DebugMode::StyleMask && !args.no_postprocess {
848            let ids = renderer.style_ids();
849            if i == 0 {
850                println!("{}", flint_render::style_mask::legend_line(ids));
851            }
852            super::style_sheet::draw_legend(&mut img, &flint_render::style_mask::legend(ids));
853        }
854        if let Some(sweep) = &args.sweep {
855            println!("  {} = {}", sweep.name(), sweep.values[i]);
856        }
857        frames.push(img);
858    }
859
860    let img = match &args.sweep {
861        None => frames.swap_remove(0),
862        Some(sweep) => {
863            let over = match &look {
864                Some(look) => format!("  over {}", look.name),
865                None => String::new(),
866            };
867            super::style_sheet::compose(
868                &frames,
869                &super::style_sheet::SheetText {
870                    title: scene_file.scene.name.clone(),
871                    subtitle: format!("sweep {}{over}", sweep.name()),
872                    labels: sweep.values.clone(),
873                },
874            )
875        }
876    };
877    img.save(&args.output)
878        .context(format!("Failed to save image to {}", args.output))?;
879
880    println!(
881        "Rendered {}x{} image to {}",
882        img.width(),
883        img.height(),
884        args.output
885    );
886
887    Ok(())
888}

890/// `def` (the scene's block under any look and sweep column, or none) as
891/// the renderer's config, with the CLI's post flags and `--style` words
892/// over it.
893fn post_process_config(
894    args: &RenderArgs,
895    def: Option<&flint_scene::PostProcessDef>,
896) -> Result<flint_render::PostProcessConfig> {
897    {
898        let mut pp_config = if let Some(pp_def) = def {
899            flint_player::post_process_config_from_def(pp_def)
900        } else {
901            flint_render::PostProcessConfig::default()
902        };
903
904        // CLI overrides
905        if args.no_postprocess {
906            pp_config.enabled = false;
907        }
908        if let Some(intensity) = args.bloom_intensity {
909            pp_config.bloom_intensity = intensity;
910            pp_config.bloom_enabled = true;
911        }
912        if let Some(threshold) = args.bloom_threshold {
913            pp_config.bloom_threshold = threshold;
914        }
915        if let Some(exposure) = args.exposure {
916            pp_config.exposure = exposure;
917        }
918        if let Some(radius) = args.ssao_radius {
919            pp_config.ssao_radius = radius;
920        }
921        if let Some(intensity) = args.ssao_intensity {
922            pp_config.ssao_intensity = intensity;
923            if intensity <= 0.0 {
924                pp_config.ssao_enabled = false;
925            }
926        }
927        if let Some(samples) = args.ssao_samples {
928            pp_config.ssao_samples = samples.clamp(1, 64);
929        }
930        if let Some(density) = args.fog_density {
931            pp_config.fog_density = density;
932            pp_config.fog_enabled = density > 0.0;
933        }
934        if let Some(color) = args.fog_color {
935            pp_config.fog_color = color;
936        }
937        if let Some(falloff) = args.fog_height_falloff {
938            pp_config.fog_height_falloff = falloff;
939            pp_config.fog_height_enabled = true;
940        }
941        if let Some(intensity) = args.dither_intensity {
942            pp_config.dither_intensity = intensity;
943            pp_config.dither_enabled = intensity > 0.0;
944        }
945        if let Some(desaturate) = args.desaturate {
946            pp_config.desaturate = desaturate;
947        }
948        if let Some(strength) = args.dof {
949            pp_config.dof_strength = strength;
950        }
951        if let Some(distance) = args.dof_focus {
952            pp_config.dof_focus_distance = distance;
953        }
954        if let Some(range) = args.dof_range {
955            pp_config.dof_focus_range = range;
956        }
957        if let Some(mode) = args.render_mode {
958            pp_config.render_mode = mode;
959            // A mode with no explicit mix defaults to fully torn through.
960            pp_config.mode_mix = args.mode_mix.unwrap_or(1.0);
961        }
962        if let Some(mix) = args.mode_mix {
963            pp_config.mode_mix = mix;
964        }
965        if let Some(params) = args.mode_params {
966            pp_config.mode_params = params;
967        }
968        if let Some(density) = args.volumetric_density {
969            pp_config.volumetric_density = density;
970            pp_config.volumetric_enabled = density > 0.0;
971        }
972        if let Some(samples) = args.volumetric_samples {
973            pp_config.volumetric_samples = samples;
974        }
975        if let Some(radius) = args.kuwahara_radius {
976            pp_config.kuwahara_radius = radius;
977            pp_config.kuwahara_enabled = true;
978        }
979        if let Some(sharpness) = args.kuwahara_sharpness {
980            pp_config.kuwahara_sharpness = sharpness;
981            pp_config.kuwahara_enabled = true;
982        }
983        if let Some(hardness) = args.kuwahara_hardness {
984            pp_config.kuwahara_hardness = hardness;
985            pp_config.kuwahara_enabled = true;
986        }
987        if let Some(anisotropy) = args.kuwahara_anisotropy {
988            pp_config.kuwahara_anisotropy = anisotropy;
989            pp_config.kuwahara_enabled = true;
990        }
991        if let Some(depth) = args.edge_depth {
992            pp_config.edge_depth = depth;
993        }
994        if let Some(normal) = args.edge_normal {
995            pp_config.edge_normal = normal;
996        }
997        if let Some(grain) = args.film_grain {
998            pp_config.film_grain = grain;
999        }
1000        if let Some(lift) = args.grade_lift {
1001            pp_config.grade_lift = lift;
1002        }
1003        if let Some(gamma) = args.grade_gamma {
1004            pp_config.grade_gamma = gamma;
1005        }
1006        if let Some(gain) = args.grade_gain {
1007            pp_config.grade_gain = gain;
1008        }
1009        if args.fxaa {
1010            pp_config.fxaa_enabled = true;
1011        }
1012        apply_ascii_style_flags(args, &mut pp_config)?;
1013        // `--style name[:k=v,...]` (#271) last: an explicit word wins
1014        args.style.apply(&mut pp_config);
1015
1016        Ok(pp_config)
1017    }
1018}

1020/// The ASCII style's flags (#265) onto `[post_process.styles.ascii]`.
1021/// `--ascii`, or any flag that sets the look (font, ramp, colour, scope,
1022/// edges...), turns the style on; the tuning flags (`--ascii-cell`,
1023/// `--ascii-levels`, `--ascii-footprint`, `--ascii-no-edges`,
1024/// `--ascii-edge-depth`, `--ascii-edge-normal`) tune it once it's on,
1025/// from the flag or the scene.
1026fn apply_ascii_style_flags(
1027    args: &RenderArgs,
1028    pp: &mut flint_render::PostProcessConfig,
1029) -> Result<()> {
1030    const NAME: &str = "ascii";
1031    let style_only = args.ascii_font.is_some()
1032        || args.ascii_ramp.is_some()
1033        || args.ascii_color.is_some()
1034        || args.ascii_tint.is_some()
1035        || args.ascii_background.is_some()
1036        || args.ascii_brightness.is_some()
1037        || args.ascii_contrast.is_some()
1038        || args.ascii_ink.is_some()
1039        || args.ascii_palette.is_some()
1040        || args.ascii_ground.is_some()
1041        || args.ascii_glow.is_some()
1042        || args.ascii_edges.is_some()
1043        || args.ascii_edge_threshold.is_some()
1044        || args.ascii_edge_max.is_some()
1045        || args.ascii_dither.is_some()
1046        || args.ascii_exposure.is_some()
1047        || args.ascii_normalize.is_some()
1048        || args.ascii_scope.is_some()
1049        || args.ascii_grid.is_some();
1050    if args.ascii || style_only {
1051        pp.styles.set_enabled(NAME, true);
1052    }
1053    if !pp.styles.is_enabled(NAME) {
1054        return Ok(());
1055    }
1056    let styles = &mut pp.styles;
1057    if let Some(font) = &args.ascii_font {
1058        styles.set_option(NAME, "font", font);
1059    }
1060    if let Some(ramp) = &args.ascii_ramp {
1061        styles.set_option(NAME, "ramp", ramp);
1062    }
1063    if let Some(scope) = &args.ascii_scope {
1064        styles.set_option(NAME, "scope", scope);
1065    }
1066    if let Some(grid) = &args.ascii_grid {
1067        styles.set_option(NAME, "grid", grid);
1068    }
1069    let choices = |option: &str| {
1070        flint_render::stylize::find(NAME)
1071            .and_then(|d| d.options.iter().find(|o| o.name == option))
1072            .map_or(&[][..], |o| o.choices)
1073    };
1074    if let Some(color) = &args.ascii_color {
1075        let modes = choices("color");
1076        if !modes.contains(&color.as_str()) {
1077            anyhow::bail!(
1078                "--ascii-color: unknown mode '{color}' ({})",
1079                modes.join(", ")
1080            );
1081        }
1082        styles.set_option(NAME, "color", color);
1083    }
1084    if let Some(ground) = &args.ascii_ground {
1085        let modes = choices("ground");
1086        if !modes.contains(&ground.as_str()) {
1087            anyhow::bail!(
1088                "--ascii-ground: unknown mode '{ground}' ({})",
1089                modes.join(", ")
1090            );
1091        }
1092        styles.set_option(NAME, "ground", ground);
1093    }
1094    if let Some(palette) = &args.ascii_palette {
1095        // Any name goes (a file beside the scene); the style warns once
1096        // and falls back when it can't be found
1097        styles.set_option(NAME, "palette", palette);
1098        if args.ascii_color.is_none() {
1099            styles.set_option(NAME, "color", "palette");
1100        }
1101    }
1102    if let Some(rgb) = args.ascii_tint {
1103        styles.set_color(NAME, "tint", rgb);
1104        if args.ascii_color.is_none() {
1105            styles.set_option(NAME, "color", "tint");
1106        }
1107    }
1108    for (param, value) in [
1109        ("cell", args.ascii_cell),
1110        ("levels", args.ascii_levels),
1111        ("footprint", args.ascii_footprint),
1112        ("background", args.ascii_background),
1113        ("brightness", args.ascii_brightness),
1114        ("contrast", args.ascii_contrast),
1115        ("ink", args.ascii_ink),
1116        ("glow", args.ascii_glow),
1117        ("edges", args.ascii_edges),
1118        ("edge_threshold", args.ascii_edge_threshold),
1119        ("edge_max", args.ascii_edge_max),
1120        ("edge_depth", args.ascii_edge_depth),
1121        ("edge_normal", args.ascii_edge_normal),
1122        ("dither", args.ascii_dither),
1123        ("exposure", args.ascii_exposure),
1124        ("normalize", args.ascii_normalize),
1125    ] {
1126        if let Some(value) = value {
1127            styles.set_param(NAME, param, value);
1128        }
1129    }
1130    if args.ascii_no_edges {
1131        styles.set_param(NAME, "edges", 0.0);
1132    }
1133    Ok(())
1134}

1136/// Load terrain from world entities for headless rendering (no physics).
1137fn load_terrain_for_render(
1138    world: &flint_ecs::FlintWorld,
1139    scene_path: &str,
1140    device: &wgpu::Device,
1141    queue: &wgpu::Queue,
1142    renderer: &mut SceneRenderer,
1143) {
1144    use flint_core::Transform;
1145    use flint_terrain::{Heightmap, Terrain, TerrainConfig};
1146
1147    let scene_dir = Path::new(scene_path)
1148        .parent()
1149        .unwrap_or_else(|| Path::new("."));
1150
1151    for entity in world.all_entities() {
1152        let terrain_comp = match world.get_component(entity.id, comp::TERRAIN) {
1153            Some(c) => c,
1154            None => continue,
1155        };
1156
1157        let heightmap_rel = match terrain_comp.get("heightmap").and_then(|v| v.as_str()) {
1158            Some(s) if !s.is_empty() => s.to_string(),
1159            _ => continue,
1160        };
1161
1162        let hm_path = {
1163            let p = scene_dir.join(&heightmap_rel);
1164            if p.exists() {
1165                p
1166            } else if let Some(parent) = scene_dir.parent() {
1167                let pp = parent.join(&heightmap_rel);
1168                if pp.exists() {
1169                    pp
1170                } else {
1171                    p
1172                }
1173            } else {
1174                p
1175            }
1176        };
1177
1178        let heightmap = match Heightmap::from_png(&hm_path) {
1179            Ok(hm) => hm,
1180            Err(e) => {
1181                tracing::warn!("[terrain] Failed to load heightmap: {}", e);
1182                continue;
1183            }
1184        };
1185
1186        let get_f32 = |key: &str, default: f32| -> f32 {
1187            terrain_comp.get(key).and_then(toml_f32).unwrap_or(default)
1188        };
1189
1190        let get_i32 = |key: &str, default: i32| -> i32 {
1191            terrain_comp
1192                .get(key)
1193                .and_then(|v| v.as_integer())
1194                .map(|i| i as i32)
1195                .unwrap_or(default)
1196        };
1197
1198        let get_str = |key: &str| -> String {
1199            terrain_comp
1200                .get(key)
1201                .and_then(|v| v.as_str())
1202                .unwrap_or("")
1203                .to_string()
1204        };
1205
1206        let config = TerrainConfig {
1207            heightmap_path: heightmap_rel,
1208            width: get_f32("width", 256.0),
1209            depth: get_f32("depth", 256.0),
1210            height_scale: get_f32("height_scale", 50.0),
1211            chunk_resolution: get_i32("chunk_resolution", 64) as u32,
1212            texture_tile: get_f32("texture_tile", 16.0),
1213            splat_map_path: get_str("splat_map"),
1214            layer_textures: [
1215                get_str("layer0_texture"),
1216                get_str("layer1_texture"),
1217                get_str("layer2_texture"),
1218                get_str("layer3_texture"),
1219            ],
1220            metallic: get_f32("metallic", 0.0),
1221            roughness: get_f32("roughness", 0.85),
1222            grass: None,
1223        };
1224
1225        let terrain = Terrain::generate(&heightmap, &config);
1226
1227        let transform = world
1228            .get_component(entity.id, comp::TRANSFORM)
1229            .and_then(|t| {
1230                let pos = t.get("position").and_then(toml_vec3)?;
1231                Some(Transform {
1232                    position: Vec3::new(pos[0], pos[1], pos[2]),
1233                    ..Default::default()
1234                })
1235            })
1236            .unwrap_or_default();
1237
1238        renderer.load_terrain(
1239            device,
1240            queue,
1241            &terrain.chunks,
1242            &transform,
1243            config.texture_tile,
1244            config.metallic,
1245            config.roughness,
1246            &config.splat_map_path,
1247            &config.layer_textures,
1248            scene_dir,
1249        );
1250        // A style_mask on the terrain entity masks its chunks (#310)
1251        renderer.set_terrain_entity(Some(entity.id));
1252
1253        // Load grass if enabled
1254        let grass_config = flint_terrain::GrassConfig::from_component(terrain_comp, false);
1255
1256        if grass_config.enabled {
1257            let hm_data = heightmap.clone_heights();
1258            let hm_w = heightmap.width;
1259            let hm_d = heightmap.depth;
1260
1261            let splat_path = {
1262                let p = scene_dir.join(&config.splat_map_path);
1263                if p.exists() {
1264                    p
1265                } else {
1266                    scene_dir
1267                        .parent()
1268                        .map(|pp| pp.join(&config.splat_map_path))
1269                        .filter(|pp| pp.exists())
1270                        .unwrap_or(p)
1271                }
1272            };
1273
1274            let offset = [
1275                transform.position.x,
1276                transform.position.y,
1277                transform.position.z,
1278            ];
1279
1280            if let Ok(splat_img) = image::open(&splat_path) {
1281                let splat_rgba = splat_img.to_rgba8();
1282                let (sw, sh) = splat_rgba.dimensions();
1283
1284                renderer.load_grass(
1285                    device,
1286                    queue,
1287                    &grass_config,
1288                    &hm_data,
1289                    hm_w,
1290                    hm_d,
1291                    splat_rgba.as_raw(),
1292                    sw,
1293                    sh,
1294                    offset,
1295                    config.width,
1296                    config.depth,
1297                    config.height_scale,
1298                );
1299
1300                println!(
1301                    "[grass] Enabled: density={}, max_dist={}",
1302                    grass_config.density, grass_config.max_distance
1303                );
1304            } else {
1305                println!("[grass] Warning: splat map not found at {:?}", splat_path);
1306            }
1307        }
1308
1309        println!(
1310            "[terrain] Loaded terrain: {}x{} heightmap, {} chunks",
1311            heightmap.width,
1312            heightmap.depth,
1313            terrain.chunks.len()
1314        );
1315        break; // Only one terrain for now
1316    }
1317
1318    // Grass on a mesh ground (ADR 0089), as the player loads it
1319    if flint_player::load_scene_grass_ground(world, scene_path, device, queue, renderer) {
1320        println!("[grass] Enabled on grass_ground");
1321    }
