53pub fn run(args: RenderArgs) -> Result<()> {
54    // Merge explicit schemas with auto-discovered dirs from scene path
55    let mut all_schemas = args.schemas.clone();
56    for dir in flint_schema::discover_schema_dirs(&args.scene) {
57        let s = dir.to_string_lossy().into_owned();
58        if !all_schemas.contains(&s) {
59            all_schemas.push(s);
60        }
61    }
62
63    // Load schemas from all directories
64    let existing: Vec<&str> = all_schemas
65        .iter()
66        .map(|s| s.as_str())
67        .filter(|p| Path::new(p).exists())
68        .collect();
69    let registry = if !existing.is_empty() {
70        SchemaRegistry::load_from_directories(&existing).context("Failed to load schemas")?
71    } else {
72        println!("Warning: No schemas directories found");
73        SchemaRegistry::new()
74    };
75
76    // Load scene
77    let (mut world, scene_file) =
78        load_scene(&args.scene, &registry).context("Failed to load scene")?;
79    println!("Loaded scene: {}", scene_file.scene.name);
80    println!("Entities: {}", world.entity_count());
81
82    // Create headless context
83    let ctx = pollster::block_on(HeadlessContext::new(args.width, args.height))
84        .context("Failed to create headless render context")?;
85
86    // Configure camera — scene-level first, then CLI overrides take precedence
87    let mut camera = Camera::new();
88    camera.aspect = ctx.aspect_ratio();
89
90    // Apply scene-level camera configuration
91    let mut scene_set_position = false;
92    if let Some(cam_def) = &scene_file.camera {
93        if cam_def.projection == "orthographic" {
94            camera.orthographic = true;
95            if cam_def.ortho_height > 0.0 {
96                camera.ortho_height = cam_def.ortho_height;
97            }
98        }
99        if let Some(pos) = cam_def.position {
100            camera.position = Vec3::new(pos[0], pos[1], pos[2]);
101            scene_set_position = true;
102        }
103        if let Some(target) = cam_def.target {
104            camera.target = Vec3::new(target[0], target[1], target[2]);
105        }
106        if let Some(fov) = cam_def.fov {
107            camera.fov = fov;
108        }
109        if let Some(near) = cam_def.near {
110            camera.near = near;
111        }
112        if let Some(far) = cam_def.far {
113            camera.far = far;
114        }
115    }
116
117    // Derive orbit parameters from position/target so update_orbit() is consistent
118    if scene_set_position {
119        let dir = camera.position - camera.target;
120        camera.distance = dir.length();
121        if camera.distance > 0.001 {
122            let n = dir * (1.0 / camera.distance);
123            camera.pitch = n.y.asin();
124            camera.yaw = n.x.atan2(n.z);
125        }
126    }
127
128    // CLI overrides take precedence
129    if let Some(d) = args.distance {
130        camera.distance = d;
131    }
132    if let Some(y) = args.yaw {
133        camera.yaw = y.to_radians();
134    }
135    if let Some(p) = args.pitch {
136        camera.pitch = p.to_radians();
137    }
138    if let Some(t) = args.target {
139        camera.target = Vec3::new(t[0], t[1], t[2]);
140    }
141    if let Some(f) = args.fov {
142        camera.fov = f;
143    }
144    camera.update_orbit();
145
146    // Create scene renderer
147    let mut renderer = SceneRenderer::new_headless(
148        &ctx.device,
149        &ctx.queue,
150        ctx.format,
151        ctx.width,
152        ctx.height,
153        RendererConfig {
154            show_grid: !args.no_grid,
155        },
156    );
157
158    // Load models and textures from the scene
159    let config = ModelLoadConfig::from_scene_path(&args.scene);
160    model_loader::load_models_from_world(
161        &mut world,
162        &mut renderer,
163        &ctx.device,
164        &ctx.queue,
165        &config,
166    );
167
168    // Generate procedural geometry from spline + spline_mesh entities
169    spline_gen::load_splines(&args.scene, &mut world, &mut renderer, None, &ctx.device);
170
171    // Apply debug state
172    if let Some(mode_str) = &args.debug_mode {
173        let mode = match mode_str.as_str() {
174            "wireframe-overlay" => DebugMode::WireframeOverlay,
175            "wireframe" => DebugMode::WireframeOnly,
176            "normals" => DebugMode::Normals,
177            "depth" => DebugMode::Depth,
178            "uv" => DebugMode::UvChecker,
179            "unlit" => DebugMode::Unlit,
180            "metalrough" => DebugMode::MetallicRoughness,
181            _ => DebugMode::Pbr,
182        };
183        renderer.set_debug_mode(mode);
184    }
185    if args.wireframe_overlay {
186        renderer.set_debug_mode(DebugMode::WireframeOverlay);
187    }
188    if args.show_normals {
189        renderer.toggle_normal_arrows();
190    }
191    if args.no_tonemapping {
192        renderer.set_tonemapping(false);
193    }
194    if args.no_shadows {
195        renderer.set_shadows(false);
196    }
197    if args.shadow_resolution != 1024 {
198        renderer.set_shadow_resolution(&ctx.device, args.shadow_resolution);
199    }
200
201    // Post-processing configuration
202    {
203        let mut pp_config = if let Some(pp_def) = &scene_file.post_process {
204            flint_player::post_process_config_from_def(pp_def)
205        } else {
206            flint_render::PostProcessConfig::default()
207        };
208
209        // CLI overrides
210        if args.no_postprocess {
211            pp_config.enabled = false;
212        }
213        if let Some(intensity) = args.bloom_intensity {
214            pp_config.bloom_intensity = intensity;
215            pp_config.bloom_enabled = true;
216        }
217        if let Some(threshold) = args.bloom_threshold {
218            pp_config.bloom_threshold = threshold;
219        }
220        if let Some(exposure) = args.exposure {
221            pp_config.exposure = exposure;
222        }
223        if let Some(radius) = args.ssao_radius {
224            pp_config.ssao_radius = radius;
225        }
226        if let Some(intensity) = args.ssao_intensity {
227            pp_config.ssao_intensity = intensity;
228            if intensity <= 0.0 {
229                pp_config.ssao_enabled = false;
230            }
231        }
232        if let Some(density) = args.fog_density {
233            pp_config.fog_density = density;
234            pp_config.fog_enabled = density > 0.0;
235        }
236        if let Some(color) = args.fog_color {
237            pp_config.fog_color = color;
238        }
239        if let Some(falloff) = args.fog_height_falloff {
240            pp_config.fog_height_falloff = falloff;
241            pp_config.fog_height_enabled = true;
242        }
243        if let Some(intensity) = args.dither_intensity {
244            pp_config.dither_intensity = intensity;
245            pp_config.dither_enabled = intensity > 0.0;
246        }
247        if let Some(mode) = args.render_mode {
248            pp_config.render_mode = mode;
249            // A mode with no explicit mix defaults to fully torn through.
250            pp_config.mode_mix = args.mode_mix.unwrap_or(1.0);
251        }
252        if let Some(mix) = args.mode_mix {
253            pp_config.mode_mix = mix;
254        }
255        if let Some(params) = args.mode_params {
256            pp_config.mode_params = params;
257        }
258        if let Some(density) = args.volumetric_density {
259            pp_config.volumetric_density = density;
260            pp_config.volumetric_enabled = density > 0.0;
261        }
262        if let Some(samples) = args.volumetric_samples {
263            pp_config.volumetric_samples = samples;
264        }
265        if let Some(radius) = args.kuwahara_radius {
266            pp_config.kuwahara_radius = radius;
267            pp_config.kuwahara_enabled = true;
268        }
269        if let Some(sharpness) = args.kuwahara_sharpness {
270            pp_config.kuwahara_sharpness = sharpness;
271            pp_config.kuwahara_enabled = true;
272        }
273        if let Some(hardness) = args.kuwahara_hardness {
274            pp_config.kuwahara_hardness = hardness;
275            pp_config.kuwahara_enabled = true;
276        }
277        if let Some(anisotropy) = args.kuwahara_anisotropy {
278            pp_config.kuwahara_anisotropy = anisotropy;
279            pp_config.kuwahara_enabled = true;
280        }
281
282        renderer.set_post_process_config(pp_config);
283        renderer.ensure_kuwahara_resources(&ctx.device, &ctx.queue);
284    }
285
286    // Load terrain (if any)
287    load_terrain_for_render(&world, &args.scene, &ctx.device, &ctx.queue, &mut renderer);
288
289    // Set scene_dir for font/texture resolution and viewport params for screen anchoring
290    renderer.scene_dir = Path::new(&args.scene).parent().map(|p| p.to_path_buf());
291    renderer.ortho_height = camera.ortho_height;
292    renderer.aspect_ratio = camera.aspect;
293
294    renderer.update_from_world(&world, &ctx.device);
295
296    // Render
297    renderer.render_to(
298        &ctx.device,
299        &ctx.queue,
300        &ctx.depth_view,
301        &camera,
302        &ctx.color_view,
303    );
304
305    // Read back pixels
306    let pixels = pollster::block_on(ctx.read_pixels()).context("Failed to read rendered pixels")?;
307
308    // Encode as PNG
309    let img = image::RgbaImage::from_raw(args.width, args.height, pixels)
310        .context("Failed to create image from pixel data")?;
311    img.save(&args.output)
312        .context(format!("Failed to save image to {}", args.output))?;
313
314    println!(
315        "Rendered {}x{} image to {}",
316        args.width, args.height, args.output
317    );
318
319    Ok(())
320}

322/// Load terrain from world entities for headless rendering (no physics).
323fn load_terrain_for_render(
324    world: &flint_ecs::FlintWorld,
325    scene_path: &str,
326    device: &wgpu::Device,
327    queue: &wgpu::Queue,
328    renderer: &mut SceneRenderer,
329) {
330    use flint_core::Transform;
331    use flint_terrain::{Heightmap, Terrain, TerrainConfig};
332
333    let scene_dir = Path::new(scene_path)
334        .parent()
335        .unwrap_or_else(|| Path::new("."));
336
337    for entity in world.all_entities() {
338        let terrain_comp = match world.get_component(entity.id, comp::TERRAIN) {
339            Some(c) => c,
340            None => continue,
341        };
342
343        let heightmap_rel = match terrain_comp.get("heightmap").and_then(|v| v.as_str()) {
344            Some(s) if !s.is_empty() => s.to_string(),
345            _ => continue,
346        };
347
348        let hm_path = {
349            let p = scene_dir.join(&heightmap_rel);
350            if p.exists() {
351                p
352            } else if let Some(parent) = scene_dir.parent() {
353                let pp = parent.join(&heightmap_rel);
354                if pp.exists() {
355                    pp
356                } else {
357                    p
358                }
359            } else {
360                p
361            }
362        };
363
364        let heightmap = match Heightmap::from_png(&hm_path) {
365            Ok(hm) => hm,
366            Err(e) => {
367                tracing::warn!("[terrain] Failed to load heightmap: {}", e);
368                continue;
369            }
370        };
371
372        let get_f32 = |key: &str, default: f32| -> f32 {
373            terrain_comp.get(key).and_then(toml_f32).unwrap_or(default)
374        };
375
376        let get_i32 = |key: &str, default: i32| -> i32 {
377            terrain_comp
378                .get(key)
379                .and_then(|v| v.as_integer())
380                .map(|i| i as i32)
381                .unwrap_or(default)
382        };
383
384        let get_str = |key: &str| -> String {
385            terrain_comp
386                .get(key)
387                .and_then(|v| v.as_str())
388                .unwrap_or("")
389                .to_string()
390        };
391
392        let config = TerrainConfig {
393            heightmap_path: heightmap_rel,
394            width: get_f32("width", 256.0),
395            depth: get_f32("depth", 256.0),
396            height_scale: get_f32("height_scale", 50.0),
397            chunk_resolution: get_i32("chunk_resolution", 64) as u32,
398            texture_tile: get_f32("texture_tile", 16.0),
399            splat_map_path: get_str("splat_map"),
400            layer_textures: [
401                get_str("layer0_texture"),
402                get_str("layer1_texture"),
403                get_str("layer2_texture"),
404                get_str("layer3_texture"),
405            ],
406            metallic: get_f32("metallic", 0.0),
407            roughness: get_f32("roughness", 0.85),
408            grass: None,
409        };
410
411        let terrain = Terrain::generate(&heightmap, &config);
412
413        let transform = world
414            .get_component(entity.id, comp::TRANSFORM)
415            .and_then(|t| {
416                let pos = t.get("position").and_then(toml_vec3)?;
417                Some(Transform {
418                    position: Vec3::new(pos[0], pos[1], pos[2]),
419                    ..Default::default()
420                })
421            })
422            .unwrap_or_default();
423
424        renderer.load_terrain(
425            device,
426            queue,
427            &terrain.chunks,
428            &transform,
429            config.texture_tile,
430            config.metallic,
431            config.roughness,
432            &config.splat_map_path,
433            &config.layer_textures,
434            scene_dir,
435        );
436
437        // Load grass if enabled
438        let grass_config = {
439            let mut gc = flint_terrain::GrassConfig::default();
440            if let Some(enabled) = terrain_comp.get("grass.enabled") {
441                if enabled.as_bool().unwrap_or(false) {
442                    gc.enabled = true;
443                    gc.density = get_f32("grass.density", gc.density);
444                    gc.max_distance = get_f32("grass.max_distance", gc.max_distance);
445                    gc.fade_start = get_f32("grass.fade_start", gc.fade_start);
446                    gc.blade_width = get_f32("grass.blade_width", gc.blade_width);
447                    gc.blade_height = get_f32("grass.blade_height", gc.blade_height);
448                    gc.height_variation =
449                        get_f32("grass.height_variation", gc.height_variation);
450                    gc.wind_speed = get_f32("grass.wind_speed", gc.wind_speed);
451                    gc.wind_strength = get_f32("grass.wind_strength", gc.wind_strength);
452                    gc.bend_radius = get_f32("grass.bend_radius", gc.bend_radius);
453                    gc.bend_strength = get_f32("grass.bend_strength", gc.bend_strength);
454                    gc.density_threshold =
455                        get_f32("grass.density_threshold", gc.density_threshold);
456                    gc.density_layer =
457                        get_i32("grass.density_layer", gc.density_layer as i32) as u32;
458                    gc.dry_amount = get_f32("grass.dry_amount", gc.dry_amount);
459
460                    if let Some(v) = terrain_comp.get("grass.color_base").and_then(toml_vec3) {
461                        gc.color_base = v;
462                    }
463                    if let Some(v) = terrain_comp.get("grass.color_tip").and_then(toml_vec3) {
464                        gc.color_tip = v;
465                    }
466                    if let Some(v) = terrain_comp.get("grass.color_dry").and_then(toml_vec3) {
467                        gc.color_dry = v;
468                    }
469                    if let Some(v) =
470                        terrain_comp.get("grass.wind_direction").and_then(toml_vec3)
471                    {
472                        gc.wind_direction = v;
473                    }
474                }
475            }
476            gc
477        };
478
479        if grass_config.enabled {
480            let hm_data = heightmap.clone_heights();
481            let hm_w = heightmap.width;
482            let hm_d = heightmap.depth;
483
484            let splat_path = {
485                let p = scene_dir.join(&config.splat_map_path);
486                if p.exists() {
487                    p
488                } else {
489                    scene_dir
490                        .parent()
491                        .map(|pp| pp.join(&config.splat_map_path))
492                        .filter(|pp| pp.exists())
493                        .unwrap_or(p)
494                }
495            };
496
497            let offset = [
498                transform.position.x,
499                transform.position.y,
500                transform.position.z,
501            ];
502
503            if let Ok(splat_img) = image::open(&splat_path) {
504                let splat_rgba = splat_img.to_rgba8();
505                let (sw, sh) = splat_rgba.dimensions();
506
507                renderer.load_grass(
508                    device,
509                    queue,
510                    &grass_config,
511                    &hm_data,
512                    hm_w,
513                    hm_d,
514                    splat_rgba.as_raw(),
515                    sw,
516                    sh,
517                    offset,
518                    config.width,
519                    config.depth,
520                    config.height_scale,
521                );
522
523                println!(
524                    "[grass] Enabled: density={}, max_dist={}",
525                    grass_config.density, grass_config.max_distance
526                );
527            } else {
528                println!(
529                    "[grass] Warning: splat map not found at {:?}",
530                    splat_path
531                );
532            }
533        }
534
535        println!(
536            "[terrain] Loaded terrain: {}x{} heightmap, {} chunks",
537            heightmap.width,
538            heightmap.depth,
539            terrain.chunks.len()
540        );
541        break; // Only one terrain for now
542    }
