1//! Tap-to-beat calibration: play the foundation bus solo, collect the
2//! player's pulses, and write the median signed offset vs the
3//! latency-compensated beat grid to `logs/latency/calibration-*.toml`.
4//! Consumed by play-chart as part of the total judgment offset.
5//! Undiegetic and ugly by design at this stage (production plan D6).
6
7use anyhow::{bail, Context, Result};
8use flint_music::clock_bridge::ClockBridge;
9use flint_music::input_stream::InputEvent;
10use flint_music::session::RealtimePlayer;
11use flint_music::{validate_manifest, validate_manifest_assets, SuiteManifest};
12use std::path::{Path, PathBuf};
13use std::time::{Duration, Instant};
14
15use flint_music::chart_session::latest_latency_ms;
16
17#[derive(clap::Args)]
18pub struct CalibrateArgs {
19    /// Path to the suite manifest (.suite.toml)
20    pub manifest: String,
21
22    /// Directory the manifest's file paths are relative to (default: cwd)
23    #[arg(long)]
24    pub base_dir: Option<String>,
25
26    /// Number of taps to collect
27    #[arg(long, default_value_t = 16)]
28    pub taps: u32,
29}
30
31pub fn run(args: CalibrateArgs) -> Result<()> {
32    let base_dir = args
33        .base_dir
34        .map(PathBuf::from)
35        .unwrap_or_else(|| PathBuf::from("."));
36
37    let manifest = SuiteManifest::load(Path::new(&args.manifest))
38        .with_context(|| format!("loading {}", args.manifest))?;
39    let mut issues = validate_manifest(&manifest);
40    issues.extend(validate_manifest_assets(&manifest, &base_dir));
41    if !issues.is_empty() {
42        for i in &issues {
43            eprintln!("{i}");
44        }
45        bail!("manifest failed validation ({} issue(s))", issues.len());
46    }
47
48    let latency = latest_latency_ms(&base_dir);
49    let latency_ms = match &latency {
50        Some((file, ms)) => {
51            println!("compensating measured output latency: {ms:.1} ms (from {file})");
52            Some(*ms)
53        }
54        None => {
55            println!("WARNING: no output-latency measurement on record; calibrating without");
56            None
57        }
58    };
59    let offset_samples =
60        (latency_ms.unwrap_or(0.0) / 1000.0 * manifest.sample_rate as f64).round() as i64;
61
62    let mut player = RealtimePlayer::start(&manifest, &base_dir, latency_ms)
63        .map_err(|e| anyhow::anyhow!("{e}"))?;
64
65    // Foundation solo: the pulse-bearing stem is the only thing to tap to.
66    let others: Vec<String> = player
67        .session
68        .mixer
69        .states()
70        .map(|(n, _)| n.to_string())
71        .filter(|n| n != "foundation")
72        .collect();
73    for name in &others {
74        if let Some(bus) = player.session.mixer.bus_mut(name) {
75            bus.set_gain_now(-60.0);
76        }
77    }
78
79    let bridge = ClockBridge::new(manifest.sample_rate);
80    let (capture_handle, input_rx) = flint_input_capture::spawn(
81        bridge.clone(),
82        flint_input_capture::CaptureConfig {
83            offset_samples,
84            ..Default::default()
85        },
86    )
87    .map_err(|e| anyhow::anyhow!("{e}"))?;
88
89    println!(
90        "tap South or the right trigger on the beat, {} times...",
91        args.taps
92    );
93
94    let sample_rate = manifest.sample_rate as f64;
95    let mut errors_ms: Vec<f64> = Vec::new();
96    while errors_ms.len() < args.taps as usize {
97        std::thread::sleep(Duration::from_millis(2));
98        player.session.pump();
99        bridge.observe(Instant::now(), player.session.clock_sample());
100
101        while let Ok(ev) = input_rx.try_recv() {
102            let InputEvent::Pulse(p) = ev else { continue };
103            if p.sample < 0 {
104                continue; // pre-roll
105            }
106            let conductor = &player.session.conductor;
107            let beat = conductor.position_at_sample(p.sample).beat;
108            let err_ms =
109                (p.sample - conductor.sample_at_beat(beat.round())) as f64 / sample_rate * 1000.0;
110            errors_ms.push(err_ms);
111            println!("  tap {:>2}: {err_ms:+.1} ms", errors_ms.len());
112            if errors_ms.len() >= args.taps as usize {
113                break;
114            }
115        }
116
117        if player.session.mixer.all_stopped() {
118            bail!(
119                "stems ended after {} tap(s); need {} — rerun",
120                errors_ms.len(),
121                args.taps
122            );
123        }
124    }
125    drop(capture_handle);
126
127    let median_ms = median(&mut errors_ms.clone());
128    let mad_ms = {
129        let mut devs: Vec<f64> = errors_ms.iter().map(|e| (e - median_ms).abs()).collect();
130        median(&mut devs)
131    };
132    println!(
133        "calibration: median {median_ms:+.1} ms, MAD {mad_ms:.1} ms over {} taps",
134        errors_ms.len()
135    );
136
137    let latency_file = latency.map(|(f, _)| f).unwrap_or_else(|| "none".into());
138    let contents = format!(
139        "# Tap-to-beat calibration (flint calibrate)\n[calibration]\nmedian_ms = {median_ms:.3}\nmad_ms = {mad_ms:.3}\ntaps = {}\nlatency_file_used = \"{latency_file}\"\n",
140        errors_ms.len()
141    );
142    let path = super::write_latency_report(&base_dir, "calibration", &contents)?;
143    println!("wrote {}", path.display());
144    Ok(())
145}
146
147fn median(vals: &mut [f64]) -> f64 {
148    vals.sort_by(|a, b| a.partial_cmp(b).unwrap());
149    vals[vals.len() / 2]
150}
