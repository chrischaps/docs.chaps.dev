31/// Write a report into `<base_dir>/logs/latency/` as
32/// `<prefix>-<host>-<epoch>.toml` — the shared naming scheme the offset
33/// readers sort lexicographically (name embeds host + epoch, so max =
34/// newest). One writer for calibrate, spike-rumble, and the input spike.
35pub fn write_latency_report(
36    base_dir: &std::path::Path,
37    prefix: &str,
38    contents: &str,
39) -> anyhow::Result<std::path::PathBuf> {
40    use anyhow::Context;
41    let dir = base_dir.join("logs/latency");
42    std::fs::create_dir_all(&dir).context("creating logs/latency")?;
43    let host = std::env::var("COMPUTERNAME")
44        .or_else(|_| std::env::var("HOSTNAME"))
45        .unwrap_or_else(|_| "host".into());
46    let epoch = std::time::SystemTime::now()
47        .duration_since(std::time::UNIX_EPOCH)
48        .map(|d| d.as_secs())
49        .unwrap_or(0);
50    let path = dir.join(format!("{prefix}-{host}-{epoch}.toml"));
51    std::fs::write(&path, contents).with_context(|| format!("writing {path:?}"))?;
52    Ok(path)
53}
