1//! Headless judgment: run a recorded or synthetic input session through the
2//! identical chart-eval → judgment → coherence pipeline with no controller,
3//! no clock, and no audio device. Deterministic: the same session and
4//! config produce a bit-identical judgment log. With `--render` it also
5//! runs the offline audio render over the same span, so listening evidence
6//! and the coherence trace come from one invocation (coherence→mixer
7//! coupling is Phase 3; the render here is the plain suite).
8//!
9//! The loop body is [`flint_music::ChartCore`] — the same code the live
10//! harness runs (ADR 0016); this command is the offline front end.

78pub fn run(args: ReplayChartArgs) -> Result<()> {
79    let base_dir = args
80        .common
81        .base_dir
82        .map(PathBuf::from)
83        .unwrap_or_else(|| PathBuf::from("."));
84
85    // --- validate ------------------------------------------------------------
86    let manifest = SuiteManifest::load(Path::new(&args.common.manifest))
87        .with_context(|| format!("loading {}", args.common.manifest))?;
88    let chart = Chart::load(Path::new(&args.common.chart))
89        .with_context(|| format!("loading {}", args.common.chart))?;
90    let mut issues = validate_manifest(&manifest);
91    issues.extend(validate_chart(&chart, &manifest));
92    if args.render.is_some() {
93        issues.extend(validate_manifest_assets(&manifest, &base_dir));
94    }
95    if !issues.is_empty() {
96        for i in &issues {
97            eprintln!("{i}");
98        }
99        bail!("validation failed ({} issue(s))", issues.len());
100    }
101
102    // --- events --------------------------------------------------------------
103    // Offline is exact: no latency compensation anywhere (session samples
104    // are already compensated by whoever recorded them).
105    let conductor = Conductor::new(&manifest, None);
106    let eval = ChartEval::new(&chart, &conductor)
107        .with_context(|| format!("building chart evaluator for {}", args.common.chart))?;
108
109    // `pairs` keeps each event's raw clock sample alongside its
110    // suite-stamped event (identical until a session contains reintegration
111    // seams). Synthetic streams carry suite stamps; the reactive path maps
112    // them through its own timeline (`synthetic_stamps`).
113    let (pairs, session_header, synthetic_stamps) = match (&args.session, &args.synthetic) {
114        (Some(path), None) => {
115            let (header, pairs) = flint_music::replay::read_session_raw(Path::new(path))
116                .with_context(|| format!("reading session file {path}"))?;
117            if header.suite != manifest.id {
118                bail!(
119                    "session was recorded against suite '{}', manifest is '{}'",
120                    header.suite,
121                    manifest.id
122                );
123            }
124            println!(
125                "session: {path} ({} event(s); recorded latency {:.1} ms, calibration {:+.1} ms)",
126                pairs.len(),
127                header.latency_ms,
128                header.calibration_ms
129            );
130            (pairs, Some(header), false)
131        }
132        (None, Some(profile)) => {
133            let profile = SyntheticProfile::parse(profile)
134                .with_context(|| format!("parsing synthetic profile '{profile}'"))?;
135            let events = synthesize(&eval, &conductor, profile);
136            println!("synthetic session: {profile:?} ({} event(s))", events.len());
137            let pairs = events.into_iter().map(|ev| (ev.sample(), ev)).collect();
138            (pairs, None, true)
139        }
140        _ => bail!("exactly one of --session or --synthetic is required"),
141    };
142    let events: Vec<flint_music::input_stream::InputEvent> =
143        pairs.iter().map(|(_, ev)| ev.clone()).collect();
144
145    if let Some(path) = &args.save_session {
146        let header = flint_music::replay::SessionHeader {
147            schema: 0,
148            suite: manifest.id.clone(),
149            chart: args.common.chart.clone(),
150            sample_rate: manifest.sample_rate,
151            latency_ms: session_header.as_ref().map(|h| h.latency_ms).unwrap_or(0.0),
152            calibration_ms: session_header
153                .as_ref()
154                .map(|h| h.calibration_ms)
155                .unwrap_or(0.0),
156            coherence_config: session_header
157                .as_ref()
158                .and_then(|h| h.coherence_config.clone()),
159            capture: args
160                .synthetic
161                .as_ref()
162                .map(|p| serde_json::json!({ "synthetic": p })),
163        };
164        let mut w = flint_music::replay::SessionWriter::create(Path::new(path), &header)
165            .with_context(|| format!("creating session file {path}"))?;
166        for ev in &events {
167            w.write(ev, manifest.sample_rate)
168                .with_context(|| format!("writing session file {path}"))?;
169        }
170        w.flush()
171            .with_context(|| format!("flushing session file {path}"))?;
172        println!("session saved: {path} ({} event(s))", w.written());
173    }
174
175    // --- coherence config: explicit flag > session header snapshot >
176    // repo default file > built-in defaults ----------------------------------
177    let (coherence_cfg, coherence_source) = resolve_coherence_config(
178        args.config.as_deref().map(Path::new),
179        session_header
180            .as_ref()
181            .and_then(|h| h.coherence_config.as_ref()),
182        &base_dir,
183    )
184    .context("resolving coherence config")?;
185    let config_path = match &coherence_source {
186        CoherenceSource::Explicit(p) => {
187            println!("coherence config: {}", p.display());
188            if session_header
189                .as_ref()
190                .and_then(|h| h.coherence_config.as_ref())
191                .is_some_and(|snap| *snap != coherence_cfg.to_json())
192            {
193                println!("  (differs from the session's recorded config — offline retune)");
194            }
195            p.clone()
196        }
197        CoherenceSource::Snapshot => {
198            println!("coherence config: session header snapshot");
199            base_dir.join("config/coherence.toml")
200        }
201        CoherenceSource::DefaultFile(p) => {
202            println!("coherence config: {}", p.display());
203            p.clone()
204        }
205        CoherenceSource::Builtin(p) => {
206            println!("coherence config: built-in defaults");
207            p.clone()
208        }
209    };
210
211    // --- run the pipeline ----------------------------------------------------
212    let judgment_cfg = JudgmentConfig {
213        lean_mode: parse_lean_mode(&args.lean_mode)
214            .with_context(|| format!("parsing lean mode '{}'", args.lean_mode))?,
215        ..Default::default()
216    };
217    println!("lean mode: {}", lean_mode_name(judgment_cfg.lean_mode));
218    let eval_for_judge = ChartEval::new(&chart, &conductor)
219        .with_context(|| format!("building chart evaluator for {}", args.common.chart))?;
220    let judge = Judge::new(
221        eval_for_judge,
222        Conductor::new(&manifest, None),
223        judgment_cfg,
224    );
225    let coherence = Coherence::new(coherence_cfg);
226
227    let out_path = args
228        .out
229        .map(PathBuf::from)
230        .unwrap_or_else(|| base_dir.join("logs/judgment/replay.jsonl"));
231    let header = serde_json::json!({
232        "t": "header", "schema": 0,
233        "suite": manifest.id, "chart": args.common.chart,
234        "session": args.session, "synthetic": args.synthetic,
235        "latency_ms": session_header.as_ref().map(|h| h.latency_ms).unwrap_or(0.0),
236        "calibration_ms": session_header.as_ref().map(|h| h.calibration_ms).unwrap_or(0.0),
237        "grid_beats": judgment_cfg.grid_beats,
238        "lean_mode": lean_mode_name(judgment_cfg.lean_mode),
239        "arrival_half_beats": judgment_cfg.arrival_half_beats,
240        "coherence_config": coherence_cfg.to_json(),
241    });
242    let log = JsonlWriter::create(&out_path, &header)
243        .with_context(|| format!("creating judgment log {}", out_path.display()))?;
244
245    // --- ladder: with --render, its presence switches to the reactive loop ----
246    let default_ladder_path = base_dir.join(flint_music::chart_session::DEFAULT_LADDER_CONFIG);
247    let (ladder_cfg, ladder_path) = if let Some(path) = &args.ladder {
248        let cfg = LadderConfig::load(Path::new(path))
249            .with_context(|| format!("loading ladder config {path}"))?;
250        println!("ladder config: {path}");
251        (Some(cfg), PathBuf::from(path))
252    } else if default_ladder_path.exists() {
253        println!("ladder config: {}", default_ladder_path.display());
254        (
255            Some(LadderConfig::load(&default_ladder_path).with_context(|| {
256                format!("loading ladder config {}", default_ladder_path.display())
257            })?),
258            default_ladder_path,
259        )
260    } else {
261        (None, default_ladder_path)
262    };
263
264    let mut ladder_cfg = ladder_cfg;
265    let reactive = args.render.is_some() && ladder_cfg.is_some();
266    if args.render.is_none() && ladder_cfg.is_some() {
267        println!("(ladder inactive without --render; judgment replay is Milestone-2 semantics)");
268    }
269    let core_ladder = if reactive {
270        ladder_cfg
271            .take()
272            .expect("reactive is true only when ladder_cfg is Some (set just above)")
273    } else {
274        LadderConfig::default()
275    };
276    // Gradient config (ADR 0024): explicit path must load, the default file
277    // is optional, absent = inert built-ins (byte-identical to pre-gradient
278    // renders). Applied only inside the reactive loop.
279    let default_gradient_path = base_dir.join(flint_music::chart_session::DEFAULT_GRADIENT_CONFIG);
280    let (gradient_cfg, gradient_path) = if let Some(path) = &args.gradient {
281        let cfg = GradientConfig::load(Path::new(path))
282            .with_context(|| format!("loading gradient config {path}"))?;
283        println!("gradient config: {path} (tune bus: {})", cfg.tune.bus);
284        (cfg, PathBuf::from(path))
285    } else if default_gradient_path.exists() {
286        let cfg = GradientConfig::load(&default_gradient_path).with_context(|| {
287            format!(
288                "loading gradient config {}",
289                default_gradient_path.display()
290            )
291        })?;
292        println!(
293            "gradient config: {} (tune bus: {})",
294            default_gradient_path.display(),
295            cfg.tune.bus
296        );
297        (cfg, default_gradient_path)
298    } else {
299        (GradientConfig::default(), default_gradient_path)
300    };
301
302    let visual_eval = ChartEval::new(&chart, &conductor)
303        .with_context(|| format!("building chart evaluator for {}", args.common.chart))?;
304    let mut core = ChartCore::new(
305        judge,
306        coherence,
307        log,
308        Ladder::new(core_ladder),
309        Reintegrator::new(manifest.reintegration.clone()),
310        Conductor::new(&manifest, None),
311        config_path,
312        ladder_path,
313        GradientDriver::new(gradient_cfg),
314        gradient_path,
315        visual_eval,
316    );
317    core.sync_seam_params();
318    core.set_cues(flint_music::chart_session::resolve_cues(
319        &chart,
320        &Conductor::new(&manifest, None),
321    ));
322    if reactive {
323        let wav = args
324            .render
325            .as_ref()
326            .expect("run_reactive is entered only with --render present");
327        return run_reactive(
328            &manifest,
329            &base_dir,
330            wav,
331            core,
332            &pairs,
333            synthetic_stamps,
334            &out_path,
335        );
336    }
337
338    let mut last_bar = i64::MIN;
339    for ev in &events {
340        core.ingest(ev);
341        core.process(ev.sample())
342            .context("processing judgment events")?;
343        let bar = conductor.position_at_sample(ev.sample()).bar;
344        if bar != last_bar {
345            last_bar = bar;
346            println!(
347                "bar {:>3} | {}",
348                bar + 1,
349                coherence_meter(core.coherence().value())
350            );
351        }
352    }
353    let final_sample = events.last().map(|e| e.sample()).unwrap_or(0);
354    core.judge_finish();
355    core.process(final_sample)
356        .context("processing final judgment events")?;
357    core.flush_log().context("flushing judgment log")?;
358
359    let s = core.summary();
360    println!(
361        "done. pulses hit {} (mean |err| {:.1} ms), missed {}, spurious {} | final {}",
362        s.hits,
363        s.mean_abs_err_ms,
364        s.misses,
365        s.spurious,
366        coherence_meter(core.coherence().value())
367    );
368    println!("judgment log: {}", out_path.display());
369
370    // --- optional audio render over the same span (plain: no ladder) ----------
371    if let Some(wav) = args.render {
372        let duration_samples = final_sample.max(1);
373        let cfg = OfflineRenderConfig {
374            duration_samples,
375            ..Default::default()
376        };
377        let script = EventScript {
378            schema_version: 0,
379            events: vec![],
380        };
381        let result = flint_music::render_offline(
382            &manifest,
383            &FileStems::new(&base_dir),
384            &script,
385            &cfg,
386            |_, _| {},
387        )
388        .context("rendering suite audio offline")?;
389        flint_music::write_wav(Path::new(&wav), &result.samples, manifest.sample_rate)
390            .with_context(|| format!("writing WAV {wav}"))?;
391        println!(
392            "rendered {} ({:.1} s) alongside the coherence trace",
393            wav,
394            duration_samples as f64 / manifest.sample_rate as f64
395        );
396    }
397    Ok(())
398}

400/// The reactive render: the same judge → coherence → ladder → reintegration
401/// sequencer → mixer loop the live harness runs — [`ChartCore`], literally —
402/// inside the deterministic offline renderer (ADR 0006/0014). A recorded or
403/// synthetic fall renders its full-fail seams and reassemblies to WAV,
404/// bit-identically.
405fn run_reactive(
406    manifest: &SuiteManifest,
407    base_dir: &Path,
408    wav: &str,
409    mut core: ChartCore,
410    pairs: &[(i64, flint_music::input_stream::InputEvent)],
411    synthetic_stamps: bool,
412    out_path: &Path,
413) -> Result<()> {
414    let sample_rate = manifest.sample_rate;
415    let last_raw = pairs.last().map(|(r, _)| *r).unwrap_or(0);
416    // Generous trailing margin: a fall that begins at the last event still
417    // needs to decay, hold, seam, and reassemble — and a fully absent player
418    // loops, which is worth hearing a few times.
419    let cfg = OfflineRenderConfig {
420        duration_samples: last_raw.max(1) + 30 * sample_rate as i64,
421        ..Default::default()
422    };
423    let script = EventScript {
424        schema_version: 0,
425        events: vec![],
426    };
427
428    let mut fails = 0u64;
429    let mut cursor = 0usize;
430    let mut last_bar = i64::MIN;
431    let mut failure: Option<anyhow::Error> = None;
432
433    let result = flint_music::render_offline_with(
434        manifest,
435        &FileStems::new(base_dir),
436        &script,
437        &cfg,
438        |pos, session| {
439            if failure.is_some() {
440                return;
441            }
442            let raw = session.raw_clock_sample();
443            if raw < 0 {
444                return;
445            }
446            let mut fallible = || -> Result<()> {
447                let offset = session.timeline_offset();
448                while cursor < pairs.len() && pairs[cursor].0 <= raw {
449                    let (r, ev) = &pairs[cursor];
450                    cursor += 1;
451                    let suite_ev = if synthetic_stamps {
452                        ev.with_sample(r - offset)
453                    } else {
454                        ev.clone()
455                    };
456                    if suite_ev.sample() >= 0 {
457                        core.ingest(&suite_ev);
458                    }
459                }
460                let now_suite = session.clock_sample();
461                core.advance_to(now_suite);
462                core.process(now_suite)
463                    .context("processing judgment events")?;
464
465                let (_seq, seq_events) = core
466                    .step_seq(session, pos)
467                    .context("stepping reintegration sequencer")?;
468                for ev in &seq_events {
469                    if let ReintegrationEvent::FullFail {
470                        at_suite_sample,
471                        re_entry_sample,
472                        seam_suite_sample,
473                    } = ev
474                    {
475                        fails += 1;
476                        println!(
477                            "FULL FAIL at sample {at_suite_sample}: reintegrating to \
478                             {re_entry_sample} (seam at {seam_suite_sample})"
479                        );
480                    }
481                }
482
483                if pos.bar != last_bar && pos.sample >= 0 {
484                    last_bar = pos.bar;
485                    let rung = match core.ladder().rung() {
486                        Some(r) => format!(" | rung {} ({})", core.ladder().level(), r.name),
487                        None => String::new(),
488                    };
489                    println!(
490                        "bar {:>3} | {}{rung}",
491                        pos.bar + 1,
492                        coherence_meter(core.coherence().value())
493                    );
494                }
495                Ok(())
496            };
497            if let Err(e) = fallible() {
498                failure = Some(e);
499            }
500        },
501    )
502    .context("rendering reactive suite audio offline")?;
503    if let Some(e) = failure {
504        return Err(e);
505    }
506
507    let s = core.finish().context("finishing judgment core")?;
508    println!(
509        "done. pulses hit {} (mean |err| {:.1} ms), missed {}, spurious \
510         {} | {fails} full-fail(s) | final {}",
511        s.hits,
512        s.mean_abs_err_ms,
513        s.misses,
514        s.spurious,
515        coherence_meter(core.coherence().value())
516    );
517    println!("judgment log: {}", out_path.display());
518
519    flint_music::write_wav(Path::new(wav), &result.samples, sample_rate)
520        .with_context(|| format!("writing WAV {wav}"))?;
521    println!(
522        "rendered {} ({:.1} s, reactive: ladder + reintegration live)",
523        wav,
524        cfg.duration_samples as f64 / sample_rate as f64
525    );
526    Ok(())
