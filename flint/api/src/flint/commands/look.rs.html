1//! Looks (#279): named post-processing presets an agent can put over any
2//! scene, and sweeps that render one setting at several values side by side.
3//!
4//! A look is a `styles/<name>.look.toml`, beside the `*.style.toml`
5//! generation guides: a `[look]` table that says what it is and what to tune,
6//! and a `[post_process]` block in exactly the scene format. The block is
7//! merged into the scene's own `[post_process]` key by key (tables merge,
8//! everything else replaces), so a look changes only what it names.
9//!
10//! ```text
11//! flint render demo/pbr_test.scene.toml --look ascii_noir -o noir.png
12//! flint render demo/pbr_test.scene.toml --look ascii_noir --sweep ascii.cell=6,9,14 -o cells.png
13//! flint render demo/pbr_test.scene.toml --sweep look=ascii_noir,matrix,ink_wash,graphite -o looks.png
14//! flint render --look help
15//! ```
16

52/// A look as written.
53#[derive(Debug, Clone)]
54pub struct Look {
55    pub name: String,
56    pub description: String,
57    /// Sweeps worth trying first, in `--sweep` syntax.
58    pub tune: Vec<String>,
59    /// Advice for whoever tunes it (an agent, mostly).
60    pub notes: String,
61    /// Where it came from: a path, or `built-in`.
62    pub source: String,
63    /// The `[post_process]` block to merge.
64    pub post_process: toml::Table,
65}
66
67impl Look {
68    /// Parse a look's text; `name` is used when `[look] name` is absent.
69    pub fn parse(name: &str, text: &str, source: &str) -> Result<Self> {
70        let mut root: toml::Table =
71            toml::from_str(text).with_context(|| format!("look '{name}' ({source})"))?;
72        let meta = match root.remove("look") {
73            Some(toml::Value::Table(t)) => t,
74            Some(_) => bail!("look '{name}' ({source}): [look] must be a table"),
75            None => toml::Table::new(),
76        };
77        let post_process = match root.remove("post_process") {
78            Some(toml::Value::Table(t)) => t,
79            Some(_) => bail!("look '{name}' ({source}): [post_process] must be a table"),
80            None => bail!("look '{name}' ({source}) has no [post_process] block"),
81        };
82        if let Some(key) = root.keys().next() {
83            bail!("look '{name}' ({source}): unknown top-level table [{key}] (a look has [look] and [post_process])");
84        }
85        let text_of = |key: &str| {
86            meta.get(key)
87                .and_then(|v| v.as_str())
88                .unwrap_or_default()
89                .to_string()
90        };
91        let tune = meta
92            .get("tune")
93            .and_then(|v| v.as_array())
94            .map(|a| {
95                a.iter()
96                    .filter_map(|v| v.as_str().map(str::to_string))
97                    .collect()
98            })
99            .unwrap_or_default();
100        let look = Self {
101            name: meta
102                .get("name")
103                .and_then(|v| v.as_str())
104                .unwrap_or(name)
105                .to_string(),
106            description: text_of("description"),
107            notes: text_of("notes"),
108            tune,
109            source: source.to_string(),
110            post_process,
111        };
112        // Catch a misspelt field or style now, not as a silent default later.
113        look.apply(None)
114            .with_context(|| format!("look '{}' ({source})", look.name))?;
115        Ok(look)
116    }
117
118    /// The scene's `[post_process]` with this look merged over it.
119    pub fn apply(&self, scene: Option<&PostProcessDef>) -> Result<PostProcessDef> {
120        overlay(scene, &self.post_process)
121    }
122}

124/// `base` with `over` merged into it, checked by deserializing the result.
125/// Keys `PostProcessDef` doesn't have are refused, so a typo can't vanish.
126pub fn overlay(base: Option<&PostProcessDef>, over: &toml::Table) -> Result<PostProcessDef> {
127    let mut table = match base {
128        Some(def) => {
129            toml::Table::try_from(def).context("serializing the scene's [post_process]")?
130        }
131        None => toml::Table::new(),
132    };
133    let known = known_fields();
134    for key in over.keys() {
135        if !known.iter().any(|k| k == key) {
136            bail!(
137                "[post_process] has no field '{key}'{}",
138                suggestion(key, &known)
139            );
140        }
141    }
142    merge(&mut table, over);
143    let def: PostProcessDef = toml::Value::Table(table)
144        .try_into()
145        .map_err(|e| anyhow!("[post_process]: {e}"))?;
146    // The style tables are checked against the registry, as a scene's are.
147    let (_, warnings) = flint_render::stylize::StyleSettings::from_tables(&def.styles);
148    if let Some(first) = warnings.first() {
149        bail!("{first}");
150    }
151    Ok(def)
152}

154/// Every field a `[post_process]` block can name.
155fn known_fields() -> Vec<String> {
156    let def: PostProcessDef = toml::Value::Table(toml::Table::new())
157        .try_into()
158        .expect("an empty [post_process] deserializes");
159    let mut fields: Vec<String> = toml::Table::try_from(&def)
160        .map(|t| t.keys().cloned().collect())
161        .unwrap_or_default();
162    // Skipped when empty, so the empty block doesn't show it
163    fields.push("styles".to_string());
164    fields
165}

167/// ", did you mean 'x'?" when one known name is close.
168fn suggestion(key: &str, known: &[String]) -> String {
169    known
170        .iter()
171        .map(|k| (strsim_distance(key, k), k))
172        .filter(|(d, _)| *d <= 3)
173        .min_by_key(|(d, _)| *d)
174        .map(|(_, k)| format!(", did you mean '{k}'?"))
175        .unwrap_or_default()
176}

178/// Levenshtein distance, small and good enough for field names.
179fn strsim_distance(a: &str, b: &str) -> usize {
180    let b: Vec<char> = b.chars().collect();
181    let mut row: Vec<usize> = (0..=b.len()).collect();
182    for (i, ca) in a.chars().enumerate() {
183        let mut prev = row[0];
184        row[0] = i + 1;
185        for (j, cb) in b.iter().enumerate() {
186            let cur = row[j + 1];
187            row[j + 1] = if ca == *cb {
188                prev
189            } else {
190                1 + prev.min(row[j]).min(row[j + 1])
191            };
192            prev = cur;
193        }
194    }
195    row[b.len()]
196}

198/// Merge `over` into `base`: tables recurse, everything else replaces.
199pub fn merge(base: &mut toml::Table, over: &toml::Table) {
200    for (key, value) in over {
201        match (base.get_mut(key), value) {
202            (Some(toml::Value::Table(b)), toml::Value::Table(o)) => merge(b, o),
203            _ => {
204                base.insert(key.clone(), value.clone());
205            }
206        }
207    }
208}

210/// The folders searched for `<name>.look.toml`, nearest first: the scene's
211/// `styles/`, its parent's (a game's `levels/` sits beside `styles/`), and
212/// the working directory's. A file found there wins over a built-in look of
213/// the same name.
214fn search_dirs(scene: &str) -> Vec<PathBuf> {
215    let mut dirs = Vec::new();
216    if let Some(dir) = Path::new(scene).parent() {
217        dirs.push(dir.join("styles"));
218        if let Some(parent) = dir.parent() {
219            dirs.push(parent.join("styles"));
220        }
221    }
222    dirs.push(PathBuf::from("styles"));
223    dirs
224}

226/// Find a look by name or path.
227pub fn find(name_or_path: &str, scene: &str) -> Result<Look> {
228    let as_path = Path::new(name_or_path);
229    if name_or_path.ends_with(".toml") {
230        let text = std::fs::read_to_string(as_path)
231            .with_context(|| format!("reading look {}", as_path.display()))?;
232        let stem = file_stem(as_path);
233        return Look::parse(&stem, &text, &as_path.display().to_string());
234    }
235    for dir in search_dirs(scene) {
236        let path = dir.join(format!("{name_or_path}.look.toml"));
237        if path.is_file() {
238            let text = std::fs::read_to_string(&path)
239                .with_context(|| format!("reading look {}", path.display()))?;
240            return Look::parse(name_or_path, &text, &path.display().to_string());
241        }
242    }
243    if let Some((name, text)) = BUILTIN.iter().find(|(n, _)| *n == name_or_path) {
244        return Look::parse(name, text, "built-in");
245    }
246    let names: Vec<String> = list(scene).into_iter().map(|l| l.name).collect();
247    bail!(
248        "unknown look '{name_or_path}'; known: {} (`--look help` describes them)",
249        names.join(", ")
250    )
251}
252
253fn file_stem(path: &Path) -> String {
254    let name = path
255        .file_name()
256        .map(|n| n.to_string_lossy().into_owned())
257        .unwrap_or_default();
258    name.strip_suffix(".look.toml")
259        .or_else(|| name.strip_suffix(".toml"))
260        .unwrap_or(&name)
261        .to_string()
262}

264/// Every look reachable from `scene`: the files on disk, then the built-ins
265/// they don't shadow, sorted by name. Files that don't parse are skipped
266/// with a warning.
267pub fn list(scene: &str) -> Vec<Look> {
268    let mut looks: Vec<Look> = Vec::new();
269    for dir in search_dirs(scene) {
270        let Ok(entries) = std::fs::read_dir(&dir) else {
271            continue;
272        };
273        let mut paths: Vec<PathBuf> = entries
274            .flatten()
275            .map(|e| e.path())
276            .filter(|p| p.to_string_lossy().ends_with(".look.toml"))
277            .collect();
278        paths.sort();
279        for path in paths {
280            let stem = file_stem(&path);
281            if looks.iter().any(|l| l.name == stem) {
282                continue;
283            }
284            match std::fs::read_to_string(&path)
285                .map_err(anyhow::Error::from)
286                .and_then(|t| Look::parse(&stem, &t, &path.display().to_string()))
287            {
288                Ok(look) => looks.push(look),
289                Err(e) => eprintln!("warning: skipping {}: {e:#}", path.display()),
290            }
291        }
292    }
293    for (name, text) in BUILTIN {
294        if !looks.iter().any(|l| l.name == *name) {
295            if let Ok(look) = Look::parse(name, text, "built-in") {
296                looks.push(look);
297            }
298        }
299    }
300    looks.sort_by(|a, b| a.name.cmp(&b.name));
301    looks
302}

304/// `--look help`: each look, what it is and what to sweep first.
305pub fn help(scene: &str) -> String {
306    let mut out = String::from(
307        "Looks (styles/<name>.look.toml; a file beside the scene wins over a built-in):\n",
308    );
309    for look in list(scene) {
310        out.push_str(&format!("\n  {}  [{}]\n", look.name, look.source));
311        if !look.description.is_empty() {
312            out.push_str(&format!("    {}\n", look.description));
313        }
314        for sweep in &look.tune {
315            out.push_str(&format!("    tune: --sweep {sweep}\n"));
316        }
317        for line in look.notes.lines().filter(|l| !l.trim().is_empty()) {
318            out.push_str(&format!("    | {}\n", line.trim()));
319        }
320    }
321    out.push_str(
322        "\nSweep any registered style key (`--style help` lists them), any [post_process] \
323         field as post.<field>, or look=a,b,c.\n",
324    );
325    out
326}

330/// What a sweep varies.
331#[derive(Debug, Clone, PartialEq)]
332pub enum SweepKey {
333    /// `look=a,b,c`: each column a different look over the scene.
334    Look,
335    /// `<style>.<key>=...`: a registered style's param or option.
336    Style { style: String, key: String },
337    /// `post.<field>=...`: a `[post_process]` field.
338    Post { field: String },
339}
340
341/// `--sweep KEY=V1,V2,...`, parsed and checked.
342#[derive(Debug, Clone)]
343pub struct Sweep {
344    pub key: SweepKey,
345    pub values: Vec<String>,
346}
347
348/// Most columns a sheet takes; past this the cells get too small to judge.
349pub const MAX_COLUMNS: usize = 12;
350
351impl Sweep {
352    /// Parse `key=v1,v2,...`. Values split on `|` instead when one appears,
353    /// so a vector can be swept: `post.fog_color=0,0,0|0.7,0.75,0.82`.
354    pub fn parse(spec: &str) -> Result<Self, String> {
355        let (key, values) = spec
356            .split_once('=')
357            .ok_or_else(|| format!("expected KEY=V1,V2,... (got '{spec}')"))?;
358        let key = key.trim();
359        let separator = if values.contains('|') { '|' } else { ',' };
360        let values: Vec<String> = values
361            .split(separator)
362            .map(|v| v.trim().to_string())
363            .filter(|v| !v.is_empty())
364            .collect();
365        if values.len() < 2 {
366            return Err(format!("--sweep {key}: give at least two values"));
367        }
368        if values.len() > MAX_COLUMNS {
369            return Err(format!(
370                "--sweep {key}: {} values; a sheet takes at most {MAX_COLUMNS}",
371                values.len()
372            ));
373        }
374        let key = if key == "look" {
375            SweepKey::Look
376        } else if let Some(field) = key.strip_prefix("post.") {
377            if !known_fields().iter().any(|k| k == field) {
378                return Err(format!(
379                    "--sweep: [post_process] has no field '{field}'{}",
380                    suggestion(field, &known_fields())
381                ));
382            }
383            SweepKey::Post {
384                field: field.to_string(),
385            }
386        } else if let Some((style, k)) = key.split_once('.') {
387            for value in &values {
388                match flint_render::stylize::StyleSpec::parse(&format!("{style}:{k}={value}")) {
389                    Ok(_) => {}
390                    // Maybe a user style (#274), registered only once the
391                    // scene's folder is read; the column's apply checks it
392                    Err(e) if e.starts_with("unknown style") => {}
393                    Err(e) => return Err(format!("--sweep {key}={value}: {e}")),
394                }
395                // A style clamps a param to its range, so a value past it
396                // would draw the same column twice; say so instead.
397                let param = flint_render::stylize::find(style)
398                    .and_then(|d| d.params.iter().find(|p| p.name == k));
399                if let (Some(p), Ok(n)) = (param, value.parse::<f32>()) {
400                    if n < p.min || n > p.max {
401                        return Err(format!(
402                            "--sweep {key}={value}: outside {style}.{k}'s range {}..{}",
403                            p.min, p.max
404                        ));
405                    }
406                }
407            }
408            SweepKey::Style {
409                style: style.to_string(),
410                key: k.to_string(),
411            }
412        } else {
413            return Err(format!(
414                "--sweep: '{key}' is not look, post.<field> or <style>.<key> \
415                 (e.g. ascii.cell; `--style help` lists the keys)"
416            ));
417        };
418        Ok(Self { key, values })
419    }
420
421    /// The key as written.
422    pub fn name(&self) -> String {
423        match &self.key {
424            SweepKey::Look => "look".into(),
425            SweepKey::Style { style, key } => format!("{style}.{key}"),
426            SweepKey::Post { field } => format!("post.{field}"),
427        }
428    }
429
430    /// Column `i`'s `[post_process]`: the scene's, under the base look (or
431    /// the column's own, for a look sweep), under a post field's value.
432    pub fn column_def(
433        &self,
434        i: usize,
435        scene: Option<&PostProcessDef>,
436        base: Option<&Look>,
437        scene_path: &str,
438    ) -> Result<Option<PostProcessDef>> {
439        let value = &self.values[i];
440        match &self.key {
441            SweepKey::Look => Ok(Some(find(value, scene_path)?.apply(scene)?)),
442            SweepKey::Style { .. } => base_def(scene, base),
443            SweepKey::Post { field } => {
444                let based = base_def(scene, base)?;
445                let mut over = toml::Table::new();
446                over.insert(field.clone(), parse_value(value));
447                Ok(Some(overlay(based.as_ref(), &over).with_context(|| {
448                    format!("--sweep post.{field}={value}")
449                })?))
450            }
451        }
452    }
453
454    /// Column `i`'s `--style` word, for a style sweep.
455    pub fn column_style(&self, i: usize) -> Option<String> {
456        match &self.key {
457            SweepKey::Style { style, key } => Some(format!("{style}:{key}={}", self.values[i])),
458            _ => None,
459        }
460    }
461}

463/// The scene's `[post_process]` under the base look, if any.
464pub fn base_def(
465    scene: Option<&PostProcessDef>,
466    look: Option<&Look>,
467) -> Result<Option<PostProcessDef>> {
468    match look {
469        Some(look) => Ok(Some(look.apply(scene)?)),
470        None => Ok(scene.cloned()),
471    }
472}

474/// A sweep value as TOML: a number, bool or array when it reads as one
475/// (`0.7,0.75,0.82` becomes an array), otherwise a string.
476fn parse_value(text: &str) -> toml::Value {
477    let try_parse = |t: &str| {
478        toml::from_str::<toml::Table>(&format!("v = {t}"))
479            .ok()
480            .and_then(|mut table| table.remove("v"))
481    };
482    if text.contains(',') {
483        if let Some(v) = try_parse(&format!("[{text}]")) {
484            return v;
485        }
486    }
487    try_parse(text).unwrap_or_else(|| toml::Value::String(text.to_string()))
488}
