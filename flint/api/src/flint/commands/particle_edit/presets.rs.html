1//! Starter effects. Each is a complete `*.particles.toml` embedded at build
2//! time; `bootstrap` writes one to disk when `flint edit new.particles.toml`
3//! targets a file that does not exist yet.
4
5use std::path::Path;
6
7use anyhow::{Context, Result};
8use flint_particles::{load_effect_from_str, ParticleEffect};
9
10pub const PRESETS: &[(&str, &str)] = &[
11    ("fire", include_str!("../../../presets/fire.particles.toml")),
12    (
13        "smoke",
14        include_str!("../../../presets/smoke.particles.toml"),
15    ),
16    (
17        "sparks",
18        include_str!("../../../presets/sparks.particles.toml"),
19    ),
20    ("rain", include_str!("../../../presets/rain.particles.toml")),
21];
22
23pub fn names() -> impl Iterator<Item = &'static str> {
24    PRESETS.iter().map(|(n, _)| *n)
25}
26
27/// Parsed preset by name.
28pub fn preset(name: &str) -> Option<ParticleEffect> {
29    PRESETS
30        .iter()
31        .find(|(n, _)| *n == name)
32        .and_then(|(n, text)| load_effect_from_str(text, n).ok())
33}
34
35/// Create `path` from a preset, naming the effect after the file stem.
36pub fn bootstrap(path: &Path, preset_name: &str) -> Result<ParticleEffect> {
37    let (_, text) = PRESETS
38        .iter()
39        .find(|(n, _)| *n == preset_name)
40        .with_context(|| {
41            format!(
42                "unknown preset '{preset_name}' (available: {})",
43                names().collect::<Vec<_>>().join(", ")
44            )
45        })?;
46    let stem = flint_particles::loader::effect_stem(path).unwrap_or_else(|| "effect".to_string());
47    let text = text.replacen(
48        &format!("name = \"{preset_name}\""),
49        &format!("name = \"{stem}\""),
50        1,
51    );
52    if let Some(parent) = path.parent() {
53        if !parent.as_os_str().is_empty() {
54            std::fs::create_dir_all(parent)
55                .with_context(|| format!("creating {}", parent.display()))?;
56        }
57    }
58    std::fs::write(path, &text).with_context(|| format!("writing {}", path.display()))?;
59    load_effect_from_str(&text, &path.display().to_string()).map_err(|e| anyhow::anyhow!(e))
60}
