1//! Flags shared verbatim between subcommands, embedded via
2//! `#[command(flatten)]` so each flag is declared exactly once.
3//!
4//! Only flags whose semantics *and* help text are identical across the
5//! commands belong here — a flag whose doc comment differs per command
6//! (e.g. `--ladder`, whose replay-chart help documents the `--render`
7//! coupling) stays declared on each command so `--help` keeps saying the
8//! right thing.
9
10/// The manifest/chart/base-dir trio shared by `play-chart` and
11/// `replay-chart`.
12#[derive(clap::Args)]
13pub struct ChartCommonArgs {
14    /// Path to the suite manifest (.suite.toml)
15    pub manifest: String,
16
17    /// Beatmap chart (.chart.toml) for the suite
18    #[arg(long)]
19    pub chart: String,
20
21    /// Directory the manifest's file paths are relative to (default: cwd)
22    #[arg(long)]
23    pub base_dir: Option<String>,
24}

26pub(crate) fn parse_debug_mode(s: &str) -> Result<String, String> {
27    match s {
28        "wireframe" | "normals" | "depth" | "uv" | "unlit" | "metalrough" => Ok(s.to_string()),
29        _ => Err(format!(
30            "unknown debug mode '{}'; valid values: wireframe, normals, depth, uv, unlit, metalrough",
31            s
32        )),
33    }
34}

36pub(crate) fn parse_vec3(s: &str) -> Result<[f32; 3], String> {
37    let parts: Vec<&str> = s.split(',').collect();
38    if parts.len() != 3 {
39        return Err(format!(
40            "expected 3 comma-separated values, got {}",
41            parts.len()
42        ));
43    }
44    let x: f32 = parts[0]
45        .trim()
46        .parse()
47        .map_err(|e| format!("invalid x: {}", e))?;
48    let y: f32 = parts[1]
49        .trim()
50        .parse()
51        .map_err(|e| format!("invalid y: {}", e))?;
52    let z: f32 = parts[2]
53        .trim()
54        .parse()
55        .map_err(|e| format!("invalid z: {}", e))?;
56    Ok([x, y, z])
57}

59pub(crate) fn parse_vec4(s: &str) -> Result<[f32; 4], String> {
60    let parts: Vec<&str> = s.split(',').collect();
61    if parts.len() != 4 {
62        return Err(format!(
63            "expected 4 comma-separated values, got {}",
64            parts.len()
65        ));
66    }
67    let mut out = [0.0f32; 4];
68    for (i, p) in parts.iter().enumerate() {
69        out[i] = p
70            .trim()
71            .parse::<f32>()
72            .map_err(|e| format!("invalid number '{}': {}", p, e))?;
73    }
74    Ok(out)
75}
