1//! Track editor — loads a scene with spline data and launches the interactive editor.
2
3use anyhow::{Context, Result};
4use flint_core::toml_util::toml_f32;
5use flint_scene::load_scene;
6use flint_schema::SchemaRegistry;
7use flint_viewer::spline_editor::{ControlPoint, SplineEditorConfig};
8use std::path::Path;
9
10pub struct EditArgs {
11    pub scene: String,
12    pub schemas: Vec<String>,
13}
14
15pub fn run(args: EditArgs) -> Result<()> {
16    // Merge explicit schemas with auto-discovered dirs from scene path
17    let mut all_schemas = args.schemas.clone();
18    for dir in flint_schema::discover_schema_dirs(&args.scene) {
19        let s = dir.to_string_lossy().into_owned();
20        if !all_schemas.contains(&s) {
21            all_schemas.push(s);
22        }
23    }
24
25    // Load schemas from all directories
26    let existing: Vec<&str> = all_schemas
27        .iter()
28        .map(|s| s.as_str())
29        .filter(|p| Path::new(p).exists())
30        .collect();
31    let registry = if !existing.is_empty() {
32        SchemaRegistry::load_from_directories(&existing).context("Failed to load schemas")?
33    } else {
34        println!("Warning: No schemas directories found");
35        SchemaRegistry::new()
36    };
37
38    // Load scene
39    let (world, scene_file) = load_scene(&args.scene, &registry).context("Failed to load scene")?;
40    println!("Loaded scene: {}", scene_file.scene.name);
41    println!("Entities: {}", world.entity_count());
42
43    // Find the first entity with a `spline` component
44    let scene_dir = Path::new(&args.scene)
45        .parent()
46        .unwrap_or_else(|| Path::new("."));
47
48    let mut spline_file_path: Option<String> = None;
49    let mut spline_name = String::from("Track");
50
51    for entity in world.all_entities() {
52        let spline_file = world
53            .get_components(entity.id)
54            .and_then(|c| c.get("spline").cloned())
55            .and_then(|s| s.get("file").and_then(|v| v.as_str().map(String::from)));
56
57        if let Some(file_rel) = spline_file {
58            // Resolve path: scene dir, then parent
59            let full_path = {
60                let p = scene_dir.join(&file_rel);
61                if p.exists() {
62                    p
63                } else if let Some(parent) = scene_dir.parent() {
64                    let pp = parent.join(&file_rel);
65                    if pp.exists() {
66                        pp
67                    } else {
68                        p
69                    }
70                } else {
71                    p
72                }
73            };
74
75            if full_path.exists() {
76                spline_file_path = Some(full_path.to_string_lossy().to_string());
77                // Try to get a name from the entity
78                spline_name = entity.name.clone();
79                println!("Found spline: {} -> {}", entity.name, full_path.display());
80                break;
81            } else {
82                tracing::warn!("Spline file not found: {}", full_path.display());
83            }
84        }
85    }
86
87    let spline_path =
88        spline_file_path.context("No entity with a `spline` component found in the scene")?;
89
90    // Parse control points from the spline TOML file
91    let content = std::fs::read_to_string(&spline_path)
92        .with_context(|| format!("Failed to read spline file: {}", spline_path))?;
93    let value: toml::Value =
94        toml::from_str(&content).with_context(|| "Failed to parse spline TOML")?;
95
96    let spline_section = value
97        .get("spline")
98        .context("Missing [spline] section in spline file")?;
99    let closed = spline_section
100        .get("closed")
101        .and_then(|v| v.as_bool())
102        .unwrap_or(true);
103    let name = spline_section
104        .get("name")
105        .and_then(|v| v.as_str())
106        .unwrap_or(&spline_name)
107        .to_string();
108
109    let spacing = value
110        .get("sampling")
111        .and_then(|s| s.get("spacing"))
112        .and_then(toml_f32)
113        .unwrap_or(2.0);
114
115    let points_arr = value
116        .get("control_points")
117        .and_then(|v| v.as_array())
118        .context("Missing [[control_points]] in spline file")?;
119
120    let mut control_points = Vec::new();
121    for pt in points_arr {
122        let pos = pt
123            .get("position")
124            .and_then(|v| v.as_array())
125            .context("Control point missing position")?;
126        if pos.len() < 3 {
127            continue;
128        }
129        let x = toml_f32(&pos[0]).unwrap_or(0.0);
130        let y = toml_f32(&pos[1]).unwrap_or(0.0);
131        let z = toml_f32(&pos[2]).unwrap_or(0.0);
132        let twist = pt.get("twist").and_then(toml_f32).unwrap_or(0.0);
133
134        control_points.push(ControlPoint {
135            position: [x, y, z],
136            twist,
137        });
138    }
139
140    println!(
141        "Loaded {} control points from {} (closed={}, spacing={:.1})",
142        control_points.len(),
143        spline_path,
144        closed,
145        spacing
146    );
147
148    // Build editor config
149    let editor_config = SplineEditorConfig {
150        control_points,
151        closed,
152        spacing,
153        name,
154        file_path: spline_path,
155    };
156
157    // Launch viewer in editor mode with file watching enabled
158    let schemas_refs: Vec<&str> = args.schemas.iter().map(|s| s.as_str()).collect();
159    flint_viewer::app::run_editor(&args.scene, true, &schemas_refs, editor_config)
160}
