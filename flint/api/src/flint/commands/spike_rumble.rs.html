1//! H1 rumble spike (ADR 0025): prove force feedback fires on this stack,
2//! let the operator feel the three weight-axis prototypes (tick / thump /
3//! grind), time both command paths, and commit the report to
4//! `logs/latency/` beside the audio-latency and input-granularity spikes.
5
6use anyhow::Result;
7use std::path::PathBuf;
8
9#[derive(clap::Args)]
10pub struct SpikeRumbleArgs {
11    /// Directory whose logs/latency/ receives the report (default: cwd)
12    #[arg(long)]
13    pub base_dir: Option<String>,
14
15    /// Skip the operator-felt tick/thump/grind demo (timing only)
16    #[arg(long)]
17    pub no_feel: bool,
18}
19
20pub fn run(args: SpikeRumbleArgs) -> Result<()> {
21    let base_dir = args
22        .base_dir
23        .map(PathBuf::from)
24        .unwrap_or_else(|| PathBuf::from("."));
25
26    println!("rumble command-path spike (hold the controller)");
27    let report = flint_input_capture::rumble::spike_rumble(!args.no_feel)
28        .map_err(|e| anyhow::anyhow!("{e}"))?;
29    println!(
30        "direct set_ff_state: median {:.1} us, p99 {:.1} us, max {:.1} us (n={})",
31        report.direct.median_us, report.direct.p99_us, report.direct.max_us, report.direct.n
32    );
33    if let Some(ep) = &report.effect_play {
34        println!(
35            "Effect::play call:   median {:.1} us (n={}) — plus the ff server's structural \
36             0-50 ms tick quantization on top",
37            ep.median_us, ep.n
38        );
39    }
40
41    let path = super::write_latency_report(&base_dir, "rumble", &report.to_toml())?;
42    println!("wrote {}", path.display());
43    Ok(())
44}
