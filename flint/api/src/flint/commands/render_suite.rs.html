1//! Render a scripted suite session to WAV, offline and deterministic — the
2//! agent-facing "programmatic ears". Validates the manifest, runs the event
3//! script through the same `SuiteSession` used live, and writes 32-bit float
4//! stereo. Prints the same status lines as `play-suite` so offline and
5//! realtime evidence are textually comparable.
6
7use anyhow::{bail, Context, Result};
8use flint_music::event_script::EventScript;
9use flint_music::offline::{render_offline, write_wav, OfflineRenderConfig};
10use flint_music::session::FileStems;
11use flint_music::status::status_line;
12use flint_music::{validate_manifest, validate_manifest_assets, Conductor, SuiteManifest};
13use std::path::{Path, PathBuf};
14
15#[derive(clap::Args)]
16pub struct RenderSuiteArgs {
17    /// Path to the suite manifest (.suite.toml)
18    pub manifest: String,
19
20    /// Event script (.events.toml) of scheduled bus changes and markers
21    #[arg(long)]
22    pub script: Option<String>,
23
24    /// Output WAV path (32-bit float stereo)
25    #[arg(long, short)]
26    pub output: String,
27
28    /// Directory the manifest's file paths are relative to (default: cwd)
29    #[arg(long)]
30    pub base_dir: Option<String>,
31
32    /// Render this many bars (default: length of the longest stem)
33    #[arg(long, conflicts_with = "duration_seconds")]
34    pub duration_bars: Option<i64>,
35
36    /// Render this many seconds
37    #[arg(long)]
38    pub duration_seconds: Option<f64>,
39
40    /// Status line cadence: bar (default) or beat
41    #[arg(long)]
42    pub status_every: Option<String>,
43
44    /// Processing chunk size in frames (= scheduling granularity)
45    #[arg(long, default_value_t = 128)]
46    pub chunk_frames: usize,
47}
48
49pub fn run(args: RenderSuiteArgs) -> Result<()> {
50    let base_dir = args
51        .base_dir
52        .map(PathBuf::from)
53        .unwrap_or_else(|| PathBuf::from("."));
54
55    // --- validate first, same contract as play-suite -------------------------
56    let manifest = SuiteManifest::load(Path::new(&args.manifest))
57        .with_context(|| format!("loading {}", args.manifest))?;
58    let mut issues = validate_manifest(&manifest);
59    issues.extend(validate_manifest_assets(&manifest, &base_dir));
60    if !issues.is_empty() {
61        for i in &issues {
62            eprintln!("{i}");
63        }
64        bail!(
65            "manifest failed validation ({} issue(s)); not rendering",
66            issues.len()
67        );
68    }
69
70    let script = match &args.script {
71        Some(path) => EventScript::load(Path::new(path))
72            .map_err(|e| anyhow::anyhow!("{e}"))
73            .with_context(|| format!("loading event script {path}"))?,
74        None => EventScript {
75            schema_version: 0,
76            events: vec![],
77        },
78    };
79
80    // --- duration: explicit bars/seconds, else the longest stem --------------
81    let conductor = Conductor::new(&manifest, None);
82    let duration_samples = if let Some(bars) = args.duration_bars {
83        if bars < 1 {
84            bail!("--duration-bars must be >= 1");
85        }
86        conductor.sample_at_bar(bars, 0.0)
87    } else if let Some(secs) = args.duration_seconds {
88        (secs * manifest.sample_rate as f64).round() as i64
89    } else {
90        stem_duration_samples(&manifest, &base_dir)
91            .context("no --duration-bars/--duration-seconds and no probeable stems")?
92    };
93
94    let status_bars = match args.status_every.as_deref() {
95        None | Some("bar") => true,
96        Some("beat") => false,
97        Some(other) => bail!("--status-every must be 'bar' or 'beat', got '{other}'"),
98    };
99
100    println!(
101        "rendering suite '{}': {} samples ({:.1} s) at {} Hz, {} scripted events, chunk {} frames",
102        manifest.id,
103        duration_samples,
104        duration_samples as f64 / manifest.sample_rate as f64,
105        manifest.sample_rate,
106        script.events.len(),
107        args.chunk_frames,
108    );
109
110    let cfg = OfflineRenderConfig {
111        duration_samples,
112        chunk_frames: args.chunk_frames,
113    };
114    let mut last_line = (i64::MIN, i64::MIN); // (bar, whole beat)
115    let stems = FileStems::new(&base_dir);
116    let result = render_offline(&manifest, &stems, &script, &cfg, |pos, mixer| {
117        if pos.sample < 0 {
118            return;
119        }
120        let key = (
121            pos.bar,
122            if status_bars {
123                0
124            } else {
125                pos.beat_in_bar as i64
126            },
127        );
128        if key != last_line {
129            last_line = key;
130            let section = conductor.section_at_sample(pos.sample);
131            println!("{}", status_line(pos, section, mixer));
132        }
133    })
134    .map_err(|e| anyhow::anyhow!("{e}"))?;
135
136    for f in &result.fired {
137        println!(
138            "fired @ sample {:>9}{}: {:?} {:?}",
139            f.sample,
140            if f.late { " (LATE)" } else { "" },
141            f.bus.as_deref().unwrap_or("-"),
142            f.action
143        );
144    }
145    let out_path = Path::new(&args.output);
146    write_wav(out_path, &result.samples, manifest.sample_rate)
147        .map_err(|e| anyhow::anyhow!("{e}"))?;
148    println!(
149        "wrote {} ({} frames, {:.1} s)",
150        out_path.display(),
151        result.samples.len() / 2,
152        (result.samples.len() / 2) as f64 / manifest.sample_rate as f64
153    );
154    Ok(())
155}

157/// Duration of the longest playable stem, via the probe (also what the
158/// validator uses, so this agrees with the validated durations).
159fn stem_duration_samples(manifest: &SuiteManifest, base_dir: &Path) -> Option<i64> {
160    manifest
161        .buses
162        .values()
163        .filter_map(|decl| decl.file.as_ref())
164        .filter_map(|file| flint_music::probe::probe_audio(&base_dir.join(file)).ok())
165        .map(|info| info.frames as i64)
166        .max()
167}
