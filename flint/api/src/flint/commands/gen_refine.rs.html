1//! The mesh [`Previewer`] for `flint gen --ai-refine --reference <image>`:
2//! it lets Claude see what a spec builds, and the refine measure every
3//! round against the reference (#183).
4//!
5//! `flint-procgen-ai` can't draw a mesh (it depends on `flint-procgen`
6//! only), so this module draws for it: generate the spec, export a GLB, and
7//! render it with the same headless model render as
8//! `flint edit model.glb --render`. `gen_ai::build_agent` attaches it.
9
10use std::path::PathBuf;
11
12use anyhow::{Context, Result};
13use flint_procgen::{export_glb, GeneratorOutput, GeneratorRegistry, ProcGenSpec};
14use flint_procgen_ai::Previewer;
15
16use super::preview;
17
18/// Side of the square render Claude sees and the loop measures. Small
19/// enough to be cheap in tokens, big enough to show a silhouette.
20pub const PREVIEW_SIZE: u32 = 512;
21
22/// Renders mesh specs through the headless model renderer.
23pub struct MeshRenderPreviewer {
24    size: u32,
25    scratch: PathBuf,
26}
27
28impl MeshRenderPreviewer {
29    /// A previewer rendering `size` x `size` images, using `scratch` for the
30    /// GLB it renders from.
31    pub fn new(size: u32, scratch: PathBuf) -> Self {
32        Self { size, scratch }
33    }
34
35    /// The PNG `spec` renders to, or why it couldn't.
36    pub fn render(&self, spec: &ProcGenSpec) -> Result<Vec<u8>> {
37        let mut registry = GeneratorRegistry::new();
38        flint_procgen::register_built_in_generators(&mut registry);
39        let glb = match registry.generate_from_spec(spec)? {
40            GeneratorOutput::Mesh(m) => export_glb::mesh_to_glb(&m)?,
41            // Only the full-detail level: LODs would stack on one another.
42            GeneratorOutput::MeshWithLods(l) => {
43                export_glb::mesh_to_glb(l.first().context("no LOD levels")?)?
44            }
45            GeneratorOutput::SkinnedMesh(m) => export_glb::skinned_mesh_to_glb(&m)?,
46            _ => anyhow::bail!("not a mesh"),
47        };
48        std::fs::create_dir_all(&self.scratch)?;
49        let path = self.scratch.join(format!(
50            "refine-preview-{}-{}.glb",
51            std::process::id(),
52            unique()
53        ));
54        std::fs::write(&path, glb)?;
55        let rendered = preview::render_model(&path, self.size, self.size);
56        let _ = std::fs::remove_file(&path);
57        let mut png = std::io::Cursor::new(Vec::new());
58        rendered?.write_to(&mut png, image::ImageFormat::Png)?;
59        Ok(png.into_inner())
60    }
61}
62
63fn unique() -> u64 {
64    use std::sync::atomic::{AtomicU64, Ordering};
65    static NEXT: AtomicU64 = AtomicU64::new(0);
66    NEXT.fetch_add(1, Ordering::Relaxed)
67}
68
69impl Previewer for MeshRenderPreviewer {
70    fn preview(&self, spec: &ProcGenSpec) -> Option<Vec<u8>> {
71        match self.render(spec) {
72            Ok(png) => Some(png),
73            Err(e) => {
74                // A texture lands here by design; the agent shows its pixels.
75                if !e.to_string().contains("not a mesh") {
76                    eprintln!("warning: couldn't render a preview: {e:#}");
77                }
78                None
79            }
80        }
81    }
