1//! A sweep's contact sheet (#279): the same frame at several values of one
2//! setting, side by side, each labelled with what it was given. The letters
3//! are the ASCII style's own built-in 8x8 glyphs (#264), so the sheet speaks
4//! in the same type the styles draw with.
5
6use flint_render::glyph_atlas::GlyphAtlas;
7use image::{Rgba, RgbaImage};
8
9/// Ink on the sheet.
10pub const PAPER: Rgba<u8> = Rgba([16, 17, 19, 255]);
11pub const FRAME: Rgba<u8> = Rgba([46, 48, 54, 255]);
12pub const TITLE: Rgba<u8> = Rgba([236, 178, 92, 255]);
13pub const LABEL: Rgba<u8> = Rgba([226, 220, 206, 255]);
14pub const DIM: Rgba<u8> = Rgba([128, 128, 134, 255]);
15
16const MARGIN: u32 = 24;
17const GUTTER: u32 = 16;
18/// Columns before a sheet wraps to another row.
19const PER_ROW: usize = 4;
20const GLYPH: u32 = 8;
21
22/// Everything on a sheet that isn't a picture.
23pub struct SheetText {
24    /// Top left, large: the scene.
25    pub title: String,
26    /// Under it: what was swept, and over what.
27    pub subtitle: String,
28    /// One per cell: the value it took.
29    pub labels: Vec<String>,
30}
31
32/// How big the letters are: a glyph pixel per `scale` image pixels.
33fn text_scale(cell_width: u32) -> u32 {
34    (cell_width / 220).clamp(1, 4)
35}
36
37/// Lay `cells` (all one size) out in rows of up to four, under a title,
38/// each with its label beneath.
39pub fn compose(cells: &[RgbaImage], text: &SheetText) -> RgbaImage {
40    let (cell_w, cell_h) = cells.first().map_or((1, 1), RgbaImage::dimensions);
41    let columns = cells.len().clamp(1, PER_ROW) as u32;
42    let rows = cells.len().div_ceil(PER_ROW).max(1) as u32;
43    let scale = text_scale(cell_w);
44    let title_scale = scale + 1;
45    let line = GLYPH * scale;
46
47    let head = GLYPH * title_scale + 6 * scale + line + 10 * scale;
48    let label_band = 6 * scale + line + 4 * scale;
49    let row_h = cell_h + 2 + label_band;
50    let width = MARGIN * 2 + columns * (cell_w + 2) + (columns - 1) * GUTTER;
51    let height = MARGIN * 2 + head + rows * row_h + (rows - 1) * GUTTER;
52    let mut sheet = RgbaImage::from_pixel(width, height, PAPER);
53    let font = GlyphAtlas::builtin();
54
55    let mut y = MARGIN;
56    draw_text(
57        &mut sheet,
58        &font,
59        MARGIN,
60        y,
61        title_scale,
62        TITLE,
63        &text.title,
64    );
65    y += GLYPH * title_scale + 6 * scale;
66    draw_text(&mut sheet, &font, MARGIN, y, scale, DIM, &text.subtitle);
67    let top = MARGIN + head;
68
69    for (i, cell) in cells.iter().enumerate() {
70        let (col, row) = ((i % PER_ROW) as u32, (i / PER_ROW) as u32);
71        let x = MARGIN + col * (cell_w + 2 + GUTTER);
72        let y = top + row * (row_h + GUTTER);
73        frame(&mut sheet, x, y, cell_w + 2, cell_h + 2);
74        image::imageops::replace(&mut sheet, cell, i64::from(x + 1), i64::from(y + 1));
75        if let Some(label) = text.labels.get(i) {
76            let fit = fit_text(label, cell_w, scale);
77            draw_text(
78                &mut sheet,
79                &font,
80                x + 1,
81                y + cell_h + 2 + 6 * scale,
82                scale,
83                LABEL,
84                &fit,
85            );
86        }
87    }
88    sheet
89}

91/// `text` shortened with `~` so it fits `width` pixels.
92fn fit_text(text: &str, width: u32, scale: u32) -> String {
93    let max = (width / (GLYPH * scale)).max(1) as usize;
94    let chars: Vec<char> = text.chars().collect();
95    if chars.len() <= max {
96        return text.to_string();
97    }
98    let mut short: String = chars[..max.saturating_sub(1)].iter().collect();
99    short.push('~');
100    short
101}

103/// A one-pixel frame around a cell.
104fn frame(sheet: &mut RgbaImage, x: u32, y: u32, w: u32, h: u32) {
105    for i in 0..w {
106        sheet.put_pixel(x + i, y, FRAME);
107        sheet.put_pixel(x + i, y + h - 1, FRAME);
108    }
109    for j in 0..h {
110        sheet.put_pixel(x, y + j, FRAME);
111        sheet.put_pixel(x + w - 1, y + j, FRAME);
112    }
113}

115/// Draw `text` with the built-in glyphs, `scale` image pixels a glyph pixel,
116/// blended by coverage. Characters the font lacks draw as `?`; text past
117/// the sheet's edge is clipped.
118pub fn draw_text(
119    sheet: &mut RgbaImage,
120    font: &GlyphAtlas,
121    x: u32,
122    y: u32,
123    scale: u32,
124    color: Rgba<u8>,
125    text: &str,
126) {
127    let (w, h) = sheet.dimensions();
128    for (i, ch) in text.chars().enumerate() {
129        let ch = if font.contains(ch) { ch } else { '?' };
130        let gx = x + i as u32 * GLYPH * scale;
131        for py in 0..GLYPH {
132            for px in 0..GLYPH {
133                let a = font.sample(ch, px, py).unwrap_or(0);
134                if a == 0 {
135                    continue;
136                }
137                for sy in 0..scale {
138                    for sx in 0..scale {
139                        let (ox, oy) = (gx + px * scale + sx, y + py * scale + sy);
140                        if ox >= w || oy >= h {
141                            continue;
142                        }
143                        let under = *sheet.get_pixel(ox, oy);
144                        sheet.put_pixel(ox, oy, blend(under, color, a));
145                    }
146                }
147            }
148        }
149    }
150}

152/// The style mask view's legend (#309), in the image's lower left: a
153/// swatch of each style id's hue, its id and its name, on a dark card.
154/// Nothing styled still gets a card saying so, so an empty mask reads as
155/// empty rather than broken.
156pub fn draw_legend(img: &mut RgbaImage, entries: &[flint_render::style_mask::LegendEntry]) {
157    let (w, h) = img.dimensions();
158    let scale = (w / 640).clamp(1, 3);
159    let line = GLYPH * scale;
160    let pad = 6 * scale;
161    let gap = 4 * scale;
162    let rows: Vec<(Option<[f32; 3]>, String)> = if entries.is_empty() {
163        vec![(None, "no styled entities".to_string())]
164    } else {
165        entries
166            .iter()
167            .map(|e| (Some(e.color), format!("{} {}", e.id, e.name)))
168            .collect()
169    };
170    let longest = rows
171        .iter()
172        .map(|(_, t)| t.chars().count())
173        .max()
174        .unwrap_or(0) as u32;
175    let swatch = line;
176    let card_w = (pad * 2 + swatch + gap * 2 + longest * line).min(w);
177    let header = line + gap;
178    let card_h = (pad * 2 + header + rows.len() as u32 * (line + gap) - gap).min(h);
179    let margin = 8 * scale;
180    let x0 = margin.min(w.saturating_sub(card_w));
181    let y0 = h.saturating_sub(card_h + margin);
182    for y in y0..(y0 + card_h).min(h) {
183        for x in x0..(x0 + card_w).min(w) {
184            let under = *img.get_pixel(x, y);
185            img.put_pixel(x, y, blend(under, PAPER, 215));
186        }
187    }
188    let font = GlyphAtlas::builtin();
189    draw_text(img, &font, x0 + pad, y0 + pad, scale, DIM, "style mask");
190    for (i, (color, text)) in rows.iter().enumerate() {
191        let y = y0 + pad + header + i as u32 * (line + gap);
192        let mut tx = x0 + pad;
193        if let Some([r, g, b]) = color {
194            let byte = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
195            let c = Rgba([byte(*r), byte(*g), byte(*b), 255]);
196            for sy in y..(y + swatch).min(h) {
197                for sx in tx..(tx + swatch).min(w) {
198                    img.put_pixel(sx, sy, c);
199                }
200            }
201            tx += swatch + gap * 2;
202        }
203        draw_text(img, &font, tx, y, scale, LABEL, text);
204    }
205}

207fn blend(under: Rgba<u8>, over: Rgba<u8>, alpha: u8) -> Rgba<u8> {
208    let a = u32::from(alpha);
209    let mix = |u: u8, o: u8| ((u32::from(u) * (255 - a) + u32::from(o) * a) / 255) as u8;
210    Rgba([
211        mix(under[0], over[0]),
212        mix(under[1], over[1]),
213        mix(under[2], over[2]),
214        255,
215    ])
216}

233    fn count(sheet: &RgbaImage, color: Rgba<u8>) -> usize {
234        sheet.pixels().filter(|p| **p == color).count()
235    }

237    #[test]
238    fn every_cell_lands_whole() {
239        let cells = vec![RgbaImage::from_pixel(64, 48, RED); 3];
240        let sheet = compose(&cells, &text(&["6", "8", "12"]));
241        assert_eq!(
242            count(&sheet, RED),
243            3 * 64 * 48,
244            "a cell was cropped or drawn over"

246        assert_eq!(sheet.width(), 2 * MARGIN + 3 * 66 + 2 * GUTTER);
247    }

249    #[test]
250    fn cells_sit_in_order_and_wrap_after_four() {
251        let mut cells = vec![RgbaImage::from_pixel(32, 32, RED); 5];
252        cells[1] = RgbaImage::from_pixel(32, 32, BLUE);
253        cells[4] = RgbaImage::from_pixel(32, 32, BLUE);
254        let sheet = compose(&cells, &text(&["a", "b", "c", "d", "e"]));
255        let blue: Vec<(u32, u32)> = sheet
256            .enumerate_pixels()
257            .filter(|(_, _, p)| **p == BLUE)
258            .map(|(x, y, _)| (x, y))
259            .collect();
260        let min_x_row0 = blue
261            .iter()
262            .filter(|(_, y)| *y < sheet.height() / 2)
263            .map(|p| p.0)
264            .min();
265        assert_eq!(
266            min_x_row0,
267            Some(MARGIN + 34 + GUTTER + 1),
268            "second cell, first row"
269        );
270        let row1: Vec<_> = blue
271            .iter()
272            .filter(|(_, y)| *y > sheet.height() / 2)
273            .collect();
274        assert_eq!(
275            row1.iter().map(|p| p.0).min(),
276            Some(MARGIN + 1),
277            "fifth cell starts row two"
278        );
279        assert_eq!(
280            sheet.width(),
281            2 * MARGIN + 4 * 34 + 3 * GUTTER,
282            "four across at most"
283        );
284    }

297        use flint_render::style_mask::{debug_color, LegendEntry};
298        let entry = |id: u16, name: &str| LegendEntry {
