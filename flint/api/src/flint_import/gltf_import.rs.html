13/// Import a glTF or GLB file
14pub fn import_gltf<P: AsRef<Path>>(path: P) -> Result<ImportResult> {
15    let path = path.as_ref();
16    let (document, buffers, images) = gltf::import(path)
17        .map_err(|e| FlintError::ImportError(format!("Failed to import glTF: {}", e)))?;
18
19    let hash = ContentHash::from_file(path)
20        .map_err(|e| FlintError::ImportError(format!("Failed to hash file: {}", e)))?;
21
22    let file_name = path
23        .file_stem()
24        .and_then(|n| n.to_str())
25        .unwrap_or("unnamed")
26        .to_string();
27
28    let ext = path
29        .extension()
30        .and_then(|e| e.to_str())
31        .unwrap_or("glb")
32        .to_string();
33
34    // Build a map from glTF node index -> skin index for mesh-skin association
35    let mut node_skin_map: HashMap<usize, usize> = HashMap::new();
36    for node in document.nodes() {
37        if let Some(skin) = node.skin() {
38            // Associate the mesh node with its skin
39            if node.mesh().is_some() {
40                node_skin_map.insert(node.mesh().unwrap().index(), skin.index());
41            }
42        }
43    }
44
45    let mut meshes = Vec::new();
46    let mut total_vertices = 0u64;
47    let mut imported_nodes: Vec<ImportedNode> = Vec::new();
48    let mut root_nodes: Vec<usize> = Vec::new();
49
50    // Track which glTF node index maps to which ImportedNode index
51    let mut gltf_node_to_imported: HashMap<usize, usize> = HashMap::new();
52
53    // Walk the scene graph via scenes → root nodes → recursive children.
54    // This extracts meshes per-node (preserving transforms) instead of per-mesh.
55    let scene_root_nodes: Vec<gltf::Node> =
56        document.scenes().flat_map(|scene| scene.nodes()).collect();
57
58    // Recursive helper: extract a node and all its children
59    fn walk_node(
60        node: &gltf::Node,
61        buffers: &[gltf::buffer::Data],
62        node_skin_map: &HashMap<usize, usize>,
63        meshes: &mut Vec<ImportedMesh>,
64        imported_nodes: &mut Vec<ImportedNode>,
65        gltf_node_to_imported: &mut HashMap<usize, usize>,
66        total_vertices: &mut u64,
67    ) -> usize {
68        let (translation, rotation, scale) = node.transform().decomposed();
69
70        let node_name = node
71            .name()
72            .map(String::from)
73            .unwrap_or_else(|| format!("node_{}", node.index()));
74
75        let skin_index = if node.skin().is_some() {
76            node.skin().map(|s| s.index())
77        } else {
78            // Check if this node's mesh is associated with a skin via another node
79            node.mesh()
80                .and_then(|m| node_skin_map.get(&m.index()).copied())
81        };
82
83        // Extract mesh primitives if this node has a mesh
84        let mut mesh_primitive_indices = Vec::new();
85        if let Some(mesh) = node.mesh() {
86            let mesh_name = mesh
87                .name()
88                .map(String::from)
89                .unwrap_or_else(|| node_name.clone());
90
91            for primitive in mesh.primitives() {
92                let reader = primitive.reader(|buffer| Some(&buffers[buffer.index()]));
93
94                let positions: Vec<[f32; 3]> = reader
95                    .read_positions()
96                    .map(|iter| iter.collect())
97                    .unwrap_or_default();
98
99                let normals: Vec<[f32; 3]> = reader
100                    .read_normals()
101                    .map(|iter| iter.collect())
102                    .unwrap_or_default();
103
104                let uvs: Vec<[f32; 2]> = reader
105                    .read_tex_coords(0)
106                    .map(|iter| iter.into_f32().collect())
107                    .unwrap_or_default();
108
109                let indices: Vec<u32> = reader
110                    .read_indices()
111                    .map(|iter| iter.into_u32().collect())
112                    .unwrap_or_default();
113
114                let material_index = primitive.material().index();
115
116                let joint_indices: Option<Vec<[u16; 4]>> =
117                    reader.read_joints(0).map(|iter| iter.into_u16().collect());
118
119                let joint_weights: Option<Vec<[f32; 4]>> =
120                    reader.read_weights(0).map(|iter| iter.into_f32().collect());
121
122                *total_vertices += positions.len() as u64;
123
124                let prim_index = meshes.len();
125                meshes.push(ImportedMesh {
126                    name: mesh_name.clone(),
127                    positions,
128                    normals,
129                    uvs,
130                    indices,
131                    material_index,
132                    joint_indices,
133                    joint_weights,
134                    skin_index,
135                });
136                mesh_primitive_indices.push(prim_index);
137            }
138        }
139
140        // Reserve this node's index
141        let node_index = imported_nodes.len();
142        gltf_node_to_imported.insert(node.index(), node_index);
143        imported_nodes.push(ImportedNode {
144            name: node_name,
145            translation,
146            rotation,
147            scale,
148            mesh_primitive_indices,
149            children: Vec::new(), // filled in below
150            skin_index,
151        });
152
153        // Recurse into children
154        let child_indices: Vec<usize> = node
155            .children()
156            .map(|child| {
157                walk_node(
158                    &child,
159                    buffers,
160                    node_skin_map,
161                    meshes,
162                    imported_nodes,
163                    gltf_node_to_imported,
164                    total_vertices,
165                )
166            })
167            .collect();
168
169        imported_nodes[node_index].children = child_indices;
170
171        node_index
172    }
173
174    for root_node in &scene_root_nodes {
175        let idx = walk_node(
176            root_node,
177            &buffers,
178            &node_skin_map,
179            &mut meshes,
180            &mut imported_nodes,
181            &mut gltf_node_to_imported,
182            &mut total_vertices,
183        );
184        root_nodes.push(idx);
185    }
186
187    let mut textures = Vec::new();
188    for (i, image) in images.iter().enumerate() {
189        let tex_name = document
190            .textures()
191            .nth(i)
192            .and_then(|t| t.name().map(String::from))
193            .unwrap_or_else(|| format!("texture_{}", i));
194
195        let format = match image.format {
196            gltf::image::Format::R8 => "r8",
197            gltf::image::Format::R8G8 => "rg8",
198            gltf::image::Format::R8G8B8 => "rgb8",
199            gltf::image::Format::R8G8B8A8 => "rgba8",
200            gltf::image::Format::R16 => "r16",
201            gltf::image::Format::R16G16 => "rg16",
202            gltf::image::Format::R16G16B16 => "rgb16",
203            gltf::image::Format::R16G16B16A16 => "rgba16",
204            gltf::image::Format::R32G32B32FLOAT => "rgb32f",
205            gltf::image::Format::R32G32B32A32FLOAT => "rgba32f",
206        };
207
208        textures.push(ImportedTexture {
209            name: tex_name,
210            width: image.width,
211            height: image.height,
212            format: format.to_string(),
213            data: image.pixels.clone(),
214        });
215    }
216
217    let mut materials = Vec::new();
218    for material in document.materials() {
219        let mat_name = material
220            .name()
221            .map(String::from)
222            .unwrap_or_else(|| format!("material_{}", material.index().unwrap_or(0)));
223
224        let pbr = material.pbr_metallic_roughness();
225        let base_color = pbr.base_color_factor();
226        let metallic = pbr.metallic_factor();
227        let roughness = pbr.roughness_factor();
228
229        let base_color_texture = pbr.base_color_texture().map(|info| {
230            info.texture()
231                .name()
232                .map(String::from)
233                .unwrap_or_else(|| format!("texture_{}", info.texture().index()))
234        });
235
236        let normal_texture = material.normal_texture().map(|info| {
237            info.texture()
238                .name()
239                .map(String::from)
240                .unwrap_or_else(|| format!("texture_{}", info.texture().index()))
241        });
242
243        let metallic_roughness_texture = pbr.metallic_roughness_texture().map(|info| {
244            info.texture()
245                .name()
246                .map(String::from)
247                .unwrap_or_else(|| format!("texture_{}", info.texture().index()))
248        });
249
250        let alpha_mode = match material.alpha_mode() {
251            gltf::material::AlphaMode::Opaque => AlphaMode::Opaque,
252            gltf::material::AlphaMode::Mask => AlphaMode::Mask,
253            gltf::material::AlphaMode::Blend => AlphaMode::Blend,
254        };
255        let alpha_cutoff = material.alpha_cutoff().unwrap_or(0.5);
256
257        materials.push(ImportedMaterial {
258            name: mat_name,
259            base_color,
260            metallic,
261            roughness,
262            base_color_texture,
263            normal_texture,
264            metallic_roughness_texture,
265            use_vertex_color: false,
266            alpha_mode,
267            alpha_cutoff,
268        });
269    }
270
271    // --- Extract skeletons (skins) ---
272    let skeletons = extract_skeletons(&document, &buffers);
273
274    // --- Extract skeletal animation clips ---
275    let skeletal_clips = extract_skeletal_clips(&document, &buffers, &skeletons);
276
277    // --- Extract node-level animation clips ---
278    let node_clips = extract_node_clips(
279        &document,
280        &buffers,
281        &skeletons,
282        &gltf_node_to_imported,
283        &imported_nodes,
284    );
285
286    let mut properties = HashMap::new();
287    properties.insert(
288        "vertex_count".to_string(),
289        toml::Value::Integer(total_vertices as i64),
290    );
291    properties.insert(
292        "mesh_count".to_string(),
293        toml::Value::Integer(meshes.len() as i64),
294    );
295    properties.insert(
296        "material_count".to_string(),
297        toml::Value::Integer(materials.len() as i64),
298    );
299    if !skeletons.is_empty() {
300        properties.insert(
301            "skeleton_count".to_string(),
302            toml::Value::Integer(skeletons.len() as i64),
303        );
304    }
305    if !skeletal_clips.is_empty() {
306        properties.insert(
307            "skeletal_clip_count".to_string(),
308            toml::Value::Integer(skeletal_clips.len() as i64),
309        );
310    }
311    if !node_clips.is_empty() {
312        properties.insert(
313            "node_clip_count".to_string(),
314            toml::Value::Integer(node_clips.len() as i64),
315        );
316    }
317
318    let asset_meta = AssetMeta {
319        name: file_name,
320        asset_type: AssetType::Mesh,
321        hash: hash.to_prefixed_hex(),
322        source_path: Some(path.to_string_lossy().to_string()),
323        format: Some(ext),
324        properties,
325        tags: vec![],
326    };
327
328    Ok(ImportResult {
329        asset_meta,
330        meshes,
331        textures,
332        materials,
333        skeletons,
334        skeletal_clips,
335        node_clips,
336        nodes: imported_nodes,
337        root_nodes,
338    })
339}

341/// Extract skeleton data from glTF skins
342fn extract_skeletons(
343    document: &gltf::Document,
344    buffers: &[gltf::buffer::Data],
345) -> Vec<ImportedSkeleton> {
346    let mut skeletons = Vec::new();
347
348    for skin in document.skins() {
349        let skin_name = skin
350            .name()
351            .map(String::from)
352            .unwrap_or_else(|| format!("skin_{}", skin.index()));
353
354        let joint_nodes: Vec<gltf::Node> = skin.joints().collect();
355
356        // Build a map from glTF node index -> joint index within this skin
357        let mut node_to_joint: HashMap<usize, usize> = HashMap::new();
358        for (joint_idx, node) in joint_nodes.iter().enumerate() {
359            node_to_joint.insert(node.index(), joint_idx);
360        }
361
362        // Read inverse bind matrices
363        let reader = skin.reader(|buf| Some(&buffers[buf.index()]));
364        let ibms: Vec<[[f32; 4]; 4]> = reader
365            .read_inverse_bind_matrices()
366            .map(|iter| iter.collect())
367            .unwrap_or_else(|| vec![identity_4x4(); joint_nodes.len()]);
368
369        // Build joint hierarchy
370        let mut joints = Vec::with_capacity(joint_nodes.len());
371        for (joint_idx, node) in joint_nodes.iter().enumerate() {
372            let joint_name = node
373                .name()
374                .map(String::from)
375                .unwrap_or_else(|| format!("joint_{}", joint_idx));
376
377            // Find parent: walk up the glTF node tree and check if the parent is also a joint
378            let parent = find_parent_joint(document, node.index(), &node_to_joint);
379
380            let ibm = ibms.get(joint_idx).copied().unwrap_or_else(identity_4x4);
381            let (rest_translation, rest_rotation, rest_scale) = node.transform().decomposed();
382
383            joints.push(ImportedJoint {
384                name: joint_name,
385                index: joint_idx,
386                parent,
387                inverse_bind_matrix: ibm,
388                rest_translation,
389                rest_rotation,
390                rest_scale,
391            });
392        }
393
394        skeletons.push(ImportedSkeleton {
395            name: skin_name,
396            joints,
397        });
398    }
399
400    skeletons
401}

403/// Find the parent joint index for a given node, if the parent is within the skin's joint set
404fn find_parent_joint(
405    document: &gltf::Document,
406    node_index: usize,
407    node_to_joint: &HashMap<usize, usize>,
408) -> Option<usize> {
409    // Walk all nodes to find the parent of node_index
410    for node in document.nodes() {
411        for child in node.children() {
412            if child.index() == node_index {
413                // Found the parent node — check if it's a joint
414                return node_to_joint.get(&node.index()).copied();
415            }
416        }
417    }
418    None
419}

421/// Extract skeletal animation clips from glTF animations
422fn extract_skeletal_clips(
423    document: &gltf::Document,
424    buffers: &[gltf::buffer::Data],
425    skeletons: &[ImportedSkeleton],
426) -> Vec<ImportedSkeletalClip> {
427    if skeletons.is_empty() {
428        return Vec::new();
429    }
430
431    // Build a map from glTF node index -> (skeleton_index, joint_index)
432    // We rebuild from skeletons since each skeleton was built from skin joints
433    let mut node_to_skeleton_joint: HashMap<usize, (usize, usize)> = HashMap::new();
434    for skin in document.skins() {
435        let skel_idx = skin.index();
436        if skel_idx >= skeletons.len() {
437            continue;
438        }
439        for (joint_idx, node) in skin.joints().enumerate() {
440            node_to_skeleton_joint.insert(node.index(), (skel_idx, joint_idx));
441        }
442    }
443
444    let mut clips = Vec::new();
445
446    for animation in document.animations() {
447        let clip_name = animation
448            .name()
449            .map(String::from)
450            .unwrap_or_else(|| format!("clip_{}", animation.index()));
451
452        let mut channels = Vec::new();
453        let mut max_time: f32 = 0.0;
454
455        for channel in animation.channels() {
456            let target_node = channel.target().node().index();
457
458            // Only process channels that target skeleton joints
459            let Some(&(_skel_idx, joint_idx)) = node_to_skeleton_joint.get(&target_node) else {
460                continue;
461            };
462
463            let property = match channel.target().property() {
464                gltf::animation::Property::Translation => JointProperty::Translation,
465                gltf::animation::Property::Rotation => JointProperty::Rotation,
466                gltf::animation::Property::Scale => JointProperty::Scale,
467                _ => continue, // Skip morph targets etc.
468            };
469
470            let reader = channel.reader(|buf| Some(&buffers[buf.index()]));
471
472            let timestamps: Vec<f32> = reader
473                .read_inputs()
474                .map(|iter| iter.collect())
475                .unwrap_or_default();
476
477            let interpolation = match channel.sampler().interpolation() {
478                gltf::animation::Interpolation::Step => "STEP",
479                gltf::animation::Interpolation::Linear => "LINEAR",
480                gltf::animation::Interpolation::CubicSpline => "CUBICSPLINE",
481            };
482
483            let outputs: Vec<Vec<f32>> = match &property {
484                JointProperty::Translation | JointProperty::Scale => reader
485                    .read_outputs()
486                    .map(|out| match out {
487                        gltf::animation::util::ReadOutputs::Translations(iter) => {
488                            iter.map(|v| vec![v[0], v[1], v[2]]).collect()
489                        }
490                        gltf::animation::util::ReadOutputs::Scales(iter) => {
491                            iter.map(|v| vec![v[0], v[1], v[2]]).collect()
492                        }
493                        _ => Vec::new(),
494                    })
495                    .unwrap_or_default(),
496                JointProperty::Rotation => reader
497                    .read_outputs()
498                    .map(|out| match out {
499                        gltf::animation::util::ReadOutputs::Rotations(iter) => iter
500                            .into_f32()
501                            .map(|v| vec![v[0], v[1], v[2], v[3]])
502                            .collect(),
503                        _ => Vec::new(),
504                    })
505                    .unwrap_or_default(),
506            };
507
508            let keyframes = build_keyframes(&timestamps, &outputs, interpolation);
509            for kf in &keyframes {
510                if kf.time > max_time {
511                    max_time = kf.time;
512                }
513            }
514
515            channels.push(ImportedChannel {
516                joint_index: joint_idx,
517                property,
518                interpolation: interpolation.to_string(),
519                keyframes,
520            });
521        }
522
523        if !channels.is_empty() {
524            clips.push(ImportedSkeletalClip {
525                name: clip_name,
526                duration: max_time,
527                channels,
528            });
529        }
530    }
531
532    clips
533}

535/// Extract node-level animation clips from glTF animations.
536///
537/// Processes animation channels where the target node is NOT a skeleton joint,
538/// producing node transform animations that can drive entity hierarchies at runtime.
539fn extract_node_clips(
540    document: &gltf::Document,
541    buffers: &[gltf::buffer::Data],
542    _skeletons: &[ImportedSkeleton],
543    gltf_node_to_imported: &HashMap<usize, usize>,
544    imported_nodes: &[ImportedNode],
545) -> Vec<ImportedNodeClip> {
546    // Build a set of glTF node indices that are skeleton joints
547    let mut skeleton_joint_nodes: std::collections::HashSet<usize> =
548        std::collections::HashSet::new();
549    for skin in document.skins() {
550        for node in skin.joints() {
551            skeleton_joint_nodes.insert(node.index());
552        }
553    }
554
555    let mut clips = Vec::new();
556
557    for animation in document.animations() {
558        let clip_name = animation
559            .name()
560            .map(String::from)
561            .unwrap_or_else(|| format!("clip_{}", animation.index()));
562
563        let mut channels = Vec::new();
564        let mut max_time: f32 = 0.0;
565
566        for channel in animation.channels() {
567            let target_node = channel.target().node().index();
568
569            // Only process channels that target NON-skeleton nodes
570            if skeleton_joint_nodes.contains(&target_node) {
571                continue;
572            }
573
574            // Resolve to imported node index and name
575            let Some(&imported_idx) = gltf_node_to_imported.get(&target_node) else {
576                continue;
577            };
578            let node_name = imported_nodes[imported_idx].name.clone();
579
580            let property = match channel.target().property() {
581                gltf::animation::Property::Translation => JointProperty::Translation,
582                gltf::animation::Property::Rotation => JointProperty::Rotation,
583                gltf::animation::Property::Scale => JointProperty::Scale,
584                _ => continue, // Skip morph targets etc.
585            };
586
587            let reader = channel.reader(|buf| Some(&buffers[buf.index()]));
588
589            let timestamps: Vec<f32> = reader
590                .read_inputs()
591                .map(|iter| iter.collect())
592                .unwrap_or_default();
593
594            let interpolation = match channel.sampler().interpolation() {
595                gltf::animation::Interpolation::Step => "STEP",
596                gltf::animation::Interpolation::Linear => "LINEAR",
597                gltf::animation::Interpolation::CubicSpline => "CUBICSPLINE",
598            };
599
600            let outputs: Vec<Vec<f32>> = match &property {
601                JointProperty::Translation | JointProperty::Scale => reader
602                    .read_outputs()
603                    .map(|out| match out {
604                        gltf::animation::util::ReadOutputs::Translations(iter) => {
605                            iter.map(|v| vec![v[0], v[1], v[2]]).collect()
606                        }
607                        gltf::animation::util::ReadOutputs::Scales(iter) => {
608                            iter.map(|v| vec![v[0], v[1], v[2]]).collect()
609                        }
610                        _ => Vec::new(),
611                    })
612                    .unwrap_or_default(),
613                JointProperty::Rotation => reader
614                    .read_outputs()
615                    .map(|out| match out {
616                        gltf::animation::util::ReadOutputs::Rotations(iter) => iter
617                            .into_f32()
618                            .map(|v| vec![v[0], v[1], v[2], v[3]])
619                            .collect(),
620                        _ => Vec::new(),
621                    })
622                    .unwrap_or_default(),
623            };
624
625            let keyframes = build_keyframes(&timestamps, &outputs, interpolation);
626            for kf in &keyframes {
627                if kf.time > max_time {
628                    max_time = kf.time;
629                }
630            }
631
632            channels.push(ImportedNodeChannel {
633                node_index: imported_idx,
634                node_name,
635                property,
636                interpolation: interpolation.to_string(),
637                keyframes,
638            });
639        }
640
641        if !channels.is_empty() {
642            clips.push(ImportedNodeClip {
643                name: clip_name,
644                duration: max_time,
645                channels,
646            });
647        }
648    }
649
650    clips
651}

653fn identity_4x4() -> [[f32; 4]; 4] {
654    [
655        [1.0, 0.0, 0.0, 0.0],
656        [0.0, 1.0, 0.0, 0.0],
657        [0.0, 0.0, 1.0, 0.0],
658        [0.0, 0.0, 0.0, 1.0],
659    ]
660}

662/// Pair sampler timestamps with their output values.
663///
664/// For STEP/LINEAR samplers glTF stores one output per timestamp. For CUBICSPLINE it
665/// stores three per timestamp — `[in_tangent, value, out_tangent]` — so the outputs
666/// must be consumed in triples or the tangents (typically zero) end up as keyframe
667/// values and the animation collapses toward zero between keys.
668fn build_keyframes(
669    timestamps: &[f32],
670    outputs: &[Vec<f32>],
671    interpolation: &str,
672) -> Vec<ImportedKeyframe> {
673    if interpolation == "CUBICSPLINE" {
674        if outputs.len() < timestamps.len() * 3 {
675            eprintln!(
676                "warning: CUBICSPLINE sampler has {} outputs for {} timestamps (expected {}); treating as LINEAR",
677                outputs.len(),
678                timestamps.len(),
679                timestamps.len() * 3
680            );
681        } else {
682            return timestamps
683                .iter()
684                .zip(outputs.chunks_exact(3))
685                .map(|(&time, triple)| ImportedKeyframe {
686                    time,
687                    in_tangent: triple[0].clone(),
688                    value: triple[1].clone(),
689                    out_tangent: triple[2].clone(),
690                })
691                .collect();
692        }
693    }
694    timestamps
695        .iter()
