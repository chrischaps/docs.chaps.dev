5/// Result of importing a file
6#[derive(Debug)]
7pub struct ImportResult {
8    /// Metadata about the imported asset
9    pub asset_meta: AssetMeta,
10    /// Extracted meshes
11    pub meshes: Vec<ImportedMesh>,
12    /// Extracted textures
13    pub textures: Vec<ImportedTexture>,
14    /// Extracted materials
15    pub materials: Vec<ImportedMaterial>,
16    /// Extracted skeletons (skins)
17    pub skeletons: Vec<ImportedSkeleton>,
18    /// Extracted skeletal animation clips
19    pub skeletal_clips: Vec<ImportedSkeletalClip>,
20    /// Extracted node-level animation clips (non-skeletal transform animations)
21    pub node_clips: Vec<ImportedNodeClip>,
22    /// glTF scene graph nodes with transforms
23    pub nodes: Vec<ImportedNode>,
24    /// Indices of top-level (root) nodes in the scene graph
25    pub root_nodes: Vec<usize>,
26}
27
28impl ImportResult {
29    /// Compute the combined bounding box across all meshes
30    pub fn bounds(&self) -> Option<MeshBounds> {
31        self.meshes
32            .iter()
33            .filter_map(|m| m.bounds())
34            .reduce(|a, b| a.union(&b))
35    }
36
37    /// Returns true when the GLB contains node-level (non-skeletal) animation clips
38    pub fn has_node_animations(&self) -> bool {
39        !self.node_clips.is_empty()
40    }
41
42    /// Returns true when the GLB contains multiple mesh-bearing nodes or
43    /// any mesh node with a non-identity transform, meaning child entities
44    /// should be created to preserve the glTF spatial layout.
45    pub fn needs_expansion(&self) -> bool {
46        let mesh_nodes: Vec<&ImportedNode> = self
47            .nodes
48            .iter()
49            .filter(|n| !n.mesh_primitive_indices.is_empty())
50            .collect();
51
52        if mesh_nodes.len() > 1 {
53            return true;
54        }
55
56        // Single mesh node with a non-identity transform
57        if let Some(node) = mesh_nodes.first() {
58            let t = &node.translation;
59            let r = &node.rotation;
60            let s = &node.scale;
61            let is_identity = (t[0].abs() < 1e-6 && t[1].abs() < 1e-6 && t[2].abs() < 1e-6)
62                && (r[0].abs() < 1e-6
63                    && r[1].abs() < 1e-6
64                    && r[2].abs() < 1e-6
65                    && (r[3] - 1.0).abs() < 1e-6)
66                && ((s[0] - 1.0).abs() < 1e-6
67                    && (s[1] - 1.0).abs() < 1e-6
68                    && (s[2] - 1.0).abs() < 1e-6);
69            if !is_identity {
70                return true;
71            }
72        }
73
74        false
75    }
76}

78/// Axis-aligned bounding box computed from vertex positions
79#[derive(Debug, Clone, Copy)]
80pub struct MeshBounds {
81    pub min: [f32; 3],
82    pub max: [f32; 3],
83}
84
85impl MeshBounds {
86    /// Compute bounds from a set of vertex positions
87    pub fn from_positions(positions: &[[f32; 3]]) -> Option<Self> {
88        if positions.is_empty() {
89            return None;
90        }
91        let mut min = positions[0];
92        let mut max = positions[0];
93        for p in positions.iter().skip(1) {
94            for i in 0..3 {
95                if p[i] < min[i] {
96                    min[i] = p[i];
97                }
98                if p[i] > max[i] {
99                    max[i] = p[i];
100                }
101            }
102        }
103        Some(Self { min, max })
104    }
105
106    /// Size along each axis
107    pub fn size(&self) -> [f32; 3] {
108        [
109            self.max[0] - self.min[0],
110            self.max[1] - self.min[1],
111            self.max[2] - self.min[2],
112        ]
113    }
114
115    /// Merge with another bounds to get the union
116    pub fn union(&self, other: &MeshBounds) -> MeshBounds {
117        MeshBounds {
118            min: [
119                self.min[0].min(other.min[0]),
120                self.min[1].min(other.min[1]),
121                self.min[2].min(other.min[2]),
122            ],
123            max: [
124                self.max[0].max(other.max[0]),
125                self.max[1].max(other.max[1]),
126                self.max[2].max(other.max[2]),
127            ],
128        }
129    }
130}
131
132impl std::fmt::Display for MeshBounds {
133    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
134        let s = self.size();
135        write!(
136            f,
137            "{:.2} x {:.2} x {:.2} (min [{:.2}, {:.2}, {:.2}], max [{:.2}, {:.2}, {:.2}])",
138            s[0],
139            s[1],
140            s[2],
141            self.min[0],
142            self.min[1],
143            self.min[2],
144            self.max[0],
145            self.max[1],
146            self.max[2],
147        )
148    }
149}

171/// One level of detail of a model: the band of camera distances a mesh draws
172/// in, authored per glTF primitive (or per mesh, for all its primitives) as
173/// `extras = { flint_lod = { near, far, draw_distance, ref_min, ref_max } }`
174/// (every key optional). Keep a model's levels as primitives of one mesh on
175/// one node: several mesh nodes are expanded into a child entity each.
176///
177/// The distance is measured from the camera to the closest point of the
178/// reference box `ref_min..ref_max` (model space; the mesh's own bounds when
179/// absent). Every level of one group names the same box, so exactly one of
180/// them draws at any distance: a merged chunk of props and its simplified
181/// copy switch together instead of overlapping. `near..far` is a switch
182/// between levels and scales with the renderer's LOD scale; `draw_distance`
183/// is where the mesh stops drawing at all and scales with the draw-distance
184/// scale. A missing `far` or `draw_distance` is unbounded.
185#[derive(Debug, Clone, Copy, PartialEq)]
186pub struct LodBand {
187    pub near: f32,
188    pub far: f32,
189    pub draw_distance: f32,
190    pub ref_box: Option<([f32; 3], [f32; 3])>,
191}
192
193impl Default for LodBand {
194    fn default() -> Self {
195        Self {
196            near: 0.0,
197            far: f32::INFINITY,
198            draw_distance: f32::INFINITY,
199            ref_box: None,
200        }
201    }
202}
203
204impl LodBand {
205    /// The band in a glTF `extras` object, if it has a `flint_lod` table.
206    /// Malformed numbers fall back to the defaults rather than failing the
207    /// import: a model with a bad band still draws.
208    pub fn from_extras(extras: &serde_json::Value) -> Option<Self> {
209        let lod = extras.get("flint_lod")?.as_object()?;
210        let num = |k: &str| lod.get(k).and_then(|v| v.as_f64()).map(|v| v as f32);
211        let vec3 = |k: &str| -> Option<[f32; 3]> {
212            let a = lod.get(k)?.as_array()?;
213            if a.len() != 3 {
214                return None;
215            }
216            Some([
217                a[0].as_f64()? as f32,
218                a[1].as_f64()? as f32,
219                a[2].as_f64()? as f32,
220            ])
221        };
222        let positive = |v: Option<f32>| v.filter(|d| *d > 0.0).unwrap_or(f32::INFINITY);
223        Some(Self {
224            near: num("near").unwrap_or(0.0).max(0.0),
225            far: positive(num("far")),
226            draw_distance: positive(num("draw_distance")),
227            ref_box: vec3("ref_min").zip(vec3("ref_max")),
228        })
229    }
230
231    /// Whether a camera `distance` metres from the reference box draws this
232    /// band, with switch distances times `lod_scale` and the draw distance
233    /// times `draw_scale`. `near` is inclusive and `far` exclusive, so two
234    /// levels that meet at one distance never both draw.
235    pub fn visible_at(&self, distance: f32, lod_scale: f32, draw_scale: f32) -> bool {
236        distance >= self.near * lod_scale
237            && distance < self.far * lod_scale
238            && distance < self.draw_distance * draw_scale
239    }
240}

365/// An imported PBR material
366#[derive(Debug, Clone)]
367pub struct ImportedMaterial {
368    pub name: String,
369    pub base_color: [f32; 4],
370    pub metallic: f32,
371    pub roughness: f32,
372    pub base_color_texture: Option<String>,
373    pub normal_texture: Option<String>,
374    pub metallic_roughness_texture: Option<String>,
375    /// When true, the shader reads per-vertex color instead of the uniform base_color.
376    pub use_vertex_color: bool,
377    /// Alpha rendering mode from glTF
378    pub alpha_mode: AlphaMode,
379    /// Alpha cutoff for Mask mode (default 0.5)
380    pub alpha_cutoff: f32,
381    /// Light the surface gives off, as linear RGB (glTF `emissiveFactor`).
382    /// Added after lighting and before tonemapping, so it reads the same
383    /// from every side and under any light, and bloom picks it up. Zero
384    /// (the default) gives off nothing.
385    pub emissive: [f32; 3],
386    /// Multiplies `emissive`, and the per-vertex emission of
387    /// `vertex_emission`, so radiance can go past 1 (glTF
388    /// `KHR_materials_emissive_strength`). Default 1.
389    pub emissive_strength: f32,
390    /// The emissive map (glTF `emissiveTexture`), sRGB. Its rgb multiplies
391    /// `emissive` × `emissive_strength`, so painted-on lights (window grids,
392    /// signage) glow only where painted. A texture with `emissive` left at
393    /// zero gives off nothing, as in glTF. Per-vertex emission ignores it.
394    pub emissive_texture: Option<String>,
395    /// Per-vertex emission, for vertex-coloured meshes that mix lit and
396    /// glowing faces. The vertex alpha says how much of the face is
397    /// surface: 1 (every ordinary vertex colour) is lit as usual, 0 reflects
398    /// nothing and gives off its RGB times `emissive_strength` as light, and
399    /// values between blend the two. RGB may go past 1 for bright lights.
400    /// Vertex alpha is then no longer opacity; the entity material's
401    /// `opacity` still applies. Needs `use_vertex_color`.
402    pub vertex_emission: bool,
403}
404

411            metallic: 0.0,

420            emissive_strength: 1.0,
421            emissive_texture: None,
422            vertex_emission: false,
423        }
424    }
425}

427impl ImportedMaterial {
428    /// An opaque, untextured, non-metallic material that takes its colour
429    /// from the vertices, for procedural meshes coloured per vertex.

445    #[test]
446    fn reads_a_band_from_extras() {
447        let extras = serde_json::json!({
448            "flint_lod": { "near": 80, "far": 200.5, "draw_distance": 400,
449                           "ref_min": [-1, 0, -1], "ref_max": [1, 2, 1] }
450        });
451        let band = LodBand::from_extras(&extras).unwrap();
452        assert_eq!(band.near, 80.0);
453        assert_eq!(band.far, 200.5);
454        assert_eq!(band.draw_distance, 400.0);
455        assert_eq!(band.ref_box, Some(([-1.0, 0.0, -1.0], [1.0, 2.0, 1.0])));
456    }

458    #[test]
459    fn missing_keys_are_unbounded_and_other_extras_are_ignored() {
460        let band = LodBand::from_extras(&serde_json::json!({ "flint_lod": {} })).unwrap();
461        assert_eq!(band, LodBand::default());
462        assert!(LodBand::from_extras(&serde_json::json!({ "units": "m" })).is_none());
463        // a zero far means "no far", not "never draws"
464        let band = LodBand::from_extras(&serde_json::json!({ "flint_lod": { "far": 0 } })).unwrap();
465        assert!(band.far.is_infinite());
466    }

468    #[test]
469    fn adjacent_levels_hand_over_with_no_gap_and_no_overlap() {
470        let lod0 = LodBand {
471            far: 90.0,
472            ..LodBand::default()
473        };
474        let lod1 = LodBand {
475            near: 90.0,
476            far: 220.0,
477            ..LodBand::default()
478        };
479        let lod2 = LodBand {
480            near: 220.0,
481            ..LodBand::default()
482        };
483        for scale in [0.5_f32, 1.0, 1.7] {
484            for d in (0..600).map(|i| i as f32 * 0.5) {
485                let n = [lod0, lod1, lod2]
486                    .iter()
487                    .filter(|b| b.visible_at(d, scale, 1.0))
488                    .count();
489                assert_eq!(n, 1, "distance {d} at scale {scale}");
490            }
491        }
492    }

494    #[test]
