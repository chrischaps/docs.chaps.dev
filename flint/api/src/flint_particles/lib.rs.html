1//! Flint Particles - GPU-instanced particle system
2//!
3//! Provides pooled per-emitter particle simulation with:
4//! - CPU-side integration with gravity, exponential damping and forces
5//!   (wind, drag, noise turbulence, vortex, attractor)
6//! - Multi-key curves for size, colour, alpha and speed over lifetime
7//! - Burst timelines, emission over distance, velocity inheritance
8//! - Sub-emitters on birth and death
9//! - Reusable `*.particles.toml` effect assets (`particle_effect` component)
10//!   alongside the inline `particle_emitter` component
11//! - Deterministic simulation: per-emitter seeded RNG, ordered instances,
12//!   fixed-step [`ParticleSystem::simulate_to`] for headless snapshots
13//! - GPU instance packing (one shared 64-byte [`ParticleInstance`] type)
14//!
15//! Design record: ADR 0068.

48/// Default RNG seed for systems created with [`ParticleSystem::new`].
49pub const DEFAULT_SEED: u32 = 0xDEAD_BEEF;
50
51/// The particle system — implements RuntimeSystem for integration with the
52/// game loop, and exposes explicit stepping for editors and headless renders.
53pub struct ParticleSystem {
54    pub sync: ParticleSync,
55    seed: u32,
56}
57
58impl ParticleSystem {
59    pub fn new() -> Self {
60        Self::with_seed(DEFAULT_SEED)
61    }
62
63    /// Create a system whose instances derive their RNG streams from `seed`.
64    pub fn with_seed(seed: u32) -> Self {
65        Self {
66            sync: ParticleSync::new(),
67            seed,
68        }
69    }
70
71    pub fn seed(&self) -> u32 {
72        self.seed
73    }
74
75    /// Clear all particle state for a scene transition.
76    pub fn clear(&mut self) {
77        self.sync.clear();
78    }
79
80    /// One simulation step: pick up component changes, then integrate.
81    /// Does not pack instances — call [`pack`](Self::pack) once per frame
82    /// with the camera position after all stepping is done.
83    pub fn step(&mut self, world: &FlintWorld, dt: f32) {
84        self.sync.sync_from_world(world);
85        self.sync.update(dt);
86    }
87
88    /// Step detached / already-synced instances without re-reading the
89    /// world (editors driving instances directly).
90    pub fn step_detached(&mut self, dt: f32) {
91        self.sync.update(dt);
92    }
93
94    /// Deterministically advance to `seconds` from the current state in
95    /// fixed `dt` steps (the count is rounded, so `simulate_to(1.0, 1/60)`
96    /// always takes exactly 60 steps).
97    pub fn simulate_to(&mut self, world: &FlintWorld, seconds: f32, dt: f32) {
98        let dt = dt.max(1e-4);
99        let steps = (seconds / dt).round().max(0.0) as u32;
100        for _ in 0..steps {
101            self.step(world, dt);
102        }
103    }
104
105    /// Pack alive particles for upload. `camera_pos` enables
106    /// back-to-front sorting and per-emitter distance keys.
107    pub fn pack(&mut self, camera_pos: Option<[f32; 3]>) {
108        self.sync.pack_instances(camera_pos);
109    }
110}
111
112impl Default for ParticleSystem {
113    fn default() -> Self {
114        Self::new()
115    }
116}

118impl RuntimeSystem for ParticleSystem {
119    fn initialize(&mut self, world: &mut FlintWorld) -> Result<()> {
120        self.sync.sync_from_world(world);
121        let count = self.sync.emitter_count();
122        if count > 0 {
123            println!(
124                "[particles] Discovered {count} emitter(s) across {} effect instance(s)",
125                self.sync.instance_count()
126            );
127        }
128        Ok(())
129    }
130
131    fn fixed_update(&mut self, _world: &mut FlintWorld, _dt: f64) -> Result<()> {
132        // Particles are purely visual — no fixed-step needed
133        Ok(())
134    }
135
136    fn update(&mut self, world: &mut FlintWorld, dt: f64) -> Result<()> {
137        self.step(world, dt as f32);
138        Ok(())
139    }
140
141    fn shutdown(&mut self) -> Result<()> {
142        Ok(())
143    }
144
145    fn name(&self) -> &str {
146        "particles"
147    }
148}

150#[cfg(test)]
151mod tests {
152    use super::*;
153    use flint_core::components as comp;
154
155    #[test]
156    fn simulate_to_is_deterministic_and_step_counted() {
157        let build = || {
158            let mut world = FlintWorld::new();
159            let id = flint_core::EntityId(77);
160            world.spawn_with_id(id, "fx").unwrap();
161            let t: toml::Value =
162                toml::from_str("emission_rate = 120.0\nlifetime_min = 0.5\nlifetime_max = 1.0")
163                    .unwrap();
164            world.set_component(id, comp::PARTICLE_EMITTER, t).unwrap();
165            world
166        };
167        let world_a = build();
168        let world_b = build();
169        let mut a = ParticleSystem::with_seed(1);
170        let mut b = ParticleSystem::with_seed(1);
171        a.simulate_to(&world_a, 1.0, 1.0 / 60.0);
172        b.simulate_to(&world_b, 1.0, 1.0 / 60.0);
173        assert!((a.sync.time() - 1.0).abs() < 1e-4);
174        a.pack(Some([0.0, 1.0, 5.0]));
175        b.pack(Some([0.0, 1.0, 5.0]));
176        assert!(!a.sync.instance_data().is_empty());
177        assert_eq!(a.sync.instance_data(), b.sync.instance_data());
178    }
