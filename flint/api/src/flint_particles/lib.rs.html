1//! Flint Particles - GPU-instanced particle system
2//!
3//! Provides pooled per-emitter particle simulation with:
4//! - CPU-side position/velocity/lifetime integration
5//! - Swap-remove particle pool for O(1) kill
6//! - GPU instance packing for instanced draw calls
7//! - Configurable emission shapes, gravity, damping, size/color over lifetime

23/// The particle system — implements RuntimeSystem for integration with the game loop.
24pub struct ParticleSystem {
25    pub sync: ParticleSync,
26    rng: rand::ParticleRng,
27}
28
29impl ParticleSystem {
30    pub fn new() -> Self {
31        // Seed from a simple hash of the timestamp-ish value
32        let seed = 0xDEAD_BEEF_u32;
33        Self {
34            sync: ParticleSync::new(),
35            rng: rand::ParticleRng::new(seed),
36        }
37    }
38
39    /// Clear all particle state for a scene transition.
40    pub fn clear(&mut self) {
41        self.sync.clear();
42    }
43}
44
45impl Default for ParticleSystem {
46    fn default() -> Self {
47        Self::new()
48    }
49}
50
51impl RuntimeSystem for ParticleSystem {
52    fn initialize(&mut self, world: &mut FlintWorld) -> Result<()> {
53        self.sync.sync_from_world(world);
54        let count = self.sync.emitter_count();
55        if count > 0 {
56            println!("[particles] Discovered {count} emitter(s)");
57        }
58        Ok(())
59    }
60
61    fn fixed_update(&mut self, _world: &mut FlintWorld, _dt: f64) -> Result<()> {
62        // Particles are purely visual — no fixed-step needed
63        Ok(())
64    }
65
66    fn update(&mut self, world: &mut FlintWorld, dt: f64) -> Result<()> {
67        self.sync.sync_from_world(world);
68        self.sync.update(&mut self.rng, dt as f32);
69        self.sync.pack_instances();
70        Ok(())
71    }
72
73    fn shutdown(&mut self) -> Result<()> {
74        Ok(())
75    }
76
77    fn name(&self) -> &str {
78        "particles"
79    }
80}
