1//! Loading `*.particles.toml` effect assets through the vfs.
2//!
3//! Effects live in `<scene_dir>/particles/` (falling back to
4//! `<scene_dir>/../particles/` for projects that keep scenes in a
5//! subdirectory), mirroring how `animations/` holds `*.sequence.toml`.
6//! Scenes reference them by `name` through the `particle_effect` component.
7
8use crate::effect::ParticleEffect;
9use crate::ParticleSystem;
10use flint_core::vfs;
11use std::path::{Path, PathBuf};
12use std::sync::Arc;
13
14/// File suffix for effect assets.
15pub const EFFECT_SUFFIX: &str = ".particles.toml";
16
17/// Parse an effect from TOML text; `origin` labels errors.
18pub fn load_effect_from_str(text: &str, origin: &str) -> Result<ParticleEffect, String> {
19    ParticleEffect::from_toml_str(text, origin)
20}
21
22/// Load one effect file. An empty `name` falls back to the file stem
23/// (`fire.particles.toml` → `fire`).
24pub fn load_effect_from_file(path: &Path) -> Result<ParticleEffect, String> {
25    let text = vfs::read_to_string(path).map_err(|e| format!("{}: {e}", path.display()))?;
26    let origin = path.display().to_string();
27    // Allow the name to be omitted: patch it in before strict parsing.
28    let mut effect = match load_effect_from_str(&text, &origin) {
29        Ok(fx) => fx,
30        Err(e) if e.contains("missing field `name`") => {
31            let stem = effect_stem(path).unwrap_or_default();
32            let patched = format!("name = \"{stem}\"\n{text}");
33            load_effect_from_str(&patched, &origin)?
34        }
35        Err(e) => return Err(e),
36    };
37    if effect.name.is_empty() {
38        effect.name = effect_stem(path).unwrap_or_default();
39    }
40    Ok(effect)
41}

43/// `fire.particles.toml` → `Some("fire")`.
44pub fn effect_stem(path: &Path) -> Option<String> {
45    let name = path.file_name()?.to_str()?;
46    name.strip_suffix(EFFECT_SUFFIX).map(|s| s.to_string())
47}

49/// Load every `*.particles.toml` in a directory (non-recursive), returning
50/// each path with its result so callers can report failures individually.
51pub fn load_effects_from_dir(dir: &Path) -> Vec<(PathBuf, Result<ParticleEffect, String>)> {
52    let mut out = Vec::new();
53    let Ok(entries) = vfs::read_dir(dir) else {
54        return out;
55    };
56    let mut paths: Vec<PathBuf> = entries
57        .into_iter()
58        .filter(|p| {
59            p.file_name()
60                .and_then(|n| n.to_str())
61                .is_some_and(|n| n.ends_with(EFFECT_SUFFIX))
62        })
63        .collect();
64    paths.sort();
65    for path in paths {
66        let result = load_effect_from_file(&path);
67        out.push((path, result));
68    }
69    out
70}

72/// `<scene_dir>/particles/`, falling back to `<scene_dir>/../particles/`.
73pub fn resolve_particles_dir(scene_path: &str) -> Option<PathBuf> {
74    let scene_dir = Path::new(scene_path)
75        .parent()
76        .unwrap_or_else(|| Path::new("."));
77    let local = scene_dir.join("particles");
78    if vfs::is_dir(&local) {
79        return Some(local);
80    }
81    let up = scene_dir.parent()?.join("particles");
82    vfs::is_dir(&up).then_some(up)
83}

85/// Directories in which particle textures are searched, most specific
86/// first: `particles/`, the scene dir, then its parent (game root).
87pub fn texture_search_dirs(scene_path: &str) -> Vec<PathBuf> {
88    let scene_dir = Path::new(scene_path)
89        .parent()
90        .unwrap_or_else(|| Path::new("."))
91        .to_path_buf();
92    let mut dirs = Vec::new();
93    if let Some(p) = resolve_particles_dir(scene_path) {
94        dirs.push(p);
95    }
96    dirs.push(scene_dir.clone());
97    if let Some(parent) = scene_dir.parent() {
98        dirs.push(parent.to_path_buf());
99    }
100    dirs
101}

103/// Register every effect found next to `scene_path` with the system.
104/// Returns the number registered; failures are logged and skipped.
105pub fn load_particle_effects_from_world(scene_path: &str, system: &mut ParticleSystem) -> usize {
106    let Some(dir) = resolve_particles_dir(scene_path) else {
107        return 0;
108    };
109    let mut n = 0;
110    for (path, result) in load_effects_from_dir(&dir) {
111        match result {
112            Ok(fx) => {
113                println!(
114                    "Loaded particle effect: {} ({} emitter{})",
115                    fx.name,
116                    fx.emitters.len(),
117                    if fx.emitters.len() == 1 { "" } else { "s" }
118                );
119                system.sync.register_effect(Arc::new(fx));
120                n += 1;
121            }
122            Err(e) => tracing::warn!("Failed to load particle effect '{}': {e}", path.display()),
123        }
124    }
125    n
126}

132    fn demo_dir() -> PathBuf {
133        Path::new(env!("CARGO_MANIFEST_DIR"))
134            .join("../../demo")
135            .canonicalize()
136            .expect("demo dir")

139    #[test]
140    fn effect_stem_strips_suffix() {
141        assert_eq!(
142            effect_stem(Path::new("a/b/fire.particles.toml")),
143            Some("fire".into())
144        );
145        assert_eq!(effect_stem(Path::new("fire.toml")), None);
146    }
147
148    #[test]
149    fn resolves_particles_dir_next_to_demo_scene() {
150        let scene = demo_dir().join("particles_demo.scene.toml");
151        let dir = resolve_particles_dir(scene.to_str().unwrap()).expect("demo/particles exists");
152        assert!(dir.ends_with("particles"));
153        let dirs = texture_search_dirs(scene.to_str().unwrap());
154        assert_eq!(dirs[0], dir);
155    }
156
157    #[test]
158    fn loads_demo_campfire_effect() {
159        let path = demo_dir().join("particles/campfire.particles.toml");
160        let fx = load_effect_from_file(&path).expect("campfire parses");
161        assert_eq!(fx.name, "campfire");
162        assert!(fx.emitters.len() >= 3);
163        assert!(fx.emitter_index("flames").is_some());
164        fx.validate().unwrap();
165    }
166
167    #[test]
168    fn loads_every_demo_effect() {
169        let results = load_effects_from_dir(&demo_dir().join("particles"));
170        assert!(!results.is_empty());
171        for (path, r) in results {
172            r.unwrap_or_else(|e| panic!("{}: {e}", path.display()));
173        }
174    }

176    #[test]
177    fn missing_name_falls_back_to_stem() {
178        let dir = std::env::temp_dir().join(format!("flint_fx_{}", std::process::id()));
179        std::fs::create_dir_all(&dir).unwrap();
180        let path = dir.join("glow.particles.toml");
181        std::fs::write(&path, "[[emitters]]\nname = \"a\"\n").unwrap();
182        let fx = load_effect_from_file(&path).unwrap();
183        assert_eq!(fx.name, "glow");
184        let _ = std::fs::remove_dir_all(&dir);
