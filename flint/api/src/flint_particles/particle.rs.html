1//! Particle types: CPU simulation state and GPU instance data
2
3use bytemuck::{Pod, Zeroable};
4
5/// CPU-side particle state (not sent to GPU)
6#[derive(Clone)]
7pub struct Particle {
8    pub position: [f32; 3],
9    pub velocity: [f32; 3],
10    pub age: f32,
11    pub lifetime: f32,
12    pub size: f32,
13    pub rotation: f32,
14    pub color: [f32; 4],
15    pub frame: u32,
16    pub alive: bool,
17}
18
19impl Particle {
20    pub fn dead() -> Self {
21        Self {
22            position: [0.0; 3],
23            velocity: [0.0; 3],
24            age: 0.0,
25            lifetime: 0.0,
26            size: 0.0,
27            rotation: 0.0,
28            color: [0.0; 4],
29            frame: 0,
30            alive: false,
31        }
32    }
33
34    /// Normalized age in [0, 1]
35    pub fn age_ratio(&self) -> f32 {
36        if self.lifetime <= 0.0 {
37            1.0
38        } else {
39            (self.age / self.lifetime).min(1.0)
40        }
41    }
42}

44/// GPU instance data — matches WGSL `ParticleInstance` struct.
45/// 48 bytes, 16-byte aligned (3 rows of vec4).
46#[repr(C)]
47#[derive(Copy, Clone, Debug, Pod, Zeroable)]
48pub struct ParticleInstance {
49    /// World position + size packed into vec4
50    pub pos_size: [f32; 4], // xyz = position, w = size
51    /// Color with alpha
52    pub color: [f32; 4], // rgba
53    /// Rotation, sprite frame, sprite sheet dimensions
54    pub rotation_frame: [f32; 4], // x = rotation, y = frame, z = frames_x, w = frames_y
55}
56
57impl ParticleInstance {
58    pub fn from_particle(p: &Particle, frames_x: u32, frames_y: u32) -> Self {
59        Self {
60            pos_size: [p.position[0], p.position[1], p.position[2], p.size],
61            color: p.color,
62            rotation_frame: [p.rotation, p.frame as f32, frames_x as f32, frames_y as f32],
63        }
64    }
65}

67/// Swap-remove pool for O(1) particle kill and contiguous alive iteration.
68pub struct ParticlePool {
69    particles: Vec<Particle>,
70    alive_count: usize,
71}
72
73impl ParticlePool {
74    pub fn new(capacity: usize) -> Self {
75        let mut particles = Vec::with_capacity(capacity);
76        for _ in 0..capacity {
77            particles.push(Particle::dead());
78        }
79        Self {
80            particles,
81            alive_count: 0,
82        }
83    }
84
85    pub fn capacity(&self) -> usize {
86        self.particles.len()
87    }
88
89    pub fn alive_count(&self) -> usize {
90        self.alive_count
91    }
92
93    /// Spawn one particle, returning a mutable ref to initialize it.
94    /// Returns None if pool is full.
95    pub fn spawn(&mut self) -> Option<&mut Particle> {
96        if self.alive_count >= self.particles.len() {
97            return None;
98        }
99        let idx = self.alive_count;
100        self.particles[idx].alive = true;
101        self.alive_count += 1;
102        Some(&mut self.particles[idx])
103    }
104
105    /// Iterate alive particles, kill expired ones via swap-remove.
106    pub fn update_and_compact(&mut self) {
107        let mut i = 0;
108        while i < self.alive_count {
109            if !self.particles[i].alive || self.particles[i].age >= self.particles[i].lifetime {
110                self.particles[i].alive = false;
111                self.alive_count -= 1;
112                if i < self.alive_count {
113                    self.particles.swap(i, self.alive_count);
114                }
115                // Don't increment i — the swapped-in particle needs checking
116            } else {
117                i += 1;
118            }
119        }
120    }
121
122    /// Access alive particles slice for reading (first `alive_count` elements)
123    pub fn alive_slice(&self) -> &[Particle] {
124        &self.particles[..self.alive_count]
125    }
126
127    /// Access alive particles mutably
128    pub fn alive_slice_mut(&mut self) -> &mut [Particle] {
129        &mut self.particles[..self.alive_count]
130    }
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136
137    #[test]
138    fn pool_spawn_and_kill() {
139        let mut pool = ParticlePool::new(4);
140        assert_eq!(pool.alive_count(), 0);
141
142        // Spawn 3 particles
143        for i in 0..3 {
144            let p = pool.spawn().unwrap();
145            p.lifetime = 1.0;
146            p.age = 0.0;
147            p.position[0] = i as f32;
148        }
149        assert_eq!(pool.alive_count(), 3);
150
151        // Kill the middle one by aging it past lifetime
152        pool.alive_slice_mut()[1].age = 2.0;
153        pool.update_and_compact();
154        assert_eq!(pool.alive_count(), 2);
155
156        // Pool full at capacity 4 — spawn should fail after 4
157        pool.spawn().unwrap();
158        pool.spawn().unwrap();
159        assert!(pool.spawn().is_none());
160    }
161
162    #[test]
163    fn particle_instance_layout() {
164        assert_eq!(std::mem::size_of::<ParticleInstance>(), 48);
165        assert_eq!(std::mem::align_of::<ParticleInstance>(), 4);
166    }
