1//! Deterministic hash-based 3D gradient noise for turbulence forces.
2//!
3//! No tables, no allocation: each lattice corner's gradient comes from an
4//! integer hash of its coordinates and the caller's seed, so two runs with
5//! the same seed produce identical fields. Curl noise is deliberately not
6//! implemented (six extra samples per particle); see ADR 0068.
7
8#[inline]
9fn hash3(x: i32, y: i32, z: i32, seed: u32) -> u32 {
10    let mut h = (x as u32).wrapping_mul(0x8DA6_B343)
11        ^ (y as u32).wrapping_mul(0xD816_3841)
12        ^ (z as u32).wrapping_mul(0xCB1A_B31F)
13        ^ seed.wrapping_mul(0x9E37_79B9);
14    h ^= h >> 15;
15    h = h.wrapping_mul(0x2C1B_3C6D);
16    h ^= h >> 12;
17    h = h.wrapping_mul(0x297A_2D39);
18    h ^= h >> 15;
19    h
20}

22/// Pseudo-random unit-ish gradient at a lattice corner.
23#[inline]
24fn gradient(x: i32, y: i32, z: i32, seed: u32) -> [f32; 3] {
25    let h = hash3(x, y, z, seed);
26    // Three signed components from separate bit ranges, in [-1, 1].
27    let gx = ((h & 0x3FF) as f32 / 511.5) - 1.0;
28    let gy = (((h >> 10) & 0x3FF) as f32 / 511.5) - 1.0;
29    let gz = (((h >> 20) & 0x3FF) as f32 / 511.5) - 1.0;
30    [gx, gy, gz]
31}

33#[inline]
34fn fade(t: f32) -> f32 {
35    t * t * t * (t * (t * 6.0 - 15.0) + 10.0)
36}

38#[inline]
39fn lerp(a: f32, b: f32, t: f32) -> f32 {
40    a + (b - a) * t
41}

43/// Perlin-style gradient noise in roughly `[-1, 1]`.
44pub fn noise3(p: [f32; 3], seed: u32) -> f32 {
45    let xi = p[0].floor();
46    let yi = p[1].floor();
47    let zi = p[2].floor();
48    let (x0, y0, z0) = (xi as i32, yi as i32, zi as i32);
49    let fx = p[0] - xi;
50    let fy = p[1] - yi;
51    let fz = p[2] - zi;
52    let u = fade(fx);
53    let v = fade(fy);
54    let w = fade(fz);
55
56    let dot = |cx: i32, cy: i32, cz: i32, dx: f32, dy: f32, dz: f32| {
57        let g = gradient(cx, cy, cz, seed);
58        g[0] * dx + g[1] * dy + g[2] * dz
59    };
60
61    let n000 = dot(x0, y0, z0, fx, fy, fz);
62    let n100 = dot(x0 + 1, y0, z0, fx - 1.0, fy, fz);
63    let n010 = dot(x0, y0 + 1, z0, fx, fy - 1.0, fz);
64    let n110 = dot(x0 + 1, y0 + 1, z0, fx - 1.0, fy - 1.0, fz);
65    let n001 = dot(x0, y0, z0 + 1, fx, fy, fz - 1.0);
66    let n101 = dot(x0 + 1, y0, z0 + 1, fx - 1.0, fy, fz - 1.0);
67    let n011 = dot(x0, y0 + 1, z0 + 1, fx, fy - 1.0, fz - 1.0);
68    let n111 = dot(x0 + 1, y0 + 1, z0 + 1, fx - 1.0, fy - 1.0, fz - 1.0);
69
70    let nx00 = lerp(n000, n100, u);
71    let nx10 = lerp(n010, n110, u);
72    let nx01 = lerp(n001, n101, u);
73    let nx11 = lerp(n011, n111, u);
74    let nxy0 = lerp(nx00, nx10, v);
75    let nxy1 = lerp(nx01, nx11, v);
76    // Gradient noise with unit-cube gradients peaks near ±0.87; rescale.
77    (lerp(nxy0, nxy1, w) * 1.15).clamp(-1.0, 1.0)
78}

80/// Fractal sum of `octaves` noise layers (each doubling frequency, halving
81/// amplitude), normalised back into roughly `[-1, 1]`.
82pub fn fbm3(p: [f32; 3], seed: u32, octaves: u32) -> f32 {
83    let octaves = octaves.clamp(1, 6);
84    let mut sum = 0.0;
85    let mut amp = 1.0;
86    let mut norm = 0.0;
87    let mut q = p;
88    for i in 0..octaves {
89        sum += noise3(q, seed.wrapping_add(i * 0x51_7C_C1)) * amp;
90        norm += amp;
91        amp *= 0.5;
92        q = [q[0] * 2.0 + 17.3, q[1] * 2.0 - 9.1, q[2] * 2.0 + 4.7];
93    }
94    sum / norm
95}

97/// A vector field: three decorrelated noise samples, one per axis.
98pub fn noise3_vec(p: [f32; 3], seed: u32, octaves: u32) -> [f32; 3] {
99    [
100        fbm3(p, seed, octaves),
101        fbm3(
102            [p[0] + 31.7, p[1] - 12.3, p[2] + 5.9],
103            seed ^ 0x5F3_759D,
104            octaves,
105        ),
106        fbm3(
107            [p[0] - 7.1, p[1] + 23.9, p[2] - 41.3],
108            seed ^ 0x2545_F491,
109            octaves,
110        ),
111    ]
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    #[test]
119    fn noise_is_deterministic_and_bounded() {
120        let a = noise3([0.3, 1.7, -2.2], 7);
121        let b = noise3([0.3, 1.7, -2.2], 7);
122        assert_eq!(a, b);
123        for i in 0..500 {
124            let f = i as f32 * 0.137;
125            let p = [f.sin() * 10.0, f * 0.7, (f * 1.3).cos() * 5.0];
126            let n = noise3(p, 11);
127            assert!((-1.0..=1.0).contains(&n), "{n} out of range");
128            let v = noise3_vec(p, 11, 3);
129            for c in v {
130                assert!((-1.0..=1.0).contains(&c));
131            }
132        }
133    }
134
135    #[test]
136    fn seed_changes_field() {
137        let p = [1.5, 2.5, 3.5];
138        assert_ne!(noise3(p, 1), noise3(p, 2));
139    }
140
141    #[test]
142    fn noise_is_continuous() {
143        let p = [0.9999, 0.5, 0.5];
144        let q = [1.0001, 0.5, 0.5];
145        assert!((noise3(p, 3) - noise3(q, 3)).abs() < 0.01);
146    }
