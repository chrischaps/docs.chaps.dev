1//! The on-screen card for user-authored styles that don't compile (#275,
2//! ADR 0112). A broken `styles/<name>.wgsl` never takes the frame down: the
3//! style draws its last good build, or its input unchanged, and this card
4//! says why, in the corner, until the file is fixed and saved. The player
5//! and the scene viewer both draw it, over everything else.
6
7use egui::{Color32, FontId, RichText};
8
9/// The most lines of one error the card shows; the console has the rest.
10const MAX_LINES: usize = 8;
11
12/// Paint the card for `errors` (`(style, message)`, as
13/// `flint_render::stylize::custom::errors` gives them). Nothing when the
14/// list is empty. `bottom_inset` lifts it above anything the host keeps at
15/// the bottom edge.
16pub fn paint_style_errors(ctx: &egui::Context, errors: &[(String, String)], bottom_inset: f32) {
17    if errors.is_empty() {
18        return;
19    }
20    // The HUD blends in display space: translucent colours are built
21    // premultiplied by hand (see CLAUDE.md on egui 0.30's helpers)
22    let ground = Color32::from_rgba_premultiplied(14, 10, 12, 230);
23    let ember = Color32::from_rgb(255, 122, 89);
24    let ink = Color32::from_rgb(236, 226, 214);
25    let quiet = Color32::from_rgb(160, 150, 142);
26    egui::Area::new(egui::Id::new("flint_style_errors"))
27        .order(egui::Order::Foreground)
28        .anchor(egui::Align2::LEFT_BOTTOM, [12.0, -12.0 - bottom_inset])
29        .interactable(false)
30        .show(ctx, |ui| {
31            egui::Frame::none()
32                .fill(ground)
33                .stroke(egui::Stroke::new(1.0_f32, ember.gamma_multiply(0.6)))
34                .rounding(4.0)
35                .inner_margin(egui::Margin::symmetric(12.0, 9.0))
36                .show(ui, |ui| {
37                    ui.set_max_width(ctx.screen_rect().width().min(760.0) - 24.0);
38                    for (i, (style, message)) in errors.iter().enumerate() {
39                        if i > 0 {
40                            ui.add_space(6.0);
41                        }
42                        ui.label(
43                            RichText::new(format!("style \u{2018}{style}\u{2019} doesn't compile"))
44                                .color(ember)
45                                .font(FontId::proportional(14.0))
46                                .strong(),
47                        );
48                        let lines: Vec<&str> = message.lines().collect();
49                        for line in lines.iter().take(MAX_LINES) {
50                            ui.label(
51                                RichText::new(*line)
52                                    .color(ink)
53                                    .font(FontId::monospace(12.0)),
54                            );
55                        }
56                        if lines.len() > MAX_LINES {
57                            ui.label(
58                                RichText::new(format!(
59                                    "\u{2026} {} more lines in the log",
60                                    lines.len() - MAX_LINES
61                                ))
62                                .color(quiet)
63                                .font(FontId::monospace(11.0)),
64                            );
65                        }
66                    }
67                    ui.add_space(4.0);
68                    ui.label(
69                        RichText::new(
70                            "Its last good build (or its input, unchanged) draws meanwhile. \
71                             Save the file to try again.",
72                        )
73                        .color(quiet)
74                        .font(FontId::proportional(11.5)),
75                    );
76                });
77        });
78}
