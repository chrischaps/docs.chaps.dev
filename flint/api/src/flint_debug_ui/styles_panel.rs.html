1//! Styles panel (F3, #268): the scene's stylized look among the other
2//! scene panels. The same [`StylesEditor`] the F4 Rendering & Effects menu
3//! draws, plus the scene's styled entities: each entity with a
4//! `style_mask` component, the style it asks for, and the hue the mask
5//! view (`--debug-mode style`) paints that style in.
6//!
7//! The host creates it when the scene uses a style (a
8//! `[post_process.styles]` table or a `style_mask` entity), mirrors the
9//! renderer's config into it while it is clean, and while it is dirty
10//! copies its `pp.styles` (and nothing else) back into the renderer's
11//! config, as the F4 panel does for the whole config (ADR 0053).
12
13use crate::styles_ui::{StylesEdit, StylesEditor};
14use crate::DebugPanel;
15use flint_render::stylize::StyleSettings;
16use flint_render::PostProcessConfig;
17
18pub const STYLES_DEBUG_PANEL: &str = "Styles";
19
20/// One styled entity, as the host found it.
21#[derive(Debug, Clone, PartialEq)]
22pub struct StyledEntityRow {
23    pub name: String,
24    /// The style its `style_mask` names.
25    pub style: String,
26    /// The component's `enabled`.
27    pub enabled: bool,
28    /// The mask view's hue for the style's id this frame, when it has one.
29    pub color: Option<[f32; 3]>,
30}
31
32pub struct StylesDebugPanel {
33    /// The renderer's config, mirrored; only `styles` is written back.
34    pub pp: PostProcessConfig,
35    authored: Option<StyleSettings>,
36    entities: Vec<StyledEntityRow>,
37    editor: StylesEditor,
38    edit: StylesEdit,
39    open: bool,
40    dirty: bool,
41}
42
43impl StylesDebugPanel {
44    /// `authored`: the scene's own `[post_process.styles]`, for the
45    /// editor's Authored button and its markers.
46    pub fn new(authored: Option<StyleSettings>) -> Self {
47        Self {
48            pp: PostProcessConfig::default(),
49            authored,
50            entities: Vec::new(),
51            editor: StylesEditor::default(),
52            edit: StylesEdit::default(),
53            open: false,
54            dirty: false,
55        }
56    }
57
58    /// Mirror the renderer's config and the styled entities. No-op on the
59    /// config while there are unapplied edits.
60    pub fn refresh(&mut self, pp: &PostProcessConfig, entities: Vec<StyledEntityRow>) {
61        if !self.dirty {
62            self.pp = pp.clone();
63        }
64        self.entities = entities;
65    }
66
67    /// What the last edit asks of the host.
68    pub fn take_edit(&mut self) -> StylesEdit {
69        std::mem::take(&mut self.edit)
70    }
71
72    /// The styled entities shown, as last refreshed.
73    pub fn entities(&self) -> &[StyledEntityRow] {
74        &self.entities
75    }
76}
77
78impl DebugPanel for StylesDebugPanel {
79    fn name(&self) -> &str {
80        STYLES_DEBUG_PANEL
81    }
82
83    fn ui(&mut self, ui: &mut egui::Ui) {
84        let edit = self.editor.ui(ui, &mut self.pp, self.authored.as_ref());
85        if edit.changed {
86            self.edit.changed = true;
87            self.edit.needs_resources |= edit.needs_resources;
88            self.dirty = true;
89        }
90        ui.separator();
91        egui::CollapsingHeader::new(format!("Styled entities ({})", self.entities.len()))
92            .default_open(!self.entities.is_empty())
93            .show(ui, |ui| {
94                if self.entities.is_empty() {
95                    ui.weak("No entity has a style_mask: styles apply to the whole frame");
96                }
97                egui::Grid::new("styled entities")
98                    .striped(true)
99                    .min_col_width(0.0)
100                    .show(ui, |ui| {
101                        for row in &self.entities {
102                            match row.color {
103                                Some([r, g, b]) => {
104                                    let (rect, _) = ui.allocate_exact_size(
105                                        egui::vec2(10.0, 10.0),
106                                        egui::Sense::hover(),
107                                    );
108                                    // The mask view paints these display values
109                                    let byte = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
110                                    ui.painter().rect_filled(
111                                        rect,
112                                        2.0,
113                                        egui::Color32::from_rgb(byte(r), byte(g), byte(b)),
114                                    );
115                                }
116                                None => {
117                                    ui.label("");
118                                }
119                            }
120                            ui.label(&row.name);
121                            let style = if row.enabled {
122                                egui::RichText::new(&row.style)
123                            } else {
124                                egui::RichText::new(format!("{} (off)", row.style)).weak()
125                            };
126                            ui.label(style);
127                            ui.end_row();
128                        }
129                    });
130                if self.entities.iter().any(|r| r.color.is_some()) {
131                    ui.weak(
132                        "Swatches: the mask view's hue for each style (Debug mode -> Style Mask)",
133                    );
134                }
135            });
136    }
137
138    fn is_open(&self) -> bool {
139        self.open
140    }
141    fn toggle(&mut self) {
142        self.open = !self.open;
143    }
144    fn is_dirty(&self) -> bool {
145        self.dirty
146    }
147    fn clear_dirty(&mut self) {
148        self.dirty = false;
149    }
150    fn as_any(&self) -> &dyn std::any::Any {
151        self
152    }
153    fn as_any_mut(&mut self) -> &mut dyn std::any::Any {
154        self
155    }
156}
