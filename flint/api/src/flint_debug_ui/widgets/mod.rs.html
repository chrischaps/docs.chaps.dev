1//! Shared egui widgets for debug panels and editors.
2//!
3//! Every helper returns `true` when the value changed, so callers can
4//! compose them into a single `dirty` flag. The panels used to carry their
5//! own copies of `drag_f32`/`drag_vec3`; they now live here (ADR 0068).

11/// Single labeled drag value. Returns true if the value changed.
12pub fn drag_f32(
13    ui: &mut egui::Ui,
14    label: &str,
15    value: &mut f32,
16    speed: f64,
17    range: std::ops::RangeInclusive<f64>,
18) -> bool {
19    let before = *value;
20    ui.horizontal(|ui| {
21        ui.label(label);
22        ui.add(
23            egui::DragValue::new(value)
24                .speed(speed)
25                .range(range)
26                .max_decimals(3),
27        );
28    });
29    *value != before
30}

47/// Labeled `min`/`max` pair on one row. Keeps `min <= max`. Returns true if
48/// either changed.
49pub fn drag_range_f32(
50    ui: &mut egui::Ui,
51    label: &str,
52    min: &mut f32,
53    max: &mut f32,
54    speed: f64,
55    range: std::ops::RangeInclusive<f64>,
56) -> bool {
57    let before = (*min, *max);
58    ui.horizontal(|ui| {
59        ui.label(label);
60        ui.add(
61            egui::DragValue::new(min)
62                .speed(speed)
63                .range(range.clone())
64                .max_decimals(3),
65        );
66        ui.label("-");
67        ui.add(
68            egui::DragValue::new(max)
69                .speed(speed)
70                .range(range)
71                .max_decimals(3),
72        );
73    });
74    if *min > *max {
75        // Whichever the user dragged wins; drag the other along.
76        if (*min - before.0).abs() > (*max - before.1).abs() {
77            *max = *min;
78        } else {
79            *min = *max;
80        }
81    }
82    (*min, *max) != before
83}

107fn drag_vec3_labeled(
108    ui: &mut egui::Ui,
109    label: &str,
110    names: [&str; 3],
111    value: &mut [f32; 3],
112    speed: f64,
113    range: std::ops::RangeInclusive<f64>,
114) -> bool {
115    let before = *value;
116    ui.horizontal(|ui| {
117        ui.label(label);
118        for (i, n) in names.iter().enumerate() {
119            ui.label(*n);
120            ui.add(
121                egui::DragValue::new(&mut value[i])
122                    .speed(speed)
123                    .range(range.clone())
124                    .max_decimals(3),
125            );
126        }
127    });
128    *value != before
129}

170/// Labeled combo box over string options. Returns true if the selection changed.
171pub fn combo_str(ui: &mut egui::Ui, label: &str, value: &mut String, options: &[&str]) -> bool {
172    let before = value.clone();
173    ui.horizontal(|ui| {
174        if !label.is_empty() {
175            ui.label(label);
176        }
177        // Salt with the auto id so two combos with the same (or empty)
178        // label in one panel do not share popup state.
179        egui::ComboBox::from_id_salt(ui.next_auto_id().with(label))
180            .selected_text(value.as_str())
181            .show_ui(ui, |ui| {
182                for opt in options {
183                    ui.selectable_value(value, (*opt).to_string(), *opt);
184                }
185            });
186    });
187    *value != before
188}
