1//! Dead Calm panel (F3 family).
2//!
3//! Controls a game-side `dead_calm` component (convention: a calm script
4//! schedules a rare ocean-stillness event, publishes `phase` 0-3, `calm`
5//! 0..1 and `next_in_s`, and honors one-shot `trigger_calm` / `end_now`
6//! flags). The panel is created only when the component exists, so games
7//! without a dead calm never see it.
8//!
9//! The script owns the published fields; the panel shows them read-only via
10//! [`DeadCalmDebugPanel::sync_status`] and never writes them. The buttons
11//! queue one-shot pushes (see `take_one_shots`) that the script consumes
12//! and resets after acting on them. All tuning lives in the scene TOML —
13//! this panel is deliberately just a status line and two doorbells.
14
15use crate::DebugPanel;
16
17pub const DEAD_CALM_PHASE_NAMES: [&str; 4] =
18    ["Idle", "Falling calm", "Dead calm", "Life returning"];
19
20pub struct DeadCalmDebugPanel {
21    calm_entity_name: String,
22    open: bool,
23    // Read-only status published by the calm script.
24    phase: f32,
25    calm: f32,
26    next_in_s: f32,
27    // One-shot flags queued by the buttons, drained by the host each frame.
28    pending_trigger: bool,
29    pending_end: bool,
30}
31
32impl DeadCalmDebugPanel {
33    pub fn new(calm_entity_name: String) -> Self {
34        Self {
35            calm_entity_name,
36            open: false,
37            phase: 0.0,
38            calm: 0.0,
39            next_in_s: -1.0,
40            pending_trigger: false,
41            pending_end: false,
42        }
43    }
44
45    pub fn entity_name(&self) -> &str {
46        &self.calm_entity_name
47    }
48
49    /// Update the read-only displays from the script-published fields.
50    pub fn sync_status(&mut self, phase: f32, calm: f32, next_in_s: f32) {
51        self.phase = phase;
52        self.calm = calm;
53        self.next_in_s = next_in_s;
54    }
55
56    /// Drain the queued one-shot flags as component field writes.
57    pub fn take_one_shots(&mut self) -> Vec<(&'static str, toml::Value)> {
58        let mut fields = Vec::new();
59        if self.pending_trigger {
60            self.pending_trigger = false;
61            fields.push(("trigger_calm", toml::Value::Float(1.0)));
62        }
63        if self.pending_end {
64            self.pending_end = false;
65            fields.push(("end_now", toml::Value::Boolean(true)));
66        }
67        fields
68    }
69}
70
71impl DebugPanel for DeadCalmDebugPanel {
72    fn name(&self) -> &str {
73        "Dead Calm"
74    }
75
76    fn ui(&mut self, ui: &mut egui::Ui) {
77        let phase_i = (self.phase.round() as usize).min(3);
78        let next = if self.next_in_s >= 0.0 {
79            format!("next in {:.0}s", self.next_in_s)
80        } else {
81            "no attempt scheduled".to_string()
82        };
83        ui.label(format!(
84            "{}  ·  calm {:.2}  ·  {}",
85            DEAD_CALM_PHASE_NAMES[phase_i], self.calm, next
86        ));
87        ui.separator();
88        // The script starts only from Idle and ends only from an active
89        // calm; stray presses are consumed and ignored, so neither button
90        // needs disabling.
91        ui.horizontal(|ui| {
92            if ui.button("Calm now").clicked() {
93                self.pending_trigger = true;
94            }
95            if ui.button("End now").clicked() {
96                self.pending_end = true;
97            }
98        });
99    }
100
101    fn is_open(&self) -> bool {
102        self.open
103    }
104    fn toggle(&mut self) {
105        self.open = !self.open;
106    }
107    fn is_dirty(&self) -> bool {
108        false
109    }
110    fn clear_dirty(&mut self) {}
111    fn as_any(&self) -> &dyn std::any::Any {
112        self
113    }
114    fn as_any_mut(&mut self) -> &mut dyn std::any::Any {
115        self
116    }
117}
