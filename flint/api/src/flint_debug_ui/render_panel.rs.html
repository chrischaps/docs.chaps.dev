1//! Rendering & Effects panel (F4).
2//!
3//! One consolidated home for every render/post-effect debug control that
4//! used to live on scattered function keys (F1 debug-mode cycle, F4 shadows,
5//! F5-F10 per-effect toggles, F12 kuwahara / DoF-follow), plus the
6//! non-binary parameters those keys could never expose (SSAO radius/samples,
7//! DoF focus/range, grade lift/gamma/gain, render mode/mix/params, shadow
8//! resolution, lighting levers, FOV).
9//!
10//! Ownership model (ADR 0053): the host refreshes the panel from live
11//! renderer state every frame while the panel is clean ("live mirror"), and
12//! applies the panel's state back when the user edits something
13//! ("write-through"), routing expensive operations by the per-group change
14//! flags. Fields a script drives are visibly reclaimed next frame unless the
15//! player-only "freeze script post overrides" switch is on — the panel shows
16//! the truth about who owns a value rather than pretending to.
17
18use crate::DebugPanel;
19use flint_render::{DebugMode, LightingLevers, PostProcessConfig};

25/// Per-group change flags the host consumes on apply. Groups map to distinct
26/// renderer calls (post config upload vs shadow-pass rebuild vs lighting
27/// setters vs debug-mode pipeline swap vs camera FOV).
28#[derive(Debug, Clone, Copy, Default)]
29pub struct RenderPanelFlags {
30    pub pp_changed: bool,
31    pub mode_changed: bool,
32    pub shadows_changed: bool,
33    pub shadow_res_changed: bool,
34    pub lighting_changed: bool,
35    pub lighting_reset: bool,
36    pub fov_changed: bool,
37    /// Kuwahara was just enabled — host must call ensure_kuwahara_resources.
38    pub kuwahara_needs_resources: bool,
39    /// FXAA was just enabled — host must call ensure_fxaa_resources.
40    pub fxaa_needs_resources: bool,
41}
42
43impl RenderPanelFlags {
44    fn any(&self) -> bool {
45        self.pp_changed
46            || self.mode_changed
47            || self.shadows_changed
48            || self.shadow_res_changed
49            || self.lighting_changed
50            || self.lighting_reset
51            || self.fov_changed
52    }
53}

55pub struct RenderDebugPanel {
56    /// Edited in place; uploaded wholesale by the host on pp_changed.
57    pub pp: PostProcessConfig,
58    pub debug_mode: DebugMode,
59    pub shadows_enabled: bool,
60    pub shadow_resolution: u32,
61    pub lighting: LightingLevers,
62    pub fov_deg: f32,
63    /// Player-only: skip the script/ladder post-override stamp while true so
64    /// panel edits to contended fields stick. Shown only when `show_freeze`.
65    pub freeze_scripts: bool,
66    show_freeze: bool,
67    flags: RenderPanelFlags,
68    open: bool,

73    /// `show_freeze`: true in the player (scripts drive post fields there),
74    /// false in the viewer (no scripts, nothing to freeze).
75    pub fn new(show_freeze: bool) -> Self {
76        Self {
77            pp: PostProcessConfig::default(),
78            debug_mode: DebugMode::Pbr,
79            shadows_enabled: true,
80            shadow_resolution: 2048,
81            lighting: LightingLevers {
82                ambient_sky: [0.0; 3],
83                ambient_ground: [0.0; 3],
84                diffuse_wrap: 0.0,
85                oren_nayar: 0.0,
86                sheen_color: [0.0; 3],
87                sheen_strength: 0.0,
88            },
89            fov_deg: 65.0,
90            freeze_scripts: false,
91            show_freeze,
92            flags: RenderPanelFlags::default(),
93            open: false,
94            dirty: false,
95        }
96    }

98    /// Mirror live renderer state into the widgets. No-op while the panel
99    /// has unapplied edits so an in-progress drag is never yanked; never
100    /// sets dirty (the mirror is not an edit).
101    #[allow(clippy::too_many_arguments)]
102    pub fn refresh(
103        &mut self,
104        pp: &PostProcessConfig,
105        debug_mode: DebugMode,
106        shadows_enabled: bool,
107        shadow_resolution: u32,
108        lighting: LightingLevers,
109        fov_deg: f32,
110    ) {
111        if self.dirty {
112            return;
113        }
114        self.pp = pp.clone();
115        self.debug_mode = debug_mode;
116        self.shadows_enabled = shadows_enabled;
117        self.shadow_resolution = shadow_resolution;
118        self.lighting = lighting;
119        self.fov_deg = fov_deg;
120    }
121
122    /// Consume the per-group change flags (host apply pass).
123    pub fn take_flags(&mut self) -> RenderPanelFlags {
124        std::mem::take(&mut self.flags)
125    }

419fn fov_label(fov_deg: f32) -> &'static str {
420    match fov_deg {
421        f if f < 50.0 => "narrow",
422        f if f < 72.0 => "natural",
423        f if f < 90.0 => "wide",
424        _ => "fisheye",
425    }
426}

428fn section(ui: &mut egui::Ui, title: &str, default_open: bool, add: impl FnOnce(&mut egui::Ui)) {
429    egui::CollapsingHeader::new(title)
430        .default_open(default_open)
431        .show(ui, add);
432}

434fn slider(
435    ui: &mut egui::Ui,
436    label: &str,
437    value: &mut f32,
438    range: std::ops::RangeInclusive<f32>,
439) -> bool {
440    ui.horizontal(|ui| {
441        ui.label(label);
442        ui.add(egui::Slider::new(value, range)).changed()
443    })
444    .inner
445}

463        ui.label(label);

469fn color_row(ui: &mut egui::Ui, label: &str, rgb: &mut [f32; 3]) -> bool {
470    ui.horizontal(|ui| {
471        ui.label(label);
472        ui.color_edit_button_rgb(rgb).changed()
473    })
474    .inner
475}
476
477fn rgb_drags(
478    ui: &mut egui::Ui,
479    label: &str,
480    rgb: &mut [f32; 3],
481    range: std::ops::RangeInclusive<f32>,
482    speed: f32,
483) -> bool {
484    ui.horizontal(|ui| {
485        ui.label(label);
486        let mut changed = false;
487        for v in rgb.iter_mut() {
488            changed |= ui
489                .add(egui::DragValue::new(v).speed(speed).range(range.clone()))
490                .changed();
491        }
492        changed
493    })
494    .inner
495}
