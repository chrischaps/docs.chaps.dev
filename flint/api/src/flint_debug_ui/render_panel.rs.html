1//! Rendering & Effects panel (F4).
2//!
3//! One consolidated home for every render/post-effect debug control that
4//! used to live on scattered function keys (F1 debug-mode cycle, F4 shadows,
5//! F5-F10 per-effect toggles, F12 kuwahara / DoF-follow), plus the
6//! non-binary parameters those keys could never expose (SSAO radius/samples,
7//! DoF focus/range, grade lift/gamma/gain, render mode/mix/params, shadow
8//! resolution, lighting levers, FOV).
9//!
10//! Ownership model (ADR 0053): the host refreshes the panel from live
11//! renderer state every frame while the panel is clean ("live mirror"), and
12//! applies the panel's state back when the user edits something
13//! ("write-through"), routing expensive operations by the per-group change
14//! flags. Fields a script drives are visibly reclaimed next frame unless the
15//! player-only "freeze script post overrides" switch is on — the panel shows
16//! the truth about who owns a value rather than pretending to.
17

28/// Per-group change flags the host consumes on apply. Groups map to distinct
29/// renderer calls (post config upload vs shadow-pass rebuild vs lighting
30/// setters vs debug-mode pipeline swap vs camera FOV).
31#[derive(Debug, Clone, Copy, Default)]
32pub struct RenderPanelFlags {
33    pub pp_changed: bool,
34    pub mode_changed: bool,
35    pub shadows_changed: bool,
36    pub shadow_res_changed: bool,
37    pub lighting_changed: bool,
38    pub lighting_reset: bool,
39    pub fov_changed: bool,
40    /// A stylize pass (Kuwahara or a registered style) was just enabled —
41    /// host must call `ensure_style_resources`.
42    pub style_needs_resources: bool,
43    /// FXAA was just enabled — host must call ensure_fxaa_resources.
44    pub fxaa_needs_resources: bool,
45}
46
47impl RenderPanelFlags {
48    fn any(&self) -> bool {
49        self.pp_changed
50            || self.mode_changed
51            || self.shadows_changed
52            || self.shadow_res_changed
53            || self.lighting_changed
54            || self.lighting_reset
55            || self.fov_changed
56    }
57}

112    /// Mirror live renderer state into the widgets. No-op while the panel
113    /// has unapplied edits so an in-progress drag is never yanked; never
114    /// sets dirty (the mirror is not an edit).
115    #[allow(clippy::too_many_arguments)]
116    pub fn refresh(
117        &mut self,
118        pp: &PostProcessConfig,
119        debug_mode: DebugMode,
120        shadows_enabled: bool,
121        shadow_resolution: u32,
122        lighting: LightingLevers,
123        fov_deg: f32,
124    ) {
125        if self.dirty {
126            return;
127        }
128        self.pp = pp.clone();
129        self.debug_mode = debug_mode;
130        self.shadows_enabled = shadows_enabled;
131        self.shadow_resolution = shadow_resolution;
132        self.lighting = lighting;
133        self.fov_deg = fov_deg;
134    }

142    /// Mirror the renderer's style ids into the Style Mask legend. Cheap
143    /// every frame: it rebuilds only when the ids changed.
144    pub fn set_style_ids(&mut self, ids: &StyleIds) {
145        let same = self.style_legend.len() == ids.len()
146            && self
147                .style_legend
148                .iter()
149                .zip(ids.iter())
150                .all(|(e, (id, name))| e.id == id && e.name == name);
151        if !same {
152            self.style_legend = legend(ids);
153        }
154    }

156    /// The Style Mask legend's rows, in id order.
157    pub fn style_legend(&self) -> &[LegendEntry] {
158        &self.style_legend
159    }

517/// The Style Mask view's legend: a swatch of each id's hue, its id and
518/// the style variant it stands for.
519fn style_legend_ui(ui: &mut egui::Ui, rows: &[LegendEntry]) {
520    if rows.is_empty() {
521        ui.weak("No styled entities: the mask is empty");
522        return;
523    }
524    for row in rows {
525        ui.horizontal(|ui| {
526            let (rect, _) = ui.allocate_exact_size(egui::vec2(10.0, 10.0), egui::Sense::hover());
527            // The mask view paints these display values
528            let byte = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
529            let [r, g, b] = row.color;
530            ui.painter().rect_filled(
531                rect,
532                2.0,
533                egui::Color32::from_rgb(byte(r), byte(g), byte(b)),
534            );
535            ui.label(format!("{}  {}", row.id, row.name))
536                .on_hover_text(row.color_name);
537        });
538    }
539    ui.weak("Grey: unstyled. White: where ids meet.");
540}

542fn fov_label(fov_deg: f32) -> &'static str {
543    match fov_deg {
544        f if f < 50.0 => "narrow",
545        f if f < 72.0 => "natural",
546        f if f < 90.0 => "wide",
547        _ => "fisheye",
548    }
549}

601    ui: &mut egui::Ui,
602    label: &str,
603    rgb: &mut [f32; 3],
604    range: std::ops::RangeInclusive<f32>,
605    speed: f32,
606) -> bool {
607    ui.horizontal(|ui| {
608        ui.label(label);
609        let mut changed = false;
610        for v in rgb.iter_mut() {
611            changed |= ui
612                .add(egui::DragValue::new(v).speed(speed).range(range.clone()))
613                .changed();
614        }
615        changed
616    })
617    .inner
618}
