1//! Camera tuning panel (F3 family).
2//!
3//! Keys on a `camera_tuning` component (engine schema): field `fov_deg` is
4//! the perspective vertical field of view in degrees. The player applies it
5//! to the live render camera at scene load and whenever the panel edits it,
6//! so games can tune FOV by feel — wide enough to see the foreground, narrow
7//! enough to avoid fisheye — then Commit to File to persist the choice.
8//! The panel is created only when the component exists.
9
10use crate::DebugPanel;
11use flint_core::toml_util::toml_f32;
12use flint_scene::SceneDocument;
13use std::path::PathBuf;
14
15#[derive(Debug, Clone, PartialEq)]
16pub struct CameraPanelConfig {
17    pub fov_deg: f32,
18}
19
20impl CameraPanelConfig {
21    pub fn from_component(value: &toml::Value) -> Self {
22        Self {
23            fov_deg: value.get("fov_deg").and_then(toml_f32).unwrap_or(65.0),
24        }
25    }
26
27    pub fn to_fields(&self) -> Vec<(&'static str, toml::Value)> {
28        vec![("fov_deg", toml::Value::Float(self.fov_deg as f64))]
29    }
30}
31
32pub struct CameraDebugPanel {
33    config: CameraPanelConfig,
34    original: CameraPanelConfig,
35    scene_path: PathBuf,
36    entity_name: String,
37    open: bool,
38    dirty: bool,
39}
40
41impl CameraDebugPanel {
42    pub fn new(config: CameraPanelConfig, scene_path: PathBuf, entity_name: String) -> Self {
43        Self {
44            original: config.clone(),
45            config,
46            scene_path,
47            entity_name,
48            open: false,
49            dirty: false,
50        }
51    }
52
53    pub fn config(&self) -> &CameraPanelConfig {
54        &self.config
55    }
56
57    pub fn entity_name(&self) -> &str {
58        &self.entity_name
59    }
60
61    /// Keep the slider tracking the live camera (e.g. a script override)
62    /// while not being edited.
63    pub fn sync_fov(&mut self, fov_deg: f32) {
64        if !self.dirty {
65            self.config.fov_deg = fov_deg;
66        }
67    }
68
69    fn commit_to_file(&self) -> Result<(), String> {
70        let mut doc = SceneDocument::from_file(&self.scene_path)?;
71        for (field, value) in self.config.to_fields() {
72            doc.patch_field(&self.entity_name, "camera_tuning", field, &value)?;
73        }
74        doc.save(&self.scene_path)
75    }
76}
77
78fn label_for(fov_deg: f32) -> &'static str {
79    match fov_deg {
80        f if f < 50.0 => "narrow",
81        f if f < 72.0 => "natural",
82        f if f < 90.0 => "wide",
83        _ => "fisheye",
84    }
85}
86
87impl DebugPanel for CameraDebugPanel {
88    fn name(&self) -> &str {
89        "Camera"
90    }
91
92    fn ui(&mut self, ui: &mut egui::Ui) {
93        let before = self.config.fov_deg;
94        ui.horizontal(|ui| {
95            ui.label("Vertical FOV");
96            ui.add(
97                egui::Slider::new(&mut self.config.fov_deg, 40.0..=110.0)
98                    .fixed_decimals(1)
99                    .suffix("°"),
100            );
101            ui.label(label_for(self.config.fov_deg));
102        });
103        if (self.config.fov_deg - before).abs() > f32::EPSILON {
104            self.dirty = true;
105        }
106
107        ui.separator();
108        ui.horizontal(|ui| {
109            if ui.button("Reset").clicked() {
110                self.config = self.original.clone();
111                self.dirty = true;
112            }
113            if ui.button("Commit to File").clicked() {
114                if let Err(e) = self.commit_to_file() {
115                    tracing::error!("CameraDebugPanel: commit_to_file failed: {}", e);
116                }
117            }
118        });
119    }
120
121    fn is_open(&self) -> bool {
122        self.open
123    }
124    fn toggle(&mut self) {
125        self.open = !self.open;
126    }
127    fn is_dirty(&self) -> bool {
128        self.dirty
129    }
130    fn clear_dirty(&mut self) {
131        self.dirty = false;
132    }
133    fn as_any(&self) -> &dyn std::any::Any {
134        self
135    }
136    fn as_any_mut(&mut self) -> &mut dyn std::any::Any {
137        self
138    }
