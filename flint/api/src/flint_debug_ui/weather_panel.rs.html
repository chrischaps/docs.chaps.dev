1//! Weather panel (F3 family).
2//!
3//! Controls a game-side `weather` component (convention: a weather script
4//! walks a continuous `state` 0..4 = clear/scattered/overcast/rain/storm,
5//! publishes `wind`/`sea`, and honors `forced_state` (-1 = autonomous),
6//! plus one-shot `snap` / `strike_now` flags). The panel is created only
7//! when the component exists, so games without weather never see it.
8//!
9//! The script owns `state`/`wind`/`sea`; the panel shows them read-only
10//! via [`WeatherDebugPanel::sync_status`] and never writes them. One-shot
11//! buttons queue a single `true` push (see `take_one_shots`) that the
12//! script clears after acting on it.
13

19pub const WEATHER_STATE_NAMES: [&str; 5] = ["Clear", "Scattered", "Overcast", "Rain", "Storm"];

34impl WeatherPanelConfig {
35    pub fn from_component(value: &toml::Value) -> Self {
36        let f = |name: &str, dv: f32| value.get(name).and_then(toml_f32).unwrap_or(dv);
37        Self {
38            forced_state: f("forced_state", -1.0),
39            transition_s: f("transition_s", 60.0),
40            wind_tau_s: f("wind_tau_s", 20.0),
41            sea_build_tau_s: f("sea_build_tau_s", 120.0),
42            sea_decay_tau_s: f("sea_decay_tau_s", 240.0),
43            amp_storm: f("amp_storm", 5.5),
44            rain_rate_max: f("rain_rate_max", 4000.0),
45            gust_strength: f("gust_strength", 0.15),
46            raft_assist: f("raft_assist", 1.0),
47        }
48    }
49
50    pub fn to_fields(&self) -> Vec<(&'static str, toml::Value)> {
51        vec![
52            ("forced_state", toml::Value::Float(self.forced_state as f64)),
53            ("transition_s", toml::Value::Float(self.transition_s as f64)),
54            ("wind_tau_s", toml::Value::Float(self.wind_tau_s as f64)),
55            (
56                "sea_build_tau_s",
57                toml::Value::Float(self.sea_build_tau_s as f64),
58            ),
59            (
60                "sea_decay_tau_s",
61                toml::Value::Float(self.sea_decay_tau_s as f64),
62            ),
63            ("amp_storm", toml::Value::Float(self.amp_storm as f64)),
64            (
65                "rain_rate_max",
66                toml::Value::Float(self.rain_rate_max as f64),
67            ),
68            (
69                "gust_strength",
70                toml::Value::Float(self.gust_strength as f64),
71            ),
72            ("raft_assist", toml::Value::Float(self.raft_assist as f64)),
73        ]
74    }
75}

77pub struct WeatherDebugPanel {
78    config: WeatherPanelConfig,
79    original: WeatherPanelConfig,
80    scene_path: PathBuf,
81    weather_entity_name: String,
82    open: bool,
83    dirty: bool,
84    // Read-only status published by the weather script.
85    state: f32,
86    wind: f32,
87    sea: f32,
88    // One-shot flags queued by buttons, drained by the host each frame.
89    pending_snap: bool,
90    pending_strike: bool,
91}
92
93impl WeatherDebugPanel {
94    pub fn new(
95        config: WeatherPanelConfig,
96        scene_path: PathBuf,
97        weather_entity_name: String,
98    ) -> Self {
99        Self {
100            original: config.clone(),
101            config,
102            scene_path,
103            weather_entity_name,
104            open: false,
105            dirty: false,
106            state: 0.0,
107            wind: 0.0,
108            sea: 0.0,
109            pending_snap: false,
110            pending_strike: false,
111        }
112    }
113
114    pub fn config(&self) -> &WeatherPanelConfig {
115        &self.config
116    }
117
118    pub fn entity_name(&self) -> &str {
119        &self.weather_entity_name
120    }
121
122    /// Update the read-only displays from the script-published fields.
123    pub fn sync_status(&mut self, state: f32, wind: f32, sea: f32) {
124        self.state = state;
125        self.wind = wind;
126        self.sea = sea;
127    }
128
129    /// Drain queued one-shot flags as component field writes.
130    pub fn take_one_shots(&mut self) -> Vec<(&'static str, toml::Value)> {
131        let mut fields = Vec::new();
132        if self.pending_snap {
133            self.pending_snap = false;
134            fields.push(("snap", toml::Value::Boolean(true)));
135        }
136        if self.pending_strike {
137            self.pending_strike = false;
138            fields.push(("strike_now", toml::Value::Boolean(true)));
139        }
140        fields
141    }
142
143    fn commit_to_file(&self) -> Result<(), String> {
144        let mut doc = SceneDocument::from_file(&self.scene_path)?;
145        for (field, value) in self.config.to_fields() {
146            doc.patch_field(&self.weather_entity_name, "weather", field, &value)?;
147        }
148        doc.save(&self.scene_path)
149    }
150}

152fn state_label(state: f32) -> String {
153    let clamped = state.clamp(0.0, 4.0);
154    let i = (clamped.round() as usize).min(4);
155    let name = WEATHER_STATE_NAMES[i];
156    if (clamped - clamped.round()).abs() > 0.1 {
157        format!("{name}ish ({clamped:.2})")
158    } else {
159        name.to_string()
160    }
161}

163impl DebugPanel for WeatherDebugPanel {
164    fn name(&self) -> &str {
165        "Weather"
166    }
167
168    fn ui(&mut self, ui: &mut egui::Ui) {
169        let mut changed = false;
170
171        // Status line: what the sky/wind/sea are actually doing.
172        ui.label(format!(
173            "{}  ·  wind {:.1} m/s  ·  sea {:.0}%",
174            state_label(self.state),
175            self.wind,
176            self.sea * 100.0
177        ));
178        ui.separator();
179
180        // Auto vs forced state.
181        let mut auto = self.config.forced_state < 0.0;
182        let before_auto = auto;
183        ui.checkbox(&mut auto, "Autonomous drift");
184        if auto != before_auto {
185            self.config.forced_state = if auto {
186                -1.0
187            } else {
188                self.state.round().clamp(0.0, 4.0)
189            };
190            changed = true;
191        }
192        if !auto {
193            ui.horizontal(|ui| {
194                ui.label("Force");
195                let mut idx = (self.config.forced_state.round() as usize).min(4);
196                let before = idx;
197                egui::ComboBox::from_id_salt("weather_force_state")
198                    .selected_text(WEATHER_STATE_NAMES[idx])
199                    .show_ui(ui, |ui| {
200                        for (i, name) in WEATHER_STATE_NAMES.iter().enumerate() {
201                            ui.selectable_value(&mut idx, i, *name);
202                        }
203                    });
204                if idx != before {
205                    self.config.forced_state = idx as f32;
206                    changed = true;
207                }
208                if ui.button("Snap").clicked() {
209                    self.pending_snap = true;
210                }
211            });
212        }
213
214        if ui.button("Strike now").clicked() {
215            self.pending_strike = true;
216        }
217
218        ui.separator();
219        let drag = |ui: &mut egui::Ui,
220                    label: &str,
221                    value: &mut f32,
222                    speed: f64,
223                    range: std::ops::RangeInclusive<f64>|
224         -> bool {
225            let before = *value;
226            ui.horizontal(|ui| {
227                ui.label(label);
228                ui.add(egui::DragValue::new(value).speed(speed).range(range));
229            });
230            (*value - before).abs() > f32::EPSILON
231        };
232        changed |= drag(
233            ui,
234            "Transition (s)",
235            &mut self.config.transition_s,
236            1.0,
237            1.0..=600.0,
238        );
239        changed |= drag(
240            ui,
241            "Wind tau (s)",
242            &mut self.config.wind_tau_s,
243            0.5,
244            0.5..=300.0,
245        );
246        changed |= drag(
247            ui,
248            "Sea build tau (s)",
249            &mut self.config.sea_build_tau_s,
250            1.0,
251            1.0..=1800.0,
252        );
253        changed |= drag(
254            ui,
255            "Sea decay tau (s)",
256            &mut self.config.sea_decay_tau_s,
257            1.0,
258            1.0..=1800.0,
259        );
260        changed |= drag(
261            ui,
262            "Storm amplitude",
263            &mut self.config.amp_storm,
264            0.05,
265            0.5..=10.0,
266        );
267        changed |= drag(
268            ui,
269            "Rain rate max",
270            &mut self.config.rain_rate_max,
271            25.0,
272            0.0..=10000.0,
273        );
274        changed |= drag(
275            ui,
276            "Gust strength",
277            &mut self.config.gust_strength,
278            0.005,
279            0.0..=1.0,
280        );
281        changed |= drag(
282            ui,
283            "Raft assist",
284            &mut self.config.raft_assist,
285            0.02,
286            0.0..=3.0,
287        );
288
289        if changed {
290            self.dirty = true;
291        }
292
293        ui.separator();
294        ui.horizontal(|ui| {
295            if ui.button("Reset").clicked() {
296                self.config = self.original.clone();
297                self.dirty = true;
298            }
299            if ui.button("Commit to File").clicked() {
300                if let Err(e) = self.commit_to_file() {
301                    tracing::error!("WeatherDebugPanel: commit_to_file failed: {}", e);
302                }
303            }
304        });
305    }
