1//! Live-tuning panel for the `ocean` component (F3 family).
2//!
3//! Edits are pushed by the player into the world's `ocean` component, so the
4//! renderer (spectrum regen on param change) and the script API
5//! (`ocean_height`) both pick them up on the next frame — one source of
6//! truth, three consumers.

15fn drag_i64(
16    ui: &mut egui::Ui,
17    label: &str,
18    value: &mut i64,
19    range: std::ops::RangeInclusive<i64>,
20) -> bool {
21    let before = *value;
22    ui.horizontal(|ui| {
23        ui.label(label);
24        ui.add(egui::DragValue::new(value).speed(0.1).range(range));
25    });
26    *value != before
27}
28
29fn color_rgba(ui: &mut egui::Ui, label: &str, value: &mut [f32; 4]) -> bool {
30    let mut rgb = [value[0], value[1], value[2]];
31    let mut changed = false;
32    ui.horizontal(|ui| {
33        ui.label(label);
34        if ui.color_edit_button_rgb(&mut rgb).changed() {
35            changed = true;
36        }
37    });
38    if changed {
39        value[0] = rgb[0];
40        value[1] = rgb[1];
41        value[2] = rgb[2];
42    }
43    changed
44}

217    open: bool,

219}

221impl OceanDebugPanel {
222    pub fn new(config: OceanPanelConfig, scene_path: PathBuf, ocean_entity_name: String) -> Self {
223        Self {
224            original: config.clone(),
225            config,
226            scene_path,
227            ocean_entity_name,
228            open: false,
229            dirty: false,
230        }
231    }
232
233    pub fn config(&self) -> &OceanPanelConfig {
234        &self.config
235    }
236
237    pub fn entity_name(&self) -> &str {
238        &self.ocean_entity_name
239    }
240
241    fn commit_to_file(&self) -> Result<(), String> {
242        let mut doc = SceneDocument::from_file(&self.scene_path)?;
243        for (field, value) in self.config.to_fields() {
244            doc.patch_field(&self.ocean_entity_name, "ocean", field, &value)?;
245        }
246        doc.save(&self.scene_path)
247    }
248
249    fn format_toml(&self) -> String {
250        self.config
251            .to_fields()
252            .iter()
253            .map(|(name, value)| format!("{name} = {value}\n"))
254            .collect()
255    }
256}
