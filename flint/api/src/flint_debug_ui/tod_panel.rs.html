1//! Day / time scrubber panel (F3 family).
2//!
3//! Controls a game-side `time_of_day` component (convention: fields
4//! `time_hours` 0-24, `day_length_sec`, `auto_advance`, `sun_path_tilt_deg`,
5//! and optionally a script-owned `day` counter; a game script interpolates
6//! keyframes from it). The panel is created only when the component exists,
7//! so games without a day/night cycle never see it.
8//!
9//! While `auto_advance` is on, the game script owns `time_hours`; the player
10//! keeps the slider display in sync via [`TimeOfDayDebugPanel::sync_time`].
11//! Drag the slider (or uncheck auto) to force a time — the debug slider and
12//! natural passage share the game script's single interpolation path, so
13//! scrubbing is seamless.
14//!
15//! The `day` counter is different: the game script owns it outright
16//! (midnight wraps increment it; >12h `time_hours` jumps adjust it), so the
17//! panel never writes it as part of its persistent config. Day edits are a
18//! one-shot override the host drains via [`TimeOfDayDebugPanel::take_day_set`]
19//! and the display tracks the live value via
20//! [`TimeOfDayDebugPanel::sync_day`]. Scenes without a materialized `day`
21//! field (flat-rate scenes) simply hide the day UI.
22
23use crate::DebugPanel;
24use flint_core::toml_util::toml_f32;
25use flint_scene::SceneDocument;
26use std::path::PathBuf;
27
28#[derive(Debug, Clone, PartialEq)]
29pub struct TimeOfDayPanelConfig {
30    pub time_hours: f32,
31    pub day_length_sec: f32,
32    pub auto_advance: bool,
33    pub sun_path_tilt_deg: f32,
34    /// Optional per-day multipliers on `day_length_sec`, indexed by day
35    /// number; the last entry repeats for every later day. Games that
36    /// compress their opening days declare the same curve their time script
37    /// applies, so the panel can report the real length. Empty = flat rate.
38    pub day_length_ramp: Vec<f32>,
39}
40
41impl TimeOfDayPanelConfig {
42    pub fn from_component(value: &toml::Value) -> Self {
43        let f = |name: &str, dv: f32| value.get(name).and_then(toml_f32).unwrap_or(dv);
44        Self {
45            time_hours: f("time_hours", 10.5),
46            day_length_sec: f("day_length_sec", 600.0),
47            auto_advance: value
48                .get("auto_advance")
49                .and_then(|v| v.as_bool())
50                .unwrap_or(true),
51            sun_path_tilt_deg: f("sun_path_tilt_deg", 28.0),
52            day_length_ramp: value
53                .get("day_length_ramp")
54                .and_then(|v| v.as_array())
55                .map(|a| a.iter().filter_map(toml_f32).collect())
56                .unwrap_or_default(),
57        }
58    }
59
60    pub fn to_fields(&self) -> Vec<(&'static str, toml::Value)> {
61        vec![
62            ("time_hours", toml::Value::Float(self.time_hours as f64)),
63            (
64                "day_length_sec",
65                toml::Value::Float(self.day_length_sec as f64),
66            ),
67            ("auto_advance", toml::Value::Boolean(self.auto_advance)),
68            (
69                "sun_path_tilt_deg",
70                toml::Value::Float(self.sun_path_tilt_deg as f64),
71            ),
72        ]
73    }
74}

76pub struct TimeOfDayDebugPanel {
77    config: TimeOfDayPanelConfig,
78    original: TimeOfDayPanelConfig,
79    scene_path: PathBuf,
80    tod_entity_name: String,
81    open: bool,
82    dirty: bool,
83    /// Live script-owned day counter, `None` in scenes without a `day` field.
84    day: Option<f32>,
85    /// Pending one-shot day override; drained by the host each frame.
86    pending_day: Option<f32>,
87}
88
89impl TimeOfDayDebugPanel {
90    pub fn new(config: TimeOfDayPanelConfig, scene_path: PathBuf, tod_entity_name: String) -> Self {
91        Self {
92            original: config.clone(),
93            config,
94            scene_path,
95            tod_entity_name,
96            open: false,
97            dirty: false,
98            day: None,
99            pending_day: None,
100        }
101    }
102
103    pub fn config(&self) -> &TimeOfDayPanelConfig {
104        &self.config
105    }
106
107    pub fn entity_name(&self) -> &str {
108        &self.tod_entity_name
109    }
110
111    /// Keep the slider tracking the game-advanced time while not being edited.
112    pub fn sync_time(&mut self, hours: f32) {
113        if !self.dirty {
114            self.config.time_hours = hours;
115        }
116    }
117
118    /// Track the script-owned day counter (`None` = field not in this scene).
119    pub fn sync_day(&mut self, day: Option<f32>) {
120        if self.pending_day.is_none() {
121            self.day = day;
122        }
123    }
124
125    /// Drain the one-shot day override, if the user edited the day this frame.
126    pub fn take_day_set(&mut self) -> Option<f32> {
127        self.pending_day.take()
128    }
129
130    fn commit_to_file(&self) -> Result<(), String> {
131        let mut doc = SceneDocument::from_file(&self.scene_path)?;
132        for (field, value) in self.config.to_fields() {
133            doc.patch_field(&self.tod_entity_name, "time_of_day", field, &value)?;
134        }
135        doc.save(&self.scene_path)
136    }
137}

139fn format_clock(hours: f32) -> String {
140    let h = (hours.floor() as i32).rem_euclid(24);
141    let m = (hours.fract() * 60.0).floor() as i32;
142    format!("{h:02}:{m:02}")
143}
144
145/// Day-length multiplier for `day` under a game-declared ramp.
146///
147/// `ramp` is indexed by day number and its last entry repeats forever, so a
148/// game that compresses its opening days declares e.g. `[0.1, 0.2, 0.4, 1.0]`
149/// on its `time_of_day` component. An empty ramp means a flat rate.
150fn ramp_fraction(day: f32, ramp: &[f32]) -> f32 {
151    if ramp.is_empty() {
152        return 1.0;
153    }
154    let idx = (day.round().max(0.0) as usize).min(ramp.len() - 1);
155    ramp[idx]
156}

158fn label_for(hours: f32) -> &'static str {
159    match hours {
160        h if h < 4.5 => "night",
161        h if h < 6.5 => "dawn",
162        h if h < 10.0 => "morning",
163        h if h < 15.0 => "midday",
164        h if h < 18.5 => "golden hour",
165        h if h < 19.8 => "sunset",
166        h if h < 21.0 => "dusk",
167        _ => "night",
168    }
169}

171impl DebugPanel for TimeOfDayDebugPanel {
172    fn name(&self) -> &str {
173        "Day / Time"
174    }
175
176    fn ui(&mut self, ui: &mut egui::Ui) {
177        let mut changed = false;
178
179        // ── Clock line: "Day 3 · 14:32 · midday" ────────────────────────
180        let clock = format_clock(self.config.time_hours);
181        let phase = label_for(self.config.time_hours);
182        let status = match self.day {
183            Some(d) => format!("Day {}  ·  {clock}  ·  {phase}", d.round() as i64),
184            None => format!("{clock}  ·  {phase}"),
185        };
186        ui.label(egui::RichText::new(status).strong());
187        ui.separator();
188
189        let before = self.config.time_hours;
190        ui.horizontal(|ui| {
191            ui.label("Time");
192            ui.add(
193                egui::Slider::new(&mut self.config.time_hours, 0.0..=24.0)
194                    .fixed_decimals(2)
195                    .suffix(" h"),
196            );
197        });
198        // Presets jump straight to a keyframe hour; a jump across midnight
199        // reads as a wrap/scrub to the game script (day +1 / -1 by design).
200        ui.horizontal(|ui| {
201            for (name, hours) in [
202                ("Dawn", 6.0),
203                ("Noon", 12.5),
204                ("Sunset", 19.0),
205                ("Dusk", 20.4),
206                ("Night", 0.0),
207            ] {
208                if ui.small_button(name).clicked() {
209                    self.config.time_hours = hours;
210                }
211            }
212        });
213        if (self.config.time_hours - before).abs() > f32::EPSILON {
214            changed = true;
215        }
216
217        let before_auto = self.config.auto_advance;
218        ui.checkbox(&mut self.config.auto_advance, "Advance naturally");
219        changed |= self.config.auto_advance != before_auto;
220
221        // ── Day counter (script-owned; edits are one-shot overrides) ────
222        if let Some(day) = self.day {
223            ui.horizontal(|ui| {
224                ui.label("Day");
225                let mut d = self.pending_day.unwrap_or(day).round() as i32;
226                let resp = ui.add(egui::DragValue::new(&mut d).speed(0.05).range(0..=9999));
227                if resp.changed() {
228                    self.pending_day = Some(d.max(0) as f32);
229                }
230                if ui.small_button("-1").clicked() && d > 0 {
231                    self.pending_day = Some((d - 1) as f32);
232                }
233                if ui.small_button("+1").clicked() {
234                    self.pending_day = Some((d + 1) as f32);
235                }
236            });
237        }
238
239        ui.horizontal(|ui| {
240            ui.label("Day length (s)");
241            let before = self.config.day_length_sec;
242            ui.add(
243                egui::DragValue::new(&mut self.config.day_length_sec)
244                    .speed(5.0)
245                    .range(30.0..=86400.0),
246            );
247            changed |= (self.config.day_length_sec - before).abs() > f32::EPSILON;
248        });
249        // Effective length under a game-declared ramp (ramped scenes only).
250        if let Some(day) = self.day {
251            if !self.config.day_length_ramp.is_empty() {
252                let frac = ramp_fraction(
253                    self.pending_day.unwrap_or(day),
254                    &self.config.day_length_ramp,
255                );
256                let eff = self.config.day_length_sec * frac;
257                ui.weak(format!(
258                    "effective {eff:.0} s this day (ramp x{frac:.1}) · {:.1} s / game hour",
259                    eff / 24.0
260                ));
261            }
262        }
263
264        ui.horizontal(|ui| {
265            ui.label("Sun path tilt");
266            let before = self.config.sun_path_tilt_deg;
267            ui.add(
268                egui::DragValue::new(&mut self.config.sun_path_tilt_deg)
269                    .speed(0.5)
270                    .range(0.0..=75.0)
271                    .suffix("°"),
272            );
273            changed |= (self.config.sun_path_tilt_deg - before).abs() > f32::EPSILON;
274        });
275
276        if changed {
277            self.dirty = true;
278        }
279
280        ui.separator();
281        ui.horizontal(|ui| {
282            if ui.button("Reset").clicked() {
283                self.config = self.original.clone();
284                self.dirty = true;
285            }
286            if ui.button("Commit to File").clicked() {
287                if let Err(e) = self.commit_to_file() {
288                    tracing::error!("TimeOfDayDebugPanel: commit_to_file failed: {}", e);
289                }
290            }
291        });
292    }

314#[cfg(test)]
315mod ramp_tests {
316    use super::{ramp_fraction, TimeOfDayPanelConfig};
317
318    #[test]
319    fn empty_ramp_is_flat() {
320        assert_eq!(ramp_fraction(0.0, &[]), 1.0);
321        assert_eq!(ramp_fraction(99.0, &[]), 1.0);
322    }
323
324    #[test]
325    fn ramp_indexes_by_day_and_holds_the_last_entry() {
326        let ramp = [0.1, 0.2, 0.4, 0.7, 1.0];
327        assert_eq!(ramp_fraction(0.0, &ramp), 0.1);
328        assert_eq!(ramp_fraction(1.0, &ramp), 0.2);
329        assert_eq!(ramp_fraction(3.0, &ramp), 0.7);
330        assert_eq!(ramp_fraction(4.0, &ramp), 1.0);
331        // Past the end the last entry repeats forever.
332        assert_eq!(ramp_fraction(400.0, &ramp), 1.0);
333    }
334
335    #[test]
336    fn negative_day_clamps_to_the_first_entry() {
337        assert_eq!(ramp_fraction(-3.0, &[0.1, 1.0]), 0.1);
338    }
339
340    #[test]
341    fn absent_ramp_field_parses_as_empty() {
342        let v: toml::Value = toml::from_str("time_hours = 6.0").unwrap();
343        assert!(TimeOfDayPanelConfig::from_component(&v)
344            .day_length_ramp
345            .is_empty());
346    }
347
348    #[test]
349    fn ramp_field_parses_from_the_component() {
350        let v: toml::Value = toml::from_str("day_length_ramp = [0.1, 0.5, 1.0]").unwrap();
351        assert_eq!(
352            TimeOfDayPanelConfig::from_component(&v).day_length_ramp,
353            vec![0.1, 0.5, 1.0]
354        );
355    }
