3use serde::{Deserialize, Serialize};

6/// Root structure of a scene TOML file
7#[derive(Debug, Clone, Serialize, Deserialize)]
8pub struct SceneFile {
9    pub scene: SceneMetadata,
10    #[serde(default)]
11    pub environment: Option<EnvironmentDef>,
12    #[serde(default)]
13    pub post_process: Option<PostProcessDef>,
14    #[serde(default)]
15    pub entities: HashMap<String, EntityDef>,
16}
17
18/// Post-processing settings for the scene
19#[derive(Debug, Clone, Serialize, Deserialize)]
20pub struct PostProcessDef {
21    #[serde(default = "default_true")]
22    pub bloom_enabled: bool,
23    #[serde(default = "default_bloom_intensity")]
24    pub bloom_intensity: f32,
25    #[serde(default = "default_bloom_threshold")]
26    pub bloom_threshold: f32,
27    #[serde(default)]
28    pub vignette_enabled: bool,
29    #[serde(default = "default_vignette_intensity")]
30    pub vignette_intensity: f32,
31    #[serde(default = "default_vignette_smoothness")]
32    pub vignette_smoothness: f32,
33    #[serde(default = "default_exposure")]
34    pub exposure: f32,
35    #[serde(default = "default_true")]
36    pub ssao_enabled: bool,
37    #[serde(default = "default_ssao_radius")]
38    pub ssao_radius: f32,
39    #[serde(default = "default_ssao_intensity")]

83/// Scene metadata
84#[derive(Debug, Clone, Serialize, Deserialize)]
85pub struct SceneMetadata {
86    pub name: String,
87    #[serde(default = "default_version")]
88    pub version: String,
89    #[serde(default)]
90    pub description: Option<String>,
91    #[serde(default)]
92    pub input_config: Option<String>,
93}
94
95fn default_version() -> String {
96    "1.0".to_string()
97}

99/// Definition of an entity in a scene file
100#[derive(Debug, Clone, Serialize, Deserialize)]
101pub struct EntityDef {
102    /// Optional archetype name
103    #[serde(skip_serializing_if = "Option::is_none")]
104    pub archetype: Option<String>,
105    /// Optional parent entity name
106    #[serde(skip_serializing_if = "Option::is_none")]
107    pub parent: Option<String>,
108    /// Component data - all other fields are treated as components
109    #[serde(flatten)]
110    pub components: HashMap<String, toml::Value>,
111}
112
113impl EntityDef {
114    pub fn new() -> Self {
115        Self {
116            archetype: None,
117            parent: None,
118            components: HashMap::new(),
119        }
120    }
121
122    pub fn with_archetype(mut self, archetype: impl Into<String>) -> Self {
123        self.archetype = Some(archetype.into());
124        self
125    }
126
127    pub fn with_parent(mut self, parent: impl Into<String>) -> Self {
128        self.parent = Some(parent.into());
129        self
130    }
131
132    pub fn with_component(mut self, name: impl Into<String>, data: toml::Value) -> Self {
133        self.components.insert(name.into(), data);
134        self
135    }
136}
137
138impl Default for EntityDef {
139    fn default() -> Self {
140        Self::new()
141    }
142}
143
144impl SceneFile {
145    /// Create a new scene file
146    pub fn new(name: impl Into<String>) -> Self {
147        Self {
148            scene: SceneMetadata {
149                name: name.into(),
150                version: default_version(),
151                description: None,
152                input_config: None,
153            },
154            environment: None,
155            post_process: None,
156            entities: HashMap::new(),
157        }
158    }
159
160    /// Add an entity to the scene
161    pub fn add_entity(&mut self, name: impl Into<String>, entity: EntityDef) {
162        self.entities.insert(name.into(), entity);
163    }
164}
