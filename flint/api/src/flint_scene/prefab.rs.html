2
3use crate::format::{PrefabFile, SceneFile};
4use flint_core::{FlintError, Result};
5use std::collections::HashMap;
6use std::fs;
7use std::path::Path;

9/// Expand all prefab instances in a scene file, inserting their entities into `scene.entities`.
10/// Must be called before entity creation (two-pass load).
11pub fn expand_prefabs(scene: &mut SceneFile, scene_path: &Path) -> Result<()> {
12    if scene.prefabs.is_empty() {
13        return Ok(());
14    }
15
16    let scene_dir = scene_path.parent().unwrap_or_else(|| Path::new("."));
17
18    // Cache loaded prefab templates by template name
19    let mut cache: HashMap<String, PrefabFile> = HashMap::new();
20
21    // Collect instances (we need to drain prefabs to avoid borrow conflict)
22    let instances: Vec<_> = scene.prefabs.drain().collect();
23
24    for (_instance_name, instance) in &instances {
25        // Load template (cached)
26        let prefab = if let Some(cached) = cache.get(&instance.template) {
27            cached.clone()
28        } else {
29            let prefab_path = resolve_prefab_path(scene_dir, &instance.template)?;
30            let content = fs::read_to_string(&prefab_path).map_err(|e| {
31                FlintError::ParseError(format!(
32                    "Failed to read prefab '{}': {}",
33                    prefab_path.display(),
34                    e
35                ))
36            })?;
37            let prefab: PrefabFile = toml::from_str(&content).map_err(|e| {
38                FlintError::ParseError(format!(
39                    "Failed to parse prefab '{}': {}",
40                    prefab_path.display(),
41                    e
42                ))
43            })?;
44            cache.insert(instance.template.clone(), prefab.clone());
45            prefab
46        };
47
48        let prefix = &instance.prefix;
49
50        // Build variable map for substitution
51        let mut vars = HashMap::new();
52        vars.insert("PREFIX".to_string(), prefix.clone());
53
54        // Expand each template entity
55        for (entity_suffix, template_def) in &prefab.entities {
56            let expanded_name = format!("{}_{}", prefix, entity_suffix);
57
58            // Check for collision
59            if scene.entities.contains_key(&expanded_name) {
60                return Err(FlintError::ParseError(format!(
61                    "Prefab expansion collision: entity '{}' already exists in scene",
62                    expanded_name
63                )));
64            }
65
66            // Clone and transform the entity definition
67            let mut entity = template_def.clone();
68
69            // Prefix parent references within the prefab
70            if let Some(ref parent) = entity.parent {
71                entity.parent = Some(format!("{}_{}", prefix, parent));
72            }
73
74            // Substitute ${PREFIX} in all string component values
75            for (_comp_name, comp_value) in entity.components.iter_mut() {
76                substitute_in_value(comp_value, &vars);
77            }
78
79            // Apply per-instance overrides (deep merge at field level)
80            if let Some(entity_overrides) = instance.overrides.get(entity_suffix) {
81                for (comp_name, override_value) in entity_overrides {
82                    deep_merge_component(&mut entity.components, comp_name, override_value.clone());
83                }
84            }
85
86            scene.entities.insert(expanded_name, entity);
87        }
88    }
89
90    Ok(())
91}

93/// Search for a prefab file: tries `scene_dir/prefabs/{name}.prefab.toml`
94/// then `scene_dir/../prefabs/{name}.prefab.toml` (project root pattern).
95fn resolve_prefab_path(scene_dir: &Path, template_name: &str) -> Result<std::path::PathBuf> {
96    let filename = format!("{}.prefab.toml", template_name);
97
98    // Try scene_dir/prefabs/
99    let path1 = scene_dir.join("prefabs").join(&filename);
100    if path1.exists() {
101        return Ok(path1);
102    }
103
104    // Try scene_dir/../prefabs/ (project root when scenes/ is a subdirectory)
105    if let Some(parent) = scene_dir.parent() {
106        let path2 = parent.join("prefabs").join(&filename);
107        if path2.exists() {
108            return Ok(path2);
109        }
110    }
111
112    Err(FlintError::ParseError(format!(
113        "Prefab template '{}' not found (searched {}/prefabs/ and parent)",
114        template_name,
115        scene_dir.display()
116    )))
117}

119/// Recursively substitute `${VAR}` patterns in string values within a toml::Value tree.
120fn substitute_in_value(value: &mut toml::Value, vars: &HashMap<String, String>) {
121    match value {
122        toml::Value::String(s) => {
123            for (key, replacement) in vars {
124                let pattern = format!("${{{}}}", key);
125                if s.contains(&pattern) {
126                    *s = s.replace(&pattern, replacement);
127                }
128            }
129        }
130        toml::Value::Table(table) => {
131            for (_k, v) in table.iter_mut() {
132                substitute_in_value(v, vars);
133            }
134        }
135        toml::Value::Array(arr) => {
136            for v in arr.iter_mut() {
137                substitute_in_value(v, vars);
138            }
139        }
140        _ => {} // integers, floats, booleans — no substitution needed
141    }
142}

144/// Deep-merge override data into an entity's component map.
145/// If the component already exists as a Table, merge fields. Otherwise replace.
146fn deep_merge_component(
147    components: &mut HashMap<String, toml::Value>,
148    comp_name: &str,
149    override_value: toml::Value,
150) {
151    if let Some(existing) = components.get_mut(comp_name) {
152        if let (toml::Value::Table(existing_table), toml::Value::Table(override_table)) =
153            (existing, &override_value)
154        {
155            // Field-level merge: override wins per field
156            for (field, val) in override_table {
157                existing_table.insert(field.clone(), val.clone());
158            }
159            return;
160        }
161    }
162    // No existing component or type mismatch: just insert
163    components.insert(comp_name.to_string(), override_value);
164}
