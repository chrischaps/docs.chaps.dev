21/// Convert a `PostProcessDef` (scene TOML) to a `PostProcessConfig` (renderer).
22///
23/// Centralises the field-by-field copy so that new fields are never forgotten.
24pub fn post_process_config_from_def(pp_def: &PostProcessDef) -> PostProcessConfig {
25    let mut config = PostProcessConfig::default();
26    config.bloom_enabled = pp_def.bloom_enabled;
27    config.bloom_intensity = pp_def.bloom_intensity;
28    config.bloom_threshold = pp_def.bloom_threshold;
29    config.vignette_enabled = pp_def.vignette_enabled;
30    config.vignette_intensity = pp_def.vignette_intensity;
31    config.vignette_smoothness = pp_def.vignette_smoothness;
32    config.exposure = pp_def.exposure;
33    config.ssao_enabled = pp_def.ssao_enabled;
34    config.ssao_radius = pp_def.ssao_radius;
35    config.ssao_intensity = pp_def.ssao_intensity;
36    config.fog_enabled = pp_def.fog_enabled;
37    config.fog_color = pp_def.fog_color;
38    config.fog_density = pp_def.fog_density;
39    config.fog_start = pp_def.fog_start;
40    config.fog_end = pp_def.fog_end;
41    config.fog_height_enabled = pp_def.fog_height_enabled;
42    config.fog_height_falloff = pp_def.fog_height_falloff;
43    config.fog_height_origin = pp_def.fog_height_origin;
44    config
45}

47/// Load terrain from world entities. Free function to avoid borrow conflicts.
48pub(super) fn load_terrain_from_world_inner(
49    world: &FlintWorld,
50    scene_path: &str,
51    device: &wgpu::Device,
52    queue: &wgpu::Queue,
53    scene_renderer: &mut SceneRenderer,
54    physics: &mut PhysicsSystem,
55    terrain_out: &mut Option<(flint_terrain::Terrain, flint_terrain::TerrainConfig)>,
56) {
57    use flint_core::Transform;
58    use flint_terrain::{Heightmap, Terrain, TerrainConfig};
59
60    let scene_dir = Path::new(scene_path)
61        .parent()
62        .unwrap_or_else(|| Path::new("."));
63
64    for entity in world.all_entities() {
65        let terrain_comp = match world.get_component(entity.id, comp::TERRAIN) {
66            Some(c) => c,
67            None => continue,
68        };
69
70        let heightmap_rel = match terrain_comp.get("heightmap").and_then(|v| v.as_str()) {
71            Some(s) if !s.is_empty() => s.to_string(),
72            _ => {
73                tracing::warn!("Terrain entity '{}': missing heightmap path", entity.name);
74                continue;
75            }
76        };
77
78        // Resolve heightmap path (scene dir, then parent)
79        let hm_path = {
80            let p = scene_dir.join(&heightmap_rel);
81            if p.exists() {
82                p
83            } else if let Some(parent) = scene_dir.parent() {
84                let pp = parent.join(&heightmap_rel);
85                if pp.exists() {
86                    pp
87                } else {
88                    p
89                }
90            } else {
91                p
92            }
93        };
94
95        let heightmap = match Heightmap::from_png(&hm_path) {
96            Ok(hm) => hm,
97            Err(e) => {
98                tracing::warn!("Failed to load heightmap: {}", e);
99                continue;
100            }
101        };
102
103        let get_f32 = |key: &str, default: f32| -> f32 {
104            terrain_comp.get(key).and_then(toml_f32).unwrap_or(default)
105        };
106
107        let get_i32 = |key: &str, default: i32| -> i32 {
108            terrain_comp
109                .get(key)
110                .and_then(|v| v.as_integer())
111                .map(|i| i as i32)
112                .unwrap_or(default)
113        };
114
115        let get_str = |key: &str| -> String {
116            terrain_comp
117                .get(key)
118                .and_then(|v| v.as_str())
119                .unwrap_or("")
120                .to_string()
121        };
122
123        let config = TerrainConfig {
124            heightmap_path: heightmap_rel,
125            width: get_f32("width", 256.0),
126            depth: get_f32("depth", 256.0),
127            height_scale: get_f32("height_scale", 50.0),
128            chunk_resolution: get_i32("chunk_resolution", 64) as u32,
129            texture_tile: get_f32("texture_tile", 16.0),
130            splat_map_path: get_str("splat_map"),
131            layer_textures: [
132                get_str("layer0_texture"),
133                get_str("layer1_texture"),
134                get_str("layer2_texture"),
135                get_str("layer3_texture"),
136            ],
137            metallic: get_f32("metallic", 0.0),
138            roughness: get_f32("roughness", 0.85),
139        };
140
141        let terrain = Terrain::generate(&heightmap, &config);
142
143        // Get entity transform
144        let transform = world
145            .get_component(entity.id, comp::TRANSFORM)
146            .and_then(|t| {
147                let pos = t.get("position").and_then(toml_vec3).unwrap_or([0.0; 3]);
148                Some(Transform {
149                    position: FlintVec3 {
150                        x: pos[0],
151                        y: pos[1],
152                        z: pos[2],
153                    },
154                    ..Default::default()
155                })
156            })
157            .unwrap_or_default();
158
159        // Upload to GPU
160        scene_renderer.load_terrain(
161            device,
162            queue,
163            &terrain.chunks,
164            &transform,
165            config.texture_tile,
166            config.metallic,
167            config.roughness,
168            &config.splat_map_path,
169            &config.layer_textures,
170            scene_dir,
171        );
172
173        // Register physics collider
174        let (verts, tris) = terrain.trimesh_data();
175        let offset = [
176            transform.position.x,
177            transform.position.y,
178            transform.position.z,
179        ];
180        let offset_verts: Vec<[f32; 3]> = verts
181            .iter()
182            .map(|v| [v[0] + offset[0], v[1] + offset[1], v[2] + offset[2]])
183            .collect();
184        physics.register_static_trimesh(
185            entity.id,
186            offset_verts,
187            tris,
188            0.8,
189            0.1,
190        );
191
192        tracing::info!(
193            "Loaded terrain '{}': {}x{} heightmap, {} chunks, {}x{} world units",
194            entity.name,
195            heightmap.width,
196            heightmap.depth,
197            terrain.chunks.len(),
198            config.width,
199            config.depth,
200        );
201
202        *terrain_out = Some((terrain, config));
203        break; // Only one terrain entity for now
204    }
205}

207/// Build a ModelLoadConfig with catalog-resolved path overrides.
208pub(super) fn build_model_load_config(
209    scene_path: &str,
210    world: &FlintWorld,
211    catalog: Option<&AssetCatalog>,
212    store: Option<&ContentStore>,
213) -> ModelLoadConfig {
214    let mut config = ModelLoadConfig::from_scene_path(scene_path);
215
216    let (Some(cat), Some(st)) = (catalog, store) else {
217        return config;
218    };
219
220    for entity in world.all_entities() {
221        if let Some(comps) = world.get_components(entity.id) {
222            // Resolve model assets
223            if let Some(model) = comps.get(comp::MODEL) {
224                if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
225                    if !config.overrides.contains_key(name) {
226                        if let Some(path) = resolve_catalog(cat, st, name) {
227                            config.overrides.insert(name.to_string(), path);
228                        }
229                    }
230                }
231            }
232            // Resolve textures
233            for comp_name in &[comp::MATERIAL, comp::SPRITE] {
234                if let Some(comp) = comps.get(*comp_name) {
235                    if let Some(tex) = comp.get("texture").and_then(|v| v.as_str()) {
236                        if !config.overrides.contains_key(tex) {
237                            if let Some(path) = resolve_catalog(cat, st, tex) {
238                                config.overrides.insert(tex.to_string(), path);
239                            }
240                        }
241                    }
242                }
243            }
244        }
245    }
246
247    config
248}

250/// Look up an asset name in the catalog and content store.
251pub(super) fn resolve_catalog(
252    catalog: &AssetCatalog,
253    store: &ContentStore,
254    name: &str,
255) -> Option<PathBuf> {
256    let meta = catalog.get(name)?;
257    let hash = flint_core::ContentHash::from_prefixed_hex(&meta.hash)?;
258    store.get(&hash)
259}

261/// Register skeletal data from loaded models into the animation system.
262pub(super) fn register_skeletal_data(
263    load_result: &model_loader::ModelLoadResult,
264    animation: &mut AnimationSystem,
265) {
266    for loaded in &load_result.models {
267        if loaded.is_skinned {
268            if let Some(ref import_result) = loaded.import_result {
269                for imported_skel in &import_result.skeletons {
270                    let skeleton = Skeleton::from_imported(imported_skel);
271                    animation.add_skeleton(loaded.entity_id, skeleton);
272                }
273                for imported_clip in &import_result.skeletal_clips {
274                    let clip = SkeletalClip::from_imported(imported_clip);
275                    println!(
276                        "  Skeletal clip: {} ({:.1}s, {} tracks)",
277                        clip.name,
278                        clip.duration,
279                        clip.joint_tracks.len()
280                    );
281                    animation.add_skeletal_clip(clip);
282                }
283            }
284        }
285    }
286}

312/// Load audio files referenced by audio_source components and preload all .ogg files from audio/
313pub(super) fn load_audio_from_world(world: &FlintWorld, audio: &mut AudioSystem, scene_path: &str) {
314    let scene_dir = Path::new(scene_path)
315        .parent()
316        .unwrap_or_else(|| Path::new("."));
317
318    // Build list of directories to search: scene dir first, then parent (game root)
319    let mut search_dirs = vec![scene_dir.to_path_buf()];
320    if let Some(parent) = scene_dir.parent() {
321        search_dirs.push(parent.to_path_buf());
322    }
323
324    // Load audio files referenced by audio_source components
325    for entity in world.all_entities() {
326        let audio_file = world
327            .get_components(entity.id)
328            .and_then(|components| components.get(comp::AUDIO_SOURCE).cloned())
329            .and_then(|audio_src| {
330                audio_src
331                    .get("file")
332                    .and_then(|v| v.as_str().map(String::from))
333            });
334
335        if let Some(file_name) = audio_file {
336            if audio.engine.has_sound(&file_name) {
337                continue;
338            }
339
340            let mut loaded = false;
341            for dir in &search_dirs {
342                let audio_path = dir.join(&file_name);
343                if audio_path.exists() {
344                    match audio.engine.load_sound(&file_name, &audio_path) {
345                        Ok(_) => {
346                            println!("Loaded audio: {}", file_name);
347                            loaded = true;
348                        }
349                        Err(e) => {
350                            tracing::warn!("Failed to load audio '{}': {:?}", file_name, e);
351                        }
352                    }
353                    break;
354                }
355            }
356            if !loaded {
357                tracing::warn!("Audio file not found: {}", file_name);
358            }
359        }
360    }
361
362    // Preload all audio files from the audio/ directory (for script-triggered sounds)
363    for dir in &search_dirs {
364        let audio_dir = dir.join("audio");
365        if audio_dir.is_dir() {
366            if let Ok(entries) = std::fs::read_dir(&audio_dir) {
367                for entry in entries.flatten() {
368                    let path = entry.path();
369                    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("");
370                    if ext == "ogg" || ext == "wav" || ext == "mp3" || ext == "flac" {
371                        let rel_name =
372                            format!("audio/{}", path.file_name().unwrap().to_string_lossy());
373                        if audio.engine.has_sound(&rel_name) {
374                            continue;
375                        }
376                        match audio.engine.load_sound(&rel_name, &path) {
377                            Ok(_) => {
378                                println!("Preloaded audio: {}", rel_name);
379                            }
380                            Err(e) => {
381                                tracing::warn!("Failed to preload audio '{}': {:?}", rel_name, e);
382                            }
383                        }
384                    }
385                }
386            }
387        }
388    }
389}

391/// Load `.anim.toml` files from the `animations/` directory next to the scene.
392/// Also checks one level up (game root) for projects that use a `scenes/` subdirectory.
393pub(super) fn load_animations_from_world(scene_path: &str, animation: &mut AnimationSystem) {
394    let scene_dir = Path::new(scene_path)
395        .parent()
396        .unwrap_or_else(|| Path::new("."));
397
398    let mut anim_dir = scene_dir.join("animations");
399    if !anim_dir.is_dir() {
400        // Check parent directory (game project structure)
401        if let Some(parent) = scene_dir.parent() {
402            let parent_anim = parent.join("animations");
403            if parent_anim.is_dir() {
404                anim_dir = parent_anim;
405            } else {
406                return;
407            }
408        } else {
409            return;
410        }
411    }
412
413    let entries = match std::fs::read_dir(&anim_dir) {
414        Ok(e) => e,
415        Err(_) => return,
416    };
417
418    for entry in entries.flatten() {
419        let path = entry.path();
420        if path
421            .file_name()
422            .and_then(|n| n.to_str())
423            .map_or(false, |n| n.ends_with(".anim.toml"))
424        {
425            match flint_animation::loader::load_clip_from_file(&path) {
426                Ok(clip) => {
427                    println!("Loaded animation: {} ({:.1}s)", clip.name, clip.duration);
428                    animation.add_property_clip(clip);
429                }
430                Err(e) => {
431                    tracing::warn!("Failed to load animation '{}': {:?}", path.display(), e);
432                }
433            }
434        }
435    }
436}

495/// Resolve a scene path relative to the current scene.
496/// If target is already an absolute path or starts from a known root, use it directly.
497/// Otherwise resolve relative to the current scene's directory.
498pub(super) fn resolve_scene_path(current_scene: &str, target: &str) -> String {
499    let target_path = Path::new(target);
500    if target_path.is_absolute() || target_path.exists() {
501        return target.to_string();
502    }
503
504    let current_dir = Path::new(current_scene)
505        .parent()
506        .unwrap_or_else(|| Path::new("."));
507
508    let resolved = current_dir.join(target);
509    if resolved.exists() {
510        return resolved.to_string_lossy().to_string();
511    }
512
513    // Try parent directory (game root)
514    if let Some(parent) = current_dir.parent() {
515        let parent_resolved = parent.join(target);
516        if parent_resolved.exists() {
517            return parent_resolved.to_string_lossy().to_string();
518        }
519    }
520
521    // Return as-is, let scene loader report the error
522    target.to_string()
523}

532/// Resolve unresolved model/texture asset names against procgen specs.
533///
534/// For each entity with a `model.asset` or texture name that is not already
535/// resolved (not in `config.overrides` and not in the mesh cache), check if
536/// a matching procgen spec exists. If so, generate and upload directly to GPU.
537pub(super) fn resolve_procgen_assets(
538    world: &FlintWorld,
539    resolver: &mut flint_procgen::ProcGenResolver,
540    renderer: &mut SceneRenderer,
541    device: &wgpu::Device,
542    queue: &wgpu::Queue,
543    config: &flint_render::model_loader::ModelLoadConfig,
544) {
545    // Collect unresolved asset names to avoid borrow conflicts
546    let mut unresolved_models: Vec<String> = Vec::new();
547    let mut unresolved_textures: Vec<String> = Vec::new();
548
549    for entity in world.all_entities() {
550        if let Some(comps) = world.get_components(entity.id) {
551            // Check model assets
552            if let Some(model) = comps.get(comp::MODEL) {
553                if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
554                    if !config.overrides.contains_key(name)
555                        && !renderer.mesh_cache().contains(name)
556                        && resolver.has_spec(name)
557                    {
558                        if !unresolved_models.contains(&name.to_string()) {
559                            unresolved_models.push(name.to_string());
560                        }
561                    }
562                }
563            }
564
565            // Check texture assets (material + sprite)
566            for comp_name in &[comp::MATERIAL, comp::SPRITE] {
567                if let Some(comp_data) = comps.get(*comp_name) {
568                    if let Some(tex) = comp_data.get("texture").and_then(|v| v.as_str()) {
569                        if !config.overrides.contains_key(tex)
570                            && resolver.has_spec(tex)
571                        {
572                            if !unresolved_textures.contains(&tex.to_string()) {
573                                unresolved_textures.push(tex.to_string());
574                            }
575                        }
576                    }
577                }
578            }
579        }
580    }
581
582    // Generate and upload each unresolved asset
583    for name in &unresolved_models {
584        match resolver.generate(name) {
585            Ok(output) => match output {
586                flint_procgen::GeneratorOutput::Mesh(mesh) => {
587                    upload_procgen_mesh(name, &mesh, renderer, device);
588                    tracing::info!("Procgen resolved mesh '{}' ({} tris)", name, mesh.triangle_count());
589                }
590                flint_procgen::GeneratorOutput::MeshWithLods(lods) => {
591                    if let Some(mesh) = lods.first() {
592                        upload_procgen_mesh(name, mesh, renderer, device);
593                        tracing::info!("Procgen resolved mesh '{}' (LOD0, {} tris)", name, mesh.triangle_count());
594                    }
595                }
596                flint_procgen::GeneratorOutput::Image(img) => {
597                    upload_procgen_image(name, &img, renderer, device, queue);
598                    tracing::info!("Procgen resolved image '{}' ({}x{})", name, img.width, img.height);
599                }
600                flint_procgen::GeneratorOutput::ImageSet(images) => {
601                    for (i, img) in images.iter().enumerate() {
602                        let tex_name = if i == 0 {
603                            name.clone()
604                        } else {
605                            format!("{}_{}", name, match img.channel_semantics {
606                            flint_procgen::ChannelSemantics::Color => "color",
607                            flint_procgen::ChannelSemantics::Normal => "normal",
608                            flint_procgen::ChannelSemantics::Roughness => "roughness",
609                            flint_procgen::ChannelSemantics::Metallic => "metallic",
610                            flint_procgen::ChannelSemantics::Height => "height",
611                            flint_procgen::ChannelSemantics::Mask => "mask",
612                        })
613                        };
614                        upload_procgen_image(&tex_name, img, renderer, device, queue);
615                    }
616                    tracing::info!("Procgen resolved image set '{}' ({} maps)", name, images.len());
617                }
618                _ => {
619                    tracing::warn!("Procgen spec '{}' produced unsupported output type for model", name);
620                }
621            },
622            Err(e) => {
623                tracing::warn!("Procgen generation failed for '{}': {}", name, e);
624            }
625        }
626    }
627
628    for name in &unresolved_textures {
629        if unresolved_models.contains(name) {
630            continue; // Already handled above
631        }
632        match resolver.generate(name) {
633            Ok(output) => match output {
634                flint_procgen::GeneratorOutput::Image(img) => {
635                    upload_procgen_image(name, &img, renderer, device, queue);
636                    tracing::info!("Procgen resolved texture '{}' ({}x{})", name, img.width, img.height);
637                }
638                flint_procgen::GeneratorOutput::ImageSet(images) => {
639                    for (i, img) in images.iter().enumerate() {
640                        let tex_name = if i == 0 {
641                            name.clone()
642                        } else {
643                            format!("{}_{}", name, match img.channel_semantics {
644                            flint_procgen::ChannelSemantics::Color => "color",
645                            flint_procgen::ChannelSemantics::Normal => "normal",
646                            flint_procgen::ChannelSemantics::Roughness => "roughness",
647                            flint_procgen::ChannelSemantics::Metallic => "metallic",
648                            flint_procgen::ChannelSemantics::Height => "height",
649                            flint_procgen::ChannelSemantics::Mask => "mask",
650                        })
651                        };
652                        upload_procgen_image(&tex_name, img, renderer, device, queue);
653                    }
654                    tracing::info!("Procgen resolved texture set '{}' ({} maps)", name, images.len());
655                }
656                _ => {
657                    tracing::warn!("Procgen spec '{}' produced non-image output for texture ref", name);
658                }
659            },
660            Err(e) => {
661                tracing::warn!("Procgen generation failed for texture '{}': {}", name, e);
662            }
663        }
664    }
665
666    if !unresolved_models.is_empty() || !unresolved_textures.is_empty() {
667        let stats = resolver.cache_stats();
668        tracing::debug!(
669            "Procgen cache: {} entries, {} bytes, {} hits / {} misses",
670            stats.entry_count,
671            stats.total_bytes,
672            stats.hits,
673            stats.misses,
674        );
675    }
676}

678/// Upload a procgen mesh to the GPU mesh cache.
679///
680/// Converts `flint_procgen::Vertex` → `flint_render::Vertex` (drops tangent,
681/// adds vertex color from `MaterialData`). Handles multi-submesh by uploading
682/// each submesh as a separate mesh entry.
683fn upload_procgen_mesh(
684    name: &str,
685    mesh: &flint_procgen::MeshData,
686    renderer: &mut SceneRenderer,
687    device: &wgpu::Device,
688) {
689    let default_material = flint_procgen::MaterialData::default();
690
691    if mesh.submeshes.is_empty() || mesh.submeshes.len() == 1 {
692        // Single mesh — convert all vertices with the first material's color
693        let mat = mesh.materials.first().unwrap_or(&default_material);
694        let vertices: Vec<flint_render::Vertex> = mesh
695            .vertices
696            .iter()
697            .map(|v| flint_render::Vertex {
698                position: v.position,
699                normal: v.normal,
700                color: mat.base_color,
701                uv: v.uv,
702            })
703            .collect();
704
705        let imported_mat = flint_import::ImportedMaterial {
706            name: mat.name.clone(),
707            base_color: mat.base_color,
708            metallic: mat.metallic,
709            roughness: mat.roughness,
710            base_color_texture: None,
711            normal_texture: None,
712            metallic_roughness_texture: None,
713            use_vertex_color: true,
714            alpha_mode: flint_import::AlphaMode::Opaque,
715            alpha_cutoff: 0.5,
716        };
717
718        renderer.load_procedural_mesh(device, name, &vertices, &mesh.indices, imported_mat);
719    } else {
720        // Multi-submesh: upload each as "{name}_sub{i}"
721        for (i, sub) in mesh.submeshes.iter().enumerate() {
722            let mat = mesh
723                .materials
724                .get(sub.material_index)
725                .unwrap_or(&default_material);
726
727            let start = sub.index_start as usize;
728            let count = sub.index_count as usize;
729            let sub_indices = &mesh.indices[start..start + count];
730
731            // Find min/max index to compact vertices
732            let min_idx = *sub_indices.iter().min().unwrap_or(&0) as usize;
733            let max_idx = *sub_indices.iter().max().unwrap_or(&0) as usize;
734
735            let vertices: Vec<flint_render::Vertex> = mesh.vertices
736                [min_idx..=max_idx]
737                .iter()
738                .map(|v| flint_render::Vertex {
739                    position: v.position,
740                    normal: v.normal,
741                    color: mat.base_color,
742                    uv: v.uv,
743                })
744                .collect();
745
746            let compacted_indices: Vec<u32> = sub_indices
747                .iter()
748                .map(|&idx| idx - min_idx as u32)
749                .collect();
750
751            let sub_name = format!("{name}_sub{i}");
752            let imported_mat = flint_import::ImportedMaterial {
753                name: mat.name.clone(),
754                base_color: mat.base_color,
755                metallic: mat.metallic,
756                roughness: mat.roughness,
757                base_color_texture: None,
758                normal_texture: None,
759                metallic_roughness_texture: None,
760                use_vertex_color: true,
761                alpha_mode: flint_import::AlphaMode::Opaque,
762                alpha_cutoff: 0.5,
763            };
764
765            renderer.load_procedural_mesh(
766                device,
767                &sub_name,
768                &vertices,
769                &compacted_indices,
770                imported_mat,
771            );
772        }
773    }
774}

776/// Upload a procgen image to the GPU texture cache.
777fn upload_procgen_image(
778    name: &str,
779    img: &flint_procgen::ImageData,
780    renderer: &mut SceneRenderer,
781    device: &wgpu::Device,
782    queue: &wgpu::Queue,
783) {
784    let is_normal = matches!(
785        img.channel_semantics,
786        flint_procgen::ChannelSemantics::Normal
787    );
788    match renderer.load_texture_rgba(device, queue, name, img.width, img.height, &img.pixels, is_normal) {
789        Ok(true) => {}
790        Ok(false) => {
791            tracing::debug!("Procgen texture '{}' already cached", name);
792        }
793        Err(e) => {
794            tracing::warn!("Failed to upload procgen texture '{}': {}", name, e);
795        }
796    }
797}

801/// Enqueue unresolved procgen assets for frame-budgeted generation.
802///
803/// Walks entities the same way as `resolve_procgen_assets()` but submits
804/// requests to the resolver's queue instead of generating synchronously.
805/// Used for dynamically spawned entities and runtime asset loading.
806#[allow(dead_code)]
807pub(super) fn enqueue_procgen_assets(
808    world: &FlintWorld,
809    resolver: &mut flint_procgen::ProcGenResolver,
810    renderer: &SceneRenderer,
811    config: &flint_render::model_loader::ModelLoadConfig,
812) {
813    for entity in world.all_entities() {
814        let comps = match world.get_components(entity.id) {
815            Some(c) => c,
816            None => continue,
817        };
818
819        // Read entity position for priority calculation
820        let position = comps
821            .get(comp::TRANSFORM)
822            .and_then(|t| t.get("position").and_then(toml_vec3))
823            .unwrap_or([0.0; 3]);
824
825        // Check model assets
826        if let Some(model) = comps.get(comp::MODEL) {
827            if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
828                if !config.overrides.contains_key(name)
829                    && !renderer.mesh_cache().contains(name)
830                    && resolver.has_spec(name)
831                {
832                    resolver.enqueue(name, position, flint_procgen::AssetKind::Model);
833                }
834            }
835        }
836
837        // Check texture assets (material + sprite)
838        for comp_name in &[comp::MATERIAL, comp::SPRITE] {
839            if let Some(comp_data) = comps.get(*comp_name) {
840                if let Some(tex) = comp_data.get("texture").and_then(|v| v.as_str()) {
841                    if !config.overrides.contains_key(tex) && resolver.has_spec(tex) {
842                        resolver.enqueue(tex, position, flint_procgen::AssetKind::Texture);
843                    }
844                }
845            }
846        }
847    }
848}
849
850/// Upload completed queued assets to the GPU.
851pub(super) fn upload_completed_procgen_assets(
852    completed: &[flint_procgen::CompletedAsset],
853    renderer: &mut SceneRenderer,
854    device: &wgpu::Device,
855    queue: &wgpu::Queue,
856) {
857    for asset in completed {
858        match (&asset.output, asset.asset_kind) {
859            (flint_procgen::GeneratorOutput::Mesh(mesh), _) => {
860                upload_procgen_mesh(&asset.spec_name, mesh, renderer, device);
861                tracing::info!(
862                    "Queued procgen resolved mesh '{}' ({} tris)",
863                    asset.spec_name,
864                    mesh.triangle_count()
865                );
866            }
867            (flint_procgen::GeneratorOutput::MeshWithLods(lods), _) => {
868                if let Some(mesh) = lods.first() {
869                    upload_procgen_mesh(&asset.spec_name, mesh, renderer, device);
870                    tracing::info!(
871                        "Queued procgen resolved mesh '{}' (LOD0, {} tris)",
872                        asset.spec_name,
873                        mesh.triangle_count()
874                    );
875                }
876            }
877            (flint_procgen::GeneratorOutput::Image(img), _) => {
878                upload_procgen_image(&asset.spec_name, img, renderer, device, queue);
879                tracing::info!(
880                    "Queued procgen resolved image '{}' ({}x{})",
881                    asset.spec_name,
882                    img.width,
883                    img.height
884                );
885            }
886            (flint_procgen::GeneratorOutput::ImageSet(images), _) => {
887                for (i, img) in images.iter().enumerate() {
888                    let tex_name = if i == 0 {
889                        asset.spec_name.clone()
890                    } else {
891                        format!(
892                            "{}_{}",
893                            asset.spec_name,
894                            match img.channel_semantics {
895                                flint_procgen::ChannelSemantics::Color => "color",
896                                flint_procgen::ChannelSemantics::Normal => "normal",
897                                flint_procgen::ChannelSemantics::Roughness => "roughness",
898                                flint_procgen::ChannelSemantics::Metallic => "metallic",
899                                flint_procgen::ChannelSemantics::Height => "height",
900                                flint_procgen::ChannelSemantics::Mask => "mask",
901                            }
902                        )
903                    };
904                    upload_procgen_image(&tex_name, img, renderer, device, queue);
905                }
906                tracing::info!(
907                    "Queued procgen resolved image set '{}' ({} maps)",
908                    asset.spec_name,
909                    images.len()
910                );
911            }
912            _ => {
913                tracing::warn!(
914                    "Queued procgen spec '{}' produced unsupported output type",
915                    asset.spec_name
916                );
917            }
918        }
919    }
920}
