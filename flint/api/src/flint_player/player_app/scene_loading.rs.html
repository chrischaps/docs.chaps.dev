36pub(super) fn load_terrain_from_world_inner(
37    world: &FlintWorld,
38    scene_path: &str,
39    device: &wgpu::Device,
40    queue: &wgpu::Queue,
41    scene_renderer: &mut SceneRenderer,
42    physics: &mut PhysicsSystem,
43    terrain_out: &mut Option<(flint_terrain::Terrain, flint_terrain::TerrainConfig)>,
44    grass_info_out: &mut Option<LoadedGrassInfo>,
45) {
46    use flint_core::Transform;
47    use flint_terrain::{Heightmap, Terrain, TerrainConfig};
48
49    let scene_dir = Path::new(scene_path)
50        .parent()
51        .unwrap_or_else(|| Path::new("."));
52
53    for entity in world.all_entities() {
54        let terrain_comp = match world.get_component(entity.id, comp::TERRAIN) {
55            Some(c) => c,
56            None => continue,
57        };
58
59        let heightmap_rel = match terrain_comp.get("heightmap").and_then(|v| v.as_str()) {
60            Some(s) if !s.is_empty() => s.to_string(),
61            _ => {
62                tracing::warn!("Terrain entity '{}': missing heightmap path", entity.name);
63                continue;
64            }
65        };
66
67        // Resolve heightmap path (scene dir, then parent)
68        let hm_path = {
69            let p = scene_dir.join(&heightmap_rel);
70            if p.exists() {
71                p
72            } else if let Some(parent) = scene_dir.parent() {
73                let pp = parent.join(&heightmap_rel);
74                if pp.exists() {
75                    pp
76                } else {
77                    p
78                }
79            } else {
80                p
81            }
82        };
83
84        let heightmap = match Heightmap::from_png(&hm_path) {
85            Ok(hm) => hm,
86            Err(e) => {
87                tracing::warn!("Failed to load heightmap: {}", e);
88                continue;
89            }
90        };
91
92        let get_f32 = |key: &str, default: f32| -> f32 {
93            terrain_comp.get(key).and_then(toml_f32).unwrap_or(default)
94        };
95
96        let get_i32 = |key: &str, default: i32| -> i32 {
97            terrain_comp
98                .get(key)
99                .and_then(|v| v.as_integer())
100                .map(|i| i as i32)
101                .unwrap_or(default)
102        };
103
104        let get_str = |key: &str| -> String {
105            terrain_comp
106                .get(key)
107                .and_then(|v| v.as_str())
108                .unwrap_or("")
109                .to_string()
110        };
111
112        let config = TerrainConfig {
113            heightmap_path: heightmap_rel,
114            width: get_f32("width", 256.0),
115            depth: get_f32("depth", 256.0),
116            height_scale: get_f32("height_scale", 50.0),
117            chunk_resolution: get_i32("chunk_resolution", 64) as u32,
118            texture_tile: get_f32("texture_tile", 16.0),
119            splat_map_path: get_str("splat_map"),
120            layer_textures: [
121                get_str("layer0_texture"),
122                get_str("layer1_texture"),
123                get_str("layer2_texture"),
124                get_str("layer3_texture"),
125            ],
126            metallic: get_f32("metallic", 0.0),
127            roughness: get_f32("roughness", 0.85),
128            grass: None,
129        };
130
131        let terrain = Terrain::generate(&heightmap, &config);
132
133        // Get entity transform
134        let transform = world
135            .get_component(entity.id, comp::TRANSFORM)
136            .and_then(|t| {
137                let pos = t.get("position").and_then(toml_vec3).unwrap_or([0.0; 3]);
138                Some(Transform {
139                    position: FlintVec3 {
140                        x: pos[0],
141                        y: pos[1],
142                        z: pos[2],
143                    },
144                    ..Default::default()
145                })
146            })
147            .unwrap_or_default();
148
149        // Upload to GPU
150        scene_renderer.load_terrain(
151            device,
152            queue,
153            &terrain.chunks,
154            &transform,
155            config.texture_tile,
156            config.metallic,
157            config.roughness,
158            &config.splat_map_path,
159            &config.layer_textures,
160            scene_dir,
161        );
162
163        // Register physics collider
164        let (verts, tris) = terrain.trimesh_data();
165        let offset = [
166            transform.position.x,
167            transform.position.y,
168            transform.position.z,
169        ];
170        let offset_verts: Vec<[f32; 3]> = verts
171            .iter()
172            .map(|v| [v[0] + offset[0], v[1] + offset[1], v[2] + offset[2]])
173            .collect();
174        physics.register_static_trimesh(entity.id, offset_verts, tris, 0.8, 0.1);
175
176        // Load grass if enabled
177        let grass_config = {
178            let mut gc = flint_terrain::GrassConfig::default();
179            if let Some(enabled) = terrain_comp.get("grass.enabled") {
180                if enabled.as_bool().unwrap_or(false) {
181                    gc.enabled = true;
182                    gc.density = get_f32("grass.density", gc.density);
183                    gc.max_distance = get_f32("grass.max_distance", gc.max_distance);
184                    gc.fade_start = get_f32("grass.fade_start", gc.fade_start);
185                    gc.blade_width = get_f32("grass.blade_width", gc.blade_width);
186                    gc.blade_height = get_f32("grass.blade_height", gc.blade_height);
187                    gc.height_variation = get_f32("grass.height_variation", gc.height_variation);
188                    gc.wind_speed = get_f32("grass.wind_speed", gc.wind_speed);
189                    gc.wind_strength = get_f32("grass.wind_strength", gc.wind_strength);
190                    gc.bend_radius = get_f32("grass.bend_radius", gc.bend_radius);
191                    gc.bend_strength = get_f32("grass.bend_strength", gc.bend_strength);
192                    gc.density_threshold = get_f32("grass.density_threshold", gc.density_threshold);
193                    gc.density_layer =
194                        get_i32("grass.density_layer", gc.density_layer as i32) as u32;
195                    gc.dry_amount = get_f32("grass.dry_amount", gc.dry_amount);
196
197                    // Parse vec3 fields
198                    if let Some(v) = terrain_comp.get("grass.color_base").and_then(toml_vec3) {
199                        gc.color_base = v;
200                    }
201                    if let Some(v) = terrain_comp.get("grass.color_tip").and_then(toml_vec3) {
202                        gc.color_tip = v;
203                    }
204                    if let Some(v) = terrain_comp.get("grass.color_dry").and_then(toml_vec3) {
205                        gc.color_dry = v;
206                    }
207                    if let Some(v) = terrain_comp.get("grass.wind_direction").and_then(toml_vec3) {
208                        gc.wind_direction = v;
209                    }
210                }
211            }
212            gc
213        };
214
215        if grass_config.enabled {
216            // Get raw heightmap data and splat map data for GPU upload
217            let hm_data = heightmap.clone_heights();
218            let hm_w = heightmap.width;
219            let hm_d = heightmap.depth;
220
221            // Load splat map pixels for grass compute shader
222            let splat_path = {
223                let p = scene_dir.join(&config.splat_map_path);
224                if p.exists() {
225                    p
226                } else {
227                    scene_dir
228                        .parent()
229                        .map(|pp| pp.join(&config.splat_map_path))
230                        .filter(|pp| pp.exists())
231                        .unwrap_or(p)
232                }
233            };
234
235            if let Ok(splat_img) = image::open(&splat_path) {
236                let splat_rgba = splat_img.to_rgba8();
237                let (sw, sh) = splat_rgba.dimensions();
238
239                scene_renderer.load_grass(
240                    device,
241                    queue,
242                    &grass_config,
243                    &hm_data,
244                    hm_w,
245                    hm_d,
246                    splat_rgba.as_raw(),
247                    sw,
248                    sh,
249                    offset,
250                    config.width,
251                    config.depth,
252                    config.height_scale,
253                );
254
255                tracing::info!(
256                    "Grass enabled: density={}, max_dist={}",
257                    grass_config.density,
258                    grass_config.max_distance
259                );
260
261                *grass_info_out = Some(LoadedGrassInfo {
262                    config: grass_config.clone(),
263                    terrain_entity_name: entity.name.clone(),
264                });
265            } else {
266                tracing::warn!("Grass enabled but splat map not found at {:?}", splat_path);
267            }
268        }
269
270        tracing::info!(
271            "Loaded terrain '{}': {}x{} heightmap, {} chunks, {}x{} world units",
272            entity.name,
273            heightmap.width,
274            heightmap.depth,
275            terrain.chunks.len(),
276            config.width,
277            config.depth,
278        );
279
280        *terrain_out = Some((terrain, config));
281        break; // Only one terrain entity for now
282    }
283}

285/// Build a ModelLoadConfig with catalog-resolved path overrides.
286pub(super) fn build_model_load_config(
287    scene_path: &str,
288    world: &FlintWorld,
289    catalog: Option<&AssetCatalog>,
290    store: Option<&ContentStore>,
291) -> ModelLoadConfig {
292    let mut config = ModelLoadConfig::from_scene_path(scene_path);
293
294    let (Some(cat), Some(st)) = (catalog, store) else {
295        return config;
296    };
297
298    for entity in world.all_entities() {
299        if let Some(comps) = world.get_components(entity.id) {
300            // Resolve model assets
301            if let Some(model) = comps.get(comp::MODEL) {
302                if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
303                    if !config.overrides.contains_key(name) {
304                        if let Some(path) = resolve_catalog(cat, st, name) {
305                            config.overrides.insert(name.to_string(), path);
306                        }
307                    }
308                }
309            }
310            // Resolve textures
311            for comp_name in &[comp::MATERIAL, comp::SPRITE] {
312                if let Some(comp) = comps.get(*comp_name) {
313                    if let Some(tex) = comp.get("texture").and_then(|v| v.as_str()) {
314                        if !config.overrides.contains_key(tex) {
315                            if let Some(path) = resolve_catalog(cat, st, tex) {
316                                config.overrides.insert(tex.to_string(), path);
317                            }
318                        }
319                    }
320                }
321            }
322        }
323    }
324
325    config
326}

329pub(super) fn resolve_catalog(
330    catalog: &AssetCatalog,
331    store: &ContentStore,
332    name: &str,
333) -> Option<PathBuf> {
334    let meta = catalog.get(name)?;
335    let hash = flint_core::ContentHash::from_prefixed_hex(&meta.hash)?;
336    store.get(&hash)
337}

339/// Register skeletal data from loaded models into the animation system.
340pub(super) fn register_skeletal_data(
341    load_result: &model_loader::ModelLoadResult,
342    animation: &mut AnimationSystem,
343) {
344    // Skeletons built from the first import of each asset, so entities that hit
345    // the mesh cache (`import_result: None` — every 2nd+ entity sharing a
346    // skinned asset) still get their own skeleton and animate independently.
347    let mut skeletons_by_asset: HashMap<String, Vec<Skeleton>> = HashMap::new();
348    let mut deferred: Vec<(EntityId, String)> = Vec::new();
349
350    for loaded in &load_result.models {
351        if !loaded.is_skinned {
352            continue;
353        }
354        let Some(ref import_result) = loaded.import_result else {
355            deferred.push((loaded.entity_id, loaded.asset_name.clone()));
356            continue;
357        };
358        let skeletons: Vec<Skeleton> = import_result
359            .skeletons
360            .iter()
361            .map(Skeleton::from_imported)
362            .collect();
363        for skeleton in &skeletons {
364            animation.add_skeleton(loaded.entity_id, skeleton.clone());
365        }
366        skeletons_by_asset
367            .entry(loaded.asset_name.clone())
368            .or_insert(skeletons);
369        for imported_clip in &import_result.skeletal_clips {
370            let clip = SkeletalClip::from_imported(imported_clip);
371            println!(
372                "  Skeletal clip: {} ({:.1}s, {} tracks)",
373                clip.name,
374                clip.duration,
375                clip.joint_tracks.len()
376            );
377            animation.add_skeletal_clip(clip);
378        }
379    }
380
381    for (entity_id, asset_name) in deferred {
382        match skeletons_by_asset.get(&asset_name) {
383            Some(skeletons) => {
384                for skeleton in skeletons {
385                    animation.add_skeleton(entity_id, skeleton.clone());
386                }
387            }
388            None => tracing::warn!(
389                "skinned entity {:?} uses `{}` but no import of that asset produced a skeleton this load; it will render in rest pose",
390                entity_id,
391                asset_name
392            ),
393        }
394    }
395}

414        }

416            animation.register_node_entity(loaded.entity_id, node_map.clone());
417        }
418    }

421/// Load audio files referenced by audio_source components and (unless the
422/// scene opts out via `[scene] preload_audio = false`) preload every audio
423/// file from audio/ for script-triggered sounds.
424pub(super) fn load_audio_from_world(
425    world: &FlintWorld,
426    audio: &mut AudioSystem,
427    scene_path: &str,
428    preload_all: bool,
429) {
430    let scene_dir = Path::new(scene_path)
431        .parent()
432        .unwrap_or_else(|| Path::new("."));
433
434    // Build list of directories to search: scene dir first, then parent (game root)
435    let mut search_dirs = vec![scene_dir.to_path_buf()];
436    if let Some(parent) = scene_dir.parent() {
437        search_dirs.push(parent.to_path_buf());
438    }
439
440    // Load audio files referenced by audio_source components
441    for entity in world.all_entities() {
442        let audio_file = world
443            .get_components(entity.id)
444            .and_then(|components| components.get(comp::AUDIO_SOURCE).cloned())
445            .and_then(|audio_src| {
446                audio_src
447                    .get("file")
448                    .and_then(|v| v.as_str().map(String::from))
449            });
450
451        if let Some(file_name) = audio_file {
452            if audio.engine.has_sound(&file_name) {
453                continue;
454            }
455
456            let mut loaded = false;
457            for dir in &search_dirs {
458                let audio_path = dir.join(&file_name);
459                if audio_path.exists() {
460                    match audio.engine.load_sound(&file_name, &audio_path) {
461                        Ok(_) => {
462                            println!("Loaded audio: {}", file_name);
463                            loaded = true;
464                        }
465                        Err(e) => {
466                            tracing::warn!("Failed to load audio '{}': {:?}", file_name, e);
467                        }
468                    }
469                    break;
470                }
471            }
472            if !loaded {
473                tracing::warn!("Audio file not found: {}", file_name);
474            }
475        }
476    }
477
478    // Preload all audio files from the audio/ directory (for script-triggered
479    // sounds). Scenes with no audio can skip this — decoding a large stem
480    // library dominates their load time.
481    if !preload_all {
482        tracing::info!("Scene opts out of audio preload (preload_audio = false)");
483        return;
484    }
485    for dir in &search_dirs {
486        let audio_dir = dir.join("audio");
487        if audio_dir.is_dir() {
488            if let Ok(entries) = std::fs::read_dir(&audio_dir) {
489                for entry in entries.flatten() {
490                    let path = entry.path();
491                    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("");
492                    if ext == "ogg" || ext == "wav" || ext == "mp3" || ext == "flac" {
493                        let rel_name =
494                            format!("audio/{}", path.file_name().unwrap().to_string_lossy());
495                        if audio.engine.has_sound(&rel_name) {
496                            continue;
497                        }
498                        match audio.engine.load_sound(&rel_name, &path) {
499                            Ok(_) => {
500                                println!("Preloaded audio: {}", rel_name);
501                            }
502                            Err(e) => {
503                                tracing::warn!("Failed to preload audio '{}': {:?}", rel_name, e);
504                            }
505                        }
506                    }
507                }
508            }
509        }
510    }
511}

513/// Load image files referenced by `particle_emitter.texture` fields into the
514/// renderer's texture cache. The cache key is the field string verbatim (the
515/// particle draw path looks textures up by that name); the file is searched
516/// in the scene dir, then its parent (game root) — same as audio and models.
517pub(super) fn load_particle_textures_from_world(
518    world: &FlintWorld,
519    renderer: &mut SceneRenderer,
520    device: &wgpu::Device,
521    queue: &wgpu::Queue,
522    scene_path: &str,
523) {
524    let scene_dir = Path::new(scene_path)
525        .parent()
526        .unwrap_or_else(|| Path::new("."));
527    let mut search_dirs = vec![scene_dir.to_path_buf()];
528    if let Some(parent) = scene_dir.parent() {
529        search_dirs.push(parent.to_path_buf());
530    }
531
532    for entity in world.all_entities() {
533        let tex_name = world
534            .get_components(entity.id)
535            .and_then(|components| components.get(comp::PARTICLE_EMITTER))
536            .and_then(|emitter| emitter.get("texture"))
537            .and_then(|v| v.as_str().map(String::from))
538            .filter(|s| !s.is_empty());
539
540        if let Some(tex_name) = tex_name {
541            let mut found = false;
542            for dir in &search_dirs {
543                let path = dir.join(&tex_name);
544                if path.exists() {
545                    match renderer.load_texture_file(device, queue, &tex_name, &path) {
546                        Ok(true) => println!("Loaded particle texture: {tex_name}"),
547                        Ok(false) => {} // already cached
548                        Err(e) => {
549                            tracing::warn!("Failed to load particle texture '{}': {}", tex_name, e)
550                        }
551                    }
552                    found = true;
553                    break;
554                }
555            }
556            if !found {
557                tracing::warn!("Particle texture not found: {}", tex_name);
558            }
559        }
560    }
561}

563/// Load `.anim.toml` files from the `animations/` directory next to the scene.
564/// Also checks one level up (game root) for projects that use a `scenes/` subdirectory.
565pub(super) fn load_animations_from_world(scene_path: &str, animation: &mut AnimationSystem) {
566    let scene_dir = Path::new(scene_path)
567        .parent()
568        .unwrap_or_else(|| Path::new("."));
569
570    let mut anim_dir = scene_dir.join("animations");
571    if !anim_dir.is_dir() {
572        // Check parent directory (game project structure)
573        if let Some(parent) = scene_dir.parent() {
574            let parent_anim = parent.join("animations");
575            if parent_anim.is_dir() {
576                anim_dir = parent_anim;
577            } else {
578                return;
579            }
580        } else {
581            return;
582        }
583    }
584
585    let entries = match std::fs::read_dir(&anim_dir) {
586        Ok(e) => e,
587        Err(_) => return,
588    };
589
590    for entry in entries.flatten() {
591        let path = entry.path();
592        if path
593            .file_name()
594            .and_then(|n| n.to_str())
595            .map_or(false, |n| n.ends_with(".anim.toml"))
596        {
597            match flint_animation::loader::load_clip_from_file(&path) {
598                Ok(clip) => {
599                    println!("Loaded animation: {} ({:.1}s)", clip.name, clip.duration);
600                    animation.add_property_clip(clip);
601                }
602                Err(e) => {
603                    tracing::warn!("Failed to load animation '{}': {:?}", path.display(), e);
604                }
605            }
606        }
607    }
608}

610/// Load `.sequence.toml` files from the `animations/` directory next to the scene
611/// (or one level up). Sequences are referenced by `name` from `animator.sequence`.
612pub(super) fn load_sequences_from_world(scene_path: &str, animation: &mut AnimationSystem) {
613    let Some(anim_dir) = resolve_animations_dir(scene_path) else {
614        return;
615    };
616    let entries = match std::fs::read_dir(&anim_dir) {
617        Ok(e) => e,
618        Err(_) => return,
619    };
620    for entry in entries.flatten() {
621        let path = entry.path();
622        if path
623            .file_name()
624            .and_then(|n| n.to_str())
625            .is_some_and(|n| n.ends_with(".sequence.toml"))
626        {
627            match flint_animation::load_sequence_from_file(&path) {
628                Ok(seq) => {
629                    println!(
630                        "Loaded sequence: {} ({:.1}s, {} events)",
631                        seq.name,
632                        seq.resolved_duration(),
633                        seq.events.len()
634                    );
635                    animation.add_sequence(seq);
636                }
637                Err(e) => {
638                    tracing::warn!("Failed to load sequence '{}': {:?}", path.display(), e);
639                }
640            }
641        }
642    }
643}

645/// `<scene_dir>/animations/`, falling back to `<scene_dir>/../animations/`
646/// for projects that keep scenes in a subdirectory.
647fn resolve_animations_dir(scene_path: &str) -> Option<std::path::PathBuf> {
648    let scene_dir = Path::new(scene_path)
649        .parent()
650        .unwrap_or_else(|| Path::new("."));
651    let local = scene_dir.join("animations");
652    if local.is_dir() {
653        return Some(local);
654    }
655    let up = scene_dir.parent()?.join("animations");
656    up.is_dir().then_some(up)
657}

659/// Load `.sprite.toml` files from the `animations/` directory next to the scene.
660/// Each file can define multiple sprite animation clips under `[animation.NAME]` sections.
661pub(super) fn load_sprite_animations_from_world(scene_path: &str, animation: &mut AnimationSystem) {
662    let scene_dir = Path::new(scene_path)
663        .parent()
664        .unwrap_or_else(|| Path::new("."));
665
666    let mut anim_dir = scene_dir.join("animations");
667    if !anim_dir.is_dir() {
668        if let Some(parent) = scene_dir.parent() {
669            let parent_anim = parent.join("animations");
670            if parent_anim.is_dir() {
671                anim_dir = parent_anim;
672            } else {
673                return;
674            }
675        } else {
676            return;
677        }
678    }
679
680    let entries = match std::fs::read_dir(&anim_dir) {
681        Ok(e) => e,
682        Err(_) => return,
683    };
684
685    for entry in entries.flatten() {
686        let path = entry.path();
687        if path
688            .file_name()
689            .and_then(|n| n.to_str())
690            .map_or(false, |n| n.ends_with(".sprite.toml"))
691        {
692            match flint_animation::sprite_clip::load_sprite_clips_from_file(&path) {
693                Ok(clips) => {
694                    for clip in clips {
695                        println!(
696                            "Loaded sprite animation: {} ({}ms, {} frames)",
697                            clip.name,
698                            clip.total_duration_ms(),
699                            clip.frames.len()
700                        );
701                        animation.add_sprite_clip(clip);
702                    }
703                }
704                Err(e) => {
705                    tracing::warn!(
706                        "Failed to load sprite animation '{}': {:?}",
707                        path.display(),
708                        e
709                    );
710                }
711            }
712        }
713    }
714}

716/// Resolve a scene path relative to the current scene.
717/// If target is already an absolute path or starts from a known root, use it directly.
718/// Otherwise resolve relative to the current scene's directory.
719pub(super) fn resolve_scene_path(current_scene: &str, target: &str) -> String {
720    let target_path = Path::new(target);
721    if target_path.is_absolute() || target_path.exists() {
722        return target.to_string();
723    }
724
725    let current_dir = Path::new(current_scene)
726        .parent()
727        .unwrap_or_else(|| Path::new("."));
728
729    let resolved = current_dir.join(target);
730    if resolved.exists() {
731        return resolved.to_string_lossy().to_string();
732    }
733
734    // Try parent directory (game root)
735    if let Some(parent) = current_dir.parent() {
736        let parent_resolved = parent.join(target);
737        if parent_resolved.exists() {
738            return parent_resolved.to_string_lossy().to_string();
739        }
740    }
741
742    // Return as-is, let scene loader report the error
743    target.to_string()
744}

753/// Resolve unresolved model/texture asset names against procgen specs.
754///
755/// For each entity with a `model.asset` or texture name that is not already
756/// resolved (not in `config.overrides` and not in the mesh cache), check if
757/// a matching procgen spec exists. If so, generate and upload directly to GPU.
758pub(super) fn resolve_procgen_assets(
759    world: &FlintWorld,
760    resolver: &mut flint_procgen::ProcGenResolver,
761    renderer: &mut SceneRenderer,
762    device: &wgpu::Device,
763    queue: &wgpu::Queue,
764    config: &flint_render::model_loader::ModelLoadConfig,
765) {
766    // Collect unresolved asset names to avoid borrow conflicts
767    let mut unresolved_models: Vec<String> = Vec::new();
768    let mut unresolved_textures: Vec<String> = Vec::new();
769
770    for entity in world.all_entities() {
771        if let Some(comps) = world.get_components(entity.id) {
772            // Check model assets
773            if let Some(model) = comps.get(comp::MODEL) {
774                if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
775                    if !config.overrides.contains_key(name)
776                        && !renderer.mesh_cache().contains(name)
777                        && resolver.has_spec(name)
778                    {
779                        if !unresolved_models.contains(&name.to_string()) {
780                            unresolved_models.push(name.to_string());
781                        }
782                    }
783                }
784            }
785
786            // Check texture assets (material + sprite)
787            for comp_name in &[comp::MATERIAL, comp::SPRITE] {
788                if let Some(comp_data) = comps.get(*comp_name) {
789                    if let Some(tex) = comp_data.get("texture").and_then(|v| v.as_str()) {
790                        if !config.overrides.contains_key(tex) && resolver.has_spec(tex) {
791                            if !unresolved_textures.contains(&tex.to_string()) {
792                                unresolved_textures.push(tex.to_string());
793                            }
794                        }
795                    }
796                }
797            }
798        }
799    }
800
801    // Generate and upload each unresolved asset
802    for name in &unresolved_models {
803        match resolver.generate(name) {
804            Ok(output) => match output {
805                flint_procgen::GeneratorOutput::Mesh(mesh) => {
806                    upload_procgen_mesh(name, &mesh, renderer, device);
807                    tracing::info!(
808                        "Procgen resolved mesh '{}' ({} tris)",
809                        name,
810                        mesh.triangle_count()
811                    );
812                }
813                flint_procgen::GeneratorOutput::MeshWithLods(lods) => {
814                    if let Some(mesh) = lods.first() {
815                        upload_procgen_mesh(name, mesh, renderer, device);
816                        tracing::info!(
817                            "Procgen resolved mesh '{}' (LOD0, {} tris)",
818                            name,
819                            mesh.triangle_count()
820                        );
821                    }
822                }
823                flint_procgen::GeneratorOutput::Image(img) => {
824                    upload_procgen_image(name, &img, renderer, device, queue);
825                    tracing::info!(
826                        "Procgen resolved image '{}' ({}x{})",
827                        name,
828                        img.width,
829                        img.height
830                    );
831                }
832                flint_procgen::GeneratorOutput::ImageSet(images) => {
833                    for (i, img) in images.iter().enumerate() {
834                        let tex_name = if i == 0 {
835                            name.clone()
836                        } else {
837                            format!(
838                                "{}_{}",
839                                name,
840                                match img.channel_semantics {
841                                    flint_procgen::ChannelSemantics::Color => "color",
842                                    flint_procgen::ChannelSemantics::Normal => "normal",
843                                    flint_procgen::ChannelSemantics::Roughness => "roughness",
844                                    flint_procgen::ChannelSemantics::Metallic => "metallic",
845                                    flint_procgen::ChannelSemantics::Height => "height",
846                                    flint_procgen::ChannelSemantics::Mask => "mask",
847                                }
848                            )
849                        };
850                        upload_procgen_image(&tex_name, img, renderer, device, queue);
851                    }
852                    tracing::info!(
853                        "Procgen resolved image set '{}' ({} maps)",
854                        name,
855                        images.len()
856                    );
857                }
858                _ => {
859                    tracing::warn!(
860                        "Procgen spec '{}' produced unsupported output type for model",
861                        name
862                    );
863                }
864            },
865            Err(e) => {
866                tracing::warn!("Procgen generation failed for '{}': {}", name, e);
867            }
868        }
869    }
870
871    for name in &unresolved_textures {
872        if unresolved_models.contains(name) {
873            continue; // Already handled above
874        }
875        match resolver.generate(name) {
876            Ok(output) => match output {
877                flint_procgen::GeneratorOutput::Image(img) => {
878                    upload_procgen_image(name, &img, renderer, device, queue);
879                    tracing::info!(
880                        "Procgen resolved texture '{}' ({}x{})",
881                        name,
882                        img.width,
883                        img.height
884                    );
885                }
886                flint_procgen::GeneratorOutput::ImageSet(images) => {
887                    for (i, img) in images.iter().enumerate() {
888                        let tex_name = if i == 0 {
889                            name.clone()
890                        } else {
891                            format!(
892                                "{}_{}",
893                                name,
894                                match img.channel_semantics {
895                                    flint_procgen::ChannelSemantics::Color => "color",
896                                    flint_procgen::ChannelSemantics::Normal => "normal",
897                                    flint_procgen::ChannelSemantics::Roughness => "roughness",
898                                    flint_procgen::ChannelSemantics::Metallic => "metallic",
899                                    flint_procgen::ChannelSemantics::Height => "height",
900                                    flint_procgen::ChannelSemantics::Mask => "mask",
901                                }
902                            )
903                        };
904                        upload_procgen_image(&tex_name, img, renderer, device, queue);
905                    }
906                    tracing::info!(
907                        "Procgen resolved texture set '{}' ({} maps)",
908                        name,
909                        images.len()
910                    );
911                }
912                _ => {
913                    tracing::warn!(
914                        "Procgen spec '{}' produced non-image output for texture ref",
915                        name
916                    );
917                }
918            },
919            Err(e) => {
920                tracing::warn!("Procgen generation failed for texture '{}': {}", name, e);
921            }
922        }
923    }
924
925    if !unresolved_models.is_empty() || !unresolved_textures.is_empty() {
926        let stats = resolver.cache_stats();
927        tracing::debug!(
928            "Procgen cache: {} entries, {} bytes, {} hits / {} misses",
929            stats.entry_count,
930            stats.total_bytes,
931            stats.hits,
932            stats.misses,
933        );
934    }
935}

937/// Upload a procgen mesh to the GPU mesh cache.
938///
939/// Converts `flint_procgen::Vertex` → `flint_render::Vertex` (drops tangent,
940/// adds vertex color from `MaterialData`). Handles multi-submesh by uploading
941/// each submesh as a separate mesh entry.
942fn upload_procgen_mesh(
943    name: &str,
944    mesh: &flint_procgen::MeshData,
945    renderer: &mut SceneRenderer,
946    device: &wgpu::Device,
947) {
948    let default_material = flint_procgen::MaterialData::default();
949
950    if mesh.submeshes.is_empty() || mesh.submeshes.len() == 1 {
951        // Single mesh — convert all vertices with the first material's color
952        let mat = mesh.materials.first().unwrap_or(&default_material);
953        let vertices: Vec<flint_render::Vertex> = mesh
954            .vertices
955            .iter()
956            .map(|v| flint_render::Vertex {
957                position: v.position,
958                normal: v.normal,
959                color: mat.base_color,
960                uv: v.uv,
961            })
962            .collect();
963
964        let imported_mat = flint_import::ImportedMaterial {
965            name: mat.name.clone(),
966            base_color: mat.base_color,
967            metallic: mat.metallic,
968            roughness: mat.roughness,
969            base_color_texture: None,
970            normal_texture: None,
971            metallic_roughness_texture: None,
972            use_vertex_color: true,
973            alpha_mode: flint_import::AlphaMode::Opaque,
974            alpha_cutoff: 0.5,
975        };
976
977        renderer.load_procedural_mesh(device, name, &vertices, &mesh.indices, imported_mat);
978    } else {
979        // Multi-submesh: upload each as "{name}_sub{i}"
980        for (i, sub) in mesh.submeshes.iter().enumerate() {
981            let mat = mesh
982                .materials
983                .get(sub.material_index)
984                .unwrap_or(&default_material);
985
986            let start = sub.index_start as usize;
987            let count = sub.index_count as usize;
988            let sub_indices = &mesh.indices[start..start + count];
989
990            // Find min/max index to compact vertices
991            let min_idx = *sub_indices.iter().min().unwrap_or(&0) as usize;
992            let max_idx = *sub_indices.iter().max().unwrap_or(&0) as usize;
993
994            let vertices: Vec<flint_render::Vertex> = mesh.vertices[min_idx..=max_idx]
995                .iter()
996                .map(|v| flint_render::Vertex {
997                    position: v.position,
998                    normal: v.normal,
999                    color: mat.base_color,
1000                    uv: v.uv,
1001                })
1002                .collect();
1003
1004            let compacted_indices: Vec<u32> = sub_indices
1005                .iter()
1006                .map(|&idx| idx - min_idx as u32)
1007                .collect();
1008
1009            let sub_name = format!("{name}_sub{i}");
1010            let imported_mat = flint_import::ImportedMaterial {
1011                name: mat.name.clone(),
1012                base_color: mat.base_color,
1013                metallic: mat.metallic,
1014                roughness: mat.roughness,
1015                base_color_texture: None,
1016                normal_texture: None,
1017                metallic_roughness_texture: None,
1018                use_vertex_color: true,
1019                alpha_mode: flint_import::AlphaMode::Opaque,
1020                alpha_cutoff: 0.5,
1021            };
1022
1023            renderer.load_procedural_mesh(
1024                device,
1025                &sub_name,
1026                &vertices,
1027                &compacted_indices,
1028                imported_mat,
1029            );
1030        }
1031    }
1032}

1034/// Upload a procgen image to the GPU texture cache.
1035fn upload_procgen_image(
1036    name: &str,
1037    img: &flint_procgen::ImageData,
1038    renderer: &mut SceneRenderer,
1039    device: &wgpu::Device,
1040    queue: &wgpu::Queue,
1041) {
1042    let is_normal = matches!(
1043        img.channel_semantics,
1044        flint_procgen::ChannelSemantics::Normal
1045    );
1046    match renderer.load_texture_rgba(
1047        device,
1048        queue,
1049        name,
1050        img.width,
1051        img.height,
1052        &img.pixels,
1053        is_normal,
1054    ) {
1055        Ok(true) => {}
1056        Ok(false) => {
1057            tracing::debug!("Procgen texture '{}' already cached", name);
1058        }
1059        Err(e) => {
1060            tracing::warn!("Failed to upload procgen texture '{}': {}", name, e);
1061        }
1062    }
1063}

1067/// Enqueue unresolved procgen assets for frame-budgeted generation.
1068///
1069/// Walks entities the same way as `resolve_procgen_assets()` but submits
1070/// requests to the resolver's queue instead of generating synchronously.
1071/// Used for dynamically spawned entities and runtime asset loading.
1072#[allow(dead_code)]
1073pub(super) fn enqueue_procgen_assets(
1074    world: &FlintWorld,
1075    resolver: &mut flint_procgen::ProcGenResolver,
1076    renderer: &SceneRenderer,
1077    config: &flint_render::model_loader::ModelLoadConfig,
1078) {
1079    for entity in world.all_entities() {
1080        let comps = match world.get_components(entity.id) {
1081            Some(c) => c,
1082            None => continue,
1083        };
1084
1085        // Read entity position for priority calculation
1086        let position = comps
1087            .get(comp::TRANSFORM)
1088            .and_then(|t| t.get("position").and_then(toml_vec3))
1089            .unwrap_or([0.0; 3]);
1090
1091        // Check model assets
1092        if let Some(model) = comps.get(comp::MODEL) {
1093            if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
1094                if !config.overrides.contains_key(name)
1095                    && !renderer.mesh_cache().contains(name)
1096                    && resolver.has_spec(name)
1097                {
1098                    resolver.enqueue(name, position, flint_procgen::AssetKind::Model);
1099                }
1100            }
1101        }
1102
1103        // Check texture assets (material + sprite)
1104        for comp_name in &[comp::MATERIAL, comp::SPRITE] {
1105            if let Some(comp_data) = comps.get(*comp_name) {
1106                if let Some(tex) = comp_data.get("texture").and_then(|v| v.as_str()) {
1107                    if !config.overrides.contains_key(tex) && resolver.has_spec(tex) {
1108                        resolver.enqueue(tex, position, flint_procgen::AssetKind::Texture);
1109                    }
1110                }
1111            }
1112        }
1113    }
1114}

1167                            }
1168                        )
1169                    };
1170                    upload_procgen_image(&tex_name, img, renderer, device, queue);
1171                }
1172                tracing::info!(
1173                    "Queued procgen resolved image set '{}' ({} maps)",
1174                    asset.spec_name,
1175                    images.len()
1176                );
1177            }
1178            _ => {
1179                tracing::warn!(
1180                    "Queued procgen spec '{}' produced unsupported output type",

1183            }
1184        }
