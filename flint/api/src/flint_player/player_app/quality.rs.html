1//! Runtime quality settings (FlintKart#55): one path from every source to
2//! the renderer. The desktop's `--msaa`, `--render-scale`, `--quality` and
3//! `FLINT_SHADOWS` / `FLINT_SHADOW_RES`, the page's `?msaa=`, `?scale=` and
4//! `?quality=`, an Android dev profile and a script's `set_quality` all end
5//! in [`PlayerApp::apply_quality`], between frames.
6//!
7//! The render scale stays the player's: the setting's `scale` is the
8//! quality's `render_scale`, and `update_render_scale` still caps it by the
9//! pixel ratio and steps it under load each frame. Everything else is the
10//! renderer's (`SceneRenderer::set_quality`).
11
12use super::PlayerApp;
13use flint_render::QualitySettings;
14
15impl PlayerApp {
16    /// The settings startup asks for: the MSAA and render-scale fields, the
17    /// shadow switches in the environment, then `quality_overrides`.
18    fn startup_quality(&self) -> QualitySettings {
19        let mut q = QualitySettings {
20            render_scale: self.render_scale.scale,
21            render_scale_auto: self.render_scale.dynamic,
22            msaa: self.msaa_sample_count,
23            ..QualitySettings::default()
24        };
25        // FLINT_SHADOWS=0 starts with the shadow pass off, so a perf run
26        // can A/B its cost headlessly.
27        if std::env::var("FLINT_SHADOWS").is_ok_and(|v| v == "0") {
28            q.shadows = false;
29        }
30        // FLINT_SHADOW_RES=1024 sizes each cascade's map (FlintKart#54).
31        if let Some(res) = std::env::var("FLINT_SHADOW_RES")
32            .ok()
33            .and_then(|v| v.parse::<u32>().ok())
34        {
35            q.shadow_resolution = res;
36        }
37        for e in q.apply_str(&self.quality_overrides) {
38            tracing::warn!("--quality: {e}");
39        }
40        q.clamped()
41    }
42
43    /// Apply the startup settings and tell scripts what the GPU is. Call
44    /// once the renderer exists and before any script's `on_init`.
45    pub(super) fn init_quality(&mut self) {
46        if let Some(context) = &self.render_context {
47            let info = flint_render::gpu_info(&context.adapter);
48            tracing::info!(
49                "[quality] GPU {:?}: {} {} on {} ({})",
50                info.name,
51                info.vendor,
52                info.device_type,
53                info.backend,
54                info.class
55            );
56            self.script.set_gpu_info(info);
57        }
58        let q = self.startup_quality();
59        self.apply_quality(q);
60        tracing::info!("[quality] {}", self.quality.to_query());
61    }
62
63    /// Apply `q` now. Call between frames: an MSAA change rebuilds the
64    /// scene pipelines, a shadow-resolution change the shadow maps.
65    pub(crate) fn apply_quality(&mut self, q: QualitySettings) {
66        let q = q.clamped();
67        if q.render_scale_auto != self.render_scale.dynamic || !q.render_scale_auto {
68            // A fresh start under the new ceiling (and none when fixed).
69            self.dynamic_scale = Default::default();
70        }
71        self.render_scale.scale = q.render_scale;
72        self.render_scale.dynamic = q.render_scale_auto;
73        let mut applied = q;
74        if let (Some(context), Some(renderer)) = (&self.render_context, &mut self.scene_renderer) {
75            // The scale is update_render_scale's to set (with the pixel
76            // ratio cap and the dynamic step); keep the one drawn now so
77            // the HDR chain is rebuilt once, there, not twice.
78            let r = QualitySettings {
79                render_scale: renderer.render_scale(),
80                ..q
81            };
82            applied = QualitySettings {
83                render_scale: q.render_scale,
84                render_scale_auto: q.render_scale_auto,
85                ..renderer.set_quality(&context.device, &r)
86            };
87        }
88        self.quality = applied;
89        self.script.set_quality_state(applied);
90    }
91
92    /// A script's `set_quality` from this frame, applied before the frame
93    /// draws.
94    pub(super) fn apply_script_quality(&mut self) {
95        if let Some(q) = self.script.take_quality_request() {
96            if q != self.quality {
97                tracing::info!("[quality] script: {}", q.clamped().to_query());
98                self.apply_quality(q);
99            } else {
100                // Asked for what's in force; just hand the state back.
101                self.script.set_quality_state(self.quality);
102            }
103        }
104    }
105
106    /// After Android brings the surface back: the renderer and its settings
107    /// outlive the surface, but the surface may come back another size, so
108    /// apply the settings again over the resized chain.
109    #[cfg_attr(not(target_os = "android"), allow(dead_code))]
110    pub(super) fn reapply_quality(&mut self) {
111        let q = self.quality;
112        self.apply_quality(q);
113    }
114
115    /// The quality settings in force.
116    pub fn quality(&self) -> &QualitySettings {
117        &self.quality
118    }
119}
