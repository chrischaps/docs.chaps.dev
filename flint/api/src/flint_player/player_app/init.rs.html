142    /// Apply scene-level camera configuration from `CameraDef`
143    pub(super) fn apply_camera_def(&mut self) {
144        if let Some(cam) = &self.scene_camera {
145            if cam.projection == "orthographic" {
146                self.camera.orthographic = true;
147                if cam.ortho_height > 0.0 {
148                    self.camera.ortho_height = cam.ortho_height;
149                }
150            } else {
151                self.camera.orthographic = false;
152                self.camera.ortho_height = 0.0;
153            }
154            if let Some(pos) = cam.position {
155                self.camera.position = flint_core::Vec3::new(pos[0], pos[1], pos[2]);
156            }
157            if let Some(target) = cam.target {
158                self.camera.target = flint_core::Vec3::new(target[0], target[1], target[2]);
159            }
160            if let Some(fov) = cam.fov {
161                self.camera.fov = fov;
162            }
163            if let Some(near) = cam.near {
164                self.camera.near = near;
165            }
166            if let Some(far) = cam.far {
167                self.camera.far = far;
168            }
169
170            // Derive orbit parameters from position/target so update_orbit() stays consistent
171            if cam.position.is_some() {
172                let dir = self.camera.position - self.camera.target;
173                let dist = dir.length();
174                if dist > 0.001 {
175                    self.camera.distance = dist;
176                    let n = dir * (1.0 / dist);
177                    self.camera.pitch = n.y.asin();
178                    self.camera.yaw = n.x.atan2(n.z);
179                }
180            }
181        }
182    }

184    pub(super) fn initialize(&mut self, event_loop: &ActiveEventLoop) -> Result<()> {
185        // Window, then renderer + model/procgen loading (context and
186        // renderer stay locals until the environment pass is done — they
187        // are stored on self only once fully built, as before).
188        let window = self.create_window(event_loop)?;
189        let (render_context, mut scene_renderer) = self.init_render_and_models(&window)?;
190
191        // Input bindings, gamepad backend, egui overlay
192        self.init_input_and_egui(&window, &render_context);
193
194        // Splines, ambient/wrap/sheen, skybox, terrain, camera, post config
195        self.apply_scene_environment(&render_context, &mut scene_renderer);
196
197        self.render_context = Some(render_context);
198        self.scene_renderer = Some(scene_renderer);
199
200        // Physics → audio → music session → animation → particles → scripts
201        // (audio before the scene-declared session, session before script
202        // init so the conducted context exists from the scripts' first
203        // frame — F3/F4 ordering).
204        self.init_game_systems()?;
205
206        // Capture cursor for first-person look (only if FPS player exists).
207        // On Android, always set cursor_captured = true so touch input flows
208        // without requiring a click-to-capture gate.
209        #[cfg(target_os = "android")]
210        {
211            self.cursor_captured = true;
212        }
213        #[cfg(not(target_os = "android"))]
214        if self.physics.has_player_entity() {
215            self.capture_cursor();
216        }
217
218        Ok(())
219    }

241    fn init_render_and_models(
242        &mut self,
243        window: &Arc<Window>,
244    ) -> Result<(RenderContext, SceneRenderer)> {
245        // Initialize rendering
246        let render_context = pollster::block_on(RenderContext::new(window.clone()))
247            .context("Failed to initialize render context")?;
248
249        self.camera.aspect = render_context.aspect_ratio();
250        self.camera.fov = 70.0; // Slightly wider FOV for first-person
251
252        let mut scene_renderer = SceneRenderer::new(
253            &render_context,
254            flint_render::RendererConfig {
255                sample_count: self.msaa_sample_count,
256                ..Default::default()
257            },
258        );
259
260        // Rebuild component index as a safety net after scene loading
261        self.world.rebuild_component_index();
262
263        // Set scene_dir for font/texture resolution
264        scene_renderer.scene_dir = Path::new(&self.scene_path)
265            .parent()
266            .map(|p| p.to_path_buf());
267
268        // Load models from world (including skeletal data)
269        let config = build_model_load_config(
270            &self.scene_path,
271            &self.world,
272            self.catalog.as_ref(),
273            self.content_store.as_ref(),
274        );
275
276        // Discover procgen specs and resolve unresolved assets before model loading
277        {
278            let scene_dir = Path::new(&self.scene_path)
279                .parent()
280                .unwrap_or(Path::new("."));
281            let mut spec_dirs = vec![scene_dir.join("specs")];
282            if let Some(parent) = scene_dir.parent() {
283                spec_dirs.push(parent.join("specs"));
284            }
285            spec_dirs.push(scene_dir.join("models"));
286            let dir_refs: Vec<&Path> = spec_dirs
287                .iter()
288                .filter(|d| d.is_dir())
289                .map(|d| d.as_path())
290                .collect();
291            self.procgen_resolver.discover_and_index(&dir_refs);
292
293            resolve_procgen_assets(
294                &self.world,
295                &mut self.procgen_resolver,
296                &mut scene_renderer,
297                &render_context.device,
298                &render_context.queue,
299                &config,
300            );
301        }
302
303        let load_result = model_loader::load_models_from_world(
304            &mut self.world,
305            &mut scene_renderer,
306            &render_context.device,
307            &render_context.queue,
308            &config,
309        );
310        register_skeletal_data(&load_result, &mut self.animation);
311        register_node_animation_data(&load_result, &mut self.animation);
312        self.skeletal_entity_assets = load_result.skinned_entities;
313        scene_renderer.update_from_world(&self.world, &render_context.device);
314
315        Ok((render_context, scene_renderer))
316    }

346    fn apply_scene_environment(
347        &mut self,
348        render_context: &RenderContext,
349        scene_renderer: &mut SceneRenderer,
350    ) {
351        // Generate procedural geometry from spline + spline_mesh entities
352        crate::spline_gen::load_splines(
353            &self.scene_path,
354            &mut self.world,
355            scene_renderer,
356            Some(&mut self.physics),
357            &render_context.device,
358        );
359
360        // Refresh renderer with any new procedural meshes
361        scene_renderer.update_from_world(&self.world, &render_context.device);
362
363        // Scene-authored hemisphere ambient + diffuse wrap (absent = renderer default)
364        if let Some((sky, ground)) = self.scene_ambient {
365            scene_renderer.set_ambient(sky, ground);
366        }
367        if let Some(wrap) = self.scene_diffuse_wrap {
368            scene_renderer.set_diffuse_wrap(wrap);
369        }
370        if let Some(oren) = self.scene_oren_nayar {
371            scene_renderer.set_oren_nayar(oren);
372        }
373        if let Some((color, strength)) = self.scene_sheen {
374            scene_renderer.set_sheen(color, strength);
375        }
376
377        // Load skybox if configured
378        if let Some(skybox_rel) = &self.skybox_path {
379            let scene_dir = Path::new(&self.scene_path)
380                .parent()
381                .unwrap_or_else(|| Path::new("."));
382
383            // Search scene dir first, then parent (game root)
384            let skybox_path = {
385                let p = scene_dir.join(skybox_rel);
386                if p.exists() {
387                    p
388                } else if let Some(parent) = scene_dir.parent() {
389                    parent.join(skybox_rel)
390                } else {
391                    p
392                }
393            };
394
395            if skybox_path.exists() {
396                scene_renderer.load_skybox(
397                    &render_context.device,
398                    &render_context.queue,
399                    &skybox_path,
400                );
401            } else {
402                tracing::warn!("Skybox file not found: {}", skybox_path.display());
403            }
404        }
405
406        // Load terrain from world
407        #[cfg(feature = "debug-hud")]
408        self.debug_panels.clear();
409        self.load_terrain_from_world(
410            &render_context.device,
411            &render_context.queue,
412            scene_renderer,
413        );
414
415        // Set terrain height callback for scripts
416        self.update_terrain_height_fn();
417
418        // Apply scene-level camera configuration
419        self.apply_camera_def();
420        self.apply_camera_tuning();
421
422        // Apply scene-level post-processing config
423        if let Some(pp_def) = &self.scene_post_process {
424            scene_renderer
425                .set_post_process_config(scene_loading::post_process_config_from_def(pp_def));
426            scene_renderer.ensure_kuwahara_resources(&render_context.device, &render_context.queue);
427            scene_renderer.ensure_fxaa_resources(&render_context.device);
428        }
429    }

431    fn init_game_systems(&mut self) -> Result<()> {
432        // Initialize physics
433        self.physics
434            .initialize(&mut self.world)
435            .context("Failed to initialize physics")?;
436
437        // Initialize audio
438        load_audio_from_world(
439            &self.world,
440            &mut self.audio,
441            &self.scene_path,
442            self.scene_preload_audio,
443        );
444        self.audio
445            .initialize(&mut self.world)
446            .unwrap_or_else(|e| tracing::warn!("Audio init failed: {:?}", e));
447
448        // Scene-declared music session (F3): started before script init so a
449        // conducted context exists from the scripts' first frame (F4).
450        self.start_music_session();
451
452        // Initialize animation
453        load_animations_from_world(&self.scene_path, &mut self.animation);
454        load_sprite_animations_from_world(&self.scene_path, &mut self.animation);
455        load_sequences_from_world(&self.scene_path, &mut self.animation);
456        self.animation
457            .initialize(&mut self.world)
458            .unwrap_or_else(|e| tracing::warn!("Animation init failed: {:?}", e));
459
460        // Initialize particles
461        self.particles
462            .initialize(&mut self.world)
463            .unwrap_or_else(|e| tracing::warn!("Particles init failed: {:?}", e));
464        self.load_particle_textures();
465
466        // Initialize scripting (state_scope required so on_init can access persist store)
467        load_scripts_from_world(&self.scene_path, &mut self.script);
468        self.script.set_current_scene(&self.scene_path);
469        {
470            let _state_scope = self.script.state_scope(
471                &mut self.state_machine,
472                &mut self.persistent_store,
473                &self.physics,
474            );
475            self.script
476                .initialize(&mut self.world)
477                .unwrap_or_else(|e| tracing::warn!("Script init failed: {:?}", e));
478        }
479
480        Ok(())
481    }

533    /// Update the terrain height callback on the script system.
534    /// Call after terrain loading or clearing.
535    pub(super) fn update_terrain_height_fn(&mut self) {
536        if let Some((ref terrain, ref config)) = self.terrain {
537            let heights = terrain.heightmap.clone_heights();
538            let hm_w = terrain.heightmap.width;
539            let hm_d = terrain.heightmap.depth;
540            let world_w = config.width;
541            let world_d = config.depth;
542            let height_scale = config.height_scale;
543            self.script
544                .set_terrain_height_fn(Some(Box::new(move |x: f32, z: f32| {
545                    // Convert world coords to normalized UV
546                    let u = x / world_w;
547                    let v = z / world_d;
548                    // Bilinear sample from heights array
549                    let fx = u * (hm_w as f32 - 1.0);
550                    let fz = v * (hm_d as f32 - 1.0);
551                    let ix = (fx as u32).min(hm_w - 2);
552                    let iz = (fz as u32).min(hm_d - 2);
553                    let tx = fx - ix as f32;
554                    let tz = fz - iz as f32;
555                    let idx = |col: u32, row: u32| -> f32 { heights[(row * hm_w + col) as usize] };
556                    let h00 = idx(ix, iz);
557                    let h10 = idx(ix + 1, iz);
558                    let h01 = idx(ix, iz + 1);
559                    let h11 = idx(ix + 1, iz + 1);
560                    let h = h00 * (1.0 - tx) * (1.0 - tz)
561                        + h10 * tx * (1.0 - tz)
562                        + h01 * (1.0 - tx) * tz
563                        + h11 * tx * tz;
564                    h * height_scale
565                })));
566        } else {
567            self.script.set_terrain_height_fn(None);
568        }
569    }

571    pub(super) fn configure_input_bindings(&mut self) -> Result<()> {
572        self.input
573            .load_bindings(InputConfig::built_in_defaults())
574            .context("failed to load built-in input defaults")?;
575
576        let paths = resolve_input_paths(
577            Path::new(&self.scene_path),
578            self.scene_input_config.as_deref(),
579            self.input_config_override.as_deref(),
580        );
581
582        if let Some(path) = &paths.game_default {
583            if path.exists() {
584                let cfg = InputConfig::load_from_file(path).with_context(|| {
585                    format!("failed to load game input config '{}'", path.display())
586                })?;
587                self.input
588                    .merge_bindings(cfg)
589                    .context("failed to merge game input config")?;
590            }
591        }
592
593        if let Some(path) = &paths.user_override {
594            if path.exists() {
595                let cfg = InputConfig::load_from_file(path).with_context(|| {
596                    format!("failed to load user input config '{}'", path.display())
597                })?;
598                self.user_override_config = cfg.clone();
599                self.input
600                    .merge_bindings(cfg)
601                    .context("failed to merge user input config")?;
602            }
603        }
604
605        if let Some(path) = &paths.cli_override {
606            if path.exists() {
607                let cfg = InputConfig::load_from_file(path).with_context(|| {
608                    format!("failed to load CLI input config '{}'", path.display())
609                })?;
610                self.input
611                    .merge_bindings(cfg)
612                    .context("failed to merge CLI input config")?;
613            }
614        }
615
616        if self.user_override_config.version == 0 {
617            self.user_override_config.version = 1;
618        }
619        if self.user_override_config.game_id.trim().is_empty() {
620            self.user_override_config.game_id = self.input.config().game_id.clone();
621        }
622
623        self.input_config_paths = Some(paths);
624        Ok(())
625    }

644    /// Load a sprite texture for UI rendering. Called lazily when a draw_sprite
645    /// command references a name not yet in ui_textures.
646    pub fn load_ui_texture(&mut self, name: &str) -> bool {
647        if self.ui_textures.contains_key(name) {
648            return true;
649        }
650
651        let scene_dir = Path::new(&self.scene_path)
652            .parent()
653            .unwrap_or_else(|| Path::new("."));
654
655        // Search: scene_dir/sprites/{name} → game_root/sprites/{name} → scene_dir/{name}
656        let candidates = [
657            scene_dir.join("sprites").join(name),
658            scene_dir
659                .parent()
660                .map(|p| p.join("sprites").join(name))
661                .unwrap_or_default(),
662            scene_dir.join(name),
663        ];
664
665        for path in &candidates {
666            if path.exists() {
667                if let Ok(img) = image::open(path) {
668                    let rgba = img.to_rgba8();
669                    let (w, h) = rgba.dimensions();
670                    let color_image =
671                        egui::ColorImage::from_rgba_unmultiplied([w as usize, h as usize], &rgba);
672                    let tex_handle =
673                        self.egui_ctx
674                            .load_texture(name, color_image, egui::TextureOptions::LINEAR);
675                    self.ui_textures.insert(name.to_string(), tex_handle);
676                    println!("Loaded UI sprite: {}", name);
677                    return true;
678                }
679            }
680        }
681
682        tracing::warn!("UI sprite not found: {}", name);
683        false
684    }
685
686    /// Pre-scan draw commands and load any sprite textures that haven't been loaded yet
687    pub(super) fn load_pending_sprites(&mut self) {
688        let sprite_names: Vec<String> = self
689            .draw_commands
690            .iter()
691            .filter_map(|cmd| {
692                if let DrawCommand::Sprite { name, .. } = cmd {
693                    if !self.ui_textures.contains_key(name.as_str()) {
694                        Some(name.clone())
695                    } else {
696                        None
697                    }
698                } else {
699                    None
700                }
701            })
702            .collect();
703
704        for name in sprite_names {
705            self.load_ui_texture(&name);
706        }
707    }
