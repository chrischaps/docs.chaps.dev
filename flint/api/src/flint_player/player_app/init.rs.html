173    /// Install a game's extension. Its script API is registered right away,
174    /// so add extensions before the event loop starts; after that, scripts
175    /// already compiled still see the new functions, because Rhai resolves
176    /// calls when they run. See `PlayerExtension` for when each hook runs.
177    pub fn add_extension(&mut self, mut extension: Box<dyn crate::extension::PlayerExtension>) {
178        extension.register_script_api(self.script.rhai_engine_mut());
179        self.extensions.push(extension);
180        // A late extension still gets its `fonts` hook.
181        if self.ui_fonts_installed {
182            self.ui_fonts_installed = false;
183            self.refresh_project_fonts();
184        }
185        // And its debug panels, before the next scene load builds them again.
186        #[cfg(feature = "debug-hud")]
187        self.create_extension_debug_panels();
188    }

201    /// Take the scene file's player settings: audio preload, skybox,
202    /// ambient and shading levers, camera and post-processing. Both entries
203    /// (`main.rs` and the web's `flint_web.rs`) call it after `new`.
204    pub fn apply_scene_settings(&mut self, scene_file: &flint_scene::SceneFile) {
205        self.scene_preload_audio = scene_file.scene.preload_audio;
206
207        // Skybox path + ambient from scene environment settings
208        if let Some(env) = &scene_file.environment {
209            self.skybox_path = env.skybox.clone();
210            if env.ambient_sky.is_some() || env.ambient_ground.is_some() {
211                self.scene_ambient = Some((
212                    env.ambient_sky
213                        .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_SKY),
214                    env.ambient_ground
215                        .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_GROUND),
216                ));
217            }
218            self.scene_diffuse_wrap = env.diffuse_wrap;
219            self.scene_oren_nayar = env.oren_nayar;
220            self.scene_sheen = env
221                .sheen_strength
222                .map(|s| (env.sheen_color.unwrap_or([1.0; 3]), s));
223        }
224
225        self.scene_camera = scene_file.camera.clone();
226        // User styles first: the post_process tables are read against the
227        // registry, and a table for an unloaded style is dropped
228        super::scene_loading::load_user_styles(&self.scene_path);
229        self.scene_post_process = scene_file.post_process.clone();
230    }

245    /// Apply scene-level camera configuration from `CameraDef`
246    pub(super) fn apply_camera_def(&mut self) {
247        if let Some(cam) = &self.scene_camera {
248            if cam.projection == "orthographic" {
249                self.camera.orthographic = true;
250                if cam.ortho_height > 0.0 {
251                    self.camera.ortho_height = cam.ortho_height;
252                }
253            } else {
254                self.camera.orthographic = false;
255                self.camera.ortho_height = 0.0;
256            }
257            if let Some(pos) = cam.position {
258                self.camera.position = flint_core::Vec3::new(pos[0], pos[1], pos[2]);
259            }
260            if let Some(target) = cam.target {
261                self.camera.target = flint_core::Vec3::new(target[0], target[1], target[2]);
262            }
263            if let Some(fov) = cam.fov {
264                self.camera.fov = fov;
265            }
266            if let Some(near) = cam.near {
267                self.camera.near = near;
268            }
269            if let Some(far) = cam.far {
270                self.camera.far = far;
271            }
272
273            // Derive orbit parameters from position/target so update_orbit() stays consistent
274            if cam.position.is_some() {
275                let dir = self.camera.position - self.camera.target;
276                let dist = dir.length();
277                if dist > 0.001 {
278                    self.camera.distance = dist;
279                    let n = dir * (1.0 / dist);
280                    self.camera.pitch = n.y.asin();
281                    self.camera.yaw = n.x.atan2(n.z);
282                }
283            }
284        }
285    }

287    /// The first `resumed`: the window, then the GPU device. A desktop
288    /// blocks on the device and finishes here. A browser can't block, so on
289    /// the web the device request runs as a promise and `finish_initialize`
290    /// follows when it lands (see `web_startup.rs`).
291    pub(super) fn initialize(&mut self, event_loop: &ActiveEventLoop) -> Result<()> {
292        let window = self.create_window(event_loop)?;
293        #[cfg(not(target_arch = "wasm32"))]
294        {
295            let render_context = pollster::block_on(RenderContext::new(window.clone()))
296                .context("Failed to initialize render context")?;
297            self.finish_initialize(window, render_context)
298        }
299        #[cfg(target_arch = "wasm32")]
300        {
301            self.request_render_context(window);
302            Ok(())
303        }
304    }

306    /// Everything after the device: renderer, models, input, egui, the
307    /// scene's environment and the game systems.
308    pub(super) fn finish_initialize(
309        &mut self,
310        window: Arc<Window>,
311        render_context: RenderContext,
312    ) -> Result<()> {
313        // Renderer + model/procgen loading (context and renderer stay locals
314        // until the environment pass is done — they are stored on self only
315        // once fully built, as before).
316        let (render_context, mut scene_renderer) = self.init_render_and_models(render_context)?;
317
318        // Input bindings, gamepad backend, egui overlay
319        self.init_input_and_egui(&window, &render_context);
320
321        // Project fonts (fonts/*.ttf|otf) and the real text measurer for scripts
322        self.refresh_project_fonts();
323        self.install_text_measurer();
324
325        // Splines, ambient/wrap/sheen, skybox, terrain, camera, post config
326        self.apply_scene_environment(&render_context, &mut scene_renderer);
327
328        self.render_context = Some(render_context);
329        self.scene_renderer = Some(scene_renderer);
330        self.update_vsync_pacing();
331        // Quality settings (FlintKart#55) before any script's on_init, so
332        // get_quality() and gpu_info() answer from the first frame.
333        self.init_quality();
334
335        // Saved persistent store (save/persist.toml) before any script's
336        // on_init runs, so persist_get sees last session's values.
337        self.load_persistent_store();
338
339        // Physics → audio → music session → animation → particles → scripts
340        // (audio before the scene-declared session, session before script
341        // init so the conducted context exists from the scripts' first
342        // frame — F3/F4 ordering).
343        self.init_game_systems()?;
344
345        // Capture cursor for first-person look (only if FPS player exists).
346        // On Android, always set cursor_captured = true so touch input flows
347        // without requiring a click-to-capture gate.
348        #[cfg(target_os = "android")]
349        {
350            self.cursor_captured = true;
351        }
352        #[cfg(not(target_os = "android"))]
353        if self.physics.has_player_entity() {
354            self.capture_cursor();
355        }
356
357        Ok(())
358    }

360    fn create_window(&mut self, event_loop: &ActiveEventLoop) -> Result<Arc<Window>> {
361        let window_attrs = Window::default_attributes().with_title("Flint Player");
362        // In a browser the page's canvas is the window, sized by its CSS.
363        #[cfg(target_arch = "wasm32")]
364        let window_attrs = super::web_startup::canvas_attributes(window_attrs);
365        #[cfg(not(any(target_os = "android", target_arch = "wasm32")))]
366        let window_attrs = window_attrs.with_inner_size(PhysicalSize::new(
367            self.window_size.0.max(1),
368            self.window_size.1.max(1),
369        ));
370
371        let window = Arc::new(
372            event_loop
373                .create_window(window_attrs)
374                .context("Failed to create player window")?,
375        );
376
377        if self.fullscreen {
378            window.set_fullscreen(Some(winit::window::Fullscreen::Borderless(None)));
379        }
380
381        self.window = Some(window.clone());
382
383        Ok(window)
384    }

386    fn init_render_and_models(
387        &mut self,
388        render_context: RenderContext,
389    ) -> Result<(RenderContext, SceneRenderer)> {
390        self.camera.aspect = render_context.aspect_ratio();
391        self.camera.fov = 70.0; // Slightly wider FOV for first-person
392
393        let mut scene_renderer = SceneRenderer::new(
394            &render_context,
395            flint_render::RendererConfig {
396                sample_count: self.msaa_sample_count,
397                ..Default::default()
398            },
399        );
400        // FLINT_SHADOWS / FLINT_SHADOW_RES and the rest of the quality
401        // settings are applied by init_quality once the renderer is stored.
402
403        // Rebuild component index as a safety net after scene loading
404        self.world.rebuild_component_index();
405
406        // Set scene_dir for font/texture resolution
407        scene_renderer.scene_dir = Path::new(&self.scene_path)
408            .parent()
409            .map(|p| p.to_path_buf());
410
411        // Load models from world (including skeletal data)
412        let config = build_model_load_config(
413            &self.scene_path,
414            &self.world,
415            self.catalog.as_ref(),
416            self.content_store.as_ref(),
417        );
418
419        // The procgen disk tier (#181), set up once, before the first scene's
420        // assets are resolved.
421        if self.procgen_resolver.disk_stats().is_none() {
422            let scene_dir = Path::new(&self.scene_path)
423                .parent()
424                .unwrap_or(Path::new("."));
425            if let Some(disk) = self
426                .procgen_disk_cache
427                .then(|| procgen_disk_config(scene_dir))
428                .flatten()
429            {
430                println!("[procgen] disk cache: {}", disk.dir.display());
431                self.procgen_resolver = flint_procgen::ProcGenResolver::with_config(
432                    flint_procgen::ProcGenCacheConfig {
433                        disk: Some(disk),
434                        ..Default::default()
435                    },
436                );
437            }
438        }
439
440        // Procgen before the model loader: catalog > procgen > file (#32).
441        super::procgen_assets::resolve_scene(
442            &self.scene_path,
443            &self.world,
444            &mut self.procgen_resolver,
445            self.procgen_mode,
446            super::procgen_assets::GpuSink {
447                renderer: &mut scene_renderer,
448                device: &render_context.device,
449                queue: &render_context.queue,
450                config: &config,
451            },
452        );
453
454        let load_result = model_loader::load_models_from_world(
455            &mut self.world,
456            &mut scene_renderer,
457            &render_context.device,
458            &render_context.queue,
459            &config,
460        );
461        register_skeletal_data(&load_result, &mut self.animation);
462        register_node_animation_data(&load_result, &mut self.animation);
463        self.skeletal_entity_assets = load_result.skinned_entities;
464        scene_renderer.update_from_world(&self.world, &render_context.device);
465
466        Ok((render_context, scene_renderer))
467    }

469    fn init_input_and_egui(&mut self, window: &Arc<Window>, render_context: &RenderContext) {
470        // Load input configs with deterministic layering.
471        self.configure_input_bindings()
472            .unwrap_or_else(|e| tracing::warn!("Input config load error: {e:#}"));
473
474        // Initialize gamepad backend (best-effort).
475        self.gilrs = Gilrs::new().ok();
476
477        // Initialize egui for HUD overlay. egui assumes textures of at most
478        // 2048 px a side until told otherwise, and panics on a wider one
479        // (Float's 2713 px title). Give it the device's limit, both for the
480        // passes egui-winit feeds it and now, since UI sprites load before
481        // the first pass.
482        let max_texture_side = render_context.device.limits().max_texture_dimension_2d as usize;
483        self.egui_ctx
484            .input_mut(|i| i.max_texture_side = max_texture_side);
485        let egui_winit = egui_winit::State::new(
486            self.egui_ctx.clone(),
487            egui::ViewportId::ROOT,
488            window,
489            Some(window.scale_factor() as f32),
490            None,
491            Some(max_texture_side),
492        );
493        let egui_renderer = egui_wgpu::Renderer::new(
494            &render_context.device,
495            render_context.hud_format,
496            None,
497            1,
498            false,
499        );
500        self.egui_winit = Some(egui_winit);
501        self.egui_renderer = Some(egui_renderer);
502    }

504    fn apply_scene_environment(
505        &mut self,
506        render_context: &RenderContext,
507        scene_renderer: &mut SceneRenderer,
508    ) {
509        // Generate procedural geometry from spline + spline_mesh entities
510        crate::spline_gen::load_splines(
511            &self.scene_path,
512            &mut self.world,
513            scene_renderer,
514            Some(&mut self.physics),
515            &render_context.device,
516            &render_context.queue,
517        );
518
519        // Refresh renderer with any new procedural meshes
520        scene_renderer.update_from_world(&self.world, &render_context.device);
521
522        // Scene-authored hemisphere ambient + diffuse wrap (absent = renderer default)
523        if let Some((sky, ground)) = self.scene_ambient {
524            scene_renderer.set_ambient(sky, ground);
525        }
526        if let Some(wrap) = self.scene_diffuse_wrap {
527            scene_renderer.set_diffuse_wrap(wrap);
528        }
529        if let Some(oren) = self.scene_oren_nayar {
530            scene_renderer.set_oren_nayar(oren);
531        }
532        if let Some((color, strength)) = self.scene_sheen {
533            scene_renderer.set_sheen(color, strength);
534        }
535
536        // Load skybox if configured
537        if let Some(skybox_rel) = &self.skybox_path {
538            let scene_dir = Path::new(&self.scene_path)
539                .parent()
540                .unwrap_or_else(|| Path::new("."));
541
542            // Search scene dir first, then parent (game root)
543            let skybox_path = {
544                let p = scene_dir.join(skybox_rel);
545                if vfs::exists(&p) {
546                    p
547                } else if let Some(parent) = scene_dir.parent() {
548                    parent.join(skybox_rel)
549                } else {
550                    p
551                }
552            };
553
554            if vfs::exists(&skybox_path) {
555                scene_renderer.load_skybox(
556                    &render_context.device,
557                    &render_context.queue,
558                    &skybox_path,
559                );
560            } else {
561                tracing::warn!("Skybox file not found: {}", skybox_path.display());
562            }
563        }
564
565        // Load terrain from world
566        #[cfg(feature = "debug-hud")]
567        self.debug_panels.clear();
568        self.load_terrain_from_world(
569            &render_context.device,
570            &render_context.queue,
571            scene_renderer,
572        );
573
574        // Set terrain height callback for scripts
575        self.update_terrain_height_fn();
576
577        // Apply scene-level camera configuration
578        self.apply_camera_def();
579        self.apply_camera_tuning();
580
581        // Apply scene-level post-processing config
582        if let Some(pp_def) = &self.scene_post_process {
583            scene_renderer
584                .set_post_process_config(scene_loading::post_process_config_from_def(pp_def));
585            scene_renderer.ensure_style_resources(&render_context.device);
586            scene_renderer.ensure_fxaa_resources(&render_context.device);
587        }
588    }

590    fn init_game_systems(&mut self) -> Result<()> {
591        // Initialize physics
592        self.physics
593            .initialize(&mut self.world)
594            .context("Failed to initialize physics")?;
595
596        // Initialize audio
597        load_audio_from_world(
598            &self.world,
599            &mut self.audio,
600            &self.scene_path,
601            self.scene_preload_audio,
602            // In the browser the audio may already be here at start, and
603            // decoding it all in one frame is the stall #124 removed:
604            // queue it. The desktop decodes up front, as a load can afford.
605            cfg!(target_arch = "wasm32"),
606        );
607        self.audio
608            .initialize(&mut self.world)
609            .unwrap_or_else(|e| tracing::warn!("Audio init failed: {:?}", e));
610
611        // Scene-declared music session (F3): started before script init so a
612        // conducted context exists from the scripts' first frame (F4).
613        self.start_music_session();
614
615        // Initialize animation
616        load_animations_from_world(&self.scene_path, &mut self.animation);
617        load_sprite_animations_from_world(&self.scene_path, &mut self.animation);
618        load_sequences_from_world(&self.scene_path, &mut self.animation);
619        self.animation
620            .initialize(&mut self.world)
621            .unwrap_or_else(|e| tracing::warn!("Animation init failed: {:?}", e));
622
623        // Initialize particles: register `particles/*.particles.toml` first so
624        // `particle_effect` components resolve on the first sync.
625        flint_particles::load_particle_effects_from_world(&self.scene_path, &mut self.particles);
626        self.particles
627            .initialize(&mut self.world)
628            .unwrap_or_else(|e| tracing::warn!("Particles init failed: {:?}", e));
629        self.load_particle_textures();
630
631        // Initialize scripting (state_scope required so on_init can access persist store)
632        load_scripts_from_world(&self.scene_path, &mut self.script);
633        self.script.set_current_scene(&self.scene_path);
634        {
635            let _state_scope = self.script.state_scope(
636                &mut self.state_machine,
637                &mut self.persistent_store,
638                &self.physics,
639            );
640            self.script
641                .initialize(&mut self.world)
642                .unwrap_or_else(|e| tracing::warn!("Script init failed: {:?}", e));
643        }
644
645        Ok(())
646    }

700    /// Update the terrain height callback on the script system.
701    /// Call after terrain loading or clearing.
702    pub(super) fn update_terrain_height_fn(&mut self) {
703        if let Some((ref terrain, ref config)) = self.terrain {
704            let heights = terrain.heightmap.clone_heights();
705            let hm_w = terrain.heightmap.width;
706            let hm_d = terrain.heightmap.depth;
707            let world_w = config.width;
708            let world_d = config.depth;
709            let height_scale = config.height_scale;
710            self.script
711                .set_terrain_height_fn(Some(Box::new(move |x: f32, z: f32| {
712                    // Convert world coords to normalized UV
713                    let u = x / world_w;
714                    let v = z / world_d;
715                    // Bilinear sample from heights array
716                    let fx = u * (hm_w as f32 - 1.0);
717                    let fz = v * (hm_d as f32 - 1.0);
718                    let ix = (fx as u32).min(hm_w - 2);
719                    let iz = (fz as u32).min(hm_d - 2);
720                    let tx = fx - ix as f32;
721                    let tz = fz - iz as f32;
722                    let idx = |col: u32, row: u32| -> f32 { heights[(row * hm_w + col) as usize] };
723                    let h00 = idx(ix, iz);
724                    let h10 = idx(ix + 1, iz);
725                    let h01 = idx(ix, iz + 1);
726                    let h11 = idx(ix + 1, iz + 1);
727                    let h = h00 * (1.0 - tx) * (1.0 - tz)
728                        + h10 * tx * (1.0 - tz)
729                        + h01 * (1.0 - tx) * tz
730                        + h11 * tx * tz;
731                    h * height_scale
732                })));
733        } else {
734            self.script.set_terrain_height_fn(None);
735        }
736    }

738    pub(super) fn configure_input_bindings(&mut self) -> Result<()> {
739        self.input
740            .load_bindings(InputConfig::built_in_defaults())
741            .context("failed to load built-in input defaults")?;
742
743        let mut paths = resolve_input_paths(
744            Path::new(&self.scene_path),
745            self.scene_input_config.as_deref(),
746            self.input_config_override.as_deref(),
747        );
748
749        if let Some(path) = &paths.game_default {
750            if vfs::exists(path) {
751                let cfg = InputConfig::load_from_file(path).with_context(|| {
752                    format!("failed to load game input config '{}'", path.display())
753                })?;
754                self.input
755                    .merge_bindings(cfg)
756                    .context("failed to merge game input config")?;
757            }
758        }
759
760        // Now the game default has set the game id, find the player's own
761        // rebinds: they're saved under it.
762        paths.user_override =
763            resolve_user_override(Path::new(&self.scene_path), &self.input.config().game_id);
764        // The player's copy (crate::user_files, localStorage on the web), else
765        // one the game ships in its vfs bundle.
766        if let Some(path) = &paths.user_override {
767            let cfg = read_user_override_file(path).with_context(|| {
768                format!("failed to load user input config '{}'", path.display())
769            })?;
770            if let Some(cfg) = cfg {
771                self.user_override_config = cfg.clone();
772                self.input
773                    .merge_bindings(cfg)
774                    .context("failed to merge user input config")?;
775            }
776        }
777
778        if let Some(path) = &paths.cli_override {
779            if vfs::exists(path) {
780                let cfg = InputConfig::load_from_file(path).with_context(|| {
781                    format!("failed to load CLI input config '{}'", path.display())
782                })?;
783                self.input
784                    .merge_bindings(cfg)
785                    .context("failed to merge CLI input config")?;
786            }
787        }
788
789        if self.user_override_config.version == 0 {
790            self.user_override_config.version = 1;
791        }
792        if self.user_override_config.game_id.trim().is_empty() {
793            self.user_override_config.game_id = self.input.config().game_id.clone();
794        }
795
796        self.input_config_paths = Some(paths);
797        Ok(())
798    }

817    /// (Re)load `<project>/fonts/` into egui when the project root changed
818    /// since the last scan (initial load and scene transitions). Extensions'
819    /// `fonts` hooks edit the definitions before they're installed.
820    pub(super) fn refresh_project_fonts(&mut self) {
821        let root = super::fonts::find_fonts_dir(&self.scene_path);
822        if self.ui_fonts_installed && self.ui_fonts_root == root {
823            return; // same project root as the last scan
824        }
825        let mut defs = super::fonts::project_font_definitions(&self.scene_path);
826        for ext in self.extensions.iter_mut() {
827            ext.fonts(&mut defs);
828        }
829        self.ui_fonts = super::fonts::script_families(&defs);
830        self.egui_ctx.set_fonts(defs);
831        self.ui_fonts_warned.clear();
832        self.ui_fonts_root = root;
833        self.ui_fonts_installed = true;
834    }

836    /// Give scripts a `measure_text` that lays text out with the same font
837    /// stack the HUD draws with. Before egui's first pass (fonts not built
838    /// yet) and for families not installed in the *active* definitions,
839    /// the closure degrades gracefully instead of panicking.
840    pub(super) fn install_text_measurer(&mut self) {
841        let ctx = self.egui_ctx.clone();
842        let ready = self.egui_fonts_ready.clone();
843        self.script.set_text_measurer(Some(Arc::new(
844            move |text: &str, size: f32, font: Option<&str>, spacing: f32| {
845                if !ready.load(std::sync::atomic::Ordering::Relaxed) {
846                    let chars = text.chars().count() as f32;
847                    return (chars * size * 0.6 + spacing * (chars - 1.0).max(0.0), size);
848                }
849                ctx.fonts(|f| {
850                    let family = font
851                        .map(|n| egui::FontFamily::Name(n.into()))
852                        .filter(|fam| f.lock().fonts.definitions().families.contains_key(fam))
853                        .unwrap_or(egui::FontFamily::Proportional);
854                    let job = super::fonts::text_layout_job(
855                        text,
856                        egui::FontId::new(size, family),
857                        egui::Color32::WHITE,
858                        spacing,
859                    );
860                    let galley = f.layout_job(job);
861                    (galley.size().x, galley.size().y)
862                })
863            },
864        )));
865    }

867    /// Load a sprite texture for UI rendering. Called lazily when a draw_sprite
868    /// command references a name not yet in ui_textures.
869    pub fn load_ui_texture(&mut self, name: &str) -> bool {
870        if self.ui_textures.contains_key(name) {
871            return true;
872        }
873
874        let scene_dir = Path::new(&self.scene_path)
875            .parent()
876            .unwrap_or_else(|| Path::new("."));
877
878        // Search: scene_dir/sprites/{name} → game_root/sprites/{name} → scene_dir/{name}
879        let candidates = [
880            scene_dir.join("sprites").join(name),
881            scene_dir
882                .parent()
883                .map(|p| p.join("sprites").join(name))
884                .unwrap_or_default(),
885            scene_dir.join(name),
886        ];
887
888        for path in &candidates {
889            if vfs::exists(path) {
890                if let Ok(img) = flint_render::open_image(path) {
891                    let rgba = img.to_rgba8();
892                    let (w, h) = rgba.dimensions();
893                    let color_image =
894                        egui::ColorImage::from_rgba_unmultiplied([w as usize, h as usize], &rgba);
895                    let tex_handle =
896                        self.egui_ctx
897                            .load_texture(name, color_image, egui::TextureOptions::LINEAR);
898                    self.ui_textures.insert(name.to_string(), tex_handle);
899                    println!("Loaded UI sprite: {}", name);
900                    return true;
901                }
902            }
903        }
904
905        tracing::warn!("UI sprite not found: {}", name);
906        false
907    }

909    /// Pre-scan draw commands and load any sprite textures that haven't been loaded yet
910    pub(super) fn load_pending_sprites(&mut self) {
911        let sprite_names: Vec<String> = self
912            .draw_commands
913            .iter()
914            .filter_map(|item| {
915                if let DrawCommand::Sprite { name, .. } = &item.cmd {
916                    if !self.ui_textures.contains_key(name.as_str()) {
917                        Some(name.clone())
918                    } else {
919                        None
920                    }
921                } else {
922                    None
923                }
924            })
925            .collect();
926
927        for name in sprite_names {
928            self.load_ui_texture(&name);
929        }
930    }

933/// Where the procgen disk tier goes for a scene in `scene_dir` (#181):
934/// `FLINT_PROCGEN_CACHE=<dir>` if set (`0`, `off` or empty: no disk tier),
935/// else `.flint/procgen-cache/` in the project, the scene folder's parent
936/// when that holds `schemas/` or `specs/` (a `levels/` layout, as in
937/// `showcase_game`), otherwise the scene folder itself. Until projects have
938/// a root marker and a config file (#179), that guess is the project.
939fn procgen_disk_config(scene_dir: &Path) -> Option<flint_procgen::DiskCacheConfig> {
940    if let Ok(value) = std::env::var("FLINT_PROCGEN_CACHE") {
941        return match value.trim() {
942            "" | "0" | "off" | "false" => None,
943            dir => Some(flint_procgen::DiskCacheConfig::new(dir)),
944        };
945    }
946    let project = scene_dir
947        .parent()
948        .filter(|p| vfs::is_dir(p.join("schemas")) || vfs::is_dir(p.join("specs")))
949        .unwrap_or(scene_dir);
950    Some(flint_procgen::DiskCacheConfig::in_project(project))
