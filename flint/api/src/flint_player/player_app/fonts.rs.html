1//! Project fonts for the script HUD.
2//!
3//! Scans `<project>/fonts/*.ttf|*.otf` (non-recursive; the project root is
4//! the scene file's parent directory, or its parent — the same rule
5//! `load_ui_texture` uses for `sprites/`) and registers every file with egui
6//! under its file stem as a named `FontFamily`. An optional
7//! `<project>/fonts/fonts.toml` manifest adds aliases:
8//!
9//! ```toml
10//! [[font]]
11//! name = "display"                     # alias → FontFamily::Name("display")
12//! file = "BarlowCondensed-BlackItalic.ttf"
13//!
14//! [egui]                               # optional (ADR 0081)
15//! proportional = "body"                # egui's default text face
16//! monospace = "code"                   # egui's monospace face
17//! ```
18//!
19//! The `[egui]` names are families (a file stem or an alias). The named face
20//! goes first in egui's `Proportional` or `Monospace` family, ahead of egui's
21//! own fonts, so a game's egui UI draws in it and keeps egui's fonts for the
22//! glyphs it lacks. Every named family keeps egui's proportional fonts as
23//! fallbacks the same way.
24//!
25//! An extension can add fonts of its own with `PlayerExtension::fonts`,
26//! which edits the definitions before Flint installs them, so they survive
27//! every reload.
28//!
29//! Both the renderer (`hud_render`) and the script `measure_text` closure
30//! build their layout jobs through [`text_layout_job`] so measurement and
31//! drawing can never disagree.
32

38/// One manifest alias entry.
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub struct FontAlias {
41    pub name: String,
42    pub file: String,
43}
44
45/// Parse a `fonts.toml` manifest. Entries missing `name` or `file` are
46/// dropped with a warning; a malformed document yields an empty list.
47pub fn parse_manifest(text: &str) -> Vec<FontAlias> {
48    let doc: toml::Value = match text.parse() {
49        Ok(v) => v,
50        Err(e) => {
51            tracing::warn!("fonts.toml: parse error: {e}");
52            return Vec::new();
53        }
54    };
55    let Some(entries) = doc.get("font").and_then(|v| v.as_array()) else {
56        return Vec::new();
57    };
58    entries
59        .iter()
60        .filter_map(|e| {
61            let name = e.get("name").and_then(|v| v.as_str());
62            let file = e.get("file").and_then(|v| v.as_str());
63            match (name, file) {
64                (Some(n), Some(f)) if !n.is_empty() && !f.is_empty() => Some(FontAlias {
65                    name: n.to_string(),
66                    file: f.to_string(),
67                }),
68                _ => {
69                    tracing::warn!("fonts.toml: [[font]] entry needs both `name` and `file`: {e}");
70                    None
71                }
72            }
73        })
74        .collect()
75}

77/// The `[egui]` table of a `fonts.toml`: which families egui's default
78/// families start with. `None` keeps egui's own face.
79#[derive(Debug, Clone, Default, PartialEq, Eq)]
80pub struct EguiDefaults {
81    pub proportional: Option<String>,
82    pub monospace: Option<String>,
83}
84
85/// Parse the `[egui]` table of a `fonts.toml` manifest. A missing table, a
86/// malformed document or a non-string value yields `None` for that slot
87/// (`parse_manifest` already warns about a parse error).
88pub fn parse_egui_defaults(text: &str) -> EguiDefaults {
89    let Ok(doc) = text.parse::<toml::Value>() else {
90        return EguiDefaults::default();
91    };
92    let Some(table) = doc.get("egui") else {
93        return EguiDefaults::default();
94    };
95    let slot = |key: &str| match table.get(key) {
96        Some(toml::Value::String(s)) if !s.is_empty() => Some(s.clone()),
97        Some(other) => {
98            tracing::warn!("fonts.toml: [egui] {key} should be a family name, not {other}");
99            None
100        }
101        None => None,
102    };
103    EguiDefaults {
104        proportional: slot("proportional"),
105        monospace: slot("monospace"),
106    }
107}

109/// Resolve the family table for a set of font files plus manifest aliases.
110/// Keys are family names (file stems first, then aliases); values are the
111/// `font_data` key each family points at (always a file stem). Aliases whose
112/// file is not among `stems` are dropped with a warning; an alias may not
113/// shadow a real stem.
114pub fn resolve_families(stems: &[String], aliases: &[FontAlias]) -> BTreeMap<String, String> {
115    let mut families: BTreeMap<String, String> =
116        stems.iter().map(|s| (s.clone(), s.clone())).collect();
117    for alias in aliases {
118        let target = Path::new(&alias.file)
119            .file_stem()
120            .map(|s| s.to_string_lossy().into_owned())
121            .unwrap_or_default();
122        if !stems.contains(&target) {
123            tracing::warn!(
124                "fonts.toml: alias `{}` points at `{}`, which is not in fonts/",
125                alias.name,
126                alias.file
127            );
128            continue;
129        }
130        if stems.contains(&alias.name) {
131            tracing::warn!(
132                "fonts.toml: alias `{}` shadows a font file of the same name; ignored",
133                alias.name
134            );
135            continue;
136        }
137        families.insert(alias.name.clone(), target);
138    }
139    families
140}

142/// The project root for a scene path: `scene_dir/fonts` if it exists,
143/// else `scene_dir/../fonts`. Returns the `fonts/` directory candidate that
144/// exists, or `None`.
145pub fn find_fonts_dir(scene_path: &str) -> Option<PathBuf> {
146    let scene_dir = Path::new(scene_path)
147        .parent()
148        .unwrap_or_else(|| Path::new("."));
149    let mut candidates = vec![scene_dir.join("fonts")];
150    if let Some(parent) = scene_dir.parent() {
151        candidates.push(parent.join("fonts"));
152    }
153    candidates.into_iter().find(|p| vfs::is_dir(p))
154}

156/// Scan a `fonts/` directory: read every `.ttf`/`.otf` (non-recursive) and
157/// the optional manifest. Returns `(stem → bytes, aliases, egui defaults)`.
158fn scan_fonts_dir(dir: &Path) -> (BTreeMap<String, Vec<u8>>, Vec<FontAlias>, EguiDefaults) {
159    let mut data = BTreeMap::new();
160    let entries = match vfs::read_dir(dir) {
161        Ok(e) => e,
162        Err(e) => {
163            tracing::warn!("fonts: cannot read {}: {e}", dir.display());
164            return (data, Vec::new(), EguiDefaults::default());
165        }
166    };
167    for path in entries {
168        if !vfs::is_file(&path) {
169            continue;
170        }
171        let ext = path
172            .extension()
173            .map(|e| e.to_string_lossy().to_ascii_lowercase())
174            .unwrap_or_default();
175        if ext != "ttf" && ext != "otf" {
176            continue;
177        }
178        let Some(stem) = path.file_stem().map(|s| s.to_string_lossy().into_owned()) else {
179            continue;
180        };
181        match vfs::read(&path) {
182            Ok(bytes) => {
183                data.insert(stem, bytes);
184            }
185            Err(e) => tracing::warn!("fonts: failed to read {}: {e}", path.display()),
186        }
187    }
188    let manifest = dir.join("fonts.toml");
189    match vfs::read_to_string(&manifest) {
190        Ok(text) => (data, parse_manifest(&text), parse_egui_defaults(&text)),
191        Err(_) => (data, Vec::new(), EguiDefaults::default()),
192    }
193}

195/// egui's definitions with a project's fonts added: each file under its
196/// stem, a named family per stem and alias, and `defaults` first in egui's
197/// own families. Every named family falls back to egui's proportional fonts.
198/// `families` is `resolve_families`' table.
199pub fn build_definitions(
200    data: BTreeMap<String, Vec<u8>>,
201    families: &BTreeMap<String, String>,
202    defaults: &EguiDefaults,
203) -> egui::FontDefinitions {
204    let mut defs = egui::FontDefinitions::default();
205    let fallbacks = defs
206        .families
207        .get(&egui::FontFamily::Proportional)
208        .cloned()
209        .unwrap_or_default();
210    for (stem, bytes) in data {
211        defs.font_data
212            .insert(stem, Arc::new(egui::FontData::from_owned(bytes)));
213    }
214    for (family, target) in families {
215        let mut list = vec![target.clone()];
216        list.extend(fallbacks.iter().cloned());
217        defs.families
218            .insert(egui::FontFamily::Name(family.as_str().into()), list);
219    }
220    for (slot, key, name) in [
221        (
222            egui::FontFamily::Proportional,
223            "proportional",
224            &defaults.proportional,
225        ),
226        (
227            egui::FontFamily::Monospace,
228            "monospace",
229            &defaults.monospace,
230        ),
231    ] {
232        let Some(name) = name else { continue };
233        let Some(target) = families.get(name) else {
234            tracing::warn!("fonts.toml: [egui] {key} = `{name}` is not a font in fonts/");
235            continue;
236        };
237        let list = defs.families.entry(slot).or_default();
238        list.retain(|f| f != target);
239        list.insert(0, target.clone());
240    }
241    defs
242}

244/// The project's font definitions for a scene path (see `build_definitions`).
245/// With no `fonts/` directory these are egui's defaults, so a scene change
246/// to a font-less project drops stale families.
247pub fn project_font_definitions(scene_path: &str) -> egui::FontDefinitions {
248    let Some(dir) = find_fonts_dir(scene_path) else {
249        return egui::FontDefinitions::default();
250    };
251    let (data, aliases, defaults) = scan_fonts_dir(&dir);
252    let stems: Vec<String> = data.keys().cloned().collect();
253    let families = resolve_families(&stems, &aliases);
254    if families.is_empty() {
255        tracing::info!("fonts: {} contains no .ttf/.otf files", dir.display());
256    } else {
257        tracing::info!(
258            "fonts: loaded {} famil{} from {}: {}",
259            families.len(),
260            if families.len() == 1 { "y" } else { "ies" },
261            dir.display(),
262            families
263                .keys()
264                .map(|s| s.as_str())
265                .collect::<Vec<_>>()
266                .join(", ")
267        );
268    }
269    build_definitions(data, &families, &defaults)
270}

272/// The family names scripts may refer to: every named family in `defs`,
273/// the project's and any an extension added.
274pub fn script_families(defs: &egui::FontDefinitions) -> HashSet<String> {
275    defs.families
276        .keys()
277        .filter_map(|f| match f {
278            egui::FontFamily::Name(name) => Some(name.to_string()),
279            _ => None,
280        })
281        .collect()
282}

284/// Resolve a script-facing family name to an egui `FontId`, falling back to
285/// the default proportional font (and warning once per unknown name).
286pub fn resolve_font_id(
287    size: f32,
288    font: Option<&str>,
289    known: &HashSet<String>,
290    warned: &mut HashSet<String>,
291) -> egui::FontId {
292    match font {
293        Some(name) if known.contains(name) => {
294            egui::FontId::new(size, egui::FontFamily::Name(name.into()))
295        }
296        Some(name) => {
297            if warned.insert(name.to_string()) {
298                tracing::warn!(
299                    "font `{name}` not found under fonts/ (known: {:?}); using default",
300                    known
301                );
302            }
303            egui::FontId::proportional(size)
304        }
305        None => egui::FontId::proportional(size),
306    }
307}

309/// The one layout-job builder shared by the HUD renderer and `measure_text`:
310/// single line, no wrapping, optional extra letter spacing.
311pub fn text_layout_job(
312    text: &str,
313    font_id: egui::FontId,
314    color: egui::Color32,
315    letter_spacing: f32,
316) -> egui::text::LayoutJob {
317    let format = egui::TextFormat {
318        font_id,
319        color,
320        extra_letter_spacing: letter_spacing,
321        ..Default::default()
322    };
323    let mut job = egui::text::LayoutJob::single_section(text.to_owned(), format);
324    job.wrap = egui::text::TextWrapping::no_max_width();
325    job.break_on_newline = false;
326    job
327}

426    /// The named face leads egui's family and egui's fonts stay behind it;
427    /// named families fall back to egui's proportional fonts; an unknown
428    /// default leaves egui's family alone.
429    #[test]
430    fn definitions_put_the_named_defaults_first() {
431        let egui_defaults = egui::FontDefinitions::default();
432        let proportional = &egui_defaults.families[&egui::FontFamily::Proportional];
433        let monospace = &egui_defaults.families[&egui::FontFamily::Monospace];
434
435        let data: BTreeMap<String, Vec<u8>> = [
436            ("Body".to_string(), vec![0u8]),
437            ("Title".to_string(), vec![1u8]),
438        ]
439        .into();
440        let stems: Vec<String> = data.keys().cloned().collect();
441        let families = resolve_families(
442            &stems,
443            &[FontAlias {
444                name: "body".into(),
445                file: "Body.ttf".into(),
446            }],
447        );
448        let defs = build_definitions(
449            data,
450            &families,
451            &EguiDefaults {
452                proportional: Some("body".into()),
453                monospace: Some("ghost".into()),
454            },
455        );
456
457        let p = &defs.families[&egui::FontFamily::Proportional];
458        assert_eq!(p[0], "Body");
459        assert_eq!(&p[1..], &proportional[..]);
460        assert_eq!(&defs.families[&egui::FontFamily::Monospace], monospace);
461        let title = &defs.families[&egui::FontFamily::Name("Title".into())];
462        assert_eq!(title[0], "Title");
463        assert_eq!(&title[1..], &proportional[..]);
464        assert!(defs.font_data.contains_key("Body"));
465
466        let mut names: Vec<String> = script_families(&defs).into_iter().collect();
467        names.sort();
468        assert_eq!(names, ["Body", "Title", "body"]);
469    }
