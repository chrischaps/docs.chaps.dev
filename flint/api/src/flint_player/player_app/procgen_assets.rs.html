1//! Procgen in the player's asset resolution (#32): **catalog > procgen > file**.
2//!
3//! At scene load every `model.asset` and `material`/`sprite` `texture` name
4//! is resolved in that order. A catalog entry (`ModelLoadConfig::overrides`)
5//! wins outright. A name with a matching `*.procgen.toml` (`meta.name`) is
6//! generated through the [`ProcGenResolver`]'s cache and uploaded under that
7//! name, *before* the file loader runs, so the file loader finds it cached
8//! and leaves it alone. Anything else is the file loader's, as before.
9//!
10//! When procgen can't make an asset (an unregistered generator, a generator
11//! error, an output the renderer can't draw) it warns and falls through to
12//! the file. If there is no file either, a placeholder is uploaded under the
13//! name and stays: a magenta box for a model, a 1x1 magenta texel for a
14//! texture, so a missing asset is seen rather than silently absent.
15//!
16//! With [`ProcGenMode::Deferred`] nothing is generated at load: each asset
17//! gets its placeholder at once and is queued on the resolver (Task 6.3's
18//! frame-budgeted queue); [`apply_completed`] swaps the real asset in under
19//! the same name as it finishes, and [`apply_failures`] falls back to the
20//! file for the ones that fail.
21//!
22//! Everything here speaks to the renderer through [`AssetSink`], so the
23//! resolution order is tested without a GPU; [`GpuSink`] is the real one.
24

39/// When the player generates a scene's procgen assets.
40#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
41pub enum ProcGenMode {
42    /// Generate during scene load; the first frame shows the real assets.
43    #[default]
44    Immediate,
45    /// Show placeholders at once and generate on the frame-budgeted queue.
46    Deferred,
47}

49/// The magenta of "this asset is missing or still coming", in linear RGB.
50pub const PLACEHOLDER_RGBA: [f32; 4] = [0.85, 0.05, 0.7, 1.0];
51/// The 1x1 placeholder texel (sRGB bytes).
52pub const PLACEHOLDER_TEXEL: [u8; 4] = [255, 0, 255, 255];

54/// One asset name the scene asks for, with somewhere to stand for queue priority.
55#[derive(Debug, Clone, PartialEq)]
56pub struct AssetRequest {
57    pub name: String,
58    pub kind: AssetKind,
59    pub position: [f32; 3],
60}
61
62/// What became of one request.
63#[derive(Debug, Clone, PartialEq, Eq)]
64pub enum Outcome {
65    /// Generated now (a cache miss).
66    Generated,
67    /// Served from the resolver's cache, no generation.
68    CacheHit,
69    /// Placeholder shown; queued for generation.
70    Deferred,
71    /// Procgen couldn't make it (the reason); the file loader takes it.
72    FileFallback(String),
73    /// Procgen couldn't make it (the reason) and there is no file: placeholder for good.
74    Placeholder(String),
75}

77/// Where the resolved assets go. The player's is [`GpuSink`].
78pub trait AssetSink {
79    /// A mesh is already uploaded under `name` (a previous scene, say).
80    fn has_mesh(&self, name: &str) -> bool;
81    /// The file loader would find a file for `name`.
82    fn has_file(&self, name: &str, kind: AssetKind) -> bool;
83    /// Upload (or replace) the mesh `name`.
84    fn upload_mesh(&mut self, name: &str, mesh: &MeshData);
85    /// Upload (or replace) the texture `name`.
86    fn upload_image(&mut self, name: &str, image: &ImageData);
87    /// Upload (or replace) the placeholder for `name`.
88    fn upload_placeholder(&mut self, name: &str, kind: AssetKind);
89    /// Drop what is under `name` and load its file instead. Returns whether it loaded.
90    fn load_file(&mut self, name: &str, kind: AssetKind) -> bool;
91}

93/// The directories searched for `*.procgen.toml`, least specific first: the
94/// resolver lets later directories override earlier ones, so a spec beside
95/// the scene beats a game-wide one of the same name.
96pub fn spec_dirs(scene_path: &str) -> Vec<PathBuf> {
97    let scene_dir = Path::new(scene_path).parent().unwrap_or(Path::new("."));
98    let mut dirs = Vec::new();
99    if let Some(parent) = scene_dir.parent() {
100        dirs.push(parent.join("specs"));
101    }
102    dirs.push(scene_dir.join("specs"));
103    dirs.push(scene_dir.join("models"));
104    dirs.into_iter().filter(|d| vfs::is_dir(d)).collect()
105}
106
107/// Index the scene's specs into `resolver`.
108pub fn discover_specs(resolver: &mut ProcGenResolver, scene_path: &str) {
109    let dirs = spec_dirs(scene_path);
110    let refs: Vec<&Path> = dirs.iter().map(PathBuf::as_path).collect();
111    resolver.discover_and_index(&refs);
112}

114/// The names in `world` that procgen should provide: those with a spec, not
115/// in the catalog (`config.overrides`), and (for models) not already uploaded.
116/// Each (name, kind) once, at the first entity that asks.
117pub fn collect_requests(
118    world: &FlintWorld,
119    config: &ModelLoadConfig,
120    resolver: &ProcGenResolver,
121    sink: &dyn AssetSink,
122) -> Vec<AssetRequest> {
123    let mut seen: HashSet<(String, bool)> = HashSet::new();
124    let mut requests = Vec::new();
125    let mut want = |name: &str, kind: AssetKind, position: [f32; 3]| {
126        if name.is_empty() || config.overrides.contains_key(name) || !resolver.has_spec(name) {
127            return; // catalog wins; no spec is the file loader's
128        }
129        if kind == AssetKind::Model && sink.has_mesh(name) {
130            return;
131        }
132        if seen.insert((name.to_string(), kind == AssetKind::Model)) {
133            requests.push(AssetRequest {
134                name: name.to_string(),
135                kind,
136                position,
137            });
138        }
139    };
140
141    for entity in world.all_entities() {
142        let Some(comps) = world.get_components(entity.id) else {
143            continue;
144        };
145        let position = comps
146            .get(comp::TRANSFORM)
147            .and_then(|t| t.get("position").and_then(toml_vec3))
148            .unwrap_or([0.0; 3]);
149        if let Some(model) = comps.get(comp::MODEL) {
150            if !model_loader::is_runtime_model(model) {
151                if let Some(name) = model.get("asset").and_then(|v| v.as_str()) {
152                    want(name, AssetKind::Model, position);
153                }
154            }
155        }
156        for comp_name in [comp::MATERIAL, comp::SPRITE] {
157            if let Some(tex) = comps
158                .get(comp_name)
159                .and_then(|c| c.get("texture"))
160                .and_then(|v| v.as_str())
161            {
162                want(tex, AssetKind::Texture, position);
163            }
164        }
165    }
166    requests
167}

169/// Resolve each request through procgen, falling through to the file and
170/// then to a placeholder. Returns each request's outcome, in order.
171pub fn resolve(
172    requests: &[AssetRequest],
173    resolver: &mut ProcGenResolver,
174    sink: &mut dyn AssetSink,
175    mode: ProcGenMode,
176) -> Vec<(AssetRequest, Outcome)> {
177    let mut outcomes = Vec::with_capacity(requests.len());
178    for req in requests {
179        let outcome = resolve_one(req, resolver, sink, mode);
180        log_outcome(req, &outcome);
181        outcomes.push((req.clone(), outcome));
182    }
183    outcomes
184}
185
186fn resolve_one(
187    req: &AssetRequest,
188    resolver: &mut ProcGenResolver,
189    sink: &mut dyn AssetSink,
190    mode: ProcGenMode,
191) -> Outcome {
192    // A spec naming a generator that isn't registered can never succeed:
193    // say so now rather than queue it.
194    if let Err(reason) = resolver.check_generator(&req.name) {
195        return fall_through(req, reason, sink);
196    }
197
198    let cached = resolver.is_cached(&req.name);
199    if mode == ProcGenMode::Deferred && !cached {
200        sink.upload_placeholder(&req.name, req.kind);
201        resolver.enqueue(&req.name, req.position, req.kind);
202        return Outcome::Deferred;
203    }
204
205    match resolver.generate(&req.name) {
206        Ok(output) => match upload_output(&req.name, req.kind, &output, sink) {
207            Ok(()) if cached => Outcome::CacheHit,
208            Ok(()) => Outcome::Generated,
209            Err(reason) => fall_through(req, reason, sink),
210        },
211        Err(e) => fall_through(req, e.to_string(), sink),
212    }
213}
214
215/// Procgen gave up on `req`: the file loader takes it if it can, otherwise a
216/// placeholder stays under its name.
217fn fall_through(req: &AssetRequest, reason: String, sink: &mut dyn AssetSink) -> Outcome {
218    if sink.has_file(&req.name, req.kind) {
219        Outcome::FileFallback(reason)
220    } else {
221        sink.upload_placeholder(&req.name, req.kind);
222        Outcome::Placeholder(reason)
223    }
224}
225
226fn log_outcome(req: &AssetRequest, outcome: &Outcome) {
227    let what = match req.kind {
228        AssetKind::Model => "model",
229        AssetKind::Texture => "texture",
230    };
231    match outcome {
232        Outcome::Generated => tracing::info!("Procgen generated {what} '{}'", req.name),
233        Outcome::CacheHit => tracing::info!("Procgen {what} '{}' from cache", req.name),
234        Outcome::Deferred => tracing::info!("Procgen {what} '{}' queued (placeholder)", req.name),
235        Outcome::FileFallback(reason) => tracing::warn!(
236            "Procgen couldn't make {what} '{}' ({reason}); loading its file",
237            req.name
238        ),
239        Outcome::Placeholder(reason) => tracing::warn!(
240            "Procgen couldn't make {what} '{}' ({reason}) and there is no file; showing a placeholder",
241            req.name
242        ),
243    }
244}

246/// The texture name each map of an image set goes under: the first (the
247/// colour) under the spec's own name, the rest as `<name>_<map>`, which is
248/// how the material path finds companion maps.
249pub fn image_set_name(name: &str, index: usize, semantics: ChannelSemantics) -> String {
250    if index == 0 {
251        return name.to_string();
252    }
253    let suffix = match semantics {
254        ChannelSemantics::Color => "color",
255        ChannelSemantics::Normal => "normal",
256        ChannelSemantics::Roughness => "roughness",
257        ChannelSemantics::Metallic => "metallic",
258        ChannelSemantics::Height => "height",
259        ChannelSemantics::Mask => "mask",
260    };
261    format!("{name}_{suffix}")
262}

264/// Upload one generator output under `name`, or say why it can't be drawn.
265pub fn upload_output(
266    name: &str,
267    kind: AssetKind,
268    output: &GeneratorOutput,
269    sink: &mut dyn AssetSink,
270) -> Result<(), String> {
271    match (output, kind) {
272        (GeneratorOutput::Mesh(mesh), AssetKind::Model) => sink.upload_mesh(name, mesh),
273        (GeneratorOutput::MeshWithLods(lods), AssetKind::Model) => match lods.first() {
274            Some(mesh) => sink.upload_mesh(name, mesh),
275            None => return Err("it produced no LODs".into()),
276        },
277        (GeneratorOutput::Image(image), _) => sink.upload_image(name, image),
278        (GeneratorOutput::ImageSet(images), _) if !images.is_empty() => {
279            for (i, image) in images.iter().enumerate() {
280                sink.upload_image(&image_set_name(name, i, image.channel_semantics), image);
281            }
282        }
283        (other, kind) => {
284            return Err(format!(
285                "a {} can't be drawn as a {}",
286                output_label(other),
287                match kind {
288                    AssetKind::Model => "model",
289                    AssetKind::Texture => "texture",
290                }
291            ))
292        }
293    }
294    Ok(())
295}
296
297fn output_label(output: &GeneratorOutput) -> &'static str {
298    match output {
299        GeneratorOutput::Mesh(_) => "mesh",
300        GeneratorOutput::MeshWithLods(_) => "LOD mesh",
301        GeneratorOutput::Image(_) => "image",
302        GeneratorOutput::ImageSet(_) => "image set",
303        GeneratorOutput::SkinnedMesh(_) => "skinned mesh",
304        GeneratorOutput::Sound(_) => "sound",
305        #[allow(unreachable_patterns)]
306        _ => "output",
307    }
308}

310/// Swap queued assets in over their placeholders as they finish.
311pub fn apply_completed(completed: &[CompletedAsset], sink: &mut dyn AssetSink) {
312    for asset in completed {
313        match upload_output(&asset.spec_name, asset.asset_kind, &asset.output, sink) {
314            Ok(()) => tracing::info!("Procgen swapped in '{}'", asset.spec_name),
315            Err(reason) => fail_queued(&asset.spec_name, asset.asset_kind, &reason, sink),
316        }
317    }
318}
319
320/// Queued assets that failed: load the file in place of the placeholder,
321/// or keep the placeholder when there is none.
322pub fn apply_failures(failures: &[FailedAsset], sink: &mut dyn AssetSink) {
323    for failed in failures {
324        fail_queued(&failed.spec_name, failed.asset_kind, &failed.error, sink);
325    }
326}
327
328fn fail_queued(name: &str, kind: AssetKind, reason: &str, sink: &mut dyn AssetSink) {
329    if sink.has_file(name, kind) && sink.load_file(name, kind) {
330        tracing::warn!("Procgen couldn't make '{name}' ({reason}); loaded its file");
331    } else {
332        tracing::warn!("Procgen couldn't make '{name}' ({reason}); keeping the placeholder");
333    }
334}

338/// [`AssetSink`] over the scene renderer.
339pub struct GpuSink<'a> {
340    pub renderer: &'a mut SceneRenderer,
341    pub device: &'a wgpu::Device,
342    pub queue: &'a wgpu::Queue,
343    pub config: &'a ModelLoadConfig,
344}
345
346impl GpuSink<'_> {
347    fn material(mat: &flint_procgen::MaterialData) -> flint_import::ImportedMaterial {
348        flint_import::ImportedMaterial {
349            name: mat.name.clone(),
350            base_color: mat.base_color,
351            metallic: mat.metallic,
352            roughness: mat.roughness,
353            use_vertex_color: true,
354            alpha_mode: flint_import::AlphaMode::Opaque,
355            alpha_cutoff: 0.5,
356            ..Default::default()
357        }
358    }
359}
360
361impl AssetSink for GpuSink<'_> {
362    fn has_mesh(&self, name: &str) -> bool {
363        self.renderer.mesh_cache().contains(name)
364    }
365
366    fn has_file(&self, name: &str, kind: AssetKind) -> bool {
367        match kind {
368            AssetKind::Model => model_loader::resolve_model_path(self.config, name).is_some(),
369            AssetKind::Texture => model_loader::resolve_texture_path(self.config, name).is_some(),
370        }
371    }
372
373    fn upload_mesh(&mut self, name: &str, mesh: &MeshData) {
374        let default_material = flint_procgen::MaterialData::default();
375        // One part per sub-mesh (one for the whole mesh when it has none),
376        // all under `name`: a tree's bark and leaves draw as one model.
377        let whole = [flint_procgen::SubMesh {
378            index_start: 0,
379            index_count: mesh.indices.len() as u32,
380            material_index: 0,
381        }];
382        let subs: &[flint_procgen::SubMesh] = if mesh.submeshes.is_empty() {
383            &whole
384        } else {
385            &mesh.submeshes
386        };
387
388        let mut parts = Vec::with_capacity(subs.len());
389        for (i, sub) in subs.iter().enumerate() {
390            let mat = mesh
391                .materials
392                .get(sub.material_index)
393                .unwrap_or(&default_material);
394            let start = sub.index_start as usize;
395            let end = (start + sub.index_count as usize).min(mesh.indices.len());
396            let indices = &mesh.indices[start.min(end)..end];
397            if indices.is_empty() {
398                continue;
399            }
400            // Compact the vertex range this part touches.
401            let lo = *indices.iter().min().unwrap() as usize;
402            let hi = *indices.iter().max().unwrap() as usize;
403            let vertices: Vec<flint_render::Vertex> = mesh.vertices[lo..=hi]
404                .iter()
405                .map(|v| flint_render::Vertex {
406                    position: v.position,
407                    normal: v.normal,
408                    color: mat.base_color,
409                    uv: v.uv,
410                })
411                .collect();
412            let indices: Vec<u32> = indices.iter().map(|&ix| ix - lo as u32).collect();
413
414            let mut material = Self::material(mat);
415            if let Some(normal) = &mat.normal_map {
416                let tex = format!("{name}_part{i}_normal");
417                self.upload_image(&tex, normal);
418                material.normal_texture = Some(tex);
419            }
420            parts.push((vertices, indices, material));
421        }
422        self.renderer
423            .load_procedural_mesh_parts(self.device, name, parts);
424    }
425
426    fn upload_image(&mut self, name: &str, image: &ImageData) {
427        let is_normal = matches!(image.channel_semantics, ChannelSemantics::Normal);
428        if let Err(e) = self.renderer.replace_texture_rgba(
429            self.device,
430            self.queue,
431            name,
432            image.width,
433            image.height,
434            &image.pixels,
435            is_normal,
436        ) {
437            tracing::warn!("Failed to upload procgen texture '{name}': {e}");
438        }
439    }
440
441    fn upload_placeholder(&mut self, name: &str, kind: AssetKind) {
442        match kind {
443            AssetKind::Model => {
444                // A unit box standing on the entity's origin, as a model's base does.
445                let mut box_mesh = flint_render::create_box_mesh(1.0, 1.0, 1.0, PLACEHOLDER_RGBA);
446                for v in &mut box_mesh.vertices {
447                    v.position[1] += 0.5;
448                }
449                let material = flint_import::ImportedMaterial {
450                    name: format!("{name} (placeholder)"),
451                    base_color: PLACEHOLDER_RGBA,
452                    roughness: 0.6,
453                    use_vertex_color: true,
454                    ..Default::default()
455                };
456                self.renderer.load_procedural_mesh_parts(
457                    self.device,
458                    name,
459                    vec![(box_mesh.vertices, box_mesh.indices, material)],
460                );
461            }
462            AssetKind::Texture => {
463                if let Err(e) = self.renderer.replace_texture_rgba(
464                    self.device,
465                    self.queue,
466                    name,
467                    1,
468                    1,
469                    &PLACEHOLDER_TEXEL,
470                    false,
471                ) {
472                    tracing::warn!("Failed to upload placeholder texture '{name}': {e}");
473                }
474            }
475        }
476    }
477
478    fn load_file(&mut self, name: &str, kind: AssetKind) -> bool {
479        match kind {
480            AssetKind::Model => {
481                let Some(path) = model_loader::resolve_model_path(self.config, name) else {
482                    return false;
483                };
484                match flint_import::import_gltf(&path) {
485                    Ok(import) => {
486                        self.renderer.remove_mesh(name);
487                        self.renderer.mesh_cache_mut().upload_imported(
488                            self.device,
489                            name,
490                            &import,
491                            [0.7, 0.7, 0.7, 1.0],
492                        );
493                        true
494                    }
495                    Err(e) => {
496                        tracing::warn!("Failed to load '{}': {e}", path.display());
497                        false
498                    }
499                }
500            }
501            AssetKind::Texture => {
502                let Some(path) = model_loader::resolve_texture_path(self.config, name) else {
503                    return false;
504                };
505                self.renderer.remove_texture(name);
506                self.renderer
507                    .load_texture_file(self.device, self.queue, name, &path)
508                    .is_ok()
509            }
510        }
511    }
512}

514/// Resolve the scene's procgen assets into the renderer: discover specs,
515/// then catalog > procgen > file > placeholder for each name. Runs before
516/// the model loader, which then skips every name this uploaded.
517pub fn resolve_scene(
518    scene_path: &str,
519    world: &FlintWorld,
520    resolver: &mut ProcGenResolver,
521    mode: ProcGenMode,
522    mut sink: GpuSink<'_>,
523) {
524    let started = web_time::Instant::now();
525    discover_specs(resolver, scene_path);
526    let requests = collect_requests(world, sink.config, resolver, &sink);
527    if requests.is_empty() {
528        return;
529    }
530    let before = resolver.cache_stats();
531    resolve(&requests, resolver, &mut sink, mode);
532    let stats = resolver.cache_stats();
533    // One line a load: how long procgen held the scene up, and where the
534    // assets came from (#181: the disk tier is what makes a relaunch fast).
535    // Deferred assets are queued, so they appear in none of the three.
536    println!(
537        "[procgen] {} asset(s) in {:.0} ms: {} generated, {} from disk, {} from memory",
538        requests.len(),
539        started.elapsed().as_secs_f64() * 1e3,
540        stats.misses - before.misses,
541        stats.disk_hits - before.disk_hits,
542        stats.hits - before.hits,
543    );
544    tracing::debug!(
545        "Procgen cache: {} entries, {} of {} bytes, {} hits / {} misses, {} evicted",
546        stats.entry_count,
547        stats.total_bytes,
548        stats.max_memory_bytes,
549        stats.hits,
550        stats.misses,
551        stats.evictions,
552    );
553}
