1//! The music-session lifecycle inside the player app (Phase 4 F3, ADR 0019).
2//!
3//! A scene declares its suite with a `music_session` component
4//! (`schemas/components/music_session.toml`, game-side); the player starts a
5//! [`ChartSession`] on the shared kira `AudioManager` (ADR 0017), hands the
6//! gamepad to the 1 kHz capture thread for the session's duration, and
7//! down-samples the same capture drain into `InputState` so the script/action
8//! surface keeps working frame-quantized (ADR 0018). Headless (no audio
9//! device) degrades to no-session, matching the engine's Option-audio
10//! convention.
11//!
12//! While a session is active only lean + pulse reach `InputState` — every
13//! other pad control is dropped inside the capture crate (accepted gap,
14//! ADR 0018). Keyboard stays winit throughout.
15

30/// Gamepad slot the down-sampled events claim in `InputState`.
31const GAMEPAD_SLOT: u32 = 0;
32/// The pulse verb's button name in `InputState`'s gamepad vocabulary.
33const PULSE_BUTTON: &str = "South";
34/// Teardown fade: long enough to avoid a click, short enough to feel like
35/// the world simply lets go.
36const STOP_FADE_MS: f64 = 50.0;
37
38/// Ladder visuals → engine post params (ADR 0021; F4/F6 move authorship into
39/// scripts via conducted params, superseding this direct path script-wins).
40/// The scales are provisional, owned by the Phase 4 [H] look-check: the
41/// ladder's `blur` becomes zoom-style radial blur (world-losing-focus),
42/// `chromatic` and `desaturate` map 1:1 (rung values are authored 0..1 and
43/// the composite shader's ash-grey formula does its own darkening).
44pub const MUSIC_BLUR_SCALE: f32 = 0.6;
45pub const MUSIC_CHROMATIC_SCALE: f32 = 1.0;
46pub const MUSIC_DESATURATE_SCALE: f32 = 1.0;
47
48/// The scene's authored values for the three ladder-driven post params,
49/// captured at session start. The merge writes `base + rung × scale` every
50/// frame (not only when a rung is active) so a ladder that recovers to clean
51/// actually restores the authored look — the renderer config is sticky, and
52/// an only-when-degraded write would leave the last degraded values applied
53/// forever (ADR 0021).
54#[derive(Debug, Clone, Copy, Default)]
55pub struct LadderPostBase {
56    pub radial_blur: f32,
57    pub chromatic_aberration: f32,
58    pub desaturate: f32,
59}
60
61/// Merge the ladder visuals into this frame's post overrides. Script
62/// overrides win (only `None` slots are filled); preroll leaves the world
63/// untouched.
64pub fn merge_ladder_postprocess(
65    radial_blur: &mut Option<f32>,
66    chromatic_aberration: &mut Option<f32>,
67    desaturation: &mut Option<f32>,
68    vf: &VisualFrame,
69    base: LadderPostBase,
70) {
71    if vf.preroll {
72        return;
73    }
74    if radial_blur.is_none() {
75        *radial_blur = Some(base.radial_blur + vf.blur * MUSIC_BLUR_SCALE);
76    }
77    if chromatic_aberration.is_none() {
78        *chromatic_aberration =
79            Some(base.chromatic_aberration + vf.chromatic * MUSIC_CHROMATIC_SCALE);
80    }
81    if desaturation.is_none() {
82        *desaturation = Some((base.desaturate + vf.desaturate * MUSIC_DESATURATE_SCALE).min(1.0));
83    }
84}

86/// One-shot restore of the authored values after session teardown, so the
87/// config does not end stuck mid-ladder. Script overrides still win.
88pub fn restore_ladder_postprocess(
89    radial_blur: &mut Option<f32>,
90    chromatic_aberration: &mut Option<f32>,
91    desaturation: &mut Option<f32>,
92    base: LadderPostBase,
93) {
94    if radial_blur.is_none() {
95        *radial_blur = Some(base.radial_blur);
96    }
97    if chromatic_aberration.is_none() {
98        *chromatic_aberration = Some(base.chromatic_aberration);
99    }
100    if desaturation.is_none() {
101        *desaturation = Some(base.desaturate);
102    }
103}

105/// One scene's music session: the reactive loop on the shared manager plus
106/// the InputState down-sampling state and permanent, near-free evidence
107/// counters (the F2 pass criteria, re-checked on every real run).
108///
109/// `ChartSession`'s internal field order drops the capture guard before the
110/// kira handles; `PlayerApp`'s field order drops those handles before the
111/// shared manager (ADR 0017).
112pub struct MusicSession {
113    session: ChartSession,
114    /// A down-sampled pulse press is released at the start of the *next*
115    /// tick so action edge detection sees a full down/up pair (ADR 0018).
116    pulse_release_pending: bool,
117    events_total: u64,
118    monotonic_violations: u64,
119    last_event_sample: i64,
120    /// Max stem skew that survived the ADR 0017 re-read discriminator
121    /// (a one-callback-buffer reporting transient collapses on re-read).
122    max_confirmed_skew_ms: f64,
123    last_skew_check: Instant,
124    /// Component's `quit_on_finish`: when the session ends on its own the
125    /// player exits instead of idling in the post-session neutral world.
126    quit_on_finish: bool,
127    /// Armed only when no gamepad was visible at session start.
128    #[cfg(feature = "debug-hud")]
129    debug_keys: Option<DebugKeyInput>,
130}

132/// Debug keyboard input (P5, 2026-08-27, `debug-hud` builds only): when a
133/// session starts with NO gamepad visible, arrows = lean and Space = pulse
134/// feed the session's ordinary input channel, stamped through the same
135/// ClockBridge arithmetic as the capture thread (compensated clock sample,
136/// monotonic-guarded). A stand-in for seam/feedback iteration when no pad is
137/// at hand — digital lean at a fixed radius, frame-quantized stamps, no
138/// keyboard-specific calibration — never an [H] feel surface. Covers the
139/// prototype verbs only (lean + pulse); sway/pressure/flick stay silent.
140#[cfg(feature = "debug-hud")]
141struct DebugKeyInput {
142    tx: std::sync::mpsc::Sender<InputEvent>,
143    bridge: flint_music::ClockBridge,
144    offset_samples: i64,
145    last_sample: i64,
146    last_lean: (f64, f64),
147}
148
149/// Digital lean radius: matches the chart's authored target radii (~0.8–0.85)
150/// so a held arrow scores like a landed lean rather than a rail-pinned one.
151#[cfg(feature = "debug-hud")]
152const KEY_LEAN_R: f64 = 0.85;
153
154#[cfg(feature = "debug-hud")]
155impl DebugKeyInput {
156    fn feed(&mut self, input: &InputState) {
157        use winit::keyboard::KeyCode;
158        let mut x: f64 = 0.0;
159        let mut y: f64 = 0.0;
160        if input.is_key_down(KeyCode::ArrowRight) {
161            x += 1.0;
162        }
163        if input.is_key_down(KeyCode::ArrowLeft) {
164            x -= 1.0;
165        }
166        if input.is_key_down(KeyCode::ArrowUp) {
167            y += 1.0;
168        }
169        if input.is_key_down(KeyCode::ArrowDown) {
170            y -= 1.0;
171        }
172        let mag = (x * x + y * y).sqrt();
173        if mag > 0.0 {
174            x = x / mag * KEY_LEAN_R;
175            y = y / mag * KEY_LEAN_R;
176        }
177        let pulse = input.is_key_just_pressed(KeyCode::Space);
178        let lean_dirty = (x, y) != self.last_lean;
179        if !lean_dirty && !pulse {
180            return;
181        }
182        let Some(clock_sample) = self.bridge.sample_at(Instant::now()) else {
183            return; // bridge not fed yet (session preroll warm-up)
184        };
185        // Same monotonic guard as the capture thread: a bridge refit can
186        // step the mapping backwards by a hair; judgment requires
187        // nondecreasing stamps.
188        let sample = (clock_sample - self.offset_samples).max(self.last_sample);
189        self.last_sample = sample;
190        if lean_dirty {
191            let _ = self
192                .tx
193                .send(InputEvent::Lean(flint_music::LeanSample { sample, x, y }));
194            self.last_lean = (x, y);
195        }
196        if pulse {
197            let _ = self.tx.send(InputEvent::Pulse(flint_music::PulseEvent {
198                sample,
199                kind: "pulse".into(),
200                direction: None,
201            }));
202        }
203    }
204}

216    /// Parse the component, start the suite on the player's shared manager,
217    /// and spawn capture. `Ok(None)` when no audio device exists (headless —
218    /// no session, everything else runs).
219    pub fn start(
220        base_dir: &Path,
221        comp: &toml::Value,
222        audio: &mut AudioEngine,
223    ) -> Result<Option<Self>> {
224        let manifest = comp_str(comp, "manifest")
225            .ok_or_else(|| anyhow!("music_session component: `manifest` is required"))?;
226        let chart = comp_str(comp, "chart")
227            .ok_or_else(|| anyhow!("music_session component: `chart` is required"))?;
228        // Declared-for-later field (F7): parsed so a populated field is
229        // never a silent lie, but this lifecycle does not read it yet.
230        if let Some(t) = comp_str(comp, "tuning_config") {
231            tracing::debug!("music_session: tuning_config `{t}` declared; not read until F7");
232        }
233        // `bindings` is declarative (ADR 0020): scripts load through the one
234        // script-discovery owner — a `script` component on a scene entity —
235        // never through this field. Validate it so a typo is loud.
236        if let Some(b) = comp_str(comp, "bindings") {
237            let path = base_dir.join(&b);
238            if vfs::exists(&path) {
239                tracing::info!(
240                    "music_session: bindings `{b}` present (load it via a `script` component)"
241                );
242            } else {
243                tracing::warn!(
244                    "music_session: bindings `{b}` does not exist at {} — this field is \
245                     declarative; the script loads via a `script` component on a scene entity",
246                    path.display()
247                );
248            }
249        }
250        let lean_mode = comp_str(comp, "lean_mode").unwrap_or_else(|| "arrival".to_string());
251        let lean_mode =
252            parse_lean_mode(&lean_mode).map_err(|e| anyhow!("music_session component: {e}"))?;
253        // Coercion helper because TOML `16` and `16.0` are different types.
254        let bars = comp
255            .get("bars")
256            .and_then(flint_core::toml_util::toml_f64)
257            .map(|b| b.max(0.0).round() as u64)
258            .filter(|&b| b > 0);
259        let quit_on_finish = comp
260            .get("quit_on_finish")
261            .and_then(|v| v.as_bool())
262            .unwrap_or(false);
263
264        // Timing offsets from the committed logs in <base_dir>/logs/latency/.
265        // Shared resolver (one wording with the CLI harness); operator-facing
266        // session diagnostics stay on println deliberately — tracing's
267        // default level filters info, and the Phase-3 double-failure lesson
268        // is that input/timing liveness must be LOUD.
269        let offsets = flint_music::chart_session::resolve_timing_offsets(base_dir);
270        for n in &offsets.notices {
271            println!("[music] {n}");
272        }
273        let (latency_ms, calibration_ms) = (offsets.latency_ms, offsets.calibration_ms);
274
275        let cfg = ChartSessionConfig {
276            manifest: base_dir.join(&manifest),
277            chart: base_dir.join(&chart),
278            base_dir: base_dir.to_path_buf(),
279            coherence_config: comp_str(comp, "coherence_config").map(|p| base_dir.join(p)),
280            ladder_config: comp_str(comp, "ladder_config").map(|p| base_dir.join(p)),
281            gradient_config: comp_str(comp, "gradient_config").map(|p| base_dir.join(p)),
282            haptics_config: comp_str(comp, "haptics_config").map(|p| base_dir.join(p)),
283            record: None,
284            bars,
285            lean_mode,
286            latency_ms,
287            calibration_ms,
288        };
289
290        // The shared manager (ADR 0017): the borrow ends at start — the
291        // session keeps only kira handles, the manager stays in the engine.
292        let Some(manager) = audio.manager_mut() else {
293            println!("[music] no audio device — session skipped");
294            return Ok(None);
295        };
296        let (mut session, notices) =
297            ChartSession::open_shared(&cfg, manager).map_err(|e| anyhow!("{e}"))?;
298        for n in notices {
299            println!("[music] {n}");
300        }
301
302        // Gamepad handoff (ADR 0018): the capture thread owns the pad for the
303        // run. Capture failure is non-fatal — the session plays without input
304        // (and the bar-2 NO INPUT notice will not fire: no receiver attached).
305        let offset_samples =
306            judgment_offset_samples(latency_ms, calibration_ms, session.sample_rate());
307        let verb_map = match comp_str(comp, "input_map") {
308            Some(m) => flint_input_capture::VerbMap::parse(&m)
309                .map_err(|e| anyhow!("music_session component: {e}"))?,
310            None => flint_input_capture::VerbMap::default(),
311        };
312        // Debug keyboard fallback (debug-hud builds): with no pad visible,
313        // arm arrows/Space instead of spawning capture — the session gets a
314        // live receiver either way, so the NO INPUT diagnostics stay honest.
315        #[cfg(feature = "debug-hud")]
316        let debug_keys = match flint_input_capture::connected_gamepads() {
317            Ok(pads) if pads.is_empty() => {
318                let (tx, rx) = std::sync::mpsc::channel();
319                let bridge = session.bridge();
320                session.set_input(rx, Box::new(()));
321                println!(
322                    "[music] *** NO GAMEPADS VISIBLE — debug KEYBOARD input armed \
323                     (debug-hud build): arrows = lean, Space = pulse. Prototype \
324                     verbs only; not a feel surface."
325                );
326                Some(DebugKeyInput {
327                    tx,
328                    bridge,
329                    offset_samples,
330                    last_sample: i64::MIN,
331                    last_lean: (0.0, 0.0),
332                })
333            }
334            _ => None,
335        };
336        #[cfg(feature = "debug-hud")]
337        let attach_pad = debug_keys.is_none();
338        #[cfg(not(feature = "debug-hud"))]
339        let attach_pad = true;
340        if attach_pad {
341            for n in
342                flint_input_capture::attach_session_input(&mut session, verb_map, offset_samples)
343            {
344                println!("[music] {n}");
345            }
346        }
347
348        Ok(Some(Self {
349            session,
350            pulse_release_pending: false,
351            events_total: 0,
352            monotonic_violations: 0,
353            last_event_sample: i64::MIN,
354            max_confirmed_skew_ms: 0.0,
355            last_skew_check: Instant::now(),
356            quit_on_finish,
357            #[cfg(feature = "debug-hud")]
358            debug_keys,
359        }))
360    }

362    /// One player frame. Returns `true` when the session is over (naturally
363    /// or on error) — the caller tears down via [`MusicSession::stop`].
364    pub fn tick(&mut self, input: &mut InputState) -> bool {
365        if self.pulse_release_pending {
366            input.process_gamepad_button_up(GAMEPAD_SLOT, PULSE_BUTTON);
367            self.pulse_release_pending = false;
368        }
369
370        // Debug keyboard stand-in: synthesize this frame's events BEFORE the
371        // drain so they judge on this tick. The events then round-trip into
372        // InputState through the ordinary down-sample tap below, exactly
373        // like gamepad events.
374        #[cfg(feature = "debug-hud")]
375        if let Some(dk) = self.debug_keys.as_mut() {
376            dk.feed(input);
377        }
378
379        // Split borrows: the tap closure feeds InputState and the evidence
380        // counters while `session` runs the drain (ADR 0018 down-sample).
381        let Self {
382            session,
383            pulse_release_pending,
384            events_total,
385            monotonic_violations,
386            last_event_sample,
387            ..
388        } = self;
389        let outcome = session.tick_with(|ev| {
390            *events_total += 1;
391            let sample = ev.sample();
392            if sample < *last_event_sample {
393                *monotonic_violations += 1;
394            }
395            *last_event_sample = sample;
396            match ev {
397                InputEvent::Lean(l) => {
398                    input.process_gamepad_axis(GAMEPAD_SLOT, "LeftStickX", l.x as f32);
399                    input.process_gamepad_axis(GAMEPAD_SLOT, "LeftStickY", l.y as f32);
400                }
401                InputEvent::Pulse(_) => {
402                    input.process_gamepad_button_down(GAMEPAD_SLOT, PULSE_BUTTON);
403                    *pulse_release_pending = true;
404                }
405                // The InputState down-sample carries the prototype verbs
406                // only; sway/pressure reach scripts via the conducted
407                // surface, not InputState (ADR 0018 scope).
408                InputEvent::Sway(_) | InputEvent::Pressure(_) => {}
409            }
410        });
411        let outcome = match outcome {
412            Ok(o) => o,
413            Err(e) => {
414                tracing::warn!("music session tick failed, ending session: {e}");
415                return true;
416            }
417        };
418        for n in &outcome.notices {
419            println!("[music] {n}");
420        }
421
422        // Once per second: stem-skew evidence with the ADR 0017 re-read
423        // discriminator (an exactly-one-buffer transient collapses on
424        // re-read; real de-sync persists and grows).
425        if self.last_skew_check.elapsed().as_secs_f64() >= 1.0 {
426            self.last_skew_check = Instant::now();
427            let mut skew_ms = self.session.max_stem_skew() * 1000.0;
428            if skew_ms > 0.5 {
429                // NOT a no-op: max_stem_skew() live-queries kira playback
430                // positions, so this second call is a genuine re-read taken
431                // one instant later. A reporting transient (exactly one
432                // audio-callback buffer of handle-position staleness, ADR
433                // 0017) collapses to ~0 on the re-read and the min() keeps
434                // the smaller value; real de-sync survives both reads.
435                skew_ms = skew_ms.min(self.session.max_stem_skew() * 1000.0);
436                if skew_ms > 0.5 {
437                    tracing::warn!("stem skew {skew_ms:.3} ms survived a re-read");
438                }
439            }
440            self.max_confirmed_skew_ms = self.max_confirmed_skew_ms.max(skew_ms);
441        }
442
443        outcome.state == Tick::Finished
444    }

483    /// The frame → script-POD conversion (ADR 0020): the only place
484    /// flint-music types meet flint-script types — flint-script never
485    /// depends on flint-music.
486    pub fn conducted_snapshot(&self) -> flint_script::ConductedSnapshot {
487        let vf = self.session.visual_frame();
488        // Next-window telegraph (P5 immediate-feedback pass): the nearest
489        // unconsumed judgment window inside a short horizon, so the scene
490        // can telegraph that input is expected. Read-only guide-frame query,
491        // per-frame cost is a filter over the window list — cheap.
492        let (next_pulse_beats, pulse_window_open) = {
493            let gf = self.session.guide_frame(16.0);
494            let beats = gf
495                .windows
496                .iter()
497                .map(|w| w.beats_until)
498                .fold(f64::INFINITY, f64::min);
499            (
500                if beats.is_finite() { beats } else { 1e6 },
501                gf.windows.iter().any(|w| w.open_now),
502            )
503        };
504        flint_script::ConductedSnapshot {
505            next_pulse_beats,
506            pulse_window_open,
507            lean: [vf.lean[0] as f64, vf.lean[1] as f64],
508            sway: [vf.sway[0] as f64, vf.sway[1] as f64],
509            pressure_l: vf.pressure_l as f64,
510            pressure_r: vf.pressure_r as f64,
511            cues: vf
512                .cues
513                .iter()
514                .map(|c| flint_script::ConductedCue {
515                    name: c.name.clone(),
516                    age: c.age_s,
517                    // Flatten to primitives; nested tables/arrays dropped
518                    // (author flat params — ADR 0033).
519                    params: c
520                        .params
521                        .iter()
522                        .filter_map(|(k, v)| {
523                            let p = match v {
524                                toml::Value::Float(f) => flint_script::CueParam::Number(*f),
525                                toml::Value::Integer(i) => {
526                                    flint_script::CueParam::Number(*i as f64)
527                                }
528                                toml::Value::String(s) => flint_script::CueParam::Text(s.clone()),
529                                toml::Value::Boolean(b) => flint_script::CueParam::Flag(*b),
530                                _ => return None,
531                            };
532                            Some((k.clone(), p))
533                        })
534                        .collect(),
535                })
536                .collect(),
537            target: [vf.target[0] as f64, vf.target[1] as f64],
538            next_target: [vf.next_target[0] as f64, vf.next_target[1] as f64],
539            next_target_beats: vf.next_target_beats,
540            coherence: vf.coherence as f64,
541            beat_phase: vf.beat_phase as f64,
542            bar_phase: vf.bar_phase as f64,
543            bar: vf.bar,
544            beat: vf.beat,
545            section: vf.section,
546            pulses: vf
547                .pulses
548                .iter()
549                .map(|p| flint_script::ConductedPulse {
550                    age: p.age_s,
551                    err_ms: p.err_ms,
552                    kind: p.kind.as_str().to_string(),
553                })
554                .collect(),
555            desaturate: vf.desaturate as f64,
556            blur: vf.blur as f64,
557            chromatic: vf.chromatic as f64,
558            reassembly: vf.reassembly as f64,
559            rewind: vf.rewind as f64,
560            no_input: vf.no_input,
561            preroll: vf.preroll,
562        }
563    }

565    /// Teardown: stop the stems with a fade FIRST — on the shared manager,
566    /// dropping handles does not stop sounds — then finish (which drops the
567    /// capture guard before anything else). Prints the summary plus the
568    /// F2-criteria evidence line.
569    pub fn stop(mut self) {
570        self.session.stop_audio(STOP_FADE_MS);
571        let evidence = format!(
572            "[music] session evidence: max confirmed skew {:.3} ms | events {} | \
573             monotonic violations {}",
574            self.max_confirmed_skew_ms, self.events_total, self.monotonic_violations,
575        );
576        match self.session.finish() {
577            Ok(notices) => {
578                for n in notices {
579                    println!("[music] {n}");
580                }
581            }
582            Err(e) => tracing::warn!("music session finish failed: {e}"),
583        }
584        println!("{evidence}");
585    }
586}
