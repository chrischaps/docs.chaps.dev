1//! The music-session lifecycle inside the player app (Phase 4 F3, ADR 0019).
2//!
3//! A scene declares its suite with a `music_session` component
4//! (`schemas/components/music_session.toml`, game-side); the player starts a
5//! [`ChartSession`] on the shared kira `AudioManager` (ADR 0017), hands the
6//! gamepad to the 1 kHz capture thread for the session's duration, and
7//! down-samples the same capture drain into `InputState` so the script/action
8//! surface keeps working frame-quantized (ADR 0018). Headless (no audio
9//! device) degrades to no-session, matching the engine's Option-audio
10//! convention.
11//!
12//! While a session is active only lean + pulse reach `InputState` — every
13//! other pad control is dropped inside the capture crate (accepted gap,
14//! ADR 0018). Keyboard stays winit throughout.
15

29/// Gamepad slot the down-sampled events claim in `InputState`.
30const GAMEPAD_SLOT: u32 = 0;
31/// The pulse verb's button name in `InputState`'s gamepad vocabulary.
32const PULSE_BUTTON: &str = "South";
33/// Teardown fade: long enough to avoid a click, short enough to feel like
34/// the world simply lets go.
35const STOP_FADE_MS: f64 = 50.0;
36
37/// Ladder visuals → engine post params (ADR 0021; F4/F6 move authorship into
38/// scripts via conducted params, superseding this direct path script-wins).
39/// The scales are provisional, owned by the Phase 4 [H] look-check: the
40/// ladder's `blur` becomes zoom-style radial blur (world-losing-focus),
41/// `chromatic` and `desaturate` map 1:1 (rung values are authored 0..1 and
42/// the composite shader's ash-grey formula does its own darkening).
43pub const MUSIC_BLUR_SCALE: f32 = 0.6;
44pub const MUSIC_CHROMATIC_SCALE: f32 = 1.0;
45pub const MUSIC_DESATURATE_SCALE: f32 = 1.0;
46
47/// The scene's authored values for the three ladder-driven post params,
48/// captured at session start. The merge writes `base + rung × scale` every
49/// frame (not only when a rung is active) so a ladder that recovers to clean
50/// actually restores the authored look — the renderer config is sticky, and
51/// an only-when-degraded write would leave the last degraded values applied
52/// forever (ADR 0021).
53#[derive(Debug, Clone, Copy, Default)]
54pub struct LadderPostBase {
55    pub radial_blur: f32,
56    pub chromatic_aberration: f32,
57    pub desaturate: f32,
58}
59
60/// Merge the ladder visuals into this frame's post overrides. Script
61/// overrides win (only `None` slots are filled); preroll leaves the world
62/// untouched.
63pub fn merge_ladder_postprocess(
64    radial_blur: &mut Option<f32>,
65    chromatic_aberration: &mut Option<f32>,
66    desaturation: &mut Option<f32>,
67    vf: &VisualFrame,
68    base: LadderPostBase,
69) {
70    if vf.preroll {
71        return;
72    }
73    if radial_blur.is_none() {
74        *radial_blur = Some(base.radial_blur + vf.blur * MUSIC_BLUR_SCALE);
75    }
76    if chromatic_aberration.is_none() {
77        *chromatic_aberration =
78            Some(base.chromatic_aberration + vf.chromatic * MUSIC_CHROMATIC_SCALE);
79    }
80    if desaturation.is_none() {
81        *desaturation = Some((base.desaturate + vf.desaturate * MUSIC_DESATURATE_SCALE).min(1.0));
82    }
83}

85/// One-shot restore of the authored values after session teardown, so the
86/// config does not end stuck mid-ladder. Script overrides still win.
87pub fn restore_ladder_postprocess(
88    radial_blur: &mut Option<f32>,
89    chromatic_aberration: &mut Option<f32>,
90    desaturation: &mut Option<f32>,
91    base: LadderPostBase,
92) {
93    if radial_blur.is_none() {
94        *radial_blur = Some(base.radial_blur);
95    }
96    if chromatic_aberration.is_none() {
97        *chromatic_aberration = Some(base.chromatic_aberration);
98    }
99    if desaturation.is_none() {
100        *desaturation = Some(base.desaturate);
101    }
102}

104/// One scene's music session: the reactive loop on the shared manager plus
105/// the InputState down-sampling state and permanent, near-free evidence
106/// counters (the F2 pass criteria, re-checked on every real run).
107///
108/// `ChartSession`'s internal field order drops the capture guard before the
109/// kira handles; `PlayerApp`'s field order drops those handles before the
110/// shared manager (ADR 0017).
111pub struct MusicSession {
112    session: ChartSession,
113    /// A down-sampled pulse press is released at the start of the *next*
114    /// tick so action edge detection sees a full down/up pair (ADR 0018).
115    pulse_release_pending: bool,
116    events_total: u64,
117    monotonic_violations: u64,
118    last_event_sample: i64,
119    /// Max stem skew that survived the ADR 0017 re-read discriminator
120    /// (a one-callback-buffer reporting transient collapses on re-read).
121    max_confirmed_skew_ms: f64,
122    last_skew_check: Instant,
123    /// Component's `quit_on_finish`: when the session ends on its own the
124    /// player exits instead of idling in the post-session neutral world.
125    quit_on_finish: bool,
126    /// Armed only when no gamepad was visible at session start.
127    #[cfg(feature = "debug-hud")]
128    debug_keys: Option<DebugKeyInput>,
129}

131/// Debug keyboard input (P5, 2026-08-27, `debug-hud` builds only): when a
132/// session starts with NO gamepad visible, arrows = lean and Space = pulse
133/// feed the session's ordinary input channel, stamped through the same
134/// ClockBridge arithmetic as the capture thread (compensated clock sample,
135/// monotonic-guarded). A stand-in for seam/feedback iteration when no pad is
136/// at hand — digital lean at a fixed radius, frame-quantized stamps, no
137/// keyboard-specific calibration — never an [H] feel surface. Covers the
138/// prototype verbs only (lean + pulse); sway/pressure/flick stay silent.
139#[cfg(feature = "debug-hud")]
140struct DebugKeyInput {
141    tx: std::sync::mpsc::Sender<InputEvent>,
142    bridge: flint_music::ClockBridge,
143    offset_samples: i64,
144    last_sample: i64,
145    last_lean: (f64, f64),
146}
147
148/// Digital lean radius: matches the chart's authored target radii (~0.8–0.85)
149/// so a held arrow scores like a landed lean rather than a rail-pinned one.
150#[cfg(feature = "debug-hud")]
151const KEY_LEAN_R: f64 = 0.85;
152
153#[cfg(feature = "debug-hud")]
154impl DebugKeyInput {
155    fn feed(&mut self, input: &InputState) {
156        use winit::keyboard::KeyCode;
157        let mut x: f64 = 0.0;
158        let mut y: f64 = 0.0;
159        if input.is_key_down(KeyCode::ArrowRight) {
160            x += 1.0;
161        }
162        if input.is_key_down(KeyCode::ArrowLeft) {
163            x -= 1.0;
164        }
165        if input.is_key_down(KeyCode::ArrowUp) {
166            y += 1.0;
167        }
168        if input.is_key_down(KeyCode::ArrowDown) {
169            y -= 1.0;
170        }
171        let mag = (x * x + y * y).sqrt();
172        if mag > 0.0 {
173            x = x / mag * KEY_LEAN_R;
174            y = y / mag * KEY_LEAN_R;
175        }
176        let pulse = input.is_key_just_pressed(KeyCode::Space);
177        let lean_dirty = (x, y) != self.last_lean;
178        if !lean_dirty && !pulse {
179            return;
180        }
181        let Some(clock_sample) = self.bridge.sample_at(Instant::now()) else {
182            return; // bridge not fed yet (session preroll warm-up)
183        };
184        // Same monotonic guard as the capture thread: a bridge refit can
185        // step the mapping backwards by a hair; judgment requires
186        // nondecreasing stamps.
187        let sample = (clock_sample - self.offset_samples).max(self.last_sample);
188        self.last_sample = sample;
189        if lean_dirty {
190            let _ = self
191                .tx
192                .send(InputEvent::Lean(flint_music::LeanSample { sample, x, y }));
193            self.last_lean = (x, y);
194        }
195        if pulse {
196            let _ = self.tx.send(InputEvent::Pulse(flint_music::PulseEvent {
197                sample,
198                kind: "pulse".into(),
199                direction: None,
200            }));
201        }
202    }
203}

215    /// Parse the component, start the suite on the player's shared manager,
216    /// and spawn capture. `Ok(None)` when no audio device exists (headless —
217    /// no session, everything else runs).
218    pub fn start(
219        base_dir: &Path,
220        comp: &toml::Value,
221        audio: &mut AudioEngine,
222    ) -> Result<Option<Self>> {
223        let manifest = comp_str(comp, "manifest")
224            .ok_or_else(|| anyhow!("music_session component: `manifest` is required"))?;
225        let chart = comp_str(comp, "chart")
226            .ok_or_else(|| anyhow!("music_session component: `chart` is required"))?;
227        // Declared-for-later field (F7): parsed so a populated field is
228        // never a silent lie, but this lifecycle does not read it yet.
229        if let Some(t) = comp_str(comp, "tuning_config") {
230            tracing::debug!("music_session: tuning_config `{t}` declared; not read until F7");
231        }
232        // `bindings` is declarative (ADR 0020): scripts load through the one
233        // script-discovery owner — a `script` component on a scene entity —
234        // never through this field. Validate it so a typo is loud.
235        if let Some(b) = comp_str(comp, "bindings") {
236            let path = base_dir.join(&b);
237            if path.exists() {
238                tracing::info!(
239                    "music_session: bindings `{b}` present (load it via a `script` component)"
240                );
241            } else {
242                tracing::warn!(
243                    "music_session: bindings `{b}` does not exist at {} — this field is \
244                     declarative; the script loads via a `script` component on a scene entity",
245                    path.display()
246                );
247            }
248        }
249        let lean_mode = comp_str(comp, "lean_mode").unwrap_or_else(|| "arrival".to_string());
250        let lean_mode =
251            parse_lean_mode(&lean_mode).map_err(|e| anyhow!("music_session component: {e}"))?;
252        // Coercion helper because TOML `16` and `16.0` are different types.
253        let bars = comp
254            .get("bars")
255            .and_then(flint_core::toml_util::toml_f64)
256            .map(|b| b.max(0.0).round() as u64)
257            .filter(|&b| b > 0);
258        let quit_on_finish = comp
259            .get("quit_on_finish")
260            .and_then(|v| v.as_bool())
261            .unwrap_or(false);
262
263        // Timing offsets from the committed logs in <base_dir>/logs/latency/.
264        // Shared resolver (one wording with the CLI harness); operator-facing
265        // session diagnostics stay on println deliberately — tracing's
266        // default level filters info, and the Phase-3 double-failure lesson
267        // is that input/timing liveness must be LOUD.
268        let offsets = flint_music::chart_session::resolve_timing_offsets(base_dir);
269        for n in &offsets.notices {
270            println!("[music] {n}");
271        }
272        let (latency_ms, calibration_ms) = (offsets.latency_ms, offsets.calibration_ms);
273
274        let cfg = ChartSessionConfig {
275            manifest: base_dir.join(&manifest),
276            chart: base_dir.join(&chart),
277            base_dir: base_dir.to_path_buf(),
278            coherence_config: comp_str(comp, "coherence_config").map(|p| base_dir.join(p)),
279            ladder_config: comp_str(comp, "ladder_config").map(|p| base_dir.join(p)),
280            gradient_config: comp_str(comp, "gradient_config").map(|p| base_dir.join(p)),
281            haptics_config: comp_str(comp, "haptics_config").map(|p| base_dir.join(p)),
282            record: None,
283            bars,
284            lean_mode,
285            latency_ms,
286            calibration_ms,
287        };
288
289        // The shared manager (ADR 0017): the borrow ends at start — the
290        // session keeps only kira handles, the manager stays in the engine.
291        let Some(manager) = audio.manager_mut() else {
292            println!("[music] no audio device — session skipped");
293            return Ok(None);
294        };
295        let (mut session, notices) =
296            ChartSession::open_shared(&cfg, manager).map_err(|e| anyhow!("{e}"))?;
297        for n in notices {
298            println!("[music] {n}");
299        }
300
301        // Gamepad handoff (ADR 0018): the capture thread owns the pad for the
302        // run. Capture failure is non-fatal — the session plays without input
303        // (and the bar-2 NO INPUT notice will not fire: no receiver attached).
304        let offset_samples =
305            judgment_offset_samples(latency_ms, calibration_ms, session.sample_rate());
306        let verb_map = match comp_str(comp, "input_map") {
307            Some(m) => flint_input_capture::VerbMap::parse(&m)
308                .map_err(|e| anyhow!("music_session component: {e}"))?,
309            None => flint_input_capture::VerbMap::default(),
310        };
311        // Debug keyboard fallback (debug-hud builds): with no pad visible,
312        // arm arrows/Space instead of spawning capture — the session gets a
313        // live receiver either way, so the NO INPUT diagnostics stay honest.
314        #[cfg(feature = "debug-hud")]
315        let debug_keys = match flint_input_capture::connected_gamepads() {
316            Ok(pads) if pads.is_empty() => {
317                let (tx, rx) = std::sync::mpsc::channel();
318                let bridge = session.bridge();
319                session.set_input(rx, Box::new(()));
320                println!(
321                    "[music] *** NO GAMEPADS VISIBLE — debug KEYBOARD input armed \
322                     (debug-hud build): arrows = lean, Space = pulse. Prototype \
323                     verbs only; not a feel surface."
324                );
325                Some(DebugKeyInput {
326                    tx,
327                    bridge,
328                    offset_samples,
329                    last_sample: i64::MIN,
330                    last_lean: (0.0, 0.0),
331                })
332            }
333            _ => None,
334        };
335        #[cfg(feature = "debug-hud")]
336        let attach_pad = debug_keys.is_none();
337        #[cfg(not(feature = "debug-hud"))]
338        let attach_pad = true;
339        if attach_pad {
340            for n in
341                flint_input_capture::attach_session_input(&mut session, verb_map, offset_samples)
342            {
343                println!("[music] {n}");
344            }
345        }
346
347        Ok(Some(Self {
348            session,
349            pulse_release_pending: false,
350            events_total: 0,
351            monotonic_violations: 0,
352            last_event_sample: i64::MIN,
353            max_confirmed_skew_ms: 0.0,
354            last_skew_check: Instant::now(),
355            quit_on_finish,
356            #[cfg(feature = "debug-hud")]
357            debug_keys,
358        }))
359    }

361    /// One player frame. Returns `true` when the session is over (naturally
362    /// or on error) — the caller tears down via [`MusicSession::stop`].
363    pub fn tick(&mut self, input: &mut InputState) -> bool {
364        if self.pulse_release_pending {
365            input.process_gamepad_button_up(GAMEPAD_SLOT, PULSE_BUTTON);
366            self.pulse_release_pending = false;
367        }
368
369        // Debug keyboard stand-in: synthesize this frame's events BEFORE the
370        // drain so they judge on this tick. The events then round-trip into
371        // InputState through the ordinary down-sample tap below, exactly
372        // like gamepad events.
373        #[cfg(feature = "debug-hud")]
374        if let Some(dk) = self.debug_keys.as_mut() {
375            dk.feed(input);
376        }
377
378        // Split borrows: the tap closure feeds InputState and the evidence
379        // counters while `session` runs the drain (ADR 0018 down-sample).
380        let Self {
381            session,
382            pulse_release_pending,
383            events_total,
384            monotonic_violations,
385            last_event_sample,
386            ..
387        } = self;
388        let outcome = session.tick_with(|ev| {
389            *events_total += 1;
390            let sample = ev.sample();
391            if sample < *last_event_sample {
392                *monotonic_violations += 1;
393            }
394            *last_event_sample = sample;
395            match ev {
396                InputEvent::Lean(l) => {
397                    input.process_gamepad_axis(GAMEPAD_SLOT, "LeftStickX", l.x as f32);
398                    input.process_gamepad_axis(GAMEPAD_SLOT, "LeftStickY", l.y as f32);
399                }
400                InputEvent::Pulse(_) => {
401                    input.process_gamepad_button_down(GAMEPAD_SLOT, PULSE_BUTTON);
402                    *pulse_release_pending = true;
403                }
404                // The InputState down-sample carries the prototype verbs
405                // only; sway/pressure reach scripts via the conducted
406                // surface, not InputState (ADR 0018 scope).
407                InputEvent::Sway(_) | InputEvent::Pressure(_) => {}
408            }
409        });
410        let outcome = match outcome {
411            Ok(o) => o,
412            Err(e) => {
413                tracing::warn!("music session tick failed, ending session: {e}");
414                return true;
415            }
416        };
417        for n in &outcome.notices {
418            println!("[music] {n}");
419        }
420
421        // Once per second: stem-skew evidence with the ADR 0017 re-read
422        // discriminator (an exactly-one-buffer transient collapses on
423        // re-read; real de-sync persists and grows).
424        if self.last_skew_check.elapsed().as_secs_f64() >= 1.0 {
425            self.last_skew_check = Instant::now();
426            let mut skew_ms = self.session.max_stem_skew() * 1000.0;
427            if skew_ms > 0.5 {
428                // NOT a no-op: max_stem_skew() live-queries kira playback
429                // positions, so this second call is a genuine re-read taken
430                // one instant later. A reporting transient (exactly one
431                // audio-callback buffer of handle-position staleness, ADR
432                // 0017) collapses to ~0 on the re-read and the min() keeps
433                // the smaller value; real de-sync survives both reads.
434                skew_ms = skew_ms.min(self.session.max_stem_skew() * 1000.0);
435                if skew_ms > 0.5 {
436                    tracing::warn!("stem skew {skew_ms:.3} ms survived a re-read");
437                }
438            }
439            self.max_confirmed_skew_ms = self.max_confirmed_skew_ms.max(skew_ms);
440        }
441
442        outcome.state == Tick::Finished
443    }

482    /// The frame → script-POD conversion (ADR 0020): the only place
483    /// flint-music types meet flint-script types — flint-script never
484    /// depends on flint-music.
485    pub fn conducted_snapshot(&self) -> flint_script::ConductedSnapshot {
486        let vf = self.session.visual_frame();
487        // Next-window telegraph (P5 immediate-feedback pass): the nearest
488        // unconsumed judgment window inside a short horizon, so the scene
489        // can telegraph that input is expected. Read-only guide-frame query,
490        // per-frame cost is a filter over the window list — cheap.
491        let (next_pulse_beats, pulse_window_open) = {
492            let gf = self.session.guide_frame(16.0);
493            let beats = gf
494                .windows
495                .iter()
496                .map(|w| w.beats_until)
497                .fold(f64::INFINITY, f64::min);
498            (
499                if beats.is_finite() { beats } else { 1e6 },
500                gf.windows.iter().any(|w| w.open_now),
501            )
502        };
503        flint_script::ConductedSnapshot {
504            next_pulse_beats,
505            pulse_window_open,
506            lean: [vf.lean[0] as f64, vf.lean[1] as f64],
507            sway: [vf.sway[0] as f64, vf.sway[1] as f64],
508            pressure_l: vf.pressure_l as f64,
509            pressure_r: vf.pressure_r as f64,
510            cues: vf
511                .cues
512                .iter()
513                .map(|c| flint_script::ConductedCue {
514                    name: c.name.clone(),
515                    age: c.age_s,
516                    // Flatten to primitives; nested tables/arrays dropped
517                    // (author flat params — ADR 0033).
518                    params: c
519                        .params
520                        .iter()
521                        .filter_map(|(k, v)| {
522                            let p = match v {
523                                toml::Value::Float(f) => flint_script::CueParam::Number(*f),
524                                toml::Value::Integer(i) => {
525                                    flint_script::CueParam::Number(*i as f64)
526                                }
527                                toml::Value::String(s) => flint_script::CueParam::Text(s.clone()),
528                                toml::Value::Boolean(b) => flint_script::CueParam::Flag(*b),
529                                _ => return None,
530                            };
531                            Some((k.clone(), p))
532                        })
533                        .collect(),
534                })
535                .collect(),
536            target: [vf.target[0] as f64, vf.target[1] as f64],
537            next_target: [vf.next_target[0] as f64, vf.next_target[1] as f64],
538            next_target_beats: vf.next_target_beats,
539            coherence: vf.coherence as f64,
540            beat_phase: vf.beat_phase as f64,
541            bar_phase: vf.bar_phase as f64,
542            bar: vf.bar,
543            beat: vf.beat,
544            section: vf.section,
545            pulses: vf
546                .pulses
547                .iter()
548                .map(|p| flint_script::ConductedPulse {
549                    age: p.age_s,
550                    err_ms: p.err_ms,
551                    kind: p.kind.as_str().to_string(),
552                })
553                .collect(),
554            desaturate: vf.desaturate as f64,
555            blur: vf.blur as f64,
556            chromatic: vf.chromatic as f64,
557            reassembly: vf.reassembly as f64,
558            rewind: vf.rewind as f64,
559            no_input: vf.no_input,
560            preroll: vf.preroll,
561        }
562    }

564    /// Teardown: stop the stems with a fade FIRST — on the shared manager,
565    /// dropping handles does not stop sounds — then finish (which drops the
566    /// capture guard before anything else). Prints the summary plus the
567    /// F2-criteria evidence line.
568    pub fn stop(mut self) {
569        self.session.stop_audio(STOP_FADE_MS);
570        let evidence = format!(
571            "[music] session evidence: max confirmed skew {:.3} ms | events {} | \
572             monotonic violations {}",
573            self.max_confirmed_skew_ms, self.events_total, self.monotonic_violations,
574        );
575        match self.session.finish() {
576            Ok(notices) => {
577                for n in notices {
578                    println!("[music] {n}");
579                }
580            }
581            Err(e) => tracing::warn!("music session finish failed: {e}"),
582        }
583        println!("{evidence}");
584    }
585}
