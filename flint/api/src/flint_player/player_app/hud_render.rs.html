3use flint_script::context::DrawCommand;
4use std::collections::HashMap;

6pub(super) fn to_color32(c: &[f32; 4]) -> egui::Color32 {
7    egui::Color32::from_rgba_unmultiplied(
8        (c[0] * 255.0) as u8,
9        (c[1] * 255.0) as u8,
10        (c[2] * 255.0) as u8,
11        (c[3] * 255.0) as u8,
12    )
13}

15/// Render script-issued 2D draw commands via egui layer painter.
16/// Uses `ctx.layer_painter()` directly instead of `egui::Area` to avoid
17/// zero-size clipping when only painter calls are used (no widgets).
18pub(super) fn render_draw_commands(
19    ctx: &egui::Context,
20    commands: &[DrawCommand],
21    ui_textures: &HashMap<String, egui::TextureHandle>,
22) {
23    if commands.is_empty() {
24        return;
25    }
26
27    // Sort by layer (stable sort preserves insertion order within same layer)
28    let mut sorted: Vec<&DrawCommand> = commands.iter().collect();
29    sorted.sort_by_key(|cmd| cmd.layer());
30
31    let layer_id = egui::LayerId::new(egui::Order::Foreground, egui::Id::new("script_ui_overlay"));
32    let painter = ctx.layer_painter(layer_id);
33
34    for cmd in &sorted {
35        match cmd {
36            DrawCommand::Text {
37                x,
38                y,
39                text,
40                size,
41                color,
42                align,
43                stroke,
44                ..
45            } => {
46                let anchor = match align {
47                    1 => egui::Align2::CENTER_TOP,
48                    2 => egui::Align2::RIGHT_TOP,
49                    _ => egui::Align2::LEFT_TOP,
50                };
51                let font = egui::FontId::proportional(*size);
52                let pos = egui::Pos2::new(*x, *y);
53
54                // Draw stroke (outline) by rendering text at 8 compass offsets
55                if let Some((stroke_color, stroke_width)) = stroke {
56                    let sc = to_color32(stroke_color);
57                    let w = *stroke_width;
58                    for &(dx, dy) in &[
59                        (-w, 0.0),
60                        (w, 0.0),
61                        (0.0, -w),
62                        (0.0, w),
63                        (-w, -w),
64                        (w, -w),
65                        (-w, w),
66                        (w, w),
67                    ] {
68                        painter.text(
69                            egui::Pos2::new(pos.x + dx, pos.y + dy),
70                            anchor,
71                            text,
72                            font.clone(),
73                            sc,
74                        );
75                    }
76                }
77
78                painter.text(pos, anchor, text, font, to_color32(color));
79            }
80
81            DrawCommand::RectFilled {
82                x,
83                y,
84                w,
85                h,
86                color,
87                rounding,
88                ..
89            } => {
90                let rect =
91                    egui::Rect::from_min_size(egui::Pos2::new(*x, *y), egui::Vec2::new(*w, *h));
92                painter.rect_filled(rect, *rounding, to_color32(color));
93            }
94
95            DrawCommand::RectOutline {
96                x,
97                y,
98                w,
99                h,
100                color,
101                thickness,
102                ..
103            } => {
104                let rect =
105                    egui::Rect::from_min_size(egui::Pos2::new(*x, *y), egui::Vec2::new(*w, *h));
106                painter.rect_stroke(rect, 0.0, egui::Stroke::new(*thickness, to_color32(color)));
107            }
108
109            DrawCommand::CircleFilled {
110                x,
111                y,
112                radius,
113                color,
114                ..
115            } => {
116                painter.circle_filled(egui::Pos2::new(*x, *y), *radius, to_color32(color));
117            }
118
119            DrawCommand::CircleOutline {
120                x,
121                y,
122                radius,
123                color,
124                thickness,
125                ..
126            } => {
127                painter.circle_stroke(
128                    egui::Pos2::new(*x, *y),
129                    *radius,
130                    egui::Stroke::new(*thickness, to_color32(color)),
131                );
132            }
133
134            DrawCommand::Line {
135                x1,
136                y1,
137                x2,
138                y2,
139                color,
140                thickness,
141                ..
142            } => {
143                painter.line_segment(
144                    [egui::Pos2::new(*x1, *y1), egui::Pos2::new(*x2, *y2)],
145                    egui::Stroke::new(*thickness, to_color32(color)),
146                );
147            }
148
149            DrawCommand::Sprite {
150                x,
151                y,
152                w,
153                h,
154                name,
155                uv,
156                tint,
157                ..
158            } => {
159                if let Some(tex_handle) = ui_textures.get(name.as_str()) {
160                    let rect =
161                        egui::Rect::from_min_size(egui::Pos2::new(*x, *y), egui::Vec2::new(*w, *h));
162                    let uv_rect = egui::Rect::from_min_max(
163                        egui::Pos2::new(uv[0], uv[1]),
164                        egui::Pos2::new(uv[2], uv[3]),
165                    );
166                    painter.image(tex_handle.id(), rect, uv_rect, to_color32(tint));
167                }
168            }
169        }
170    }
