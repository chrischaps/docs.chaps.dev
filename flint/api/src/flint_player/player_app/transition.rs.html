20/// Scene transition lifecycle phase
21#[derive(Debug, Clone)]
22pub(super) enum TransitionPhase {
23    /// Normal gameplay
24    Idle,
25    /// Playing exit transition — scripts draw fade-out visuals
26    Exiting { target_scene: String, elapsed: f32 },
27    /// Loading the new scene (synchronous, happens in one frame)
28    Loading { target_scene: String },
29    /// Playing enter transition — scripts draw fade-in visuals
30    Entering { elapsed: f32 },
31}
32
33impl TransitionPhase {
34    pub(super) fn is_idle(&self) -> bool {
35        matches!(self, TransitionPhase::Idle)
36    }
37}

40    /// Advance the transition phase based on elapsed time and script signals.
41    pub(super) fn advance_transition(&mut self) {
42        match &self.transition_phase {
43            TransitionPhase::Idle => {}
44            TransitionPhase::Exiting {
45                target_scene,
46                elapsed,
47            } => {
48                let new_elapsed = elapsed + self.clock.delta_time as f32;
49                let target = target_scene.clone();
50                self.transition_phase = TransitionPhase::Exiting {
51                    target_scene: target.clone(),
52                    elapsed: new_elapsed,
53                };
54                // Check if a complete_transition event was fired
55                // (We use a sentinel event name to signal completion)
56            }
57            TransitionPhase::Loading { target_scene } => {
58                let target = target_scene.clone();
59                self.execute_scene_transition(&target);
60                self.transition_phase = TransitionPhase::Entering { elapsed: 0.0 };
61            }
62            TransitionPhase::Entering { elapsed } => {
63                let new_elapsed = elapsed + self.clock.delta_time as f32;
64                self.transition_phase = TransitionPhase::Entering {
65                    elapsed: new_elapsed,
66                };
67            }
68        }
69    }

71    /// Load a chunk TOML file as additional entities with offset.
72    pub(super) fn load_chunk(&mut self, path: &str, offset_x: f32, offset_y: f32, chunk_id: &str) {
73        if self.loaded_chunks.contains_key(chunk_id) {
74            tracing::warn!("Chunk '{}' already loaded, skipping", chunk_id);
75            return;
76        }
77
78        // Resolve path relative to scene directory
79        let scene_dir = Path::new(&self.scene_path)
80            .parent()
81            .unwrap_or_else(|| Path::new("."));
82        let chunk_path = scene_dir.join(path);
83
84        // Parse chunk as a scene file
85        let content = match std::fs::read_to_string(&chunk_path) {
86            Ok(c) => c,
87            Err(e) => {
88                tracing::error!("Failed to read chunk '{}': {}", chunk_path.display(), e);
89                return;
90            }
91        };
92        let scene_file: flint_scene::SceneFile = match toml::from_str(&content) {
93            Ok(s) => s,
94            Err(e) => {
95                tracing::error!("Failed to parse chunk '{}': {}", chunk_path.display(), e);
96                return;
97            }
98        };
99
100        // Load schema registry for archetype defaults
101        let registry = if self.schema_paths.is_empty() {
102            flint_schema::SchemaRegistry::load_from_directory("schemas")
103                .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
104        } else {
105            let existing: Vec<&str> = self
106                .schema_paths
107                .iter()
108                .map(|s| s.as_str())
109                .filter(|p| Path::new(p).exists())
110                .collect();
111            if existing.is_empty() {
112                flint_schema::SchemaRegistry::new()
113            } else {
114                flint_schema::SchemaRegistry::load_from_directories(&existing)
115                    .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
116            }
117        };
118
119        let mut spawned_ids = Vec::new();
120
121        // Spawn entities with prefixed names
122        for (name, entity_def) in &scene_file.entities {
123            let prefixed_name = format!("{}_{}", chunk_id, name);
124            let id = match self.world.spawn(prefixed_name.clone()) {
125                Ok(id) => id,
126                Err(e) => {
127                    tracing::error!("Failed to spawn chunk entity '{}': {:?}", prefixed_name, e);
128                    continue;
129                }
130            };
131            spawned_ids.push(id);
132
133            // Set archetype + defaults
134            if let Some(archetype) = &entity_def.archetype {
135                if let Some(comps) = self.world.get_components_mut(id) {
136                    comps.archetype = Some(archetype.clone());
137                    if let Some(arch_schema) = registry.get_archetype(archetype) {
138                        for (comp_name, defaults) in &arch_schema.defaults {
139                            if !comps.has(comp_name) {
140                                comps.set(comp_name.clone(), defaults.clone());
141                            }
142                        }
143                    }
144                }
145            }
146
147            // Set component data
148            for (comp_name, comp_data) in &entity_def.components {
149                let _ = self.world.merge_component(id, comp_name, comp_data.clone());
150            }
151
152            // Apply position offset
153            if let Some(transform) = self.world.get_transform(id) {
154                let new_x = transform.position.x + offset_x;
155                let new_y = transform.position.y + offset_y;
156                let new_z = transform.position.z;
157                if let Some(comps) = self.world.get_components_mut(id) {
158                    comps.set_field(
159                        comp::TRANSFORM,
160                        "position",
161                        toml::Value::Array(vec![
162                            toml::Value::Float(new_x as f64),
163                            toml::Value::Float(new_y as f64),
164                            toml::Value::Float(new_z as f64),
165                        ]),
166                    );
167                }
168            } else {
169                // Entity has no transform yet — create one with the offset
170                if let Some(comps) = self.world.get_components_mut(id) {
171                    comps.set_field(
172                        comp::TRANSFORM,
173                        "position",
174                        toml::Value::Array(vec![
175                            toml::Value::Float(offset_x as f64),
176                            toml::Value::Float(offset_y as f64),
177                            toml::Value::Float(0.0),
178                        ]),
179                    );
180                }
181            }
182        }
183
184        // Re-sync physics for new entities
185        self.physics
186            .sync_new_entities(&mut self.world, &spawned_ids);
187
188        // Re-sync scripts for new entities
189        let chunk_scripts_dir = chunk_path
190            .parent()
191            .unwrap_or_else(|| Path::new("."))
192            .join("scripts");
193        for &id in &spawned_ids {
194            if let Some(comps) = self.world.get_components(id) {
195                if let Some(script_comp) = comps.get(comp::SCRIPT) {
196                    if let Some(source) = script_comp.get("source").and_then(|v| v.as_str()) {
197                        // Try chunk's own scripts dir first, then scene's scripts dir
198                        let scene_scripts_dir = Path::new(&self.scene_path)
199                            .parent()
200                            .unwrap_or_else(|| Path::new("."))
201                            .join("scripts");
202                        let script_path = if chunk_scripts_dir.join(source).exists() {
203                            chunk_scripts_dir.join(source)
204                        } else {
205                            scene_scripts_dir.join(source)
206                        };
207                        if script_path.exists() {
208                            self.script.load_script_for_entity(id, &script_path);
209                        }
210                    }
211                }
212            }
213        }
214
215        // Initialize scripts for new entities
216        self.script
217            .initialize_entities(&mut self.world, &spawned_ids);
218
219        println!(
220            "[chunk] Loaded '{}' ({} entities) at offset ({}, {})",
221            chunk_id,
222            spawned_ids.len(),
223            offset_x,
224            offset_y
225        );
226        self.loaded_chunks.insert(chunk_id.to_string(), spawned_ids);
227    }

229    /// Unload a previously loaded chunk by despawning all its entities.
230    pub(super) fn unload_chunk(&mut self, chunk_id: &str) {
231        if let Some(entity_ids) = self.loaded_chunks.remove(chunk_id) {
232            let count = entity_ids.len();
233            for id in &entity_ids {
234                self.physics.remove_entity(*id);
235                self.script.remove_entity(*id);
236                let _ = self.world.despawn(*id);
237            }
238            println!("[chunk] Unloaded '{}' ({} entities)", chunk_id, count);
239        } else {
240            tracing::warn!("Chunk '{}' not loaded, cannot unload", chunk_id);
241        }
242    }

244    /// Unload the current scene and load a new one.
245    pub(super) fn execute_scene_transition(&mut self, target_scene: &str) {
246        // Teardown: scripts told, music session stopped BEFORE audio.clear()
247        // (ADR 0017 producer→handles→device order), systems/world/terrain/
248        // transient render state cleared.
249        self.teardown_current_scene();
250
251        // Load: resolve + schema discovery + scene parse; on failure the
252        // transition aborts here (world already cleared, same as before).
253        if !self.load_target_scene(target_scene) {
254            return;
255        }
256
257        // Bring-up: camera, models, terrain, system re-init (audio before
258        // the scene-declared music session, session before script init so
259        // the conducted context exists from the scripts' first frame),
260        // on_scene_enter, cursor.
261        self.bring_up_scene();
262    }

264    fn teardown_current_scene(&mut self) {
265        println!("[transition] Unloading current scene...");
266
267        // Call on_scene_exit on all scripts
268        {
269            let _state_scope = self.script.state_scope(
270                &mut self.state_machine,
271                &mut self.persistent_store,
272                &self.physics,
273            );
274            self.script.call_scene_exits(&mut self.world);
275        }
276
277        // Music session first: capture guard drops and stems stop BEFORE the
278        // audio system clears (ADR 0017 producer→handles→device order).
279        self.stop_music_session();
280
281        // Clear all systems
282        self.script.clear();
283        self.audio.clear();
284        self.physics.clear();
285        self.animation.clear();
286        self.particles.clear();
287        self.procgen_resolver.clear_queue();
288
289        // Clear world
290        self.world = FlintWorld::new();
291
292        // Clear terrain
293        self.terrain = None;
294        if let Some(renderer) = &mut self.scene_renderer {
295            renderer.clear_terrain();
296        }
297
298        // Clear transient rendering state
299        self.skeletal_entity_assets.clear();
300        self.ui_textures.clear();
301        self.draw_commands.clear();
302        self.loaded_chunks.clear();
303    }

305    fn load_target_scene(&mut self, target_scene: &str) -> bool {
306        println!("[transition] Loading scene: {}", target_scene);
307
308        // Resolve scene path relative to current scene
309        let new_scene_path = resolve_scene_path(&self.scene_path, target_scene);
310
311        // Auto-discover schema dirs from the new scene path and merge
312        for dir in flint_schema::discover_schema_dirs(&new_scene_path) {
313            let s = dir.to_string_lossy().into_owned();
314            if !self.schema_paths.contains(&s) {
315                self.schema_paths.push(s);
316            }
317        }
318
319        // Load schema registry
320        let registry = if self.schema_paths.is_empty() {
321            // Try default schemas/ dir
322            flint_schema::SchemaRegistry::load_from_directory("schemas")
323                .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
324        } else {
325            let existing: Vec<&str> = self
326                .schema_paths
327                .iter()
328                .map(|s| s.as_str())
329                .filter(|p| Path::new(p).exists())
330                .collect();
331            if existing.is_empty() {
332                flint_schema::SchemaRegistry::new()
333            } else {
334                flint_schema::SchemaRegistry::load_from_directories(&existing)
335                    .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
336            }
337        };
338
339        // Parse and load scene
340        match flint_scene::load_scene(&new_scene_path, &registry) {
341            Ok((world, scene_file)) => {
342                self.world = world;
343                self.scene_path = new_scene_path.clone();
344                self.skybox_path = scene_file
345                    .environment
346                    .as_ref()
347                    .and_then(|env| env.skybox.clone());
348                self.scene_ambient = scene_file.environment.as_ref().and_then(|env| {
349                    (env.ambient_sky.is_some() || env.ambient_ground.is_some()).then(|| {
350                        (
351                            env.ambient_sky
352                                .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_SKY),
353                            env.ambient_ground
354                                .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_GROUND),
355                        )
356                    })
357                });
358                self.scene_diffuse_wrap = scene_file
359                    .environment
360                    .as_ref()
361                    .and_then(|env| env.diffuse_wrap);
362                self.scene_oren_nayar = scene_file
363                    .environment
364                    .as_ref()
365                    .and_then(|env| env.oren_nayar);
366                self.scene_sheen = scene_file.environment.as_ref().and_then(|env| {
367                    env.sheen_strength
368                        .map(|s| (env.sheen_color.unwrap_or([1.0; 3]), s))
369                });
370                self.scene_camera = scene_file.camera.clone();
371                self.scene_post_process = scene_file.post_process.clone();
372                self.scene_input_config = scene_file.scene.input_config.clone();
373                self.scene_preload_audio = scene_file.scene.preload_audio;
374            }
375            Err(e) => {
376                tracing::error!("Failed to load scene '{}': {:?}", new_scene_path, e);
377                return false;
378            }
379        }
380
381        // Rebuild component index after loading new scene
382        self.world.rebuild_component_index();
383
384        true
385    }

387    fn bring_up_scene(&mut self) {
388        // Apply camera config before borrowing renderer
389        self.apply_camera_def();
390
391        // Reload models
392        if let (Some(renderer), Some(context)) = (&mut self.scene_renderer, &self.render_context) {
393            renderer.scene_dir = Path::new(&self.scene_path)
394                .parent()
395                .map(|p| p.to_path_buf());
396
397            let config = build_model_load_config(
398                &self.scene_path,
399                &self.world,
400                self.catalog.as_ref(),
401                self.content_store.as_ref(),
402            );
403
404            // Discover procgen specs and resolve unresolved assets before model loading
405            {
406                let scene_dir = Path::new(&self.scene_path)
407                    .parent()
408                    .unwrap_or(Path::new("."));
409                let mut spec_dirs = vec![scene_dir.join("specs")];
410                if let Some(parent) = scene_dir.parent() {
411                    spec_dirs.push(parent.join("specs"));
412                }
413                spec_dirs.push(scene_dir.join("models"));
414                let dir_refs: Vec<&Path> = spec_dirs
415                    .iter()
416                    .filter(|d| d.is_dir())
417                    .map(|d| d.as_path())
418                    .collect();
419                self.procgen_resolver.discover_and_index(&dir_refs);
420
421                resolve_procgen_assets(
422                    &self.world,
423                    &mut self.procgen_resolver,
424                    renderer,
425                    &context.device,
426                    &context.queue,
427                    &config,
428                );
429            }
430
431            let load_result = model_loader::load_models_from_world(
432                &mut self.world,
433                renderer,
434                &context.device,
435                &context.queue,
436                &config,
437            );
438            register_skeletal_data(&load_result, &mut self.animation);
439            register_node_animation_data(&load_result, &mut self.animation);
440            self.skeletal_entity_assets = load_result.skinned_entities;
441            renderer.update_from_world(&self.world, &context.device);
442
443            // Reload splines
444            crate::spline_gen::load_splines(
445                &self.scene_path,
446                &mut self.world,
447                renderer,
448                Some(&mut self.physics),
449                &context.device,
450            );
451            renderer.update_from_world(&self.world, &context.device);
452
453            // Scene-authored hemisphere ambient + wrap; reset first so the
454            // previous scene's values never leak into one that doesn't
455            // author them (reset clears both ambient and wrap)
456            renderer.reset_ambient();
457            if let Some((sky, ground)) = self.scene_ambient {
458                renderer.set_ambient(sky, ground);
459            }
460            if let Some(wrap) = self.scene_diffuse_wrap {
461                renderer.set_diffuse_wrap(wrap);
462            }
463            if let Some(oren) = self.scene_oren_nayar {
464                renderer.set_oren_nayar(oren);
465            }
466            if let Some((color, strength)) = self.scene_sheen {
467                renderer.set_sheen(color, strength);
468            }
469
470            // Reload skybox
471            if let Some(skybox_rel) = &self.skybox_path {
472                let scene_dir = Path::new(&self.scene_path)
473                    .parent()
474                    .unwrap_or_else(|| Path::new("."));
475                let skybox_path = {
476                    let p = scene_dir.join(skybox_rel);
477                    if p.exists() {
478                        p
479                    } else if let Some(parent) = scene_dir.parent() {
480                        parent.join(skybox_rel)
481                    } else {
482                        p
483                    }
484                };
485                if skybox_path.exists() {
486                    renderer.load_skybox(&context.device, &context.queue, &skybox_path);
487                }
488            }
489
490            // Apply post-process config
491            if let Some(pp_def) = &self.scene_post_process {
492                renderer.set_post_process_config(
493                    super::scene_loading::post_process_config_from_def(pp_def),
494                );
495                renderer.ensure_kuwahara_resources(&context.device, &context.queue);
496                renderer.ensure_fxaa_resources(&context.device);
497            }
498        }
499
500        // Reload terrain
501        #[cfg(feature = "debug-hud")]
502        self.debug_panels.clear();
503        if let (Some(renderer), Some(context)) = (&mut self.scene_renderer, &self.render_context) {
504            let mut grass_info = None;
505            load_terrain_from_world_inner(
506                &self.world,
507                &self.scene_path,
508                &context.device,
509                &context.queue,
510                renderer,
511                &mut self.physics,
512                &mut self.terrain,
513                &mut grass_info,
514            );
515
516            // Create grass debug panel if grass was loaded
517            #[cfg(feature = "debug-hud")]
518            if let Some(info) = grass_info {
519                let panel = flint_debug_ui::GrassDebugPanel::new(
520                    info.config,
521                    std::path::PathBuf::from(&self.scene_path),
522                    info.terrain_entity_name,
523                );
524                self.debug_panels.push(Box::new(panel));
525            }
526            #[cfg(not(feature = "debug-hud"))]
527            let _ = grass_info;
528        }
529        #[cfg(feature = "debug-hud")]
530        {
531            self.create_ocean_debug_panel();
532            self.create_tod_debug_panel();
533            self.create_weather_debug_panel();
534            self.create_reality_debug_panel();
535            self.create_visitor_debug_panel();
536            self.create_dead_calm_debug_panel();
537            self.create_camera_debug_panel();
538        }
539        self.apply_camera_tuning();
540
541        // Update terrain height callback for scripts
542        self.update_terrain_height_fn();
543
544        // Re-initialize systems
545        self.physics
546            .initialize(&mut self.world)
547            .unwrap_or_else(|e| tracing::warn!("Physics init failed: {:?}", e));
548
549        load_audio_from_world(
550            &self.world,
551            &mut self.audio,
552            &self.scene_path,
553            self.scene_preload_audio,
554        );
555        self.audio
556            .initialize(&mut self.world)
557            .unwrap_or_else(|e| tracing::warn!("Audio init failed: {:?}", e));
558
559        // Scene-declared music session on the freshly initialized audio
560        // (before scripts, so a conducted context exists from their first
561        // frame — F4). Music→music transitions: the fresh gilrs above is
562        // simply dropped again by the new session's handoff.
563        self.start_music_session();
564
565        load_animations_from_world(&self.scene_path, &mut self.animation);
566        load_sprite_animations_from_world(&self.scene_path, &mut self.animation);
567        load_sequences_from_world(&self.scene_path, &mut self.animation);
568        self.animation
569            .initialize(&mut self.world)
570            .unwrap_or_else(|e| tracing::warn!("Animation init failed: {:?}", e));
571
572        self.particles
573            .initialize(&mut self.world)
574            .unwrap_or_else(|e| tracing::warn!("Particles init failed: {:?}", e));
575        self.load_particle_textures();
576
577        load_scripts_from_world(&self.scene_path, &mut self.script);
578        self.script
579            .initialize(&mut self.world)
580            .unwrap_or_else(|e| tracing::warn!("Script init failed: {:?}", e));
581
582        // Call on_scene_enter on new scripts
583        self.script.set_current_scene(&self.scene_path);
584        {
585            let _state_scope = self.script.state_scope(
586                &mut self.state_machine,
587                &mut self.persistent_store,
588                &self.physics,
589            );
590            self.script.call_scene_enters(&mut self.world);
591        }
592
593        // Recapture cursor if player exists
594        if self.physics.has_player_entity() && !self.cursor_captured {
595            self.capture_cursor();
596        }
597
598        println!("[transition] Scene loaded: {}", self.scene_path);
599    }
