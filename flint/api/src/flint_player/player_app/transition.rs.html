21/// Scene transition lifecycle phase
22#[derive(Debug, Clone)]
23pub(super) enum TransitionPhase {
24    /// Normal gameplay
25    Idle,
26    /// Playing exit transition — scripts draw fade-out visuals
27    Exiting { target_scene: String, elapsed: f32 },
28    /// Loading the new scene (synchronous, happens in one frame)
29    Loading { target_scene: String },
30    /// Playing enter transition — scripts draw fade-in visuals
31    Entering { elapsed: f32 },
32}
33
34impl TransitionPhase {
35    pub(super) fn is_idle(&self) -> bool {
36        matches!(self, TransitionPhase::Idle)
37    }
38}

41    /// Advance the transition phase based on elapsed time and script signals.
42    pub(super) fn advance_transition(&mut self) {
43        match &self.transition_phase {
44            TransitionPhase::Idle => {}
45            TransitionPhase::Exiting {
46                target_scene,
47                elapsed,
48            } => {
49                let new_elapsed = elapsed + self.clock.delta_time as f32;
50                let target = target_scene.clone();
51                self.transition_phase = TransitionPhase::Exiting {
52                    target_scene: target.clone(),
53                    elapsed: new_elapsed,
54                };
55                // Check if a complete_transition event was fired
56                // (We use a sentinel event name to signal completion)
57            }
58            TransitionPhase::Loading { target_scene } => {
59                let target = target_scene.clone();
60                self.execute_scene_transition(&target);
61                self.transition_phase = TransitionPhase::Entering { elapsed: 0.0 };
62            }
63            TransitionPhase::Entering { elapsed } => {
64                let new_elapsed = elapsed + self.clock.delta_time as f32;
65                self.transition_phase = TransitionPhase::Entering {
66                    elapsed: new_elapsed,
67                };
68            }
69        }
70    }

72    /// Load a chunk TOML file as additional entities with offset.
73    pub(super) fn load_chunk(&mut self, path: &str, offset_x: f32, offset_y: f32, chunk_id: &str) {
74        if self.loaded_chunks.contains_key(chunk_id) {
75            tracing::warn!("Chunk '{}' already loaded, skipping", chunk_id);
76            return;
77        }
78
79        // Resolve path relative to scene directory
80        let scene_dir = Path::new(&self.scene_path)
81            .parent()
82            .unwrap_or_else(|| Path::new("."));
83        let chunk_path = scene_dir.join(path);
84
85        // Parse chunk as a scene file
86        let content = match vfs::read_to_string(&chunk_path) {
87            Ok(c) => c,
88            Err(e) => {
89                tracing::error!("Failed to read chunk '{}': {}", chunk_path.display(), e);
90                return;
91            }
92        };
93        let scene_file: flint_scene::SceneFile = match toml::from_str(&content) {
94            Ok(s) => s,
95            Err(e) => {
96                tracing::error!("Failed to parse chunk '{}': {}", chunk_path.display(), e);
97                return;
98            }
99        };
100
101        // Load schema registry for archetype defaults
102        let registry = if self.schema_paths.is_empty() {
103            flint_schema::SchemaRegistry::load_from_directory("schemas")
104                .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
105        } else {
106            let existing: Vec<&str> = self
107                .schema_paths
108                .iter()
109                .map(|s| s.as_str())
110                .filter(|p| vfs::exists(p))
111                .collect();
112            if existing.is_empty() {
113                flint_schema::SchemaRegistry::new()
114            } else {
115                flint_schema::SchemaRegistry::load_from_directories(&existing)
116                    .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
117            }
118        };
119
120        let mut spawned_ids = Vec::new();
121
122        // Spawn entities with prefixed names
123        for (name, entity_def) in &scene_file.entities {
124            let prefixed_name = format!("{}_{}", chunk_id, name);
125            let id = match self.world.spawn(prefixed_name.clone()) {
126                Ok(id) => id,
127                Err(e) => {
128                    tracing::error!("Failed to spawn chunk entity '{}': {:?}", prefixed_name, e);
129                    continue;
130                }
131            };
132            spawned_ids.push(id);
133
134            // Set archetype + defaults
135            if let Some(archetype) = &entity_def.archetype {
136                if let Some(comps) = self.world.get_components_mut(id) {
137                    comps.archetype = Some(archetype.clone());
138                    if let Some(arch_schema) = registry.get_archetype(archetype) {
139                        for (comp_name, defaults) in &arch_schema.defaults {
140                            if !comps.has(comp_name) {
141                                comps.set(comp_name.clone(), defaults.clone());
142                            }
143                        }
144                    }
145                }
146            }
147
148            // Set component data
149            for (comp_name, comp_data) in &entity_def.components {
150                let _ = self.world.merge_component(id, comp_name, comp_data.clone());
151            }
152
153            // Apply position offset
154            if let Some(transform) = self.world.get_transform(id) {
155                let new_x = transform.position.x + offset_x;
156                let new_y = transform.position.y + offset_y;
157                let new_z = transform.position.z;
158                if let Some(comps) = self.world.get_components_mut(id) {
159                    comps.set_field(
160                        comp::TRANSFORM,
161                        "position",
162                        toml::Value::Array(vec![
163                            toml::Value::Float(new_x as f64),
164                            toml::Value::Float(new_y as f64),
165                            toml::Value::Float(new_z as f64),
166                        ]),
167                    );
168                }
169            } else {
170                // Entity has no transform yet — create one with the offset
171                if let Some(comps) = self.world.get_components_mut(id) {
172                    comps.set_field(
173                        comp::TRANSFORM,
174                        "position",
175                        toml::Value::Array(vec![
176                            toml::Value::Float(offset_x as f64),
177                            toml::Value::Float(offset_y as f64),
178                            toml::Value::Float(0.0),
179                        ]),
180                    );
181                }
182            }
183        }
184
185        // Re-sync physics for new entities
186        self.physics
187            .sync_new_entities(&mut self.world, &spawned_ids);
188
189        // Re-sync scripts for new entities
190        let chunk_scripts_dir = chunk_path
191            .parent()
192            .unwrap_or_else(|| Path::new("."))
193            .join("scripts");
194        for &id in &spawned_ids {
195            if let Some(comps) = self.world.get_components(id) {
196                if let Some(script_comp) = comps.get(comp::SCRIPT) {
197                    if let Some(source) = script_comp.get("source").and_then(|v| v.as_str()) {
198                        // Try chunk's own scripts dir first, then scene's scripts dir
199                        let scene_scripts_dir = Path::new(&self.scene_path)
200                            .parent()
201                            .unwrap_or_else(|| Path::new("."))
202                            .join("scripts");
203                        let script_path = if vfs::exists(chunk_scripts_dir.join(source)) {
204                            chunk_scripts_dir.join(source)
205                        } else {
206                            scene_scripts_dir.join(source)
207                        };
208                        if vfs::exists(&script_path) {
209                            self.script.load_script_for_entity(id, &script_path);
210                        }
211                    }
212                }
213            }
214        }
215
216        // Initialize scripts for new entities
217        self.script
218            .initialize_entities(&mut self.world, &spawned_ids);
219
220        println!(
221            "[chunk] Loaded '{}' ({} entities) at offset ({}, {})",
222            chunk_id,
223            spawned_ids.len(),
224            offset_x,
225            offset_y
226        );
227        self.loaded_chunks.insert(chunk_id.to_string(), spawned_ids);
228    }

230    /// Unload a previously loaded chunk by despawning all its entities.
231    pub(super) fn unload_chunk(&mut self, chunk_id: &str) {
232        if let Some(entity_ids) = self.loaded_chunks.remove(chunk_id) {
233            let count = entity_ids.len();
234            for id in &entity_ids {
235                self.physics.remove_entity(*id);
236                self.script.remove_entity(*id);
237                let _ = self.world.despawn(*id);
238            }
239            println!("[chunk] Unloaded '{}' ({} entities)", chunk_id, count);
240        } else {
241            tracing::warn!("Chunk '{}' not loaded, cannot unload", chunk_id);
242        }
243    }

245    /// Unload the current scene and load a new one.
246    pub(super) fn execute_scene_transition(&mut self, target_scene: &str) {
247        // Teardown: scripts told, music session stopped BEFORE audio.clear()
248        // (ADR 0017 producer→handles→device order), systems/world/terrain/
249        // transient render state cleared.
250        self.teardown_current_scene();
251
252        // The outgoing scene's on_scene_exit has run: persist whatever it
253        // (and the scene before it) stored, before the path changes.
254        self.flush_persistent_store(true, "scene transition");
255
256        // Load: resolve + schema discovery + scene parse; on failure the
257        // transition aborts here (world already cleared, same as before).
258        if !self.load_target_scene(target_scene) {
259            return;
260        }
261
262        // Bring-up: camera, models, terrain, system re-init (audio before
263        // the scene-declared music session, session before script init so
264        // the conducted context exists from the scripts' first frame),
265        // on_scene_enter, cursor.
266        self.bring_up_scene();
267    }

269    fn teardown_current_scene(&mut self) {
270        println!("[transition] Unloading current scene...");
271
272        // Call on_scene_exit on all scripts
273        {
274            let _state_scope = self.script.state_scope(
275                &mut self.state_machine,
276                &mut self.persistent_store,
277                &self.physics,
278            );
279            self.script.call_scene_exits(&mut self.world);
280        }
281
282        // Music session first: capture guard drops and stems stop BEFORE the
283        // audio system clears (ADR 0017 producer→handles→device order).
284        self.stop_music_session();
285
286        // Clear all systems
287        self.script.clear();
288        self.audio.clear();
289        self.physics.clear();
290        self.animation.clear();
291        self.particles.clear();
292        self.procgen_resolver.clear_queue();
293
294        // Clear world
295        self.world = FlintWorld::new();
296
297        // Clear terrain
298        self.terrain = None;
299        if let Some(renderer) = &mut self.scene_renderer {
300            renderer.clear_terrain();
301            renderer.unload_grass();
302        }
303
304        // Clear transient rendering state
305        self.skeletal_entity_assets.clear();
306        self.ui_textures.clear();
307        self.draw_commands.clear();
308        self.loaded_chunks.clear();
309    }

311    fn load_target_scene(&mut self, target_scene: &str) -> bool {
312        println!("[transition] Loading scene: {}", target_scene);
313
314        // Resolve scene path relative to current scene
315        let new_scene_path = resolve_scene_path(&self.scene_path, target_scene);
316
317        // Auto-discover schema dirs from the new scene path and merge
318        for dir in flint_schema::discover_schema_dirs(&new_scene_path) {
319            let s = dir.to_string_lossy().into_owned();
320            if !self.schema_paths.contains(&s) {
321                self.schema_paths.push(s);
322            }
323        }
324
325        // Load schema registry
326        let registry = if self.schema_paths.is_empty() {
327            // Try default schemas/ dir
328            flint_schema::SchemaRegistry::load_from_directory("schemas")
329                .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
330        } else {
331            let existing: Vec<&str> = self
332                .schema_paths
333                .iter()
334                .map(|s| s.as_str())
335                .filter(|p| vfs::exists(p))
336                .collect();
337            if existing.is_empty() {
338                flint_schema::SchemaRegistry::new()
339            } else {
340                flint_schema::SchemaRegistry::load_from_directories(&existing)
341                    .unwrap_or_else(|_| flint_schema::SchemaRegistry::new())
342            }
343        };
344
345        // Parse and load scene
346        match flint_scene::load_scene(&new_scene_path, &registry) {
347            Ok((world, scene_file)) => {
348                self.world = world;
349                self.scene_path = new_scene_path.clone();
350                // A slow-motion moment never outlives the scene that asked for it
351                self.clock.time_scale = 1.0;
352                self.skybox_path = scene_file
353                    .environment
354                    .as_ref()
355                    .and_then(|env| env.skybox.clone());
356                self.scene_ambient = scene_file.environment.as_ref().and_then(|env| {
357                    (env.ambient_sky.is_some() || env.ambient_ground.is_some()).then(|| {
358                        (
359                            env.ambient_sky
360                                .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_SKY),
361                            env.ambient_ground
362                                .unwrap_or(flint_render::LightUniforms::DEFAULT_AMBIENT_GROUND),
363                        )
364                    })
365                });
366                self.scene_diffuse_wrap = scene_file
367                    .environment
368                    .as_ref()
369                    .and_then(|env| env.diffuse_wrap);
370                self.scene_oren_nayar = scene_file
371                    .environment
372                    .as_ref()
373                    .and_then(|env| env.oren_nayar);
374                self.scene_sheen = scene_file.environment.as_ref().and_then(|env| {
375                    env.sheen_strength
376                        .map(|s| (env.sheen_color.unwrap_or([1.0; 3]), s))
377                });
378                self.scene_camera = scene_file.camera.clone();
379                super::scene_loading::load_user_styles(&new_scene_path);
380                self.scene_post_process = scene_file.post_process.clone();
381                self.scene_input_config = scene_file.scene.input_config.clone();
382                self.scene_preload_audio = scene_file.scene.preload_audio;
383            }
384            Err(e) => {
385                tracing::error!("Failed to load scene '{}': {:?}", new_scene_path, e);
386                return false;
387            }
388        }
389
390        // Rebuild component index after loading new scene
391        self.world.rebuild_component_index();
392
393        true
394    }

396    fn bring_up_scene(&mut self) {
397        // Apply camera config before borrowing renderer
398        self.apply_camera_def();
399
400        // Reload models
401        if let (Some(renderer), Some(context)) = (&mut self.scene_renderer, &self.render_context) {
402            renderer.scene_dir = Path::new(&self.scene_path)
403                .parent()
404                .map(|p| p.to_path_buf());
405
406            let config = build_model_load_config(
407                &self.scene_path,
408                &self.world,
409                self.catalog.as_ref(),
410                self.content_store.as_ref(),
411            );
412
413            // Procgen before the model loader: catalog > procgen > file (#32).
414            super::procgen_assets::resolve_scene(
415                &self.scene_path,
416                &self.world,
417                &mut self.procgen_resolver,
418                self.procgen_mode,
419                super::procgen_assets::GpuSink {
420                    renderer,
421                    device: &context.device,
422                    queue: &context.queue,
423                    config: &config,
424                },
425            );
426
427            let load_result = model_loader::load_models_from_world(
428                &mut self.world,
429                renderer,
430                &context.device,
431                &context.queue,
432                &config,
433            );
434            register_skeletal_data(&load_result, &mut self.animation);
435            register_node_animation_data(&load_result, &mut self.animation);
436            self.skeletal_entity_assets = load_result.skinned_entities;
437            renderer.update_from_world(&self.world, &context.device);
438
439            // Reload splines
440            crate::spline_gen::load_splines(
441                &self.scene_path,
442                &mut self.world,
443                renderer,
444                Some(&mut self.physics),
445                &context.device,
446                &context.queue,
447            );
448            renderer.update_from_world(&self.world, &context.device);
449            // The new scene's first frame has no past: styles that keep
450            // history (ASCII's temporal stability, #277) start over
451            renderer.reset_style_history();
452
453            // Scene-authored hemisphere ambient + wrap; reset first so the
454            // previous scene's values never leak into one that doesn't
455            // author them (reset clears both ambient and wrap)
456            renderer.reset_ambient();
457            if let Some((sky, ground)) = self.scene_ambient {
458                renderer.set_ambient(sky, ground);
459            }
460            if let Some(wrap) = self.scene_diffuse_wrap {
461                renderer.set_diffuse_wrap(wrap);
462            }
463            if let Some(oren) = self.scene_oren_nayar {
464                renderer.set_oren_nayar(oren);
465            }
466            if let Some((color, strength)) = self.scene_sheen {
467                renderer.set_sheen(color, strength);
468            }
469
470            // Reload skybox
471            if let Some(skybox_rel) = &self.skybox_path {
472                let scene_dir = Path::new(&self.scene_path)
473                    .parent()
474                    .unwrap_or_else(|| Path::new("."));
475                let skybox_path = {
476                    let p = scene_dir.join(skybox_rel);
477                    if vfs::exists(&p) {
478                        p
479                    } else if let Some(parent) = scene_dir.parent() {
480                        parent.join(skybox_rel)
481                    } else {
482                        p
483                    }
484                };
485                if vfs::exists(&skybox_path) {
486                    renderer.load_skybox(&context.device, &context.queue, &skybox_path);
487                }
488            }
489
490            // Apply post-process config
491            if let Some(pp_def) = &self.scene_post_process {
492                renderer.set_post_process_config(
493                    super::scene_loading::post_process_config_from_def(pp_def),
494                );
495                renderer.ensure_style_resources(&context.device);
496                renderer.ensure_fxaa_resources(&context.device);
497            }
498        }
499
500        // Reload terrain
501        // The scene panels go with the scene. Rendering & Effects (F4) and
502        // the game's own panels belong to the player and are kept (open or
503        // closed as they were); before #268 they were dropped here and
504        // never made again, so F4 went dead after the first transition.
505        #[cfg(feature = "debug-hud")]
506        self.drop_scene_debug_panels();
507        if let (Some(renderer), Some(context)) = (&mut self.scene_renderer, &self.render_context) {
508            let mut grass_info = None;
509            load_terrain_from_world_inner(
510                &self.world,
511                &self.scene_path,
512                &context.device,
513                &context.queue,
514                renderer,
515                &mut self.physics,
516                &mut self.terrain,
517                &mut grass_info,
518            );
519
520            // Create grass debug panel if grass was loaded
521            #[cfg(feature = "debug-hud")]
522            if let Some(info) = grass_info {
523                let panel = flint_debug_ui::GrassDebugPanel::new(
524                    info.config,
525                    std::path::PathBuf::from(&self.scene_path),
526                    info.terrain_entity_name,
527                )
528                .with_component(info.component);
529                self.debug_panels.push(Box::new(panel));
530            }
531            #[cfg(not(feature = "debug-hud"))]
532            let _ = grass_info;
533        }
534        #[cfg(feature = "debug-hud")]
535        {
536            self.create_ocean_debug_panel();
537            self.create_tod_debug_panel();
538            self.create_weather_debug_panel();
539            self.create_reality_debug_panel();
540            self.create_visitor_debug_panel();
541            self.create_dead_calm_debug_panel();
542            self.create_camera_debug_panel();
543            self.create_particles_debug_panel();
544            self.create_styles_debug_panel();
545            self.create_render_debug_panel();
546            self.create_extension_debug_panels();
547        }
548        self.apply_camera_tuning();
549
550        // Update terrain height callback for scripts
551        self.update_terrain_height_fn();
552
553        // Project fonts may differ when the new scene lives in another project
554        self.refresh_project_fonts();
555
556        // Re-initialize systems
557        self.physics
558            .initialize(&mut self.world)
559            .unwrap_or_else(|e| tracing::warn!("Physics init failed: {:?}", e));
560
561        load_audio_from_world(
562            &self.world,
563            &mut self.audio,
564            &self.scene_path,
565            self.scene_preload_audio,
566            // In the browser the audio may already be here at a scene change, and
567            // decoding it all in one frame is the stall #124 removed:
568            // queue it. The desktop decodes up front, as a load can afford.
569            cfg!(target_arch = "wasm32"),
570        );
571        self.audio
572            .initialize(&mut self.world)
573            .unwrap_or_else(|e| tracing::warn!("Audio init failed: {:?}", e));
574
575        // Scene-declared music session on the freshly initialized audio
576        // (before scripts, so a conducted context exists from their first
577        // frame — F4). Music→music transitions: the fresh gilrs above is
578        // simply dropped again by the new session's handoff.
579        self.start_music_session();
580
581        load_animations_from_world(&self.scene_path, &mut self.animation);
582        load_sprite_animations_from_world(&self.scene_path, &mut self.animation);
583        load_sequences_from_world(&self.scene_path, &mut self.animation);
584        self.animation
585            .initialize(&mut self.world)
586            .unwrap_or_else(|e| tracing::warn!("Animation init failed: {:?}", e));
587
588        flint_particles::load_particle_effects_from_world(&self.scene_path, &mut self.particles);
589        self.particles
590            .initialize(&mut self.world)
591            .unwrap_or_else(|e| tracing::warn!("Particles init failed: {:?}", e));
592        self.load_particle_textures();
593
594        load_scripts_from_world(&self.scene_path, &mut self.script);
595        self.script
596            .initialize(&mut self.world)
597            .unwrap_or_else(|e| tracing::warn!("Script init failed: {:?}", e));
598
599        // Call on_scene_enter on new scripts
600        self.script.set_current_scene(&self.scene_path);
601        {
602            let _state_scope = self.script.state_scope(
603                &mut self.state_machine,
604                &mut self.persistent_store,
605                &self.physics,
606            );
607            self.script.call_scene_enters(&mut self.world);
608        }
609
610        // Recapture cursor if player exists
611        if self.physics.has_player_entity() && !self.cursor_captured {
612            self.capture_cursor();
613        }
614
615        println!("[transition] Scene loaded: {}", self.scene_path);
616    }
