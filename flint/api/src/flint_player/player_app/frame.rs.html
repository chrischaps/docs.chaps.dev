31        // On Android, window may be None between suspended() and resumed()
32        if self.window.is_none() {
33            return;
34        }
35        let Some(context) = &self.render_context else {
36            return;
37        };
38        let Some(renderer) = &mut self.scene_renderer else {
39            return;
40        };

42        let output = match context.surface.get_current_texture() {
43            Ok(output) => output,
44            Err(wgpu::SurfaceError::Lost | wgpu::SurfaceError::Outdated) => {

102            }
103            if let Some(fd) = self.pp_fog_density_override {

491    pub(super) fn tick(&mut self) {
492        // Music session drain + guide/timeline panel feed (F3, ADR 0018).
493        self.tick_music_session();
494
495        // Advance game clock
496        self.clock.tick();
497
498        // Advance transition phase timing
499        self.advance_transition();
500
501        // Read active state config to decide which systems run
502        let config = self.state_machine.active_config().clone();
503
504        let has_fps_player = self.physics.has_player_entity();
505
506        // Fixed-timestep physics + FPS camera/listener follow
507        self.tick_fixed_physics(&config, has_fps_player);
508
509        // Physics + input events — scripts and audio both consume them
510        let game_events = self.collect_game_events();
511
512        // Set state machine + persistent store + physics pointers for script
513        // access. The RAII guard must outlive every script call this frame —
514        // including the sprite-anim-end callbacks inside tick_av_systems —
515        // so it lives here, in tick's own scope, not in a sub-step.
516        let _state_scope = self.script.state_scope(
517            &mut self.state_machine,
518            &mut self.persistent_store,
519            &self.physics,
520        );
521        self.script.set_current_scene(&self.scene_path);
522
523        // Script context (transition state, camera, input/events, conducted
524        // parameters — session tick above precedes set_conducted, which
525        // precedes provide_context/on_update: F4/ADR 0020 ordering), then
526        // on_update.
527        self.run_scripts(&config, &game_events);
528
529        // Script camera overrides (chase cam etc.) + roll basis + listener
530        self.apply_script_camera_overrides(has_fps_player);
531
532        // on_draw_ui (always runs), script commands, draw-command collection
533        self.drive_script_ui();
534
535        // Audio triggers, animations + sprite-end callbacks, bone matrices,
536        // bone probes, particles
537        self.tick_av_systems(&config, &game_events);
538
539        // Frame-budgeted procgen queue
540        self.tick_procgen();
541
542        // Renderer sync: transforms, particle upload, grass/ocean time
543        self.sync_renderer();
544
545        // Script post-override drain + music-session ladder merge (ADR 0021)
546        self.drain_post_overrides();
547
548        // Script audio/cursor frame overrides, then per-frame input clear
549        self.finish_frame();
550    }

552    fn finish_frame(&mut self) {
553        // Apply audio low-pass filter override from scripts
554        if let Some(cutoff) = self.script.take_audio_overrides() {
555            self.audio.set_filter_cutoff(cutoff);
556        }
557
558        // Apply cursor capture request from scripts (skipped while a debug
559        // panel is open — the panel needs the mouse).
560        if let Some(capture) = self.script.take_cursor_capture_override() {
561            #[cfg(feature = "debug-hud")]
562            let panel_open = self.debug_panels.iter().any(|p| p.is_open());
563            #[cfg(not(feature = "debug-hud"))]
564            let panel_open = false;
565            if capture && !panel_open && !self.cursor_captured {
566                self.capture_cursor();
567            } else if !capture && self.cursor_captured {
568                self.release_cursor();
569            }
570        }
571
572        // Clear per-frame input state
573        self.input.end_frame();
574    }

576    fn tick_music_session(&mut self) {
577        // Music session (F3, ADR 0018): while a session is active the capture
578        // thread owns the pad — its drain replaces the player's own polling,
579        // feeding the Judge at full precision and InputState down-sampled.
580        let session_finished = if let Some(ms) = &mut self.music_session {
581            ms.tick(&mut self.input)
582        } else {
583            self.poll_gamepad_events();
584            false
585        };
586        if session_finished {
587            // Natural mid-scene finish: gilrs back, the scene keeps running —
588            // unless the component asked for `quit_on_finish`.
589            let quit = self
590                .music_session
591                .as_ref()
592                .is_some_and(music_session::MusicSession::quit_on_finish);
593            self.stop_music_session();
594            if quit {
595                println!("[music] quit_on_finish — exiting player");
596                self.music_exit_requested = true;
597            }
598        }
599
600        // Feed the Music Guide + Manifest Map panels — only while summoned,
601        // so a closed panel costs nothing (history accumulates in the core
602        // regardless).
603        #[cfg(feature = "debug-hud")]
604        if let Some(ms) = &self.music_session {
605            for panel in &mut self.debug_panels {
606                if !panel.is_open() {
607                    continue;
608                }

610                    if let Some(p) = panel
611                        .as_any_mut()
612                        .downcast_mut::<music_guide_panel::MusicGuidePanel>()
613                    {
614                        p.set_data(
615                            ms.visual_frame(),
616                            ms.guide_frame(music_guide_panel::GUIDE_HORIZON_BEATS),
617                        );
618                    }
619                } else if panel.name() == timeline_panel::MANIFEST_MAP_PANEL {
620                    if let Some(p) = panel
621                        .as_any_mut()
622                        .downcast_mut::<timeline_panel::ManifestMapPanel>()
623                    {
624                        p.set_frame(ms.timeline_frame());
625                    }
626                }

631    fn tick_fixed_physics(&mut self, config: &StateConfig, has_fps_player: bool) {
632        // Fixed-timestep physics loop (skip when paused, but still consume steps to avoid spiral)
633        while self.clock.should_fixed_update() {
634            let dt = self.clock.fixed_timestep;
635
636            if config.physics == SystemPolicy::Run {
637                if has_fps_player {
638                    self.physics
639                        .update_character(&self.input, &mut self.world, dt);
640                }
641                self.physics
642                    .fixed_update(&mut self.world, dt)
643                    .unwrap_or_else(|e| tracing::warn!("Physics error: {:?}", e));
644            }
645
646            self.clock.consume_fixed_step();
647        }
648
649        // Update camera from player character position (FPS mode)
650        if has_fps_player {
651            let cam_pos = self.physics.camera_position(&self.world);
652            let cam_target = self.physics.camera_target(cam_pos);
653            self.camera.update_first_person(
654                cam_pos,
655                self.physics.camera_yaw(),
656                self.physics.camera_pitch(),
657            );
658            self.camera.target = cam_target;
659
660            if config.audio == SystemPolicy::Run {
661                self.audio.update_listener(
662                    cam_pos,
663                    self.physics.camera_yaw(),
664                    self.physics.camera_pitch(),
665                );
666            }
667        }
668    }

683    fn run_scripts(&mut self, config: &StateConfig, game_events: &Vec<flint_runtime::GameEvent>) {
684        // Set transition state for script access
685        match &self.transition_phase {
686            TransitionPhase::Idle => {
687                self.script.set_transition_state(-1.0, "idle");
688            }
689            TransitionPhase::Exiting { elapsed, .. } => {
690                self.script.set_transition_state(*elapsed as f64, "exiting");
691            }
692            TransitionPhase::Loading { .. } => {
693                self.script.set_transition_state(1.0, "loading");
694            }
695            TransitionPhase::Entering { elapsed } => {
696                self.script
697                    .set_transition_state(*elapsed as f64, "entering");
698            }
699        }
700
701        // Script system: provide camera context, then run updates
702        self.script
703            .set_camera(self.camera.position_array(), self.camera.forward_vector());
704        self.script.provide_context(
705            &self.input,
706            &game_events,
707            self.clock.total_time,
708            self.clock.delta_time,
709        );
710        let screen_rect = self.egui_ctx.screen_rect();
711        self.script
712            .set_screen_size(screen_rect.width(), screen_rect.height());
713
714        // Sync loaded chunk IDs so scripts can query is_chunk_loaded()
715        let chunk_ids: HashSet<String> = self.loaded_chunks.keys().cloned().collect();
716        self.script.set_loaded_chunk_ids(chunk_ids);
717
718        // Conducted parameters (F4, ADR 0020): the music session's per-frame
719        // state for scene bindings; neutral defaults when no session (also
720        // resets the frame after a session ends).
721        self.script.set_conducted(
722            self.music_session
723                .as_ref()
724                .map(|ms| ms.conducted_snapshot())
725                .unwrap_or_default(),
726        );
727
728        // Only run on_update when scripts are not paused
729        if config.scripts == SystemPolicy::Run {
730            self.script
731                .update(&mut self.world, self.clock.delta_time)
732                .unwrap_or_else(|e| tracing::warn!("Script error: {:?}", e));
733        }
734    }

736    fn apply_script_camera_overrides(&mut self, has_fps_player: bool) {
737        // Apply script camera overrides (for non-FPS camera modes like chase camera)
738        let cam = self.script.take_camera_overrides();
739        if let Some(pos) = cam.position {
740            self.camera.position = flint_core::Vec3::new(pos[0], pos[1], pos[2]);
741        }
742        if let Some(target) = cam.target {
743            self.camera.target = flint_core::Vec3::new(target[0], target[1], target[2]);
744        }
745        if let Some(fov) = cam.fov {
746            self.camera.fov = fov;
747        }
748        if let Some(ortho) = cam.orthographic {
749            self.camera.orthographic = ortho;
750        }
751        if let Some(height) = cam.ortho_height {
752            self.camera.ortho_height = height;
753        }
754        // Roll rebuilds the up vector from the view basis each frame it is set;
755        // when absent the up vector must reset to world up (overrides are
756        // one-frame take()s — without the reset, roll would stick after a
757        // script stops setting it, e.g. hot-reload into a compile error).
758        match cam.roll {
759            Some(roll) => {
760                let forward = flint_core::Vec3::new(
761                    self.camera.target.x - self.camera.position.x,
762                    self.camera.target.y - self.camera.position.y,
763                    self.camera.target.z - self.camera.position.z,
764                )
765                .normalized();
766                let right = forward.cross(&flint_core::Vec3::UP);
767                let right_len = (right.dot(&right)).sqrt();
768                if right_len > 1e-4 {
769                    let right = FlintVec3::new(
770                        right.x / right_len,
771                        right.y / right_len,
772                        right.z / right_len,
773                    );
774                    let base_up = right.cross(&forward);
775                    let (sin_r, cos_r) = roll.sin_cos();
776                    self.camera.up = FlintVec3::new(
777                        base_up.x * cos_r + right.x * sin_r,
778                        base_up.y * cos_r + right.y * sin_r,
779                        base_up.z * cos_r + right.z * sin_r,
780                    );
781                }
782                // Degenerate (looking straight up/down): keep the previous up.
783            }
784            None => {
785                self.camera.up = flint_core::Vec3::UP;
786            }
787        }
788
789        // Update audio listener for script-driven cameras (chase cam, etc.)
790        if !has_fps_player && cam.position.is_some() {
791            let cam_pos = self.camera.position;
792            let dir = flint_core::Vec3::new(
793                self.camera.target.x - cam_pos.x,
794                self.camera.target.y - cam_pos.y,
795                self.camera.target.z - cam_pos.z,
796            );
797            let yaw = dir.x.atan2(dir.z);
798            let horiz = (dir.x * dir.x + dir.z * dir.z).sqrt();
799            let pitch = (-dir.y).atan2(horiz);
800            self.audio.update_listener(cam_pos, yaw, pitch);
801        }
802    }

821    fn tick_av_systems(
822        &mut self,
823        config: &StateConfig,
824        game_events: &Vec<flint_runtime::GameEvent>,
825    ) {
826        // Audio triggers from game events (skip when paused)
827        if config.audio == SystemPolicy::Run {
828            self.audio.process_events(&game_events, &self.world);
829            self.audio
830                .update(&mut self.world, self.clock.delta_time)
831                .ok();
832        }
833
834        // Advance animations (skip when paused)
835        if config.animation == SystemPolicy::Run {
836            self.animation
837                .update(&mut self.world, self.clock.delta_time)
838                .ok();
839
840            // Deliver sprite animation end events to scripts
841            let sprite_events = self.animation.drain_sprite_events();
842            if !sprite_events.is_empty() {
843                self.script
844                    .call_sprite_anim_ends(&mut self.world, &sprite_events);
845            }
846
847            // Deliver animation sequence cues to the owning entity's script
848            let cues = self.animation.drain_sequence_cues();
849            if !cues.is_empty() {
850                for cue in &cues {
851                    println!(
852                        "[sequence] {:?} '{}' cue '{}' at {:.2}s",
853                        cue.entity_id, cue.sequence, cue.cue, cue.time
854                    );
855                }
856                self.script.call_sequence_cues(&mut self.world, &cues);
857            }
858        }
859
860        // Push skeletal bone matrices to GPU
861        if let (Some(renderer), Some(context)) = (&mut self.scene_renderer, &self.render_context) {
862            for (entity_id, asset_name) in &self.skeletal_entity_assets {
863                if let Some(matrices) = self.animation.bone_matrices(entity_id) {
864                    renderer.update_bone_matrices(
865                        &context.device,
866                        &context.queue,
867                        *entity_id,
868                        asset_name,
869                        matrices,
870                    );
871                }
872            }
873        }
874
875        // Publish bone_probe joints: model-local joint positions written
876        // into the component right after the pose computation, so scripts
877        // read this frame's pose (e.g. seat_camera following the eye).
878        {
879            let probe_ids: Vec<flint_core::EntityId> = self
880                .world
881                .entities_with_component(flint_core::components::BONE_PROBE)
882                .iter()
883                .copied()
884                .collect();
885            for id in probe_ids {
886                let joint = self
887                    .world
888                    .get_components(id)
889                    .and_then(|c| c.get_field(flint_core::components::BONE_PROBE, "joint"))
890                    .and_then(|v| v.as_str())
891                    .map(String::from);
892                if let Some(joint) = joint {
893                    if let Some(pos) = self.animation.joint_position(&id, &joint) {
894                        for (field, value) in [("x", pos[0]), ("y", pos[1]), ("z", pos[2])] {
895                            let _ = self.world.set_field(
896                                id,
897                                flint_core::components::BONE_PROBE,
898                                field,
899                                toml::Value::Float(value as f64),
900                            );
901                        }
902                    }
903                }
904            }
905        }
906
907        // Advance particle simulation (skip when paused)
908        if config.particles == SystemPolicy::Run {
909            self.particles
910                .update(&mut self.world, self.clock.delta_time)
911                .ok();
912        }
913    }

982    fn drain_post_overrides(&mut self) {
983        // Drain script post-processing overrides for this frame
984        let pp = self.script.take_postprocess_overrides();
985        self.pp_vignette_override = pp.vignette;
986        self.pp_bloom_override = pp.bloom;
987        self.pp_exposure_override = pp.exposure;
988        self.pp_chromatic_aberration_override = pp.chromatic_aberration;
989        self.pp_radial_blur_override = pp.radial_blur;
990        self.pp_ssao_intensity_override = pp.ssao_intensity;
991        self.pp_fog_density_override = pp.fog_density;
992        self.pp_desaturation_override = pp.desaturation;
993        // DoF is script-owned (ADR 0027) — never touched by the ladder merge
994        // below; there is no restore machinery because owning scripts write it
995        // every frame and scene loads reset the sticky config from the def.
996        self.pp_dof_strength_override = pp.dof_strength;
997        self.pp_dof_focus_distance_override = pp.dof_focus_distance;
998        self.pp_dof_focus_range_override = pp.dof_focus_range;

1001        self.pp_render_mode_override = pp_mode;
1002        self.pp_mode_params_override = pp_mode_params;
1003
1004        // Music-session ladder/reintegration visuals (ADR 0021). Script
1005        // overrides win — F4/F6 script-authored rung visuals supersede this
1006        // direct path with no code change and no double application. The
1007        // merge writes base + rung every frame so a recovered ladder actively
1008        // restores the scene's authored post config (the renderer config is
1009        // sticky); after teardown a one-shot restore does the same.
1010        if let Some(ms) = &self.music_session {
1011            let vf = ms.visual_frame();
1012            music_session::merge_ladder_postprocess(
1013                &mut self.pp_radial_blur_override,
1014                &mut self.pp_chromatic_aberration_override,
1015                &mut self.pp_desaturation_override,
1016                &vf,
1017                self.music_pp_base.unwrap_or_default(),
1018            );
1019        } else if let Some(base) = self.music_pp_restore.take() {
1020            music_session::restore_ladder_postprocess(
1021                &mut self.pp_radial_blur_override,
1022                &mut self.pp_chromatic_aberration_override,
1023                &mut self.pp_desaturation_override,
1024                base,
1025            );
1026        }
1027    }

1029    fn render_hud(&mut self, target_view: &wgpu::TextureView) {
1030        // Lazy-load any sprite textures referenced by draw commands
1031        // (must happen before egui_winit borrow)
1032        self.load_pending_sprites();
1033
1034        let Some(window) = &self.window else { return };
1035        let Some(context) = &self.render_context else {
1036            return;
1037        };
1038        let Some(egui_winit) = &mut self.egui_winit else {
1039            return;
1040        };
1041
1042        let raw_input = egui_winit.take_egui_input(window);
1043
1044        let draw_commands = std::mem::take(&mut self.draw_commands);
1045        #[cfg(feature = "debug-hud")]
1046        let mut debug_panels = std::mem::take(&mut self.debug_panels);
1047        let ui_textures = &self.ui_textures;
1048
1049        let show_stats = self.show_stats;
1050        let stats_data = if show_stats {
1051            self.scene_renderer.as_ref().map(|r| {
1052                let mut stats = r.collect_stats();
1053                // FPS smoothing: rolling window of delta_time samples
1054                self.stats_frame_times.push_back(self.clock.delta_time);
1055                while self.stats_frame_times.len() > 60 {
1056                    self.stats_frame_times.pop_front();
1057                }
1058                let avg_dt: f64 = self.stats_frame_times.iter().sum::<f64>()
1059                    / self.stats_frame_times.len().max(1) as f64;
1060                stats.fps = (1.0 / avg_dt) as f32;
1061                stats.frame_time_ms = (avg_dt * 1000.0) as f32;
1062                if let Some(ctx) = &self.render_context {
1063                    stats.resolution = [ctx.config.width, ctx.config.height];
1064                }
1065                stats
1066            })
1067        } else {
1068            None
1069        };
1070
1071        let full_output = self.egui_ctx.run(raw_input, |ctx| {
1072            // Script UI first: it shares the panels' background layer, so
1073            // issuing it before the panels puts the panels in front of it
1074            // (title card, HUD) — see render_draw_commands.
1075            render_draw_commands(ctx, &draw_commands, ui_textures);
1076
1077            #[cfg(feature = "debug-hud")]
1078            {
1079                // Full-width bottom strips first (e.g. the timeline panel);
1080                // the panel draws its own compact labels (a heading would
1081                // eat the vertical budget).

1084                        && matches!(panel.layout(), flint_debug_ui::PanelLayout::Bottom)
1085                    {
1086                        let panel_name = panel.name().to_owned();

1088                            .exact_height(112.0)
1089                            .show(ctx, |ui| {
1090                                panel.ui(ui);
1091                            });
1092                    }
1093                }
1094
1095                // Open side-panel indices, in creation order.
1096                let open: Vec<usize> = debug_panels
1097                    .iter()
1098                    .enumerate()
1099                    .filter(|(_, p)| {
1100                        p.is_open() && matches!(p.layout(), flint_debug_ui::PanelLayout::SideRight)
1101                    })
1102                    .map(|(i, _)| i)
1103                    .collect();
1104
1105                if !open.is_empty() {
1106                    // Logical points (screen_rect), never config.width (physical px).
1107                    let screen_w = ctx.screen_rect().width();
1108                    // Panels take at most ~half the window; fewer columns when narrow.
1109                    let max_cols = (((screen_w * 0.5) / 300.0).floor() as usize).clamp(1, 3);
1110                    let names: Vec<String> = open
1111                        .iter()
1112                        .map(|&i| debug_panels[i].name().to_owned())
1113                        .collect();
1114                    let name_refs: Vec<&str> = names.iter().map(String::as_str).collect();
1115                    let columns = flint_debug_ui::assign_columns(&name_refs, max_cols);
1116
1117                    for (col_idx, col) in columns.iter().enumerate() {
1118                        egui::SidePanel::right(egui::Id::new(("debug_col", col_idx)))
1119                            .default_width(300.0)
1120                            .min_width(280.0)
1121                            .show(ctx, |ui| {
1122                                egui::ScrollArea::vertical()
1123                                    .auto_shrink([false, false])
1124                                    .show(ui, |ui| {
1125                                        for &slot in col {
1126                                            let panel = &mut debug_panels[open[slot]];
1127                                            let name = panel.name().to_owned();
1128                                            egui::CollapsingHeader::new(
1129                                                egui::RichText::new(name).heading(),
1130                                            )
1131                                            .default_open(true)
1132                                            .show(ui, |ui| panel.ui(ui));
1133                                            ui.separator();
1134                                        }
1135                                    });
1136                            });
1137                    }
1138                }
1139            }
1140            if let Some(ref stats) = stats_data {
1141                render_stats_overlay(ctx, stats);
1142            }
1143        });
1144
1145        self.draw_commands = draw_commands;
1146        #[cfg(feature = "debug-hud")]
1147        {
1148            self.debug_panels = debug_panels;
1149        }
1150
1151        egui_winit.handle_platform_output(window, full_output.platform_output);
1152
1153        let paint_jobs = self
1154            .egui_ctx
1155            .tessellate(full_output.shapes, full_output.pixels_per_point);
1156
1157        let screen_descriptor = egui_wgpu::ScreenDescriptor {
1158            size_in_pixels: [context.config.width, context.config.height],
1159            pixels_per_point: full_output.pixels_per_point,
1160        };
1161
1162        let mut egui_renderer = self.egui_renderer.take().unwrap();
1163
1164        let mut encoder = context
1165            .device
1166            .create_command_encoder(&wgpu::CommandEncoderDescriptor {
1167                label: Some("HUD Encoder"),
1168            });
1169
1170        for (id, image_delta) in &full_output.textures_delta.set {
1171            egui_renderer.update_texture(&context.device, &context.queue, *id, image_delta);
1172        }
1173
1174        egui_renderer.update_buffers(
1175            &context.device,
1176            &context.queue,
1177            &mut encoder,
1178            &paint_jobs,
1179            &screen_descriptor,
1180        );
1181
1182        {
1183            let render_pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
1184                label: Some("HUD Render Pass"),
1185                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
1186                    view: target_view,
1187                    resolve_target: None,
1188                    ops: wgpu::Operations {
1189                        load: wgpu::LoadOp::Load, // overlay on top of 3D
1190                        store: wgpu::StoreOp::Store,
1191                    },
1192                })],
1193                depth_stencil_attachment: None,
1194                timestamp_writes: None,
1195                occlusion_query_set: None,
1196            });
1197
1198            let mut render_pass = render_pass.forget_lifetime();
1199            egui_renderer.render(&mut render_pass, &paint_jobs, &screen_descriptor);
1200        }
1201
1202        context.queue.submit(std::iter::once(encoder.finish()));
1203
1204        for id in &full_output.textures_delta.free {
1205            egui_renderer.free_texture(id);
1206        }
1207
1208        self.egui_renderer = Some(egui_renderer);
1209    }
1210}
