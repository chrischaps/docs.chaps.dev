38/// Tracks Android device IDs that produce gamepad button keycodes,
39/// assigning each a stable u32 slot for InputState's gamepad API.
40#[cfg(target_os = "android")]
41pub(super) struct AndroidGamepadTracker {
42    known_device_ids: std::collections::HashSet<winit::event::DeviceId>,
43    device_slots: HashMap<winit::event::DeviceId, u32>,
44    next_slot: u32,
45}
46
47#[cfg(target_os = "android")]
48impl AndroidGamepadTracker {
49    pub(super) fn new() -> Self {
50        Self {
51            known_device_ids: std::collections::HashSet::new(),
52            device_slots: HashMap::new(),
53            next_slot: 0,
54        }
55    }

57    /// Register a device as a gamepad and return its slot index.
58    /// True once any device has been registered as a gamepad.
59    #[cfg(target_os = "android")]
60    fn any_registered(&self) -> bool {
61        !self.device_slots.is_empty()
62    }
63
64    fn register_device(&mut self, device_id: winit::event::DeviceId) -> u32 {
65        self.known_device_ids.insert(device_id);
66        *self.device_slots.entry(device_id).or_insert_with(|| {
67            let slot = self.next_slot;
68            self.next_slot += 1;
69            slot
70        })
71    }
72
73    fn is_gamepad(&self, device_id: winit::event::DeviceId) -> bool {
74        self.known_device_ids.contains(&device_id)
75    }
76
77    fn slot_for(&self, device_id: winit::event::DeviceId) -> Option<u32> {
78        self.device_slots.get(&device_id).copied()
79    }
80}

82/// Pixels of a touchpad's smooth scroll that make one wheel line, the
83/// figure egui-winit uses too.
84const PIXELS_PER_LINE: f32 = 50.0;
85
86/// A wheel event in lines (notches): up and right are positive, as winit
87/// reports them. A touchpad's pixel scroll is scaled to lines, so wheel
88/// bindings and `mouse_wheel_*()` feel alike on either.
89pub(super) fn wheel_lines(delta: MouseScrollDelta) -> (f32, f32) {
90    match delta {
91        MouseScrollDelta::LineDelta(x, y) => (x, y),
92        MouseScrollDelta::PixelDelta(p) => {
93            (p.x as f32 / PIXELS_PER_LINE, p.y as f32 / PIXELS_PER_LINE)
94        }
95    }
96}

128    pub(super) fn poll_gamepad_events(&mut self) {
129        let mut events = Vec::new();
130        if let Some(gilrs) = &mut self.gilrs {
131            while let Some(event) = gilrs.next_event() {
132                events.push(event);
133            }
134        }
135
136        // gamepad_connected(): gilrs knows on desktop; on Android (no gilrs
137        // backend) a pad exists once any gamepad-shaped event has been seen.
138        #[cfg(not(target_os = "android"))]
139        {
140            let connected = self
141                .gilrs
142                .as_ref()
143                .map(|g| g.gamepads().next().is_some())
144                .unwrap_or(false);
145            self.input.set_gamepad_connected(connected);
146        }
147        #[cfg(target_os = "android")]
148        {
149            if self.android_gamepad.any_registered() || !events.is_empty() {
150                self.input.set_gamepad_connected(true);
151            }
152        }
153
154        for event in events {
155            let gamepad = gamepad_id_to_u32(event.id);
156            self.real_pad_slots.insert(gamepad);
157            // A real pad is ignored while the window is unfocused (ADR 0110),
158            // and by a capture, which hears its script's pad alone.
159            let deaf = !self.window_focused || self.capture.is_capture();
160            if deaf && event.event != EventType::Disconnected {
161                continue;
162            }
163            match event.event {
164                EventType::ButtonPressed(button, _) => {
165                    let name = format!("{button:?}");
166                    if self.try_capture_rebind(Binding::GamepadButton {
167                        button: name.clone(),
168                        gamepad: flint_runtime::GamepadSelector::Any,
169                    }) {
170                        continue;
171                    }
172                    self.input.process_gamepad_button_down(gamepad, name);
173                }
174                EventType::ButtonReleased(button, _) => {
175                    let name = format!("{button:?}");
176                    self.input.process_gamepad_button_up(gamepad, name);
177                }
178                EventType::ButtonChanged(button, value, _) => {
179                    let name = format!("{button:?}");
180                    self.input
181                        .process_gamepad_button_changed(gamepad, name, value);
182                }
183                EventType::AxisChanged(axis, value, _) => {
184                    let name = format!("{axis:?}");
185                    if self.pending_rebind.is_some() && value.abs() >= 0.45 {
186                        let direction = if value < 0.0 {
187                            Some(flint_runtime::AxisDirection::Negative)
188                        } else {
189                            Some(flint_runtime::AxisDirection::Positive)
190                        };
191                        if self.try_capture_rebind(Binding::GamepadAxis {
192                            axis: name.clone(),
193                            gamepad: flint_runtime::GamepadSelector::Any,
194                            deadzone: 0.15,
195                            scale: 1.0,
196                            invert: false,
197                            threshold: Some(0.35),
198                            direction,
199                        }) {
200                            continue;
201                        }
202                    }
203                    self.input.process_gamepad_axis(gamepad, name, value);
204                }
205                EventType::Disconnected => {
206                    self.input.clear_gamepad(gamepad);
207                }
208                _ => {}
209            }
210        }
211
212        // Android: poll trigger/right-stick axes from JNI bridge
213        #[cfg(target_os = "android")]
214        if let Some(reader) = self.android_axis_reader {
215            let axes = reader();
216            let [lt, rt, rs_x, rs_y] = axes;
217            // Use slot 0 — the JNI bridge doesn't distinguish multiple gamepads
218            let slot = 0u32;
219            // Feed triggers as LeftZ/RightZ to match existing input.toml bindings
220            // (accelerate = gamepad_axis "RightZ", brake = gamepad_axis "LeftZ")
221            self.input.process_gamepad_axis(slot, "LeftZ", lt);
222            self.input.process_gamepad_axis(slot, "RightZ", rt);
223            // Right stick axes for future use
224            self.input.process_gamepad_axis(slot, "RightStickX", rs_x);
225            self.input.process_gamepad_axis(slot, "RightStickY", rs_y);
226        }
227    }

229    fn try_capture_rebind(&mut self, binding: Binding) -> bool {
230        let Some(pending) = self.pending_rebind.take() else {
231            return false;
232        };
233
234        if let Err(e) = self
235            .input
236            .rebind_action(&pending.action, binding, pending.mode)
237        {
238            tracing::warn!("Failed to rebind action '{}': {:?}", pending.action, e);
239            return true;
240        }
241
242        if let Some(action_cfg) = self.input.action_config(&pending.action) {
243            self.user_override_config
244                .actions
245                .insert(pending.action.clone(), action_cfg);
246        }
247        if self.user_override_config.game_id.trim().is_empty() {
248            self.user_override_config.game_id = self.input.config().game_id.clone();
249        }
250
251        if let Err(e) = self.persist_user_overrides() {
252            tracing::warn!("Failed to save input overrides: {e:#}");
253        }
254
255        true
256    }

258    fn persist_user_overrides(&mut self) -> Result<()> {
259        let Some(paths) = &mut self.input_config_paths else {
260            return Ok(());
261        };
262
263        let scene_path = Path::new(&self.scene_path);
264        let game_id = &self.input.config().game_id;
265        let mut target =
266            user_override_save_path(scene_path, game_id, paths.user_override.as_deref());
267
268        if let Err(err) = write_user_override_file(&target, &self.user_override_config) {
269            let fallback = fallback_user_override_path(scene_path, game_id);
270            if fallback != target {
271                write_user_override_file(&fallback, &self.user_override_config)?;
272                target = fallback;
273            } else {
274                return Err(err);
275            }
276        }
277
278        paths.user_override = Some(target);
279        Ok(())
280    }

282    /// Capture the cursor for mouse-look (ADR 0110). While it's captured the
283    /// cursor reads as the window centre, its moves reach no one, and its
284    /// buttons and wheel reach the game but not egui. Without a window
285    /// (headless tests) only the state changes.
286    pub(super) fn capture_cursor(&mut self) {
287        // Fingers on the on-screen controls need the cursor free (#133).
288        if self.input.touch_controls_visible() {
289            return;
290        }
291        if let Some(window) = &self.window {
292            // A browser locks the pointer only from a gesture: the lock is
293            // asked for once a frame while one is live (`sync_pointer_lock`).
294            #[cfg(target_arch = "wasm32")]
295            window.set_cursor_visible(false);
296            #[cfg(not(any(target_os = "android", target_arch = "wasm32")))]
297            {
298                // Try confined first, then locked
299                let _ = window
300                    .set_cursor_grab(CursorGrabMode::Confined)
301                    .or_else(|_| window.set_cursor_grab(CursorGrabMode::Locked));
302                window.set_cursor_visible(false);
303            }
304        }
305        self.cursor_captured = true;
306        self.centre_the_cursor();
307        // egui stops hearing the pointer, so it mustn't think it's still
308        // hovering whatever it was over.
309        if let Some(egui_winit) = &mut self.egui_winit {
310            egui_winit.egui_input_mut().events.push(egui::Event::PointerGone);
311        }
312    }

314    pub(super) fn release_cursor(&mut self) {
315        if let Some(window) = &self.window {
316            // On the web the lock is let go of by the page, not winit: winit
317            // calls `exitPointerLock` without asking whether the browser has
318            // one, and iOS WebKit hasn't, so the first finger's release threw
319            // through the event loop and froze the game (Float#3).
320            #[cfg(target_arch = "wasm32")]
321            super::web_page::exit_lock();
322            #[cfg(not(target_os = "android"))]
323            {
324                #[cfg(not(target_arch = "wasm32"))]
325                let _ = window.set_cursor_grab(CursorGrabMode::None);
326                window.set_cursor_visible(true);
327                // The cursor comes back where it read while captured, so
328                // nothing jumps under it.
329                let (w, h) = self.input.screen_size();
330                let _ = window.set_cursor_position(winit::dpi::PhysicalPosition::new(
331                    w / 2.0,
332                    h / 2.0,
333                ));
334            }
335        }
336        self.cursor_captured = false;
337    }

373    /// The window gained or lost focus. Losing it releases every real pad,
374    /// and their events are dropped until it's back; then the sticks are
375    /// read again where they stand. Buttons need a fresh press. The input
376    /// script's pad is left alone: captures run unfocused.
377    pub(super) fn window_focus_changed(&mut self, focused: bool) {
378        self.window_focused = focused;
379        if !focused {
380            for slot in self.real_pad_slots.iter().copied() {
381                self.input.clear_gamepad(slot);
382            }
383        }
384        let Some(gilrs) = &mut self.gilrs else {
385            return;
386        };
387        // Drain what came in before the change, so it isn't replayed.
388        while gilrs.next_event().is_some() {}
389        if !focused || self.capture.is_capture() {
390            return;
391        }
392        for (id, pad) in gilrs.gamepads() {
393            let slot = gamepad_id_to_u32(id);
394            self.real_pad_slots.insert(slot);
395            for axis in STICKS {
396                let value = pad.value(axis);
397                if value != 0.0 {
398                    self.input
399                        .process_gamepad_axis(slot, format!("{axis:?}"), value);
400                }
401            }
402        }
403    }
404}

418impl ApplicationHandler for PlayerApp {
419    fn resumed(&mut self, event_loop: &ActiveEventLoop) {
420        // No window yet means first launch. (On the web the window exists
421        // while the device is still on its way; don't start twice.)
422        if self.window.is_none() && self.render_context.is_none() {
423            // First launch — full initialization
424            if let Err(e) = self.initialize(event_loop) {
425                tracing::error!("Failed to initialize player: {e:#}");
426                event_loop.exit();
427            }
428        } else {
429            // Subsequent resume (Android returning from background).
430            // Recreate window and surface — GPU context is still valid.
431            #[cfg(target_os = "android")]
432            {
433                let window_attrs = Window::default_attributes().with_title("Flint Player");
434                match event_loop.create_window(window_attrs) {
435                    Ok(window) => {
436                        let window = Arc::new(window);
437                        self.window = Some(window.clone());
438                        if let Some(ctx) = &mut self.render_context {
439                            if let Err(e) = ctx.recreate_surface(window.clone()) {
440                                tracing::error!("Failed to recreate surface: {e}");
441                                return;
442                            }
443                            self.camera.aspect = ctx.aspect_ratio();
444                            let size = ctx.size;
445                            if let Some(renderer) = &mut self.scene_renderer {
446                                renderer.resize_postprocess(&ctx.device, size.width, size.height);
447                            }
448                            self.input
449                                .set_screen_size(size.width as f64, size.height as f64);
450                        }
451                        // The renderer kept its quality settings through the
452                        // suspend; apply them over the resized chain
453                        // (FlintKart#55).
454                        self.reapply_quality();
455                        self.cursor_captured = true;
456                    }
457                    Err(e) => {
458                        tracing::error!("Failed to recreate window on resume: {e}");
459                    }
460                }
461            }
462        }
463    }

467        // Drop the window reference — a new one will be created in resumed().

474    fn window_event(
475        &mut self,
476        event_loop: &ActiveEventLoop,
477        _window_id: WindowId,
478        event: WindowEvent,
479    ) {
480        // Not ready: the device is still on its way (web startup, #99), so
481        // there is nothing to tick, draw or steer. Each event is a chance to
482        // pick it up.
483        if self.render_context.is_none() {
484            #[cfg(target_arch = "wasm32")]
485            self.take_ready_render_context(event_loop);
486            if self.render_context.is_none() {
487                return;
488            }
489        }
490        if let WindowEvent::Focused(focused) = event {
491            self.window_focus_changed(focused);
492        }
493        if self.capture_drops(&event) {
494            return;
495        }
496        if self.taken_before_the_game(&event) {
497            return;
498        }
499
500        match event {
501            WindowEvent::CloseRequested => {
502                event_loop.exit();
503            }
504
505            WindowEvent::Resized(new_size) => {
506                self.handle_resize(new_size);
507            }
508
509            WindowEvent::Moved(_) | WindowEvent::ScaleFactorChanged { .. } => {
510                self.update_vsync_pacing();
511            }
512
513            WindowEvent::KeyboardInput {
514                device_id, event, ..
515            } => {
516                self.handle_keyboard_input(event_loop, device_id, event);
517            }
518
519            WindowEvent::MouseInput { state, button, .. } => {
520                self.handle_mouse_input(state, button);
521            }
522
523            WindowEvent::CursorMoved { position, .. } => {
524                self.input.process_mouse_move(position.x, position.y);
525            }
526
527            WindowEvent::MouseWheel { delta, .. } => {
528                let (dx, dy) = wheel_lines(delta);
529                self.input.process_mouse_wheel(dx, dy);
530            }
531
532            WindowEvent::Touch(touch) => {
533                self.handle_touch(touch);
534            }
535
536            WindowEvent::RedrawRequested => {
537                // A refresh the frame-rate pacer skips (#140) does nothing.
538                #[cfg(target_arch = "wasm32")]
539                if !self.web_frame_begin() {
540                    return;
541                }
542                self.tick();
543                self.render();
544                #[cfg(target_arch = "wasm32")]
545                self.web_frame_end();
546                // The frame's end, for the profiler (ADR 0109).
547                profiling::finish_frame!();
548                if self.music_exit_requested || self.exit_requested {
549                    event_loop.exit();
550                }
551            }
552
553            _ => {}
554        }
555    }

563    fn device_event(
564        &mut self,
565        _event_loop: &ActiveEventLoop,
566        _device_id: winit::event::DeviceId,
567        event: DeviceEvent,
568    ) {
569        // Raw motion flows while the cursor is captured, the window has
570        // focus and no text field has the keyboard (ADR 0110), and never
571        // into a capture (`look` is the script's).
572        if !self.cursor_captured
573            || !self.window_focused
574            || self.text_has_focus()
575            || self.capture.is_capture()
576        {
577            return;
578        }
579
580        // On the web the pointer moves the view only once the browser has
581        // locked it; before the first click it is just crossing the page.
582        #[cfg(target_arch = "wasm32")]
583        if !self.pointer_lock.held {
584            return;
585        }
586
587        if let DeviceEvent::MouseMotion { delta } = event {
588            self.input.process_mouse_raw_delta(delta.0, delta.1);
589        }
590    }

611    pub(super) fn handle_resize(&mut self, new_size: winit::dpi::PhysicalSize<u32>) {
612            if let Some(context) = &mut self.render_context {
613                context.resize(new_size);
614                self.camera.aspect = context.aspect_ratio();
615                // Resize post-processing HDR buffer and bloom chain
616                if let Some(renderer) = &mut self.scene_renderer {
617                    renderer.resize_postprocess(
618                        &context.device,
619                        new_size.width,
620                        new_size.height,
621                    );
622                }
623            }
624            self.input
625                .set_screen_size(new_size.width as f64, new_size.height as f64);
626            #[cfg(target_arch = "wasm32")]
627            self.measure_touch_scale(new_size);
628            if self.cursor_captured {
629                self.centre_the_cursor();
630            }
631    }

633    /// Pace the clock to the window's monitor when frames wait for vblank
634    /// (#406 in The Capitol). On start, and again when the window moves, since
635    /// another monitor can have another refresh. `FLINT_PACING=0` turns it off.
636    pub(super) fn update_vsync_pacing(&mut self) {
637        // On the web the frame-rate pacer sets the period (#140).
638        if self.frame_pacer.is_some() {
639            return;
640        }
641        let on_vblank = self
642            .render_context
643            .as_ref()
644            .is_some_and(|c| c.presents_on_vblank());
645        let disabled = std::env::var("FLINT_PACING").is_ok_and(|v| v == "0");
646        let period = (on_vblank && !disabled)
647            .then(|| self.window.as_ref()?.current_monitor()?.refresh_rate_millihertz())
648            .flatten()
649            .filter(|&mhz| mhz > 0)
650            .map(|mhz| 1000.0 / mhz as f64);
651        if period != self.clock.vsync_period() {
652            tracing::info!(
653                "frame pacing: {}",
654                period.map_or("off".to_string(), |p| format!("{:.2} Hz", 1.0 / p))
655            );
656            self.clock.set_vsync_period(period);
657        }
658    }

660    fn handle_keyboard_input(
661        &mut self,
662        event_loop: &ActiveEventLoop,
663        device_id: winit::event::DeviceId,
664        event: winit::event::KeyEvent,
665    ) {
666            // Suppress unused variable warning on non-Android platforms
667            let _ = &device_id;
668            #[cfg(target_arch = "wasm32")]
669            let _ = &event_loop;
670
671            if let PhysicalKey::Code(key_code) = event.physical_key {
672                match event.state {
673                    ElementState::Pressed => {
674                        // Handle escape to toggle cursor capture
675                        if key_code == KeyCode::Escape {
676                            if self.cursor_captured {
677                                self.release_cursor();
678                            } else {
679                                // A page has nowhere to exit to.
680                                #[cfg(not(any(target_os = "android", target_arch = "wasm32")))]
681                                event_loop.exit();
682                            }
683                            return;
684                        }
685
686                        // Debug keys, shared with `--input-script` (H9)
687                        self.handle_debug_key(key_code);
688
689                        self.input.process_key_down(key_code);
690
691                        // On Android, DPad maps to ArrowKeys — also fire gamepad
692                        // button events so configs with gamepad DPad bindings work.
693                        #[cfg(target_os = "android")]
694                        {
695                            let dpad_button = match key_code {
696                                KeyCode::ArrowUp => Some("DPadUp"),
697                                KeyCode::ArrowDown => Some("DPadDown"),
698                                KeyCode::ArrowLeft => Some("DPadLeft"),
699                                KeyCode::ArrowRight => Some("DPadRight"),
700                                _ => None,
701                            };
702                            if let Some(btn) = dpad_button {
703                                let slot = self.android_gamepad.register_device(device_id);
704                                self.input.process_gamepad_button_down(slot, btn);
705                            }
706                        }
707                    }
708                    ElementState::Released => {
709                        self.input.process_key_up(key_code);
710
711                        #[cfg(target_os = "android")]
712                        {
713                            let dpad_button = match key_code {
714                                KeyCode::ArrowUp => Some("DPadUp"),
715                                KeyCode::ArrowDown => Some("DPadDown"),
716                                KeyCode::ArrowLeft => Some("DPadLeft"),
717                                KeyCode::ArrowRight => Some("DPadRight"),
718                                _ => None,
719                            };
720                            if let Some(btn) = dpad_button {
721                                if let Some(slot) = self.android_gamepad.slot_for(device_id) {
722                                    self.input.process_gamepad_button_up(slot, btn);
723                                }
724                            }
725                        }
726                    }
727                }
728            }
729
730            // Android: intercept gamepad buttons delivered as unidentified native keycodes
731            #[cfg(target_os = "android")]
732            if let PhysicalKey::Unidentified(NativeKeyCode::Android(keycode)) =
733                event.physical_key
734            {
735                if let Some(button_name) = Self::android_keycode_to_gamepad_button(keycode) {
736                    let slot = self.android_gamepad.register_device(device_id);
737                    match event.state {
738                        ElementState::Pressed => {
739                            self.input.process_gamepad_button_down(slot, button_name);
740                        }
741                        ElementState::Released => {
742                            self.input.process_gamepad_button_up(slot, button_name);
743                        }
744                    }
745                }
746            }
747    }

749    /// The player's debug keys: F2 stats overlay, F3/F4 panels, the music
750    /// guide and timeline, F9 force-fail, F11 fullscreen. Called for a real
751    /// key press and for one the input script presses, so a capture can
752    /// open the overlay (H9). Escape stays window-only: it exits.
753    pub(super) fn handle_debug_key(&mut self, key_code: KeyCode) {
754        match key_code {
755            KeyCode::F2 => {
756                self.show_stats = !self.show_stats;
757            }
758            #[cfg(feature = "debug-hud")]
759            KeyCode::F3 => {
760                // Toggle the scene-component debug panels
761                // (grass, ocean, ...). The Rendering &
762                // Effects panel is excluded — it has its
763                // own key (F4) and would otherwise flip
764                // out of phase with it.
765                let scene_panels: Vec<usize> = self
766                    .debug_panels
767                    .iter()
768                    .enumerate()
769                    .filter(|(_, p)| {
770                        p.name() != flint_debug_ui::RENDER_DEBUG_PANEL
771                    })
772                    .map(|(i, _)| i)
773                    .collect();
774                if scene_panels.is_empty() {
775                    tracing::info!("No debug panels in current scene");
776                } else {
777                    // Toggle the panels, then adjust cursor outside the borrow
778                    for i in scene_panels {
779                        self.debug_panels[i].toggle();
780                    }
781                    // Cursor follows ALL panels (incl. the
782                    // render panel, which F3 leaves alone).
783                    let any_open =
784                        self.debug_panels.iter().any(|p| p.is_open());
785                    if any_open {
786                        self.release_cursor();
787                    } else if self.physics.has_player_entity() {
788                        self.capture_cursor();
789                    }
790                }
791            }
792            // Music Guide overlay (ADR 0035): the Phase 4
793            // debug-surface key — all F-keys are taken.
794            #[cfg(feature = "debug-hud")]
795            KeyCode::Backquote => {
796                self.toggle_named_panel(
797                    music_guide_panel::MUSIC_GUIDE_PANEL,
798                    "no music session running — nothing to guide",
799                );
800            }
801            // Manifest Map timeline strip: Backslash
802            // (unbound elsewhere; Backquote is the guide's).
803            #[cfg(feature = "debug-hud")]
804            KeyCode::Backslash => {
805                self.toggle_named_panel(
806                    timeline_panel::MANIFEST_MAP_PANEL,
807                    "no music session running — nothing to map",
808                );
809            }
810            // Debug force-fail: trigger the full
811            // reintegration (rewind → seam → reassembly)
812            // without having to play to a fail — seam
813            // iteration must not depend on first clearing
814            // the arming threshold (ADR 0064 follow-up).
815            #[cfg(feature = "debug-hud")]
816            KeyCode::F9 => {
817                if let Some(ms) = self.music_session.as_mut() {
818                    ms.debug_force_fail();
819                    println!(
820                        "[music] DEBUG: full-fail forced — rewind incoming"
821                    );
822                } else {
823                    println!(
824                        "[music] no music session running — nothing to fail"
825                    );
826                }
827            }
828            // Rendering & Effects menu (ADR 0053): the
829            // one home for every render/post debug
830            // control the old F1/F4-F10/F12 keys used to
831            // flip blindly, plus their non-binary params.
832            #[cfg(feature = "debug-hud")]
833            KeyCode::F4 => {
834                self.toggle_named_panel(
835                    flint_debug_ui::RENDER_DEBUG_PANEL,
836                    "no renderer active — nothing to tune",
837                );
838            }
839            KeyCode::F11 => {
840                if let Some(window) = &self.window {
841                    if window.fullscreen().is_some() {
842                        window.set_fullscreen(None);
843                    } else {
844                        window.set_fullscreen(Some(
845                            winit::window::Fullscreen::Borderless(None),
846                        ));
847                    }
848                }
849            }
850            _ => {}
851        }
852    }

854    /// Whether egui or an extension takes a window event before the player
855    /// and scripts see it.
856    fn taken_before_the_game(&mut self, event: &WindowEvent) -> bool {
857        if let WindowEvent::ModifiersChanged(modifiers) = event {
858            self.key_modifiers = modifiers.state();
859        }
860        // A captured cursor's moves reach no one, real or scripted: it reads
861        // as the window centre (ADR 0110).
862        if self.cursor_captured && matches!(event, WindowEvent::CursorMoved { .. }) {
863            return true;
864        }
865        // The on-screen touch controls sit above egui and the extensions:
866        // a finger on a stick or button goes straight to them (#133). Any
867        // finger landing brings them up first, even one egui then takes,
868        // or the first touch on a page with an extension would show none.
869        if let WindowEvent::Touch(touch) = event {
870            if touch.phase == winit::event::TouchPhase::Started {
871                self.input.touch_controls_mut().note_touch();
872            }
873            if self.touch_controls_take(touch) {
874                return false;
875            }
876        }
877        // Input egui wants stops here (see egui_input for the rules). A
878        // captured cursor's buttons and wheel skip egui, but for the release
879        // of a press egui took before the capture (ADR 0110).
880        let skip_egui = self.cursor_captured && self.egui_input.skips_captured(event);
881        if self.egui_sees_input() && !skip_egui {
882            if let (Some(egui_winit), Some(window)) = (&mut self.egui_winit, &self.window) {
883                let response = egui_winit.on_window_event(window, event);
884                let wants_keyboard = super::egui_input::text_field_focused(&self.egui_ctx);
885                let routed = super::egui_input::classify(event);
886                if self
887                    .egui_input
888                    .consumes(routed, response.consumed, wants_keyboard)
889                {
890                    return true;
891                }
892            }
893        }
894
895        // Game extensions next; they may consume input but never lifecycle
896        // events (see PlayerExtension::on_window_event and on_key).
897        crate::extension::dispatch_window_event(&mut self.extensions, event, self.key_modifiers)
898    }

900    /// Whether egui sees input before the game: whenever it can have
901    /// interactive UI on screen, an open debug panel or an extension's ui().
902    fn egui_sees_input(&self) -> bool {
903        #[cfg(feature = "debug-hud")]
904        let any_panel_open = self.debug_panels.iter().any(|p| p.is_open());
905        #[cfg(not(feature = "debug-hud"))]
906        let any_panel_open = false;
907        any_panel_open || !self.extensions.is_empty()
908    }

910    /// An event from `--input-script`, routed as a real device's would be.
911    pub(super) fn scripted_event(&mut self, event: ScriptEvent) {
912        match event {
913            ScriptEvent::Pointer(pointer) => self.scripted_pointer_event(pointer),
914            ScriptEvent::Key(key, state) => self.scripted_key(key, state),
915            ScriptEvent::Modifiers(state) => {
916                // A real `ModifiersChanged`: egui-winit keeps the modifiers
917                // for the events after it, extensions see it too, and so
918                // does the next key's `KeyInput`.
919                self.taken_before_the_game(&WindowEvent::ModifiersChanged(state.into()));
920            }
921            // Raw motion takes a real mouse's gate (ADR 0110), but for the
922            // window's focus: captures run unfocused.
923            // A finger takes a real one's path, but is already in surface
924            // pixels (no web rescale, no Android joystick guess).
925            ScriptEvent::Touch(id, phase, x, y) => {
926                let event = WindowEvent::Touch(winit::event::Touch {
927                    device_id: winit::event::DeviceId::dummy(),
928                    phase,
929                    location: winit::dpi::PhysicalPosition::new(x, y),
930                    force: None,
931                    id,
932                });
933                if !self.taken_before_the_game(&event) {
934                    self.touch_on_surface(id, phase, x, y);
935                }
936            }
937            ScriptEvent::Look(dx, dy) => {
938                if self.cursor_captured && !self.text_has_focus() {
939                    self.input.process_mouse_raw_delta(dx, dy);
940                }
941            }
942            ScriptEvent::Text(text) => {
943                if self.egui_sees_input() {
944                    if let Some(egui_winit) = &mut self.egui_winit {
945                        egui_winit
946                            .egui_input_mut()
947                            .events
948                            .push(egui::Event::Text(text));
949                    }
950                }
951            }
952        }
953    }

955    /// A key from `--input-script` (ADR 0097). egui gets what egui-winit
956    /// would have made of it, and the same gate as a real key decides
957    /// whether the game does: a key typed into a focused field stops there.
958    /// Then extensions' `on_key` (ADR 0103), which may take it. Their
959    /// `on_window_event` never sees one: it takes a winit `KeyboardInput`,
960    /// whose `KeyEvent` can't be built outside winit.
961    fn scripted_key(&mut self, key: KeyCode, state: ElementState) {
962        let pressed = state == ElementState::Pressed;
963        if self.egui_sees_input() {
964            if let Some(egui_winit) = &mut self.egui_winit {
965                let raw = egui_winit.egui_input_mut();
966                let events = super::egui_input::scripted_key_events(key, pressed, raw.modifiers);
967                raw.events.extend(events);
968            }
969            let held = Held::Key(PhysicalKey::Code(key));
970            let routed = if pressed {
971                Routed::Press(held)
972            } else {
973                Routed::Release(held)
974            };
975            let wants_keyboard = super::egui_input::text_field_focused(&self.egui_ctx);
976            if self.egui_input.consumes(routed, false, wants_keyboard) {
977                return;
978            }
979        }
980        let mut buf = [0; 4];
981        let character = super::egui_input::typed_char(key, self.key_modifiers.shift_key())
982            .map(|c| &*c.encode_utf8(&mut buf));
983        let input = crate::extension::KeyInput {
984            key,
985            state,
986            repeat: false,
987            character,
988            modifiers: self.key_modifiers,
989            scripted: true,
990        };
991        if crate::extension::dispatch_key(&mut self.extensions, &input) {
992            return;
993        }
994        // As a real key, through the debug keys (H9), but Esc doesn't quit.
995        if pressed {
996            self.handle_debug_key(key);
997            self.input.process_key_down(key);
998        } else {
999            self.input.process_key_up(key);
1000        }
1001    }

1003    /// A pointer event from `--input-script` (ADR 0082). It takes the path a
1004    /// real one does: egui, then extensions, then `InputState`.
1005    pub(super) fn scripted_pointer_event(&mut self, event: PointerEvent) {
1006        let event = event.to_window_event();
1007        if self.taken_before_the_game(&event) {
1008            return;
1009        }
1010        match event {
1011            WindowEvent::MouseInput { state, button, .. } => {
1012                self.handle_mouse_input(state, button);
1013            }
1014            WindowEvent::CursorMoved { position, .. } => {
1015                self.input.process_mouse_move(position.x, position.y);
1016            }
1017            WindowEvent::MouseWheel { delta, .. } => {
1018                let (dx, dy) = wheel_lines(delta);
1019                self.input.process_mouse_wheel(dx, dy);
1020            }
1021            _ => {}
1022        }
1023    }

1025    fn handle_mouse_input(&mut self, state: ElementState, button: MouseButton) {
1026            // Mouse button state always reaches InputState so scripts can
1027            // read it (`is_mouse_pressed`, menus, screen-space UI) and the
1028            // mouse-as-touch emulation keeps working. FPS scenes (a player
1029            // entity exists) additionally treat the first left click as
1030            // click-to-capture, which hides the cursor for mouse-look.
1031            if let Some(btn) = match button {
1032                MouseButton::Left => Some(0),
1033                MouseButton::Right => Some(1),
1034                MouseButton::Middle => Some(2),
1035                _ => None,
1036            } {
1037                match state {
1038                    ElementState::Pressed => self.input.process_mouse_button_down(btn),
1039                    ElementState::Released => self.input.process_mouse_button_up(btn),
1040                }
1041            }
1042
1043            if !self.cursor_captured
1044                && self.physics.has_player_entity()
1045                && state == ElementState::Pressed
1046                && button == MouseButton::Left
1047            {
1048                #[cfg(feature = "debug-hud")]
1049                let panel_open = self.debug_panels.iter().any(|p| p.is_open());
1050                #[cfg(not(feature = "debug-hud"))]
1051                let panel_open = false;
1052                if !panel_open {
1053                    self.capture_cursor();
1054                }
1055            }
1056    }

1058    fn handle_touch(&mut self, touch: winit::event::Touch) {
1059            // Android: joystick MotionEvents arrive as Touch with axis values
1060            // in [-1, 1]. Route these to gamepad axis input instead of touch.
1061            #[cfg(target_os = "android")]
1062            {
1063                let x = touch.location.x;
1064                let y = touch.location.y;
1065
1066                // If this device was already identified as a gamepad via button
1067                // presses, treat its touch events as stick axis data.
1068                if self.android_gamepad.is_gamepad(touch.device_id) {
1069                    if let Some(slot) = self.android_gamepad.slot_for(touch.device_id) {
1070                        self.input
1071                            .process_gamepad_axis(slot, "LeftStickX", x as f32);
1072                        self.input
1073                            .process_gamepad_axis(slot, "LeftStickY", y as f32);
1074                        return;
1075                    }
1076                }
1077
1078                // Heuristic: if coordinates are in joystick range [-1.5, 1.5]
1079                // and this touch id has no prior Started event (process_touch_move
1080                // would silently drop it anyway), treat it as a joystick from an
1081                // unregistered gamepad device.
1082                if x.abs() <= 1.5 && y.abs() <= 1.5 && !self.input.has_active_touch(touch.id) {
1083                    let slot = self.android_gamepad.register_device(touch.device_id);
1084                    self.input
1085                        .process_gamepad_axis(slot, "LeftStickX", x as f32);
1086                    self.input
1087                        .process_gamepad_axis(slot, "LeftStickY", y as f32);
1088                    return;
1089                }
1090            }
1091
1092            // Normal touch processing
1093            let x = touch.location.x;
1094            let y = touch.location.y;
1095            #[cfg(target_arch = "wasm32")]
1096            let (x, y) = self.touch_to_surface(x, y);
1097            self.touch_on_surface(touch.id, touch.phase, x, y);
1098    }

1100    /// A finger at (x, y) in surface pixels, real or scripted.
1101    pub(super) fn touch_on_surface(
1102        &mut self,
1103        id: u64,
1104        phase: winit::event::TouchPhase,
1105        x: f64,
1106        y: f64,
1107    ) {
1108            self.input.disable_touch_emulation();
1109            match phase {
1110                winit::event::TouchPhase::Started => {
1111                    self.input.process_touch_start(id, x, y);
1112                }
1113                winit::event::TouchPhase::Moved => {
1114                    self.input.process_touch_move(id, x, y);
1115                }
1116                winit::event::TouchPhase::Ended => {
1117                    // Update position from the End event before gesture detection —
1118                    // fast swipes may have few/no Move events between Start and End
1119                    self.input.process_touch_move(id, x, y);
1120                    self.input.process_touch_end(id);
1121                }
1122                winit::event::TouchPhase::Cancelled => {
1123                    self.input.process_touch_cancel(id);
1124                }
1125            }
1126    }
