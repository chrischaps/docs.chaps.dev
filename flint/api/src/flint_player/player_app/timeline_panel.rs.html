1//! The "Manifest Map" debug panel: a full-width bottom strip mapping the
2//! whole suite — sections, bar ruler, tempo/meter changes, re-entry points —
3//! with a playhead riding the current sample and this-run history (judged
4//! pulses, full-fail seams). Toggled with Backslash while a music session
5//! runs; compiled out entirely without the `debug-hud` feature. Judgment-
6//! shaped dev surface by design — it never ships in the felt experience
7//! (the no-gamified-UI rule).
8
9use flint_debug_ui::{DebugPanel, PanelLayout};
10use flint_music::chart_session::{PulseKind, SeamMarkKind, TimelineFrame, TimelineMap};
11
12pub const MANIFEST_MAP_PANEL: &str = "Manifest Map";
13
14/// Strip heights (logical points).
15const SECTION_H: f32 = 18.0;
16const RULER_H: f32 = 14.0;
17const HISTORY_H: f32 = 40.0;
18
19pub struct ManifestMapPanel {
20    open: bool,
21    map: TimelineMap,
22    frame: TimelineFrame,
23}
24
25impl ManifestMapPanel {
26    pub fn new() -> Self {
27        Self {
28            open: false,
29            map: TimelineMap::default(),
30            frame: TimelineFrame::default(),
31        }
32    }
33
34    /// The static suite structure — set once at panel registration.
35    pub fn set_map(&mut self, map: TimelineMap) {
36        self.map = map;
37    }
38
39    /// Per-frame playhead + history feed (only while open).
40    pub fn set_frame(&mut self, frame: TimelineFrame) {
41        self.frame = frame;
42    }
43}

45fn seam_color(kind: SeamMarkKind) -> egui::Color32 {
46    match kind {
47        SeamMarkKind::FullFail => egui::Color32::from_rgb(235, 90, 90),
48        SeamMarkKind::Seam => egui::Color32::from_rgb(255, 165, 70),
49        SeamMarkKind::ReassemblyComplete => egui::Color32::from_rgb(120, 180, 255),
50    }
51}
52
53impl DebugPanel for ManifestMapPanel {
54    fn name(&self) -> &str {
55        MANIFEST_MAP_PANEL
56    }
57
58    fn layout(&self) -> PanelLayout {
59        PanelLayout::Bottom
60    }
61
62    fn ui(&mut self, ui: &mut egui::Ui) {
63        let map = &self.map;
64        let f = &self.frame;
65        if map.end_sample <= 0 {
66            ui.label(egui::RichText::new("no suite map").weak());
67            return;
68        }
69
70        // Status line: the raw/suite clock relationship the strip visualizes.
71        ui.label(
72            egui::RichText::new(format!(
73                "raw {}  offset {}  suite {}{}",
74                f.raw_clock_sample,
75                f.timeline_offset,
76                f.playhead_sample,
77                if f.preroll { "  (preroll)" } else { "" }
78            ))
79            .monospace()
80            .size(10.0)
81            .weak(),
82        );
83
84        let height = SECTION_H + RULER_H + HISTORY_H;
85        let (resp, painter) = ui.allocate_painter(
86            egui::vec2(ui.available_width(), height),
87            egui::Sense::hover(),
88        );
89        let rect = resp.rect;
90
91        // Suite-sample → x. Domain starts at min(playhead, 0) so the preroll
92        // playhead is visible approaching zero.
93        let min_s = f.playhead_sample.min(0);
94        let span = (map.end_sample - min_s).max(1) as f32;
95        let to_x = |sample: i64| rect.left() + (sample - min_s) as f32 / span * rect.width();
96
97        let sec_top = rect.top();
98        let ruler_top = sec_top + SECTION_H;
99        let hist_top = ruler_top + RULER_H;
100        let hist_mid = hist_top + HISTORY_H * 0.5;
101
102        // Preroll region: everything left of sample 0, shaded dark.
103        if min_s < 0 {
104            let zero_x = to_x(0);
105            painter.rect_filled(
106                egui::Rect::from_min_max(rect.min, egui::pos2(zero_x, rect.bottom())),
107                0.0,
108                egui::Color32::from_gray(18),
109            );
110        }
111
112        // Lane 1 — sections band: alternating fills, names when they fit,
113        // re-entry sections outlined with a diamond at their start.
114        for (i, s) in map.sections.iter().enumerate() {
115            let x0 = to_x(s.start_sample);
116            let x1 = to_x(s.end_sample);
117            let band = egui::Rect::from_min_max(
118                egui::pos2(x0, sec_top),
119                egui::pos2(x1, sec_top + SECTION_H),
120            );
121            let fill = if i % 2 == 0 {
122                egui::Color32::from_gray(48)
123            } else {
124                egui::Color32::from_gray(38)
125            };
126            painter.rect_filled(band, 0.0, fill);
127            if s.re_entry {
128                painter.rect_stroke(
129                    band.shrink(0.5),
130                    0.0,
131                    egui::Stroke::new(1.0, egui::Color32::from_rgb(255, 200, 120)),
132                );
133                let c = egui::pos2(x0 + 5.0, sec_top + SECTION_H * 0.5);
134                let r = 3.5;
135                painter.add(egui::Shape::convex_polygon(
136                    vec![
137                        egui::pos2(c.x, c.y - r),
138                        egui::pos2(c.x + r, c.y),
139                        egui::pos2(c.x, c.y + r),
140                        egui::pos2(c.x - r, c.y),
141                    ],
142                    egui::Color32::from_rgb(255, 200, 120),
143                    egui::Stroke::NONE,
144                ));
145            }
146            let galley = painter.layout_no_wrap(
147                s.name.clone(),
148                egui::FontId::monospace(9.0),
149                egui::Color32::from_gray(200),
150            );
151            let text_x = x0 + if s.re_entry { 11.0 } else { 3.0 };
152            if text_x + galley.size().x <= x1 - 2.0 {
153                painter.galley(
154                    egui::pos2(text_x, sec_top + (SECTION_H - galley.size().y) * 0.5),
155                    galley,
156                    egui::Color32::from_gray(200),
157                );
158            }
159        }
160
161        // Lane 2 — bar ruler, density-thinned so ticks never smear.
162        if map.bars.len() > 1 {
163            let px_per_bar = rect.width() / map.bars.len() as f32;
164            let mut stride = 1usize;
165            while px_per_bar * (stride as f32) < 6.0 {
166                stride *= 2;
167            }
168            let label_ok = px_per_bar * stride as f32 >= 30.0;
169            for b in map.bars.iter().filter(|b| b.bar % stride as i64 == 0) {
170                let x = to_x(b.sample);
171                painter.line_segment(
172                    [egui::pos2(x, ruler_top), egui::pos2(x, ruler_top + 5.0)],
173                    egui::Stroke::new(1.0, egui::Color32::from_gray(110)),
174                );
175                if label_ok {
176                    painter.text(
177                        egui::pos2(x + 2.0, ruler_top + RULER_H * 0.5),
178                        egui::Align2::LEFT_CENTER,
179                        b.bar.to_string(),
180                        egui::FontId::monospace(8.0),
181                        egui::Color32::from_gray(140),
182                    );
183                }
184            }
185        }
186
187        // Tempo/meter change flags (the sample-0 anchor is the opening
188        // tempo, not a change — flag it only when later anchors exist).
189        let show_first = map.tempo_marks.len() > 1;
190        for (i, t) in map.tempo_marks.iter().enumerate() {
191            if i == 0 && !show_first {
192                continue;
193            }
194            let x = to_x(t.sample);
195            let color = egui::Color32::from_rgb(150, 210, 255);
196            painter.line_segment(
197                [
198                    egui::pos2(x, ruler_top),
199                    egui::pos2(x, hist_top + HISTORY_H),
200                ],
201                egui::Stroke::new(1.0, color),
202            );
203            painter.text(
204                egui::pos2(x + 3.0, ruler_top + RULER_H - 2.0),
205                egui::Align2::LEFT_BOTTOM,
206                format!("{}/{} {:.0}", t.beats_per_bar, t.beat_unit, t.bpm),
207                egui::FontId::monospace(8.0),
208                color,
209            );
210        }
211
212        // Lane 3 — this-run history. Pulse ticks: hit up-green, miss
213        // down-red, spurious mid-yellow (the guide panel's color language).
214        painter.line_segment(
215            [
216                egui::pos2(rect.left(), hist_mid),
217                egui::pos2(rect.right(), hist_mid),
218            ],
219            egui::Stroke::new(0.5, egui::Color32::from_gray(55)),
220        );
221        for p in &f.pulses {
222            let x = to_x(p.sample);
223            let (y0, y1, color) = match p.kind {
224                PulseKind::Hit => (
225                    hist_mid - 12.0,
226                    hist_mid,
227                    egui::Color32::from_rgb(140, 220, 130),
228                ),
229                PulseKind::Miss => (
230                    hist_mid,
231                    hist_mid + 12.0,
232                    egui::Color32::from_rgb(235, 90, 90),
233                ),
234                PulseKind::Spurious => (
235                    hist_mid - 5.0,
236                    hist_mid + 5.0,
237                    egui::Color32::from_rgb(230, 210, 110),
238                ),
239            };
240            painter.line_segment(
241                [egui::pos2(x, y0), egui::pos2(x, y1)],
242                egui::Stroke::new(1.0, color),
243            );
244        }
245
246        // Seams: full-fail line with a connector back to its re-entry point,
247        // an orange chevron at each committed re-entry, a blue reassembly tick.
248        for s in &f.seams {
249            let x = to_x(s.suite_sample);
250            let color = seam_color(s.kind);
251            match s.kind {
252                SeamMarkKind::FullFail => {
253                    painter.line_segment(
254                        [egui::pos2(x, hist_top), egui::pos2(x, hist_top + HISTORY_H)],
255                        egui::Stroke::new(1.5, color),
256                    );
257                    let rx = to_x(s.re_entry_sample);
258                    let y = hist_top + 3.0;
259                    painter.line_segment(
260                        [egui::pos2(x, y), egui::pos2(rx, y)],
261                        egui::Stroke::new(1.0, color.gamma_multiply(0.6)),
262                    );
263                    // Arrowhead at the re-entry end: the rewind's direction.
264                    painter.line_segment(
265                        [egui::pos2(rx, y), egui::pos2(rx + 4.0, y - 3.0)],
266                        egui::Stroke::new(1.0, color.gamma_multiply(0.6)),
267                    );
268                    painter.line_segment(
269                        [egui::pos2(rx, y), egui::pos2(rx + 4.0, y + 3.0)],
270                        egui::Stroke::new(1.0, color.gamma_multiply(0.6)),
271                    );
272                }
273                SeamMarkKind::Seam => {
274                    let y = hist_top + HISTORY_H - 5.0;
275                    let stroke = egui::Stroke::new(1.5, color);
276                    painter.line_segment([egui::pos2(x - 4.0, y - 4.0), egui::pos2(x, y)], stroke);
277                    painter.line_segment([egui::pos2(x, y), egui::pos2(x - 4.0, y + 4.0)], stroke);
278                }
279                SeamMarkKind::ReassemblyComplete => {
280                    painter.line_segment(
281                        [
282                            egui::pos2(x, hist_top + HISTORY_H - 8.0),
283                            egui::pos2(x, hist_top + HISTORY_H),
284                        ],
285                        egui::Stroke::new(1.5, color),
286                    );
287                }
288            }
289        }
290
291        // Playhead: full height, weak while preroll.
292        let px = to_x(f.playhead_sample);
293        let (ph_color, ph_w) = if f.preroll {
294            (egui::Color32::from_gray(120), 1.0)
295        } else {
296            (egui::Color32::WHITE, 2.0)
297        };
298        painter.line_segment(
299            [egui::pos2(px, rect.top()), egui::pos2(px, rect.bottom())],
300            egui::Stroke::new(ph_w, ph_color),
301        );
302        painter.add(egui::Shape::convex_polygon(
303            vec![
304                egui::pos2(px - 4.0, rect.top()),
305                egui::pos2(px + 4.0, rect.top()),
306                egui::pos2(px, rect.top() + 5.0),
307            ],
308            ph_color,
309            egui::Stroke::NONE,
310        ));
311
312        // Hover: cursor x → sample/bar readout.
313        if let Some(pos) = resp.hover_pos() {
314            let sample = min_s + ((pos.x - rect.left()) / rect.width().max(1.0) * span) as i64;
315            let bar = match map.bars.binary_search_by_key(&sample, |b| b.sample) {
316                Ok(i) => map.bars[i].bar,
317                Err(0) => -1,
318                Err(i) => map.bars[i - 1].bar,
319            };
320            resp.on_hover_text(
321                egui::RichText::new(format!("sample {sample}  bar {bar}"))
322                    .monospace()
323                    .size(10.0),
324            );
325        }
326    }
