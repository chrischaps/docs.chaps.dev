1//! The style commands scripts queue (#269): `set_entity_style`,
2//! `set_style_param` and `clear_entity_style`, applied to the world by
3//! `flint_render::entity_style`. The renderer reads the world after
4//! scripts each frame, so a change shows the frame it was made.
5
6use flint_core::EntityId;
7use flint_ecs::FlintWorld;
8use flint_render::entity_style;
9use flint_runtime::GameEvent;
10use flint_script::context::ScriptCommand;
11
12/// The event a style change fires to scripts' `on_event`, with
13/// `#{ entity, style }` (`style` is `""` once cleared).
14pub const STYLE_CHANGED: &str = "style_changed";
15
16/// What applying one command did.
17#[derive(Debug)]
18pub enum StyleApplied {
19    /// Not a style command: the caller handles it.
20    NotAStyleCommand,
21    /// Applied. Setting or clearing a style hands back the `style_changed`
22    /// event to fire; a param change fires none (it may come every frame).
23    Done(Option<GameEvent>),
24    /// Refused, with what to tell the script's author.
25    Failed(String),
26}
27
28fn changed(entity_id: i64, style: &str) -> GameEvent {
29    let mut data = toml::Table::new();
30    data.insert("entity".into(), toml::Value::Integer(entity_id));
31    data.insert("style".into(), style.into());
32    GameEvent::Custom {
33        name: STYLE_CHANGED.into(),
34        data: toml::Value::Table(data),
35    }
36}
37
38/// Apply `cmd` to `world` if it's a style command.
39pub fn apply_style_command(world: &mut FlintWorld, cmd: &ScriptCommand) -> StyleApplied {
40    let id = |e: i64| EntityId(e.max(0) as u64);
41    match cmd {
42        ScriptCommand::SetEntityStyle { entity_id, style } => {
43            match entity_style::set_entity_style(world, id(*entity_id), style) {
44                Ok(()) => StyleApplied::Done(Some(changed(*entity_id, style))),
45                Err(e) => StyleApplied::Failed(format!("set_entity_style: {e}")),
46            }
47        }
48        ScriptCommand::SetStyleParam {
49            entity_id,
50            param,
51            value,
52        } => match entity_style::set_style_param(world, id(*entity_id), param, value.clone()) {
53            Ok(_) => StyleApplied::Done(None),
54            Err(e) => StyleApplied::Failed(format!("set_style_param: {e}")),
55        },
56        ScriptCommand::ClearEntityStyle { entity_id } => {
57            let had = entity_style::clear_entity_style(world, id(*entity_id));
58            StyleApplied::Done(had.then(|| changed(*entity_id, "")))
59        }
60        _ => StyleApplied::NotAStyleCommand,
61    }
62}
