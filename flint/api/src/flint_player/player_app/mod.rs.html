105/// Frame-capture settings for the player (`--screenshot` family of flags).
106///
107/// Times are compared against `GameClock::total_time`, so they are game
108/// seconds since the first tick, not wall-clock seconds since launch.
109#[derive(Debug, Clone, Default)]
110pub struct CaptureConfig {
111    /// Output path. Base name for numbered files when `every` is set
112    /// (`hud.png` -> `hud_0001.png`, `hud_0002.png`, ...).
113    pub path: Option<std::path::PathBuf>,
114    /// Game time (s) at which to take the single screenshot. Default 0 =
115    /// first frame.
116    pub at: f64,
117    /// Interval (s) between repeated screenshots; None = single shot.
118    pub every: Option<f64>,
119    /// Game time (s) at which the player exits on its own.
120    pub exit_after: Option<f64>,
121    /// Timed input played into `InputState` (`--input-script`).
122    pub input_script: Option<input_script::InputScript>,
123    /// Open the F2 stats overlay on the first frame (`--stats`). F2 still
124    /// closes it.
125    pub show_stats: bool,
126    /// Internal: the first capture has been written.
127    pub(crate) taken: bool,
128    /// Internal: number of numbered frames written so far.
129    pub(crate) sequence: u32,
130    /// Internal: game time of the next scheduled interval capture.
131    pub(crate) next_due: f64,
132    /// Internal: frames presented and game time accumulated after the first
133    /// `PERF_SKIP_SECS`, reported on `--exit-after` as an average frame time.
134    pub(crate) perf_frames: u32,
135    pub(crate) perf_time: f64,
136    /// Internal: render stats summed over the same frames, reported beside
137    /// the frame time.
138    pub(crate) perf_draws: u64,
139    pub(crate) perf_entity_draws: u64,
140    pub(crate) perf_culled: u64,
141    /// Of those, set aside by level of detail / draw distance (#56)
142    pub(crate) perf_lod_culled: u64,
143    pub(crate) perf_triangles: u64,
144    pub(crate) perf_shadow_triangles: u64,
145    /// GPU frame times summed over frames that had one (ADR 0108)
146    pub(crate) perf_gpu_ms: f64,
147    pub(crate) perf_gpu_frames: u64,
148    /// The same frames' dt and present-to-present times, reported as a
149    /// spread so uneven pacing shows at an even mean.
150    pub(crate) perf_pacing: pacing::Pacing,
151}
152
153/// Game seconds ignored by the `--exit-after` frame-time report (loading,
154/// shader warm-up, first-frame texture uploads).
155pub(crate) const PERF_SKIP_SECS: f64 = 2.0;
156
157impl CaptureConfig {
158    /// Whether this run is a capture: it shoots, stops on its own or plays
159    /// an input script. A capture hears its script alone: real keys, the
160    /// mouse, the wheel and real pads are dropped before egui or the game
161    /// sees them, so a stray scroll over its window can't move its shot
162    /// (ADR 0110's amendment).
163    pub fn is_capture(&self) -> bool {
164        self.path.is_some() || self.exit_after.is_some() || self.input_script.is_some()
165    }
166
167    /// Build a capture schedule. `every <= 0` means single-shot.
168    pub fn new(
169        path: Option<std::path::PathBuf>,
170        at: f64,
171        every: Option<f64>,
172        exit_after: Option<f64>,
173    ) -> Self {
174        Self {
175            path,
176            at,
177            every: every.filter(|s| *s > 0.0),
178            exit_after,
179            ..Default::default()
180        }
181    }
182}
