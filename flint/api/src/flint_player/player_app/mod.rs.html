42/// Game-side day/night component driven by a script; the player only knows
43/// it to offer the F3 time scrubber (see flint-debug-ui tod_panel).
44#[cfg(feature = "debug-hud")]
45const TIME_OF_DAY_COMPONENT: &str = "time_of_day";
46#[cfg(feature = "debug-hud")]
47const WEATHER_COMPONENT: &str = "weather";
48/// Reality-tear controller (rare render-mode world events) driven by a
49/// game-side script; the F3 Reality panel forces/ends tears for tuning.
50#[cfg(feature = "debug-hud")]
51const REALITY_COMPONENT: &str = "reality";
52/// Second-raft visitor event controller driven by a game-side script; the
53/// F3 Visitor panel shows its phase/day and can force a visit for tuning.
54#[cfg(feature = "debug-hud")]
55const RAFT_VISITOR_COMPONENT: &str = "raft_visitor";
56/// Dead-calm ocean-stillness event controller driven by a game-side script;
57/// the F3 Dead Calm panel shows its phase/envelope and can force/end one.
58#[cfg(feature = "debug-hud")]
59const DEAD_CALM_COMPONENT: &str = "dead_calm";
60/// Live-tunable camera settings applied to the render camera at scene load
61/// and edited through the F3 Camera panel (see flint-debug-ui camera_panel).
