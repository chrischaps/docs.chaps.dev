1//! Spline-based procedural geometry.
2//!
3//! Two-pass initialization:
4//! 1. Load `.spline.toml` files for entities with a `spline` component,
5//!    sample the path, and store `spline_data` on each entity.
6//! 2. For entities with a `spline_mesh` component, sweep a rectangular
7//!    cross-section along the referenced spline and upload the resulting
8//!    mesh + trimesh collider.
9

23fn toml_f32(v: &toml::Value) -> Option<f32> {
24    v.as_float()
25        .map(|f| f as f32)
26        .or_else(|| v.as_integer().map(|i| i as f32))
27}
28
29fn toml_f32_array(arr: &[toml::Value]) -> Option<Vec<f32>> {
30    arr.iter().map(toml_f32).collect()
31}

35struct SplineDef {
36    closed: bool,
37    spacing: f32,
38    control_points: Vec<SplineControlPoint>,
39}
40
41fn parse_spline_file(path: &Path) -> Option<SplineDef> {
42    let content = std::fs::read_to_string(path).ok()?;
43    let value: toml::Value = toml::from_str(&content).ok()?;
44
45    let spline_section = value.get("spline")?;
46    let closed = spline_section
47        .get("closed")
48        .and_then(|v| v.as_bool())
49        .unwrap_or(true);
50
51    let spacing = value
52        .get("sampling")
53        .and_then(|s| s.get("spacing"))
54        .and_then(toml_f32)
55        .unwrap_or(2.0);
56
57    let points_arr = value.get("control_points")?.as_array()?;
58    let mut control_points = Vec::new();
59
60    for pt in points_arr {
61        let pos = pt.get("position")?.as_array()?;
62        if pos.len() < 3 {
63            continue;
64        }
65        let vals = toml_f32_array(pos)?;
66        let twist = pt.get("twist").and_then(toml_f32).unwrap_or(0.0);
67
68        control_points.push(SplineControlPoint {
69            position: Vec3::new(vals[0], vals[1], vals[2]),
70            twist,
71        });
72    }
73
74    if control_points.len() < 2 {
75        eprintln!(
76            "Spline needs at least 2 control points, got {}",
77            control_points.len()
78        );
79        return None;
80    }
81
82    Some(SplineDef {
83        closed,
84        spacing,
85        control_points,
86    })
87}

89/// Store sampled spline data as flat arrays in a `spline_data` ECS component.
90fn store_spline_data(
91    world: &mut FlintWorld,
92    entity_id: flint_core::EntityId,
93    samples: &[SplineSample],
94    closed: bool,
95) {
96    let to_val = |f: f32| toml::Value::Float(f as f64);
97
98    let positions_x: Vec<toml::Value> = samples.iter().map(|s| to_val(s.position.x)).collect();
99    let positions_y: Vec<toml::Value> = samples.iter().map(|s| to_val(s.position.y)).collect();
100    let positions_z: Vec<toml::Value> = samples.iter().map(|s| to_val(s.position.z)).collect();
101    let forwards_x: Vec<toml::Value> = samples.iter().map(|s| to_val(s.forward.x)).collect();
102    let forwards_y: Vec<toml::Value> = samples.iter().map(|s| to_val(s.forward.y)).collect();
103    let forwards_z: Vec<toml::Value> = samples.iter().map(|s| to_val(s.forward.z)).collect();
104    let rights_x: Vec<toml::Value> = samples.iter().map(|s| to_val(s.right.x)).collect();
105    let rights_y: Vec<toml::Value> = samples.iter().map(|s| to_val(s.right.y)).collect();
106    let rights_z: Vec<toml::Value> = samples.iter().map(|s| to_val(s.right.z)).collect();
107    let ups_x: Vec<toml::Value> = samples.iter().map(|s| to_val(s.up.x)).collect();
108    let ups_y: Vec<toml::Value> = samples.iter().map(|s| to_val(s.up.y)).collect();
109    let ups_z: Vec<toml::Value> = samples.iter().map(|s| to_val(s.up.z)).collect();
110    let twists: Vec<toml::Value> = samples.iter().map(|s| to_val(s.twist)).collect();
111    let t_values: Vec<toml::Value> = samples.iter().map(|s| to_val(s.t)).collect();
112
113    let mut table = toml::map::Map::new();
114    table.insert("positions_x".into(), toml::Value::Array(positions_x));
115    table.insert("positions_y".into(), toml::Value::Array(positions_y));
116    table.insert("positions_z".into(), toml::Value::Array(positions_z));
117    table.insert("forwards_x".into(), toml::Value::Array(forwards_x));
118    table.insert("forwards_y".into(), toml::Value::Array(forwards_y));
119    table.insert("forwards_z".into(), toml::Value::Array(forwards_z));
120    table.insert("rights_x".into(), toml::Value::Array(rights_x));
121    table.insert("rights_y".into(), toml::Value::Array(rights_y));
122    table.insert("rights_z".into(), toml::Value::Array(rights_z));
123    table.insert("ups_x".into(), toml::Value::Array(ups_x));
124    table.insert("ups_y".into(), toml::Value::Array(ups_y));
125    table.insert("ups_z".into(), toml::Value::Array(ups_z));
126    table.insert("twists".into(), toml::Value::Array(twists));
127    table.insert("t_values".into(), toml::Value::Array(t_values));
128    table.insert(
129        "sample_count".into(),
130        toml::Value::Integer(samples.len() as i64),
131    );
132    table.insert("closed".into(), toml::Value::Boolean(closed));
133
134    let _ = world.set_component(entity_id, "spline_data", toml::Value::Table(table));
135}

139struct SplineMeshDef {
140    spline_entity: String,
141    width: f32,
142    height: f32,
143    offset: [f32; 2], // [right, up]
144    color: [f32; 4],
145    stripe_color: Option<[f32; 4]>,
146    stripe_width: f32,
147    friction: f32,
148    restitution: f32,
149    metallic: f32,
150    roughness: f32,
151}
152
153fn parse_spline_mesh_component(comp: &toml::Value) -> Option<SplineMeshDef> {
154    let spline_entity = comp.get("spline")?.as_str()?.to_string();
155    let width = comp.get("width").and_then(toml_f32)?;
156    let height = comp.get("height").and_then(toml_f32)?;
157
158    let offset = if let Some(arr) = comp.get("offset").and_then(|v| v.as_array()) {
159        let vals = toml_f32_array(arr)?;
160        if vals.len() >= 2 {
161            [vals[0], vals[1]]
162        } else {
163            [0.0, 0.0]
164        }
165    } else {
166        [0.0, 0.0]
167    };
168
169    let color = if let Some(arr) = comp.get("color").and_then(|v| v.as_array()) {
170        let vals = toml_f32_array(arr)?;
171        if vals.len() >= 4 {
172            [vals[0], vals[1], vals[2], vals[3]]
173        } else if vals.len() >= 3 {
174            [vals[0], vals[1], vals[2], 1.0]
175        } else {
176            [0.5, 0.5, 0.5, 1.0]
177        }
178    } else {
179        [0.5, 0.5, 0.5, 1.0]
180    };
181
182    let stripe_color = if let Some(arr) = comp.get("stripe_color").and_then(|v| v.as_array()) {
183        let vals = toml_f32_array(arr)?;
184        if vals.len() >= 4 {
185            Some([vals[0], vals[1], vals[2], vals[3]])
186        } else if vals.len() >= 3 {
187            Some([vals[0], vals[1], vals[2], 1.0])
188        } else {
189            None
190        }
191    } else {
192        None
193    };
194
195    let stripe_width = comp.get("stripe_width").and_then(toml_f32).unwrap_or(2.0);
196
197    let friction = comp.get("friction").and_then(toml_f32).unwrap_or(0.5);
198    let restitution = comp.get("restitution").and_then(toml_f32).unwrap_or(0.1);
199    let metallic = comp.get("metallic").and_then(toml_f32).unwrap_or(0.0);
200    let roughness = comp.get("roughness").and_then(toml_f32).unwrap_or(0.5);
201
202    Some(SplineMeshDef {
203        spline_entity,
204        width,
205        height,
206        offset,
207        color,
208        stripe_color,
209        stripe_width,
210        friction,
211        restitution,
212        metallic,
213        roughness,
214    })
215}

217/// Generate a rectangular cross-section mesh swept along spline samples.
218///
219/// At each sample, 4 corners are computed:
220///   TL ──── TR
221///   │        │
222///   BL ──── BR
223///
224/// Between consecutive samples, 4 quad faces are generated
225/// (top, bottom, left-side, right-side). For open splines,
226/// 2 end-cap quads are added.
227fn generate_cross_section_mesh(
228    samples: &[SplineSample],
229    closed: bool,
230    width: f32,
231    height: f32,
232    offset: [f32; 2],
233    color: [f32; 4],
234    stripe_color: Option<[f32; 4]>,
235    stripe_width: f32,
236    spacing: f32,
237) -> (Vec<Vertex>, Vec<u32>, Vec<[f32; 3]>, Vec<[u32; 3]>) {
238    let n = samples.len();
239    if n < 2 {
240        return (Vec::new(), Vec::new(), Vec::new(), Vec::new());
241    }
242
243    let hw = width / 2.0;
244    let hh = height / 2.0;
245
246    // Pre-compute 4 corners at each sample in world space
247    // Order: BL, BR, TR, TL (bottom-left, bottom-right, top-right, top-left)
248    let mut corners: Vec<[Vec3; 4]> = Vec::with_capacity(n);
249    for s in samples {
250        let center = s.position + s.right * offset[0] + s.up * offset[1];
251        let bl = center + s.right * (-hw) + s.up * (-hh);
252        let br = center + s.right * hw + s.up * (-hh);
253        let tr = center + s.right * hw + s.up * hh;
254        let tl = center + s.right * (-hw) + s.up * hh;
255        corners.push([bl, br, tr, tl]);
256    }
257
258    // Each longitudinal segment contributes 4 quads (8 triangles, 8 vertices unique per face)
259    // We use per-face vertices for correct normals.
260    let num_segs = if closed { n } else { n - 1 };
261    // 4 faces * 4 verts each = 16 verts per segment
262    let vert_cap = num_segs * 16 + if !closed { 8 } else { 0 };
263    let tri_cap = num_segs * 8 + if !closed { 4 } else { 0 };
264
265    let mut vertices = Vec::with_capacity(vert_cap);
266    let mut indices = Vec::with_capacity(tri_cap * 3);
267    let mut phys_verts = Vec::with_capacity(vert_cap);
268    let mut phys_tris = Vec::with_capacity(tri_cap);
269
270    for seg in 0..num_segs {
271        let next = if closed { (seg + 1) % n } else { seg + 1 };
272        let [c_bl, c_br, c_tr, c_tl] = corners[seg];
273        let [n_bl, n_br, n_tr, n_tl] = corners[next];
274
275        let u0 = seg as f32 / num_segs as f32;
276        let u1 = (seg + 1) as f32 / num_segs as f32;
277
278        // Per-segment color: alternate between primary and stripe color
279        let seg_color = if let Some(sc) = stripe_color {
280            let dist = seg as f32 * spacing;
281            let stripe_index = (dist / stripe_width).floor() as i32;
282            if stripe_index % 2 == 0 { color } else { sc }
283        } else {
284            color
285        };
286
287        // --- Top face (TL, TR → NTL, NTR) ---
288        // Normal: average of up vectors
289        let top_normal = ((samples[seg].up + samples[next].up) * 0.5).normalized();
290        let base = vertices.len() as u32;
291        vertices.extend_from_slice(&[
292            Vertex { position: c_tl.to_array(), normal: top_normal.to_array(), color: seg_color, uv: [0.0, u0] },
293            Vertex { position: c_tr.to_array(), normal: top_normal.to_array(), color: seg_color, uv: [1.0, u0] },
294            Vertex { position: n_tl.to_array(), normal: top_normal.to_array(), color: seg_color, uv: [0.0, u1] },
295            Vertex { position: n_tr.to_array(), normal: top_normal.to_array(), color: seg_color, uv: [1.0, u1] },
296        ]);
297        indices.extend_from_slice(&[base, base + 1, base + 2, base + 1, base + 3, base + 2]);
298        let pb = phys_verts.len() as u32;
299        phys_verts.extend_from_slice(&[c_tl.to_array(), c_tr.to_array(), n_tl.to_array(), n_tr.to_array()]);
300        phys_tris.push([pb, pb + 1, pb + 2]);
301        phys_tris.push([pb + 1, pb + 3, pb + 2]);
302
303        // --- Bottom face (BL, BR → NBL, NBR) ---
304        let bot_normal = ((samples[seg].up + samples[next].up) * -0.5).normalized();
305        let base = vertices.len() as u32;
306        vertices.extend_from_slice(&[
307            Vertex { position: c_br.to_array(), normal: bot_normal.to_array(), color: seg_color, uv: [0.0, u0] },
308            Vertex { position: c_bl.to_array(), normal: bot_normal.to_array(), color: seg_color, uv: [1.0, u0] },
309            Vertex { position: n_br.to_array(), normal: bot_normal.to_array(), color: seg_color, uv: [0.0, u1] },
310            Vertex { position: n_bl.to_array(), normal: bot_normal.to_array(), color: seg_color, uv: [1.0, u1] },
311        ]);
312        indices.extend_from_slice(&[base, base + 1, base + 2, base + 1, base + 3, base + 2]);
313        let pb = phys_verts.len() as u32;
314        phys_verts.extend_from_slice(&[c_br.to_array(), c_bl.to_array(), n_br.to_array(), n_bl.to_array()]);
315        phys_tris.push([pb, pb + 1, pb + 2]);
316        phys_tris.push([pb + 1, pb + 3, pb + 2]);
317
318        // --- Left side face (TL, BL → NTL, NBL) ---
319        let left_normal = ((samples[seg].right + samples[next].right) * -0.5).normalized();
320        let base = vertices.len() as u32;
321        vertices.extend_from_slice(&[
322            Vertex { position: c_bl.to_array(), normal: left_normal.to_array(), color: seg_color, uv: [0.0, u0] },
323            Vertex { position: c_tl.to_array(), normal: left_normal.to_array(), color: seg_color, uv: [1.0, u0] },
324            Vertex { position: n_bl.to_array(), normal: left_normal.to_array(), color: seg_color, uv: [0.0, u1] },
325            Vertex { position: n_tl.to_array(), normal: left_normal.to_array(), color: seg_color, uv: [1.0, u1] },
326        ]);
327        indices.extend_from_slice(&[base, base + 1, base + 2, base + 1, base + 3, base + 2]);
328        let pb = phys_verts.len() as u32;
329        phys_verts.extend_from_slice(&[c_bl.to_array(), c_tl.to_array(), n_bl.to_array(), n_tl.to_array()]);
330        phys_tris.push([pb, pb + 1, pb + 2]);
331        phys_tris.push([pb + 1, pb + 3, pb + 2]);
332
333        // --- Right side face (BR, TR → NBR, NTR) ---
334        let right_normal = ((samples[seg].right + samples[next].right) * 0.5).normalized();
335        let base = vertices.len() as u32;
336        vertices.extend_from_slice(&[
337            Vertex { position: c_tr.to_array(), normal: right_normal.to_array(), color: seg_color, uv: [0.0, u0] },
338            Vertex { position: c_br.to_array(), normal: right_normal.to_array(), color: seg_color, uv: [1.0, u0] },
339            Vertex { position: n_tr.to_array(), normal: right_normal.to_array(), color: seg_color, uv: [0.0, u1] },
340            Vertex { position: n_br.to_array(), normal: right_normal.to_array(), color: seg_color, uv: [1.0, u1] },
341        ]);
342        indices.extend_from_slice(&[base, base + 1, base + 2, base + 1, base + 3, base + 2]);
343        let pb = phys_verts.len() as u32;
344        phys_verts.extend_from_slice(&[c_tr.to_array(), c_br.to_array(), n_tr.to_array(), n_br.to_array()]);
345        phys_tris.push([pb, pb + 1, pb + 2]);
346        phys_tris.push([pb + 1, pb + 3, pb + 2]);
347    }
348
349    // End caps for open splines
350    if !closed && n >= 2 {
351        // Front cap (at sample 0): BL, BR, TR, TL — normal = -forward
352        let cap_normal = (samples[0].forward * -1.0).to_array();
353        let [c_bl, c_br, c_tr, c_tl] = corners[0];
354        let base = vertices.len() as u32;
355        vertices.extend_from_slice(&[
356            Vertex { position: c_bl.to_array(), normal: cap_normal, color, uv: [0.0, 0.0] },
357            Vertex { position: c_br.to_array(), normal: cap_normal, color, uv: [1.0, 0.0] },
358            Vertex { position: c_tr.to_array(), normal: cap_normal, color, uv: [1.0, 1.0] },
359            Vertex { position: c_tl.to_array(), normal: cap_normal, color, uv: [0.0, 1.0] },
360        ]);
361        // Wind CCW when viewed from outside (facing -forward)
362        indices.extend_from_slice(&[base, base + 2, base + 1, base, base + 3, base + 2]);
363        let pb = phys_verts.len() as u32;
364        phys_verts.extend_from_slice(&[c_bl.to_array(), c_br.to_array(), c_tr.to_array(), c_tl.to_array()]);
365        phys_tris.push([pb, pb + 2, pb + 1]);
366        phys_tris.push([pb, pb + 3, pb + 2]);
367
368        // Back cap (at last sample): normal = +forward
369        let last = n - 1;
370        let cap_normal = samples[last].forward.to_array();
371        let [c_bl, c_br, c_tr, c_tl] = corners[last];
372        let base = vertices.len() as u32;
373        vertices.extend_from_slice(&[
374            Vertex { position: c_bl.to_array(), normal: cap_normal, color, uv: [0.0, 0.0] },
375            Vertex { position: c_br.to_array(), normal: cap_normal, color, uv: [1.0, 0.0] },
376            Vertex { position: c_tr.to_array(), normal: cap_normal, color, uv: [1.0, 1.0] },
377            Vertex { position: c_tl.to_array(), normal: cap_normal, color, uv: [0.0, 1.0] },
378        ]);
379        indices.extend_from_slice(&[base, base + 1, base + 2, base, base + 2, base + 3]);
380        let pb = phys_verts.len() as u32;
381        phys_verts.extend_from_slice(&[c_bl.to_array(), c_br.to_array(), c_tr.to_array(), c_tl.to_array()]);
382        phys_tris.push([pb, pb + 1, pb + 2]);
383        phys_tris.push([pb, pb + 2, pb + 3]);
384    }
385
386    (vertices, indices, phys_verts, phys_tris)
387}

389/// Helper to set model.asset on an entity so the renderer draws it.

402/// Load all spline data and generate geometry for spline_mesh entities.
403///
404/// Called from `PlayerApp::initialize()`. Scans the world for:
405/// - Entities with a `spline` component (Pass 1: load + sample)
406/// - Entities with a `spline_mesh` component (Pass 2: geometry)
407pub fn load_splines(
408    scene_path: &str,
409    world: &mut FlintWorld,
410    renderer: &mut SceneRenderer,
411    mut physics: Option<&mut PhysicsSystem>,
412    device: &wgpu::Device,
413) {
414    let scene_dir = Path::new(scene_path)
415        .parent()
416        .unwrap_or_else(|| Path::new("."));
417
418    // ── Pass 1: Load spline data ──
419
420    // Collect spline entity names and their sampled data
421    // (we need to collect first to avoid borrowing world during iteration)
422    struct SplineInfo {
423        entity_id: flint_core::EntityId,
424        entity_name: String,
425        samples: Vec<SplineSample>,
426        closed: bool,
427        spacing: f32,
428    }
429    let mut spline_infos: Vec<SplineInfo> = Vec::new();
430
431    for entity in world.all_entities() {
432        let spline_file = world
433            .get_components(entity.id)
434            .and_then(|c| c.get("spline").cloned())
435            .and_then(|s| s.get("file").and_then(|v| v.as_str().map(String::from)));
436
437        if let Some(file_rel) = spline_file {
438            // Resolve path: scene dir, then parent
439            let spline_path = {
440                let p = scene_dir.join(&file_rel);
441                if p.exists() {
442                    p
443                } else if let Some(parent) = scene_dir.parent() {
444                    parent.join(&file_rel)
445                } else {
446                    p
447                }
448            };
449
450            match parse_spline_file(&spline_path) {
451                Some(def) => {
452                    let samples = if def.closed {
453                        spline::sample_closed_spline(&def.control_points, def.spacing)
454                    } else {
455                        spline::sample_open_spline(&def.control_points, def.spacing)
456                    };
457
458                    println!(
459                        "Loaded spline '{}': {} control points → {} samples ({})",
460                        entity.name,
461                        def.control_points.len(),
462                        samples.len(),
463                        if def.closed { "closed" } else { "open" },
464                    );
465
466                    spline_infos.push(SplineInfo {
467                        entity_id: entity.id,
468                        entity_name: entity.name.clone(),
469                        samples,
470                        closed: def.closed,
471                        spacing: def.spacing,
472                    });
473                }
474                None => {
475                    eprintln!("Failed to parse spline file: {}", spline_path.display());
476                }
477            }
478        }
479    }
480
481    // Store spline data on entities
482    for info in &spline_infos {
483        store_spline_data(world, info.entity_id, &info.samples, info.closed);
484    }
485
486    // Build lookup: entity name → index into spline_infos
487    let spline_lookup: HashMap<&str, usize> = spline_infos
488        .iter()
489        .enumerate()
490        .map(|(i, info)| (info.entity_name.as_str(), i))
491        .collect();
492
493    // ── Pass 2: Generate spline_mesh geometry ──
494
495    // Collect mesh entities first
496    struct MeshJob {
497        entity_id: flint_core::EntityId,
498        entity_name: String,
499        def: SplineMeshDef,
500    }
501    let mut mesh_jobs: Vec<MeshJob> = Vec::new();
502
503    for entity in world.all_entities() {
504        let mesh_comp = world
505            .get_components(entity.id)
506            .and_then(|c| c.get("spline_mesh").cloned());
507
508        if let Some(comp) = mesh_comp {
509            match parse_spline_mesh_component(&comp) {
510                Some(def) => {
511                    mesh_jobs.push(MeshJob {
512                        entity_id: entity.id,
513                        entity_name: entity.name.clone(),
514                        def,
515                    });
516                }
517                None => {
518                    eprintln!(
519                        "Failed to parse spline_mesh on entity '{}'",
520                        entity.name
521                    );
522                }
523            }
524        }
525    }
526
527    for job in &mesh_jobs {
528        let spline_idx = match spline_lookup.get(job.def.spline_entity.as_str()) {
529            Some(&idx) => idx,
530            None => {
531                eprintln!(
532                    "spline_mesh '{}' references unknown spline entity '{}'",
533                    job.entity_name, job.def.spline_entity
534                );
535                continue;
536            }
537        };
538        let info = &spline_infos[spline_idx];
539
540        let (verts, indices, phys_verts, phys_tris) = generate_cross_section_mesh(
541            &info.samples,
542            info.closed,
543            job.def.width,
544            job.def.height,
545            job.def.offset,
546            job.def.color,
547            job.def.stripe_color,
548            job.def.stripe_width,
549            info.spacing,
550        );
551
552        if verts.is_empty() {
553            continue;
554        }
555
556        println!(
557            "  Mesh '{}': {}x{} @ offset [{:.1}, {:.1}] → {} verts, {} tris",
558            job.entity_name,
559            job.def.width,
560            job.def.height,
561            job.def.offset[0],
562            job.def.offset[1],
563            verts.len(),
564            phys_tris.len(),
565        );
566
567        // Upload render mesh
568        let material = ImportedMaterial {
569            name: job.entity_name.clone(),
570            base_color: job.def.color,
571            metallic: job.def.metallic,
572            roughness: job.def.roughness,
573            base_color_texture: None,
574            normal_texture: None,
575            metallic_roughness_texture: None,
576            use_vertex_color: job.def.stripe_color.is_some(),
577            alpha_mode: flint_import::AlphaMode::Opaque,
578            alpha_cutoff: 0.5,
579        };
580        renderer.load_procedural_mesh(device, &job.entity_name, &verts, &indices, material);
581
582        // Register trimesh collider (when physics is available)
583        if let Some(ref mut phys) = physics {
584            phys.sync.register_static_trimesh(
585                job.entity_id,
586                &mut phys.physics_world,
587                phys_verts,
588                phys_tris,
589                job.def.friction,
590                job.def.restitution,
591            );
592        }
593
594        // Set model.asset so renderer draws it
595        set_model_asset(world, job.entity_id, &job.entity_name);
596    }
597
598    if !spline_infos.is_empty() || !mesh_jobs.is_empty() {
599        println!(
600            "Spline generation complete: {} splines, {} meshes.",
601            spline_infos.len(),
602            mesh_jobs.len(),
603        );
604    }
605}
