1//! Player extensions: Rust code an embedding game installs into
2//! `PlayerApp` (the first step toward the roadmap's plugin system; ADR 0071).
3//!
4//! A game that wraps `PlayerApp` in its own binary can add native Rhai
5//! functions, run per-frame Rust before the scripts, and see window events
6//! before the player does. Every method has a no-op default, and a player
7//! with no extensions behaves exactly as it did before they existed.
8//!
9//! ```ignore
10//! struct MyGame { /* shared state */ }
11//!
12//! impl PlayerExtension for MyGame {
13//!     fn register_script_api(&mut self, engine: &mut rhai::Engine) {
14//!         engine.register_fn("my_score", || 42_i64);
15//!     }
16//!     fn pre_update(&mut self, ctx: &mut ExtensionCtx) {
17//!         // apply network state to ctx.world before on_update runs
18//!     }
19//! }
20//!
21//! player.add_extension(Box::new(MyGame { .. }));
22//! ```
23

33/// Hooks an embedding game installs with `PlayerApp::add_extension`.
34///
35/// Extensions run in the order they were added. All methods default to
36/// doing nothing, so an extension implements only what it needs.
37pub trait PlayerExtension {
38    /// Register native functions, types or modules on the script engine.
39    ///
40    /// Called once, from `add_extension`. The engine outlives scene
41    /// transitions, so registrations persist for the whole run. Flint's own
42    /// API is registered first; a function registered here with the same
43    /// name and parameter types replaces the engine's.
44    ///
45    /// The engine uses rhai's `sync` feature, so registered closures and
46    /// the state they capture must be `Send + Sync` (`Arc<Mutex<_>>`,
47    /// atomics).
48    fn register_script_api(&mut self, _engine: &mut rhai::Engine) {}
49
50    /// Per-frame Rust, run after the clock ticks and before the scripts'
51    /// `on_update`, so anything written to the world here is visible to
52    /// scripts in the same frame.
53    fn pre_update(&mut self, _ctx: &mut ExtensionCtx) {}
54
55    /// See a window event before the player does. Return `true` to consume
56    /// it. Only input events (keyboard, IME, mouse buttons, wheel, cursor
57    /// movement, touch) can be consumed; lifecycle events such as
58    /// `Resized`, `CloseRequested` and `RedrawRequested` always reach the
59    /// player whatever this returns. The first extension to consume an
60    /// event stops it reaching later extensions.
61    ///
62    /// egui sees events first (see `ui`): input it wants, such as typing
63    /// in a focused text field or a click on an egui window, never arrives
64    /// here.
65    fn on_window_event(&mut self, _event: &WindowEvent) -> bool {
66        false
67    }
68
69    /// See a key go down or up before the player does: a real one, or one
70    /// an `--input-script` plays (ADR 0103). Return `true` to consume it,
71    /// as for `on_window_event`.
72    ///
73    /// A scripted key has no winit `KeyEvent` (nothing outside winit can
74    /// build one), so it never reaches `on_window_event`. Read keys here and
75    /// a capture can press them. A real key comes here straight after this
76    /// extension's `on_window_event` passes on its `KeyboardInput`, so
77    /// handle a key in one hook or the other, not both. Keys winit can't
78    /// name (`PhysicalKey::Unidentified`), and the synthetic presses winit
79    /// sends for keys held as the window gains focus, come only to
80    /// `on_window_event`.
81    ///
82    /// egui sees keys first, as it does window events: a key typed into a
83    /// focused field never arrives here.
84    fn on_key(&mut self, _key: &KeyInput) -> bool {
85        false
86    }
87
88    /// Draw the game's own egui UI. Called every frame from inside the
89    /// player's egui pass, after the scripts' `on_draw_ui` and before the
90    /// debug panels, so it draws over the script HUD and under the panels.
91    ///
92    /// Installing any extension routes window events through egui first.
93    /// An event egui wants is consumed, and extensions, the player and
94    /// scripts never see it: keys and IME while a widget has keyboard
95    /// focus (a `TextEdit` being typed in, so Escape there unfocuses the
96    /// field rather than quitting), and clicks and the wheel over an egui
97    /// area. Clicks on empty space still reach the game.
98    ///
99    /// Use windows and areas, or side and top panels. A `CentralPanel`
100    /// covers the whole window, so it takes every click from the game.
101    fn ui(&mut self, _ctx: &egui::Context) {}
102
103    /// Add to or change egui's fonts before the player installs them
104    /// (ADR 0081). `defs` already holds the project's `fonts/` (each file
105    /// under its stem, the `fonts.toml` aliases, and its `[egui]` defaults).
106    ///
107    /// Called whenever the player sets egui's fonts: at start, on a scene
108    /// change into another project root, and when an extension is added
109    /// after the fonts are in. Fonts set here survive those reloads, which
110    /// fonts set with `Context::set_fonts` from `ui` do not. A named family
111    /// added here is one scripts can draw in too.
112    ///
113    /// Families exist from the first egui pass after the call, so laying
114    /// text out in one from `ui` is safe on every frame.
115    fn fonts(&mut self, _defs: &mut egui::FontDefinitions) {}
116
117    /// Name the game's own debug panels (ADR 0107). Each joins the
118    /// engine's (grass, ocean, camera, ...): F3 opens and closes it with
119    /// them, and it docks in their columns under its name. The player asks
120    /// when it builds its panels, on each scene load and when the extension
121    /// is added, so a name given later appears at the next scene load.
122    /// Empty, the default, adds none.
123    ///
124    /// Only a player built with its `debug-hud` feature (the default) has
125    /// debug panels. Without it this is never called.
126    fn debug_panels(&self) -> Vec<String> {
127        Vec::new()
128    }
129
130    /// Draw the contents of the debug panel `name`, one that
131    /// [`debug_panels`](Self::debug_panels) named. Called every frame while
132    /// the panel is open, after [`ui`](Self::ui), so it reads the game's
133    /// state as `ui` left it. The heading and the scrolling are the
134    /// player's.
135    fn debug_panel_ui(&mut self, _name: &str, _ui: &mut egui::Ui) {}
136}

138/// What an extension can reach during `pre_update`.
139pub struct ExtensionCtx<'a> {
140    pub world: &'a mut FlintWorld,
141    /// Already ticked for this frame.
142    pub clock: &'a GameClock,
143    /// This frame's input, before scripts read it.
144    pub input: &'a InputState,
145    /// Last frame's camera; scripts and physics may move it later this frame.
146    pub camera: &'a Camera,
147    /// The renderer and GPU handles, for uploading meshes and textures.
148    /// `None` until the window and renderer exist.
149    pub gpu: Option<ExtensionGpu<'a>>,
150    /// The store scripts read and write with `persist_get` / `persist_set`
151    /// (`save/persist.toml`; ADR 0080). Writes here are visible to scripts
152    /// this frame, and join the player's save: one second after a burst of
153    /// changes, on scene transitions and on exit. Call
154    /// [`PersistentStore::request_flush`] to save at the end of this frame
155    /// instead. Scripts can read every key, so keep secrets out of it.
156    pub persist: &'a mut PersistentStore,
157    /// The player's audio: play sounds, load sounds made in memory, and set
158    /// the mixer buses (ADR 0084). See [`ExtensionAudio`].
159    pub audio: ExtensionAudio<'a>,
160    /// The meshes uploaded as ray targets ([`MeshOptions::ray_target`]),
161    /// where the last frame drew them (ADR 0105): cast a ray or a segment
162    /// through it to find the first thing drawn along it. Empty until the
163    /// first frame is drawn. Scripts cast through the same scene.
164    pub rays: std::sync::Arc<flint_core::ray_mesh::RayScene>,
165    /// Events fired with `fire_event`, queued for this frame's scripts.
166    pub(crate) fired_events: &'a mut Vec<GameEvent>,
167    pub(crate) cursor_captured: bool,
168}
169
170impl ExtensionCtx<'_> {
171    /// Whether the cursor is captured now (ADR 0110), as scripts'
172    /// `is_cursor_captured()` reads it. While it is, `input.mouse_position`
173    /// is the window centre and clicks skip egui.
174    pub fn is_cursor_captured(&self) -> bool {
175        self.cursor_captured
176    }
177
178    /// Fire a custom game event, as a script's `fire_event_data` does.
179    ///
180    /// It joins this frame's events, so scripts' `on_event(name, data)`
181    /// receives it before their `on_update` runs this frame. Audio triggers
182    /// see it too. A table `data` arrives in scripts as a map; anything else
183    /// arrives as `#{ value: data }`.
184    ///
185    /// ```ignore
186    /// let mut data = toml::Table::new();
187    /// data.insert("count".into(), 3.into());
188    /// ctx.fire_event("inventory_changed", data);
189    /// ```
190    pub fn fire_event(&mut self, name: impl Into<String>, data: impl Into<toml::Value>) {
191        self.fired_events.push(GameEvent::Custom {
192            name: name.into(),
193            data: data.into(),
194        });
195    }
196
197    /// Draw the runtime mesh `mesh` at `entity`'s transform, by setting the
198    /// entity's `model` to `{ asset = mesh, runtime = true }`. The mesh may be
199    /// uploaded before or after (see [`ExtensionGpu::upload_mesh`]); until it
200    /// exists the entity draws nothing. Any other `model` fields are kept.
201    pub fn attach_mesh(&mut self, entity: EntityId, mesh: &str) -> flint_core::Result<()> {
202        let mut model = toml::Table::new();
203        model.insert("asset".into(), mesh.into());
204        model.insert("runtime".into(), true.into());
205        self.world
206            .merge_component(entity, flint_core::components::MODEL, model.into())
207    }
208}

210/// The player's audio, as an extension reaches it (ADR 0084).
211///
212/// Sounds are cached by name, the same cache scripts' `play_sound` and
213/// `audio_source.file` read: a sound loaded here can be played from a
214/// script, and a file the scene preloaded (`audio/<file>`) can be played
215/// from here. Loading needs no audio device; playing without one does
216/// nothing, so a game needn't check.
217///
218/// ```ignore
219/// // Once: a sound made in code, mono at 44.1 kHz.
220/// ctx.audio.load_samples("blip", 44_100, &[&samples])?;
221/// // Any frame:
222/// ctx.audio.play("blip", &SoundOptions::new().volume(0.5).pitch(1.2))?;
223/// ctx.audio.play("audio/door.ogg", &SoundOptions::new().at(pos).distances(2.0, 60.0))?;
224/// // Heard from far off, fading like a sound in the open:
225/// ctx.audio.play("bell", &SoundOptions::new().at(pos).distances(10.0, 400.0).rolloff(0.7))?;
226/// ctx.audio.set_bus_volume(Bus::Music, 0.4);
227/// ```
228pub struct ExtensionAudio<'a> {
229    pub(crate) system: &'a mut AudioSystem,
230}
231
232impl<'a> ExtensionAudio<'a> {
233    pub(crate) fn new(system: &'a mut AudioSystem) -> Self {
234        Self { system }
235    }
236
237    /// Whether there's a device to play on. Without one, every sound is
238    /// silent, though loading still works.
239    pub fn is_available(&self) -> bool {
240        self.system.engine.is_available()
241    }
242
243    /// Whether a sound is cached under `name`.
244    pub fn has_sound(&self, name: &str) -> bool {
245        self.system.engine.has_sound(name)
246    }
247
248    /// Cache a sound made in memory: one channel (heard on both sides) or
249    /// two (left, right) of samples in -1..1 at `sample_rate` Hz. Replaces
250    /// any sound cached under `name`; one already playing plays on.
251    ///
252    /// A looping `audio_source` naming a sound that isn't cached yet stays
253    /// silent, so load before spawning the entity that plays it.
254    pub fn load_samples(
255        &mut self,
256        name: &str,
257        sample_rate: u32,
258        channels: &[&[f32]],
259    ) -> flint_core::Result<()> {
260        self.system
261            .engine
262            .load_sound_from_samples(name, sample_rate, channels)
263    }
264
265    /// Cache an encoded sound held in memory (WAV, OGG, FLAC or MP3), such
266    /// as one from `include_bytes!`. Replaces any sound cached under `name`.
267    pub fn load_bytes(&mut self, name: &str, bytes: Vec<u8>) -> flint_core::Result<()> {
268        self.system.engine.load_sound_from_bytes(name, bytes)
269    }
270
271    /// Play the cached sound `name` once. An unknown name is an error;
272    /// with no device it does nothing.
273    pub fn play(&mut self, name: &str, options: &SoundOptions) -> flint_core::Result<()> {
274        let engine = &mut self.system.engine;
275        if !engine.has_sound(name) {
276            return Err(flint_core::FlintError::AudioError(format!(
277                "Sound not cached: {name}"
278            )));
279        }
280        if !engine.is_available() {
281            return Ok(());
282        }
283        match options.at {
284            Some(at) => engine.play_oneshot_at(
285                name,
286                at,
287                options.volume,
288                options.pitch,
289                options.distances,
290                options.rolloff,
291                options.bus,
292            ),
293            None => engine
294                .play_non_spatial(name, options.volume, options.pitch, false, options.bus)
295                .map(drop),
296        }
297    }
298
299    /// Set a mixer bus's volume (linear: 0 mutes, 1 is unity). It scales
300    /// everything on the bus, scripts' sounds and `audio_source`s included,
301    /// on top of their own volumes. It stays set across scene transitions.
302    pub fn set_bus_volume(&mut self, bus: Bus, volume: f64) {
303        self.system.set_bus_volume(bus, volume);
304    }
305}

307/// How [`ExtensionAudio::play`] plays a sound.
308#[derive(Debug, Clone, Copy, PartialEq)]
309pub struct SoundOptions {
310    /// Linear: 1 is the sound as made.
311    pub volume: f64,
312    /// Playback rate: 2 is an octave up and twice as fast.
313    pub pitch: f64,
314    pub bus: Bus,
315    /// Where the sound is, world units, so it pans and fades with distance
316    /// from the listener. `None` plays it flat, from nowhere.
317    pub at: Option<Vec3>,
318    /// For a placed sound, `(min, max)` from the listener: full volume
319    /// within `min`, silent past `max`.
320    pub distances: (f32, f32),
321    /// The curve between them (see [`flint_audio::engine::DEFAULT_ROLLOFF`]):
322    /// 2 holds the sound up and drops it near `max`; below 1 it falls away
323    /// soon after `min` and tails off, like a real sound in the open.
324    pub rolloff: f32,
325}
326
327impl Default for SoundOptions {
328    fn default() -> Self {
329        Self {
330            volume: 1.0,
331            pitch: 1.0,
332            bus: Bus::Sfx,
333            at: None,
334            distances: flint_audio::engine::DEFAULT_DISTANCES,
335            rolloff: flint_audio::engine::DEFAULT_ROLLOFF,
336        }
337    }
338}
339
340impl SoundOptions {
341    /// Full volume, normal pitch, on the sfx bus, from nowhere.
342    pub fn new() -> Self {
343        Self::default()
344    }
345    pub fn volume(self, volume: f64) -> Self {
346        Self { volume, ..self }
347    }
348    pub fn pitch(self, pitch: f64) -> Self {
349        Self { pitch, ..self }
350    }
351    pub fn bus(self, bus: Bus) -> Self {
352        Self { bus, ..self }
353    }
354    /// Place the sound at `at`.
355    pub fn at(self, at: Vec3) -> Self {
356        Self {
357            at: Some(at),
358            ..self
359        }
360    }
361    /// The falloff of a placed sound (default 1 to 25).
362    pub fn distances(self, min: f32, max: f32) -> Self {
363        Self {
364            distances: (min, max),
365            ..self
366        }
367    }
368    /// The falloff curve of a placed sound (default 2).
369    pub fn rolloff(self, rolloff: f32) -> Self {
370        Self { rolloff, ..self }
371    }
372}

374/// The renderer and the device and queue it draws with.
375pub struct ExtensionGpu<'a> {
376    pub renderer: &'a mut SceneRenderer,
377    pub device: &'a wgpu::Device,
378    pub queue: &'a wgpu::Queue,
379}
380
381impl ExtensionGpu<'_> {
382    /// Create or replace the runtime mesh `name`. Entities whose
383    /// `model.asset` is `name` draw it this frame (see
384    /// [`ExtensionCtx::attach_mesh`]). Replacing frees the old GPU buffers
385    /// once this frame's draw lists let go of them, so a mesh can be rebuilt
386    /// as often as its data changes.
387    ///
388    /// The mesh is culled against the camera by the bounds of `vertices`.
389    /// Use [`ImportedMaterial::vertex_colored`] to colour per vertex; vertex
390    /// colours are linear, so convert sRGB with `flint_render::srgb_to_linear`.
391    pub fn upload_mesh(
392        &mut self,
393        name: &str,
394        vertices: &[Vertex],
395        indices: &[u32],
396        material: ImportedMaterial,
397    ) {
398        self.renderer
399            .load_procedural_mesh(self.device, name, vertices, indices, material);
400    }
401
402    /// [`upload_mesh`](Self::upload_mesh), with [`MeshOptions`]: a mesh
403    /// uploaded with `ray_target` is met by rays cast through
404    /// [`ExtensionCtx::rays`] and by scripts' `raycast_meshes` and
405    /// `screen_to_mesh`, wherever an entity draws it (ADR 0105). An upload
406    /// without it, or [`remove_mesh`](Self::remove_mesh), stops it being one.
407    pub fn upload_mesh_with(
408        &mut self,
409        name: &str,
410        vertices: &[Vertex],
411        indices: &[u32],
412        material: ImportedMaterial,
413        options: MeshOptions,
414    ) {
415        self.upload_mesh(name, vertices, indices, material);
416        if options.ray_target {
417            let positions: Vec<[f32; 3]> = vertices.iter().map(|v| v.position).collect();
418            let mesh = flint_core::ray_mesh::RayMesh::new(&positions, indices);
419            self.renderer
420                .set_ray_mesh(name, Some(std::sync::Arc::new(mesh)));
421        }
422    }
423
424    /// Remove the runtime mesh `name`; entities naming it draw nothing
425    /// until it is uploaded again. Returns whether it existed.
426    pub fn remove_mesh(&mut self, name: &str) -> bool {
427        self.renderer.remove_mesh(name)
428    }
429
430    /// Grow compute grass on the game's own ground: a density mask and a
431    /// height over a world rectangle (ADR 0089). Replaces any grass loaded,
432    /// a terrain's included. Call again whenever the ground changes, and
433    /// [`set_grass_config`](Self::set_grass_config) to retune it without
434    /// re-uploading the mask. Errs only when the mask or heights don't
435    /// match their stated sizes; on a device with no compute grass it does
436    /// nothing.
437    pub fn set_grass_ground(
438        &mut self,
439        config: &flint_terrain::GrassConfig,
440        ground: &GrassGround,
441    ) -> Result<(), String> {
442        self.renderer
443            .set_grass_ground(self.device, self.queue, config, ground)
444    }
445
446    /// Change the grass's look (colours, wind, density, reach) and keep its
447    /// ground. The instance buffer grows on the next frame if it must.
448    pub fn set_grass_config(&mut self, config: flint_terrain::GrassConfig) {
449        self.renderer.set_grass_config(config);
450    }
451
452    /// Repaint part of the grass ground's mask in place (ADR 0093): the
453    /// `width × height` texels at (`x`, `y`), from `bytes`, one a texel,
454    /// row by row toward +Z. Much cheaper than a whole
455    /// [`set_grass_ground`](Self::set_grass_ground) for a small change,
456    /// such as a clearing round something placed. Errs when no ground is
457    /// set, or the rectangle runs off the mask or `bytes` is the wrong
458    /// length; on a device with no compute grass it does nothing.
459    pub fn update_grass_mask(
460        &mut self,
461        x: u32,
462        y: u32,
463        width: u32,
464        height: u32,
465        bytes: &[u8],
466    ) -> Result<(), String> {
467        self.renderer
468            .update_grass_mask(self.queue, x, y, width, height, bytes)
469    }
470
471    /// Remove the grass, whatever ground it grows on.
472    pub fn clear_grass(&mut self) {
473        self.renderer.unload_grass();
474    }
475
476    /// Lay a ground fog over a world rectangle (ADR 0106): each texel of
477    /// the map has a floor and an amount, and the fog lies on that floor,
478    /// thinning upward, so a mist can fill valleys at many heights where
479    /// the height fog is one level layer. Replaces any set. How it looks
480    /// is the post-process config's `ground_fog_*` (it shows only while
481    /// `ground_fog_density` is over 0), so set the map once, and again
482    /// when the ground changes, and change the look as often as you like.
483    /// Errs when the map's texels don't fill its size, or a value isn't
484    /// finite; a map with no fog on it clears it.
485    pub fn set_ground_fog(&mut self, fog: &crate::GroundFog) -> Result<(), String> {
486        self.renderer.set_ground_fog(self.device, self.queue, fog)
487    }
488
489    /// Remove the ground fog. Returns whether one was set.
490    pub fn clear_ground_fog(&mut self) -> bool {
491        self.renderer.clear_ground_fog()
492    }
493
494    /// Draw water on the game's own mesh (ADR 0094): a lake at its own
495    /// level, a river stepping down a valley. It shades as the `ocean` does
496    /// (waves, Fresnel, refraction, the cel ramp, shoreline foam), with the
497    /// body's tint, wave scale and flow; fast flow foams white. Build the
498    /// mesh with [`WaterBodyMesh::lake`](crate::WaterBodyMesh::lake) or
499    /// [`river`](crate::WaterBodyMesh::river), or supply your own. Replaces
500    /// any body set with this `id`. Errs when the mesh's indices are bad.
501    pub fn set_water_body(
502        &mut self,
503        id: &str,
504        mesh: &crate::WaterBodyMesh,
505        look: crate::WaterBodyLook,
506    ) -> Result<(), String> {
507        self.renderer.set_water_body(self.device, id, mesh, look)
508    }
509
510    /// Change a water body's look and keep its mesh. Returns whether it existed.
511    pub fn set_water_body_look(&mut self, id: &str, look: crate::WaterBodyLook) -> bool {
512        self.renderer.set_water_body_look(id, look)
513    }
514
515    /// Remove the water body set with `id`. Returns whether it existed.
516    pub fn remove_water_body(&mut self, id: &str) -> bool {
517        self.renderer.remove_water_body(id)
518    }
519
520    /// Remove every water body the game set (a scene's `water_body`
521    /// components stay).
522    pub fn clear_water_bodies(&mut self) {
523        self.renderer.clear_water_bodies();
524    }
525}

527/// How a runtime mesh takes part beyond being drawn, for
528/// [`ExtensionGpu::upload_mesh_with`].
529#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
530pub struct MeshOptions {
531    /// Rays meet it (ADR 0105): the camera can keep out of it and a click
532    /// can pick it. Its triangles are kept on the CPU in a bounding volume
533    /// hierarchy, built at upload, so mark only what rays need to meet.
534    pub ray_target: bool,
535}

537/// Whether an extension may consume `event`. Lifecycle events must always
538/// reach the player, or a consuming extension could stall the frame loop
539/// or leave the surface the wrong size.
540pub(crate) fn is_consumable(event: &WindowEvent) -> bool {
541    matches!(
542        event,
543        WindowEvent::KeyboardInput { .. }
544            | WindowEvent::Ime(_)
545            | WindowEvent::MouseInput { .. }
546            | WindowEvent::MouseWheel { .. }
547            | WindowEvent::CursorMoved { .. }
548            | WindowEvent::Touch(_)
549    )
550}

552/// A key going down or up, for [`PlayerExtension::on_key`]: a real one or
553/// one an `--input-script` plays.
554#[derive(Debug, Clone, Copy, PartialEq, Eq)]
555pub struct KeyInput<'a> {
556    /// The key where it sits on the keyboard, whatever the layout.
557    pub key: KeyCode,
558    pub state: ElementState,
559    /// The OS repeating a held key. Never set on a scripted key.
560    pub repeat: bool,
561    /// The character the key stands for (winit's `logical_key`, when it's
562    /// a character), on the release as on the press, and under Ctrl too:
563    /// `?` for Shift and Slash on a US layout. A scripted key is read on a
564    /// US layout.
565    pub character: Option<&'a str>,
566    /// The modifiers held as the key moved.
567    pub modifiers: ModifiersState,
568    /// Whether an `--input-script` pressed it.
569    pub scripted: bool,
570}
571
572impl KeyInput<'_> {
573    /// A first press: down, and not the OS repeating it.
574    pub fn pressed(&self) -> bool {
575        self.state == ElementState::Pressed && !self.repeat
576    }
577}
578
579/// The `KeyInput` a real `KeyboardInput` is, if `on_key` should see it.
580fn key_input<'a>(event: &'a WindowEvent, modifiers: ModifiersState) -> Option<KeyInput<'a>> {
581    let WindowEvent::KeyboardInput {
582        event,
583        is_synthetic: false,
584        ..
585    } = event
586    else {
587        return None;
588    };
589    let PhysicalKey::Code(key) = event.physical_key else {
590        return None;
591    };
592    let character = match &event.logical_key {
593        Key::Character(c) => Some(c.as_str()),
594        _ => None,
595    };
596    Some(KeyInput {
597        key,
598        state: event.state,
599        repeat: event.repeat,
600        character,
601        modifiers,
602        scripted: false,
603    })
604}
605
606/// Offer `event` to each extension in order. Returns whether one consumed
607/// it; the rest are skipped once one has. Non-consumable events go to
608/// every extension and are never reported as consumed. A key goes to each
609/// extension's `on_window_event` and then its `on_key`, with `modifiers`
610/// the ones held now.
611pub(crate) fn dispatch_window_event(
612    extensions: &mut [Box<dyn PlayerExtension>],
613    event: &WindowEvent,
614    modifiers: ModifiersState,
615) -> bool {
616    let consumable = is_consumable(event);
617    let key = key_input(event, modifiers);
618    for ext in extensions.iter_mut() {
619        if ext.on_window_event(event) && consumable {
620            return true;
621        }
622        if key.as_ref().is_some_and(|key| ext.on_key(key)) {
623            return true;
624        }
625    }
626    false
627}
628
629/// Offer a scripted key to each extension's `on_key` in order. Returns
630/// whether one consumed it.
631pub(crate) fn dispatch_key(extensions: &mut [Box<dyn PlayerExtension>], key: &KeyInput) -> bool {
632    extensions.iter_mut().any(|ext| ext.on_key(key))
633}

680            self.log

682                .unwrap()

684        }

686            self.log

691        }
692        fn on_key(&mut self, key: &KeyInput) -> bool {
693            self.log.lock().unwrap().push(format!(
694                "{} key {:?} {:?}",

697            self.consume
698        }
699    }
