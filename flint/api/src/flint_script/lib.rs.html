1//! Flint Script — Rhai scripting system
2//!
3//! Integrates the Rhai scripting engine into the Flint game loop:
4//! - `ScriptEngine` — compiles and runs .rhai scripts, manages per-entity state
5//! - `ScriptSync` — discovers entities with `script` component, handles hot-reload
6//! - `ScriptSystem` — implements `RuntimeSystem` for game loop integration
7//!
8//! Scripts can read/write ECS data, respond to collisions/triggers/input,
9//! control animation and audio, and hot-reload while the game is running.

97    /// Provide input and timing context for the current frame.
98    /// Called by PlayerApp before update().
99    pub fn provide_context(
100        &mut self,
101        input: &InputState,
102        events: &[GameEvent],
103        total_time: f64,
104        delta_time: f64,
105    ) {
106        // Snapshot input state
107        let touches: Vec<(i64, f64, f64)> = input
108            .active_touches()
109            .values()
110            .map(|t| (t.id as i64, t.position.0, t.position.1))
111            .collect();
112        let touch_just_started: Vec<(i64, f64, f64)> = input
113            .touches_just_started_set()
114            .iter()
115            .filter_map(|id| input.touch_position(*id).map(|(x, y)| (*id as i64, x, y)))
116            .collect();
117        let touch_just_ended: Vec<i64> = input
118            .touches_just_ended_set()
119            .iter()
120            .map(|id| *id as i64)
121            .collect();
122        let touch_taps: Vec<(f64, f64)> = input.touch_taps().to_vec();
123        let touch_swipes: Vec<(String, f64, f64)> = input
124            .touch_swipes()
125            .iter()
126            .map(|(dir, x, y)| (format!("{dir:?}").to_lowercase(), *x, *y))
127            .collect();
128
129        let snapshot = InputSnapshot {
130            actions_pressed: snapshot_actions(input, true),
131            actions_just_pressed: snapshot_actions(input, false),
132            actions_just_released: snapshot_actions_released(input),
133            action_values: snapshot_action_values(input),
134            any_just_pressed: input.any_just_pressed(),
135            last_device: input.last_device().as_str().to_string(),
136            gamepad_connected: input.gamepad_connected(),
137            touch_controls_visible: input.touch_controls_visible(),
138            mouse_delta: input.raw_mouse_delta(),
139            mouse_wheel: {
140                let (x, y) = input.mouse_wheel_delta();
141                (x as f64, y as f64)
142            },
143            mouse_buttons_down: (0..3_u32)
144                .filter(|b| input.is_mouse_button_down(*b))
145                .map(|b| b as i64)
146                .collect(),
147            mouse_buttons_just_pressed: (0..3_u32)
148                .filter(|b| input.is_mouse_button_just_pressed(*b))
149                .map(|b| b as i64)
150                .collect(),
151            touches,
152            touch_just_started,
153            touch_just_ended,
154            touch_taps,
155            touch_swipes,
156        };
157
158        self.engine
159            .provide_context(snapshot, delta_time, total_time);
160        self.pending_events = events.to_vec();
161    }

281    /// Load a script file for a specific entity (for chunk loading).
282    /// The entity must already have a `script` component in the ECS.
283    pub fn load_script_for_entity(
284        &mut self,
285        entity_id: flint_core::EntityId,
286        script_path: &std::path::Path,
287    ) {
288        match self.engine.compile_file(script_path) {
289            Ok(ast) => {
290                let source = script_path
291                    .file_name()
292                    .and_then(|f| f.to_str())
293                    .unwrap_or("unknown.rhai")
294                    .to_string();
295                println!("[script] Loaded (chunk): {:?} → {}", entity_id, source);
296                self.engine.add_script(entity_id, ast, source);
297                self.sync.discovered.insert(entity_id);
298            }
299            Err(e) => {
300                tracing::warn!("Compile error for chunk entity {:?}: {}", entity_id, e);
301            }
302        }
303    }

512impl RuntimeSystem for ScriptSystem {
513    fn initialize(&mut self, world: &mut FlintWorld) -> Result<()> {
514        // Discover scripts from world
515        self.sync.discover_and_load(world, &mut self.engine);
516
517        // Call on_init() for all loaded scripts
518        self.engine.call_inits(world);
519
520        let count = self.engine.scripts.len();
521        if count > 0 {
522            println!("Script system initialized ({} scripts)", count);
523        }
524
525        Ok(())
526    }
527
528    fn fixed_update(&mut self, _world: &mut FlintWorld, _dt: f64) -> Result<()> {
529        // Scripts run at variable rate (like animation), not fixed timestep
530        Ok(())
531    }
532
533    fn update(&mut self, world: &mut FlintWorld, _dt: f64) -> Result<()> {
534        // A perf readout times each step below, and each script's on_update.
535        self.engine.clear_update_times();
536        let timing = self.engine.timing();
537        let now = || timing.then(web_time::Instant::now);
538
539        // Check for hot-reloaded scripts
540        let began = now();
541        self.sync.check_hot_reload(&mut self.engine);
542
543        // Check for edited .ui.toml / .style.toml / token files
544        crate::lock_or_recover(&self.engine.ctx)
545            .ui_system
546            .poll_reload();
547        if let Some(began) = began {
548            self.engine.add_update_time("[hot reload]", began);
549        }
550
551        // Discover any new script entities
552        let began = now();
553        self.sync.discover_and_load(world, &mut self.engine);
554
555        // Call on_init for newly added scripts
556        self.engine.call_inits(world);
557        if let Some(began) = began {
558            self.engine.add_update_time("[load, on_init]", began);
559        }
560
561        // Process events (collision, trigger, action callbacks)
562        let began = now();
563        let events = std::mem::take(&mut self.pending_events);
564        self.engine.process_events(&events, world);
565        if let Some(began) = began {
566            self.engine.add_update_time("[events]", began);
567        }
568
569        // Call on_update() for all scripts
570        self.engine.call_updates(world);
571
572        Ok(())
573    }
574
575    fn shutdown(&mut self) -> Result<()> {
576        let count = self.engine.scripts.len();
577        if count > 0 {
578            println!("Script system shut down ({} scripts)", count);
579        }
580        Ok(())
581    }

584        "script"
585    }

588/// Snapshot which actions are pressed/just-pressed from InputState.
589/// Dynamically iterates all registered action names instead of a hard-coded list.
590fn snapshot_actions(input: &InputState, pressed: bool) -> std::collections::HashSet<String> {
591    let mut set = std::collections::HashSet::new();
592    for action in input.action_names() {
593        let active = if pressed {
594            input.is_action_pressed(&action)
595        } else {
596            input.is_action_just_pressed(&action)
597        };
598        if active {
599            set.insert(action);
600        }
601    }
602    set
603}
604
605fn snapshot_actions_released(input: &InputState) -> std::collections::HashSet<String> {
606    let mut set = std::collections::HashSet::new();
607    for action in input.actions_just_released() {
608        set.insert(action);
609    }
610    set
611}
612
613fn snapshot_action_values(input: &InputState) -> std::collections::HashMap<String, f64> {
614    let mut map = std::collections::HashMap::new();
615    for action in input.action_names() {
616        let val = input.action_value(&action);
617        if val.abs() > 0.001 {
618            map.insert(action, val as f64);
619        }
620    }
621    map
622}

699/// Lock a shared script-state mutex, recovering from poisoning instead of
700/// propagating the panic (ADR 0039). The guarded values are always-valid
701/// snapshots written whole each frame, so the last-written state is safe to
702/// read after a writer panicked; propagating would turn one panic into a
703/// panic on every later script call, contradicting the crate's
704/// log-and-continue policy. Logs once per process.
705pub(crate) fn lock_or_recover<T>(m: &std::sync::Mutex<T>) -> std::sync::MutexGuard<'_, T> {
706    m.lock().unwrap_or_else(|poisoned| {
707        static LOGGED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
708        if !LOGGED.swap(true, std::sync::atomic::Ordering::Relaxed) {
709            tracing::error!(
710                "script context mutex poisoned (a thread panicked holding it);                  recovering with the last-written snapshot (ADR 0039)"
711            );
712        }
713        poisoned.into_inner()
714    })
