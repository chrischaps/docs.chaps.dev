1//! Flint Script — Rhai scripting system
2//!
3//! Integrates the Rhai scripting engine into the Flint game loop:
4//! - `ScriptEngine` — compiles and runs .rhai scripts, manages per-entity state
5//! - `ScriptSync` — discovers entities with `script` component, handles hot-reload
6//! - `ScriptSystem` — implements `RuntimeSystem` for game loop integration
7//!
8//! Scripts can read/write ECS data, respond to collisions/triggers/input,
9//! control animation and audio, and hot-reload while the game is running.
10

49    /// Provide input and timing context for the current frame.
50    /// Called by PlayerApp before update().
51    pub fn provide_context(
52        &mut self,
53        input: &InputState,
54        events: &[GameEvent],
55        total_time: f64,
56        delta_time: f64,
57    ) {
58        // Snapshot input state
59        let snapshot = InputSnapshot {
60            actions_pressed: snapshot_actions(input, true),
61            actions_just_pressed: snapshot_actions(input, false),
62            actions_just_released: snapshot_actions_released(input),
63            action_values: snapshot_action_values(input),
64            mouse_delta: input.raw_mouse_delta(),
65        };
66
67        self.engine.provide_context(snapshot, delta_time, total_time);
68        self.pending_events = events.to_vec();
69    }

189impl RuntimeSystem for ScriptSystem {
190    fn initialize(&mut self, world: &mut FlintWorld) -> Result<()> {
191        // Discover scripts from world
192        self.sync.discover_and_load(world, &mut self.engine);
193
194        // Call on_init() for all loaded scripts
195        self.engine.call_inits(world);
196
197        let count = self.engine.scripts.len();
198        if count > 0 {
199            println!("Script system initialized ({} scripts)", count);
200        }
201
202        Ok(())
203    }
204
205    fn fixed_update(&mut self, _world: &mut FlintWorld, _dt: f64) -> Result<()> {
206        // Scripts run at variable rate (like animation), not fixed timestep
207        Ok(())
208    }
209
210    fn update(&mut self, world: &mut FlintWorld, _dt: f64) -> Result<()> {
211        // Check for hot-reloaded scripts
212        self.sync.check_hot_reload(&mut self.engine);
213
214        // Discover any new script entities
215        self.sync.discover_and_load(world, &mut self.engine);
216
217        // Call on_init for newly added scripts
218        self.engine.call_inits(world);
219
220        // Process events (collision, trigger, action callbacks)
221        let events = std::mem::take(&mut self.pending_events);
222        self.engine.process_events(&events, world);
223
224        // Call on_update() for all scripts
225        self.engine.call_updates(world);
226
227        Ok(())
228    }
229
230    fn shutdown(&mut self) -> Result<()> {
231        let count = self.engine.scripts.len();
232        if count > 0 {
233            println!("Script system shut down ({} scripts)", count);
234        }
235        Ok(())
236    }
237
238    fn name(&self) -> &str {
239        "script"
240    }
241}

243/// Snapshot which actions are pressed/just-pressed from InputState.
244/// Dynamically iterates all registered action names instead of a hard-coded list.
245fn snapshot_actions(input: &InputState, pressed: bool) -> std::collections::HashSet<String> {
246    let mut set = std::collections::HashSet::new();
247    for action in input.action_names() {
248        let active = if pressed {
249            input.is_action_pressed(&action)
250        } else {
251            input.is_action_just_pressed(&action)
252        };
253        if active {
254            set.insert(action);
255        }
256    }
257    set
258}
259
260fn snapshot_actions_released(input: &InputState) -> std::collections::HashSet<String> {
261    let mut set = std::collections::HashSet::new();
262    for action in input.actions_just_released() {
263        set.insert(action);
264    }
265    set
266}
267
268fn snapshot_action_values(input: &InputState) -> std::collections::HashMap<String, f64> {
269    let mut map = std::collections::HashMap::new();
270    for action in input.action_names() {
271        let val = input.action_value(&action);
272        if val.abs() > 0.001 {
273            map.insert(action, val as f64);
274        }
275    }
276    map
277}
