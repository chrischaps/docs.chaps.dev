1//! Flint Script — Rhai scripting system
2//!
3//! Integrates the Rhai scripting engine into the Flint game loop:
4//! - `ScriptEngine` — compiles and runs .rhai scripts, manages per-entity state
5//! - `ScriptSync` — discovers entities with `script` component, handles hot-reload
6//! - `ScriptSystem` — implements `RuntimeSystem` for game loop integration
7//!
8//! Scripts can read/write ECS data, respond to collisions/triggers/input,
9//! control animation and audio, and hot-reload while the game is running.

82    /// Provide input and timing context for the current frame.
83    /// Called by PlayerApp before update().
84    pub fn provide_context(
85        &mut self,
86        input: &InputState,
87        events: &[GameEvent],
88        total_time: f64,
89        delta_time: f64,
90    ) {
91        // Snapshot input state
92        let touches: Vec<(i64, f64, f64)> = input
93            .active_touches()
94            .values()
95            .map(|t| (t.id as i64, t.position.0, t.position.1))
96            .collect();
97        let touch_just_started: Vec<(i64, f64, f64)> = input
98            .touches_just_started_set()
99            .iter()
100            .filter_map(|id| input.touch_position(*id).map(|(x, y)| (*id as i64, x, y)))
101            .collect();
102        let touch_just_ended: Vec<i64> = input
103            .touches_just_ended_set()
104            .iter()
105            .map(|id| *id as i64)
106            .collect();
107        let touch_taps: Vec<(f64, f64)> = input.touch_taps().to_vec();
108        let touch_swipes: Vec<(String, f64, f64)> = input
109            .touch_swipes()
110            .iter()
111            .map(|(dir, x, y)| (format!("{dir:?}").to_lowercase(), *x, *y))
112            .collect();
113
114        let snapshot = InputSnapshot {
115            actions_pressed: snapshot_actions(input, true),
116            actions_just_pressed: snapshot_actions(input, false),
117            actions_just_released: snapshot_actions_released(input),
118            action_values: snapshot_action_values(input),
119            any_just_pressed: input.any_just_pressed(),
120            mouse_delta: input.raw_mouse_delta(),
121            touches,
122            touch_just_started,
123            touch_just_ended,
124            touch_taps,
125            touch_swipes,
126        };
127
128        self.engine
129            .provide_context(snapshot, delta_time, total_time);
130        self.pending_events = events.to_vec();
131    }

191    /// Load a script file for a specific entity (for chunk loading).
192    /// The entity must already have a `script` component in the ECS.
193    pub fn load_script_for_entity(
194        &mut self,
195        entity_id: flint_core::EntityId,
196        script_path: &std::path::Path,
197    ) {
198        match self.engine.compile_file(script_path) {
199            Ok(ast) => {
200                let source = script_path
201                    .file_name()
202                    .and_then(|f| f.to_str())
203                    .unwrap_or("unknown.rhai")
204                    .to_string();
205                println!("[script] Loaded (chunk): {:?} → {}", entity_id, source);
206                self.engine.add_script(entity_id, ast, source);
207                self.sync.discovered.insert(entity_id);
208            }
209            Err(e) => {
210                tracing::warn!("Compile error for chunk entity {:?}: {}", entity_id, e);
211            }
212        }
213    }

352impl RuntimeSystem for ScriptSystem {
353    fn initialize(&mut self, world: &mut FlintWorld) -> Result<()> {
354        // Discover scripts from world
355        self.sync.discover_and_load(world, &mut self.engine);
356
357        // Call on_init() for all loaded scripts
358        self.engine.call_inits(world);
359
360        let count = self.engine.scripts.len();
361        if count > 0 {
362            println!("Script system initialized ({} scripts)", count);
363        }
364
365        Ok(())
366    }
367
368    fn fixed_update(&mut self, _world: &mut FlintWorld, _dt: f64) -> Result<()> {
369        // Scripts run at variable rate (like animation), not fixed timestep
370        Ok(())
371    }
372
373    fn update(&mut self, world: &mut FlintWorld, _dt: f64) -> Result<()> {
374        // Check for hot-reloaded scripts
375        self.sync.check_hot_reload(&mut self.engine);
376
377        // Discover any new script entities
378        self.sync.discover_and_load(world, &mut self.engine);
379
380        // Call on_init for newly added scripts
381        self.engine.call_inits(world);
382
383        // Process events (collision, trigger, action callbacks)
384        let events = std::mem::take(&mut self.pending_events);
385        self.engine.process_events(&events, world);
386
387        // Call on_update() for all scripts
388        self.engine.call_updates(world);
389
390        Ok(())
391    }
392
393    fn shutdown(&mut self) -> Result<()> {
394        let count = self.engine.scripts.len();
395        if count > 0 {
396            println!("Script system shut down ({} scripts)", count);
397        }
398        Ok(())
399    }
400
401    fn name(&self) -> &str {
402        "script"
403    }
404}

406/// Snapshot which actions are pressed/just-pressed from InputState.
407/// Dynamically iterates all registered action names instead of a hard-coded list.
408fn snapshot_actions(input: &InputState, pressed: bool) -> std::collections::HashSet<String> {
409    let mut set = std::collections::HashSet::new();
410    for action in input.action_names() {
411        let active = if pressed {
412            input.is_action_pressed(&action)
413        } else {
414            input.is_action_just_pressed(&action)
415        };
416        if active {
417            set.insert(action);
418        }
419    }
420    set
421}
422
423fn snapshot_actions_released(input: &InputState) -> std::collections::HashSet<String> {
424    let mut set = std::collections::HashSet::new();
425    for action in input.actions_just_released() {
426        set.insert(action);
427    }
428    set
429}

431fn snapshot_action_values(input: &InputState) -> std::collections::HashMap<String, f64> {
432    let mut map = std::collections::HashMap::new();
433    for action in input.action_names() {
434        let val = input.action_value(&action);
435        if val.abs() > 0.001 {
436            map.insert(action, val as f64);
437        }
438    }
439    map
440}

485/// Lock a shared script-state mutex, recovering from poisoning instead of
486/// propagating the panic (ADR 0039). The guarded values are always-valid
487/// snapshots written whole each frame, so the last-written state is safe to
488/// read after a writer panicked; propagating would turn one panic into a
489/// panic on every later script call, contradicting the crate's
490/// log-and-continue policy. Logs once per process.
491pub(crate) fn lock_or_recover<T>(m: &std::sync::Mutex<T>) -> std::sync::MutexGuard<'_, T> {
492    m.lock().unwrap_or_else(|poisoned| {
493        static LOGGED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
494        if !LOGGED.swap(true, std::sync::atomic::Ordering::Relaxed) {
495            tracing::error!(
496                "script context mutex poisoned (a thread panicked holding it);                  recovering with the last-written snapshot (ADR 0039)"
497            );
498        }
499        poisoned.into_inner()
500    })
