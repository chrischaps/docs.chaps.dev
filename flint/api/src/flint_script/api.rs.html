49// ─── Quality settings API (FlintKart#55) ──────────────────
50//
51// `set_quality(#{ msaa: 4, shadow_cascades: 2 })` changes the renderer's
52// load levers from the next frame; `get_quality()` reads them back;
53// `gpu_info()` says what the GPU is. Keys and ranges are

56/// A script value as a quality value: bools, integers and floats.
57fn quality_value(v: &Dynamic) -> Option<flint_core::quality::QualityValue> {
58    use flint_core::quality::QualityValue;
59    if let Ok(b) = v.as_bool() {
60        Some(QualityValue::Bool(b))
61    } else if let Ok(i) = v.as_int() {
62        Some(QualityValue::Int(i))
63    } else if let Ok(f) = v.as_float() {
64        Some(QualityValue::Float(f))
65    } else {
66        None
67    }
68}

70/// A quality value as a script value.
71fn quality_dynamic(v: flint_core::quality::QualityValue) -> Dynamic {
72    use flint_core::quality::QualityValue;
73    match v {
74        QualityValue::Bool(b) => b.into(),
75        QualityValue::Int(i) => i.into(),
76        QualityValue::Float(f) => f.into(),
77    }
78}

80/// Merge `settings`' keys into `c.quality` and ask for it to be applied.
81/// True when every key was taken; a bad key is warned about and skipped.
82pub(crate) fn request_quality(c: &mut ScriptCallContext, settings: &Map) -> bool {
83    let mut all_taken = true;
84    let mut quality = c.quality;
85    for (key, value) in settings {
86        let result = match quality_value(value) {
87            Some(v) => quality.set(key.as_str(), v),
88            None => Err(flint_core::quality::QualityError::Malformed(format!(
89                "{key}={value}"
90            ))),
91        };
92        if let Err(e) = result {
93            tracing::warn!("set_quality: {e}");
94            all_taken = false;
95        }
96    }
97    if quality != c.quality {
98        c.quality = quality;
99        c.quality_requested = true;
100    }
101    all_taken
102}

104fn register_quality_api(engine: &mut Engine, ctx: Arc<Mutex<ScriptCallContext>>) {
105    // set_quality(map) -> bool
106    {
107        let ctx = ctx.clone();
108        engine.register_fn("set_quality", move |settings: Map| -> bool {
109            let mut c = crate::lock_or_recover(&ctx);
110            request_quality(&mut c, &settings)
111        });
112    }
113
114    // set_quality(key, value) -> bool
115    {
116        let ctx = ctx.clone();
117        engine.register_fn("set_quality", move |key: &str, value: Dynamic| -> bool {
118            let mut c = crate::lock_or_recover(&ctx);
119            let mut settings = Map::new();
120            settings.insert(key.into(), value);
121            request_quality(&mut c, &settings)
122        });
123    }
124
125    // get_quality() -> map of every setting
126    {
127        let ctx = ctx.clone();
128        engine.register_fn("get_quality", move || -> Map {
129            let c = crate::lock_or_recover(&ctx);
130            c.quality
131                .entries()
132                .into_iter()
133                .map(|(k, v)| (k.into(), quality_dynamic(v)))
134                .collect()
135        });
136    }
137
138    // get_quality(key) -> the value, or () for an unknown key
139    {
140        let ctx = ctx.clone();
141        engine.register_fn("get_quality", move |key: &str| -> Dynamic {
142            let c = crate::lock_or_recover(&ctx);
143            c.quality.get(key).map_or(Dynamic::UNIT, quality_dynamic)
144        });
145    }
146
147    // gpu_info() -> #{ name, vendor, device_type, backend, driver, class, platform }
148    {
149        let ctx = ctx.clone();
150        engine.register_fn("gpu_info", move || -> Map {
151            let c = crate::lock_or_recover(&ctx);
152            let g = &c.gpu_info;
153            let mut map = Map::new();
154            for (k, v) in [
155                ("name", &g.name),
156                ("vendor", &g.vendor),
157                ("device_type", &g.device_type),
158                ("backend", &g.backend),
159                ("driver", &g.driver),
160                ("class", &g.class),
161                ("platform", &g.platform),
162            ] {
163                map.insert(k.into(), v.clone().into());
164            }
165            map
166        });
167    }
168}

351// ─── Ocean API ───────────────────────────────────────────
352//
353// Samples the same Gerstner spectrum the renderer displaces vertices with
354// (flint_core::ocean is the single source of truth, and total_time is the
355// same clock the player hands the renderer as ocean_time). The spectrum is
356// cached and regenerated only when the `ocean` component's params change.
357
358fn register_ocean_api(engine: &mut Engine, ctx: Arc<Mutex<ScriptCallContext>>) {
359    use flint_core::ocean::{OceanParams, WaveSpectrum};
360
361    type SpectrumCache = Arc<Mutex<Option<WaveSpectrum>>>;
362    let cache: SpectrumCache = Arc::new(Mutex::new(None));
363
364    // Reads the first `ocean` component and returns (spectrum ready?, time).
365    fn with_spectrum<R>(
366        ctx: &Arc<Mutex<ScriptCallContext>>,
367        cache: &Arc<Mutex<Option<WaveSpectrum>>>,
368        f: impl FnOnce(&WaveSpectrum, f64) -> R,
369    ) -> Option<R> {
370        let c = ctx.lock().unwrap();
371        let world = unsafe { c.world_ref() };
372        let entity_id = world
373            .entities_with_component(comp::OCEAN)
374            .iter()
375            .next()
376            .copied()?;
377        let ocean_comp = world
378            .get_components(entity_id)
379            .and_then(|comps| comps.get(comp::OCEAN).cloned())?;
380        let params = OceanParams::from_component(&ocean_comp);
381
382        let mut cached = cache.lock().unwrap();
383        let needs_regen = cached.as_ref().map(|s| s.params != params).unwrap_or(true);
384        if needs_regen {
385            *cached = Some(WaveSpectrum::generate(&params));
386        }
387        Some(f(cached.as_ref().unwrap(), c.total_time))
388    }
389
390    // ocean_height(x, z) -> f64 — Eulerian surface height at world (x, z).
391    {
392        let ctx = ctx.clone();
393        let cache = cache.clone();
394        engine.register_fn("ocean_height", move |x: f64, z: f64| -> f64 {
395            with_spectrum(&ctx, &cache, |spec, t| {
396                spec.sample_height(x as f32, z as f32, t) as f64
397            })
398            .unwrap_or(0.0)
399        });
400    }
401
402    // ocean_velocity_y(x, z) -> f64 — vertical surface velocity (m/s) at
403    // world (x, z). Analytic ∂h/∂t; pairs with ocean_height for impact cues.
404    {
405        let ctx = ctx.clone();
406        let cache = cache.clone();
407        engine.register_fn("ocean_velocity_y", move |x: f64, z: f64| -> f64 {
408            with_spectrum(&ctx, &cache, |spec, t| {
409                spec.sample_velocity_y(x as f32, z as f32, t) as f64
410            })
411            .unwrap_or(0.0)
412        });
413    }
414
415    // ocean_normal(x, z) -> #{x, y, z} — surface normal at world (x, z).
416    {
417        let ctx = ctx.clone();
418        let cache = cache.clone();
419        engine.register_fn("ocean_normal", move |x: f64, z: f64| -> Map {
420            let n = with_spectrum(&ctx, &cache, |spec, t| {
421                spec.sample_normal(x as f32, z as f32, t)
422            })
423            .unwrap_or([0.0, 1.0, 0.0]);
424            let mut map = Map::new();
425            map.insert("x".into(), Dynamic::from(n[0] as f64));
426            map.insert("y".into(), Dynamic::from(n[1] as f64));
427            map.insert("z".into(), Dynamic::from(n[2] as f64));
428            map
429        });
430    }
431}

975    {
976        let ctx = ctx.clone();
977        engine.register_fn(
978            "spline_closest_point",
979            move |spline_id: i64, qx: f64, qy: f64, qz: f64| -> Dynamic {
980                if spline_id < 0 {
981                    return Dynamic::UNIT;
982                }
983                let c = crate::lock_or_recover(&ctx);
984                let world = unsafe { c.world_ref() };
985                let eid = EntityId::from_raw(spline_id as u64);
986                let comps = match world.get_components(eid) {
987                    Some(c) => c,
988                    None => return Dynamic::UNIT,
989                };
990                let sd = match comps.get(comp::SPLINE_DATA) {
991                    Some(v) => v,
992                    None => return Dynamic::UNIT,
993                };
994                let table = match sd.as_table() {
995                    Some(t) => t,
996                    None => return Dynamic::UNIT,
997                };
998
999                let count = match table.get("sample_count").and_then(|v| v.as_integer()) {
1000                    Some(n) if n > 0 => n as usize,
1001                    _ => return Dynamic::UNIT,
1002                };
1003
1004                let px_arr = match table.get("positions_x").and_then(|v| v.as_array()) {
1005                    Some(a) => a,
1006                    None => return Dynamic::UNIT,
1007                };
1008                let py_arr = match table.get("positions_y").and_then(|v| v.as_array()) {
1009                    Some(a) => a,
1010                    None => return Dynamic::UNIT,
1011                };
1012                let pz_arr = match table.get("positions_z").and_then(|v| v.as_array()) {
1013                    Some(a) => a,
1014                    None => return Dynamic::UNIT,
1015                };
1016                let t_arr = match table.get("t_values").and_then(|v| v.as_array()) {
1017                    Some(a) => a,
1018                    None => return Dynamic::UNIT,
1019                };
1020
1021                let mut best_dist_sq = f64::MAX;
1022                let mut best_idx = 0usize;
1023
1024                for i in 0..count.min(px_arr.len()).min(pz_arr.len()).min(t_arr.len()) {
1025                    let sx = px_arr[i].as_float().unwrap_or(0.0);
1026                    let sy = py_arr[i].as_float().unwrap_or(0.0);
1027                    let sz = pz_arr[i].as_float().unwrap_or(0.0);
1028                    let dx = qx - sx;
1029                    let dy = qy - sy;
1030                    let dz = qz - sz;
1031                    let dist_sq = dx * dx + dy * dy + dz * dz;
1032                    if dist_sq < best_dist_sq {
1033                        best_dist_sq = dist_sq;
1034                        best_idx = i;
1035                    }
1036                }
1037
1038                let mut map = Map::new();
1039                map.insert(
1040                    "t".into(),
1041                    Dynamic::from(t_arr[best_idx].as_float().unwrap_or(0.0)),
1042                );
1043                map.insert(
1044                    "x".into(),
1045                    Dynamic::from(px_arr[best_idx].as_float().unwrap_or(0.0)),
1046                );
1047                map.insert(
1048                    "y".into(),
1049                    Dynamic::from(py_arr[best_idx].as_float().unwrap_or(0.0)),
1050                );
1051                map.insert(
1052                    "z".into(),
1053                    Dynamic::from(pz_arr[best_idx].as_float().unwrap_or(0.0)),
1054                );

1210
1211                let count = match table.get("sample_count").and_then(|v| v.as_integer()) {
1212                    Some(n) if n > 1 => n as usize,
1213                    _ => return Dynamic::UNIT,
1214                };
1215
1216                let px_arr = match table.get("positions_x").and_then(|v| v.as_array()) {
1217                    Some(a) => a,
1218                    None => return Dynamic::UNIT,
1219                };
1220                let py_arr = match table.get("positions_y").and_then(|v| v.as_array()) {
1221                    Some(a) => a,
1222                    None => return Dynamic::UNIT,
1223                };
1224                let pz_arr = match table.get("positions_z").and_then(|v| v.as_array()) {
1225                    Some(a) => a,
1226                    None => return Dynamic::UNIT,
1227                };
1228                let t_arr = match table.get("t_values").and_then(|v| v.as_array()) {
1229                    Some(a) => a,
1230                    None => return Dynamic::UNIT,
1231                };
1232
1233                let len = count
1234                    .min(px_arr.len())
1235                    .min(py_arr.len())
1236                    .min(pz_arr.len())
1237                    .min(t_arr.len());
1238                if len < 2 {
1239                    return Dynamic::UNIT;
1240                }
1241
1242                // Wrap t into [0, 1) for closed splines
1243                let t_wrapped = ((t % 1.0) + 1.0) % 1.0;
1244
1245                // Binary search for the bracketing segment
1246                let mut lo = 0usize;
1247                let mut hi = len - 1;
1248                while lo + 1 < hi {
1249                    let mid = (lo + hi) / 2;
1250                    if t_arr[mid].as_float().unwrap_or(0.0) <= t_wrapped {
1251                        lo = mid;
1252                    } else {
1253                        hi = mid;
1254                    }
1255                }
1256
1257                let t0 = t_arr[lo].as_float().unwrap_or(0.0);
1258                let t1 = t_arr[hi].as_float().unwrap_or(0.0);
1259                let seg_len = t1 - t0;
1260                let frac = if seg_len.abs() > 1e-9 {
1261                    (t_wrapped - t0) / seg_len
1262                } else {
1263                    0.0
1264                };
1265
1266                let x0 = px_arr[lo].as_float().unwrap_or(0.0);
1267                let y0 = py_arr[lo].as_float().unwrap_or(0.0);
1268                let z0 = pz_arr[lo].as_float().unwrap_or(0.0);
1269                let x1 = px_arr[hi].as_float().unwrap_or(0.0);
1270                let y1 = py_arr[hi].as_float().unwrap_or(0.0);
1271                let z1 = pz_arr[hi].as_float().unwrap_or(0.0);
1272
1273                let px = x0 + (x1 - x0) * frac;
1274                let py = y0 + (y1 - y0) * frac;
1275                let pz = z0 + (z1 - z0) * frac;
1276
1277                // Forward direction: tangent along the spline
1278                let mut fx = x1 - x0;
1279                let mut fy = y1 - y0;
1280                let mut fz = z1 - z0;
1281                let flen = (fx * fx + fy * fy + fz * fz).sqrt();
1282                if flen > 1e-9 {
1283                    fx /= flen;
1284                    fy /= flen;
1285                    fz /= flen;
1286                }
1287
1288                // Right vector: cross(forward, up) where up = (0,1,0)
1289                let mut rx = fz;
1290                let ry = 0.0_f64;
1291                let mut rz = -fx;
1292                let rlen = (rx * rx + rz * rz).sqrt();
1293                if rlen > 1e-9 {
1294                    rx /= rlen;
1295                    rz /= rlen;
1296                }
1297
1298                let mut map = Map::new();
1299                map.insert("x".into(), Dynamic::from(px));
1300                map.insert("y".into(), Dynamic::from(py));
1301                map.insert("z".into(), Dynamic::from(pz));
1302                map.insert("fwd_x".into(), Dynamic::from(fx));
1303                map.insert("fwd_y".into(), Dynamic::from(fy));
1304                map.insert("fwd_z".into(), Dynamic::from(fz));
1305                map.insert("right_x".into(), Dynamic::from(rx));
1306                map.insert("right_y".into(), Dynamic::from(ry));
1307                map.insert("right_z".into(), Dynamic::from(rz));
1308                Dynamic::from(map)
1309            },
1310        );
1311    }
1312}

4335/// Project root for a scene path: the scene directory's parent, or "." when
4336/// the path is empty or has no such parent. The same rule `load_ui` and the
4337/// player's `save/persist.toml` use.
4338pub fn project_root_for(scene_path: &str) -> std::path::PathBuf {
4339    if scene_path.is_empty() {
4340        return std::path::PathBuf::from(".");
4341    }
4342    // scenes/oval_plus.scene.toml → parent "scenes" → parent "" → "."
4343    let root = std::path::Path::new(scene_path)
4344        .parent()
4345        .and_then(|dir| dir.parent())
4346        .unwrap_or(std::path::Path::new("."));
4347    if root.as_os_str().is_empty() {
4348        std::path::PathBuf::from(".")
4349    } else {
4350        root.to_path_buf()
4351    }
4352}

4354/// Read and cache a TOML file relative to the project root, returning the
4355/// whole document as a Rhai map (`()` on failure, which is also logged).
4356fn load_data_impl(c: &mut ScriptCallContext, path: &str, reload: bool) -> Dynamic {
4357    let full = project_root_for(&c.current_scene_path).join(path);
4358    let key = full.to_string_lossy().to_string();
4359    if reload {
4360        c.data_cache.remove(&key);
4361    }
4362    if let Some(v) = c.data_cache.get(&key) {
4363        return toml_to_dynamic(v.clone());
4364    }
4365    let parsed = vfs::read_to_string(&full)
4366        .map_err(|e| e.to_string())
4367        .and_then(|text| text.parse::<toml::Value>().map_err(|e| e.to_string()));
4368    match parsed {
4369        Ok(v) => {
4370            let d = toml_to_dynamic(v.clone());
4371            c.data_cache.insert(key, v);
4372            d
4373        }
4374        Err(e) => {
4375            let message = format!("load_data('{}') failed (resolved to {}): {}", path, key, e);
4376            tracing::warn!("{}", message);
4377            c.commands.push(ScriptCommand::Log {
4378                level: LogLevel::Warn,
4379                message,
4380            });
4381            Dynamic::UNIT
4382        }
4383    }
4384}

4409/// Convert a toml::Value to rhai::Dynamic
4410pub fn toml_to_dynamic(val: toml::Value) -> Dynamic {
4411    match val {
4412        toml::Value::Boolean(b) => Dynamic::from(b),
4413        toml::Value::Integer(i) => Dynamic::from(i),
4414        toml::Value::Float(f) => Dynamic::from(f),
4415        toml::Value::String(s) => Dynamic::from(s),
4416        toml::Value::Array(arr) => {
4417            let items: Vec<Dynamic> = arr.into_iter().map(toml_to_dynamic).collect();
4418            Dynamic::from(items)
4419        }
4420        toml::Value::Table(table) => {
4421            let mut map = Map::new();
4422            for (k, v) in table {
4423                map.insert(k.into(), toml_to_dynamic(v));
4424            }
4425            Dynamic::from(map)
4426        }
4427        toml::Value::Datetime(dt) => Dynamic::from(dt.to_string()),
4428    }
4429}
4430
4431/// Convert a rhai::Dynamic to toml::Value
4432pub fn dynamic_to_toml(val: &Dynamic) -> Option<toml::Value> {
4433    if val.is_bool() {
4434        return val.as_bool().ok().map(toml::Value::Boolean);
4435    }
4436    if val.is_int() {
4437        return val.as_int().ok().map(toml::Value::Integer);
4438    }
4439    if val.is_float() {
4440        return val.as_float().ok().map(toml::Value::Float);
4441    }
4442    if val.is_string() {
4443        return val.clone().into_string().ok().map(toml::Value::String);
4444    }
4445    if val.is_array() {
4446        let arr = val.clone().into_array().ok()?;
4447        let items: Vec<toml::Value> = arr.iter().filter_map(dynamic_to_toml).collect();
4448        return Some(toml::Value::Array(items));
4449    }
4450    if val.is_map() {
4451        return Some(map_to_toml_from_dynamic(val));
4452    }
4453    None
4454}
4455
4456fn map_to_toml(map: &Map) -> toml::Value {
4457    let mut table = toml::map::Map::new();
4458    for (k, v) in map {
4459        if let Some(tv) = dynamic_to_toml(v) {
4460            table.insert(k.to_string(), tv);
4461        }
4462    }
4463    toml::Value::Table(table)
4464}
4465
4466fn map_to_toml_from_dynamic(val: &Dynamic) -> toml::Value {
4467    if let Some(map) = val.clone().try_cast::<Map>() {
4468        return map_to_toml(&map);
4469    }
4470    toml::Value::String(val.to_string())
4471}

4892    // camera_follow(entity_id, offset_x, offset_y, speed, deadzone_w, deadzone_h)
4893    {
4894        let ctx = ctx.clone();
4895        engine.register_fn(
4896            "camera_follow",
4897            move |entity_id: i64,
4898                  offset_x: f64,
4899                  offset_y: f64,
4900                  speed: f64,
4901                  deadzone_w: f64,
4902                  deadzone_h: f64| {
4903                let mut c = crate::lock_or_recover(&ctx);
4904                let dt = c.delta_time as f32;
4905                if dt <= 0.0 {
4906                    return;
4907                }
4908
4909                // Read target entity position
4910                let (target_x, target_y) = {
4911                    let world = unsafe { c.world_ref() };
4912                    match world.get_transform(flint_core::EntityId::from_raw(entity_id as u64)) {
4913                        Some(t) => (
4914                            t.position.x + offset_x as f32,
4915                            t.position.y + offset_y as f32,
4916                        ),
4917                        None => return,
4918                    }
4919                };
4920
4921                // Initialize follow position on first call
4922                if !c.camera_follow.initialized {
4923                    c.camera_follow.current_x = target_x;
4924                    c.camera_follow.current_y = target_y;
4925                    c.camera_follow.initialized = true;
4926                }
4927
4928                // Deadzone: only track when target is outside deadzone rectangle
4929                let dx = target_x - c.camera_follow.current_x;
4930                let dy = target_y - c.camera_follow.current_y;
4931                let half_w = deadzone_w as f32 * 0.5;
4932                let half_h = deadzone_h as f32 * 0.5;
4933
4934                // Compute tracking target — nearest edge of deadzone
4935                let track_x = if dx.abs() > half_w {
4936                    target_x - dx.signum() * half_w
4937                } else {
4938                    c.camera_follow.current_x
4939                };
4940                let track_y = if dy.abs() > half_h {
4941                    target_y - dy.signum() * half_h
4942                } else {
4943                    c.camera_follow.current_y
4944                };
4945
4946                // Frame-rate-independent exponential smoothing
4947                let t = 1.0 - (-(speed as f32) * dt).exp();
4948                c.camera_follow.current_x += (track_x - c.camera_follow.current_x) * t;
4949                c.camera_follow.current_y += (track_y - c.camera_follow.current_y) * t;
4950
4951                // Advance shake
4952                let mut shake_offset_x = 0.0f32;
4953                let mut shake_offset_y = 0.0f32;
4954                if c.shake.amplitude > 0.001 {
4955                    c.shake.phase += dt;
4956                    c.shake.amplitude *= (-(c.shake.decay) * dt).exp();
4957                    let freq = c.shake.frequency;
4958                    shake_offset_x = c.shake.amplitude * (freq * c.shake.phase).sin();
4959                    shake_offset_y = c.shake.amplitude * (freq * 1.37 * c.shake.phase).cos();
4960                    if c.shake.amplitude < 0.001 {
4961                        c.shake.amplitude = 0.0;
4962                        c.shake.phase = 0.0;
4963                    }
4964                }
4965
4966                let final_x = c.camera_follow.current_x + shake_offset_x;
4967                let final_y = c.camera_follow.current_y + shake_offset_y;
4968
4969                // Z=10 ensures all sprite layers (z_offset = layer*0.01) are in front of the camera
4970                c.camera_position_override = Some([final_x, final_y, 10.0]);
4971                c.camera_target_override = Some([final_x, final_y, -1.0]);
4972            },
4973        );
4974    }
4975
4976    // camera_follow_position() -> #{x, y}
4977    {
4978        let ctx = ctx.clone();
4979        engine.register_fn("camera_follow_position", move || -> Map {
4980            let c = crate::lock_or_recover(&ctx);
4981            let mut map = Map::new();

4989    {
4990        let ctx = ctx.clone();
4991        engine.register_fn("camera_follow_set", move |x: f64, y: f64| {
4992            let mut c = crate::lock_or_recover(&ctx);

4994            c.camera_follow.current_y = y as f32;
4995            c.camera_follow.initialized = true;
4996        });
4997    }
4998
4999    // camera_shake(amplitude, frequency, decay)
5000    {
5001        let ctx = ctx.clone();
5002        engine.register_fn(
5003            "camera_shake",
5004            move |amplitude: f64, frequency: f64, decay: f64| {
5005                let mut c = crate::lock_or_recover(&ctx);
5006                // Stack: take max of current + new amplitude
5007                let new_amp = (c.shake.amplitude + amplitude as f32).max(amplitude as f32);
5008                c.shake.amplitude = new_amp;
5009                c.shake.frequency = frequency as f32;
5010                c.shake.decay = decay as f32;
5011                // Don't reset phase — let it continue naturally
5012            },
5013        );
5014    }
5015
5016    // camera_shake_stop()
5017    {
5018        let ctx = ctx.clone();
5019        engine.register_fn("camera_shake_stop", move || {
5020            let mut c = crate::lock_or_recover(&ctx);
5021            c.shake.amplitude = 0.0;
5022            c.shake.phase = 0.0;
5023        });
5024    }
5025
5026    // camera_apply_shake() — for manual camera control without camera_follow
5027    {
5028        let ctx = ctx.clone();
5029        engine.register_fn("camera_apply_shake", move || {
5030            let mut c = crate::lock_or_recover(&ctx);
5031            let dt = c.delta_time as f32;
5032            if dt <= 0.0 || c.shake.amplitude < 0.001 {
5033                return;
5034            }
5035
5036            c.shake.phase += dt;
5037            c.shake.amplitude *= (-(c.shake.decay) * dt).exp();
5038            let freq = c.shake.frequency;
5039            let shake_x = c.shake.amplitude * (freq * c.shake.phase).sin();
5040            let shake_y = c.shake.amplitude * (freq * 1.37 * c.shake.phase).cos();
5041
5042            if c.shake.amplitude < 0.001 {
5043                c.shake.amplitude = 0.0;
5044                c.shake.phase = 0.0;
5045            }
5046
5047            // Apply shake offset to existing camera override (or current camera position)
5048            if let Some(pos) = c.camera_position_override.as_mut() {
5049                pos[0] += shake_x;
5050                pos[1] += shake_y;
5051            } else {
5052                let base = c.camera_position;
5053                c.camera_position_override = Some([base[0] + shake_x, base[1] + shake_y, base[2]]);
5054            }
5055            if let Some(target) = c.camera_target_override.as_mut() {
5056                target[0] += shake_x;
5057                target[1] += shake_y;
5058            }
5059        });
5060    }

5326    text: &str,
5327    size: f32,
5328    opts: &Map,
5329) -> flint_core::world_text::WorldText {

5332        o.align

5334        1
5335    };

5338            Some(s) => match s.as_str() {
5339                "top" => 0,
5340                "bottom" => 2,

5352        }
5353    };

5375/// Measure text with the host's installed measurer, falling back to the
5376/// glyph-count estimate (0.6 em per char) when none is installed.
5377pub(crate) fn measure_text_with(
5378    c: &ScriptCallContext,
5379    text: &str,
5380    size: f32,
5381    font: Option<&str>,
5382    spacing: f32,
5383) -> (f32, f32) {
5384    match &c.text_measurer {
5385        Some(m) => m(text, size, font, spacing),
5386        None => {
5387            let chars = text.chars().count() as f32;
5388            (chars * size * 0.6 + spacing * (chars - 1.0).max(0.0), size)
5389        }
5390    }
5391}

5802/// Project a world-space segment with the frame's camera and record it as
5803/// a HUD line. Clipped against the near plane so a segment that passes the
5804/// camera still draws; fully-behind segments are dropped.
5805fn push_line_3d(
5806    c: &mut ScriptCallContext,
5807    a: [f64; 3],
5808    b: [f64; 3],
5809    color: [f32; 4],
5810    thickness: f32,
5811    layer: i32,
5812) {
5813    let mut p = project_clip(&c.camera_view_proj, [a[0] as f32, a[1] as f32, a[2] as f32]);
5814    let mut q = project_clip(&c.camera_view_proj, [b[0] as f32, b[1] as f32, b[2] as f32]);
5815    const NEAR_W: f32 = 1e-3;
5816    if p[3] <= NEAR_W && q[3] <= NEAR_W {
5817        return;
5818    }
5819    if p[3] <= NEAR_W || q[3] <= NEAR_W {
5820        // Move the behind-camera end onto the near plane (linear in clip space).
5821        let t = (NEAR_W - p[3]) / (q[3] - p[3]);
5822        let m = [
5823            p[0] + (q[0] - p[0]) * t,
5824            p[1] + (q[1] - p[1]) * t,
5825            p[2] + (q[2] - p[2]) * t,
5826            NEAR_W,
5827        ];
5828        if p[3] <= NEAR_W {
5829            p = m;
5830        } else {
5831            q = m;
5832        }
5833    }
5834    let (sx1, sy1) = clip_to_screen(p, c.screen_width, c.screen_height);
5835    let (sx2, sy2) = clip_to_screen(q, c.screen_width, c.screen_height);
5836    c.push_draw(DrawCommand::Line {
5837        x1: sx1,
5838        y1: sy1,
5839        x2: sx2,
5840        y2: sy2,
5841        color,
5842        thickness,
5843        layer,
5844    });
