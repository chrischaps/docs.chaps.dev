6/// Layout flow direction

23/// Resolved style for a single element — all visual properties with defaults applied
24#[derive(Debug, Clone)]
25pub struct ResolvedStyle {
26    // Position (offset from anchor/parent)
27    pub x: f32,
28    pub y: f32,
29    pub width: f32,
30    pub height: f32,
31    pub width_pct: Option<f32>,
32    pub height_pct: Option<f32>,
33    pub height_auto: bool,
34
35    // Visual
36    pub color: [f32; 4],
37    pub bg_color: [f32; 4],
38    pub font_size: f32,
39    pub text_align: TextAlign,
40    pub rounding: f32,
41    pub opacity: f32,
42    pub thickness: f32,
43    pub radius: f32,
44    pub layer: i32,
45    pub padding: [f32; 4], // L, T, R, B
46    pub stroke_color: [f32; 4],
47    pub stroke_width: f32,
48
49    // Layout
50    pub layout: LayoutFlow,
51    pub margin_bottom: f32,
52}

82/// A named style class parsed from .style.toml
83#[derive(Debug, Clone)]
84pub struct StyleClass {
85    pub name: String,
86    pub properties: HashMap<String, StyleValue>,
87}
88
89impl StyleClass {
90    /// Resolve this class into a full ResolvedStyle, applying defaults for missing properties
91    pub fn resolve(&self) -> ResolvedStyle {
92        let mut style = ResolvedStyle::default();
93
94        for (key, val) in &self.properties {
95            match key.as_str() {
96                "x" => if let StyleValue::Float(v) = val { style.x = *v; },
97                "y" => if let StyleValue::Float(v) = val { style.y = *v; },
98                "width" => if let StyleValue::Float(v) = val { style.width = *v; },
99                "height" => if let StyleValue::Float(v) = val { style.height = *v; },
100                "width_pct" => if let StyleValue::Float(v) = val { style.width_pct = Some(*v); },
101                "height_pct" => if let StyleValue::Float(v) = val { style.height_pct = Some(*v); },
102                "height_auto" => if let StyleValue::Bool(v) = val { style.height_auto = *v; },
103                "font_size" => if let StyleValue::Float(v) = val { style.font_size = *v; },
104                "rounding" => if let StyleValue::Float(v) = val { style.rounding = *v; },
105                "opacity" => if let StyleValue::Float(v) = val { style.opacity = *v; },
106                "thickness" => if let StyleValue::Float(v) = val { style.thickness = *v; },
107                "radius" => if let StyleValue::Float(v) = val { style.radius = *v; },
108                "layer" => if let StyleValue::Float(v) = val { style.layer = *v as i32; },
109                "margin_bottom" => if let StyleValue::Float(v) = val { style.margin_bottom = *v; },
110                "color" => if let StyleValue::Color(c) = val { style.color = *c; },
111                "bg_color" => if let StyleValue::Color(c) = val { style.bg_color = *c; },
112                "stroke_color" => if let StyleValue::Color(c) = val { style.stroke_color = *c; },
113                "stroke_width" => if let StyleValue::Float(v) = val { style.stroke_width = *v; },
114                "text_align" => if let StyleValue::String(s) = val {
115                    style.text_align = match s.as_str() {
116                        "center" => TextAlign::Center,
117                        "right" => TextAlign::Right,
118                        _ => TextAlign::Left,
119                    };
120                },
121                "layout" => if let StyleValue::String(s) = val {
122                    style.layout = match s.as_str() {
123                        "horizontal" => LayoutFlow::Horizontal,
124                        _ => LayoutFlow::Stack,
125                    };
126                },
127                "padding" => if let StyleValue::Color(p) = val {
128                    // Reuse Color([f32;4]) for 4-value padding
129                    style.padding = *p;
130                },
131                _ => {}
132            }
133        }
134
135        style
136    }
137
138    /// Apply runtime overrides to a resolved style
139    pub fn apply_overrides(style: &mut ResolvedStyle, overrides: &HashMap<String, StyleValue>) {
140        for (key, val) in overrides {
141            match key.as_str() {
142                "x" => if let StyleValue::Float(v) = val { style.x = *v; },
143                "y" => if let StyleValue::Float(v) = val { style.y = *v; },
144                "width" => if let StyleValue::Float(v) = val { style.width = *v; },
145                "height" => if let StyleValue::Float(v) = val { style.height = *v; },
146                "font_size" => if let StyleValue::Float(v) = val { style.font_size = *v; },
147                "rounding" => if let StyleValue::Float(v) = val { style.rounding = *v; },
148                "opacity" => if let StyleValue::Float(v) = val { style.opacity = *v; },
149                "layer" => if let StyleValue::Float(v) = val { style.layer = *v as i32; },
150                "color" => if let StyleValue::Color(c) = val { style.color = *c; },
151                "bg_color" => if let StyleValue::Color(c) = val { style.bg_color = *c; },
152                _ => {}
153            }
154        }
155    }
