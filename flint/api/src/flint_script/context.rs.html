1//! Script execution context and command types

3//! ScriptCallContext is the shared state accessed by Rhai API functions during script execution.
4//! ScriptCommand represents deferred actions (audio, events, logging) collected during a script call.

13/// Snapshot of input state for script access (no winit dependency needed)
14#[derive(Clone, Default)]
15pub struct InputSnapshot {
16    pub actions_pressed: HashSet<String>,
17    pub actions_just_pressed: HashSet<String>,
18    pub actions_just_released: HashSet<String>,
19    pub action_values: std::collections::HashMap<String, f64>,
20    pub mouse_delta: (f64, f64),
21    /// Active touches: (id, norm_x, norm_y)
22    pub touches: Vec<(i64, f64, f64)>,
23    /// Touches that just started this frame: (id, norm_x, norm_y)
24    pub touch_just_started: Vec<(i64, f64, f64)>,
25    /// Touch IDs that just ended this frame
26    pub touch_just_ended: Vec<i64>,
27    /// Tap positions detected this frame (norm_x, norm_y)
28    pub touch_taps: Vec<(f64, f64)>,
29}

79/// 2D draw command issued by scripts each frame (immediate mode)
80#[derive(Debug, Clone)]
81pub enum DrawCommand {
82    Text {
83        x: f32,
84        y: f32,
85        text: String,
86        size: f32,
87        color: [f32; 4],
88        layer: i32,
89        /// 0 = left (default), 1 = center, 2 = right
90        align: u8,
91        /// Optional stroke (outline): color + pixel width
92        stroke: Option<([f32; 4], f32)>,
93    },
94    RectFilled {
95        x: f32,
96        y: f32,
97        w: f32,
98        h: f32,
99        color: [f32; 4],
100        rounding: f32,
101        layer: i32,
102    },
103    RectOutline {
104        x: f32,
105        y: f32,
106        w: f32,
107        h: f32,
108        color: [f32; 4],
109        thickness: f32,
110        layer: i32,
111    },
112    CircleFilled {
113        x: f32,
114        y: f32,
115        radius: f32,
116        color: [f32; 4],
117        layer: i32,
118    },
119    CircleOutline {
120        x: f32,
121        y: f32,
122        radius: f32,
123        color: [f32; 4],
124        thickness: f32,
125        layer: i32,
126    },
127    Line {
128        x1: f32,
129        y1: f32,
130        x2: f32,
131        y2: f32,
132        color: [f32; 4],
133        thickness: f32,
134        layer: i32,
135    },
136    Sprite {
137        x: f32,
138        y: f32,
139        w: f32,
140        h: f32,
141        name: String,
142        uv: [f32; 4],
143        tint: [f32; 4],
144        layer: i32,
145    },
146}
147
148impl DrawCommand {
149    pub fn layer(&self) -> i32 {
150        match self {
151            DrawCommand::Text { layer, .. } => *layer,
152            DrawCommand::RectFilled { layer, .. } => *layer,
153            DrawCommand::RectOutline { layer, .. } => *layer,
154            DrawCommand::CircleFilled { layer, .. } => *layer,
155            DrawCommand::CircleOutline { layer, .. } => *layer,
156            DrawCommand::Line { layer, .. } => *layer,
157            DrawCommand::Sprite { layer, .. } => *layer,
158        }
159    }
160}

162/// Shared context set before each script call and read by registered Rhai functions.
163///
164/// Safety: the `world` pointer is only valid during the scope of `call_update` /
165/// `process_events`. It is set to null immediately after each call batch.
166pub struct ScriptCallContext {
167    /// Raw pointer to the FlintWorld — valid only during call scope
168    pub world: *mut FlintWorld,
169    /// Raw pointer to the PhysicsSystem — valid only during call scope
170    pub physics: *const PhysicsSystem,
171    /// Camera position and direction for weapon aiming
172    pub camera_position: [f32; 3],
173    pub camera_direction: [f32; 3],
174    /// Entity currently being scripted
175    pub current_entity: EntityId,
176    /// Accumulated commands to be drained after all scripts run
177    pub commands: Vec<ScriptCommand>,
178    /// Accumulated 2D draw commands for the current frame
179    pub draw_commands: Vec<DrawCommand>,
180    /// Input snapshot for this frame
181    pub input: InputSnapshot,
182    /// Frame delta time
183    pub delta_time: f64,
184    /// Total elapsed game time
185    pub total_time: f64,
186    /// Screen dimensions in pixels (set before scripts run)
187    pub screen_width: f32,
188    pub screen_height: f32,
189    /// Script-driven camera overrides (set by set_camera_position/set_camera_target/set_camera_fov)
190    pub camera_position_override: Option<[f32; 3]>,
191    pub camera_target_override: Option<[f32; 3]>,
192    pub camera_fov_override: Option<f32>,
193    /// Script-driven camera orthographic override
194    pub camera_orthographic_override: Option<bool>,
195    /// Script-driven camera ortho_height override
196    pub camera_ortho_height_override: Option<f32>,
197    /// Script-driven post-processing overrides
198    pub postprocess_vignette_override: Option<f32>,
199    pub postprocess_bloom_override: Option<f32>,
200    pub postprocess_exposure_override: Option<f32>,
201    pub postprocess_chromatic_aberration_override: Option<f32>,
202    pub postprocess_radial_blur_override: Option<f32>,
203    pub postprocess_ssao_intensity_override: Option<f32>,
204    pub postprocess_fog_density_override: Option<f32>,
205    /// Script-driven audio low-pass filter override (cutoff frequency in Hz)
206    pub audio_lowpass_cutoff_override: Option<f32>,
207    /// Raw pointer to the GameStateMachine — valid only during call scope
208    pub state_machine: *mut GameStateMachine,
209    /// Raw pointer to the PersistentStore — valid only during call scope
210    pub persistent_store: *mut PersistentStore,
211    /// Transition progress (0.0-1.0 during transitions, -1.0 when idle)
212    pub transition_progress: f64,
213    /// Current transition phase name ("idle", "exiting", "entering")
214    pub transition_phase: String,
215    /// Path of the currently loaded scene
216    pub current_scene_path: String,
217    /// Data-driven UI system
218    pub ui_system: UiSystem,
219    /// Terrain height sampling callback — set by PlayerApp if terrain is loaded
220    pub terrain_height_fn: Option<Box<dyn Fn(f32, f32) -> f32 + Send + Sync>>,
221}

223// SAFETY: ScriptCallContext is only accessed from the main thread within
224// controlled call scopes. The world pointer is valid only during those scopes.
225unsafe impl Send for ScriptCallContext {}
226unsafe impl Sync for ScriptCallContext {}

228impl Default for ScriptCallContext {

272    /// Get a reference to the world. Panics if called outside a valid scope.
273    ///
274    /// # Safety
275    /// Caller must ensure the world pointer was set and is still valid
276    /// (i.e., called within the scope of `call_update` or `process_events`).
277    pub unsafe fn world_ref(&self) -> &FlintWorld {
278        assert!(
279            !self.world.is_null(),
280            "ScriptCallContext: world pointer is null (called outside scope)"
281        );
282        unsafe { &*self.world }
283    }
284
285    /// Get a mutable reference to the world. Panics if called outside a valid scope.
286    ///
287    /// # Safety
288    /// Caller must ensure the world pointer was set, is still valid, and no other
289    /// references to the world exist (i.e., called within the scope of
290    /// `call_update` or `process_events`).
291    #[allow(clippy::mut_from_ref)]
292    pub unsafe fn world_mut(&self) -> &mut FlintWorld {
293        assert!(
294            !self.world.is_null(),
295            "ScriptCallContext: world pointer is null (called outside scope)"
296        );
297        unsafe { &mut *self.world }
298    }
299
300    /// Get a reference to the physics system. Returns None if not set.
301    ///
302    /// # Safety
303    /// Caller must ensure the physics pointer was set and is still valid.
304    pub unsafe fn physics_ref(&self) -> Option<&PhysicsSystem> {
305        if self.physics.is_null() {
306            None
307        } else {
308            Some(unsafe { &*self.physics })
309        }
310    }
