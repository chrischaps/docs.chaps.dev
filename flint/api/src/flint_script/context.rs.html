13/// Snapshot of input state for script access (no winit dependency needed)
14#[derive(Clone, Default)]
15pub struct InputSnapshot {
16    pub actions_pressed: HashSet<String>,

19    pub action_values: std::collections::HashMap<String, f64>,

31/// Deferred commands produced by scripts, processed by PlayerApp after script update
32#[derive(Debug, Clone)]
33pub enum ScriptCommand {
34    PlaySound { name: String, volume: f64 },
35    PlaySoundAt { name: String, position: (f64, f64, f64), volume: f64 },
36    StopSound { name: String },
37    FireEvent { name: String, data: toml::Value },
38    Log { level: LogLevel, message: String },
39    EmitBurst { entity_id: i64, count: i64 },
40    LoadScene { path: String },
41    ReloadScene,
42    PushState { name: String },
43    PopState,
44    ReplaceState { name: String },
45}

47/// 2D draw command issued by scripts each frame (immediate mode)
48#[derive(Debug, Clone)]
49pub enum DrawCommand {
50    Text {
51        x: f32, y: f32,
52        text: String,
53        size: f32,
54        color: [f32; 4],
55        layer: i32,
56        /// 0 = left (default), 1 = center, 2 = right
57        align: u8,
58        /// Optional stroke (outline): color + pixel width
59        stroke: Option<([f32; 4], f32)>,
60    },
61    RectFilled {
62        x: f32, y: f32, w: f32, h: f32,
63        color: [f32; 4],
64        rounding: f32,
65        layer: i32,
66    },
67    RectOutline {
68        x: f32, y: f32, w: f32, h: f32,
69        color: [f32; 4],
70        thickness: f32,
71        layer: i32,
72    },
73    CircleFilled {
74        x: f32, y: f32,
75        radius: f32,
76        color: [f32; 4],
77        layer: i32,
78    },
79    CircleOutline {
80        x: f32, y: f32,
81        radius: f32,
82        color: [f32; 4],
83        thickness: f32,
84        layer: i32,
85    },
86    Line {
87        x1: f32, y1: f32, x2: f32, y2: f32,
88        color: [f32; 4],
89        thickness: f32,
90        layer: i32,
91    },
92    Sprite {
93        x: f32, y: f32, w: f32, h: f32,
94        name: String,
95        uv: [f32; 4],
96        tint: [f32; 4],
97        layer: i32,
98    },
99}
100
101impl DrawCommand {
102    pub fn layer(&self) -> i32 {
103        match self {
104            DrawCommand::Text { layer, .. } => *layer,
105            DrawCommand::RectFilled { layer, .. } => *layer,
106            DrawCommand::RectOutline { layer, .. } => *layer,
107            DrawCommand::CircleFilled { layer, .. } => *layer,
108            DrawCommand::CircleOutline { layer, .. } => *layer,
109            DrawCommand::Line { layer, .. } => *layer,
110            DrawCommand::Sprite { layer, .. } => *layer,
111        }
112    }
113}

115/// Shared context set before each script call and read by registered Rhai functions.
116///
117/// Safety: the `world` pointer is only valid during the scope of `call_update` /
118/// `process_events`. It is set to null immediately after each call batch.
119pub struct ScriptCallContext {
120    /// Raw pointer to the FlintWorld — valid only during call scope
121    pub world: *mut FlintWorld,
122    /// Raw pointer to the PhysicsSystem — valid only during call scope
123    pub physics: *const PhysicsSystem,
124    /// Camera position and direction for weapon aiming
125    pub camera_position: [f32; 3],
126    pub camera_direction: [f32; 3],
127    /// Entity currently being scripted
128    pub current_entity: EntityId,
129    /// Accumulated commands to be drained after all scripts run
130    pub commands: Vec<ScriptCommand>,
131    /// Accumulated 2D draw commands for the current frame
132    pub draw_commands: Vec<DrawCommand>,
133    /// Input snapshot for this frame
134    pub input: InputSnapshot,
135    /// Frame delta time
136    pub delta_time: f64,
137    /// Total elapsed game time
138    pub total_time: f64,
139    /// Screen dimensions in pixels (set before scripts run)
140    pub screen_width: f32,
141    pub screen_height: f32,
142    /// Script-driven camera overrides (set by set_camera_position/set_camera_target/set_camera_fov)
143    pub camera_position_override: Option<[f32; 3]>,
144    pub camera_target_override: Option<[f32; 3]>,
145    pub camera_fov_override: Option<f32>,
146    /// Script-driven post-processing overrides
147    pub postprocess_vignette_override: Option<f32>,
148    pub postprocess_bloom_override: Option<f32>,
149    pub postprocess_exposure_override: Option<f32>,
150    pub postprocess_chromatic_aberration_override: Option<f32>,
151    pub postprocess_radial_blur_override: Option<f32>,
152    pub postprocess_ssao_intensity_override: Option<f32>,
153    pub postprocess_fog_density_override: Option<f32>,
154    /// Script-driven audio low-pass filter override (cutoff frequency in Hz)
155    pub audio_lowpass_cutoff_override: Option<f32>,
156    /// Raw pointer to the GameStateMachine — valid only during call scope
157    pub state_machine: *mut GameStateMachine,
158    /// Raw pointer to the PersistentStore — valid only during call scope
159    pub persistent_store: *mut PersistentStore,
160    /// Transition progress (0.0-1.0 during transitions, -1.0 when idle)
161    pub transition_progress: f64,
162    /// Current transition phase name ("idle", "exiting", "entering")
163    pub transition_phase: String,
164    /// Path of the currently loaded scene
165    pub current_scene_path: String,
166    /// Data-driven UI system
167    pub ui_system: UiSystem,
168    /// Terrain height sampling callback — set by PlayerApp if terrain is loaded
169    pub terrain_height_fn: Option<Box<dyn Fn(f32, f32) -> f32 + Send + Sync>>,
170}

172// SAFETY: ScriptCallContext is only accessed from the main thread within
173// controlled call scopes. The world pointer is valid only during those scopes.
174unsafe impl Send for ScriptCallContext {}
175unsafe impl Sync for ScriptCallContext {}
176
177impl Default for ScriptCallContext {
178    fn default() -> Self {
179        Self::new()
180    }
181}

219    /// Get a reference to the world. Panics if called outside a valid scope.
220    ///
221    /// # Safety
222    /// Caller must ensure the world pointer was set and is still valid
223    /// (i.e., called within the scope of `call_update` or `process_events`).
224    pub unsafe fn world_ref(&self) -> &FlintWorld {
225        assert!(!self.world.is_null(), "ScriptCallContext: world pointer is null (called outside scope)");
226        unsafe { &*self.world }
227    }
228
229    /// Get a mutable reference to the world. Panics if called outside a valid scope.
230    ///
231    /// # Safety
232    /// Caller must ensure the world pointer was set, is still valid, and no other
233    /// references to the world exist (i.e., called within the scope of
234    /// `call_update` or `process_events`).
235    #[allow(clippy::mut_from_ref)]
236    pub unsafe fn world_mut(&self) -> &mut FlintWorld {
237        assert!(!self.world.is_null(), "ScriptCallContext: world pointer is null (called outside scope)");
238        unsafe { &mut *self.world }
239    }
240
241    /// Get a reference to the physics system. Returns None if not set.
242    ///
243    /// # Safety
244    /// Caller must ensure the physics pointer was set and is still valid.
245    pub unsafe fn physics_ref(&self) -> Option<&PhysicsSystem> {
246        if self.physics.is_null() {
247            None
248        } else {
249            Some(unsafe { &*self.physics })
250        }
251    }
